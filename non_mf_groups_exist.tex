\begin{abstract}
\leanverified{Manuscript/OneSidedMFRadical/CountableNonMF}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptNotEveryCountableGroupIsMF}
Some groups are not MF. For example, $H=\EL_{12}(L_{\F_2}(1,2))$ is
a simple property-\textup{(T)} group where every homomorphism from $H$
to an MF group is trivial, so $C^*_{\mathrm r}(H)$ is a separable stably
finite $C^*$-algebra that is not MF\@.  Generally, a
normal property-\textup{(T)} subgroup contained in the commutators of a
one-sided compression of a property-\textup{(T)} subgroup is killed by every
homomorphism to an MF group.  The same criterion gives a finitely presented
torsion-free property-\textup{(T)} group with no nontrivial homomorphism to
an MF group.
\end{abstract}

\section{Introduction}

The counterexample is
\[
 H=\EL_{12}(L_{\F_2}(1,2)),
\]
the elementary group of $12\times12$ matrices over the binary Leavitt
algebra.  Khanh--Thanh show that $H$ is isomorphic to the unit group of the
binary Leavitt algebra~\cite[Proposition~4.2 and Corollary~4.4]{KhanhThanh},
which OpenAI used to construct the first nonsofic
group~\cite[Chapter~3]{OAI}.

A countable group $G$ is \emph{MF}~\cite{CDE} if there are positive
integers $d_n$ and maps $V_n\colon G\to\U(d_n)$, with $V_n(1)=1$, such that
\[
 \opnorm{V_n(gh)-V_n(g)V_n(h)}\longrightarrow0
 \qquad(g,h\in G),
\]
and
\[
 \limsup_n\opnorm{V_n(g)-1}>0
 \qquad(g\in G\setminus\{1\}).
\]
In other words, $G$ is MF if there are finite unitary matrices, one for
each element of $G$, that multiply correctly up to an error tending to zero
in operator norm while every element other than the identity stays a fixed
distance from the identity matrix.  The separation may equivalently be
required along the full sequence with a constant independent of
$g$~\cite[Propositions~2 and~7]{Korchagin}.  The strong convergence
convention of~\cite{GaoEtAl,Schafhauser} also requires the models to
reproduce the operator norms of the left regular representation, so a group
that is not MF as defined here is not MF in that convention either.
Equivalently, $G$ embeds
in the unitary group of the norm matrix corona
\[
 \mathcal Q_{\mathbf d}
 =\prod_nM_{d_n}(\C)\big/\bigoplus_nM_{d_n}(\C),
\]
where $\prod_nM_{d_n}(\C)$ is the $C^*$-algebra of bounded sequences, as
every product of $C^*$-algebras below, and $\bigoplus_nM_{d_n}(\C)$ is its
ideal of sequences with $\opnorm{x_n}\to0$.  The quotient
$\mathcal Q_{\mathbf d}$ is the corona algebra of $\bigoplus_nM_{d_n}(\C)$,
since $\prod_nM_{d_n}(\C)$ is its multiplier algebra.  We call a
homomorphism $G\to\U(\mathcal Q_{\mathbf d})$ a \emph{corona
homomorphism}.  A separable
$C^*$-algebra is MF if it embeds in a norm matrix corona~\cite{BK}.

The groups here fail to be MF because they contain a subgroup that is
conjugated into itself.  For $L\le G$ put
\[
 \operatorname{Comp}_G(L)=\{u\in G:uLu^{-1}\le L\}.
\]
An element $c$ of the centralizer $C_G(L)$ commutes with $L$, so $ucu^{-1}$
commutes with $uLu^{-1}$ but need not commute with the rest of $L$.  The
normal subgroup of $G$ generated by the commutators of all such elements
$ucu^{-1}$ with elements of $L$ is
\begin{equation}\label{eq:intrinsic-defect}
 \mathfrak D_G(L)
 =\normal{
   [ucu^{-1},\ell]:
   u\in\operatorname{Comp}_G(L),\ c\in C_G(L),\ \ell\in L
 }_G.
\end{equation}

\begin{mainthm}[one-sided compression criterion]
\label{thm:compression-criterion}
Let $G$ be countable and let $L\le G$ have property~\textup{(T)}.  If
$K\trianglelefteq G$ has property~\textup{(T)} and
$K\le\mathfrak D_G(L)$, then every homomorphism from $G$ to an MF group is
trivial on $K$.  In particular, a nontrivial such $K$ makes $G$ non-MF, and
if $G$ has property~\textup{(T)} and $\mathfrak D_G(L)=G$, then every
homomorphism from $G$ to an MF group is trivial.
\leanverified{Manuscript/OneSidedMFRadical/PrintedCriterion}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptOneSidedCompressionCriterion}
\leanverified{Manuscript/OneSidedMFRadical/PrintedCriterion}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptFullRadicalKillsMFTargets}
\end{mainthm}

Asymptotic multiplicativity in operator norm makes the conjugation maps
$\operatorname{Ad}(V_n(g))\colon x\mapsto V_n(g)xV_n(g)^*$ an asymptotic
unitary representation of $G$ on $M_{d_n}(\C)$ with its normalized
Hilbert--Schmidt inner product, because
$\opnorm{\operatorname{Ad}(A)-\operatorname{Ad}(B)}\le2\opnorm{A-B}$.
Property~\textup{(T)} applies to unitary representations on Hilbert
spaces, so it applies to this one, and its conclusions are estimates in
Hilbert--Schmidt norm.  The unitary $U$ of the compressor $u$ satisfies
$U^*PU\le P$ for the Kazhdan projection $P$ of $L$, because $uLu^{-1}\le L$
has fewer constraints and more fixed vectors, and stable finiteness of the
matrix corona forces $U^*PU=P$.  So conjugation by $u$ preserves the
Hilbert--Schmidt asymptotic commutant of $L$, and every element of
$\mathfrak D_G(L)$ tends to $1$ in Hilbert--Schmidt norm in every operator
norm asymptotic representation (Corollary~\ref{cor:defect-hs}).
Property~\textup{(T)} of $K$ turns this into triviality in operator norm:
a corona homomorphism nontrivial on $K$ compresses to the corner where $K$
has no fixed vectors, and the Kazhdan inequality there contradicts the
Hilbert--Schmidt triviality (Theorem~\ref{thm:normal-kazhdan}).\footnote{A
Hilbert--Schmidt bound on the multiplicative defect does not give an
operator norm bound on the conjugation maps, so this argument does not
apply to sofic or hyperlinear approximations.}
\leanverified{Manuscript/OneSidedMFRadical/PrintedRemarks}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptFullRadicalsDoNotEntailNonsoficity}

The stable letter of an ascending HNN extension of a property-\textup{(T)}
group along a proper self-embedding is a compressor.  The group $H$ realizes the compression inside a simple group, where
one nontrivial commutator $[ucu^{-1},\ell]$ normally generates everything.
Let
$L_{\F_2}(1,2)$ denote the binary Leavitt algebra, the unital
$\F_2$-algebra generated by $s_0,s_1,t_0,t_1$ subject to
\begin{equation}\label{eq:leavitt}
 t_i s_j=\delta_{ij},
 \qquad s_0t_0+s_1t_1=1.
\end{equation}
For a unital ring $R$, write $e_{ij}(a)=1+aE_{ij}$ and let $\EL_n(R)$ be
the subgroup of $\GL_n(R)$ generated by the elements $e_{ij}(a)$, $i\ne j$.

\begin{mainthm}[the binary Leavitt group]\label{thm:headline}
Put $R=L_{\F_2}(1,2)$ and $H=\EL_{12}(R)$.  Then $H$ is finitely
generated, nontrivial, simple, has property~\textup{(T)}, and every
homomorphism from $H$ to an MF group is trivial.  So $H$ is not MF, and
$C^*_{\mathrm r}(H)$ is separable and stably finite but not MF, while
$C^*_{\max}(H)$ is not even finite: it contains a proper isometry.
\leanverified{Manuscript/OneSidedMFRadical/ReducedCStarConsequence}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptPrintedHeadline}
\leanverified{Manuscript/OneSidedMFRadical/HeadlineTheorem}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptBinaryLeavittHeadline}
\leanverified{Manuscript/OneSidedMFRadical/ReducedCStarConsequence}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptReducedCStarConsequence}
\leanverified{Manuscript/OneSidedMFRadical/ReducedCStarConsequence}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptNotEverySeparableStablyFiniteCStarAlgebraIsMF}
\end{mainthm}

\begin{mainthm}[a torsion-free finitely presented example]
\label{thm:torsion-free}
There is a two-generated, finitely presented, torsion-free, acylindrically
hyperbolic group $Q$ with property~\textup{(T)} such that every
homomorphism from $Q$ to an MF group is trivial.  In particular, no
nontrivial quotient of $Q$ is MF\@.
\end{mainthm}

Section~\ref{sec:torsion-free} builds $Q$ from Fournier-Facio's
torsion-free property-\textup{(T)} group~\cite[\S2]{FFF} and Hull's small
cancellation theorem~\cite{Hull}.

\subsection*{Relation to prior work}

Blackadar and Kirchberg introduced MF algebras and proved that a separable
$C^*$-algebra is NF if and only if it is nuclear and MF~\cite{BK}.  Whether
every separable stably finite $C^*$-algebra is MF, the MF
problem~\cite[\S6]{GoldbringHart}, was answered negatively through the
failure of the Connes embedding problem~\cite{MIPRE}\cite[Proposition~6.1
and Remark~6.2]{GoldbringHart}.  Theorem~\ref{thm:headline} gives a
counterexample among reduced group $C^*$-algebras.  Whether every countable
group is MF was a longstanding open problem~\cite{BDL}.
\leanverified{Manuscript/OneSidedMFRadical/ReducedCStarConsequence}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptSeparableStablyFiniteNonMFCStarAlgebraExists}

Theorem~\ref{thm:compression-criterion} uses the configuration of the
nonsofic construction~\cite[Proposition~2.3]{OAI}: a property-\textup{(T)}
subgroup $L$ conjugated into itself by an element $u$, and an element $c$
commuting with $L$ whose conjugate $ucu^{-1}$ does not.  There, soficity
and the rigidity theorems of Kun~\cite{Kun16} and Kun--Thom~\cite{KT19}
force the centralizing subgroup to be locally embeddable into finite
groups, and a copy of Thompson's group $V$ gives the contradiction.  Here
the rigidity is the Kazhdan projection of $L$ in a norm matrix corona,
which stable finiteness of the corona forces to commute with the
compressor, and the contradiction is a commutator $[ucu^{-1},\ell]\ne1$
that every operator norm asymptotic representation sends to $1$ in
Hilbert--Schmidt norm.  The group $H$ is the unit group of the binary
Leavitt algebra, the group of~\cite{OAI}, and
Theorem~\ref{thm:torsion-free} reuses Fournier-Facio's
configuration~\cite{FFF}.  Bachner--Dogon--Lubotzky showed that for groups
of Deligne type, operator--Hilbert--Schmidt stability would imply that the
group is not MF~\cite[Proposition~1.5]{BDL}.  The compression relation
replaces stability.  For a short elementary construction of non-MF
generalized wreath products from one-sided compression,
see~\cite{Eckhardt}.

Leavitt introduced the algebras $L_K(1,n)$~\cite{Leavitt}.  The binary
Leavitt algebra is purely infinite simple~\cite{AbramsAranda}, is an
exchange ring~\cite{AraExchange}, and has center
$\F_2$~\cite{ArandaCrow}.  Preusser's sandwich theorem then gives
simplicity of $H$~\cite{Preusser}, and property~\textup{(T)} is
Ershov--Jaikin-Zapirain's theorem~\cite[Theorem~1.1]{EJZ}.

\section{One-sided compression}
\label{sec:compression-radical}

\subsection{Corona homomorphisms}

The image of a corona homomorphism from $G$ is a countable subgroup of a
corona unitary group, so MF, and every countable MF group embeds in the
unitary group of a norm matrix corona.
\leanverified{Manuscript/OneSidedMFRadical/PrintedDefinitions}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptCoronaImagesAndTargets}

\begin{lemma}\label{prop:mf-residual-calculus}
Let $G$ be countable and $K\le G$.  Every homomorphism from $G$ to an MF
group is trivial on $K$ if and only if every corona homomorphism from $G$
is trivial on $K$.
\leanverified{Manuscript/OneSidedMFRadical/LiteralMFClosure}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptMFResidualCalculusLiteral}
\end{lemma}

\begin{proof}
A corona homomorphism is a homomorphism to an MF group, and a homomorphism
to an MF group composed with an embedding of that group into a corona
unitary group is a corona homomorphism with the same kernel.
\end{proof}

\subsection{Kazhdan transport in normalized Hilbert--Schmidt norm}

For $a\in M_d(\C)$ write
\[
 \hsnorm{a}=\tr_d(a^*a)^{1/2},
 \qquad \tr_d=\frac1d\operatorname{Tr}.
\]
An \emph{operator norm asymptotic representation} of a countable group $G$
is a sequence of maps $V_n\colon G\to\U(d_n)$ such that $V_n(1)=1$ and
\[
 \opnorm{V_n(gh)-V_n(g)V_n(h)}\longrightarrow0
 \qquad(g,h\in G).
\]
The elements $g$ with $\hsnorm{V_n(g)-1}\to0$ form a normal subgroup of
$G$, because $\hsnorm{a}\le\opnorm{a}$ and the Hilbert--Schmidt norm is
unitarily invariant:
\begin{gather*}
 \hsnorm{V_n(gh)-1}\le\hsnorm{V_n(g)-1}+\hsnorm{V_n(h)-1}+o(1),\\
 \hsnorm{V_n(ghg^{-1})-1}=\hsnorm{V_n(h)-1}+o(1).
\end{gather*}
For $L\le G$, let
\[
 \mathcal C_2(V,L)
 =\left\{(x_n)\ \middle|\
   \begin{aligned}
    &\sup_n\opnorm{x_n}<\infty,\\
    &\hsnorm{[V_n(\ell),x_n]}\longrightarrow0 &&(\ell\in L)
   \end{aligned}\right\}
\]
be the bounded Hilbert--Schmidt asymptotic commutant of $V(L)$.  Coordinatewise
conjugation by $V_n(g)$ defines a bijection, denoted $\operatorname{Ad}(V(g))$,
of the bounded matrix sequences.

\begin{lemma}\label{lem:stable-finite}
For every sequence of positive integers $(m_n)$, the norm matrix corona
\[
 \prod_nM_{m_n}(\C)\big/\bigoplus_nM_{m_n}(\C)
\]
is stably finite.  Consequently, unitarily equivalent projections in a norm
matrix corona are equal whenever one is dominated by the other.
\leanverified{Manuscript/OneSidedMFRadical/StableFiniteness}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptNormMatrixCoronaStableFinite}
\end{lemma}

\begin{proof}
Write $A$ for the displayed corona.  If $v^*v=1$ in $A$ and $(x_n)$ is a
bounded lift of $v$, then $\opnorm{x_n^*x_n-1}\to0$, so for all large $n$
the matrix $u_n=x_n(x_n^*x_n)^{-1/2}$ is unitary and $\opnorm{u_n-x_n}\to0$;
after assigning arbitrary unitary values to the finitely many remaining
coordinates, $(u_n)$ is a unitary lift of $v$, and $vv^*=1$.  So $A$ is
finite, and so is $M_k(A)$, which is the corona with matrix sizes $km_n$.
Finally, if $p\le q$ are equivalent projections in a stably finite algebra
$B$ and $w$ is a partial isometry with $w^*w=q$ and $ww^*=p$, then
$w\in qBq$ is an isometry in the finite corner $qBq$, so $p=ww^*=q$.
\end{proof}

\begin{lemma}[one-sided order for the Kazhdan projection]
\label{lem:kazhdan-projection-order}
Let $L$ have property~\textup{(T)}, let $B$ be a unital $C^*$-algebra, and
let $\pi\colon L\to\U(B)$ be a homomorphism.  Denote by
$P\in B$ the image of the Kazhdan projection under the extension
$C^*_{\max}(L)\to B$.  If $U\in\U(B)$ satisfies
\[
 U\pi(L)U^*\subseteq\pi(L),
\]
then
\[
 U^*PU\le P.
\]
\leanverified{Manuscript/OneSidedMFRadical/KazhdanProjectionOrderLiteral}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptLiteralMaximalCStarKazhdanProjectionOrder}
\end{lemma}

\begin{proof}
Represent $B$ faithfully and nondegenerately on a Hilbert space
$\mathcal H$.  Then $P$ is the orthogonal projection onto
$\operatorname{Fix}\pi(L)$.  If $\xi$ is $L$-fixed and $\ell\in L$, then
$\pi(\ell)U^*\xi=U^*(U\pi(\ell)U^*)\xi=U^*\xi$ because $U\pi(\ell)U^*$
belongs to $\pi(L)$.  So $U^*\operatorname{Fix}\pi(L)\subseteq\operatorname{Fix}\pi(L)$,
and the range projection $U^*PU$ is dominated by $P$ in $B(\mathcal H)$,
and then in $B$ by faithfulness.
\end{proof}

\begin{theorem}[one-sided Kazhdan transport]\label{thm:transport}
Let $L\le G$ have property~\textup{(T)}, and let $u\in G$ satisfy
$uLu^{-1}\le L$.  Let $(V_n)$ be an operator norm asymptotic
representation of $G$.  Then
\[
 \operatorname{Ad}(V(u))\bigl(\mathcal C_2(V,L)\bigr)
 =\mathcal C_2(V,L).
\]
\leanverified{Manuscript/OneSidedMFRadical/TransportCommutantEquality}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptOneSidedKazhdanTransportCommutantEquality}
\end{theorem}

\begin{proof}
Fix $x=(x_n)\in\mathcal C_2(V,L)$ and a free ultrafilter $\omega$, and let
$\mathcal K_\omega$ be the ultraproduct of the Hilbert spaces
$(M_{d_n}(\C),\hsnorm{\,\cdot\,})$.  For unitaries $A,B$,
\[
 \opnorm{\operatorname{Ad}(A)-\operatorname{Ad}(B)}\le2\opnorm{A-B},
\]
so the maps $\operatorname{Ad}(V_n(g))$ are asymptotically multiplicative
in operator norm, and
\[
 \sigma(g)=\lim_\omega\operatorname{Ad}(V_n(g))
\]
is a unitary representation of $G$ on $\mathcal K_\omega$ with
$\xi=[x_n]_\omega\in\operatorname{Fix}\sigma(L)$.  The same estimate makes
\[
 \widetilde\sigma(g)=[\operatorname{Ad}(V_n(g))]_n\in\U(\mathcal B),
 \qquad
 \mathcal B=\prod_nB(M_{d_n}(\C))\big/\bigoplus_nB(M_{d_n}(\C)),
\]
a homomorphism, and $\mathcal B$ is a norm matrix corona with coordinate
sizes $d_n^2$ after a choice of matrix units.  Let $P\in\mathcal B$ be the
image of the Kazhdan projection of $L$~\cite{AkemannWalter} under
$C^*_{\max}(L)\to\mathcal B$.  On $\mathcal K_\omega$ its range is
$\operatorname{Fix}\sigma(L)$.  Put $U=\widetilde\sigma(u)$.  Since
$uLu^{-1}\le L$,
\[
 U\widetilde\sigma(L)U^*\subseteq\widetilde\sigma(L),
 \qquad\text{so}\qquad
 U^*PU\le P
\]
by Lemma~\ref{lem:kazhdan-projection-order}, and
\[
 U^*PU=P
\]
by Lemma~\ref{lem:stable-finite}, since the two projections are unitarily
equivalent.  So $U$ commutes with $P$, and
\[
\begin{aligned}
 &U\xi,\ U^*\xi\in\operatorname{Fix}\sigma(L),\qquad\text{that is,}\\
 &\lim_\omega\hsnorm{[V_n(\ell),V_n(u)^{\pm1}x_nV_n(u)^{\mp1}]}=0
 \qquad(\ell\in L).
\end{aligned}
\]
Since $\omega$ was arbitrary, the limits hold along the full sequence: a
subsequence staying above some $\varepsilon>0$ would lie in a free
ultrafilter and give a nonzero ultralimit.  So $\operatorname{Ad}(V(u))$
and its inverse both preserve $\mathcal C_2(V,L)$.
\end{proof}

\begin{corollary}\label{cor:defect-hs}
Let $L\le G$ have property~\textup{(T)} and let $(V_n)$ be an operator
norm asymptotic representation of $G$.  Then
\[
 \hsnorm{V_n(d)-1}\longrightarrow0
 \qquad(d\in\mathfrak D_G(L)).
\]
\leanverified{Manuscript/OneSidedMFRadical/DefectHS}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptCompressionDefectHSInvisible}
\leanverified{Manuscript/OneSidedMFRadical/PrintedDefectShadow}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptPrintedDefectShadowInclusion}
\end{corollary}

\begin{proof}
Let $u\in\operatorname{Comp}_G(L)$, $c\in C_G(L)$, and $\ell\in L$, and
write $\mathcal C_2=\mathcal C_2(V,L)$.  Then
\[
 (V_n(c))\in\mathcal C_2
 \ \Longrightarrow\
 (V_n(u)V_n(c)V_n(u)^*)\in\mathcal C_2
 \ \Longrightarrow\
 (V_n(ucu^{-1}))\in\mathcal C_2,
\]
the first membership because $c$ commutes with $L$, the first implication
by Theorem~\ref{thm:transport}, and the second because
$\opnorm{V_n(u)V_n(c)V_n(u)^*-V_n(ucu^{-1})}\to0$.  So
$\hsnorm{[V_n(\ell),V_n(ucu^{-1})]}\to0$, that is,
$\hsnorm{V_n([ucu^{-1},\ell])-1}\to0$.  The elements with this property
form a normal subgroup of $G$, so it contains $\mathfrak D_G(L)$.
\end{proof}

\subsection{From Hilbert--Schmidt to operator norm}

A corona homomorphism $\rho$ cannot simply be restricted to a corner: a
projection $q$ commuting with $\rho(G)$ has lifts $q_n$ that only
asymptotically commute with the lifts of $\rho(g)$, so the compressions
$q_n\rho(g)q_n$ are only approximately unitary.

\begin{lemma}[central corona corners]\label{lem:central-corona-corner}
Let \(\rho\colon G\to\U(\mathcal Q_{\mathbf d})\) be a homomorphism from a
countable group, and let \(q\in\mathcal Q_{\mathbf d}\) be a nonzero
projection commuting with \(\rho(G)\).  Then, after passing to an infinite
coordinate subsequence, there are nonzero projections
\(q_n\in M_{d_n}(\C)\), integers \(r_n=\operatorname{rank}(q_n)\), unitary
identifications \(J_n\colon\C^{r_n}\to q_n\C^{d_n}\), and an operator norm
asymptotic representation
\[
 W_n\colon G\longrightarrow\U(r_n)
\]
with \(W_n(1)=I_{r_n}\) such that, in the corona over the retained
coordinates, the class of \((J_nW_n(g)J_n^*)\) is the coordinate restriction
of \(q\rho(g)\), a unitary of the corner \(q\mathcal Q_{\mathbf d}q\) with
unit \(q\).
\leanverified{Manuscript/OneSidedMFRadical/CentralCoronaCorner}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptCentralCoronaCorner}
\end{lemma}

\begin{proof}
Lift \(q\) to projections \(q_n\) by spectral rounding, and lift each
\(\rho(g)\) to unitaries \(U_n(g)\) by polar correction, with
\(U_n(1)=I_{d_n}\).  Since \(q\ne0\), infinitely many \(q_n\) are nonzero.
Retain those coordinates and put \(r_n=\operatorname{rank}(q_n)\).  The
commutation relation in the corona gives
\(\opnorm{q_nU_n(g)-U_n(g)q_n}\to0\) for each \(g\), so
\cite[Proposition~2.4]{BDL} with \(p=\infty\) gives an operator norm
asymptotic representation \(\widetilde W_n\) of \(G\) by unitaries of the
corner \(q_nM_{d_n}(\C)q_n\) with
\(\opnorm{\widetilde W_n(g)-q_nU_n(g)q_n}\to0\).  Replacing
\(\widetilde W_n(1)\) by \(q_n\) changes it by \(o(1)\).  Choose unitaries
\(J_n\colon\C^{r_n}\to q_n\C^{d_n}\) and put
\(W_n(g)=J_n^*\widetilde W_n(g)J_n\).  Conjugation by \(J_n\) preserves
operator norms and multiplicative defects, so \((W_n)\) is an operator
norm asymptotic representation with \(W_n(1)=I_{r_n}\), and
\(\opnorm{J_nW_n(g)J_n^*-q_nU_n(g)q_n}\to0\) identifies the class of
\((J_nW_n(g)J_n^*)\) with the coordinate restriction of \(q\rho(g)\).
\end{proof}

\begin{theorem}[normal Kazhdan subgroups]\label{thm:normal-kazhdan}
Let \(G\) be countable and let \(K\trianglelefteq G\) have
property~\textup{(T)}.  If every operator norm asymptotic representation
\((V_n)\) of \(G\) satisfies \(\hsnorm{V_n(k)-1}\to0\) for all \(k\in K\),
then every corona homomorphism from \(G\) is trivial on \(K\).
\leanverified{Manuscript/OneSidedMFRadical/NormalKazhdan}{GroupApproximation.Manuscript.OneSidedMFRadical.manuscriptNormalKazhdanRadical}
\end{theorem}

\begin{proof}
Suppose that a corona homomorphism
\(\Theta\colon G\to\U(\mathcal Q_{\mathbf d})\) is nontrivial on \(K\).
Replace \(G\) by \(\Theta(G)\) and \(K\) by \(\Theta(K)\): the new \(K\)
is a nontrivial normal property-\textup{(T)} subgroup, since
property~\textup{(T)} passes to quotients, and it still satisfies the
hypothesis, because every operator norm asymptotic representation of
\(\Theta(G)\) pulls back along \(G\to\Theta(G)\) to one of \(G\).

Let \(p\in\mathcal Q_{\mathbf d}\) be the image of the Kazhdan projection
of \(K\), the projection onto the \(K\)-fixed vectors in any faithful
representation of the corona.  Since \(gKg^{-1}=K\),
\[
 \Theta(g)p\Theta(g)^*=p\qquad(g\in G),
\]
and \(q=1-p\) is nonzero, because \(p=1\) would make every element of \(K\)
act trivially.  Lemma~\ref{lem:central-corona-corner} gives an operator
norm asymptotic representation \(W_n\colon G\to\U(r_n)\) whose corona
homomorphism \(\widehat\Theta\) is the coordinate restriction of
\(g\mapsto q\Theta(g)\).  The Kazhdan projection of \(K\) under
\(\widehat\Theta\) is the coordinate restriction of \(qp=0\), so
\(\widehat\Theta|_K\) has no nonzero fixed vectors, and for a finite
Kazhdan set \(S\subseteq K\) with constant \(\kappa>0\),
\[
 b=\frac1{|S|}\sum_{s\in S}
       (\widehat\Theta(s)-1)^*(\widehat\Theta(s)-1)
 \ \ge\ \frac{\kappa^2}{|S|}\,1
\]
in \(\mathcal Q_{\mathbf r}\).  The coordinates
\(b_n=\frac1{|S|}\sum_{s\in S}(W_n(s)-I_{r_n})^*(W_n(s)-I_{r_n})\)
represent \(b\), so the negative part of \(b_n-(\kappa^2/|S|)I_{r_n}\)
tends to zero in operator norm, and normalized traces on \(M_{r_n}(\C)\)
give
tends to zero in operator norm, and taking normalized traces on
\(M_{r_n}(\C)\) gives
\[
 \frac1{|S|}\sum_{s\in S}\hsnorm{W_n(s)-I_{r_n}}^{\,2}
 \ge \frac{\kappa^2}{|S|}-o(1).
\]
After passing to a subsequence, one fixed \(s_0\in S\) stays a positive
Hilbert--Schmidt distance from \(I_{r_n}\), against the hypothesis applied
to \((W_n)\) and \(s_0\in K\).
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:compression-criterion}]
By Corollary~\ref{cor:defect-hs}, every operator norm asymptotic
representation of \(G\) satisfies \(\hsnorm{V_n(k)-1}\to0\) for
\(k\in K\le\mathfrak D_G(L)\).  Theorem~\ref{thm:normal-kazhdan} then makes
every corona homomorphism trivial on \(K\), and
Lemma~\ref{prop:mf-residual-calculus} makes every homomorphism to an MF
group trivial on \(K\).  If \(K\ne1\), then \(G\) does not embed in an MF
group, so \(G\) is not MF\@.  The case \(K=G=\mathfrak D_G(L)\) gives the
last assertion.
\end{proof}

\subsection{The maximal group \texorpdfstring{$C^*$}{C*}-algebra}

Lemma~\ref{lem:kazhdan-projection-order} also applies inside
$C^*_{\max}(G)$ itself, where a strict compression produces a proper
isometry.

\begin{lemma}[strictly dominated projections]\label{lem:proper-isometry}
Let $A$ be a unital $C^*$-algebra containing a projection $p$ and a
unitary $u$ with $p<upu^{*}$.  Then $s=pu^{*}+(1-upu^{*})$ is an isometry
that is not a unitary, so $A$ is not stably finite and has no faithful
tracial state.
\leanverified{Analysis/ProperIsometryStrictOrder}{GroupApproximation.manuscriptProperIsometryStrictOrder}
\end{lemma}

\begin{proof}
Put $q=upu^{*}$, so that $pq=qp=p$, $up(1-q)=0$, and
$pu^{*}(1-q)=pu^{*}-pu^{*}upu^{*}=0$.  Then
\begin{align*}
 s^{*}s&=upu^{*}+up(1-q)+(1-q)pu^{*}+(1-q)=q+(1-q)=1,\\
 ss^{*}&=pu^{*}up+pu^{*}(1-q)+(1-q)up+(1-q)\\
       &=p+(1-q)=1-(q-p)\ne1.
\end{align*}
A unital $C^*$-algebra containing a nonunitary isometry is not finite, and
neither is any matrix algebra over it, since the isometry embeds in the
corner.  A tracial state $\tau$ has
$\tau(q-p)=\tau(s^{*}s)-\tau(ss^{*})=0$ with $q-p$ a nonzero positive
element, so no tracial state on $A$ is faithful.
\end{proof}

\begin{proposition}[proper one-sided compression]\label{prop:max-infinite}
Let $G$ be a countable group containing a property-\textup{(T)} subgroup
$\Gamma$ and an element $t$ with $t\Gamma t^{-1}\subsetneq\Gamma$.  Then
$C^*_{\max}(G)$ contains a proper isometry.  In particular,
$C^*_{\max}(G)$ is not stably finite, admits no faithful tracial state,
and is neither residually finite-dimensional nor MF\@.
\leanverified{Analysis/StrictCompressionFromPrinted}{GroupApproximation.MaximalCStarPrintedHypotheses.manuscriptMaximalCStarRemarkFromPrintedHypotheses}
\leanverified{Analysis/QuasiRegularWitness}{GroupApproximation.QuasiRegularWitness.baseVector_apply_base}
\end{proposition}

\begin{proof}
Let $P\in C^*_{\max}(G)$ be the image of the Kazhdan projection of
$\Gamma$~\cite{AkemannWalter} under the canonical map
$C^*_{\max}(\Gamma)\to C^*_{\max}(G)$, and let $u$ be the canonical
unitary of $t$.  Conjugation by $u$ implements the isomorphism
$\Gamma\to t\Gamma t^{-1}$ on canonical unitaries, so
Lemma~\ref{lem:kazhdan-projection-order}, applied with $U=u$, gives
$P\le uPu^{*}$ with $uPu^{*}$ the image of the Kazhdan projection of
$t\Gamma t^{-1}$.  The inequality is strict: in the quasi-regular
representation of $G$ on $\ell^2(G/t\Gamma t^{-1})$, the point mass at
the base coset is fixed by $t\Gamma t^{-1}$ and moved by every element of
$\Gamma\setminus t\Gamma t^{-1}$, so the two fixed-space projections
differ.  Lemma~\ref{lem:proper-isometry} then supplies the proper
isometry and rules out stable finiteness and faithful traces.  MF
algebras are stably finite, and a separable residually
finite-dimensional $C^*$-algebra is MF~\cite{BK}.
\end{proof}

\section{The binary Leavitt group}

The Leavitt algebra $R=L_{\F_2}(1,2)$ is built so that one copy of the
free module $R$ is isomorphic to two copies: the relations~\eqref{eq:leavitt}
say that $s_0,s_1$ embed two copies of $R$ side by side and $t_0,t_1$ read
them back.  So a matrix $A$ over $R$ can be copied into one of the two
halves, with the identity on the other half, and this compression is an
injective endomorphism of the matrix ring that fixes the identity.  Inside
a large enough elementary group the compression is realized by
conjugation, which gives a compressor $\tau$ for a copy $L$ of $\EL_3(R)$;
one commutator computation gives a nontrivial element of
$\mathfrak D_H(L)$, and simplicity of $H$ turns it into all of $H$.

Throughout this section $R=L_{\F_2}(1,2)$ and
\[
 p=s_0t_0,\qquad q=s_1t_1,
\]
so that $p+q=1$ and $t_1qs_1=1$, and in particular $q\ne0$.  We use indices
$0,\ldots,11$ for $12\times12$ matrices and identify $\GL_3(R)$ with the
subgroup of $\GL_{12}(R)$ of matrices $\operatorname{diag}(A,I_9)$.  Define
$\Psi\colon M_3(R)\to M_3(R)$ by
\begin{equation}\label{eq:matrix-compression}
 \Psi(A)=qI_3+s_0A t_0,
\end{equation}
where $s_0At_0$ is the matrix $(s_0a_{ij}t_0)_{ij}$.  The relations
$q^2=q$, $qs_0=t_0q=0$, and $t_0s_0=1$ show that $\Psi$ is multiplicative,
$\Psi(I_3)=(p+q)I_3=I_3$, and $t_0\Psi(A)s_0=A$.  So $\Psi$ is an injective
unital ring endomorphism and restricts to an injective group homomorphism
$\GL_3(R)\to\GL_3(R)$.  Put
\[
 X=\begin{pmatrix}s_0I_3&s_1t_0I_3\\0&t_1I_3\end{pmatrix},
 \qquad
 Y=\begin{pmatrix}t_0I_3&0\\s_0t_1I_3&s_1I_3\end{pmatrix};
\]
block multiplication with~\eqref{eq:leavitt} gives $Y=X^{-1}$ and
$X\operatorname{diag}(A,I_3)Y=\operatorname{diag}(\Psi(A),I_3)$.  The matrix
$X$ need not lie in an elementary group, which is why there are twelve
coordinates: with
\begin{equation}\label{eq:tau}
 \tau=\operatorname{diag}(X,Y)\in\GL_{12}(R)
\end{equation}
one has
\begin{equation}\label{eq:corner-conjugation}
 \tau\,\operatorname{diag}(A,I_9)\,\tau^{-1}
 =\operatorname{diag}(\Psi(A),I_9).
\end{equation}

\begin{lemma}\label{lem:tau-elementary}
The matrix $\tau$ defined in \eqref{eq:tau} belongs to $\EL_{12}(R)$.
\leanverified{Manuscript/OneSidedMFRadical/RankTwelveConfiguration}{GroupApproximation.Manuscript.OneSidedMFRadical.RankTwelveEndpoint.printedTauGL_mem}
\end{lemma}

\begin{proof}
By the Whitehead lemma~\cite[\S3]{Milnor}, $\operatorname{diag}(X,X^{-1})$
is a product of block-triangular matrices
$\begin{psmallmatrix}I&B\\0&I\end{psmallmatrix}$ and
$\begin{psmallmatrix}I&0\\B&I\end{psmallmatrix}$ with $B\in M_6(R)$, and
each of these is the product of the elementary matrices $e_{r,6+s}(b_{rs})$,
respectively $e_{6+r,s}(b_{rs})$, since distinct off-diagonal matrix units
have product zero.
\end{proof}

Set $H=\EL_{12}(R)$, and let $L=\EL_3(R)\le H$ denote the image of the
embedding $A\mapsto\operatorname{diag}(A,I_9)$.

\begin{proposition}\label{prop:leavitt-compression}
The groups $H$ and $L$ have property~\textup{(T)}, and $\tau\in H$ satisfies
\begin{equation}\label{eq:compresses-L}
 \tau L\tau^{-1}\le L.
\end{equation}
\leanverified{Manuscript/OneSidedMFRadical/SentenceCurrentLeavittCompressionClosure}{GroupApproximation.Manuscript.OneSidedMFRadical.RankTwelveEndpoint.manuscriptCurrentLeavittCompressionProposition}
\end{proposition}

\begin{proof}
The ring $R$ is generated by $s_0,s_1,t_0,t_1$ as a unital ring, so the
theorem of Ershov and Jaikin-Zapirain gives property~\textup{(T)} for
$\EL_{12}(R)$ and $\EL_3(R)$~\cite[Theorem~1.1]{EJZ}.
Lemma~\ref{lem:tau-elementary} gives $\tau\in H$.  For $i\ne j$ and
$a\in R$, equations \eqref{eq:corner-conjugation}
and~\eqref{eq:matrix-compression} give
$\tau e_{ij}(a)\tau^{-1}=e_{ij}(s_0at_0)\in L$, and these elementary
matrices generate $L$.
\end{proof}

\begin{proposition}\label{prop:simple}
The group $H=\EL_{12}(R)$ is nontrivial and simple.
\leanverified{Manuscript/OneSidedMFRadical/RankTwelveSimplicity}{GroupApproximation.Manuscript.OneSidedMFRadical.RankTwelveEndpoint.manuscriptPropositionSimple}
\end{proposition}

\begin{proof}
The ring $R$ is purely infinite simple~\cite{AbramsAranda}, so it is an
exchange ring~\cite{AraExchange}.  Let $N\trianglelefteq H$.  Since
$N\le\GL_{12}(R)$ is normalized by $\EL_{12}(R)$, Preusser's sandwich
theorem~\cite[Theorem~3]{Preusser} gives an ideal $I\trianglelefteq R$ with
$\EL_{12}(R,I)\le N\le C_{12}(R,I)$.  Simplicity of $R$ gives $I=0$ or
$I=R$, and $I=R$ gives $N=H$.  If $I=0$, then $N$ centralizes every
$e_{ij}(a)$: centralizing $e_{ij}(1)$ for all $i\ne j$ forces $g=\lambda I_{12}$
with $\lambda\in R^\times$, and centralizing $e_{ij}(a)$ forces
$\lambda\in Z(R)^\times$.  Since $Z(R)=\F_2$~\cite[Corollary~4.3]{ArandaCrow},
$N=1$.  Finally, $e_{01}(1)\ne1$.
\end{proof}

\begin{proposition}\label{prop:defect}
Let
\[
 c=e_{34}(1),\qquad \ell=e_{12}(1),\qquad
 d=[\tau c\tau^{-1},\ell]
\]
in $H$.  Then
\[
 c\in C_H(L),\qquad d=e_{02}(q)\ne1,
 \qquad \normal d_H=H.
\]
\leanverified{Manuscript/OneSidedMFRadical/RankTwelveConfiguration}{GroupApproximation.Manuscript.OneSidedMFRadical.RankTwelveEndpoint.manuscriptPropositionDefect}
\end{proposition}

\begin{proof}
The Steinberg relations give $[e_{ij}(a),e_{34}(1)]=1$ for
$i,j\in\{0,1,2\}$, and these matrices generate $L$, so $c\in C_H(L)$.
Since $\tau=\operatorname{diag}(X,Y)$ and $E_{34}$ is supported in the
first six coordinates, $\tau E_{34}\tau^{-1}=\operatorname{diag}(XE_{34}Y,0)$
with $E_{34}$ read in $M_6(R)$, and with $\varepsilon_0,\varepsilon_1,\varepsilon_2$
the standard basis of $R^3$,
\[
 XE_{34}Y
 =\begin{pmatrix}s_1t_0\varepsilon_0\\ t_1\varepsilon_0\end{pmatrix}
  \begin{pmatrix}s_0t_1\varepsilon_1^{\mathsf T}& s_1\varepsilon_1^{\mathsf T}\end{pmatrix}
 =\begin{pmatrix}s_1t_1E_{01}&0\\0&E_{01}\end{pmatrix}
\]
by~\eqref{eq:leavitt}.  So
\[
 \tau c\tau^{-1}=1+qE_{01}+E_{34}=e_{01}(q)e_{34}(1).
\]
Direct multiplication of the matrices in~\eqref{eq:tau} gives
\[
 \tau c\tau^{-1}=e_{01}(q)e_{34}(1).
\]
The second factor commutes with $\ell$, and
$[e_{ij}(a),e_{jk}(b)]=e_{ik}(ab)$ gives $d=e_{02}(q)$, which is
nontrivial because $q\ne0$.  Since $H$ is simple, $d$ normally generates
$H$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:headline}]
Proposition~\ref{prop:leavitt-compression} gives property~\textup{(T)},
which implies finite generation~\cite[Theorem~1.3.1]{BHV}.  The
containment~\eqref{eq:compresses-L}, the centrality of $c$ relative to $L$,
and Proposition~\ref{prop:defect} put $d$ in $\mathfrak D_H(L)$, and
$\normal d_H=H$ gives $\mathfrak D_H(L)=H$.  So
Theorem~\ref{thm:compression-criterion}, applied with $K=H$, makes every
homomorphism from $H$ to an MF group trivial, and
Proposition~\ref{prop:simple} makes $H$ nontrivial and simple.  In
particular $H$ is not MF\@.  The algebra $C^*_{\mathrm r}(H)$ is separable,
and stably finite because its canonical trace is faithful.  If an embedding
$e\colon C^*_{\mathrm r}(H)\to\mathcal Q_{\mathbf d}$ existed and $p=e(1)$, then $h\mapsto e(\lambda_h)+(1-p)$ would embed $H$ in
$\U(\mathcal Q_{\mathbf d})$.
Finally, the compression~\eqref{eq:compresses-L} is strict: by
\eqref{eq:corner-conjugation} and~\eqref{eq:matrix-compression}, every
off-diagonal entry of a matrix in $\tau L\tau^{-1}$ has the form
$s_0at_0$, and
\[
 t_1(s_0at_0)s_1=0\ne1=t_1\cdot1\cdot s_1,
\]
so $e_{12}(1)\in L\setminus\tau L\tau^{-1}$.
$s_0at_0$, and $t_1(s_0at_0)s_1=0$ while $t_1\cdot1\cdot s_1=1$, so
$e_{12}(1)\in L\setminus\tau L\tau^{-1}$.
Proposition~\ref{prop:max-infinite}, applied to $L\le H$ and $\tau$,
puts a proper isometry in $C^*_{\max}(H)$.
\end{proof}

\section{A torsion-free finitely presented example}
\label{sec:torsion-free}

For an acylindrically hyperbolic group $G$, choose the generating set $A$
provided by Hull~\cite[Theorem~3.12]{Hull}.  A subgroup is \emph{suitable}
if it acts non-elementarily on $\Gamma(G,A)$ and normalizes no nontrivial
finite subgroup~\cite[Definition~1.4]{Hull}.  Hull's small cancellation
theorem~\cite[Theorem~7.1]{Hull} is stated with injectivity on a ball of
$\Gamma(G,A)$, and a ball containing a given finite set gives the following
form.

\begin{theorem}[Hull's small cancellation theorem]\label{thm:hull}
Let $G$ be acylindrically hyperbolic, let $N\le G$ be suitable with respect
to $A$, and let $g_1,\dots,g_m\in G$ and a finite subset $\Omega\subseteq G$
be given.  Then there is a surjective homomorphism $\varphi\colon G\to Q$
such that $Q$ is acylindrically hyperbolic, $\varphi|_\Omega$ is injective,
$\varphi(g_i)\in\varphi(N)$ for all $i$, and every element of finite order
in $Q$ is the image of an element of the same order in $G$.
\end{theorem}

Hull's proof treats $m=1$ by passing to $G/\normal{W}_G$ for one word $W$
and the general case by induction on $m$~\cite[proof of Theorem~7.1]{Hull},
so $\ker\varphi$ is the normal closure of $m$ elements and $Q$ is finitely
presented when $G$ is.

\begin{lemma}[saturation]\label{lem:saturation}
Let $G$ be finitely presented, torsion-free, and acylindrically hyperbolic,
let $N\trianglelefteq G$ be nontrivial, and let $\Omega\subseteq G$ be finite.
Then there is a surjective homomorphism $\varphi\colon G\to Q$ such that $Q$
is two-generated, finitely presented, torsion-free, and acylindrically
hyperbolic, $\varphi|_\Omega$ is injective, and $\varphi(N)=Q$.
\end{lemma}

\begin{proof}
The subgroup $N$ is infinite and normal, so it acts non-elementarily on
$\Gamma(G,A)$ by Osin~\cite[Lemma~7.1]{Osin}, and torsion-freeness makes
it suitable.  By Hull~\cite[Corollary~5.7 and Lemma~5.8]{Hull}, $N$ contains
two elements $h_1,h_2$ such that $N_0=\langle h_1,h_2\rangle$ is again
suitable.  Apply Theorem~\ref{thm:hull} to $N_0$ with $g_1,\dots,g_m$ a
finite generating set of $G$ and with the set $\Omega$.  Then
$Q=\langle\varphi(g_1),\dots,\varphi(g_m)\rangle\le\varphi(N_0)\le\varphi(N)\le Q$,
so $Q=\varphi(N_0)=\langle\varphi(h_1),\varphi(h_2)\rangle$ and
$\varphi(N)=Q$.  The quotient is torsion-free by Theorem~\ref{thm:hull} and
finitely presented because $\ker\varphi$ is normally generated by $m$
elements.
\end{proof}

Fournier-Facio constructs a finitely presented torsion-free group $G_0$
with property~\textup{(T)}, a subgroup $\Gamma\le G_0$ with
property~\textup{(T)}, an element $t\in G_0$ with
$t\Gamma t^{-1}\le\Gamma$, and a subgroup $J\le G_0$ isomorphic to a
finitely presented infinite simple group, such that $[\Gamma,J]=1$ and
$tJt^{-1}\le\Gamma$~\cite[\S2]{FFF}.  The group $G_0$ is obtained there
from Hull's common quotient theorem~\cite[Corollary~7.4]{Hull}, so it is
acylindrically hyperbolic.  Put $S=tJt^{-1}$.  Since $J$ is simple and
nonabelian, $J$ and $S$ are perfect.

\begin{lemma}\label{lem:commutator-in-defect}
For every homomorphism $\rho\colon G_0\to\bar G$,
\[
 \rho(S)\le\mathfrak D_{\bar G}(\rho(\Gamma)).
\]
\end{lemma}

\begin{proof}
Write $\bar\Gamma=\rho(\Gamma)$, $\bar t=\rho(t)$, $\bar J=\rho(J)$, and
$\bar S=\rho(S)=\bar t\bar J\bar t^{-1}$.  Then
$\bar t\bar\Gamma\bar t^{-1}\le\bar\Gamma$, $\bar J$ centralizes
$\bar\Gamma$, and $\bar S\le\bar\Gamma$.  For $c\in\bar J$ and
$\ell\in\bar S$, the commutator $[\bar tc\bar t^{-1},\ell]$ lies in
$\mathfrak D_{\bar G}(\bar\Gamma)$ by~\eqref{eq:intrinsic-defect}.  As $c$
ranges over $\bar J$, the element $\bar tc\bar t^{-1}$ ranges over
$\bar S$, so $[\bar S,\bar S]\le\mathfrak D_{\bar G}(\bar\Gamma)$.  The
group $\bar S$ is perfect as a quotient of $S$, so
$\bar S\le\mathfrak D_{\bar G}(\bar\Gamma)$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:torsion-free}]
Let $N$ be the normal closure of $S$ in $G_0$, and fix $s\in S$ with
$s\ne1$.  Lemma~\ref{lem:saturation} applied to $G_0$, $N$, and the finite
set $\{1,s\}$ gives $\varphi\colon G_0\to Q$ with $Q$ two-generated,
finitely presented, torsion-free, and acylindrically hyperbolic,
$\varphi(N)=Q$, and $\varphi(s)\ne1$.  The group $Q$ has
property~\textup{(T)} as a quotient of $G_0$.

Let $r\colon Q\to M$ be a homomorphism to an MF group with image $\bar G$,
and put $\rho=r\circ\varphi$.  Since $\varphi(N)=Q$, the normal closure of
$\rho(S)$ in $\bar G$ is $\rho(N)=\bar G$, and by
Lemma~\ref{lem:commutator-in-defect} it lies in the normal subgroup
$\mathfrak D_{\bar G}(\rho(\Gamma))$, so $\mathfrak D_{\bar G}(\rho(\Gamma))=\bar G$.
The group $\bar G$ has property~\textup{(T)} as a quotient of $Q$, and
$\rho(\Gamma)$ has property~\textup{(T)} as a quotient of $\Gamma$.
Theorem~\ref{thm:compression-criterion}, applied to $\bar G$ with the
subgroup $\rho(\Gamma)$ and $K=\bar G$, makes every homomorphism from
$\bar G$ to an MF group trivial.  But $\bar G$ is a subgroup of the MF
group $M$, so it is MF, and its identity map is such a homomorphism.  So
$\bar G=1$, and $r$ is trivial.  A homomorphism from a quotient of $Q$
composes to one from $Q$, so no nontrivial quotient of $Q$ is MF\@.
\end{proof}
