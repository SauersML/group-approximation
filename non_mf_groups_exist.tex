\title[Spectral compression]{Spectral compression and non-MF groups}
\author{\paperauthor}
\subjclass[2020]{Primary 46L05; Secondary 20F05, 22D25, 20E26}
\keywords{MF groups, approximate unitary representations, property~(T), spectral rigidity, permutational wreath products}

\begin{abstract}
We prove that non-MF groups exist.  More precisely, we construct a split
quotient $q\colon W\twoheadrightarrow G$ whose nontrivial kernel is normally
generated by one involution, such that $W$ is sofic and non-MF, $G$ is
residually finite and MF, and every homomorphism from $W$ to a norm matrix
corona already factors through $q$.  The same quotient is invisible to finite,
compact, and finite-dimensional linear targets.

The mechanism is a spectral rigidity theorem for one-sided compression of a
Kazhdan subgroup.  In a norm matrix corona, a commuting spectral packet can
move only inside the connected components of its spectrum.  We apply this to
compute the universal finite-dimensional visible quotient of the compression
wreath products $W_K=K^{(X)}\rtimes G$ for every nontrivial finite perfect
group $K$: every such observer forgets exactly the whole lamp subgroup.  Thus
non-MF-ness follows from an exact structural computation rather than a
separate obstruction.  The choice $K=A_5$ is needed only to make the invisible
kernel normally generated by one involution.
\end{abstract}

\section{Introduction}

A countable group is \emph{MF} if it embeds in the unitary group of a norm
matrix corona
\[
 \mathcal Q_{\mathbf d}=\prod_nM_{d_n}(\C)\big/\bigoplus_nM_{d_n}(\C).
\]
Equivalently, it has asymptotically multiplicative finite-dimensional unitary
models in operator norm which asymptotically separate its nonidentity
elements.  This is the group property of Carri\'on--Dadarlat--Eckhardt
\cite{CDE}.  Residually finite and LEF groups are MF, and amenable groups are
MF \cite{CDE,SchafhauserAF}.  Whether every countable group is MF had remained
open \cite{Schafhauser,Shulman}.

\begin{mainthm}[one invisible relation changes MF]\label{thm:headline}
There are explicit finitely generated groups $W$ and $G$, a split epimorphism
$q\colon W\twoheadrightarrow G$, and an involution $r\in W\setminus\{1\}$
such that $\ker q=\langle\!\langle r\rangle\!\rangle_W$.  The group $W$ is
sofic and non-MF, whereas $G$ is residually finite and MF.  For every norm
matrix corona, precomposition gives a bijection
\[
 \operatorname{Hom}(G,\U(\mathcal Q_{\mathbf d}))
 \xrightarrow{\ \cong\ }
 \operatorname{Hom}(W,\U(\mathcal Q_{\mathbf d})).
\]
The analogous bijection holds for finite groups, residually finite groups,
finite-dimensional linear groups over every field, finite-dimensional unitary
groups, and profinite groups.
\leanverified{Sofic/AlternatingLampExactRadical}{GroupApproximation.AlternatingLampFamily.alternatingFamilyPackage}
\leanverified{Sofic/AlternatingLampLiteralPackage}{GroupApproximation.AlternatingLampLiteral.ker_rightHom_eq_normalClosure_witnessLamp}
\end{mainthm}

Thus adding the single relation $r=1$, which every MF representation already
satisfies, changes a non-MF group into an MF group.  The relation is not
intrinsically pathological: its normal closure is locally finite, amenable,
residually finite, and MF.  What fails is extension of its finite models
through the compressed Kazhdan action.
The classical Peter--Weyl point-separation theorem upgrades the last target
list from profinite groups to all compact Hausdorff groups.

\begin{mainthm}[spectral compression]\label{thm:spectral}
Let $L\le H$ have property~\textup{(T)}, let $sLs^{-1}\le L$, and let
$\pi\colon H\to\U(\mathcal Q_{\mathbf d})$ be a homomorphism.  Suppose that
$v\in\U(\mathcal Q_{\mathbf d})$ commutes with $\pi(sLs^{-1})$ and that its
$L$-conjugates commute pairwise.  Then, for $\gamma\in L$,
\[
 \opnorm{\pi(\gamma)v\pi(\gamma)^*-v}
 \le \sup_{C\in\pi_0(\sigma(v))}\diameter(C).
\]
In particular, if every connected component of $\sigma(v)$ is a singleton,
then $v$ commutes with $\pi(L)$.
\leanverified{Analysis/SpectralComponentDiameter}{GroupApproximation.SpectralComponentDiameter.norm_sub_le_of_same_clopen_spectral_pieces}
\leanverified{Sofic/SpectralCompression}{GroupApproximation.SpectralCompression.corona_clopen_projection_collapse}
\end{mainthm}

Finite spectrum, Cantor spectrum, and every zero-dimensional compact spectrum
therefore collapse.  The proof uses clopen spectral projections, not Fourier
decomposition.  Fourier idempotents remain a useful special case but are no
longer part of the architecture.

Let
\[
 \Gamma_*=\Z^3\rtimes\SL_3(\Z),\qquad \alpha(v,A)=(2v,A),
\]
let $G_*$ be the ascending HNN extension of $\Gamma_*$ along $\alpha$, and put
$X=G_*/\Gamma_*$.  The height map $h\colon G_*\twoheadrightarrow\Z$ induces
$X\to\Z$.  For a nontrivial finite group $K$, set
$W_K=K^{(X)}\rtimes G_*$.  Write $L_K=K^{(X)}$ for its lamp subgroup and
$q_K\colon W_K\twoheadrightarrow G_*$ for the wreath projection.

\begin{mainthm}[the universal visible quotient]\label{thm:visible}
For every nontrivial finite perfect group $K$, every matrix-corona
representation of $W_K$ factors uniquely through $q_K$.  Moreover
\[
 L_K=\ker q_K
 =\Rad_{\mathrm{MF}}(W_K)=\Rad_{\mathrm{fd}}(W_K)
 =\Rad_{\mathrm{lin}}(W_K)=\Res_{\mathrm{fin}}(W_K).
\]
The quotient $G_*$ is residually finite, while $W_K$ is finitely generated,
sofic, and non-MF.  Precomposition with $q_K$ is a bijection on homomorphisms
into every norm matrix corona, finite or residually finite group, and
finite-dimensional linear group over every field, as well as every profinite
group.
\leanverified{Sofic/FinitePerfectLampExactRadical}{GroupApproximation.FinitePerfectLamp.finitePerfectPackage}
\leanverified{Sofic/UniversalVisibleQuotient}{GroupApproximation.comp_mk'_bijective}
\end{mainthm}

By Peter--Weyl separation, every homomorphism from $W_K$ to a compact
Hausdorff group also kills $L_K$.  Thus precomposition is bijective for compact
targets and $\Rad_{\mathrm{Bohr}}(W_K)=L_K$.  The formal development isolates
precisely this classical point-separation input.

Theorem~\ref{thm:headline} is the specialization $K=A_5$.  Perfectness, not
$A_5$ itself, makes the entire lamp subgroup invisible.  Simplicity makes any
nonidentity one-site lamp normally generate that subgroup, and the choice
$(12)(34)$ makes the generator an involution.  Thus $A_5$ is not essential to
the collapse or to the existence of non-MF groups.  It is the smallest
standard choice that packages the sharper one-relation statement.

\section{Exact finite-dimensional compression}\label{sec:fd}

For $L\le H$, put
\[
 \operatorname{Comp}(L)=\{s\in H:sLs^{-1}\le L\},\qquad
 G_{\mathrm{comp}}(L)=\langle\operatorname{Comp}(L)\rangle,
\]
and define
\[
 D(H,L)=\left\langle\!\left\langle
 [gzg^{-1},\ell]:g\in G_{\mathrm{comp}}(L),\ z\in C_H(L),\ \ell\in L
 \right\rangle\!\right\rangle_H.
\]

\begin{theorem}\label{thm:fd-defect}
For every field $F$, finite-dimensional $F$-vector space $V$, and homomorphism
$\rho\colon H\to\GL_F(V)$, one has $D(H,L)\le\ker\rho$.
\leanverified{Criterion/CompressionCentralizerDefect}{GroupApproximation.compressionCentralizerDefect_le_ker}
\end{theorem}

\begin{proof}
Let $C$ be the commutant of $\rho(L)$ in $\operatorname{End}_F(V)$.  If
$sLs^{-1}\le L$, conjugation by $\rho(s)$ sends $C$ injectively into $C$.
Finite dimension makes it surjective.  Thus every compressor, and hence
$G_{\mathrm{comp}}(L)$, normalizes $C$.  It follows that each displayed
commutator, and therefore its normal closure, is killed by $\rho$.
\end{proof}

No property~\textup{(T)}, trace, or unitarity occurs here.  The only use of
finite dimension is injective implies surjective.  A norm corona is not finite
dimensional and its commutants can grow; the next section gives the rigidity
that survives there.

\section{Spectral rigidity in norm coronas}\label{sec:spectral}

\begin{theorem}[projection collapse]\label{thm:projection-collapse}
Let $L\le H$ have property~\textup{(T)}, let $sLs^{-1}\le L$, and let
$\pi\colon H\to\U(\mathcal Q_{\mathbf d})$.  If a projection $p$ commutes with
$\pi(sLs^{-1})$ and the projections $\pi(\gamma)p\pi(\gamma)^*$,
$\gamma\in L$, commute pairwise, then $p$ commutes with $\pi(L)$.
\leanverified{Sofic/ProjectionCompressionCollapse}{GroupApproximation.ProjectionCompressionCollapse.corona_projection_collapse}
\end{theorem}

\begin{proof}[Proof idea]
Lift the data to bounded matrix sequences.  On the finite von Neumann algebra
generated by the asymptotic commuting orbit, the join of the displacement
supports carries a normalized rank trace.  The displacements form a
Hilbert-space cocycle for $L$.  Compressed centrality makes its restriction to
$sLs^{-1}$ vanish, while one-sided compression makes the invariant projection
subordinate to a unitarily equivalent projection.  Finiteness forces equality.
Property~\textup{(T)} then makes the cocycle a coboundary, and support
normalization forces that coboundary to vanish.
\end{proof}

Let $v$ be normal and $U\subseteq\sigma(v)$ clopen.  Continuous functional
calculus gives $p_U=1_U(v)$.  It is functorial under unitary conjugation, so
Theorem~\ref{thm:projection-collapse} gives
\[
 \pi(\gamma)p_U\pi(\gamma)^*=p_U\qquad(\gamma\in L).
\tag{3.1}
\]
Put $w=\pi(\gamma)v\pi(\gamma)^*$.  Since $v,w$ commute,
$A=C^*(1,v,w)$ is commutative.  For a character $\chi$ of $A$, put
$z=\chi(v)$ and $z'=\chi(w)$.  Evaluating (3.1) says that $z$ and $z'$ lie in
exactly the same clopen subsets of $\sigma(v)$.  In a compact Hausdorff space,
quasi-components equal connected components.  Gelfand duality therefore gives
\[
 \opnorm{v-w}=\sup_\chi|\chi(v)-\chi(w)|
 \le\sup_{C\in\pi_0(\sigma(v))}\diameter(C),
\]
proving Theorem~\ref{thm:spectral}.

\section{The affine compression geometry}\label{sec:geometry}

Use the affine $4\times4$ matrix representation of
$\Gamma_*=\Z^3\rtimes\SL_3(\Z)$ and put
$D=\operatorname{diag}(2,2,2,1)$.  Conjugation by $D$ is $\alpha$, with
$[\Gamma_*:\alpha(\Gamma_*)]=8$.  The group $\Gamma_*$ has
property~\textup{(T)} by relative property~\textup{(T)} for the translation
subgroup and property~\textup{(T)} of $\SL_3(\Z)$ \cite{BHV,Burger}.  It is
residually finite by reduction modulo primes.

Form
\[
 G_*=\langle\Gamma_*,t\mid t\gamma t^{-1}=\alpha(\gamma)\rangle.
\]
Sending $t$ to $D$ embeds $G_*$ in $\GL_4(\Z[1/2])$: the direct-limit kernel
is the union of $D^{-n}\Gamma_*D^n$, and determinant separates height.
Reduction modulo odd primes makes $G_*$ residually finite, hence MF.

Put $X=G_*/\Gamma_*$.  The proper inclusion
$t\Gamma_*t^{-1}<\Gamma_*$ supplies compression.  The height homomorphism
$h(t)=1$, $h(\Gamma_*)=0$ induces $X\to\Z$.  For finite $K$,
$W_K=K^{(X)}\rtimes G_*$ is finitely generated because the action is
transitive.  It is sofic: finite-index compression gives finite telescope
windows for the coset action, and generalized wreath approximations over
those windows give sofic models.  This is the only use of the index $8$.

\section{The exact visible quotient}\label{sec:quotient}

\begin{proposition}\label{prop:factor}
If $K$ is finite and perfect, every norm-corona representation of $W_K$
kills $L_K$ and therefore factors uniquely through $q_K$.
\leanverified{Sofic/FinitePerfectLampExactRadical}{GroupApproximation.FinitePerfectLamp.actualCoronaMFResidual_eq_lampSub}
\leanverified{Sofic/UniversalVisibleQuotient}{GroupApproximation.precomp_bijective}
\end{proposition}

\begin{proof}
Fix $k\in K$.  A one-site lamp with value $k$ is centralized by the compressed
copy $t\Gamma_*t^{-1}$, while its $\Gamma_*$-conjugates commute whenever they
occupy distinct sites.  Since $K$ is finite, the lamp has finite spectrum.
Theorem~\ref{thm:spectral} therefore makes the images of the relevant
conjugate lamps coincide.  Copies supported at distinct sites commuted before
this identification, so the resulting image of the one-site copy of $K$ is
abelian.  It kills $[K,K]$, which equals $K$ by perfectness.  Transitivity of
the $G_*$-action then kills every one-site copy and hence $L_K$.  Uniqueness of
the factorization follows from surjectivity of $q_K$.
\end{proof}

Since $q_K$ lands in the residually finite group $G_*$, no element outside
$L_K=\ker q_K$ can lie in the finite residual.  On the other hand every finite
target is a finite-dimensional unitary target, and every finite-dimensional
unitary target embeds in a norm matrix corona by constant sequences.  Together
with Proposition~\ref{prop:factor}, these containments identify the MF,
finite-dimensional unitary, and finite residuals with $L_K$.

If $\rho\colon W_K\to\GL_d(F)$, its image is finitely generated and linear,
hence residually finite by Mal'cev's theorem, so it also kills
$L_K$.  Conversely finite groups have faithful complex regular
representations.  This identifies the all-fields linear radical as well.
The abstract factorization lemma for a surjection now gives all the stated
Hom-set bijections.  Finally, Peter--Weyl separation says that a nonidentity
element of a compact Hausdorff group is detected by a finite-dimensional
unitary representation; it therefore identifies the Bohr radical with the
finite-dimensional unitary radical.  This proves Theorem~\ref{thm:visible}.

\section{One invisible relation}\label{sec:a5}

Take $K=A_5$.  Since $A_5$ is perfect,
$\ker q_{A_5}=A_5^{(X)}$ is the exact common invisible subgroup.  Let $r$ be
the one-site lamp with value
$(12)(34)$.  It is an involution.  Simplicity of $A_5$ makes its conjugates
inside one site normally generate that copy; transitivity spreads the normal
closure to every site.  Hence
\[
 \langle\!\langle r\rangle\!\rangle_{W_{A_5}}=A_5^{(X)},\qquad
 W_{A_5}/\langle\!\langle r\rangle\!\rangle\cong G_*.
\]
Every corona representation already imposes $r=1$.  Nevertheless the kernel
is nontrivial, so $W_{A_5}$ is not MF, while $G_*$ is residually finite and
MF.  This proves Theorem~\ref{thm:headline}.

The invisible subgroup $A_5^{(X)}$ is locally finite and residually finite,
hence MF on its own.  Its ambient invisibility is an extension phenomenon
created by the compressed action, not intrinsic non-approximability.

\section*{Formal verification}
The arguments specific to this paper are formalized in Lean~4: the
finite-dimensional defect theorem over every field; projection collapse;
clopen spectral functional calculus; the compact-Hausdorff component lemma and
Gelfand norm estimate; the affine compression pair; the exact MF, finite,
unitary, and all-fields linear residuals for every finite perfect lamp;
universal factorization; and the $A_5$ one-relation specialization.  The
formal Bohr statement is reduced to the point-separation conclusion of the
classical Peter--Weyl theorem, which is isolated as an explicit hypothesis.
The source contains no admitted propositions.
