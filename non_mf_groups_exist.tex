\begin{document}
\title[One-sided compression]{One-sided compression and finite-dimensional approximation}
\author{\paperauthor}
\subjclass[2020]{Primary 46L05; Secondary 20F05, 22D25, 20E26}
\keywords{MF groups, approximate unitary representations, property~(T), spectral rigidity, permutational wreath products}

\begin{abstract}
A countable group is MF if it embeds in the unitary group of a norm matrix
corona.  We prove that non-MF groups exist.  In fact there is a split epimorphism
$q\colon W\twoheadrightarrow G$, with nontrivial kernel normally generated by
one involution, such that $W$ is sofic and non-MF and $G$ is residually finite,
while every homomorphism from $W$ to a norm matrix corona factors
through $q$.

The main analytic theorem concerns a property-\textup{(T)} subgroup $L\le H$
and $s\in H$ with $sLs^{-1}\le L$.  If a normal element of a norm matrix
corona commutes with the compressed subgroup and has commuting $L$-conjugates,
then every clopen spectral projection is $L$-invariant; quantitatively, an
$L$-conjugate differs from the original element by at most the largest diameter
of a connected component of its spectrum.  For every finite group $K$, this
computes the exact radical of $W_K=K^{(X)}\rtimes G$ through the quotient
\[
 W_K\longrightarrow K_{\ab}^{(\mathbb Z)}\rtimes G.
\]
The kernel is simultaneously the MF, finite-dimensional, linear, Bohr, and
finite residual.  Finally we prove the sharp complementary result for every
countable abelian lamp group $A$:
$A^{(X)}\rtimes G$ is MF if and only if $A$ is torsion-free.
\end{abstract}

\section{Introduction}

A countable group is \emph{MF} if it embeds in the unitary group of a norm
matrix corona
\[
 \mathcal Q_{\mathbf d}=\prod_nM_{d_n}(\C)\big/\bigoplus_nM_{d_n}(\C).
\]
Equivalently, it has asymptotically multiplicative finite-dimensional unitary
models in operator norm which asymptotically separate its nonidentity
elements.  This is the group property of Carri\'on--Dadarlat--Eckhardt
\cite{CDE}.  Residually finite and LEF groups are MF, and amenable groups are
MF \cite{CDE,SchafhauserAF}.  Whether every countable group is MF had remained
open \cite{Schafhauser,Shulman}.

For a class $\mathcal C$ of target groups, write
\[
 R_{\mathcal C}(H)=
 \bigcap_{\rho\colon H\to C,\ C\in\mathcal C}\ker\rho.
\tag{1.1}
\]
All homomorphisms in (1.1) are abstract group homomorphisms.  We write
$\Rad_{\mathrm{MF}}(H)$ when the targets are the groups
$\U(\mathcal Q_{\mathbf d})$, over all sequences of positive matrix sizes;
$\Rad_{\mathrm{fd}}(H)$ when they are the groups $\U(d)$, $d\ge1$; and
$\Rad_{\mathrm{lin}}(H)$ when they are $\GL_d(F)$, over every field $F$ and
every $d\ge1$.  For compact Hausdorff targets, again with no continuity
condition on the map from the discrete group $H$, we write
$\Rad_{\mathrm{Bohr}}(H)$; this is the von Neumann radical.  Finally,
$\Res_{\mathrm{fin}}(H)$ is the intersection of the kernels of all
homomorphisms from $H$ to finite groups.

\begin{mainthm}[a single relation changes the MF property]\label{thm:headline}
There are explicit finitely generated groups $W$ and $G$, a split epimorphism
$q\colon W\twoheadrightarrow G$, and an involution $r\in W\setminus\{1\}$
such that $\ker q=\langle\!\langle r\rangle\!\rangle_W$.  The group $W$ is
sofic and non-MF, whereas $G$ is residually finite and MF.  For every norm
matrix corona, precomposition gives a bijection
\[
 \operatorname{Hom}(G,\U(\mathcal Q_{\mathbf d}))
 \xrightarrow{\ \cong\ }
 \operatorname{Hom}(W,\U(\mathcal Q_{\mathbf d})).
\]
The analogous bijection holds for residually finite targets,
finite-dimensional linear groups over every field, and compact Hausdorff
targets; this includes finite, finite-dimensional unitary, and profinite
groups.
\leanverified{Manuscript/SpectralPaper/MainTheorems}{GroupApproximation.SpectralPaper.oneInvisibleRelationPackage}
\leanverified{Manuscript/SpectralPaper/MainTheorems}{GroupApproximation.SpectralPaper.a5UniversalInvisibility}
\leanverified{Sofic/AlternatingLampLiteralPackage}{GroupApproximation.AlternatingLampLiteral.ker_rightHom_eq_normalClosure_witnessLamp}
\end{mainthm}

Every homomorphism from $W$ to a norm matrix corona maps $r$ to $1$, whereas
$W/\langle\!\langle r\rangle\!\rangle$ is MF and $W$ is not.  The normal
closure of $r$ is locally finite, amenable, residually finite, and MF.  Thus
the kernel is itself well approximated by matrices, although its faithful
matrix models do not extend to representations of $W$.

\begin{mainthm}[spectral compression]\label{thm:spectral}
Let $L\le H$ have property~\textup{(T)}, let $sLs^{-1}\le L$, and let
$\pi\colon H\to\U(\mathcal Q_{\mathbf d})$ be a homomorphism.  Suppose that
$v\in\mathcal Q_{\mathbf d}$ is normal, commutes with $\pi(sLs^{-1})$, and
has pairwise commuting $L$-conjugates.  Then every clopen spectral projection
of $v$ commutes with $\pi(L)$ and, for $\gamma\in L$,
\[
 \opnorm{\pi(\gamma)v\pi(\gamma)^*-v}
 \le \sup\{\diameter(C):C\text{ is a connected component of }\sigma(v)\}.
\]
In particular, if every connected component of $\sigma(v)$ is a singleton,
then $v$ commutes with $\pi(L)$.
\leanverified{Manuscript/SpectralPaper/MainTheorems}{GroupApproximation.SpectralPaper.spectralCompressionTheorem}
\leanverified{Manuscript/SpectralPaper/MainTheorems}{GroupApproximation.SpectralPaper.zeroDimensionalSpectralCollapseTheorem}
\end{mainthm}

Hence $v$ commutes with $\pi(L)$ whenever $\sigma(v)$ is zero-dimensional.
This includes finite and Cantor spectra.

Let
\[
 \Gamma_*=\Z^3\rtimes\SL_3(\Z),\qquad \alpha(v,A)=(2v,A),
\]
let $G_*$ be the ascending HNN extension of $\Gamma_*$ along $\alpha$, and put
$X=G_*/\Gamma_*$.  The height map $h\colon G_*\twoheadrightarrow\Z$ induces
$\eta\colon X\to\Z$.  For a group $K$, set $W_K=K^{(X)}\rtimes G_*$.  The
height map fuses all lamps at a common height after first abelianizing their
values.  It gives the canonical epimorphism
\[
 \Phi_K\colon W_K\longrightarrow
 P_K:=K_{\ab}^{(\Z)}\rtimes G_*.
\tag{1.2}
\]
Here $g\in G_*$ translates the $\Z$-coordinate by $h(g)$.

\begin{mainthm}[exact radicals for finite lamps]\label{thm:visible}
For every finite group $K$, every matrix-corona representation of $W_K$
factors uniquely through $\Phi_K$.  Moreover
\[
 \ker\Phi_K
 =\Rad_{\mathrm{MF}}(W_K)=\Rad_{\mathrm{fd}}(W_K)
 =\Rad_{\mathrm{lin}}(W_K)=\Rad_{\mathrm{Bohr}}(W_K)
 =\Res_{\mathrm{fin}}(W_K).
\]
Equivalently, precomposition with $\Phi_K$ is a bijection on homomorphisms into
every norm matrix corona, residually finite group, finite-dimensional linear
group over every field, or compact Hausdorff group.  If $K\ne1$, then $W_K$
is finitely generated, sofic, and non-MF, whereas $P_K$ is residually finite
and MF.
\end{mainthm}

Thus two configurations have the same image precisely when they have the same
abelianized total lamp value on every height fibre.  If $K$ is perfect, then
$K_{\ab}=1$, so $P_K=G_*$ and every representation in
Theorem~\ref{thm:visible} is trivial on the lamp subgroup.  For $K=A_5$, any
nonidentity one-site lamp normally generates this kernel; choosing $(12)(34)$
gives the involution in Theorem~\ref{thm:headline}.

The rigidity theorem has a sharp positive side.

\begin{mainthm}[the abelian MF boundary]\label{thm:abelian-boundary}
For every countable abelian group $A$,
\[
 A^{(X)}\rtimes G_*\text{ is MF}
 \quad\Longleftrightarrow\quad A\text{ is torsion-free}.
\]
\end{mainthm}

If $A$ has torsion, the image of a finite-order lamp has finite spectrum, and
Theorem~\ref{thm:spectral} puts a nontrivial two-site word in every
matrix-corona kernel.  If $A$ is torsion-free, a faithful character deforms
continuously to the trivial character; evaluating that deformation against
slowly varying functions on finite quotients gives a faithful corona
representation.  In particular, a relation killed by every exact
finite-dimensional representation need not be killed by every
matrix-corona representation.

\subsection*{Relation to prior work}

The MF property for groups used here is due to
Carri\'on--Dadarlat--Eckhardt \cite{CDE}; Korchagin and, more recently,
Schafhauser and Shulman developed further permanence and approximation results
\cite{Korchagin,Schafhauser,Shulman}.  Amenable groups are MF by
Schafhauser's embedding theorem \cite{SchafhauserAF}.  The use of
Hilbert--Schmidt spaces inside operator-norm approximation arguments is related
to work of Bekka, Bekka--Valette, and Dadarlat on conjugation representations
and matricial stability \cite{BekkaAmenable,BekkaValette,Dadarlat}.

One-sided conjugation of Kazhdan subgroups occurs in the first constructions
of nonsofic groups and in the subsequent wreath-product analysis of
Kun--Thom \cite{OAI,KT26}.  The relevant Hamming-metric rigidity builds on
Kun and Kun--Thom \cite{Kun16,KT19}; Alekseev--Thom prove a related theorem on
centralizers of sofic embeddings \cite{AlekseevThom}.  Those results concern
permutation or tracial ultraproducts.  Here the initial homomorphism takes
values in a norm matrix corona, and the rank-normalized Hilbert space used in
Section~\ref{sec:spectral} depends on the spectral projection being moved.
Slofstra's work on hyperlinear profiles similarly distinguishes kernels of
exact finite-dimensional representations from kernels detected by approximate
representations \cite{SlofstraCorrelations,SlofstraProfile}.

The factorization statements are also related in spirit to Grothendieck's
comparison between discrete groups and their profinite completions
\cite{Grothendieck,BridsonGrunewald}.  The epimorphism in
Theorem~\ref{thm:headline} is split, and its kernel is computed simultaneously
for norm-matrix, linear, finite, and compact targets.  For soficity we use the
theory of sofic actions on sets and the generalized wreath-product theorem of
Gao--Kunnawalkam Elayavalli--Patchell \cite{GKEP}.

Section~\ref{sec:fd} gives the exact finite-dimensional analogue of the main
rigidity theorem.  Section~\ref{sec:spectral} proves spectral rigidity in norm
coronas.  Sections~\ref{sec:geometry} and~\ref{sec:quotient} construct the
groups and compute their radicals, and Section~\ref{sec:boundary} proves the
abelian classification.

\section{Exact finite-dimensional compression}\label{sec:fd}

For $L\le H$, put
\[
 \operatorname{Comp}(L)=\{s\in H:sLs^{-1}\le L\},\qquad
 G_{\mathrm{comp}}(L)=\langle\operatorname{Comp}(L)\rangle,
\]
and define
\[
 D(H,L)=\left\langle\!\left\langle
 [gzg^{-1},\ell]:g\in G_{\mathrm{comp}}(L),\ z\in C_H(L),\ \ell\in L
 \right\rangle\!\right\rangle_H.
\]

\begin{theorem}\label{thm:fd-defect}
For every field $F$, finite-dimensional $F$-vector space $V$, and homomorphism
$\rho\colon H\to\GL_F(V)$, one has $D(H,L)\le\ker\rho$.
\leanverified{Criterion/CompressionCentralizerDefect}{GroupApproximation.compressionCentralizerDefect_le_ker}
\end{theorem}

\begin{proof}
Let $C$ be the commutant of $\rho(L)$ in $\operatorname{End}_F(V)$.  If
$sLs^{-1}\le L$, then
\[
 \rho(s)^{-1}C\rho(s)\subseteq C,
 \qquad\text{equivalently}\qquad
 C\subseteq\rho(s)C\rho(s)^{-1}.
\tag{2.1}
\]
The two commutants in the second inclusion have the same finite dimension, so
equality holds.  Hence $s$ normalizes $C$; the same is true for products and
inverses of compressors.  If $z\in C_H(L)$, then $\rho(z)\in C$, and therefore
$\rho(gzg^{-1})\in C$ for $g\in G_{\mathrm{comp}}(L)$.  It commutes with
$\rho(L)$, so every displayed commutator, and then its normal closure, lies in
$\ker\rho$.
\end{proof}

For representations into norm matrix coronas, the finite-dimensional
surjectivity argument is unavailable.  Section~\ref{sec:spectral} replaces it
with property~\textup{(T)}.

\section{Spectral rigidity in norm coronas}\label{sec:spectral}

We first record the fixed-space comparison used in the proof.  Let
$u_n\colon H\to\U(d_n)$ be an operator-norm almost representation and let
$(k_n)$ be positive integers.  The rank-normalized Hilbert ultraproduct is the
completion, modulo zero seminorm, of the matrix sequences for which the
right-hand side below is bounded:
\[
 \opnorm{(x_n)}_{2,k,\omega}^2
   =\lim_{n\to\omega}\frac{\Tr(x_n^*x_n)}{k_n}.
\tag{3.1}
\]
Conjugation by $(u_n(g))_n$ defines a unitary representation of $H$ on this
space: the operator-norm multiplicative errors vanish in the Hilbert
ultraproduct.  We denote this representation by $\Ad u$.

\begin{lemma}[fixed-space transport]\label{lem:fixed-space}
If $L\le H$ has property~\textup{(T)} and $sLs^{-1}\le L$, then every vector
in a rank-normalized Hilbert ultraproduct which is fixed by $sLs^{-1}$ is
fixed by $L$.
\end{lemma}

\begin{proof}
Let $\omega$ be the ultrafilter used to form the Hilbert ultraproduct and put
\[
 \mathcal B_\omega=\prod_\omega B(M_{d_n}(\C)),
\]
where each $B(M_{d_n}(\C))$ acts on $M_{d_n}(\C)$ by left and right
multiplication.  This is an operator-norm ultraproduct of matrix algebras and
is Dedekind finite: if $x^*x=1$, polar correction of coordinate lifts gives a
unitary representing $x$, and hence $xx^*=1$.  It acts on every
rank-normalized Hilbert ultraproduct in (3.1).

Choose a finite symmetric Kazhdan set $S\subseteq L$ and let
\[
 A=|S|^{-1}\sum_{a\in S}\Ad u(a)\in\mathcal B_\omega.
\]
The Kazhdan inequality isolates $1$ in $\sigma(A)$.  Thus continuous
functional calculus in $\mathcal B_\omega$ gives the projection
$P=1_{\{1\}}(A)$.  Its range on the Hilbert ultraproduct is exactly the
$L$-fixed subspace: one inclusion follows because $A\xi=\xi$ for an
$L$-fixed vector, and the converse is the equality case in the averaging
estimate together with the Kazhdan gap.

Let $V=\Ad u(s)$ and $Q=VPV^*$.  Conjugating the preceding range
identification shows that the range of $Q$ is the subspace fixed by
$sLs^{-1}$.  Since $sLs^{-1}\le L$, this subspace contains the $L$-fixed
subspace, so $P\le Q$.  The projections $P$ and $Q$ are unitarily conjugate
in the Dedekind-finite algebra $\mathcal B_\omega$.  A projection in such an
algebra cannot be properly contained in an equivalent projection; hence
$P=Q$.  Their ranges are equal, as required.
\end{proof}

\begin{theorem}[projection collapse]\label{thm:projection-collapse}
Let $L\le H$ have property~\textup{(T)}, let $sLs^{-1}\le L$, and let
$\pi\colon H\to\U(\mathcal Q_{\mathbf d})$.  If a projection $p$ commutes with
$\pi(sLs^{-1})$ and the projections $\pi(\gamma)p\pi(\gamma)^*$,
$\gamma\in L$, commute pairwise, then $p$ commutes with $\pi(L)$.
\leanverified{Sofic/ProjectionCompressionCollapse}{GroupApproximation.ProjectionCompressionCollapse.corona_projection_collapse}
\end{theorem}

\begin{proof}
Write $p_\gamma=\pi(\gamma)p\pi(\gamma)^*$.  Suppose that
$p_{\gamma_0}\ne p$ for some $\gamma_0\in L$.  Choose a finite symmetric
Kazhdan generating set $S=\{a_1,\ldots,a_m\}$ for $L$.  At least one $p_a$,
$a\in S$, differs from $p$.  Indeed, equality for every generator would give
equality for every element of $L$.

The projections in the orbit commute.  Hence
$q_a=(p_a-p)^2$ is the support projection of $p_a-p$, the projections $q_a$
commute, and their join is
\[
 q=1-\prod_{a\in S}(1-q_a)\ne0.
\tag{3.2}
\]
To lift $q$, first choose self-adjoint contractions $A_n$ representing it.
Then $\opnorm{A_n^2-A_n}\to0$, so cutting the spectrum of $A_n$ at $1/2$
gives projections $Q_n$ which still represent $q$.  After passing to a free
ultrafilter $\omega$ and discarding an $\omega$-null set of coordinates, we
may assume that $k_n=\operatorname{rank}Q_n>0$.  Let $\mathcal I_q$ be the
ideal in the corona represented by sequences whose ranks are $O(k_n)$.  On
this ideal the
formula
\[
 \langle\Lambda(x),\Lambda(y)\rangle
 =\lim_{n\to\omega}\frac{\Tr(y_n^*x_n)}{k_n}
\tag{3.3}
\]
is independent of the rank-controlled representatives: an operator-norm-null
change has rank $O(k_n)$ and therefore has zero seminorm in (3.3).  After
quotienting the null space and completing, it gives a Hilbert space
$\mathcal K_q$.

Choose unitary lifts $u_n(g)$ of $\pi(g)$; polar correction of arbitrary
bounded lifts provides such lifts.  They form an operator-norm almost
representation of $H$.  The displacement identity
\[
 d_{\gamma\delta}=d_\gamma+\pi(\gamma)d_\delta\pi(\gamma)^*,
 \qquad d_\gamma=p_\gamma-p,
\tag{3.4}
\]
shows first that $d_\gamma\in\mathcal I_q$ for every $\gamma\in L$: it holds
for the generators because $q$ contains their support projections, and then
for all words by (3.4).  It also shows that
\[
 \beta(\gamma)=\Lambda(d_\gamma)
\]
is a $1$-cocycle for the conjugation representation of $L$ on
$\mathcal K_q$.

This cocycle is nonzero.  To see this, put
$r_i=\prod_{j<i}(1-q_{a_j})$.  The telescoping identity
\[
 q=\sum_{i=1}^m r_iq_{a_i}
   =\sum_{i=1}^m (d_{a_i}r_i)^*d_{a_i}
\tag{3.5}
\]
uses only the commutation of the $q_{a_i}$.  The null space of (3.3) is stable
under left and right multiplication by uniformly bounded sequences.  Hence,
if every $\beta(a_i)$ were zero, every summand in (3.5) would have zero image
under $\Lambda$, and $\Lambda(q)=0$.  But
\[
 \opnorm{\Lambda(q)}^2
 =\lim_{n\to\omega}\frac{\Tr(Q_n)}{k_n}=1,
\]
a contradiction.

Property~\textup{(T)} and the Delorme--Guichardet theorem
\cite[Theorem~2.12.4]{BHV} now give $y\in\mathcal K_q$ such that
\[
 \beta(\gamma)=y-\Ad u(\gamma)y\qquad(\gamma\in L).
\tag{3.6}
\]
The hypothesis that $p$ commutes with $\pi(sLs^{-1})$ gives
$d_{s\gamma s^{-1}}=0$, hence (3.6) says that $y$ is fixed by
$sLs^{-1}$.  Lemma~\ref{lem:fixed-space} says that $y$ is fixed by $L$.
Equation (3.6) would then make $\beta$ identically zero, contradicting (3.5).
Thus $p_\gamma=p$ for every $\gamma\in L$, which is equivalent to the asserted
commutation.
\end{proof}

Let $v$ be normal and $U\subseteq\sigma(v)$ clopen.  Continuous functional
calculus gives $p_U=1_U(v)$.  Normality supplies the starred commutation
hypotheses by Fuglede's theorem, and functional calculus is functorial under
unitary conjugation.  Theorem~\ref{thm:projection-collapse} therefore gives
\[
 \pi(\gamma)p_U\pi(\gamma)^*=p_U\qquad(\gamma\in L).
\tag{3.7}
\]
Put $w=\pi(\gamma)v\pi(\gamma)^*$.  Since $v,w$ commute,
$A=C^*(1,v,w)$ is commutative.  For a character $\chi$ of $A$, put
$z=\chi(v)$ and $z'=\chi(w)$.  Applying $\chi$ to (3.7) says that $z$ and $z'$ lie in
exactly the same clopen subsets of $\sigma(v)$.  In a compact Hausdorff space,
quasi-components equal connected components.  Gelfand duality therefore gives
\[
 \opnorm{v-w}=\sup_\chi|\chi(v)-\chi(w)|
 \le\sup\{\diameter(C):C\text{ is a connected component of }\sigma(v)\},
\]
proving Theorem~\ref{thm:spectral}.

Equivalently, compression fixes the Boolean algebra of clopen spectral tests.
Under Gelfand duality it acts trivially on the maximal totally disconnected
quotient detected by those tests.  We use this interpretation only as a
description: the precise theorem is the fixed-projection statement (3.7),
which does not require conjugation to preserve a chosen commutative
subalgebra.

\section{The affine ascending HNN extension}\label{sec:geometry}

Realize $\Gamma_*=\Z^3\rtimes\SL_3(\Z)$ as the affine subgroup of
$\GL_4(\Z)$ and put
$D=\operatorname{diag}(2,2,2,1)$.  Conjugation by $D$ is $\alpha$, with
$[\Gamma_*:\alpha(\Gamma_*)]=8$.  The group $\Gamma_*$ has
property~\textup{(T)} by relative property~\textup{(T)} for the translation
subgroup and property~\textup{(T)} of $\SL_3(\Z)$ \cite{BHV,Burger}.  It is
residually finite by reduction modulo primes.

Form
\[
 G_*=\langle\Gamma_*,t\mid t\gamma t^{-1}=\alpha(\gamma)\rangle.
\]
The subgroup
\[
 T=\bigcup_{n\ge0}D^{-n}\Gamma_*D^n\le\GL_4(\Z[1/2])
\]
is the direct limit of $(\Gamma_*,\alpha)$.  Conjugation by $D$ induces the
shift automorphism of $T$, so $T\rtimes\langle D\rangle$ satisfies the defining
HNN relation.  Conversely, the normal form theorem for ascending HNN
extensions identifies the kernel of the height map with this direct limit.
Thus $t\mapsto D$ gives an embedding
$G_*\hookrightarrow\GL_4(\Z[1/2])$; the determinant $8^{h(g)}$ records the
height.  Mal'cev's theorem now implies that $G_*$ is residually finite
\cite{Malcev}, and hence MF.

Put $X=G_*/\Gamma_*$.  The relation
$t\Gamma_*t^{-1}<\Gamma_*$ is the one-sided inclusion used in
Theorem~\ref{thm:spectral}.  The height homomorphism
$h(t)=1$, $h(\Gamma_*)=0$ induces $X\to\Z$.  For finite $K$,
$W_K=K^{(X)}\rtimes G_*$ is finitely generated because the action is
transitive.

\begin{proposition}\label{prop:sofic}
For every finite group $K$, the group $W_K$ is sofic.
\end{proposition}

\begin{proof}
Put $\Gamma_n=t^{-n}\Gamma_*t^n$.  The telescope
$T=\bigcup_{n\ge0}\Gamma_n$ is a directed union of copies of the residually
finite group $\Gamma_*$, hence is sofic.  Since
$G_*=T\rtimes\Z$, permanence under extensions with amenable quotient
\cite{ElekSzabo} makes $G_*$ sofic.

We next verify that the action $G_*\curvearrowright X$ is sofic in the sense
of Gao--Kunnawalkam Elayavalli--Patchell \cite[Definition~2.1]{GKEP}.
Every finite subset of $T$ lies in some $\Gamma_n$.  Each $\Gamma_n$ has
finite orbits on $X$: the stabilizer in $\Gamma_n$ of $g\Gamma_*$ is
$\Gamma_n\cap g\Gamma_*g^{-1}$, which has finite index because any two
telescope levels are commensurable and
$[\Gamma_{j+1}:\Gamma_j]=8$.  More explicitly, write $g=at^m$ with $a\in T$
and choose a level containing $a$; inside a still larger level, both
$\Gamma_n$ and $a(t^m\Gamma_*t^{-m})a^{-1}$ have finite index.  Thus every
finite part of the restricted $T$-action is modeled exactly on a finite union
of finite orbits.

To add the stable-letter direction, take a long interval
$I_N\subset\Z$.  Place a copy of the required finite $T$-orbit model over
each point of $I_N$ and let $t$ shift the copies.  The finitely many conjugates
of the tested $T$-window which occur over $I_N$ lie in one common telescope
level, so the finite-orbit models may be chosen compatibly.  The multiplication
and orbit relations are exact away from the two ends of $I_N$; their error is
at most $2/|I_N|$.  Letting $N\to\infty$ proves that
$G_*\curvearrowright X$ is sofic.  Theorem~3.6 of \cite{GKEP} states that a
permutational wreath product of sofic groups over a sofic action is sofic.
Applying it to the finite group $K$, the sofic group $G_*$, and this action
gives the result.
\end{proof}

\section{The exact quotient for finite lamps}\label{sec:quotient}

Let $K_{\ab}=K/[K,K]$, write $\bar k$ for the class of $k$, and let
$\eta\colon X\to\Z$ be the height of a site.  For $f\in K^{(X)}$ define
\[
 (\eta_{\ab,*}f)(n)=\prod_{\eta(x)=n}\overline{f(x)}.
\tag{5.1}
\]
The product is finite and is independent of order because $K_{\ab}$ is
abelian.  The covariance identity
\[
 \eta(gx)=h(g)+\eta(x)
\tag{5.2}
\]
extends (5.1), together with the identity on $G_*$, to the epimorphism
$\Phi_K$ in (1.2).  Surjectivity follows by lifting a finitely supported
$K_{\ab}$-configuration one coordinate and one site at a time, together with
the unchanged $G_*$-coordinate.

\begin{proposition}[factorization]\label{prop:factor}
If $K$ is finite, every homomorphism
$\pi\colon W_K\to\U(\mathcal Q_{\mathbf d})$ factors uniquely through
$\Phi_K$.
\end{proposition}

\begin{proof}
Let $x_0=\Gamma_*\in X$ and put
$\Gamma_n=t^{-n}\Gamma_*t^n$ for $n\ge0$.  Each $\Gamma_n$ has
property~\textup{(T)}, and $t^n$ compresses it onto $\Gamma_*$:
\[
 t^n\Gamma_nt^{-n}=\Gamma_*\le\Gamma_n.
\]
The compressed group fixes $x_0$.  Hence a one-site lamp $k_{x_0}$ commutes
with it, while the $\Gamma_n$-conjugates of $k_{x_0}$ commute pairwise because
they are supported at individual sites.  Since $K$ is finite,
$\pi(k_{x_0})$ has finite spectrum.  Theorem~\ref{thm:spectral}, applied to
$(\Gamma_n,t^n)$, gives
\[
 \pi(k_{a x_0})=\pi(k_{x_0})\qquad(a\in\Gamma_n).
\]

Now let $x=gx_0$ and $y=wx_0$ have equal height.  Then
$g^{-1}w$ lies in the height kernel
$T=\bigcup_{n\ge0}\Gamma_n$.  Conjugating the preceding equality by
$\pi(g)$ therefore gives
\[
 \pi(k_x)=\pi(k_y)
 \quad\text{whenever}\quad \eta(x)=\eta(y).
\tag{5.3}
\]

The proper inclusion $\Gamma_*<\Gamma_1$ supplies
$a_0\in\Gamma_1\setminus\Gamma_*$.  Thus $a_0x_0\ne x_0$ and these two sites
have equal height.  Translating this pair by $g$ shows that every height fibre
contains distinct sites $x,y$.  For $k,\ell\in K$, the lamps $k_x$ and
$\ell_y$ commute.  Applying (5.3) to $\ell$ gives
$\pi(\ell_y)=\pi(\ell_x)$, and therefore
\[
 [\pi(k_x),\pi(\ell_x)]=1.
\]
Thus the image of each coordinate copy $K_x$ is abelian and its restriction
factors through $K_{\ab}$.  Equation (5.3) says that this factor depends only
on the height.  Multiplying the finitely many coordinates of a configuration
then shows that its image depends precisely on (5.1).  Hence
$\ker\Phi_K\le\ker\pi$.
The factor exists by the quotient universal property and is unique because
$\Phi_K$ is onto.
\end{proof}

The quotient in (1.2) is residually finite.  Indeed, let
$K_{\ab}\wr\Z=K_{\ab}^{(\Z)}\rtimes\Z$ be the regular wreath product.  The map
\[
 P_K\longrightarrow G_*\times(K_{\ab}\wr\Z),\qquad
 (a,g)\longmapsto\bigl(g,(a,h(g))\bigr)
\tag{5.4}
\]
is an injective homomorphism.  The first factor is residually finite.  The
second is residually finite because any prescribed finite lamp configuration
and nonzero shift are separated after reducing the sites and the shift modulo
a sufficiently large integer.  Products and subgroups of residually finite
groups are residually finite.

Proposition~\ref{prop:factor} gives
$\ker\Phi_K\le\Rad_{\mathrm{MF}}(W_K)$.  Constant matrix sequences and finite
regular representations give the chain
\[
 \Rad_{\mathrm{MF}}(W_K)\le\Rad_{\mathrm{fd}}(W_K)
 \le\Res_{\mathrm{fin}}(W_K)\le\ker\Phi_K,
\tag{5.5}
\]
where the last inclusion uses residual finiteness of $P_K$.  Thus all four
terms in (5.5) are equal.  A finite-dimensional linear image of the finitely
generated group $W_K$ is finitely generated and hence residually finite by
Mal'cev's theorem \cite{Malcev}; consequently every such representation kills
$\ker\Phi_K$.  Conversely, finite groups have faithful complex regular
representations, so $\Rad_{\mathrm{lin}}$ is the same kernel.  By Peter--Weyl
point separation \cite{PeterWeyl}, every nonidentity element of a compact
Hausdorff group is detected by a finite-dimensional unitary representation.
This identifies $\Rad_{\mathrm{Bohr}}$ as well.  Since $\Phi_K$ is onto,
containment of its kernel in the kernel of a homomorphism is equivalent to
unique factorization through $\Phi_K$.  This proves the exact part of
Theorem~\ref{thm:visible}.

If $K\ne1$, choose $1\ne k\in K$ and distinct sites $x,y$ of equal height.
Then $k_xk_y^{-1}\ne1$ but lies in $\ker\Phi_K$.  Hence $W_K$ is not MF.
Together with Proposition~\ref{prop:sofic}, this proves that sofic non-MF
groups exist.

For the advertised one-relation form, take $K=A_5$.  Since $A_5$ is perfect,
$P_{A_5}=G_*$ and $\ker\Phi_{A_5}=A_5^{(X)}$.  Let $r$ be the one-site lamp
with value $(12)(34)$.  It is an involution.  The normal closure of
$(12)(34)$ in $A_5$ is all of $A_5$, because $A_5$ is simple.  Hence the
normal closure of $r$ in $W_{A_5}$ contains the site copy at its support.  The
$G_*$-action on $X$ is transitive, so conjugating that site copy gives every
site copy.  Therefore
\[
 \langle\!\langle r\rangle\!\rangle_{W_{A_5}}=A_5^{(X)},\qquad
 W_{A_5}/\langle\!\langle r\rangle\!\rangle\cong G_*.
\tag{5.6}
\]
In particular,
\[
 \Rad_{\mathrm{MF}}(W_{A_5})
 =\ker\Phi_{A_5}
 =\langle\!\langle r\rangle\!\rangle_{W_{A_5}}.
\tag{5.7}
\]
Every homomorphism from $W_{A_5}$ to a norm matrix corona maps $r$ to $1$,
although the quotient by $r=1$ is MF and $W_{A_5}$ is not.  This proves
Theorem~\ref{thm:headline}.  The group $A_5^{(X)}$ is locally finite,
amenable, residually finite, and MF in isolation.  Thus its faithful matrix
models do not extend to representations of $W_{A_5}$.

\section{Sharpness and the MF boundary}\label{sec:boundary}

The exact finite-dimensional conclusion of Theorem~\ref{thm:fd-defect} does
not hold for all matrix-corona representations.  The following construction
gives the positive half of the classification.
Put $L_1=t\Gamma_*t^{-1}<\Gamma_*$ and
$X'=G_*/L_1$.  The map $gL_1\mapsto gt\Gamma_*$ is a $G_*$-equivariant
bijection $X'\cong X$.

\begin{proposition}[the integer-lamp group is MF]\label{prop:slow-wall}
The group $\Z^{(X)}\rtimes G_*$ admits a faithful homomorphism to the unitary
group of a norm matrix corona.
\end{proposition}

\begin{proof}
Enumerate $X'=\{r_1L_1,r_2L_1,\ldots\}$ and the nonidentity elements of
$G_*$, and fix a finite symmetric generating set $S_1$ of $L_1$.  Residual
finiteness supplies finite quotients $q_n\colon G_*\to F_n$ which separate the
first $n$ group elements and also have the following property: for
$i\ne j\le n$, the element $q_n(r_j^{-1}r_i)$ is not represented by a
$q_n(S_1)$-word of length at most $3n$.  There are only finitely many elements to
exclude at stage $n$, so one quotient can meet all these conditions.

Put $K_n=q_n(L_1)$.  Since $L_1$ and $\Gamma_*$ are nested conjugate
subgroups, their images in the finite group $F_n$ are nested conjugate finite
subgroups and therefore equal.  Let $\ell_n$ be word length on $K_n$ with
respect to $q_n(S_1)$ and define
\[
 f_n(y)=
 \begin{cases}
  \max\{1-\ell_n(y)/n,0\},&y\in K_n,\\
  0,&y\notin K_n.
 \end{cases}
\tag{6.1}
\]
Choose $\theta/2\pi$ irrational.  On $\ell^2(F_n)$ let $\lambda_n$ be the
left regular representation and let
\[
 D_n\delta_y=e^{i\theta f_n(y)}\delta_y.
\tag{6.2}
\]
For fixed $a\in L_1$, left translation changes $\ell_n$ by at most
$|a|_{S_1}$, whence
\[
 \opnorm{\lambda_n(q_n(a))D_n\lambda_n(q_n(a))^*-D_n}
 \le |\theta|\,|a|_{S_1}/n\longrightarrow0.
\tag{6.3}
\]
Thus $D=[D_n]$ commutes with the image $\Lambda(L_1)$ in the corona.  For
$x=rL_1$, put $U_x=\Lambda(r)D\Lambda(r)^*$.  This is independent of the
representative, the $U_x$ commute because their finite-stage lifts are
diagonal, and
$\Lambda(g)U_x\Lambda(g)^*=U_{gx}$.  Hence
\[
 \Psi\bigl((m_x)_{x\in X'},g\bigr)
   =\prod_x U_x^{m_x}\Lambda(g)
\tag{6.4}
\]
is a homomorphism.

It is faithful.  If $g\ne1$, then $q_n(g)\ne1$ eventually, and a diagonal
unitary times its nontrivial left translation stays at least $\sqrt2$ from the
identity on every basis vector.  If $g=1$ and $(m_x)$ is a nonzero finite
configuration, select a nonzero coefficient $m_{r_jL_1}$.  At the basis vector
$\delta_{q_n(r_j)}$, the chosen lamp contributes $e^{i\theta m_{r_jL_1}}$ and,
by the $3n$ separation, every other lamp in the support contributes $1$ for
all large $n$.  Irrationality makes the displayed phase nontrivial.  Thus no
nonidentity element is killed.
\end{proof}

We now prove the classification.  Suppose first that a countable abelian group
$A$ has a nontrivial torsion element $a$, and let
$\pi\colon A^{(X')}\rtimes G_*\to\U(\mathcal Q_{\mathbf d})$ be any
homomorphism.  The unitary $\pi(a_{L_1})$ has finite spectrum.  The
$\Gamma_*$-conjugates of $a_{L_1}$ commute, and $a_{L_1}$ centralizes $L_1$.
Theorem~\ref{thm:spectral} therefore gives
$\pi(a_{\gamma L_1})=\pi(a_{L_1})$ for every $\gamma\in\Gamma_*$.  Choosing
$\gamma\in\Gamma_*\setminus L_1$ puts the nontrivial two-site word
\[
 a_{\gamma L_1}a_{L_1}^{-1}
\tag{6.5}
\]
in $\ker\pi$.  The wreath product is therefore not MF.

Conversely, suppose $A$ is torsion-free.  Then $A\hookrightarrow
A\otimes_{\Z}\mathbb Q$.  Since this is a countable-dimensional
$\mathbb Q$-vector space, choose a linear embedding into a subspace of
$\mathbb R$ disjoint from $2\pi\mathbb Q$ except at zero.  Denote its
restriction by $\vartheta\colon A\to\mathbb R$.  Replace (6.2) by the
exact representation
\[
 D_n(a)\delta_y=e^{i\vartheta(a)f_n(y)}\delta_y
 \qquad(a\in A).
\tag{6.6}
\]
Additivity gives the lamp relations exactly, (6.3) gives covariance in the
corona, and the two separation estimates following (6.4) remain unchanged.
Since $e^{i\vartheta(a)}\ne1$ for $a\ne0$, the resulting representation of
$A^{(X)}\rtimes G_*$ is faithful.  This proves
Theorem~\ref{thm:abelian-boundary}.

For completeness, the same example separates the operator norm from its
Hilbert--Schmidt shadow.  Let $R_{\infty\to2}(H)$ be the set of $g\in H$ such
that, for every sequence $\varphi_n\colon H\to\U(d_n)$ whose multiplicative
defects tend to zero in operator norm,
$\opnorm{\varphi_n(g)-1}_{2,d_n}\to0$ in normalized Hilbert--Schmidt norm.
Clearly $\Rad_{\mathrm{MF}}(H)\le R_{\infty\to2}(H)$.

\begin{corollary}\label{cor:shadow-gap}
For $H=\Z^{(X')}\rtimes G_*$ one has
\[
 \Rad_{\mathrm{MF}}(H)\subsetneq R_{\infty\to2}(H).
\tag{6.7}
\]
\end{corollary}

\begin{proof}
Let $z$ be the generator of the lamp at $L_1$ and choose
$\gamma\in\Gamma_*\setminus L_1$.  For every operator-norm almost
representation $(\varphi_n)$, conjugation by $\varphi_n(g)$ induces a genuine
unitary representation on the normalized Hilbert--Schmidt ultraproduct: the
operator-norm multiplicative defects vanish there.  The vector represented by
$(\varphi_n(z))$ is fixed by $L_1$, because $z$ centralizes $L_1$.  Apply
Lemma~\ref{lem:fixed-space} with $L=\Gamma_*$ and $s=t$.  The vector is then
fixed by $\Gamma_*$, and in particular by $\gamma$.  Translating this equality
back to the representing matrices gives
\[
 \opnorm{\varphi_n(\gamma z\gamma^{-1}z^{-1})-1}_{2,d_n}\longrightarrow0.
\]
Thus the nontrivial two-site lamp
$\gamma z\gamma^{-1}z^{-1}$ lies in $R_{\infty\to2}(H)$.  It survives under
the faithful matrix-corona representation (6.4), so it does not lie in
$\Rad_{\mathrm{MF}}(H)$.
\end{proof}
