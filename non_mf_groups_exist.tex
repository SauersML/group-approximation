\title[Self-compression]{Self-compression and finite-dimensional invisibility}
\author{\paperauthor}
\subjclass[2020]{Primary 20C07; Secondary 16S50, 20G35, 22D10}
\keywords{finite-dimensional representations, Leavitt algebra, elementary group, property~(T), Bohr compactification}

\begin{abstract}
We construct a finitely generated Kazhdan group $H$ for which every
finite-dimensional linear representation over every field is trivial.  The
group is the concrete elementary matrix group
\[
 H=\EL_{16}\bigl(L_{\F_2}(1,2)\bigr).
\]
Its Bohr compactification is trivial, and every homomorphism from $H$ to a
residually finite group is trivial as well.

The proof has two ingredients.  First, one-sided compression of a subgroup
forces a commutator to vanish in every finite-dimensional representation: an
injective endomorphism of a finite-dimensional commutant is surjective.
Second, an explicit Whitehead matrix over the binary Leavitt algebra produces
one such commutator which normally generates $H$.  Thus a single relation
forced by finite-dimensionality annihilates every representation of the
group.
\end{abstract}

\section{Introduction}

Finite-dimensional representations often retain enough information to
separate a group.  This happens for residually finite groups, for finitely
generated linear groups by Mal'cev's theorem~\cite{Malcev}, and for maximally
almost periodic groups by definition.  The opposite extreme is a
\emph{minimally almost periodic} group: one admitting no nontrivial
finite-dimensional unitary representation.  We give a concrete Kazhdan group
at a still stronger extreme.

Let $L_{\F_2}(1,2)$ denote the binary Leavitt algebra.  It is the unital
$\F_2$-algebra generated by $s_0,s_1,t_0,t_1$ subject to
\begin{equation}\label{eq:leavitt}
 t_i s_j=\delta_{ij},
 \qquad s_0t_0+s_1t_1=1.
\end{equation}
For a unital ring $R$, write $e_{ij}(a)=1+aE_{ij}$ and let
$\EL_n(R)$ be the subgroup of $\GL_n(R)$ generated by the elements
$e_{ij}(a)$, $i\ne j$.

\begin{mainthm}\label{thm:headline}
Put $R=L_{\F_2}(1,2)$ and $H=\EL_{16}(R)$.  Then $H$ is nontrivial,
finitely generated, and has property~\textup{(T)}.  For every field $k$,
every finite-dimensional $k$-vector space $V$, and every homomorphism
\[
 \rho\colon H\longrightarrow\GL(V),
\]
one has $\rho(H)=\{1\}$.
\end{mainthm}

The conclusion is simultaneous over all characteristics.  In particular,
$H$ has no nontrivial finite quotient.  Peter--Weyl point separation gives
the compact-group form of the result.

\begin{mainthm}\label{thm:targets}
Every homomorphism from $H$ to a compact Hausdorff group or to a residually
finite group is trivial.  Equivalently, the Bohr compactification and the
profinite completion of $H$ are both trivial.
\end{mainthm}

The mechanism is intrinsic and elementary.  Suppose $L\le G$, an element
$u\in G$ satisfies $uLu^{-1}\le L$, and $c$ centralizes $L$.  Every
finite-dimensional representation of $G$ kills
\begin{equation}\label{eq:basic-defect}
 [ucu^{-1},\ell],\qquad \ell\in L.
\end{equation}
Indeed, conjugation by the inverse of the representing matrix for $u$ maps
the commutant of the image of $L$ injectively into itself.  Finite
dimensionality makes this map surjective, so conjugation by $u$ preserves the
commutant in both directions.

The binary Leavitt relations realize this configuration inside $H$.  A
$4\times4$ corner subgroup $L\cong\EL_4(R)$ is compressed into itself by an
explicit element $\tau\in H$.  A root element $c=e_{45}(1)$ centralizes that
corner, and for a suitable $\ell\in L$ the commutator in
\eqref{eq:basic-defect} is
\[
 d=e_{02}(s_1t_1).
\]
The coefficient $s_1t_1$ is full, and two Steinberg commutator identities
show that $d$ normally generates $H$.  Thus every finite-dimensional
representation kills one normal generator and is consequently trivial.

The result concerns genuine finite-dimensional representations.  Passing from
exact finite dimension to operator-norm asymptotic finite dimension is a
separate problem, and the argument below does not address it.

\subsection*{Relation to prior work}

Leavitt introduced the algebras $L_K(1,n)$ to exhibit modules of nonunique
rank~\cite{Leavitt}.  Their matrix and elementary groups retain the resulting
self-similarity.  Property~\textup{(T)} for elementary groups over finitely
generated rings follows from the work of Ershov and Jaikin-Zapirain
\cite{EJZ}; see also the account in~\cite{BHV}.  Minimally almost periodic
groups and the von Neumann radical have a long history in abstract harmonic
analysis; our result strengthens minimal almost periodicity by treating
linear representations over arbitrary fields.

The proof is also adjacent to classical rigidity of finite-dimensional
linear images.  Mal'cev proved that finitely generated linear groups are
residually finite~\cite{Malcev}, while Peter--Weyl shows that
finite-dimensional unitary representations separate points of compact
groups~\cite{PeterWeyl}.  These theorems turn the all-fields conclusion of
Theorem~\ref{thm:headline} into the target-group statements of
Theorem~\ref{thm:targets}.

\section{One-sided compression in finite dimension}

We isolate the representation-theoretic argument.  No topology, trace, or
unitarity is involved.

\begin{theorem}[the commutant theorem]\label{thm:commutant}
Let $G$ be a group, let $L\le G$, and suppose $u\in G$ satisfies
$uLu^{-1}\le L$.  If $k$ is a field, $V$ is a finite-dimensional
$k$-vector space, and $\rho\colon G\to\GL(V)$ is a homomorphism, then
\[
 \rho(u)\rho(L)'\rho(u)^{-1}=\rho(L)',
\]
where $\rho(L)'$ is the commutant of $\rho(L)$ in $\operatorname{End}_k(V)$.
Consequently, if $c\in C_G(L)$, then
\[
 \rho([ucu^{-1},\ell])=1
 \qquad(\ell\in L).
\]
\end{theorem}

\begin{proof}
Put $C=\rho(L)'$.  If $x\in C$, $h\in L$, and
$h'=uhu^{-1}\in L$, then
\begin{align*}
 \rho(h)\rho(u)^{-1}x\rho(u)
 &=\rho(u)^{-1}\rho(h')x\rho(u)\\
 &=\rho(u)^{-1}x\rho(h')\rho(u)\\
 &=\rho(u)^{-1}x\rho(u)\rho(h).
\end{align*}
Thus $\rho(u)^{-1}C\rho(u)\subseteq C$.  Conjugation by
$\rho(u)^{-1}$ is injective, and $C$ is finite-dimensional.  Its restriction
to $C$ is therefore surjective, so the inclusion is equality.  Conjugating
by $\rho(u)$ gives $\rho(u)C\rho(u)^{-1}=C$.

If $c\in C_G(L)$, then $\rho(c)\in C$, hence
$\rho(ucu^{-1})\in C$.  It commutes with every $\rho(\ell)$,
$\ell\in L$, which proves the last assertion.
\end{proof}

Let $\operatorname{Comp}_G(L)$ consist of the $u\in G$ satisfying
$uLu^{-1}\le L$, and let $G^+(L)$ be the subgroup they generate.  Define
\begin{equation}\label{eq:defect-subgroup}
 D(G,L)=\normal{[g c g^{-1},\ell]:
   g\in G^+(L),\ c\in C_G(L),\ \ell\in L}_G.
\end{equation}

\begin{corollary}\label{cor:all-compressors}
Every finite-dimensional linear representation of $G$, over every field, is
trivial on $D(G,L)$.  In particular, if $D(G,L)=G$, then every such
representation of $G$ is trivial.
\end{corollary}

\begin{proof}
For a fixed representation $\rho$, the elements $g\in G$ satisfying
$\rho(g)C\rho(g)^{-1}=C$, where $C=\rho(L)'$, form a subgroup.  Theorem
\ref{thm:commutant} puts every compressor in this subgroup.  Hence it contains
$G^+(L)$, and every generator in \eqref{eq:defect-subgroup} lies in
$\ker\rho$.  Normality of the kernel gives the result.
\end{proof}

\section{The binary Leavitt self-compression}

Throughout this section $R=L_{\F_2}(1,2)$ and
\[
 p=s_0t_0,\qquad q=s_1t_1.
\]
The relations \eqref{eq:leavitt} give
\begin{equation}\label{eq:pq}
 p+q=1,\qquad t_1qs_1=1.
\end{equation}
In particular $q\ne0$.

We use indices $0,\ldots,15$ for $16\times16$ matrices.  Embed
$M_4(R)$ as the upper-left corner of $M_{16}(R)$, with the identity on the
complement.  On $M_4(R)$ consider the unital injective endomorphism
\begin{equation}\label{eq:matrix-compression}
 \Psi(A)=qI_4+s_0A t_0.
\end{equation}
Here the coefficients multiply every entry.  Multiplicativity follows from
$t_0s_0=1$ and the vanishing of the mixed terms, while
$t_0\Psi(A)s_0=A$ proves injectivity.

Set
\[
 U=\begin{pmatrix}s_0I_4&s_1t_0I_4\\0&t_1I_4\end{pmatrix},
 \qquad
 V=\begin{pmatrix}t_0I_4&0\\s_0t_1I_4&s_1I_4\end{pmatrix}.
\]
A direct calculation using \eqref{eq:leavitt} gives $UV=VU=I_8$.  Put
\begin{equation}\label{eq:tau}
 \tau=\operatorname{diag}(U,V)\in\GL_{16}(R).
\end{equation}
The Whitehead factorization of $\operatorname{diag}(U,U^{-1})$ expresses
$\tau$ as a product of elementary matrices, so $\tau\in\EL_{16}(R)$.

Let $H=\EL_{16}(R)$ and let $L=\EL_4(R)\le H$ be the upper-left corner.
Block multiplication yields
\begin{equation}\label{eq:corner-conjugation}
 \tau\,\operatorname{diag}(A,I_{12})\,\tau^{-1}
 =\operatorname{diag}(\Psi(A),I_{12}).
\end{equation}
The endomorphism $\Psi$ carries elementary matrices to elementary matrices:
$\Psi(e_{ij}(a))=e_{ij}(s_0at_0)$.  It follows that
\begin{equation}\label{eq:compresses-L}
 \tau L\tau^{-1}\le L.
\end{equation}

\begin{remark}
The complementary term $qI_4$ in \eqref{eq:matrix-compression} is essential.
The entrywise map $a\mapsto s_0at_0$ is nonunital; $qI_4$ turns it into a
homomorphism on the corner unit group.
\end{remark}

\section{One defect normally generates}

\begin{proposition}\label{prop:defect}
Let
\[
 c=e_{45}(1),\qquad \ell=e_{12}(1),\qquad
 d=[\tau c\tau^{-1},\ell]
\]
in $H$.  Then
\[
 c\in C_H(L),\qquad d=e_{02}(q)\ne1,qquad \normal d_H=H.
\]
\end{proposition}

\begin{proof}
Every elementary generator of the upper-left $4\times4$ corner has both
indices in $\{0,1,2,3\}$.  The elementary commutator relations show that it
commutes with $c=e_{45}(1)$, so $c\in C_H(L)$.

Direct multiplication of the matrices in \eqref{eq:tau} gives
\begin{equation}\label{eq:moved-mark}
 \tau c\tau^{-1}=e_{01}(q)e_{45}(1).
\end{equation}
The second factor commutes with $\ell=e_{12}(1)$, and
$[e_{ij}(a),e_{jk}(b)]=e_{ik}(ab)$ gives
$d=e_{02}(q)$.  This is nontrivial because $q\ne0$.

Let $N=\normal d_H$.  For arbitrary $a\in R$, two further elementary
commutator identities give
\begin{align*}
 [e_{30}(at_1),e_{02}(q)]&=e_{32}(at_1q)\in N,\\
 [e_{32}(at_1q),e_{21}(s_1)]&=e_{31}(at_1qs_1)=e_{31}(a)\in N,
\end{align*}
where the last equality uses \eqref{eq:pq}.  Elementary Weyl matrices
conjugate the root position $(3,1)$ to any ordered pair of distinct indices.
Thus $N$ contains every $e_{ij}(a)$, and these elements generate $H$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:headline}]
The group $H$ is nontrivial by Proposition~\ref{prop:defect}.  The ring $R$
is finitely generated, and $H$ has property~\textup{(T)} by the
Ershov--Jaikin-Zapirain theorem for elementary groups over finitely generated
rings~\cite{EJZ}; in particular $H$ is finitely generated.

Let $k$ be a field and let $\rho\colon H\to\GL(V)$ with
$\dim_kV<\infty$.  The containment \eqref{eq:compresses-L}, the centrality of
$c$ relative to $L$, and Theorem~\ref{thm:commutant} give $\rho(d)=1$.
Proposition~\ref{prop:defect} gives $H=\normal d_H$, so $\rho$ is trivial.
\end{proof}

\section{Universal consequences}

\begin{proposition}\label{prop:factorization}
Let $T$ be a finite-dimensional linear group over a field, a residually
finite group, or a compact Hausdorff group.  Then $\Hom(H,T)$ consists only
of the trivial homomorphism.
\end{proposition}

\begin{proof}
The linear case is Theorem~\ref{thm:headline}.  If a homomorphism to a
residually finite group were nontrivial, some finite quotient of its image
would be nontrivial.  Every finite group has a faithful complex regular
representation, contradicting the linear case.

If $T$ is compact Hausdorff and $f(H)$ is nontrivial, Peter--Weyl point
separation supplies a finite-dimensional unitary representation of $T$
nontrivial on $f(H)$.  Its composition with $f$ again contradicts the linear
case.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:targets}]
Apply Proposition~\ref{prop:factorization} and the universal properties of
the Bohr compactification and profinite completion.
\end{proof}

\begin{corollary}\label{cor:cstar}
Every finite-dimensional $*$-representation of $C^*_{\max}(H)$ factors
through the augmentation $C^*_{\max}(H)\to\C$.  In particular,
$C^*_{\max}(H)$ is not residually finite-dimensional.
\end{corollary}

\begin{proof}
Such a representation restricts on the canonical group unitaries to a
finite-dimensional unitary representation of $H$, hence to the trivial
representation.  Since $H$ is nontrivial, finite-dimensional
$*$-representations do not separate the canonical image of $H$.
\end{proof}

\begin{remark}[the approximation boundary]\label{rem:boundary}
The proof uses equality inside genuine finite-dimensional commutants.  In a
norm matrix corona the relative commutant can be strictly larger than its
finite-stage antecedents.  Consequently Theorem~\ref{thm:headline} does not
by itself determine whether $H$ is MF.  Establishing or refuting a faithful
corona representation requires an additional argument.
\end{remark}
