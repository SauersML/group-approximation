\title[Full MF radical]{A simple group with full MF radical}
\author{\paperauthor}
\subjclass[2020]{Primary 46L05; Secondary 20C07, 16S50, 20G35}
\keywords{MF groups, norm matrix coronas, simple groups, property~(T), Leavitt algebra, elementary group}

\begin{abstract}
A countable group is MF if it embeds in the unitary group of a norm matrix
corona.  We construct a nontrivial finitely generated simple Kazhdan group $H$
such that every homomorphism from $H$ to an MF group is trivial.  Equivalently,
the MF radical of $H$ is all of $H$, so $H$ is non-MF.  We take
\[
 H=\EL_{16}\bigl(L_{\F_2}(1,2)\bigr).
\]
Conjugation by an explicit Whitehead matrix maps a Kazhdan subgroup
$L\cong\EL_4(L_{\F_2}(1,2))$ into itself.  If $c$ centralizes $L$, one-sided
Kazhdan transport implies that the commutators $[ucu^{-1},\ell]$ converge to
the identity in normalized Hilbert--Schmidt norm in every operator-norm
asymptotic representation.  A finite-order active-core theorem shows that
universal Hilbert--Schmidt triviality and MF invisibility coincide for a
torsion normal generator.  For our choice of $u,c$, and $\ell$, the
commutator is an involutory elementary matrix that normally generates $H$.
Every finite-dimensional linear representation of $H$ over any field is also
trivial.  The same group yields a general construction: for every countable
group $Q$, there is a split epimorphism $\pi_Q\colon W_Q\to Q$ such that
\[
 \Rad_{\mathrm{MF}}(W_Q)
 =\pi_Q^{-1}\bigl(\Rad_{\mathrm{MF}}(Q)\bigr).
\]
If $Q$ is MF, the kernel of $\pi_Q$ is the MF radical of $W_Q$ and is
normally generated by the single element $d$ above.  For every $Q$, this
kernel is itself a perfect group with full MF radical and admits an explicit
decomposition as copies of $H$ amalgamated over $\langle d\rangle$.  If every
quotient of $Q$ is MF, then the single relation $d=1$ classifies MF throughout
the quotient lattice of $W_Q$: for every normal subgroup $N$,
$W_Q/N$ is MF exactly when $d\in N$.  For $Q=\mathbb Z$, the ambient group is
finitely generated while its exact MF radical is infinitely generated but
normally generated by $d$.
\end{abstract}

\section{Introduction}

A countable group is \emph{MF} if it embeds in the unitary group of a norm
matrix corona
\[
 \mathcal Q_{\mathbf d}
 =\prod_nM_{d_n}(\C)\big/\bigoplus_nM_{d_n}(\C).
\]
Equivalently, it admits finite-dimensional unitary models whose
multiplicative defects tend to zero in operator norm and which asymptotically
separate its nonidentity elements.  This is the convention of
Carri\'on--Dadarlat--Eckhardt~\cite{CDE}.

For a countable group $G$, define its MF radical by
\[
 \Rad_{\mathrm{MF}}(G)
 =\bigcap_{\mathbf d}
   \bigcap_{\pi\colon G\to\U(\mathcal Q_{\mathbf d})}\ker\pi.
\]
The group $G$ is MF precisely when $\Rad_{\mathrm{MF}}(G)$ is trivial.  If
$\Rad_{\mathrm{MF}}(G)=G$, every homomorphism from $G$ to an MF group is
trivial.

More generally, for $N\trianglelefteq G$ define its \emph{MF closure} by
\[
 \operatorname{cl}_{\mathrm{MF}}^G(N)
 =\bigcap\{\ker f:N\le\ker f,\ f\colon G\to M,\ M\text{ MF}\}.
\]
Thus $\Rad_{\mathrm{MF}}(G)=\operatorname{cl}_{\mathrm{MF}}^G(1)$, and
$G/N$ is MF precisely when
$\operatorname{cl}_{\mathrm{MF}}^G(N)=N$.

The existence of a countable non-MF group was open~\cite{Shulman}.
Theorem~\ref{thm:headline} answers this question and proves the stronger equality
$\Rad_{\mathrm{MF}}(H)=H$ for an explicit simple group.

Let $L_{\F_2}(1,2)$ denote the binary Leavitt algebra.  It is the unital
$\F_2$-algebra generated by $s_0,s_1,t_0,t_1$ subject to
\begin{equation}\label{eq:leavitt}
 t_i s_j=\delta_{ij},
 \qquad s_0t_0+s_1t_1=1.
\end{equation}
For a unital ring $R$, write $e_{ij}(a)=1+aE_{ij}$ and let
$\EL_n(R)$ be the subgroup of $\GL_n(R)$ generated by the elements
$e_{ij}(a)$, $i\ne j$.

\begin{mainthm}\label{thm:headline}
Put $R=L_{\F_2}(1,2)$ and $H=\EL_{16}(R)$.  Then $H$ is nontrivial, simple,
finitely generated, and has property~\textup{(T)}, and
\[
 \Rad_{\mathrm{MF}}(H)=H.
\]
Equivalently, every homomorphism from $H$ to an MF group is trivial.
In particular, $H$ is non-MF.
\end{mainthm}

Call an element of $G$
\emph{universally Hilbert--Schmidt trivial} if its value converges to the
identity in normalized Hilbert--Schmidt norm in every operator-norm
asymptotic representation of $G$.

\begin{mainthm}[finite-order normal generators]\label{thm:torsion-bridge}
Let $G$ be finitely generated and let $a\in G$ have finite order and normally
generate $G$.  Then
\[
 a\in\Rad_{\mathrm{MF}}(G)
 \quad\Longleftrightarrow\quad
 a\in R_{\infty\to2}(G).
\]
Consequently, if $a$ is universally Hilbert--Schmidt trivial, then
$\Rad_{\mathrm{MF}}(G)=G$.
\end{mainthm}

Suppose that $L\le G$ has property~\textup{(T)}, that $u\in G$ satisfies
$uLu^{-1}\le L$, and that $c$ centralizes $L$.  For every operator-norm
asymptotic representation $(V_n)$ of $G$,
\begin{equation}\label{eq:basic-defect}
 \hsnorm{V_n([ucu^{-1},\ell])-1}\longrightarrow0
 \qquad(\ell\in L).
\end{equation}

Inside $H$, an explicit element $\tau$ satisfies $\tau L\tau^{-1}\le L$ for
the upper-left subgroup $L\cong\EL_4(R)$.  The element $c=e_{45}(1)$
centralizes $L$, and for $\ell=e_{12}(1)$ the corresponding commutator is
\[
 d=e_{02}(s_1t_1).
\]
Two elementary commutator identities imply that $d$ normally generates
$H$.  Since the coefficient ring has characteristic two, $d$ is an
involution.  Theorem~\ref{thm:torsion-bridge} then gives
$\Rad_{\mathrm{MF}}(H)=H$.  Thus property~\textup{(T)} is used for the
compressed subgroup $L$ in the transport step; the passage from the
Hilbert--Schmidt obstruction to the MF radical uses only the finite order and
normal generation of $d$.

For an exact finite-dimensional representation, conjugation by a compressor
defines an injective endomorphism of the commutant of $L$.  This endomorphism
is surjective because the commutant is finite-dimensional.  Consequently,
every finite-dimensional linear representation of $H$ over every field is
trivial.

The group $H$ also gives non-MF groups with prescribed MF quotients.  The
following theorem is proved in Section~\ref{sec:prescribed-quotients}.

\begin{mainthm}[prescribed MF quotients]\label{thm:prescribed-quotients}
For every countable group $Q$, there are a countable group $W_Q$ and a split
epimorphism $\pi_Q\colon W_Q\to Q$ such that
\[
 \Rad_{\mathrm{MF}}(W_Q)
 =\pi_Q^{-1}\bigl(\Rad_{\mathrm{MF}}(Q)\bigr).
\]
The group $W_Q$ is non-MF.  If $Q$ is MF, then
\[
 \Rad_{\mathrm{MF}}(W_Q)=\ker\pi_Q=\normal d_{W_Q},
\]
where $d$ is the element in Proposition~\ref{prop:defect}.
Moreover, precomposition with $\pi_Q$ induces a bijection
\[
 \operatorname{Hom}(Q,M)\longrightarrow\operatorname{Hom}(W_Q,M)
\]
for every MF group $M$.
For every $N\trianglelefteq W_Q$, this factorization determines the entire MF
closure:
\[
 \operatorname{cl}_{\mathrm{MF}}^{W_Q}(N)
 =\pi_Q^{-1}\!\left(
  \operatorname{cl}_{\mathrm{MF}}^Q(\pi_Q(N))
 \right).
\]
If every quotient of $Q$ is MF, then for every
$N\trianglelefteq W_Q$,
\[
 \operatorname{cl}_{\mathrm{MF}}^{W_Q}(N)
 =N\normal d_{W_Q},
 \qquad
 W_Q/N\text{ is MF}\quad\Longleftrightarrow\quad d\in N.
\]
Writing $K_Q=\ker\pi_Q$, one also has
\[
 \begin{aligned}
  \Rad_{\mathrm{MF}}(K_Q)&=K_Q,& [K_Q,K_Q]&=K_Q,\\
  Z(K_Q)&=1,& W_Q&\cong K_Q\rtimes Q.
 \end{aligned}
\]
The element $d$ normally generates $K_Q$ itself.  The group $K_Q$ is finitely
generated exactly when $Q$ is finite, and its isomorphism type depends only
on the cardinality of $Q$.  In particular, one fixed countable perfect group
with full MF radical occurs as $K_Q$ for every countably infinite $Q$.
\end{mainthm}

\subsection*{Relation to prior work}

Leavitt introduced the algebras $L_K(1,n)$ as examples of rings with
nonunique module rank~\cite{Leavitt}.  Their defining relations yield the
internal endomorphism of the elementary matrix group used here.  The binary
Leavitt algebra is purely infinite simple~\cite{AbramsAranda}, hence an exchange
ring~\cite{AraExchange}, and its center is the base field~\cite{ArandaCrow}.
Preusser's normal-subgroup theorem for elementary groups over exchange rings
implies that $H$ is simple~\cite{Preusser}.  The theorem of
Ershov and Jaikin-Zapirain gives property~\textup{(T)} for elementary groups
over finitely generated rings~\cite{EJZ}; standard background on
property~\textup{(T)} may be found in~\cite{BHV}.  Carri\'on, Dadarlat, and
Eckhardt relate the MF property for groups to quasidiagonality of group
$C^*$-algebras~\cite{CDE}, and Shulman studies permanence of the MF property
under amalgamated free products~\cite{Shulman}.  Related rigidity results for
metric approximations of Kazhdan groups appear in~\cite{KT19,AlekseevThom};
Dadarlat studies obstructions to matricial stability in~\cite{Dadarlat}.

\section{One-sided compression in finite dimension}

\begin{theorem}[finite-dimensional commutant rigidity]\label{thm:commutant}
Let $G$ be a group, let $L\le G$, and suppose $u\in G$ satisfies
$uLu^{-1}\le L$.  If $k$ is a field, $V$ is a finite-dimensional
$k$-vector space, and $\rho\colon G\to\GL(V)$ is a homomorphism, then
\[
 \rho(u)\rho(L)'\rho(u)^{-1}=\rho(L)',
\]
where $\rho(L)'$ is the commutant of $\rho(L)$ in $\operatorname{End}_k(V)$.
Consequently, if $c\in C_G(L)$, then
\[
 \rho([ucu^{-1},\ell])=1
 \qquad(\ell\in L).
\]
\end{theorem}

\begin{proof}
Put $C=\rho(L)'$.  If $x\in C$, $h\in L$, and
$h'=uhu^{-1}\in L$, then
\begin{align*}
 \rho(h)\rho(u)^{-1}x\rho(u)
 &=\rho(u)^{-1}\rho(h')x\rho(u)\\
 &=\rho(u)^{-1}x\rho(h')\rho(u)\\
 &=\rho(u)^{-1}x\rho(u)\rho(h).
\end{align*}
Thus $\rho(u)^{-1}C\rho(u)\subseteq C$.  Conjugation by
$\rho(u)^{-1}$ is injective, and $C$ is finite-dimensional.  Its restriction
to $C$ is therefore surjective, so the inclusion is equality.  Conjugating
by $\rho(u)$ gives $\rho(u)C\rho(u)^{-1}=C$.

If $c\in C_G(L)$, then $\rho(c)\in C$, hence
$\rho(ucu^{-1})\in C$.  It commutes with every $\rho(\ell)$,
$\ell\in L$, which proves the last assertion.
\end{proof}

Let $\operatorname{Comp}_G(L)$ consist of the $u\in G$ satisfying
$uLu^{-1}\le L$, and let $G^+(L)$ be the subgroup they generate.  Define
\begin{equation}\label{eq:defect-subgroup}
 D(G,L)=\normal{[g c g^{-1},\ell]:
   g\in G^+(L),\ c\in C_G(L),\ \ell\in L}_G.
\end{equation}

\begin{corollary}\label{cor:all-compressors}
Every finite-dimensional linear representation of $G$, over every field, is
trivial on $D(G,L)$.  In particular, if $D(G,L)=G$, then every such
representation of $G$ is trivial.
\end{corollary}

\begin{proof}
For a fixed representation $\rho$, the elements $g\in G$ for which
$\rho(g)C\rho(g)^{-1}=C$, where $C=\rho(L)'$, form a subgroup.  Theorem
\ref{thm:commutant} shows that this subgroup contains
$\operatorname{Comp}_G(L)$ and therefore $G^+(L)$.  Every commutator in
\eqref{eq:defect-subgroup} lies in $\ker\rho$, so its normal closure does as
well.
\end{proof}

\section{Kazhdan transport in normalized Hilbert--Schmidt norm}

For $a\in M_d(\C)$ write
\[
 \hsnorm{a}=\tr_d(a^*a)^{1/2},
 \qquad \tr_d=\frac1d\operatorname{Tr}.
\]
An \emph{operator-norm asymptotic representation} of a countable group $G$
is a sequence of maps $V_n\colon G\to\U(d_n)$ such that $V_n(1)=1$ and
\[
 \opnorm{V_n(gh)-V_n(g)V_n(h)}\longrightarrow0
 \qquad(g,h\in G).
\]
For such a sequence put
\[
 K_2(V)=\{g\in G:\hsnorm{V_n(g)-1}\longrightarrow0\}.
\]
The operator-norm defects also tend to zero in normalized
Hilbert--Schmidt norm, so $K_2(V)$ is a normal subgroup of $G$.  Define
\begin{equation}\label{eq:shadow-residual}
 R_{\infty\to2}(G)=\bigcap_V K_2(V),
\end{equation}
where $V$ ranges over all operator-norm asymptotic representations of $G$.
An element is universally Hilbert--Schmidt trivial precisely when it belongs
to $R_{\infty\to2}(G)$.

\begin{lemma}\label{lem:stable-finite}
For every sequence $(m_n)$, the norm matrix corona
\[
 \prod_nM_{m_n}(\C)\big/\bigoplus_nM_{m_n}(\C)
\]
is stably finite.  Consequently, unitarily equivalent projections in a norm
matrix corona are equal whenever one is dominated by the other.
\end{lemma}

\begin{proof}
Suppose that $v^*v=1$ in the corona, and let $(x_n)$ be a bounded lift of
$v$.  Then $x_n^*x_n\to1$ in norm.  For all sufficiently large $n$, the matrix
$x_n$ is invertible, and the unitary $x_n(x_n^*x_n)^{-1/2}$ differs from
$x_n$ by $o(1)$.  Thus $v$ is represented by unitaries and $vv^*=1$.  The
same argument applies to every matrix amplification of the corona.  In a
stably finite unital $C^*$-algebra, equivalent projections $p\le q$ satisfy
$p=q$.
\end{proof}

\begin{theorem}[one-sided Kazhdan transport]\label{thm:transport}
Let $L\le G$ have property~\textup{(T)}, and let $u\in G$ satisfy
$uLu^{-1}\le L$.  Let $(V_n)$ be an operator-norm asymptotic
representation of $G$, and let $(x_n)$ be uniformly bounded in operator
norm.  If
\[
 \hsnorm{V_n(\ell)x_nV_n(\ell)^*-x_n}\longrightarrow0
 \qquad(\ell\in L),
\]
then, for every $\ell\in L$,
\begin{align*}
 \hsnorm{V_n(\ell)V_n(u)x_nV_n(u)^*V_n(\ell)^*
          -V_n(u)x_nV_n(u)^*}&\longrightarrow0,\\
 \hsnorm{V_n(\ell)V_n(u)^*x_nV_n(u)V_n(\ell)^*
          -V_n(u)^*x_nV_n(u)}&\longrightarrow0.
\end{align*}
\end{theorem}

\begin{proof}
Let $\omega$ be a free ultrafilter.  Regard $M_{d_n}(\C)$ as a Hilbert
space with its normalized Hilbert--Schmidt inner product, and form the
Hilbert-space ultraproduct $\mathcal K_\omega$.  Conjugation by the $V_n(g)$
defines a unitary representation $\sigma$ of $G$ on $\mathcal K_\omega$.
The class $\xi=[x_n]_\omega$ is fixed by $\sigma(L)$.

For $g\in G$, let $\widetilde\sigma(g)$ be the class of the coordinate
operators $\operatorname{Ad}(V_n(g))$ in the norm matrix corona
\[
 \mathcal B=\prod_nB(M_{d_n}(\C))\big/\bigoplus_nB(M_{d_n}(\C)).
\]
Its action on $\mathcal K_\omega$ is $\sigma(g)$.  Let $P\in\mathcal B$ be
the image of the Kazhdan projection of $L$ under
$\widetilde\sigma|_L$.  On $\mathcal K_\omega$, its range is
$\operatorname{Fix}\sigma(L)$
\cite[Chapter~3]{BHV}.  Since
$uLu^{-1}\le L$, the unitary $U=\widetilde\sigma(u)$ satisfies
\[
 U^*PU\le P.
\]
The two projections are unitarily equivalent.  Lemma~\ref{lem:stable-finite}
therefore gives $U^*PU=P$.  Hence both $U$ and $U^*$ preserve
$\operatorname{Fix}\sigma(L)$, so $U\xi$ and $U^*\xi$ are fixed by
$\sigma(L)$.  The two limits in the statement therefore vanish along
$\omega$.  Since this holds for every free ultrafilter, both sequences
converge ordinarily.
\end{proof}

\begin{corollary}
\label{cor:defect-hs}
Let $L\le G$ have property~\textup{(T)}.  Then
\[
 D(G,L)\le R_{\infty\to2}(G).
\]
\end{corollary}

\begin{proof}
Let $(V_n)$ be an operator-norm asymptotic representation.  If
$c\in C_G(L)$, the sequence $(V_n(c))$ asymptotically commutes with
$V_n(L)$ in operator norm and hence in normalized Hilbert--Schmidt norm.
Repeated application of Theorem~\ref{thm:transport} shows that the same is
true of $(V_n(gcg^{-1}))$ for every $g\in G^+(L)$.  Therefore
\[
 \hsnorm{V_n([gcg^{-1},\ell])-1}\longrightarrow0
 \qquad(\ell\in L).
\]
The subgroup $K_2(V)$ is normal, so it contains the normal closure of these
elements.  Intersecting over $(V_n)$ proves the claim.
\end{proof}

\section{Finite-order active cores and the MF radical}

The normalized Hilbert--Schmidt norm can lose a fixed operator-norm defect
when that defect occupies a vanishing fraction of the matrix dimension.  A
finite-order normal generator prevents this loss after an intrinsic
reblocking: the active spectral subspace of the generator expands, through
fixed normal words, to an invariant subspace whose dimension is only a fixed
multiple larger.

\begin{theorem}[active-core reblocking]\label{thm:active-core}
Let $G$ be generated by a finite symmetric set $S$ containing an element
$a$ of order $m\ge2$, and suppose that $a$ normally generates $G$.  For each
$s\in S$, fix an expression of $s$ as a product of conjugates of
$a^{\pm1}$, taking the expression for $a$ to be the one-letter word $a$.
Let $L$ be the total number of occurrences of $a^{\pm1}$ in these
expressions.

If $\rho\colon G\to\U(\mathcal Q_{\mathbf d})$ satisfies $\rho(a)\ne1$,
then, after passing to an infinite coordinate subsequence, $\rho$ has unitary
lifts of the form
\[
 \sigma_n(g)\oplus 1_{d_n-k_n},
 \qquad \sigma_n(g)\in\U(k_n),
\]
where $(\sigma_n)$ is an operator-norm asymptotic representation.  Moreover,
$A_n=\sigma_n(a)$ satisfies $A_n^m=1$, and if
\[
 p_n=1_{\C\setminus\{1\}}(A_n),\qquad r_n=\operatorname{rank}p_n,
\]
then
\begin{equation}\label{eq:active-core-ranks}
 1\le r_n\le k_n\le Lr_n.
\end{equation}
Consequently, with
\[
 \eta_m=\min\{|1-\zeta|:\zeta^m=1,\ \zeta\ne1\}>0,
\]
one has
\begin{equation}\label{eq:active-core-gap}
 \hsnorm{\sigma_n(a)-1}\ge\frac{\eta_m}{\sqrt L}.
\end{equation}
If $a$ is a product of $N_g$ conjugates of $g^{\pm1}$, then also
\begin{equation}\label{eq:propagated-gap}
 \liminf_n\hsnorm{\sigma_n(g)-1}
 \ge\frac{\eta_m}{N_g\sqrt L}.
\end{equation}
\end{theorem}

\begin{proof}
Choose unitary lifts $U_n(g)$ of $\rho(g)$.  They form an operator-norm
asymptotic representation.  Since $\rho(a)^m=1$, finite-spectrum functional
calculus replaces $U_n(a)$, within $o(1)$ in operator norm, by a unitary
$B_n$ satisfying $B_n^m=1$.

Write the chosen expression for $s\in S$ as
\[
 s=\prod_{j=1}^{L_s}h_{s,j}a^{\varepsilon_{s,j}}h_{s,j}^{-1},
 \qquad \varepsilon_{s,j}\in\{1,-1\}.
\]
Define
\[
 \widetilde U_n(s)=
 \prod_{j=1}^{L_s}
 U_n(h_{s,j})B_n^{\varepsilon_{s,j}}U_n(h_{s,j})^*.
\]
For every $g\in G$, choose once and for all a word in $S$ representing $g$
and define $\widetilde U_n(g)$ by substituting these matrices.  We use the
one-letter word for $a$, so $\widetilde U_n(a)=B_n$.  The defining group
identities and asymptotic multiplicativity give
\[
 \opnorm{\widetilde U_n(g)-U_n(g)}\longrightarrow0
 \qquad(g\in G).
\]
Thus $(\widetilde U_n)$ is an operator-norm asymptotic representation and
represents the same corona homomorphism.

Let $P_n$ be the active spectral subspace of $B_n$, and let $K_n$ be the sum
of the subspaces
\[
 U_n(h_{s,j})P_n
 \qquad(s\in S,\ 1\le j\le L_s).
\]
Every factor in the displayed formula for $\widetilde U_n(s)$ is the
identity on $K_n^\perp$, preserves $K_n$, and differs from the identity by
an operator whose range lies in $K_n$.  Hence every $\widetilde U_n(g)$
preserves $K_n$ and is the identity on $K_n^\perp$.  Since the one-letter
expression for $a$ occurs in the family,
\[
 \dim P_n\le\dim K_n\le L\dim P_n.
\]
The assumption $\rho(a)\ne1$ makes $P_n$ nonzero on an infinite coordinate
subsequence.  Retain that subsequence, put $k_n=\dim K_n$, and let
$\sigma_n(g)$ be the restriction of $\widetilde U_n(g)$ to $K_n$.  This
proves the lifting assertion and \eqref{eq:active-core-ranks}.

Every nonidentity eigenvalue of $A_n$ is an $m$th root of unity, so
\[
 \hsnorm{A_n-1}^2
 \ge \eta_m^2\frac{r_n}{k_n}
 \ge\frac{\eta_m^2}{L},
\]
which proves \eqref{eq:active-core-gap}.  Finally, substitute the matrices
$\sigma_n(g)$ into a fixed normal word of length $N_g$ expressing $a$ through
$g^{\pm1}$.  Telescoping in normalized Hilbert--Schmidt norm and asymptotic
multiplicativity give
\[
 \hsnorm{\sigma_n(a)-1}
 \le N_g\hsnorm{\sigma_n(g)-1}+o(1).
\]
Together with \eqref{eq:active-core-gap}, this proves
\eqref{eq:propagated-gap}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:torsion-bridge}]
For every group,
\[
 \Rad_{\mathrm{MF}}(G)\le R_{\infty\to2}(G):
\]
an operator-norm asymptotic representation defines a homomorphism to its
norm matrix corona, and an element killed there converges to the identity in
operator norm, hence in normalized Hilbert--Schmidt norm.

Conversely, suppose $a\in R_{\infty\to2}(G)$ but
$a\notin\Rad_{\mathrm{MF}}(G)$.  Some corona homomorphism sees $a$.
Theorem~\ref{thm:active-core} produces an operator-norm asymptotic
representation for which \eqref{eq:active-core-gap} holds, contradicting the
definition of $R_{\infty\to2}(G)$.  Thus $a$ lies in the MF radical.  Since
that radical is normal and $a$ normally generates $G$, it equals $G$.
\end{proof}

\begin{corollary}[simple torsion dichotomy]\label{cor:simple-torsion}
Let $G$ be a nontrivial finitely generated simple group containing a
finite-order nonidentity element.  Then
\[
 \Rad_{\mathrm{MF}}(G)=R_{\infty\to2}(G).
\]
Exactly one of the following holds:
\begin{enumerate}
 \item every homomorphism from $G$ to a norm matrix corona is trivial;
 \item $G$ admits a faithful operator-norm asymptotic representation
 $(\sigma_n)$ such that
 \[
  \liminf_n\hsnorm{\sigma_n(g)-1}>0
  \qquad(g\ne1).
 \]
\end{enumerate}
\end{corollary}

\begin{proof}
Both displayed residuals are normal subgroups.  Simplicity and
Theorem~\ref{thm:torsion-bridge}, applied to a nonidentity torsion element,
give their equality.  If a corona homomorphism is nontrivial, it is injective.
Apply Theorem~\ref{thm:active-core}; for each $g\ne1$, simplicity puts the
chosen torsion element in the normal closure of $g$, so
\eqref{eq:propagated-gap} gives the asserted pointwise separation.  The two
alternatives are therefore exhaustive and mutually exclusive.
\end{proof}

\section{The binary Leavitt self-compression}

Throughout this section $R=L_{\F_2}(1,2)$ and
\[
 p=s_0t_0,\qquad q=s_1t_1.
\]
The relations \eqref{eq:leavitt} give
\begin{equation}\label{eq:pq}
 p+q=1,\qquad t_1qs_1=1.
\end{equation}
In particular $q\ne0$.

We use indices $0,\ldots,15$ for $16\times16$ matrices and identify
$\GL_4(R)$ with the subgroup of $\GL_{16}(R)$ consisting of the matrices
$\operatorname{diag}(A,I_{12})$.  On $M_4(R)$ consider the unital injective
endomorphism
\begin{equation}\label{eq:matrix-compression}
 \Psi(A)=qI_4+s_0A t_0.
\end{equation}
Here $s_0At_0$ denotes the matrix $(s_0a_{ij}t_0)_{ij}$.  The relations
$qs_0=t_0q=0$ and $t_0s_0=1$ show that $\Psi$ is multiplicative and unital.
The identity $t_0\Psi(A)s_0=A$ proves injectivity.  The term $qI_4$ is needed
for unitality because the map $A\mapsto s_0At_0$ sends $I_4$ to $pI_4$.

Set
\[
 U=\begin{pmatrix}s_0I_4&s_1t_0I_4\\0&t_1I_4\end{pmatrix},
 \qquad
 V=\begin{pmatrix}t_0I_4&0\\s_0t_1I_4&s_1I_4\end{pmatrix}.
\]
A direct calculation using \eqref{eq:leavitt} gives $UV=VU=I_8$.  Put
\begin{equation}\label{eq:tau}
 \tau=\operatorname{diag}(U,V)\in\GL_{16}(R).
\end{equation}
The Whitehead factorization of $\operatorname{diag}(U,U^{-1})$ expresses
$\tau$ as a product of elementary matrices, so $\tau\in\EL_{16}(R)$.

Let $H=\EL_{16}(R)$ and let $L=\EL_4(R)\le H$ be the upper-left corner.
Both groups have property~\textup{(T)} by the theorem of Ershov and
Jaikin-Zapirain~\cite{EJZ}.
Block multiplication yields
\begin{equation}\label{eq:corner-conjugation}
 \tau\,\operatorname{diag}(A,I_{12})\,\tau^{-1}
 =\operatorname{diag}(\Psi(A),I_{12}).
\end{equation}
For every $i\ne j$ and $a\in R$,
$\Psi(e_{ij}(a))=e_{ij}(s_0at_0)$.  Therefore
\begin{equation}\label{eq:compresses-L}
 \tau L\tau^{-1}\le L.
\end{equation}

\section{The full MF radical}

\begin{proposition}\label{prop:simple}
The group $H=\EL_{16}(R)$ is simple.
\end{proposition}

\begin{proof}
The ring $R=L_{\F_2}(1,2)$ is purely infinite simple
\cite{AbramsAranda} and therefore an exchange ring~\cite{AraExchange}.
Let $N\trianglelefteq H$.  Preusser's normal-subgroup theorem
\cite[Theorem~3]{Preusser} gives an ideal $I\trianglelefteq R$ such that
\[
 \EL_{16}(R,I)\le N\le C_{16}(R,I),
\]
where $C_{16}(R,I)$ is the full congruence subgroup of level $I$.
Since $R$ is simple, either $I=R$ or $I=0$.  In the first case
$\EL_{16}(R,I)=H$, so $N=H$.  In the second case
\[
 N\le C_{16}(R,0)=Z(\GL_{16}(R)).
\]

The center of $R$ is $\F_2$ by~\cite[Corollary~4.3]{ArandaCrow}.  For
completeness, suppose that $z\in Z(\GL_{16}(R))$.  Commuting $z$ with the
elementary matrices $e_{ij}(1)$ shows that $z=\lambda I_{16}$ for some
$\lambda\in R$.  Commuting with $e_{ij}(a)$ for arbitrary $a\in R$ then gives
$\lambda a=a\lambda$, so $\lambda\in Z(R)^\times=\F_2^\times=\{1\}$.
Thus $Z(\GL_{16}(R))$ is trivial, and $N=1$ when $I=0$.  Finally,
$e_{01}(1)\ne1$, so $H$ is nontrivial.
\end{proof}

\begin{proposition}\label{prop:defect}
Let
\[
 c=e_{45}(1),\qquad \ell=e_{12}(1),\qquad
 d=[\tau c\tau^{-1},\ell]
\]
in $H$.  Then
\[
 c\in C_H(L),\qquad d=e_{02}(q)\ne1,\qquad d^2=1,
 \qquad \normal d_H=H.
\]
\end{proposition}

\begin{proof}
Every elementary generator of the upper-left $4\times4$ corner has both
indices in $\{0,1,2,3\}$.  The elementary commutator relations show that it
commutes with $c=e_{45}(1)$, so $c\in C_H(L)$.

Direct multiplication of the matrices in \eqref{eq:tau} gives
\begin{equation}\label{eq:moved-mark}
 \tau c\tau^{-1}=e_{01}(q)e_{45}(1).
\end{equation}
The second factor commutes with $\ell=e_{12}(1)$, and
$[e_{ij}(a),e_{jk}(b)]=e_{ik}(ab)$ gives
$d=e_{02}(q)$.  This is nontrivial because $q\ne0$.
Since $R$ has characteristic two and $E_{02}^2=0$, one also has
$e_{02}(q)^2=1$.

Let $N=\normal d_H$.  For arbitrary $a\in R$, two further elementary
commutator identities give
\begin{align*}
 [e_{30}(at_1),e_{02}(q)]&=e_{32}(at_1q)\in N,\\
 [e_{32}(at_1q),e_{21}(s_1)]&=e_{31}(at_1qs_1)=e_{31}(a)\in N,
\end{align*}
where the last equality uses \eqref{eq:pq}.  Elementary Weyl matrices
conjugate the root position $(3,1)$ to any ordered pair of distinct indices.
Thus $N$ contains every $e_{ij}(a)$, and these elements generate $H$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:headline}]
The group $H$ is nontrivial and simple by Proposition~\ref{prop:simple}.  The
ring $R$ is finitely generated, and $H$ has property~\textup{(T)} by the
Ershov--Jaikin-Zapirain theorem for elementary groups over finitely generated
rings~\cite{EJZ}; in particular $H$ is finitely generated.

The containment \eqref{eq:compresses-L}, the centrality of $c$ relative to
$L$, and Corollary~\ref{cor:defect-hs} give
$d\in R_{\infty\to2}(H)$.  The subgroup $R_{\infty\to2}(H)$ is normal, and
Proposition~\ref{prop:defect} says that $d$ is an involution which normally
generates $H$.  Theorem~\ref{thm:torsion-bridge} gives
$d\in\Rad_{\mathrm{MF}}(H)$.  Normality of the MF radical now gives
$\Rad_{\mathrm{MF}}(H)=H$.

If $M$ is MF and $f\colon H\to M$ is a homomorphism, compose $f$ with a
faithful homomorphism from $M$ to a norm matrix corona.  Since
$\Rad_{\mathrm{MF}}(H)=H$, this composition and hence $f$ are trivial.  If
$H$ were MF, its identity homomorphism would contradict this conclusion.
\end{proof}

\begin{corollary}\label{cor:exact-fd}
For every field $k$ and every finite-dimensional $k$-vector space $V$, every
homomorphism $H\to\GL(V)$ is trivial.
\end{corollary}

\begin{proof}
Let $\rho\colon H\to\GL(V)$.  The containment
\eqref{eq:compresses-L}, the centrality of $c$ relative to $L$, and
Theorem~\ref{thm:commutant} give $\rho(d)=1$.  Since
$H=\normal d_H$ by Proposition~\ref{prop:defect}, the homomorphism $\rho$ is
trivial.
\end{proof}

In particular, every homomorphism from $H$ to a finite group, a residually
finite group, or a compact Hausdorff group is trivial.  For finite and
residually finite targets this follows by the regular representation of a
finite quotient; for compact targets it follows from Peter--Weyl point
separation~\cite{PeterWeyl}.

\section{Prescribed MF quotients}\label{sec:prescribed-quotients}

Let $A=\langle d\rangle\le H$, where $d$ is the element in
Proposition~\ref{prop:defect}.  For a countable group $Q$, define
\begin{equation}\label{eq:amalgam}
 W_Q=H*_A(Q\times A),
\end{equation}
where $A\to Q\times A$ is the inclusion into the second factor.  The maps
from $H$ and $Q\times A$ to $Q$ that are respectively trivial and the
projection onto $Q$ agree on $A$.  They therefore define an epimorphism
\[
 \pi_Q\colon W_Q\longrightarrow Q.
\]
It is split by the inclusion $Q\to Q\times A\to W_Q$.  The normal-form
theorem for amalgamated free products shows that both vertex maps in
\eqref{eq:amalgam} are injective; in particular, $d\ne1$ in $W_Q$.

\begin{proposition}[universal factorization]\label{prop:universal-factorization}
Let $T$ be a group such that every homomorphism $H\to T$ is trivial.  Then
precomposition with $\pi_Q$ is a bijection
\[
 \operatorname{Hom}(Q,T)\longrightarrow\operatorname{Hom}(W_Q,T).
\]
Furthermore,
\[
 \ker\pi_Q=\normal d_{W_Q}.
\]
\end{proposition}

\begin{proof}
Let $f\colon W_Q\to T$ be a homomorphism.  Its restriction to $H$ is
trivial, so its restriction to the amalgamated subgroup $A$ is trivial.  For
$(q,a)\in Q\times A$, we then have
\[
 f(q,a)=f(q,1)f(1,a)=f(q,1).
\]
Thus the restriction of $f$ to $Q\times A$ factors uniquely through the
projection onto $Q$.  The universal property of the amalgam shows that $f$
factors through $\pi_Q$.  The factorization is unique because $\pi_Q$ is
surjective.

The element $d$ belongs to $\ker\pi_Q$.  Conversely, quotienting $W_Q$ by
$\normal d_{W_Q}$ kills $H$, since $\normal d_H=H$, and kills the second
factor $A$ in $Q\times A$.  The resulting quotient is $Q$, with quotient map
induced by $\pi_Q$.  Hence $\ker\pi_Q=\normal d_{W_Q}$.
\end{proof}

\begin{proposition}[the hidden kernel]\label{prop:hidden-kernel}
Let $K_Q=\ker\pi_Q$ and, using the canonical section $Q\le W_Q$, put
$H_q=qHq^{-1}$ for $q\in Q$.  Then
\[
 K_Q\cong\mathop{\ast}_{A,\ q\in Q}H_q,
 \qquad W_Q\cong K_Q\rtimes Q,
\]
where the first group is the generalized free product of the copies $H_q$
with their common subgroup $A$ identified pointwise.  Conjugation by
$r\in Q$ sends $H_q$ to $H_{rq}$ and fixes $A$.  Moreover,
\[
 \begin{aligned}
  [K_Q,K_Q]&=K_Q,& \Rad_{\mathrm{MF}}(K_Q)&=K_Q,\\
  \normal d_{K_Q}&=K_Q,& Z(K_Q)&=1.
 \end{aligned}
\]
The group $K_Q$ is finitely generated if and only if $Q$ is finite, and its
isomorphism type depends only on the cardinality of $Q$.
\end{proposition}

\begin{proof}
Consider the Bass--Serre tree of the amalgam
$W_Q=H*_A(Q\times A)$.  Since $K_Q$ is normal with quotient $Q$, the quotient
graph by $K_Q$ has one $(Q\times A)$-type vertex, one $H$-type vertex for
each $q\in Q$, and one edge joining the central vertex to each of the other
vertices.  The central vertex stabilizer is
\[
 K_Q\cap(Q\times A)=A,
\]
the stabilizer at the vertex indexed by $q$ is $H_q$, and every edge
stabilizer is $A$.  The quotient graph is a tree.  Since $Q$ commutes with
$A$ inside $Q\times A$, all its edge groups are literally the same subgroup
$A$.  The Bass--Serre normal-form theorem therefore gives the displayed
generalized amalgam.

The canonical section of $\pi_Q$ gives $W_Q\cong K_Q\rtimes Q$, with the
stated action on the vertex groups.  The group $H$ is nonabelian and simple,
hence perfect.  Its conjugates $H_q$ are therefore perfect, and they generate
$K_Q$; consequently $K_Q$ is perfect.

The normal-form theorem also shows that each distinguished factor $H_q$ is
self-normalizing in $K_Q$, and that distinct distinguished factors are not
conjugate.  A central element of $K_Q$ therefore belongs to every $H_q$.  It
is central in each such factor, while $H$ is nonabelian simple and hence
centerless.  Thus $Z(K_Q)=1$.

The common element $d$ normally generates every vertex group $H_q$.
Therefore its normal closure in $K_Q$ contains every $H_q$, and hence
$\normal d_{K_Q}=K_Q$.

If $Q$ is finite, the displayed star amalgam has finitely many finitely
generated vertex groups, so $K_Q$ is finitely generated.  Conversely, every
finite subset of $K_Q$ is contained in the subamalgam supported on finitely
many indices $q\in Q$: each of its reduced normal forms involves only
finitely many vertex groups.  If $Q$ is infinite, a vertex group outside
this finite support contains an element of $H_q\setminus A$, and the normal
form theorem shows that this element is not in the generated subamalgam.
Thus $K_Q$ is not finitely generated.  Finally, the star presentation uses
only the underlying indexing set of $Q$; a bijection between two such sets
induces an isomorphism of their kernels.

Finally, let $f\colon K_Q\to M$ be a homomorphism to an MF group.  Its
restriction to every $H_q$ is trivial by Theorem~\ref{thm:headline}.  Since
these subgroups generate $K_Q$, the map $f$ is trivial.  This proves
$\Rad_{\mathrm{MF}}(K_Q)=K_Q$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:prescribed-quotients}]
Every homomorphism from $H$ to an MF group is trivial by
Theorem~\ref{thm:headline}.  Proposition~\ref{prop:universal-factorization}
therefore gives the asserted bijection for every MF group $M$.  In
particular, every homomorphism from $W_Q$ to a norm matrix corona factors
uniquely through $\pi_Q$.  Intersecting their kernels gives
\[
 \Rad_{\mathrm{MF}}(W_Q)
 =\pi_Q^{-1}\bigl(\Rad_{\mathrm{MF}}(Q)\bigr).
\]
The kernel of $\pi_Q$ belongs to this radical, and it contains the
nonidentity element $d$.  Thus $W_Q$ is non-MF.  If $Q$ is MF, its MF radical
is trivial, and the displayed equality becomes
\[
 \Rad_{\mathrm{MF}}(W_Q)=\ker\pi_Q=\normal d_{W_Q}.
\]

For a normal subgroup $N\trianglelefteq W_Q$, a homomorphism
$f\colon W_Q\to M$ to an MF group kills $N$ precisely when its unique factor
$\bar f\colon Q\to M$ kills $\pi_Q(N)$.  Intersecting the corresponding
kernels gives
\[
 \operatorname{cl}_{\mathrm{MF}}^{W_Q}(N)
 =\pi_Q^{-1}\!\left(
 \operatorname{cl}_{\mathrm{MF}}^Q(\pi_Q(N))
  \right).
\]
The structural assertions about $K_Q$ and the split extension are
Proposition~\ref{prop:hidden-kernel}.
\end{proof}

The same factorization applies to every target group $T$ for which all
homomorphisms $H\to T$ are trivial.  Corollary~\ref{cor:exact-fd} and
Peter--Weyl point separation therefore give the corresponding bijections for
finite-dimensional linear groups over arbitrary fields, residually finite
groups, and compact Hausdorff groups.

\begin{corollary}\label{cor:quotient-criterion}
Let $N\trianglelefteq W_Q$.  Then
\[
 W_Q/N\text{ is MF}
 \quad\Longleftrightarrow\quad
 \ker\pi_Q\le N\ \text{ and }\ Q/\pi_Q(N)\text{ is MF}.
\]
If every quotient of $Q$ is MF, then
\[
 W_Q/N\text{ is MF}
 \quad\Longleftrightarrow\quad d\in N.
\]
In that case the stronger identity
\[
 \operatorname{cl}_{\mathrm{MF}}^{W_Q}(N)
 =N\normal d_{W_Q}
\]
holds for every normal subgroup $N$.
In particular, $W_{\mathbb Z}$ is finitely generated, while its MF radical
$K_{\mathbb Z}$ is perfect, has full MF radical, is not finitely generated,
and is normally generated inside itself by $d$.
\end{corollary}

\begin{proof}
If $W_Q/N$ is MF, its quotient map is a homomorphism from $W_Q$ to an MF
group.  Proposition~\ref{prop:universal-factorization} shows that it factors
through $\pi_Q$, so $\ker\pi_Q\le N$.  The induced isomorphism
\[
 W_Q/N\cong Q/\pi_Q(N)
\]
then proves the first implication.  The same isomorphism proves the converse.
Suppose now that every quotient of $Q$ is MF.  The second condition in the
displayed equivalence is then automatic, and
$\ker\pi_Q=\normal d_{W_Q}$.  Since $N$ is normal, the condition
$\ker\pi_Q\le N$ is equivalent to $d\in N$.
The MF-closure formula in Theorem~\ref{thm:prescribed-quotients} gives
\[
 \operatorname{cl}_{\mathrm{MF}}^{W_Q}(N)
 =\pi_Q^{-1}(\pi_Q(N))
 =N\ker\pi_Q,
\]
which is the asserted closure formula.  Finally, every quotient of
$\mathbb Z$ is MF.  The assertions about $K_{\mathbb Z}$ follow from
Proposition~\ref{prop:hidden-kernel}; finite generation of $W_{\mathbb Z}$
follows from the amalgam presentation because $H$, $A$, and $\mathbb Z$ are
finitely generated.
\end{proof}

\begin{corollary}[universal endomorphisms and outer symmetry]
\label{cor:outer-symmetry}
There is a fixed countable centerless perfect group $K_\infty$ such that
\[
 \Rad_{\mathrm{MF}}(K_\infty)=K_\infty,
 \qquad \normal d_{K_\infty}=K_\infty,
 \qquad \operatorname{Sym}(\mathbb N)
 \hookrightarrow\operatorname{Out}(K_\infty).
\]
Here $d$ is an involution.  Consequently every countable group embeds in
$\operatorname{Out}(K_\infty)$, and every countable monoid embeds in
$\operatorname{End}(K_\infty)$.  The group $K_\infty$ is both non-Hopfian and
non-co-Hopfian.  It also absorbs every nonempty countable amalgamated power
of itself over $\langle d\rangle$.  Moreover, it occurs as the characteristic MF radical in
continuum many pairwise nonisomorphic countable split extensions having
amenable abelian universal MF quotients.
\end{corollary}

\begin{proof}
Take
\[
 K_\infty=\mathop{\ast}_{A,\,n\in\mathbb N}H_n,
\]
with all copies of $A=\langle d\rangle$ identified pointwise.  Its stated
structural properties follow from Proposition~\ref{prop:hidden-kernel}.
Every self-map $f$ of $\mathbb N$ induces an endomorphism $\Phi_f$ by sending
$H_n$ identically onto $H_{f(n)}$.  Normal form shows that $f\mapsto\Phi_f$
is injective and detects injectivity and surjectivity of $f$.  Thus it embeds
the full transformation monoid; left regular actions give every countable
monoid, while a shift and a fold show that $K_\infty$ is respectively
non-co-Hopfian and non-Hopfian.  Flattening the factor indices along
$I\times\mathbb N\cong\mathbb N$ proves the asserted countable
self-absorption.

Every permutation $\sigma$ of $\mathbb N$ induces an automorphism by sending
$H_n$ identically onto $H_{\sigma(n)}$.  If this automorphism were inner,
then $H_n$ and $H_{\sigma(n)}$ would be conjugate for every $n$.  The
normal-form theorem distinguishes the conjugacy classes of the distinguished
factors, so $\sigma(n)=n$ for every $n$.  This proves the injection into the
outer automorphism group.  Cayley's regular action embeds every countable
group in $\operatorname{Sym}(\mathbb N)$.

For an infinite set $S$ of primes, put
\[
 Q_S=\bigoplus_{p\in S}\mathbb Z/p\mathbb Z.
\]
The groups $Q_S$ are countably infinite, residually finite abelian, and hence
MF~\cite{CDE}; their sets of prime element orders recover $S$.
Theorem~\ref{thm:prescribed-quotients}
gives split extensions with kernel and MF radical $K_\infty$ and quotient
$Q_S$.  This kernel is characteristic because the MF radical is fully
invariant: every endomorphism preserves the intersection of the kernels of
all maps to MF groups.  An isomorphism of two extensions would induce an
isomorphism of their MF-radical quotients, hence of the corresponding
$Q_S$.  There are continuum many infinite sets of primes.
\end{proof}
