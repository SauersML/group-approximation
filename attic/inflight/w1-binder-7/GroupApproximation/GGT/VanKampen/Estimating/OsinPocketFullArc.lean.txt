/-!
# Pockets with a full arc are never simple

Osin, arXiv:math/0411039v3, §9, proof of Lemma 9.7(b): the subdiagram `Γ_1` with
`∂Γ_1 = s_1 t_1 s_2 t_2`.  A pocket face set lists its boundary cycle as a side, the reversed arc
`t_1` of the source cell, a side, and the arc `t_2` of the exterior boundary
(`PocketFaceSet.decomposition`).  When `t_1` is the whole boundary of the source cell, the reversed
arc is a closed walk: its first dart starts, and the dart after its last dart starts, at the vertex
of the first dart of `t_1`.  So a walk that holds anything besides the reversed arc passes that
vertex twice.  The same holds for a full exterior arc `t_2`.

This is the wrap case of the pinch (`Estimating/OsinPocketWrapRose.lean`) in every diagram.
Surgery that carries the arcs and the sides of a pocket face set into a copy (edge doublings,
pinch splits) keeps a full arc full, so it never makes such a pocket simple.  A step for the wrap
case has to change the decomposition itself.

* `PocketFullArc.cyclicRel_rotate`: a cyclic chain stays cyclic under rotation.
* `PocketFaceSet.not_simple_of_full_sourceArc`: a full source arc, with a nonempty remainder,
  gives a pocket that is not simple.
* `PocketFaceSet.not_simple_of_full_targetArc`: a full target arc, with a nonempty remainder, gives
  a pocket that is not simple.
* `PocketFullArc.wrapK_not_simple`: the wrap rose's pocket, through the general theorem.

## Manuscript status

Infrastructure for `thm:hull` (Hull's small cancellation theorem, through Osin's Lemma 9.7(b));
certifies no printed sentence on its own.
-/

/-- **A cyclic chain rotates.**  If consecutive entries of `l` are related and the last entry is
related to the first, then the last and first entries of every rotation of `l` by at most its
length are related. -/
theorem cyclicRel_rotate {α : Type*} {R : α → α → Prop} {l : List α} (hchain : l.IsChain R)
    (hclose : ∀ x ∈ l.getLast?, ∀ y ∈ l.head?, R x y) {n : ℕ} (hn : n ≤ l.length) :
    ∀ x ∈ (l.rotate n).getLast?, ∀ y ∈ (l.rotate n).head?, R x y := by
  rw [List.rotate_eq_drop_append_take hn]
  have hsplit : l.take n ++ l.drop n = l := List.take_append_drop n l
  by_cases hB : l.drop n = []
  · have hA : l.take n = l := by
      rw [hB, List.append_nil] at hsplit
      exact hsplit
    rw [hB, List.nil_append, hA]
    exact hclose
  by_cases hA : l.take n = []
  · have hD : l.drop n = l := by
      rw [hA, List.nil_append] at hsplit
      exact hsplit
    rw [hA, List.append_nil, hD]
    exact hclose
  have hchain' : (l.take n ++ l.drop n).IsChain R := by
    rw [hsplit]
    exact hchain
  obtain ⟨-, -, hjoin⟩ := List.isChain_append.mp hchain'
  intro x hx y hy
  have hBA : l.drop n ++ l.take n ≠ [] := fun h => hA (List.append_eq_nil_iff.mp h).2
  rw [List.getLast?_eq_getLast hBA, List.getLast_append_of_ne_nil hBA hA, Option.mem_some_iff]
    at hx
  rw [List.head?_eq_head hBA, List.head_append_of_ne_nil hB, Option.mem_some_iff] at hy
  subst hx hy
  exact hjoin _ (by rw [List.getLast?_eq_getLast hA]; exact Option.mem_some_iff.mpr rfl)
    _ (by rw [List.head?_eq_head hB]; exact Option.mem_some_iff.mpr rfl)

/-- A full arc of a cycle is a rotation of the cycle. -/
theorem darts_eq_rotate_of_full {α : Type*} {cycle : List α} (arc : CyclicArc cycle)
    (hfull : arc.length = cycle.length) : arc.darts = cycle.rotate arc.start.1 := by
  rw [CyclicArc.darts, arc.rotated_eq_rotate, hfull]
  exact List.take_of_length_le (by simp)

/-- The vertex after a dart is the vertex of the next dart of its face. -/
theorem vertexOf_alpha_eq_facePerm (M : CombMap.{v}) (d : M.Dart) :
    M.vertexOf (M.alpha d) = M.vertexOf (M.facePerm d) :=
  (M.vertexOf_sigma (M.alpha d)).symm

/-- The face rotation takes the last dart of a face boundary to its first. -/
theorem faceBoundary_cyclic {M : CombMap.{v}} {f : M.Face} (B : FaceBoundary M f) :
    ∀ x ∈ B.darts.getLast?, ∀ y ∈ B.darts.head?, M.facePerm x = y := by
  intro x hx y hy
  rw [List.getLast?_eq_getLast B.nonempty, Option.mem_some_iff] at hx
  rw [List.head?_eq_head B.nonempty, Option.mem_some_iff] at hy
  subst hx hy
  exact B.closes

/-- A simple closed walk closes: the vertex after its last dart is the vertex of its first. -/
theorem closes_mem {M : CombMap.{v}} {walk : List M.Dart} (hw : IsSimpleClosedWalk M walk) :
    ∀ x ∈ walk.getLast?, ∀ y ∈ walk.head?, M.vertexOf (M.alpha x) = M.vertexOf y := by
  intro x hx y hy
  rw [List.getLast?_eq_getLast hw.ne_nil, Option.mem_some_iff] at hx
  rw [List.head?_eq_head hw.ne_nil, Option.mem_some_iff] at hy
  subst hx hy
  exact hw.closes

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}

/-- The oriented exterior darts form a walk. -/
theorem outerDarts_chain (X : DiscDiagram.{u, w, v} W) :
    (outerDarts X).IsChain
      fun d e => X.toCombMap.vertexOf (X.toCombMap.alpha d) = X.toCombMap.vertexOf e := by
  show ((X.faceBoundary X.outerFace).darts.reverse.map X.toCombMap.alpha).IsChain _
  rw [List.isChain_map, List.isChain_reverse]
  refine (X.faceBoundary X.outerFace).chain.imp ?_
  intro a b hab
  rw [X.toCombMap.alpha_involutive, ← hab, vertexOf_alpha_eq_facePerm]

/-- The oriented exterior darts close up: the vertex after the last is the vertex of the first. -/
theorem outerDarts_cyclic (X : DiscDiagram.{u, w, v} W) :
    ∀ x ∈ (outerDarts X).getLast?, ∀ y ∈ (outerDarts X).head?,
      X.toCombMap.vertexOf (X.toCombMap.alpha x) = X.toCombMap.vertexOf y := by
  intro x hx y hy
  have hB := (X.faceBoundary X.outerFace).nonempty
  simp only [outerDarts, List.getLast?_map, List.getLast?_reverse, List.head?_map,
    List.head?_reverse, List.head?_eq_head hB, List.getLast?_eq_getLast hB, Option.map_some,
    Option.mem_some_iff] at hx hy
  subst hx hy
  rw [X.toCombMap.alpha_involutive, vertexOf_alpha_eq_facePerm, (X.faceBoundary X.outerFace).closes]

end PocketFullArc

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {eps : ℕ} {X : DiscDiagram.{u, w, v} W} {lo hi : ℕ}

/-- On a full source arc the face rotation takes the last dart to the first. -/
theorem facePerm_sourceArc_of_full (K : PocketFaceSet D eps X lo hi)
    (hfull : K.sourceArc.length = (cellDarts X K.source).length) :
    ∀ x ∈ K.sourceArc.darts.getLast?, ∀ y ∈ K.sourceArc.darts.head?,
      X.toCombMap.facePerm x = y := by
  rw [darts_eq_rotate_of_full K.sourceArc hfull]
  exact cyclicRel_rotate (X.faceBoundary (cell X K.source).face).chain
    (faceBoundary_cyclic (X.faceBoundary (cell X K.source).face))
    (Nat.le_of_lt_succ K.sourceArc.start.isLt)

/-- **A pocket with a full source arc is not simple**, unless its cycle is the reversed arc alone.
The first dart of the reversed arc and the dart after its last dart start at the vertex of the
first dart of the arc. -/
theorem not_simple_of_full_sourceArc (K : PocketFaceSet D eps X lo hi)
    (hfull : K.sourceArc.length = (cellDarts X K.source).length)
    (hrest : K.firstSide ++ K.secondSide ++ K.targetArc.darts ≠ []) : ¬ K.Simple := by
  intro hK
  have hS : IsSimpleClosedWalk X.toCombMap K.boundary.cycle := hK
  have hne : K.sourceArc.darts ≠ [] := by
    apply List.ne_nil_of_length_pos
    rw [CyclicArc.darts_length, hfull]
    exact List.length_pos_iff.mpr (X.faceBoundary (cell X K.source).face).nonempty
  have hlastfirst : X.toCombMap.facePerm (K.sourceArc.darts.getLast hne) =
      K.sourceArc.darts.head hne :=
    facePerm_sourceArc_of_full K hfull _
      (by rw [List.getLast?_eq_getLast hne]; exact Option.mem_some_iff.mpr rfl)
      _ (by rw [List.head?_eq_head hne]; exact Option.mem_some_iff.mpr rfl)
  have hv1 : X.toCombMap.vertexOf (X.toCombMap.alpha (K.sourceArc.darts.getLast hne)) =
      X.toCombMap.vertexOf (K.sourceArc.darts.head hne) := by
    rw [vertexOf_alpha_eq_facePerm, hlastfirst]
  have hv2 : X.toCombMap.vertexOf (X.toCombMap.alpha (X.toCombMap.alpha
      (K.sourceArc.darts.head hne))) = X.toCombMap.vertexOf (K.sourceArc.darts.head hne) := by
    rw [X.toCombMap.alpha_involutive]
  have hmem : X.toCombMap.alpha (K.sourceArc.darts.getLast hne) ∈ invDarts X K.sourceArc.darts :=
    List.mem_map_of_mem (List.mem_reverse.mpr (List.getLast_mem hne))
  have E2 : invDarts X K.sourceArc.darts = K.sourceArc.darts.tail.reverse.map X.toCombMap.alpha ++
      [X.toCombMap.alpha (K.sourceArc.darts.head hne)] := by
    conv_lhs => rw [← List.cons_head_tail hne]
    simp [invDarts]
  have hnd := hS.vertex_nodup
  by_cases h2 : K.secondSide ++ K.targetArc.darts = []
  · have hs2 : K.secondSide = [] := (List.append_eq_nil_iff.mp h2).1
    have ht2 : K.targetArc.darts = [] := (List.append_eq_nil_iff.mp h2).2
    have hs1 : K.firstSide ≠ [] := fun h => hrest (by simp [h, hs2, ht2])
    obtain ⟨b, rest, hb⟩ := List.exists_cons_of_ne_nil hs1
    have hc : K.boundary.cycle = (b :: rest) ++ invDarts X K.sourceArc.darts := by
      rw [K.decomposition, hs2, ht2, hb]
      simp
    have hcl := closes_mem hS
    rw [hc, E2] at hcl
    have hvb := hcl (X.toCombMap.alpha (K.sourceArc.darts.head hne)) (by simp) b (by simp)
    rw [hc, List.map_append, List.nodup_append] at hnd
    exact hnd.2.2 _ (List.mem_map_of_mem (by simp)) _ (List.mem_map_of_mem hmem)
      (hvb.symm.trans (hv2.trans hv1.symm))
  · obtain ⟨b, rest, hb⟩ := List.exists_cons_of_ne_nil h2
    have hc : K.boundary.cycle = (K.firstSide ++ invDarts X K.sourceArc.darts) ++ (b :: rest) := by
      rw [K.decomposition,
        List.append_assoc (K.firstSide ++ invDarts X K.sourceArc.darts) K.secondSide
          K.targetArc.darts, hb]
    have hc2 : K.boundary.cycle = (K.firstSide ++ K.sourceArc.darts.tail.reverse.map
        X.toCombMap.alpha) ++ X.toCombMap.alpha (K.sourceArc.darts.head hne) :: b :: rest := by
      rw [hc, E2]
      simp
    have hch := hS.chain
    rw [hc2] at hch
    obtain ⟨-, hxb, -⟩ := List.isChain_append_cons_cons.mp hch
    rw [hc, List.map_append, List.nodup_append] at hnd
    exact hnd.2.2 _ (List.mem_map_of_mem (List.mem_append_right _ hmem)) _
      (List.mem_map_of_mem (by simp)) (hv1.trans (hv2.symm.trans hxb))

/-- **A pocket with a full target arc is not simple**, unless its cycle is that arc alone.  The
vertex after the last dart of the arc is the vertex of its first dart, and the walk closes at the
vertex of its own first dart. -/
theorem not_simple_of_full_targetArc (K : PocketFaceSet D eps X lo hi)
    (hfull : K.targetArc.length = (outerDarts X).length)
    (hrest : K.firstSide ++ invDarts X K.sourceArc.darts ++ K.secondSide ≠ []) : ¬ K.Simple := by
  intro hK
  have hS : IsSimpleClosedWalk X.toCombMap K.boundary.cycle := hK
  have hne : K.targetArc.darts ≠ [] := by
    apply List.ne_nil_of_length_pos
    rw [CyclicArc.darts_length, hfull]
    simp only [outerDarts, List.length_map, List.length_reverse]
    exact List.length_pos_iff.mpr (X.faceBoundary X.outerFace).nonempty
  have hrel : ∀ x ∈ K.targetArc.darts.getLast?, ∀ y ∈ K.targetArc.darts.head?,
      X.toCombMap.vertexOf (X.toCombMap.alpha x) = X.toCombMap.vertexOf y := by
    rw [darts_eq_rotate_of_full K.targetArc hfull]
    exact cyclicRel_rotate (outerDarts_chain X) (outerDarts_cyclic X)
      (Nat.le_of_lt_succ K.targetArc.start.isLt)
  have hv : X.toCombMap.vertexOf (X.toCombMap.alpha (K.targetArc.darts.getLast hne)) =
      X.toCombMap.vertexOf (K.targetArc.darts.head hne) :=
    hrel _ (by rw [List.getLast?_eq_getLast hne]; exact Option.mem_some_iff.mpr rfl)
      _ (by rw [List.head?_eq_head hne]; exact Option.mem_some_iff.mpr rfl)
  obtain ⟨b, rest, hb⟩ := List.exists_cons_of_ne_nil hrest
  have hc : K.boundary.cycle = (b :: rest) ++ K.targetArc.darts := by
    rw [K.decomposition, hb]
  have hcl := closes_mem hS
  rw [hc] at hcl
  have hbr : (b :: rest) ++ K.targetArc.darts ≠ [] := by simp
  have hvb := hcl (K.targetArc.darts.getLast hne)
    (by
      rw [List.getLast?_eq_getLast hbr, List.getLast_append_of_ne_nil hbr hne]
      exact Option.mem_some_iff.mpr rfl)
    b (by simp)
  have hnd := hS.vertex_nodup
  rw [hc, List.map_append, List.nodup_append] at hnd
  exact hnd.2.2 _ (List.mem_map_of_mem (by simp)) _ (List.mem_map_of_mem (List.head_mem hne))
    (hvb.symm.trans hv)

end PocketFaceSet

/-- **The wrap rose's pocket is not simple**, through the full source arc: its first side, second
side and target arc are not all empty. -/
theorem wrapK_not_simple : ¬ OsinPocketWrapRose.wrapK.Simple :=
  PocketFaceSet.not_simple_of_full_sourceArc OsinPocketWrapRose.wrapK
    OsinPocketWrapRose.wrapK_sourceArc_length
    (by
      show ([] : List OsinPocketWrapRose.wrapDiagram.toCombMap.Dart) ++ [] ++
        OsinPocketWrapRose.wrapTargetArc.darts ≠ []
      rw [OsinPocketWrapRose.wrapTargetArc_darts]
      simp)

end PocketFullArc
