/-!
# The section pinch, one step at a time

Osin, arXiv:math/0411039v3, §9, proof of Lemma 9.7(b): the subdiagram `Γ_1` with
`∂Γ_1 = s_1 t_1 s_2 t_2`.  `PocketPinchLabelledSectionStatement` (`Estimating/OsinPocketPinchSection.lean`)
asks for the pinch under Osin's condition, on a least-area diagram with letter labels, for a pocket
in walk order whose source arc is proper.  As for the pinch at a positive side bound, a pocket in
walk order on an unpinched face set is already simple, so the section pinch follows by induction
from one step that lowers the repeated visits.  The step keeps letter labels, walk order and a
proper source arc; least area passes to every O-equivalent copy.

A vertex splitting carries the source arc and the relator words, so it keeps a proper source arc.
Both splits on main package into the step:

* `PocketPinchStepSectionStatement`: one step of the section pinch.
* `pocketPinchLabelledSectionStatement_of_stepSection`: the step gives the section pinch, by strong
  induction on the repeated visits.
* `PocketFaceSet.pinchSplit_source_proper` and `pinchSplitInside_source_proper`: a split keeps a
  proper source arc.
* `PocketFaceSet.exists_pinchStepSection_of_avoids` and `exists_pinchStepSection_of_inside`: a split
  at two darts whose merged faces avoid the face set, or lie inside it, with the turns of the cycle
  kept on one new vertex and two cycle darts on the new vertices, gives the conclusion of the step.

## Manuscript status

Infrastructure for `thm:hull` (Hull's small cancellation theorem, through Osin's Lemma 9.7(b));
certifies no printed sentence on its own.
-/

/-- **One step of the section pinch.**  Beyond thresholds for `ε` and `ρ`, for a family satisfying
Osin's condition, a pinched pocket face set in walk order of a least-area diagram with letter
labels, with a proper source arc, has an O-equivalent copy with letter labels and a pocket face set
in walk order in the same positions, with a proper source arc and fewer repeated visits. -/
def PocketPinchStepSectionStatement : Prop :=
  ∀ {G : Type u} [Group G] {Lambda : Type w} (D : RelGenSet G Lambda),
    (∃ delta : ℕ, Hyperbolic.IsFourPointHyperbolic D.alphabet.carrier delta) →
    ∀ lambda c mu : ℝ, 0 < lambda → lambda ≤ 1 → 0 ≤ c → 0 < mu → mu ≤ 1 / 16 →
      ∃ eps0 : ℕ, ∀ eps : ℕ, eps0 ≤ eps →
        ∃ rho0 : ℕ, 0 < rho0 ∧ ∀ rho : ℕ, rho0 ≤ rho →
          ∀ (W : Set (List (RelLetter G Lambda))),
            OsinCCondition D W eps mu lambda c rho →
              ∀ (X : DiscDiagram.{u, w, v} W) (lo hi : ℕ), X.LeastArea →
                (∀ d, (symmetricLabelAlphabet D).IsLetter (X.label d)) →
                ∀ K : PocketFaceSet D eps X lo hi, K.ClosedWalk →
                  K.sourceArc.length < (cellDarts X K.source).length →
                  ¬Unpinched X.toCombMap K.faces →
                    ∃ (X' : DiscDiagram.{u, w, v} W) (K' : PocketFaceSet D eps X' lo hi),
                      Nonempty (OEquivalentDiscDiagram X X') ∧
                        (∀ d, (symmetricLabelAlphabet D).IsLetter (X'.label d)) ∧
                        K'.ClosedWalk ∧ K'.sourceArc.length < (cellDarts X' K'.source).length ∧
                        K'.repeatedVisits < K.repeatedVisits

/-- **The section pinch, by induction on repeated visits.**  An unpinched pocket in walk order is
simple, with the diagram itself as the copy; a pinched one takes a step at the same thresholds,
least area passes to the copy, and O-equivalences compose. -/
theorem pocketPinchLabelledSectionStatement_of_stepSection
    (h : PocketPinchStepSectionStatement.{u, w, v}) :
    PocketPinchLabelledSectionStatement.{u, w, v} := by
  intro G _ Lambda D hhyp lambda c mu hlambda hlambda1 hc hmu hmu16
  obtain ⟨eps0, heps0⟩ := h D hhyp lambda c mu hlambda hlambda1 hc hmu hmu16
  refine ⟨eps0, fun eps heps => ?_⟩
  obtain ⟨rho0, hrho0, hrho⟩ := heps0 eps heps
  refine ⟨rho0, hrho0, fun rho hrho' W hcondition X₀ lo hi hlea₀ hlabel₀ K₀ hK₀ hprop₀ => ?_⟩
  have hstep := hrho rho hrho' W hcondition
  have H : ∀ (n : ℕ) (X : DiscDiagram.{u, w, v} W), X.LeastArea →
      (∀ d, (symmetricLabelAlphabet D).IsLetter (X.label d)) →
      ∀ K : PocketFaceSet D eps X lo hi, K.ClosedWalk →
        K.sourceArc.length < (cellDarts X K.source).length → K.repeatedVisits = n →
          ∃ (X' : DiscDiagram.{u, w, v} W) (K' : PocketFaceSet D eps X' lo hi),
            Nonempty (OEquivalentDiscDiagram X X') ∧ K'.Simple := by
    intro n
    refine Nat.strong_induction_on n ?_
    intro n ih X hlea hlabel K hK hprop hn
    by_cases hpinch : Unpinched X.toCombMap K.faces
    · exact ⟨X, K, ⟨OEquivalentDiscDiagram.refl X⟩, K.simple_of_closedWalk_of_unpinched hK hpinch⟩
    · obtain ⟨X₁, K₁, ⟨e₁⟩, hlabel₁, hK₁, hprop₁, hlt⟩ :=
        hstep X lo hi hlea hlabel K hK hprop hpinch
      obtain ⟨X₂, K₂, ⟨e₂⟩, hK₂⟩ :=
        ih K₁.repeatedVisits (lt_of_lt_of_eq hlt hn) X₁ (e₁.leastArea hlea) hlabel₁ K₁ hK₁ hprop₁
          rfl
      exact ⟨X₂, K₂, ⟨OEquivalentDiscDiagram.trans e₁ e₂⟩, hK₂⟩
  exact H _ X₀ hlea₀ hlabel₀ K₀ hK₀ hprop₀ rfl

namespace PocketFaceSet

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {eps : ℕ} {X : DiscDiagram.{u, w, v} W} {lo hi : ℕ}

/-- **A split avoiding the face set keeps a proper source arc.** -/
theorem pinchSplit_source_proper (K : PocketFaceSet D eps X lo hi) (I : PinchSplit.Input X)
    (hs : I.Avoids K.faces) (hprop : K.sourceArc.length < (cellDarts X K.source).length) :
    (K.pinchSplit I hs).sourceArc.length <
      (cellDarts I.diagram (K.pinchSplit I hs).source).length := by
  have h1 : (K.pinchSplit I hs).sourceArc.length = K.sourceArc.length :=
    K.sourceArc.mapTo_length id (I.cellDarts_eq K.source)
  have h2 : (cellDarts I.diagram (K.pinchSplit I hs).source).length =
      (cellDarts X K.source).length := by
    rw [pinchSplit_source, I.cellDarts_eq K.source, List.length_map]
  rw [h1, h2]
  exact hprop

/-- **A split inside the face set keeps a proper source arc.** -/
theorem pinchSplitInside_source_proper (K : PocketFaceSet D eps X lo hi) (I : PinchSplit.Input X)
    (hs : I.Inside K.faces) (hprop : K.sourceArc.length < (cellDarts X K.source).length) :
    (K.pinchSplitInside I hs).sourceArc.length <
      (cellDarts I.diagram (K.pinchSplitInside I hs).source).length := by
  have h1 : (K.pinchSplitInside I hs).sourceArc.length = K.sourceArc.length := by
    have h := congrArg List.length (K.pinchSplitInside_sourceArc_darts I hs)
    rwa [CyclicArc.darts_length, CyclicArc.darts_length] at h
  have h2 : (cellDarts I.diagram (K.pinchSplitInside I hs).source).length =
      (cellDarts X K.source).length := by
    rw [pinchSplitInside_source, I.cellDarts_eq K.source, List.length_map]
  rw [h1, h2]
  exact hprop

/-- **One step of the section pinch at a split avoiding the face set.**  The merged faces lie off
the face set, every turn of the cycle runs along vertex rotation without meeting a split dart, and
one cycle dart runs into `x` without passing `y`, another into `y` without passing `x`. -/
theorem exists_pinchStepSection_of_avoids
    (hlabel : ∀ d, (symmetricLabelAlphabet D).IsLetter (X.label d))
    (K : PocketFaceSet D eps X lo hi)
    (hprop : K.sourceArc.length < (cellDarts X K.source).length)
    (I : PinchSplit.Input X) (hs : I.Avoids K.faces)
    (hchain : K.boundary.cycle.IsChain fun d e => I.StretchAvoids (X.toCombMap.alpha d) e)
    (hcloses : I.StretchAvoids
      (X.toCombMap.alpha (K.boundary.cycle.getLast K.boundary.cycle_nonempty))
      (K.boundary.cycle.head K.boundary.cycle_nonempty))
    {e₁ e₂ : X.toCombMap.Dart} (he₁ : e₁ ∈ K.boundary.cycle) (he₂ : e₂ ∈ K.boundary.cycle)
    {n₁ n₂ : ℕ} (hn₁ : (X.toCombMap.sigma ^ n₁) e₁ = I.x)
    (hy₁ : ∀ i ≤ n₁, (X.toCombMap.sigma ^ i) e₁ ≠ I.y)
    (hn₂ : (X.toCombMap.sigma ^ n₂) e₂ = I.y)
    (hx₂ : ∀ i ≤ n₂, (X.toCombMap.sigma ^ i) e₂ ≠ I.x) :
    ∃ (X' : DiscDiagram.{u, w, v} W) (K' : PocketFaceSet D eps X' lo hi),
      Nonempty (OEquivalentDiscDiagram X X') ∧
        (∀ d, (symmetricLabelAlphabet D).IsLetter (X'.label d)) ∧
        K'.ClosedWalk ∧ K'.sourceArc.length < (cellDarts X' K'.source).length ∧
        K'.repeatedVisits < K.repeatedVisits :=
  ⟨I.diagram, K.pinchSplit I hs, ⟨I.oEquivalent⟩,
    I.label_isLetter (symmetricLabelAlphabet D) hlabel,
    K.pinchSplit_closedWalk I hs hchain hcloses, K.pinchSplit_source_proper I hs hprop,
    K.pinchSplit_repeatedVisits_lt_of_stretch I hs he₁ he₂ hn₁ hy₁ hn₂ hx₂⟩

/-- **One step of the section pinch at a split inside the face set.**  The merged faces lie in the
face set, every turn runs along vertex rotation in one direction or the other without meeting a
split dart, and two cycle darts lie on the new vertices of `x` and `y`. -/
theorem exists_pinchStepSection_of_inside
    (hlabel : ∀ d, (symmetricLabelAlphabet D).IsLetter (X.label d))
    (K : PocketFaceSet D eps X lo hi)
    (hprop : K.sourceArc.length < (cellDarts X K.source).length)
    (I : PinchSplit.Input X) (hs : I.Inside K.faces)
    (hchain : K.boundary.cycle.IsChain fun d e =>
      I.StretchAvoids (X.toCombMap.alpha d) e ∨ I.StretchAvoids e (X.toCombMap.alpha d))
    (hcloses :
      I.StretchAvoids (X.toCombMap.alpha (K.boundary.cycle.getLast K.boundary.cycle_nonempty))
          (K.boundary.cycle.head K.boundary.cycle_nonempty) ∨
        I.StretchAvoids (K.boundary.cycle.head K.boundary.cycle_nonempty)
          (X.toCombMap.alpha (K.boundary.cycle.getLast K.boundary.cycle_nonempty)))
    {e₁ e₂ : X.toCombMap.Dart} (he₁ : e₁ ∈ K.boundary.cycle) (he₂ : e₂ ∈ K.boundary.cycle)
    (hx : I.diagram.toCombMap.vertexOf e₁ = I.diagram.toCombMap.vertexOf I.x)
    (hy : I.diagram.toCombMap.vertexOf e₂ = I.diagram.toCombMap.vertexOf I.y) :
    ∃ (X' : DiscDiagram.{u, w, v} W) (K' : PocketFaceSet D eps X' lo hi),
      Nonempty (OEquivalentDiscDiagram X X') ∧
        (∀ d, (symmetricLabelAlphabet D).IsLetter (X'.label d)) ∧
        K'.ClosedWalk ∧ K'.sourceArc.length < (cellDarts X' K'.source).length ∧
        K'.repeatedVisits < K.repeatedVisits :=
  ⟨I.diagram, K.pinchSplitInside I hs, ⟨I.oEquivalent⟩,
    I.label_isLetter (symmetricLabelAlphabet D) hlabel,
    K.pinchSplitInside_closedWalk I hs hchain hcloses, K.pinchSplitInside_source_proper I hs hprop,
    K.pinchSplitInside_repeatedVisits_lt I hs he₁ he₂ hx hy⟩

end PocketFaceSet
