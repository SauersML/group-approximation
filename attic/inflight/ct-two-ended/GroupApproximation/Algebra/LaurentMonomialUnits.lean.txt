/-!
# Units of Laurent polynomials are monomials

`non_mf_groups_exist.tex`, the remark after `cor:dynamic-rank-budget` (tex 1780–1782):

> A unit has Laurent symbols $z^{a_-},z^{a_+}$ at the two ends and acts by these translations on
> the two sufficiently distant tails.

A Laurent symbol is an element of `k[z, z⁻¹] = AddMonoidAlgebra k ℤ`.  Over an integral domain the
units of this ring are the monomials `c z^a`: the top and bottom exponents of a product are the sums
of those of the factors, so a factorization `p q = 1` forces both supports to be single points.
Over `𝔽₂` the coefficient is `1`, which gives the printed symbols `z^a`.

* `coeff_mul_add_of_forall_le`, `coeff_mul_add_of_forall_ge`: the coefficient of a product at the
  sum of two extremal exponents;
* `exists_single_of_mul_eq_one`: if `p q = 1` then `p = c z^a`;
* `exists_single_one_of_mul_eq_one_zmodTwo`: over `𝔽₂`, `p = z^a`.
-/

namespace GroupApproximation

namespace LaurentMonomialUnits

variable {k : Type*} [CommRing k]

/-- At the sum of two top exponents, a product has the product of the top coefficients. -/
theorem coeff_mul_add_of_forall_le {p q : AddMonoidAlgebra k ℤ} {a b : ℤ}
    (ha : ∀ m ∈ p.coeff.support, m ≤ a) (hb : ∀ m ∈ q.coeff.support, m ≤ b) :
    (p * q).coeff (a + b) = p.coeff a * q.coeff b := by
  classical
  rw [AddMonoidAlgebra.coeff_mul]
  simp only [Finsupp.sum]
  rw [Finset.sum_eq_single a]
  · rw [Finset.sum_eq_single b]
    · rw [if_pos rfl]
    · intro m _ hne
      rw [if_neg (by omega)]
    · intro hb'
      rw [if_pos rfl, Finsupp.notMem_support_iff.1 hb', mul_zero]
  · intro m hm hne
    refine Finset.sum_eq_zero fun m₂ hm₂ => ?_
    have h1 := ha m hm
    have h2 := hb m₂ hm₂
    rw [if_neg (by omega)]
  · intro ha'
    refine Finset.sum_eq_zero fun m₂ _ => ?_
    rw [Finsupp.notMem_support_iff.1 ha', zero_mul, ite_self]

/-- At the sum of two bottom exponents, a product has the product of the bottom coefficients. -/
theorem coeff_mul_add_of_forall_ge {p q : AddMonoidAlgebra k ℤ} {a b : ℤ}
    (ha : ∀ m ∈ p.coeff.support, a ≤ m) (hb : ∀ m ∈ q.coeff.support, b ≤ m) :
    (p * q).coeff (a + b) = p.coeff a * q.coeff b := by
  classical
  rw [AddMonoidAlgebra.coeff_mul]
  simp only [Finsupp.sum]
  rw [Finset.sum_eq_single a]
  · rw [Finset.sum_eq_single b]
    · rw [if_pos rfl]
    · intro m _ hne
      rw [if_neg (by omega)]
    · intro hb'
      rw [if_pos rfl, Finsupp.notMem_support_iff.1 hb', mul_zero]
  · intro m hm hne
    refine Finset.sum_eq_zero fun m₂ hm₂ => ?_
    have h1 := ha m hm
    have h2 := hb m₂ hm₂
    rw [if_neg (by omega)]
  · intro ha'
    refine Finset.sum_eq_zero fun m₂ _ => ?_
    rw [Finsupp.notMem_support_iff.1 ha', zero_mul, ite_self]

theorem coeff_support_nonempty_of_mul_eq_one [Nontrivial k] {p q : AddMonoidAlgebra k ℤ}
    (h : p * q = 1) : p.coeff.support.Nonempty ∧ q.coeff.support.Nonempty := by
  classical
  have h1 : (p * q).coeff 0 = 1 := by
    rw [h, AddMonoidAlgebra.one_def, AddMonoidAlgebra.coeff_single, Finsupp.single_eq_same]
  refine ⟨Finsupp.support_nonempty_iff.2 fun hp0 => ?_, Finsupp.support_nonempty_iff.2 fun hq0 => ?_⟩
  · rw [AddMonoidAlgebra.coeff_mul, hp0, Finsupp.sum_zero_index] at h1
    exact zero_ne_one h1
  · rw [AddMonoidAlgebra.coeff_mul] at h1
    simp only [hq0, Finsupp.sum_zero_index, Finsupp.sum_zero] at h1
    exact zero_ne_one h1

/-- **Units of `k[z, z⁻¹]` are monomials** over an integral domain: `p q = 1` forces
`p = c z^a`. -/
theorem exists_single_of_mul_eq_one [IsDomain k] {p q : AddMonoidAlgebra k ℤ} (h : p * q = 1) :
    ∃ (a : ℤ) (c : k), p.coeff = Finsupp.single a c := by
  classical
  obtain ⟨hp, hq⟩ := coeff_support_nonempty_of_mul_eq_one h
  have key : ∀ n, (p * q).coeff n ≠ 0 → n = 0 := fun n hn => by
    rw [h, AddMonoidAlgebra.one_def, AddMonoidAlgebra.coeff_single, Finsupp.single_apply] at hn
    split_ifs at hn with h0
    · exact h0.symm
    · exact absurd rfl hn
  set M := p.coeff.support.max' hp with hM
  set m := p.coeff.support.min' hp with hm
  set M' := q.coeff.support.max' hq with hM'
  set m' := q.coeff.support.min' hq with hm'
  have hpM : p.coeff M ≠ 0 := Finsupp.mem_support_iff.1 (Finset.max'_mem _ _)
  have hpm : p.coeff m ≠ 0 := Finsupp.mem_support_iff.1 (Finset.min'_mem _ _)
  have hqM : q.coeff M' ≠ 0 := Finsupp.mem_support_iff.1 (Finset.max'_mem _ _)
  have hqm : q.coeff m' ≠ 0 := Finsupp.mem_support_iff.1 (Finset.min'_mem _ _)
  have htop : M + M' = 0 := key _ (by
    rw [coeff_mul_add_of_forall_le (fun n hn => Finset.le_max' _ n hn)
      (fun n hn => Finset.le_max' _ n hn)]
    exact mul_ne_zero hpM hqM)
  have hbot : m + m' = 0 := key _ (by
    rw [coeff_mul_add_of_forall_ge (fun n hn => Finset.min'_le _ n hn)
      (fun n hn => Finset.min'_le _ n hn)]
    exact mul_ne_zero hpm hqm)
  have hmM : m ≤ M := Finset.min'_le _ _ (Finset.max'_mem _ _)
  have hmM' : m' ≤ M' := Finset.min'_le _ _ (Finset.max'_mem _ _)
  refine ⟨M, p.coeff M, Finsupp.eq_single_iff.2 ⟨fun n hn => Finset.mem_singleton.2 ?_, rfl⟩⟩
  have h1 : n ≤ M := Finset.le_max' _ n hn
  have h2 : m ≤ n := Finset.min'_le _ n hn
  omega

/-- **Over `𝔽₂` a unit symbol is `z^a`** (tex 1780–1781). -/
theorem exists_single_one_of_mul_eq_one_zmodTwo {p q : AddMonoidAlgebra (ZMod 2) ℤ}
    (h : p * q = 1) : ∃ a : ℤ, p.coeff = Finsupp.single a 1 := by
  classical
  haveI : Nontrivial (ZMod 2) := ⟨⟨0, 1, by decide⟩⟩
  obtain ⟨hp, hq⟩ := coeff_support_nonempty_of_mul_eq_one h
  have key : ∀ n, (p * q).coeff n ≠ 0 → n = 0 := fun n hn => by
    rw [h, AddMonoidAlgebra.one_def, AddMonoidAlgebra.coeff_single, Finsupp.single_apply] at hn
    split_ifs at hn with h0
    · exact h0.symm
    · exact absurd rfl hn
  have hne' : ∀ r s : ZMod 2, r ≠ 0 → s ≠ 0 → r * s ≠ 0 := by decide
  have hne : ∀ {r s : ZMod 2}, r ≠ 0 → s ≠ 0 → r * s ≠ 0 := fun hr hs => hne' _ _ hr hs
  set M := p.coeff.support.max' hp with hM
  set m := p.coeff.support.min' hp with hm
  set M' := q.coeff.support.max' hq with hM'
  set m' := q.coeff.support.min' hq with hm'
  have hpM : p.coeff M ≠ 0 := Finsupp.mem_support_iff.1 (Finset.max'_mem _ _)
  have hpm : p.coeff m ≠ 0 := Finsupp.mem_support_iff.1 (Finset.min'_mem _ _)
  have hqM : q.coeff M' ≠ 0 := Finsupp.mem_support_iff.1 (Finset.max'_mem _ _)
  have hqm : q.coeff m' ≠ 0 := Finsupp.mem_support_iff.1 (Finset.min'_mem _ _)
  have htop : M + M' = 0 := key _ (by
    rw [coeff_mul_add_of_forall_le (fun n hn => Finset.le_max' _ n hn)
      (fun n hn => Finset.le_max' _ n hn)]
    exact hne hpM hqM)
  have hbot : m + m' = 0 := key _ (by
    rw [coeff_mul_add_of_forall_ge (fun n hn => Finset.min'_le _ n hn)
      (fun n hn => Finset.min'_le _ n hn)]
    exact hne hpm hqm)
  have hmM : m ≤ M := Finset.min'_le _ _ (Finset.max'_mem _ _)
  have hmM' : m' ≤ M' := Finset.min'_le _ _ (Finset.max'_mem _ _)
  have hone : p.coeff M = 1 := by
    have hcases : ∀ r : ZMod 2, r ≠ 0 → r = 1 := by decide
    exact hcases _ hpM
  refine ⟨M, Finsupp.eq_single_iff.2 ⟨fun n hn => Finset.mem_singleton.2 ?_, hone⟩⟩
  have h1 : n ≤ M := Finset.le_max' _ n hn
  have h2 : m ≤ n := Finset.min'_le _ n hn
  omega

end LaurentMonomialUnits
