import Mathlib.LinearAlgebra.Dimension.Finrank
import Mathlib.LinearAlgebra.Finsupp.Supported
import Mathlib.LinearAlgebra.Projection
import Mathlib.LinearAlgebra.Quotient.Basic

/-!
# Codimension of coordinate subspaces

`non_mf_groups_exist.tex`, the remark after `cor:dynamic-rank-budget` (tex 1782–1785):

> The tails starting at $-N$ and $N$ span a subspace of codimension $2N-1$, whose image has
> codimension $2N-1+a_+-a_-$.  Invertibility forces $a_+=a_-$

This module supplies the linear algebra of that count for coordinate subspaces of `k^{(α)}`: the span
`k^{(S)} = supported k k S` of the basis vectors indexed by `S` has codimension `#(αᶜS)` when the
complement is finite, and a linear automorphism carrying one such span onto another preserves the
codimension.

* `isCompl_supported_compl`: `k^{(Fᶜ)}` and `k^{(F)}` are complementary;
* `finrank_quotient_supported_compl`: `dim k^{(α)} ⧸ k^{(Fᶜ)} = #F`;
* `card_eq_of_map_supported_compl`: an automorphism mapping `k^{(Fᶜ)}` onto `k^{(F'ᶜ)}` forces
  `#F = #F'`.
-/

namespace GroupApproximation

namespace SupportedCodimension

open Finsupp

variable {α : Type*} (k : Type*) [CommRing k] [Nontrivial k]

/-- The coordinate subspaces `k^{(Fᶜ)}` and `k^{(F)}` are complementary. -/
theorem isCompl_supported_compl (F : Set α) : IsCompl (supported k k Fᶜ) (supported k k F) :=
  IsCompl.of_eq (by rw [← supported_inter, Set.compl_inter_self, supported_empty])
    (by rw [← supported_union, Set.compl_union_self, supported_univ])

/-- **Codimension of a coordinate subspace**: `k^{(α)} ⧸ k^{(Fᶜ)}` has dimension `#F`. -/
theorem finrank_quotient_supported_compl (F : Set α) [Fintype F] :
    Module.finrank k ((α →₀ k) ⧸ supported k k Fᶜ) = Fintype.card F := by
  rw [(Submodule.quotientEquivOfIsCompl _ _ (isCompl_supported_compl k F)).finrank_eq,
    (supportedEquivFinsupp (M := k) (R := k) F).finrank_eq, Module.finrank_finsupp_self]

/-- **Invertibility preserves the codimension** (tex 1784): a linear automorphism carrying
`k^{(Fᶜ)}` onto `k^{(F'ᶜ)}` forces `#F = #F'`. -/
theorem card_eq_of_map_supported_compl {F F' : Set α} [Fintype F] [Fintype F']
    (e : (α →₀ k) ≃ₗ[k] (α →₀ k))
    (he : (supported k k Fᶜ).map (e : (α →₀ k) →ₗ[k] (α →₀ k)) = supported k k F'ᶜ) :
    Fintype.card F = Fintype.card F' := by
  rw [← finrank_quotient_supported_compl k F, ← finrank_quotient_supported_compl k F']
  exact (Submodule.Quotient.equiv _ _ e he).finrank_eq

end SupportedCodimension

end GroupApproximation

open GroupApproximation

#audit_axioms GroupApproximation.SupportedCodimension.finrank_quotient_supported_compl
#audit_axioms GroupApproximation.SupportedCodimension.card_eq_of_map_supported_compl
