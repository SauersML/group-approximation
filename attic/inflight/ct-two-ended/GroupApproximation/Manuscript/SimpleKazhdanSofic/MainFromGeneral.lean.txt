/-!
# `thm:main` from `thm:general`

`simple_kazhdan_sofic_group.tex` at 37551fd93, section "Proof of Theorem 1" (tex l.238–271):

> Let $\Z$ act on $X$ by powers of $T$ [...] Minimality and infiniteness imply that $T$ has no periodic
> points, so the action is topologically free. The ring $R=\LC(X,\F_2)\rtimes_T\Z$ is generated by
> $S=\{1,u,u^{-1}\}\cup\{e_a:a\in A\}$ [...] So $\varphi_\ell(S)$ generates $M_{N_\ell}(\F_2)$, and
> Theorem 2 applies.

* `shiftMulAction S`, `shiftContinuousConstSMul S`: `ℤ` acts on `X` by powers of `T`;
* `ringEquiv S : R S ≃+* ClopenGroupCrossedProduct (Multiplicative ℤ) X F₂`;
* `isMinimal_shift`, `isTopologicallyFree_shift` (no periodic points, ct-involution's `zpow_apply_ne_self`),
  `perfectSpace_carrier`: an infinite minimal subshift is a Cantor set with a minimal topologically free
  action.  Perfectness is implicit in the note: an isolated point with dense orbits would make the compact
  space discrete, hence finite;
* `printedGenerators S = {1} ∪ ringGenerators S` and the piece `PrintedPeriodicMatricialStatement`
  (tex l.244–271; lane ms-core-3);
* `printedSimpleKazhdanSoficMain_of_general`, `printedBlockElementarySimple`, and the residual forms
  `printedSimpleKazhdanGeneral_of_open_pieces` and `printedSimpleKazhdanSoficMain_of_open_pieces`.
-/

section Ring

variable {A : Type} [TopologicalSpace A]

/-- `ℤ` acts on `X` by powers of `T` (tex l.239): `ξ • x = T^ξ x`. -/
noncomputable instance shiftMulAction (S : Subshift A ℤ) : MulAction (Multiplicative ℤ) S.carrier where
  smul ξ x := (subshiftHomeo S ^ toAdd ξ) x
  one_smul x := by
    show (subshiftHomeo S ^ toAdd (1 : Multiplicative ℤ)) x = x
    rw [toAdd_one, zpow_zero, Homeomorph.one_apply]
  mul_smul ξ η x := by
    show (subshiftHomeo S ^ toAdd (ξ * η)) x = (subshiftHomeo S ^ toAdd ξ) ((subshiftHomeo S ^ toAdd η) x)
    rw [toAdd_mul, zpow_add, Homeomorph.mul_apply]

theorem shift_smul_apply (S : Subshift A ℤ) (ξ : Multiplicative ℤ) (x : S.carrier) :
    ξ • x = (subshiftHomeo S ^ toAdd ξ) x :=
  rfl

instance shiftContinuousConstSMul (S : Subshift A ℤ) :
    ContinuousConstSMul (Multiplicative ℤ) S.carrier :=
  ⟨fun ξ => (subshiftHomeo S ^ toAdd ξ).continuous⟩

/-- The identity on `LC(X, F₂)`, from the coefficients of `R` to those of the group crossed product. -/
def coeffEquiv (S : Subshift A ℤ) :
    ClopenCoeff (subshiftHomeo S) (ZMod 2) ≃+* ClopenGroupCoeff (Multiplicative ℤ) S.carrier (ZMod 2) :=
  (ClopenCoeff.of (subshiftHomeo S) (ZMod 2)).symm.trans
    (ClopenGroupCoeff.of (Multiplicative ℤ) S.carrier (ZMod 2))

theorem coeffEquiv_smul (S : Subshift A ℤ) (g : Multiplicative ℤ)
    (a : ClopenCoeff (subshiftHomeo S) (ZMod 2)) :
    (coeffEquiv S).toRingHom (g • a) = g • (coeffEquiv S).toRingHom a :=
  LocallyConstant.ext fun _ => rfl

/-- **`R = LC(X, F₂) ⋊_T ℤ` is the crossed product of the action by powers of `T`** (tex l.239–240). -/
noncomputable def ringEquiv (S : Subshift A ℤ) :
    R S ≃+* ClopenGroupCrossedProduct (Multiplicative ℤ) S.carrier (ZMod 2) :=
  RingEquiv.ofBijective (Pestov91.CrossedProduct.coeffMap (coeffEquiv S).toRingHom (coeffEquiv_smul S))
    ⟨Pestov91.CrossedProduct.coeffMap_injective _ _ (coeffEquiv S).injective,
      Pestov91.CrossedProduct.coeffMap_surjective _ _ (coeffEquiv S).surjective⟩

end Ring

/-! ## Matricial rings along ring isomorphisms -/

section Transport

variable {R R' : Type*} [Ring R] [Ring R'] (e : R ≃+* R') {S : Finset R}

theorem one_mem_map_ringEquiv (h1 : 1 ∈ S) : (1 : R') ∈ S.map e.toEquiv.toEmbedding :=
  Finset.mem_map.2 ⟨1, h1, map_one e⟩

theorem closure_map_ringEquiv (hS : Subring.closure (S : Set R) = ⊤) :
    Subring.closure ((S.map e.toEquiv.toEmbedding : Finset R') : Set R') = ⊤ := by
  rw [Finset.coe_map]
  show Subring.closure (e.toRingHom '' (S : Set R)) = ⊤
  rw [← RingHom.map_closure, hS]
  exact eq_top_iff.2 fun y _ => Subring.mem_map.2 ⟨e.symm y, Subring.mem_top _, e.apply_symm_apply y⟩

theorem isMatricialVia_ringEquiv {N : ℕ → ℕ} {φ : ∀ k, R → Matrix (Fin (N k)) (Fin (N k)) (ZMod 2)}
    (h : IsMatricialVia S N φ) :
    IsMatricialVia (S.map e.toEquiv.toEmbedding) N (fun k x => φ k (e.symm x)) := by
  obtain ⟨hpos, h1, hgen, hmul⟩ := h
  refine ⟨hpos, fun k => ?_, fun k => ?_, fun r s => (hmul (e.symm r) (e.symm s)).mono fun k hk => ?_⟩
  · show φ k (e.symm 1) = 1
    rw [map_one]
    exact h1 k
  · have himg : (fun x => φ k (e.symm x)) '' ((S.map e.toEquiv.toEmbedding : Finset R') : Set R') =
        φ k '' (S : Set R) := by
      rw [Finset.coe_map, Set.image_image]
      congr 1
      funext x
      exact congrArg (φ k) (e.symm_apply_apply x)
    rw [himg]
    exact hgen k
  · show φ k (e.symm (r + s)) = φ k (e.symm r) + φ k (e.symm s) ∧
      φ k (e.symm (r * s)) = φ k (e.symm r) * φ k (e.symm s)
    rw [map_add, map_mul]
    exact hk

end Transport

/-! ## The subshift as a Cantor system -/

section Dynamics

variable {A : Type} [TopologicalSpace A] [DiscreteTopology A] [Finite A]

theorem isMinimal_shift (S : Subshift A ℤ) (hmin : IsMinimal S) :
    MulAction.IsMinimal (Multiplicative ℤ) S.carrier :=
  ⟨fun z => (minimalSubshift_dense_orbits S hmin z).mono (by
    rintro _ ⟨j, rfl⟩
    exact ⟨ofAdd j, rfl⟩)⟩

/-- **No periodic points, so the action is topologically free** (tex l.240–242). -/
theorem isTopologicallyFree_shift (S : Subshift A ℤ) (hinf : Infinite S.carrier) (hmin : IsMinimal S) :
    IsTopologicallyFree (Multiplicative ℤ) S.carrier := by
  intro ξ hξ U _ hU
  obtain ⟨z, hz⟩ := hU
  have hj : toAdd ξ ≠ 0 := fun h => hξ (by rw [← ofAdd_toAdd ξ, h, ofAdd_zero])
  haveI := hinf
  exact ⟨z, hz, zpow_apply_ne_self (subshiftHomeo S) (fun y => minimalSubshift_dense_orbits S hmin y) hj z⟩

/-- **An infinite minimal subshift has no isolated points.** -/
theorem perfectSpace_carrier (S : Subshift A ℤ) (hinf : Infinite S.carrier) (hmin : IsMinimal S) :
    PerfectSpace S.carrier := by
  refine ⟨preperfect_iff_nhds.2 fun x _ U hU => ?_⟩
  by_contra hno
  push_neg at hno
  obtain ⟨t, htU, htopen, hxt⟩ := mem_nhds_iff.1 hU
  have hsingle : IsOpen ({x} : Set S.carrier) := by
    have heq : t = {x} := Set.Subset.antisymm (fun y hy => hno y ⟨htU hy, Set.mem_univ y⟩)
      (Set.singleton_subset_iff.2 hxt)
    rwa [heq] at htopen
  haveI : DiscreteTopology S.carrier := discreteTopology_iff_isOpen_singleton.2 fun y => by
    obtain ⟨_, ⟨j, rfl⟩, hjx⟩ :=
      (minimalSubshift_dense_orbits S hmin y).exists_mem_open hsingle ⟨x, rfl⟩
    have hy : y = (subshiftHomeo S ^ (-j)) x := by
      rw [← Set.mem_singleton_iff.1 hjx, ← Homeomorph.mul_apply, ← zpow_add, neg_add_cancel, zpow_zero,
        Homeomorph.one_apply]
    rw [hy, ← Set.image_singleton]
    exact (subshiftHomeo S ^ (-j)).isOpenMap _ hsingle
  haveI := compactSpace_carrier S
  haveI := hinf
  haveI : Finite S.carrier := finite_of_compact_of_discrete
  exact not_finite S.carrier

end Dynamics

theorem fg_multiplicative_int : Group.FG (Multiplicative ℤ) := by
  refine Group.fg_def.mpr ⟨{ofAdd 1}, eq_top_iff.2 fun g _ => ?_⟩
  have hg : g = ofAdd (1 : ℤ) ^ toAdd g := by rw [← Int.ofAdd_mul, one_mul, ofAdd_toAdd]
  rw [hg]
  exact Subgroup.zpow_mem _ (Subgroup.subset_closure (Finset.mem_coe.2 (Finset.mem_singleton_self _))) _

/-! ## The periodic approximation piece -/

/-- The printed generators `S = {1, u, u⁻¹} ∪ {e_a : a ∈ A}` (tex l.243–244). -/
noncomputable def printedGenerators {A : Type} [TopologicalSpace A] [DiscreteTopology A] [DecidableEq A]
    [Finite A] (S : Subshift A ℤ) : Finset (R S) := by
  classical
  exact insert 1 (ringGenerators_finite S).toFinset

theorem one_mem_printedGenerators {A : Type} [TopologicalSpace A] [DiscreteTopology A] [DecidableEq A]
    [Finite A] (S : Subshift A ℤ) : (1 : R S) ∈ printedGenerators S := by
  classical
  unfold printedGenerators
  exact Finset.mem_insert_self _ _

theorem closure_printedGenerators {A : Type} [TopologicalSpace A] [DiscreteTopology A] [DecidableEq A]
    [Finite A] (S : Subshift A ℤ) : Subring.closure (printedGenerators S : Set (R S)) = ⊤ := by
  classical
  refine eq_top_iff.2 ((subring_closure_eq_top S).symm.le.trans (Subring.closure_mono fun x hx => ?_))
  unfold printedGenerators
  rw [Finset.coe_insert, Set.Finite.coe_toFinset]
  exact Set.mem_insert_of_mem _ hx

/-- **Periodic approximations make `R` matricial** (tex l.244–271; lane ms-core-3). -/
def PrintedPeriodicMatricialStatement : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [DecidableEq A] [Finite A] (S : Subshift A ℤ),
    Infinite S.carrier → IsMinimal S →
    ∃ (N : ℕ → ℕ) (φ : ∀ k, R S → Matrix (Fin (N k)) (Fin (N k)) (ZMod 2)),
      IsMatricialVia (printedGenerators S) N φ

/-! ## Derivation -/

/-- **`thm:main` from `thm:general`** (section "Proof of Theorem 1", tex l.238–271). -/
theorem printedSimpleKazhdanSoficMain_of_general (hgen : PrintedSimpleKazhdanGeneral)
    (hper : PrintedPeriodicMatricialStatement) : PrintedSimpleKazhdanSoficMain := by
  intro A _ _ _ S hinf hmin n hn
  classical
  haveI := compactSpace_carrier S
  haveI : Nonempty S.carrier := hinf.nonempty
  haveI := perfectSpace_carrier S hinf hmin
  haveI : Group.FG (Multiplicative ℤ) := fg_multiplicative_int
  obtain ⟨N, φ, hφ⟩ := hper A S hinf hmin
  have hpos : ∀ k, 0 < N k := And.left hφ
  obtain ⟨-, -, hsimple', -, hlim', hsimpleM, -, hexp'⟩ :=
    hgen (Multiplicative ℤ) S.carrier (isMinimal_shift S hmin) (isTopologicallyFree_shift S hinf hmin) _
      (one_mem_map_ringEquiv (ringEquiv S) (one_mem_printedGenerators S))
      (closure_map_ringEquiv (ringEquiv S) (closure_printedGenerators S)) N _
      (isMatricialVia_ringEquiv (ringEquiv S) hφ) n hn
  set eG := (elementaryGroupEquivOfRingEquiv n (ringEquiv S)).symm with heG
  set eSL := fun ℓ => elementaryBlockEquivSL n hn (N ℓ) (hpos ℓ) with heSL
  have hInf : Infinite ↥(elementaryGroup (Fin n) (R S)) := infinite_elementaryGroup S hinf hn
  have hT : HasKazhdanPropertyT.{0, 0} ↥(elementaryGroup (Fin n) (R S)) :=
    Pestov91.elementary_hasKazhdanPropertyT (printedSubshiftRingFGStatement A S) hn
  have hsimple : IsSimpleGroup ↥(elementaryGroup (Fin n) (R S)) := by
    haveI := hsimple'
    haveI : Nontrivial ↥(elementaryGroup (Fin n) (R S)) := eG.injective.nontrivial
    exact IsSimpleGroup.isSimpleGroup_of_surjective eG.toMonoidHom eG.surjective
  have hlimSL := (hlim'.map_source eG).map_target eSL
  have hexpSL := hexp'.map_target eSL
  have hML : PrintedMarkedLimitExpanders S n := by
    refine ⟨_, _, N, _, fun ℓ => ?_, hlimSL, hexpSL⟩
    haveI := hsimpleM ℓ
    haveI : Nontrivial (Matrix.SpecialLinearGroup (Fin n × Fin (N ℓ)) (ZMod 2)) :=
      (eSL ℓ).injective.nontrivial
    exact IsSimpleGroup.isSimpleGroup_of_surjective (eSL ℓ).toMonoidHom (eSL ℓ).surjective
  have hLEF : IsLEF ↥(elementaryGroup (Fin n) (R S)) :=
    printedMarkedLimitLEFStatement _ _ _ _ _ (And.left (And.right hexpSL)) hlimSL
  exact ⟨hInf, KazhdanFiniteGeneration.fg_of_hasKazhdanPropertyT _ hT, hsimple, hT, hML, hLEF,
    isSofic_of_isLEF hLEF, Pestov91.isHyperlinear_of_isLEF hLEF⟩

theorem printedSimpleKazhdanGeneral_of_open_pieces (hsimple : PrintedGeneralSimplicityStatement)
    (hmarked : PrintedMatricialMarkedLimitStatement) : PrintedSimpleKazhdanGeneral :=

/-- **`thm:main` from the open pieces of `thm:general` and the periodic approximation.** -/

    (hmarked : PrintedMatricialMarkedLimitStatement) (hper : PrintedPeriodicMatricialStatement) :
    PrintedSimpleKazhdanSoficMain :=
  printedSimpleKazhdanSoficMain_of_general (printedSimpleKazhdanGeneral_of_open_pieces hsimple hmarked)
    hper
