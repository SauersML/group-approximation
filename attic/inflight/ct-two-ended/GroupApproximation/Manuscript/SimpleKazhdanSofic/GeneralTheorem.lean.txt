/-!
# `thm:general` of `simple_kazhdan_sofic_group.tex`: statement and assembly

`simple_kazhdan_sofic_group.tex` at origin/main 37551fd93, Theorem `thm:general` (tex l.109–127):

> Let a finitely generated group $\Lambda$ act minimally and topologically freely on a Cantor set $Z$,
> and let $R=\LC(Z,\F_2)\rtimes\Lambda$ be generated by a finite set $S\ni1$. Suppose that $R$ is
> matricial, that is, there are maps $\varphi_k\colon R\to M_{N_k}(\F_2)$ with $\varphi_k(1)=I$ such that
> $\varphi_k(S)$ generates the ring $M_{N_k}(\F_2)$, and for all $r,s\in R$ we have
> $\varphi_k(r+s)=\varphi_k(r)+\varphi_k(s)$ and $\varphi_k(rs)=\varphi_k(r)\varphi_k(s)$ for large $k$.
> Then for every $n\ge3$, $\EL_n(R)$ is an infinite, finitely generated, simple Kazhdan group. Marked by
> the $e_{ij}(s)$ with $s\in S$, it is the limit of the finite simple groups $\SL_{nN_k}(\F_2)$ marked
> by the $e_{ij}(\varphi_k(s))$, whose Cayley graphs form a family of expanders.

"An action on a Cantor set is topologically free if no nontrivial element fixes a nonempty open set"
(tex l.105–106).  A Cantor set is a nonempty compact metrizable totally disconnected perfect space.

This module is additive.  `thm:main` keeps its statement and assembly in `SimpleKazhdanSofic.MainAssembly`.

## Definitions

* `ActionCoeff α k`, `ActionCrossedProduct α k`: `LC(Z, k) ⋊ Λ` for `α : Λ →* (Z ≃ₜ Z)`, with
  `ξ • f = f ∘ α(ξ)⁻¹`, so `(f u_ξ)(f' u_η) = f (f' ∘ ξ⁻¹) u_{ξη}` (tex l.135–137);
* `IsMinimalAction α`, `IsTopologicallyFree α`;
* `IsMatricialVia S N φ`: `R` is matricial via `φ_k : R → M_{N_k}(F₂)`;
* `elementaryMarking n S`, `matricialMarking n S N φ k`: the markings by the `e_ij(s)` and the `e_ij(φ_k(s))`;
* `PrintedSimpleKazhdanGeneral`.  The finite groups are `EL_n(M_{N_k}(F₂))`.  Block flattening carries
  them onto `SL_{nN_k}(F₂)` (`printedBlockElementarySpecialLinear`, lane ms-core-3), and the statement
  records this.

## Pieces (named propositions, with the owning lanes)

* `PrintedElementaryMarkingGeneration`: the `e_ij(s)`, `s ∈ S`, generate `EL_n(R)` (tex l.144–145; sk-lef-action);
* `PrintedGeneralElementaryInfinite`: `EL_n(R)` is infinite (tex l.149–150; sk-lef-action);
* `PrintedGeneralSimplicityStatement`: `EL_n(R)` is simple (section "Simplicity", tex l.151–210;
  sk-lef-simplicity and skf-consequences);
* `PrintedMatricialMarkedLimitStatement`: the marked limit, for a matricial ring with `EL_n(R)` simple
  (section "Finite simple models", tex l.211–233; ms-core-3);
* `PrintedBlockElementarySimple`: `EL_n(M_N(F₂))` is simple for `N ≥ 1` (tex l.221–224; ms-core-3);
* `PrintedMatricialExpandersStatement`: the markings `e_ij(φ_k(s))` give expanders once the orders tend to
  infinity (tex l.235–236; oa-expanders).

Property (T) is Ershov–Jaikin-Zapirain, proved in this repository.  Finite generation follows from (T).
"$N_k\to\infty$ as $G$ is infinite" (tex l.234–235) is `tendsto_card_of_isMarkedLimit`.
`printedSimpleKazhdanGeneral_of_pieces` assembles the theorem.
-/

/-! ## The crossed product of an action -/

/-- The coefficient ring `LC(Z, k)` of `LC(Z, k) ⋊ Λ`, carrying the action of `Λ` through `α`. -/
def ActionCoeff {Λ Z : Type*} [Group Λ] [TopologicalSpace Z] (_α : Λ →* (Z ≃ₜ Z)) (k : Type*) :
    Type _ :=
  LocallyConstant Z k

namespace ActionCoeff

variable {Λ Z : Type*} [Group Λ] [TopologicalSpace Z] (α : Λ →* (Z ≃ₜ Z)) (k : Type*) [Ring k]

instance instRing : Ring (ActionCoeff α k) :=
  inferInstanceAs (Ring (LocallyConstant Z k))

/-- The identity map `LC(Z, k) ≃+* ActionCoeff α k`. -/
def of : LocallyConstant Z k ≃+* ActionCoeff α k where
  toFun f := f
  invFun f := f
  left_inv _ := rfl
  right_inv _ := rfl
  map_mul' _ _ := rfl
  map_add' _ _ := rfl

/-- `Λ` acts by `ξ • f = f ∘ α(ξ)⁻¹`. -/
noncomputable instance instMulSemiringAction : MulSemiringAction Λ (ActionCoeff α k) where
  smul ξ f := of α k (LocallyConstant.comap ⟨⇑(α ξ⁻¹), (α ξ⁻¹).continuous⟩ ((of α k).symm f))
  one_smul f := LocallyConstant.ext fun z => by
    show (of α k).symm f ((α (1 : Λ)⁻¹) z) = (of α k).symm f z
    rw [inv_one, map_one, Homeomorph.one_apply]
  mul_smul ξ η f := LocallyConstant.ext fun z => by
    show (of α k).symm f ((α (ξ * η)⁻¹) z) = (of α k).symm f ((α η⁻¹) ((α ξ⁻¹) z))
    rw [mul_inv_rev, map_mul, Homeomorph.mul_apply]
  smul_zero _ := LocallyConstant.ext fun _ => rfl
  smul_add _ _ _ := LocallyConstant.ext fun _ => rfl
  smul_one _ := LocallyConstant.ext fun _ => rfl
  smul_mul _ _ _ := LocallyConstant.ext fun _ => rfl

theorem smul_of (ξ : Λ) (f : LocallyConstant Z k) :
    ξ • of α k f = of α k (LocallyConstant.comap ⟨⇑(α ξ⁻¹), (α ξ⁻¹).continuous⟩ f) :=
  rfl

end ActionCoeff

/-- **The crossed product** `LC(Z, k) ⋊ Λ` of an action by homeomorphisms (tex l.135–137). -/
abbrev ActionCrossedProduct {Λ Z : Type*} [Group Λ] [TopologicalSpace Z] (α : Λ →* (Z ≃ₜ Z))
    (k : Type*) [Ring k] : Type _ :=
  SkewMonoidAlgebra (ActionCoeff α k) Λ

section Dynamics

variable {Λ Z : Type*} [Group Λ] [TopologicalSpace Z]

/-- The action is minimal: every orbit is dense. -/
def IsMinimalAction (α : Λ →* (Z ≃ₜ Z)) : Prop :=
  ∀ z : Z, Dense (Set.range fun ξ : Λ => α ξ z)

/-- The action is topologically free: no nontrivial element fixes a nonempty open set (tex l.105–106). -/
def IsTopologicallyFree (α : Λ →* (Z ≃ₜ Z)) : Prop :=
  ∀ ξ : Λ, ξ ≠ 1 → ∀ U : Set Z, IsOpen U → U.Nonempty → ∃ z ∈ U, α ξ z ≠ z

end Dynamics

/-! ## Matricial rings and markings -/

/-- **`R` is matricial via `φ_k`** (tex l.113–117), with `N_k ≥ 1`.  The statement speaks of the finite simple
groups `SL_{nN_k}(F₂)`, so `M_{N_k}(F₂)` is a nonzero matrix algebra.  Size-zero models satisfy every other
axiom, and infinitely many of them would break the marked convergence (lane ms-core-3). -/
def IsMatricialVia {R : Type*} [Ring R] (S : Finset R) (N : ℕ → ℕ)
    (φ : ∀ k, R → Matrix (Fin (N k)) (Fin (N k)) (ZMod 2)) : Prop :=
  (∀ k, 0 < N k) ∧ (∀ k, φ k 1 = 1) ∧ (∀ k, Subring.closure (φ k '' (S : Set R)) = ⊤) ∧
    ∀ r s : R, ∀ᶠ k in atTop, φ k (r + s) = φ k r + φ k s ∧ φ k (r * s) = φ k r * φ k s

/-- The markings are indexed by pairs `i ≠ j` and elements `s ∈ S`. -/
abbrev MarkingIndex (n : ℕ) {R : Type*} (S : Finset R) : Type _ :=
  {p : Fin n × Fin n // p.1 ≠ p.2} × ↥S

/-- `EL_n(R)` marked by the `e_ij(s)`, `s ∈ S` (tex l.119). -/
noncomputable def elementaryMarking (n : ℕ) {R : Type*} [Ring R] (S : Finset R) :
    MarkingIndex n S → ↥(elementaryGroup (Fin n) R) :=
  fun p => ⟨elementaryUnit p.1.1.1 p.1.1.2 p.1.2 (p.2 : R), elementaryUnit_mem _ _ _ _⟩

/-- `EL_n(M_{N_k}(F₂))` marked by the `e_ij(φ_k(s))`, `s ∈ S` (tex l.120). -/
noncomputable def matricialMarking (n : ℕ) {R : Type*} [Ring R] (S : Finset R) (N : ℕ → ℕ)
    (φ : ∀ k, R → Matrix (Fin (N k)) (Fin (N k)) (ZMod 2)) (k : ℕ) :
    MarkingIndex n S → ↥(elementaryGroup (Fin n) (Matrix (Fin (N k)) (Fin (N k)) (ZMod 2))) :=
  fun p => ⟨elementaryUnit p.1.1.1 p.1.1.2 p.1.2 (φ k (p.2 : R)), elementaryUnit_mem _ _ _ _⟩

/-! ## The statement -/

/-- **Theorem `thm:general` as printed** (tex l.109–127), for every `n ≥ 3`. -/
def PrintedSimpleKazhdanGeneral : Prop :=
  ∀ (Λ : Type) [Group Λ] [Group.FG Λ] (Z : Type) [TopologicalSpace Z] [CompactSpace Z]
    [TopologicalSpace.MetrizableSpace Z] [TotallyDisconnectedSpace Z] [PerfectSpace Z] [Nonempty Z]
    (α : Λ →* (Z ≃ₜ Z)), IsMinimalAction α → IsTopologicallyFree α →
    ∀ S : Finset (ActionCrossedProduct α (ZMod 2)), 1 ∈ S →
      Subring.closure (S : Set (ActionCrossedProduct α (ZMod 2))) = ⊤ →
    ∀ (N : ℕ → ℕ) (φ : ∀ k, ActionCrossedProduct α (ZMod 2) → Matrix (Fin (N k)) (Fin (N k)) (ZMod 2)),
      IsMatricialVia S N φ → ∀ n : ℕ, 3 ≤ n →
        Infinite ↥(elementaryGroup (Fin n) (ActionCrossedProduct α (ZMod 2))) ∧
        Group.FG ↥(elementaryGroup (Fin n) (ActionCrossedProduct α (ZMod 2))) ∧
        IsSimpleGroup ↥(elementaryGroup (Fin n) (ActionCrossedProduct α (ZMod 2))) ∧
        HasKazhdanPropertyT.{0, 0} ↥(elementaryGroup (Fin n) (ActionCrossedProduct α (ZMod 2))) ∧
        IsMarkedLimit (elementaryMarking n S) (matricialMarking n S N φ) ∧
        (∀ k, IsSimpleGroup ↥(elementaryGroup (Fin n) (Matrix (Fin (N k)) (Fin (N k)) (ZMod 2)))) ∧
        (∀ k,
          (elementaryGroup (Fin n) (Matrix (Fin (N k)) (Fin (N k)) (ZMod 2))).map
              (elementaryBlockUnitEquiv (ι := Fin n) (κ := Fin (N k)) (R := ZMod 2)).toMonoidHom =
            (AlgebraicK.detUnits (ι := Fin n × Fin (N k)) (k := ZMod 2)).ker) ∧
        IsExpanderFamily (matricialMarking n S N φ)

/-! ## Pieces -/

/-- **Generation** (tex l.144–145): "So the $e_{ij}(s)$ with $s\in S$ generate $G$." -/
def PrintedElementaryMarkingGeneration : Prop :=
  ∀ (R : Type) [Ring R] (S : Finset R), 1 ∈ S → Subring.closure (S : Set R) = ⊤ →
    ∀ n : ℕ, 3 ≤ n → Subgroup.closure (Set.range (elementaryMarking n S)) = ⊤

/-- **Infinite** (tex l.149–150): "it is infinite because $e_{12}(\LC(Z,\F_2))$ is infinite". -/
def PrintedGeneralElementaryInfinite : Prop :=
  ∀ (Λ : Type) [Group Λ] (Z : Type) [TopologicalSpace Z] [CompactSpace Z]
    [TopologicalSpace.MetrizableSpace Z] [TotallyDisconnectedSpace Z] [PerfectSpace Z] [Nonempty Z]
    (α : Λ →* (Z ≃ₜ Z)) (n : ℕ), 3 ≤ n →
      Infinite ↥(elementaryGroup (Fin n) (ActionCrossedProduct α (ZMod 2)))

/-- **Simplicity** (section "Simplicity", tex l.151–210). -/
def PrintedGeneralSimplicityStatement : Prop :=
  ∀ (Λ : Type) [Group Λ] [Group.FG Λ] (Z : Type) [TopologicalSpace Z] [CompactSpace Z]
    [TopologicalSpace.MetrizableSpace Z] [TotallyDisconnectedSpace Z] [PerfectSpace Z] [Nonempty Z]
    (α : Λ →* (Z ≃ₜ Z)), IsMinimalAction α → IsTopologicallyFree α → ∀ n : ℕ, 3 ≤ n →
      IsSimpleGroup ↥(elementaryGroup (Fin n) (ActionCrossedProduct α (ZMod 2)))

/-- **The marked limit** (section "Finite simple models", tex l.211–233), for any unital ring generated
by a finite `S ∋ 1` and matricial via `φ_k`, when `EL_n(R)` is simple. -/
def PrintedMatricialMarkedLimitStatement : Prop :=
  ∀ (R : Type) [Ring R] (S : Finset R), 1 ∈ S → Subring.closure (S : Set R) = ⊤ →
    ∀ (N : ℕ → ℕ) (φ : ∀ k, R → Matrix (Fin (N k)) (Fin (N k)) (ZMod 2)), IsMatricialVia S N φ →
    ∀ n : ℕ, 3 ≤ n → IsSimpleGroup ↥(elementaryGroup (Fin n) R) →
      IsMarkedLimit (elementaryMarking n S) (matricialMarking n S N φ)

/-- **`EL_n(M_N(F₂)) = SL_{nN}(F₂)` is simple** for `N ≥ 1` (tex l.221–224). -/
def PrintedBlockElementarySimple : Prop :=
  ∀ n : ℕ, 3 ≤ n → ∀ N : ℕ, 0 < N →
    IsSimpleGroup ↥(elementaryGroup (Fin n) (Matrix (Fin N) (Fin N) (ZMod 2)))

/-- **Expanders** (tex l.235–236): "Since $\EL_3(F)$ has property (T), its finite quotients form a
family of expanders." -/
def PrintedMatricialExpandersStatement : Prop :=
  ∀ (R : Type) [Ring R] (S : Finset R), 1 ∈ S → Subring.closure (S : Set R) = ⊤ →
    ∀ (N : ℕ → ℕ) (φ : ∀ k, R → Matrix (Fin (N k)) (Fin (N k)) (ZMod 2)), IsMatricialVia S N φ →
    ∀ n : ℕ, 3 ≤ n →
      Tendsto (fun k => Nat.card ↥(elementaryGroup (Fin n) (Matrix (Fin (N k)) (Fin (N k)) (ZMod 2))))
        atTop atTop →
      IsExpanderFamily (matricialMarking n S N φ)

/-! ## Marked limits: orders and nontriviality -/

section MarkedLimits

variable {ι G : Type*} [Group G] {s : ι → G} {H : ℕ → Type*} [∀ ℓ, Group (H ℓ)] {σ : ∀ ℓ, ι → H ℓ}

theorem exists_word_of_isMarkedLimit (hlim : IsMarkedLimit s σ) (g : G) :
    ∃ w : FreeGroup ι, FreeGroup.lift s w = g := by
  obtain ⟨hs, -, -⟩ := hlim
  have hg : g ∈ (FreeGroup.lift s).range := by
    rw [FreeGroup.range_lift_eq_closure, hs]
    exact Subgroup.mem_top g
  exact MonoidHom.mem_range.mp hg

/-- **The orders tend to infinity** (tex l.234–235: "and $N_k\to\infty$ as $G$ is infinite"). -/
theorem tendsto_card_of_isMarkedLimit [Infinite G] (hfin : ∀ ℓ, Finite (H ℓ))
    (hlim : IsMarkedLimit s σ) : Tendsto (fun ℓ => Nat.card (H ℓ)) atTop atTop := by
  classical
  choose word hword using exists_word_of_isMarkedLimit hlim
  obtain ⟨-, -, hw⟩ := hlim
  refine Filter.tendsto_atTop.2 fun B => ?_
  obtain ⟨F, hF⟩ := Infinite.exists_subset_card_eq G B
  have hsep : ∀ᶠ ℓ in atTop, ∀ x ∈ F, ∀ y ∈ F, x ≠ y →
      FreeGroup.lift (σ ℓ) (word x * (word y)⁻¹) ≠ 1 := by
    refine (eventually_all_finset F).2 fun x _ => (eventually_all_finset F).2 fun y _ => ?_
    by_cases hxy : x = y
    · exact Eventually.of_forall fun _ hne => absurd hxy hne
    · refine (hw (word x * (word y)⁻¹)).mono fun ℓ hℓ _ h1 => hxy ?_
      have h2 := hℓ.mp h1
      rw [map_mul, map_inv, hword, hword] at h2
      exact mul_inv_eq_one.mp h2
  refine hsep.mono fun ℓ hℓ => ?_
  letI : Fintype (H ℓ) := @Fintype.ofFinite (H ℓ) (hfin ℓ)
  have hinj : Function.Injective fun x : ↥F => FreeGroup.lift (σ ℓ) (word x.1) := by
    intro x y hxy
    by_contra hne
    have hxy' : FreeGroup.lift (σ ℓ) (word x.1) = FreeGroup.lift (σ ℓ) (word y.1) := hxy
    have h1 := hℓ x.1 x.2 y.1 y.2 fun h => hne (Subtype.ext h)
    rw [map_mul, map_inv] at h1
    exact h1 (mul_inv_eq_one.mpr hxy')
  calc B = Fintype.card ↥F := by rw [Fintype.card_coe, hF]
    _ ≤ Fintype.card (H ℓ) := Fintype.card_le_of_injective _ hinj
    _ = Nat.card (H ℓ) := Nat.card_eq_fintype_card.symm

end MarkedLimits

theorem finite_elementaryGroup_matrix (n N : ℕ) :
    Finite ↥(elementaryGroup (Fin n) (Matrix (Fin N) (Fin N) (ZMod 2))) := by
  haveI : Finite (Matrix (Fin n) (Fin n) (Matrix (Fin N) (Fin N) (ZMod 2))) :=
    inferInstanceAs (Finite (Fin n → Fin n → Fin N → Fin N → ZMod 2))
  haveI : Finite (Matrix (Fin n) (Fin n) (Matrix (Fin N) (Fin N) (ZMod 2)))ˣ :=
    Finite.of_injective (fun u => (u : Matrix (Fin n) (Fin n) (Matrix (Fin N) (Fin N) (ZMod 2))))
      Units.ext
  exact Subtype.finite

    PrintedSimpleKazhdanGeneral := by

      (Exists.intro S hS : IsFinitelyGeneratedRing (ActionCrossedProduct α (ZMod 2))) hn
  have hG : IsSimpleGroup ↥(elementaryGroup (Fin n) (ActionCrossedProduct α (ZMod 2))) :=
