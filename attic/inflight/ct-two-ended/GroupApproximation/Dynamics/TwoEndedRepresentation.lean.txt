/-!
# The faithful action of the two-ended crossed product on `k^{(ℤ)}`

`non_mf_groups_exist.tex`, the remark after `cor:dynamic-rank-budget` (tex 1779–1780):

> To see this, use the faithful action on $\F_2^{(\mathbb Z)}$.

For the shift `T` on `X = ℤ ∪ {−∞, +∞}` (`TwoEndedShift.shift`) and every commutative ring `k`,
`R_X = LC(X, k) ⋊_T ℤ` acts on `k^{(ℤ)}` with basis `e_j = single j 1`: a coefficient `f` acts
diagonally by its values at the integer points, `e_j ↦ f(j) e_j`, and `u` acts by the bilateral
shift `e_j ↦ e_{j+1}` (`FinitaryGL.shiftGL`).  The covariance `u f u⁻¹ = f ∘ T⁻¹` holds because
`T⁻¹(j + 1) = j`.

* `diag k f`, `diagHom k`: the diagonal action of `LC(X, k)`;
* `shiftHom k : Multiplicative ℤ →* Module.End k (ℤ →₀ k)`: the powers of the shift;
* `rho k : ClopenCrossedProduct shift k →+* Module.End k (ℤ →₀ k)`: the action;
* `rho_apply_single_apply`: `(ρ(x) (c e_i))_l = x_{l−i}(l) · c`, where `x_m` is the Laurent
  coefficient of `x` at the exponent `m`;
* `rho_injective`: the action is faithful, since a locally constant function vanishing at every
  integer vanishes at both endpoints;
* `rho_coeff`: `ρ(f) = diag f`.
-/

namespace GroupApproximation

namespace TwoEndedShift

open Finsupp Filter Topology
open Multiplicative (ofAdd toAdd)

theorem shift_symm_pt (j : ℤ) : shift.symm (pt j) = pt (j - 1) := by
  calc shift.symm (pt j) = shift.symm (shift (pt (j - 1))) := by rw [shift_pt, sub_add_cancel]
    _ = pt (j - 1) := shift.symm_apply_apply _

theorem shift_zpow_pt (n j : ℤ) : (shift ^ n) (pt j) = pt (j + n) := by
  induction n using Int.induction_on generalizing j with
  | zero => rw [zpow_zero, Homeomorph.one_apply, add_zero]
  | succ n ih =>
      rw [zpow_add_one, Homeomorph.mul_apply, shift_pt, ih]
      congr 1
      omega
  | pred n ih =>
      rw [zpow_sub_one, Homeomorph.mul_apply, Homeomorph.inv_apply, shift_symm_pt, ih]
      congr 1
      omega

variable (k : Type*) [CommRing k]

/-- A locally constant function on `ℤ ∪ {−∞, +∞}` that vanishes at every integer is zero: it is
constant near each endpoint, and the integers accumulate at both endpoints. -/
theorem eq_zero_of_forall_pt {f : LocallyConstant TwoEnded k} (h : ∀ j, f (pt j) = 0) : f = 0 := by
  have hend : ∀ (x : TwoEnded) (l : Filter ℤ) [l.NeBot], Tendsto pt l (𝓝 x) → f x = 0 := by
    intro x l _ hx
    have hU : IsOpen {y : TwoEnded | f y = f x} := f.isLocallyConstant.isOpen_fiber (f x)
    have hev : ∀ᶠ j in l, pt j ∈ {y : TwoEnded | f y = f x} := hx (hU.mem_nhds rfl)
    obtain ⟨j, hj⟩ := hev.exists
    have hj' : f (pt j) = f x := hj
    rw [← hj', h j]
  refine LocallyConstant.ext fun x => ?_
  show f x = 0
  rcases eq_posInf_or_eq_negInf_or_exists_pt x with rfl | rfl | ⟨j, rfl⟩
  · exact hend posInf atTop tendsto_pt_atTop
  · exact hend negInf atBot tendsto_pt_atBot
  · exact h j

/-- **The diagonal action** `e_j ↦ f(j) e_j` of a coefficient `f ∈ LC(X, k)`. -/
noncomputable def diag (f : LocallyConstant TwoEnded k) : Module.End k (ℤ →₀ k) where
  toFun w := Finsupp.onFinset w.support (fun i => f (pt i) * w i)
    (fun i hi => Finsupp.mem_support_iff.2 fun h0 => hi (by rw [h0, mul_zero]))
  map_add' v w := Finsupp.ext fun i => by
    simp only [Finsupp.onFinset_apply, Finsupp.add_apply, mul_add]
  map_smul' c w := Finsupp.ext fun i => by
    simp only [Finsupp.onFinset_apply, Finsupp.smul_apply, smul_eq_mul, RingHom.id_apply]
    ring

theorem diag_apply (f : LocallyConstant TwoEnded k) (w : ℤ →₀ k) (i : ℤ) :
    diag k f w i = f (pt i) * w i :=
  rfl

theorem diag_single (f : LocallyConstant TwoEnded k) (j : ℤ) (c : k) :
    diag k f (single j c) = single j (f (pt j) * c) :=
  Finsupp.ext fun i => by
    rw [diag_apply, Finsupp.single_apply, Finsupp.single_apply]
    split_ifs with h
    · rw [h]
    · rw [mul_zero]

/-- The diagonal action as a ring homomorphism `LC(X, k) →+* End(k^{(ℤ)})`. -/
noncomputable def diagHom : LocallyConstant TwoEnded k →+* Module.End k (ℤ →₀ k) where
  toFun := diag k
  map_one' := LinearMap.ext fun w => Finsupp.ext fun i => by
    show (1 : k) * w i = w i
    rw [one_mul]
  map_mul' f g := LinearMap.ext fun w => Finsupp.ext fun i => by
    show f (pt i) * g (pt i) * w i = f (pt i) * (g (pt i) * w i)
    rw [mul_assoc]
  map_zero' := LinearMap.ext fun w => Finsupp.ext fun i => by
    show (0 : k) * w i = 0
    rw [zero_mul]
  map_add' f g := LinearMap.ext fun w => Finsupp.ext fun i => by
    show (f (pt i) + g (pt i)) * w i = f (pt i) * w i + g (pt i) * w i
    rw [add_mul]

/-- The coefficients of `R_X` acting diagonally. -/
noncomputable def coeffHom : ClopenCoeff shift k →+* Module.End k (ℤ →₀ k) :=
  (diagHom k).comp (ClopenCoeff.of shift k).symm.toRingHom

theorem coeffHom_apply (a : ClopenCoeff shift k) :
    coeffHom k a = diag k ((ClopenCoeff.of shift k).symm a) :=
  rfl

/-- The powers of the bilateral shift `e_j ↦ e_{j+1}`. -/
noncomputable def shiftHom : Multiplicative ℤ →* Module.End k (ℤ →₀ k) :=
  (Units.coeHom _).comp (zpowersHom _ (FinitaryGL.shiftGL k))

theorem shiftHom_apply (m : Multiplicative ℤ) :
    shiftHom k m = ((FinitaryGL.shiftGL k ^ toAdd m : LinearMap.GeneralLinearGroup k (ℤ →₀ k)) :
      Module.End k (ℤ →₀ k)) :=
  rfl

/-- **Covariance** in the representation: `(f ∘ T^{−m}) σ^m = σ^m f`. -/
theorem coeffHom_smul_mul_shiftHom (x : ClopenCoeff shift k) (y : Multiplicative ℤ) :
    coeffHom k (y • x) * shiftHom k y = shiftHom k y * coeffHom k x := by
  refine Finsupp.lhom_ext fun j c => ?_
  have hval : (ClopenCoeff.of shift k).symm (y • x) (pt (j + toAdd y)) =
      (ClopenCoeff.of shift k).symm x (pt j) := by
    show (ClopenCoeff.of shift k).symm x ((shift ^ (-toAdd y)) (pt (j + toAdd y))) =
      (ClopenCoeff.of shift k).symm x (pt j)
    rw [shift_zpow_pt, add_neg_cancel_right]
  rw [Module.End.mul_apply, Module.End.mul_apply, shiftHom_apply, FinitaryGL.shiftGL_zpow_single,
    coeffHom_apply, coeffHom_apply, diag_single, diag_single, FinitaryGL.shiftGL_zpow_single, hval]

/-- **The faithful action** of `R_X = LC(X, k) ⋊_T ℤ` on `k^{(ℤ)}` (tex 1780). -/
noncomputable def rho : ClopenCrossedProduct shift k →+* Module.End k (ℤ →₀ k) :=
  SkewMonoidAlgebra.liftNCRingHom (coeffHom k) (shiftHom k) fun {x y} =>
    coeffHom_smul_mul_shiftHom k x y

theorem rho_single (g : Multiplicative ℤ) (a : ClopenCoeff shift k) :
    rho k (SkewMonoidAlgebra.single g a) = coeffHom k a * shiftHom k g := by
  show SkewMonoidAlgebra.liftNC (coeffHom k : ClopenCoeff shift k →+ Module.End k (ℤ →₀ k))
    (shiftHom k) (SkewMonoidAlgebra.single g a) = coeffHom k a * shiftHom k g
  exact SkewMonoidAlgebra.liftNC_single _ _ _ _

/-- **Matrix entries of the action**: `(ρ(x) (c e_i))_l = x_{l−i}(l) · c`. -/
theorem rho_apply_single_apply (x : ClopenCrossedProduct shift k) (i l : ℤ) (c : k) :
    rho k x (single i c) l =
      (ClopenCoeff.of shift k).symm (x.coeff (ofAdd (l - i))) (pt l) * c := by
  induction x using SkewMonoidAlgebra.induction_on with
  | zero => simp
  | single g a =>
      rw [rho_single, Module.End.mul_apply, shiftHom_apply, FinitaryGL.shiftGL_zpow_single,
        coeffHom_apply, diag_single, Finsupp.single_apply, SkewMonoidAlgebra.coeff_single_apply]
      by_cases h : g = ofAdd (l - i)
      · have h' : i + toAdd g = l := by
          rw [h, toAdd_ofAdd]
          omega
        rw [if_pos h', if_pos h, h']
      · have h' : ¬ i + toAdd g = l := fun h'' => by
          have ht : toAdd g = l - i := by omega
          exact h (by rw [← ht, ofAdd_toAdd])
        rw [if_neg h', if_neg h]
        simp
  | add x y hx hy =>
      rw [map_add, LinearMap.add_apply, Finsupp.add_apply, hx, hy, SkewMonoidAlgebra.coeff_add,
        Finsupp.add_apply, map_add]
      exact (add_mul _ _ _).symm

/-- **The action is faithful** (tex 1780). -/
theorem rho_injective : Function.Injective (rho k) := by
  refine (injective_iff_map_eq_zero (rho k)).2 fun x hx => ?_
  refine SkewMonoidAlgebra.ext fun m => ?_
  have hcoeff : (ClopenCoeff.of shift k).symm (x.coeff m) = 0 := by
    refine eq_zero_of_forall_pt k fun l => ?_
    have h := rho_apply_single_apply k x (l - toAdd m) l 1
    rw [hx, LinearMap.zero_apply, Finsupp.coe_zero, Pi.zero_apply, mul_one, sub_sub_cancel,
      ofAdd_toAdd] at h
    exact h.symm
  rw [SkewMonoidAlgebra.coeff_zero, Finsupp.coe_zero, Pi.zero_apply]
  exact (ClopenCoeff.of shift k).symm.injective (hcoeff.trans (map_zero _).symm)

/-- `ρ(f) = diag f` for a coefficient `f ∈ LC(X, k)`. -/
theorem rho_coeff (f : LocallyConstant TwoEnded k) :
    rho k (ClopenCrossedProduct.coeff shift k f) = diag k f := by
  rw [ClopenCrossedProduct.coeff_apply, Pestov91.CrossedProduct.C_apply, rho_single, map_one,
    mul_one]
  exact congrArg (diag k) ((ClopenCoeff.of shift k).symm_apply_apply f)

/-- `ρ(u^n) = σ^n`: the implementing unit acts by the bilateral shift. -/
theorem rho_unit_zpow (n : ℤ) :
    rho k ((ClopenCrossedProduct.unit shift k ^ n : (ClopenCrossedProduct shift k)ˣ) :
      ClopenCrossedProduct shift k) =
      ((FinitaryGL.shiftGL k ^ n : LinearMap.GeneralLinearGroup k (ℤ →₀ k)) :
        Module.End k (ℤ →₀ k)) := by
  rw [Pestov91.CrossedProduct.val_unit_zpow, rho_single, map_one, one_mul, shiftHom_apply,
    toAdd_ofAdd]
