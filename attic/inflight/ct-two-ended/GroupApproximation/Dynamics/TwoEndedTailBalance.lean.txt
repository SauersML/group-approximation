/-!
# The two tail exponents of a unit agree, and the remainder is finitary

`non_mf_groups_exist.tex`, the remark after `cor:dynamic-rank-budget` (tex 1782–1785):

> The tails starting at $-N$ and $N$ span a subspace of codimension $2N-1$, whose image has
> codimension $2N-1+a_+-a_-$.  Invertibility forces $a_+=a_-$; removing that common shift leaves a
> finite invertible block extended by identity.

For a unit `u` of `R_X` over `𝔽₂`, `ρ(u)` translates the far tails by `a₊` and `a₋`
(`TwoEndedTailSymbols`).  Take `N` large.  The span of `e_j` for `j ≤ −N` or `j ≥ N` is the coordinate
subspace `k^{((−N, N)ᶜ)}`, of codimension `#(−N, N) = 2N − 1`.  Its image under `ρ(u)` is
`k^{((−N + a₋, N + a₊)ᶜ)}`, of codimension `2N − 1 + a₊ − a₋`.  A linear automorphism preserves
codimension (`SupportedCodimension`), so `a₊ = a₋ = a`.  Then `ρ(u) σ^{−a}` fixes every far basis
vector, so `ρ(u) σ^{−a} − 1` has finite matrix support: `ρ(u) σ^{−a} ∈ GL_fin(ℤ, 𝔽₂)`.

* `card_Ioo_neg`: `#(−N, N) = 2N − 1` and `#(−N + a₋, N + a₊) = 2N − 1 + a₊ − a₋`;
* `exists_common_tail_exponent`: `ρ(u) e_i = e_{i + a}` whenever `i ≥ N` or `i ≤ −N`;
* `units_map_mul_shiftGL_zpow_mem_GLfin`: `ρ(u) σ^{−a} ∈ GL_fin(ℤ, 𝔽₂)`.
-/

namespace GroupApproximation

namespace TwoEndedShift

open Finsupp
open Multiplicative (ofAdd toAdd)

/-- **Codimension of the tails** (tex 1782–1784): `#(b, c) = c − b − 1` for `b < c`; in particular
`#(−N, N) = 2N − 1` and `#(−N + a₋, N + a₊) = 2N − 1 + a₊ − a₋`. -/
theorem card_Ioo_eq {b c : ℤ} (h : b < c) : (Fintype.card (Set.Ioo b c) : ℤ) = c - b - 1 := by
  rw [← Set.toFinset_card, Set.toFinset_Ioo, Int.card_Ioo]
  omega

/-- The faithful action of a unit, as a linear automorphism of `𝔽₂^{(ℤ)}`. -/
noncomputable def unitEquiv (u : (ClopenCrossedProduct shift (ZMod 2))ˣ) :
    (ℤ →₀ ZMod 2) ≃ₗ[ZMod 2] (ℤ →₀ ZMod 2) :=
  LinearMap.GeneralLinearGroup.toLinearEquiv (Units.map (rho (ZMod 2)).toMonoidHom u)

theorem unitEquiv_apply (u : (ClopenCrossedProduct shift (ZMod 2))ˣ) (v : ℤ →₀ ZMod 2) :
    unitEquiv u v = rho (ZMod 2) (u : ClopenCrossedProduct shift (ZMod 2)) v :=
  rfl

/-- **Invertibility forces `a₊ = a₋`** (tex 1782–1785): a unit of `R_X` over `𝔽₂` translates both
far tails by one common exponent `a`. -/
theorem exists_common_tail_exponent (u : (ClopenCrossedProduct shift (ZMod 2))ˣ) :
    ∃ a N : ℤ, 0 ≤ N ∧ ∀ i, (N ≤ i ∨ i ≤ -N) →
      rho (ZMod 2) (u : ClopenCrossedProduct shift (ZMod 2)) (single i 1) = single (i + a) 1 := by
  obtain ⟨ap, Np, hp⟩ := exists_tail_translation_pos u
  obtain ⟨am, Nm, hm⟩ := exists_tail_translation_neg u
  obtain ⟨N, hN1, hN2, hN3, hN4⟩ : ∃ N : ℤ, Np ≤ N ∧ -N ≤ Nm ∧ -N + am < N + ap ∧ 0 ≤ N :=
    ⟨(Np.natAbs + Nm.natAbs + ap.natAbs + am.natAbs + 1 : ℕ), by omega, by omega, by omega,
      by omega⟩
  have hmap : (supported (ZMod 2) (ZMod 2) (Set.Ioo (-N) N)ᶜ).map
      ((unitEquiv u : (ℤ →₀ ZMod 2) ≃ₗ[ZMod 2] (ℤ →₀ ZMod 2)) :
        (ℤ →₀ ZMod 2) →ₗ[ZMod 2] (ℤ →₀ ZMod 2)) =
      supported (ZMod 2) (ZMod 2) (Set.Ioo (-N + am) (N + ap))ᶜ := by
    rw [supported_eq_span_single, supported_eq_span_single, Submodule.map_span]
    congr 1
    ext w
    simp only [Set.mem_image, Set.mem_compl_iff, Set.mem_Ioo, not_and_or, not_lt,
      LinearEquiv.coe_coe]
    constructor
    · rintro ⟨_, ⟨i, hi, rfl⟩, rfl⟩
      rcases hi with hi | hi
      · exact ⟨i + am, Or.inl (by omega), by rw [unitEquiv_apply, hm i (by omega)]⟩
      · exact ⟨i + ap, Or.inr (by omega), by rw [unitEquiv_apply, hp i (by omega)]⟩
    · rintro ⟨j, hj, rfl⟩
      rcases hj with hj | hj
      · exact ⟨single (j - am) 1, ⟨j - am, Or.inl (by omega), rfl⟩, by
          rw [unitEquiv_apply, hm (j - am) (by omega), sub_add_cancel]⟩
      · exact ⟨single (j - ap) 1, ⟨j - ap, Or.inr (by omega), rfl⟩, by
          rw [unitEquiv_apply, hp (j - ap) (by omega), sub_add_cancel]⟩
  have hcard := SupportedCodimension.card_eq_of_map_supported_compl (ZMod 2) (unitEquiv u) hmap
  have h1 := card_Ioo_eq (show -N < N by omega)
  have h2 := card_Ioo_eq hN3
  have ha : ap = am := by
    rw [hcard] at h1
    omega
  refine ⟨ap, N, hN4, fun i hi => ?_⟩
  rcases hi with hi | hi
  · exact hp i (by omega)
  · rw [ha]
    exact hm i (by omega)

/-- **The finitary remainder** (tex 1784–1785): removing the common shift leaves a finite invertible
block extended by identity, `ρ(u) σ^{−a} ∈ GL_fin(ℤ, 𝔽₂)`, where `ρ(u)` translates both far tails
by `a`. -/
theorem units_map_mul_shiftGL_zpow_mem_GLfin (u : (ClopenCrossedProduct shift (ZMod 2))ˣ) :
    ∃ a N : ℤ, 0 ≤ N ∧
      (∀ i, (N ≤ i ∨ i ≤ -N) →
        rho (ZMod 2) (u : ClopenCrossedProduct shift (ZMod 2)) (single i 1) = single (i + a) 1) ∧
      Units.map (rho (ZMod 2)).toMonoidHom u * FinitaryGL.shiftGL (ZMod 2) ^ (-a) ∈
        FinitaryGL.GLfin (ZMod 2) := by
  classical
  obtain ⟨a, N, hN, hu⟩ := exists_common_tail_exponent u
  refine ⟨a, N, hN, hu, ?_⟩
  set v : Module.End (ZMod 2) (ℤ →₀ ZMod 2) :=
    ((Units.map (rho (ZMod 2)).toMonoidHom u * FinitaryGL.shiftGL (ZMod 2) ^ (-a) :
      LinearMap.GeneralLinearGroup (ZMod 2) (ℤ →₀ ZMod 2)) : Module.End (ZMod 2) (ℤ →₀ ZMod 2))
    with hv
  have hcol : ∀ c : ℤ, (N + a ≤ c ∨ c ≤ -N + a) → (v - 1) (single c 1) = 0 := by
    intro c hc
    rw [LinearMap.sub_apply, Module.End.one_apply, sub_eq_zero]
    show rho (ZMod 2) (u : ClopenCrossedProduct shift (ZMod 2))
      (((FinitaryGL.shiftGL (ZMod 2) ^ (-a) : LinearMap.GeneralLinearGroup (ZMod 2)
        (ℤ →₀ ZMod 2)) : Module.End (ZMod 2) (ℤ →₀ ZMod 2)) (single c 1)) = single c 1
    rw [FinitaryGL.shiftGL_zpow_single, hu (c + -a) (by omega), neg_add_cancel_right]
  show FinitaryGL.HasFiniteMatrixSupport (ZMod 2) (v - 1)
  refine Set.Finite.subset ((Finset.Ioo (-N + a) (N + a)).finite_toSet.biUnion fun c _ =>
    ((((v - 1) (single c 1)).support : Set ℤ).toFinite.prod (Set.finite_singleton c)))
    fun p hp => ?_
  have hp' : (v - 1) (single p.2 1) p.1 ≠ 0 := hp
  have hc : p.2 ∈ Finset.Ioo (-N + a) (N + a) := by
    rw [Finset.mem_Ioo]
    by_contra hc
    rw [not_and_or, not_lt, not_lt] at hc
    exact hp' (by rw [hcol p.2 (by omega), Finsupp.coe_zero, Pi.zero_apply])
  exact Set.mem_biUnion (Finset.mem_coe.2 hc)
    ⟨Finset.mem_coe.2 (Finsupp.mem_support_iff.2 hp'), Set.mem_singleton p.2⟩

end TwoEndedShift
