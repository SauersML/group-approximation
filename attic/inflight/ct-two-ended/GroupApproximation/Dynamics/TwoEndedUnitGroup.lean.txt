/-!
# The unit group of the two-ended crossed product

`non_mf_groups_exist.tex`, the remark after `cor:dynamic-rank-budget` (tex 1774–1792):

> Nevertheless its unit group is
> \[
>  R_X^\times\cong\GL_{\mathrm{fin}}(\mathbb Z,\F_2)\rtimes\mathbb Z,
> \]
> where $\GL_{\mathrm{fin}}$ means finite matrix support of $g-1$, and
> $\mathbb Z$ acts by the bilateral shift.  [...]  removing that common
> shift leaves a finite invertible block extended by identity.
> [...]  Thus $R_X^\times$ is LEF although $\GL_3(R_X)$ is
> not MF\@.

A unit `u` of `R_X` over `𝔽₂` translates both far tails by one exponent `a = tailExponent u`
(`TwoEndedTailBalance`), and `ρ(u) σ^{−a} ∈ GL_fin(ℤ, 𝔽₂)`.  So `u ↦ (ρ(u) σ^{−a}, a)` is a
homomorphism `R_X^× → GL_fin(ℤ, 𝔽₂) ⋊ ℤ`.  It is injective because `ρ` is faithful.  It is surjective
because every finitary block times a power of `u` is a unit (`TwoEndedMatrixUnits`, lane
ct-bilateral-cell).

* `TranslatesTails u a`, `translatesTails_unique`, `translatesTails_mul`;
* `tailExponent u`, `tailExponent_mul`;
* `unitsToFinitaryShift : R_X^× →* FinitaryGL.FinitaryShift (ZMod 2)`, bijective;
* `unitsEquivFinitaryShift`, closed endpoint `printedTwoEndedUnitGroup`;
* `printedTwoEndedUnitsLEFGLThreeNotMF_of_pieces`: "R_X^× is LEF although GL_3(R_X) is not MF", from
  LEF of `GL_fin(ℤ, 𝔽₂) ⋊ ℤ` (tex 1786–1791) and `prop:bilateral-three`.
-/

namespace GroupApproximation

namespace TwoEndedShift

open Finsupp
open Multiplicative (ofAdd toAdd)

/-- `ρ(u)` translates both far tails by `a`: `ρ(u) e_i = e_{i+a}` for `i ≥ N` and for `i ≤ −N`. -/
def TranslatesTails (u : (ClopenCrossedProduct shift (ZMod 2))ˣ) (a : ℤ) : Prop :=
  ∃ N : ℤ, ∀ i, (N ≤ i ∨ i ≤ -N) →
    rho (ZMod 2) (u : ClopenCrossedProduct shift (ZMod 2)) (single i 1) = single (i + a) 1

theorem translatesTails_unique {u : (ClopenCrossedProduct shift (ZMod 2))ˣ} {a b : ℤ}
    (ha : TranslatesTails u a) (hb : TranslatesTails u b) : a = b := by
  obtain ⟨N, hN⟩ := ha
  obtain ⟨M, hM⟩ := hb
  have h := (hN (max N M) (Or.inl (le_max_left _ _))).symm.trans
    (hM (max N M) (Or.inl (le_max_right _ _)))
  have h' := (Finsupp.single_left_inj (one_ne_zero : (1 : ZMod 2) ≠ 0)).1 h
  omega

theorem translatesTails_mul {u v : (ClopenCrossedProduct shift (ZMod 2))ˣ} {a b : ℤ}
    (ha : TranslatesTails u a) (hb : TranslatesTails v b) : TranslatesTails (u * v) (b + a) := by
  obtain ⟨N, hN⟩ := ha
  obtain ⟨M, hM⟩ := hb
  refine ⟨((N.natAbs + M.natAbs + b.natAbs : ℕ) : ℤ), fun i hi => ?_⟩
  rw [Units.val_mul, map_mul, Module.End.mul_apply, hM i (by omega), hN (i + b) (by omega),
    add_assoc]

theorem exists_translatesTails (u : (ClopenCrossedProduct shift (ZMod 2))ˣ) :
    ∃ a : ℤ, TranslatesTails u a ∧
      Units.map (rho (ZMod 2)).toMonoidHom u * FinitaryGL.shiftGL (ZMod 2) ^ (-a) ∈
        FinitaryGL.GLfin (ZMod 2) := by
  obtain ⟨a, N, _, hN, hmem⟩ := units_map_mul_shiftGL_zpow_mem_GLfin u
  exact ⟨a, ⟨N, hN⟩, hmem⟩

/-- **The common tail exponent** `a` of a unit (tex 1784). -/
noncomputable def tailExponent (u : (ClopenCrossedProduct shift (ZMod 2))ˣ) : ℤ :=
  Classical.choose (exists_translatesTails u)

theorem translatesTails_tailExponent (u : (ClopenCrossedProduct shift (ZMod 2))ˣ) :
    TranslatesTails u (tailExponent u) :=
  (Classical.choose_spec (exists_translatesTails u)).1

theorem mem_GLfin_tailExponent (u : (ClopenCrossedProduct shift (ZMod 2))ˣ) :
    Units.map (rho (ZMod 2)).toMonoidHom u * FinitaryGL.shiftGL (ZMod 2) ^ (-tailExponent u) ∈
      FinitaryGL.GLfin (ZMod 2) :=
  (Classical.choose_spec (exists_translatesTails u)).2

theorem tailExponent_mul (u v : (ClopenCrossedProduct shift (ZMod 2))ˣ) :
    tailExponent (u * v) = tailExponent u + tailExponent v := by
  rw [add_comm]
  exact translatesTails_unique (translatesTails_tailExponent (u * v))
    (translatesTails_mul (translatesTails_tailExponent u) (translatesTails_tailExponent v))

theorem tailExponent_one : tailExponent (1 : (ClopenCrossedProduct shift (ZMod 2))ˣ) = 0 :=
  translatesTails_unique (translatesTails_tailExponent 1) ⟨0, fun i _ => by
    rw [Units.val_one, map_one, Module.End.one_apply, add_zero]⟩

/-- **`R_X^× → GL_fin(ℤ, 𝔽₂) ⋊ ℤ`**, `u ↦ (ρ(u) σ^{−a}, a)` with `a` the tail exponent. -/
noncomputable def unitsToFinitaryShift :
    (ClopenCrossedProduct shift (ZMod 2))ˣ →* FinitaryGL.FinitaryShift (ZMod 2) where
  toFun u :=
    ⟨⟨Units.map (rho (ZMod 2)).toMonoidHom u * FinitaryGL.shiftGL (ZMod 2) ^ (-tailExponent u),
      mem_GLfin_tailExponent u⟩, ofAdd (tailExponent u)⟩
  map_one' := by
    refine SemidirectProduct.ext (Subtype.ext ?_) ?_
    · show Units.map (rho (ZMod 2)).toMonoidHom 1 *
        FinitaryGL.shiftGL (ZMod 2) ^ (-tailExponent (1 : (ClopenCrossedProduct shift (ZMod 2))ˣ)) = 1
      rw [tailExponent_one, map_one, neg_zero, zpow_zero, one_mul]
    · show ofAdd (tailExponent (1 : (ClopenCrossedProduct shift (ZMod 2))ˣ)) = 1
      rw [tailExponent_one, ofAdd_zero]
  map_mul' u v := by
    refine SemidirectProduct.ext (Subtype.ext ?_) ?_
    · show Units.map (rho (ZMod 2)).toMonoidHom (u * v) *
          FinitaryGL.shiftGL (ZMod 2) ^ (-tailExponent (u * v)) =
        (Units.map (rho (ZMod 2)).toMonoidHom u * FinitaryGL.shiftGL (ZMod 2) ^ (-tailExponent u)) *
          (FinitaryGL.shiftGL (ZMod 2) ^ toAdd (ofAdd (tailExponent u)) *
            (Units.map (rho (ZMod 2)).toMonoidHom v *
              FinitaryGL.shiftGL (ZMod 2) ^ (-tailExponent v)) *
            (FinitaryGL.shiftGL (ZMod 2) ^ toAdd (ofAdd (tailExponent u)))⁻¹)
      rw [tailExponent_mul, map_mul, toAdd_ofAdd]
      group
    · show ofAdd (tailExponent (u * v)) = ofAdd (tailExponent u) * ofAdd (tailExponent v)
      rw [tailExponent_mul, ofAdd_add]

theorem unitsToFinitaryShift_injective : Function.Injective unitsToFinitaryShift := by
  rw [injective_iff_map_eq_one]
  intro u hu
  have hr : ofAdd (tailExponent u) = 1 := congrArg SemidirectProduct.right hu
  have hl : Units.map (rho (ZMod 2)).toMonoidHom u *
      FinitaryGL.shiftGL (ZMod 2) ^ (-tailExponent u) = 1 :=
    congrArg (fun x : FinitaryGL.FinitaryShift (ZMod 2) =>
      ((x.left : FinitaryGL.GLfin (ZMod 2)) : LinearMap.GeneralLinearGroup (ZMod 2) (ℤ →₀ ZMod 2)))
      hu
  have he : tailExponent u = 0 := Multiplicative.ofAdd_eq_one.1 hr
  rw [he, neg_zero, zpow_zero, mul_one] at hl
  refine Units.ext (rho_injective (ZMod 2) ?_)
  rw [Units.val_one, map_one]
  exact congrArg Units.val hl

theorem fixesTails_of_mem_GLfin {g : LinearMap.GeneralLinearGroup (ZMod 2) (ℤ →₀ ZMod 2)}
    (hg : g ∈ FinitaryGL.GLfin (ZMod 2)) :
    ∃ N : ℤ, ∀ i, (N ≤ i ∨ i ≤ -N) →
      (g : Module.End (ZMod 2) (ℤ →₀ ZMod 2)) (single i 1) = single i 1 := by
  obtain ⟨S, hS⟩ := (FinitaryGL.hasFiniteMatrixSupport_iff _).1 hg
  refine ⟨∑ j ∈ S, |j| + 1, fun i hi => ?_⟩
  have hiS : i ∉ S := fun h => by
    have hle : |i| ≤ ∑ j ∈ S, |j| := Finset.single_le_sum (fun j _ => abs_nonneg j) h
    have h1 := le_abs_self i
    have h2 := neg_abs_le i
    rcases hi with hi | hi <;> linarith
  have h0 := hS.apply_single_eq_zero hiS
  rw [LinearMap.sub_apply, Module.End.one_apply, sub_eq_zero] at h0
  exact h0

theorem unitsToFinitaryShift_surjective : Function.Surjective unitsToFinitaryShift := by
  rintro ⟨⟨g, hg⟩, m⟩
  obtain ⟨x, hx⟩ := exists_unit_rho_eq_mul_shiftGL_zpow (ZMod 2) hg (toAdd m)
  refine ⟨x, ?_⟩
  have hmap : Units.map (rho (ZMod 2)).toMonoidHom x = g * FinitaryGL.shiftGL (ZMod 2) ^ toAdd m :=
    Units.ext hx
  have htr : TranslatesTails x (toAdd m) := by
    obtain ⟨N, hN⟩ := fixesTails_of_mem_GLfin hg
    refine ⟨((N.natAbs + (toAdd m).natAbs : ℕ) : ℤ), fun i hi => ?_⟩
    rw [hx, Units.val_mul, Module.End.mul_apply, FinitaryGL.shiftGL_zpow_single,
      hN (i + toAdd m) (by omega)]
  have he : tailExponent x = toAdd m :=
    translatesTails_unique (translatesTails_tailExponent x) htr
  refine SemidirectProduct.ext (Subtype.ext ?_) ?_
  · show Units.map (rho (ZMod 2)).toMonoidHom x * FinitaryGL.shiftGL (ZMod 2) ^ (-tailExponent x) = g
    rw [he, hmap, mul_assoc, ← zpow_add, add_neg_cancel, zpow_zero, mul_one]
  · show ofAdd (tailExponent x) = m
    rw [he, ofAdd_toAdd]

/-- The isomorphism `R_X^× ≅ GL_fin(ℤ, 𝔽₂) ⋊ ℤ`. -/
noncomputable def unitsEquivFinitaryShift :
    (ClopenCrossedProduct shift (ZMod 2))ˣ ≃* FinitaryGL.FinitaryShift (ZMod 2) :=
  MulEquiv.ofBijective unitsToFinitaryShift
    ⟨unitsToFinitaryShift_injective, unitsToFinitaryShift_surjective⟩

/-- **tex 1774–1779**: "its unit group is `R_X^× ≅ GL_fin(ℤ, 𝔽₂) ⋊ ℤ`, where `GL_fin` means finite
matrix support of `g − 1`, and `ℤ` acts by the bilateral shift".  `R_X = LC(X, 𝔽₂) ⋊_T ℤ` for the shift
`T` on `X = ℤ ∪ {−∞, +∞}` (`TwoEndedShift.shift`); `FinitaryGL.FinitaryShift (ZMod 2)` is
`GL_fin(ℤ, 𝔽₂) ⋊ ℤ` with `ℤ` acting by conjugation with the bilateral shift. -/
def PrintedTwoEndedUnitGroup : Prop :=
  Nonempty ((ClopenCrossedProduct shift (ZMod 2))ˣ ≃* FinitaryGL.FinitaryShift (ZMod 2))

theorem printedTwoEndedUnitGroup : PrintedTwoEndedUnitGroup :=
  ⟨unitsEquivFinitaryShift⟩

/-- **tex 1791–1792**: "Thus `R_X^×` is LEF although `GL_3(R_X)` is not MF." -/
def PrintedTwoEndedUnitsLEFGLThreeNotMF : Prop :=
  IsLEF (ClopenCrossedProduct shift (ZMod 2))ˣ ∧
    ¬ IsOperatorMF (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct shift (ZMod 2)))ˣ

/-- The conclusion of the remark, from LEF of `GL_fin(ℤ, 𝔽₂) ⋊ ℤ` (tex 1786–1791) and
`prop:bilateral-three` at `n = 3`, since `R_X` is not directly finite. -/
theorem printedTwoEndedUnitsLEFGLThreeNotMF_of_pieces
    (hLEF : IsLEF (FinitaryGL.FinitaryShift (ZMod 2))) (hB : PrintedBilateralThree) :
    PrintedTwoEndedUnitsLEFGLThreeNotMF := by
  haveI : Nonempty TwoEnded := ⟨posInf⟩
  refine ⟨isLEF_of_injective unitsEquivFinitaryShift.toMonoidHom
    unitsEquivFinitaryShift.injective hLEF, fun hMF => ?_⟩
  have h := hB TwoEnded shift (ZMod 2) 3 le_rfl
  exact not_isDedekindFiniteMonoid (h.2.1.1 (h.1.1 hMF))

end TwoEndedShift
