/-!
# The two-ended shift: its core and its clopen defect

`non_mf_groups_exist.tex`, `sec:chain-core`, the remark after `cor:dynamic-rank-budget`
(tex 1769–1773):

> The period hypothesis cannot be removed from a unit-group theorem.
> For the shift on the two-ended compactification
> $X=\mathbb Z\cup\{-\infty,+\infty\}$, with its endpoints fixed,
> the core consists of those endpoints.  Over $\F_2$ the ring is not
> directly finite: $P=\{0,1,\dots\}\cup\{+\infty\}$ has defect $\{0\}$.

## The space

A point of `TwoEnded` is an up-closed subset of `ℤ`, recorded as a monotone map `ℤ → Bool`, with
the topology of pointwise convergence.  The integer `j` is `pt j`, the up-set `[j, ∞)`; `+∞` is
`posInf`, the empty up-set, and `−∞` is `negInf`, the up-set `ℤ`.  The theorem
`isTwoEndedCompactification` records that this is the two-ended compactification
`ℤ ∪ {−∞, +∞}`: every point is an integer or one of the two distinct endpoints, each integer is
isolated, `j → +∞` converges to `+∞` and `j → −∞` converges to `−∞`, and the space is compact
Hausdorff.  It is also metrizable and totally disconnected, so it lies in the printed class of
sec:chain-core.  `shift` is `j ↦ j + 1` with both endpoints fixed.

The core `CR(T)` is computed for the uniform structure of `TwoEnded`; on a compact space every
compatible uniform structure gives the same set (`Dynamics.chainRecurrentSet_eq_of_compact`).

## Results

* `chainRecurrentSet_shift`: the core is `{−∞, +∞}`;
* `isClopen_P`, `P_eq`, `shift_image_P_subset`, `P_diff_shift_image_P`: `P = {0, 1, …} ∪ {+∞}` is
  a compressed clopen set with defect `{0}`;
* `not_isDedekindFiniteMonoid`: over `𝔽₂` the ring `R_X` is not directly finite;
* `printedTwoEndedCoreAndDefect`: the closed endpoint.
-/

namespace GroupApproximation

/-- The two-ended compactification `ℤ ∪ {−∞, +∞}`, as the up-closed subsets of `ℤ` (monotone maps
`ℤ → Bool`) with the topology of pointwise convergence. -/
abbrev TwoEnded : Type := {f : ℤ → Bool // Monotone f}

/-- The integer `j`, as the up-set `[j, ∞)`. -/
def pt (j : ℤ) : TwoEnded :=
  ⟨fun i => decide (j ≤ i), fun a b hab => by
    show decide (j ≤ a) ≤ decide (j ≤ b)
    by_cases h : j ≤ a
    · rw [decide_eq_true (le_trans h hab)]
      exact le_top
    · rw [decide_eq_false h]
      exact bot_le⟩

/-- `+∞`, the empty up-set. -/
def posInf : TwoEnded := ⟨fun _ => false, monotone_const⟩

/-- `−∞`, the up-set `ℤ`. -/
def negInf : TwoEnded := ⟨fun _ => true, monotone_const⟩

theorem bool_true_ne_false : true ≠ false := by decide

theorem pt_apply (j i : ℤ) : (pt j).1 i = decide (j ≤ i) := rfl

theorem val_eq_true_of_le {x : TwoEnded} {a b : ℤ} (hab : a ≤ b) (ha : x.1 a = true) :
    x.1 b = true := by
  have h := x.2 hab
  rw [ha] at h
  exact top_le_iff.1 h

theorem val_eq_false_of_le {x : TwoEnded} {a b : ℤ} (hab : a ≤ b) (hb : x.1 b = false) :
    x.1 a = false := by
  have h := x.2 hab
  rw [hb] at h
  exact le_bot_iff.1 h

/-- Every point of `TwoEnded` is `+∞`, `−∞` or an integer. -/
theorem eq_posInf_or_eq_negInf_or_exists_pt (x : TwoEnded) :
    x = posInf ∨ x = negInf ∨ ∃ j, x = pt j := by
  by_cases ht : ∃ i, x.1 i = true
  · by_cases hf : ∃ i, x.1 i = false
    · obtain ⟨i₀, hi₀⟩ := hf
      have hbdd : ∃ b : ℤ, ∀ z : ℤ, x.1 z = true → b ≤ z := by
        refine ⟨i₀, fun z hz => ?_⟩
        by_contra hlt
        have h := val_eq_true_of_le (le_of_lt (not_le.1 hlt)) hz
        rw [hi₀] at h
        exact Bool.false_ne_true h
      obtain ⟨j, hj, hjmin⟩ := Int.exists_least_of_bdd hbdd ht
      refine Or.inr (Or.inr ⟨j, Subtype.ext (funext fun i => ?_)⟩)
      rw [pt_apply]
      by_cases hji : j ≤ i
      · rw [decide_eq_true hji]
        exact val_eq_true_of_le hji hj
      · rw [decide_eq_false hji]
        cases hxi : x.1 i
        · rfl
        · exact absurd (hjmin i hxi) hji
    · refine Or.inr (Or.inl (Subtype.ext (funext fun i => ?_)))
      cases hxi : x.1 i
      · exact absurd ⟨i, hxi⟩ hf
      · rfl
  · refine Or.inl (Subtype.ext (funext fun i => ?_))
    cases hxi : x.1 i
    · rfl
    · exact absurd ⟨i, hxi⟩ ht

theorem pt_injective : Function.Injective pt := by
  intro a b h
  have h₁ : decide (a ≤ b) = true :=
    (congrFun (congrArg Subtype.val h) b).trans (decide_eq_true (le_refl b))
  have h₂ : decide (b ≤ a) = true :=
    (congrFun (congrArg Subtype.val h) a).symm.trans (decide_eq_true (le_refl a))
  exact le_antisymm (of_decide_eq_true h₁) (of_decide_eq_true h₂)

theorem pt_ne_posInf (j : ℤ) : pt j ≠ posInf := fun h => by
  have h₁ : decide (j ≤ j) = false := congrFun (congrArg Subtype.val h) j
  rw [decide_eq_true (le_refl j)] at h₁
  exact bool_true_ne_false h₁

theorem pt_ne_negInf (j : ℤ) : pt j ≠ negInf := fun h => by
  have h₁ : decide (j ≤ j - 1) = true := congrFun (congrArg Subtype.val h) (j - 1)
  exact absurd (of_decide_eq_true h₁) (by omega)

theorem posInf_ne_negInf : posInf ≠ negInf := fun h => by
  have h₁ : false = true := congrFun (congrArg Subtype.val h) 0
  exact Bool.false_ne_true h₁

theorem continuous_apply_val (i : ℤ) : Continuous fun x : TwoEnded => x.1 i :=
  (continuous_apply i).comp continuous_subtype_val

theorem isClopen_setOf_apply_eq (i : ℤ) (b : Bool) : IsClopen {x : TwoEnded | x.1 i = b} :=
  (isClopen_discrete ({b} : Set Bool)).preimage (continuous_apply_val i)

theorem singleton_pt_eq (j : ℤ) :
    ({pt j} : Set TwoEnded) = {x | x.1 j = true} ∩ {x | x.1 (j - 1) = false} := by
  ext x
  constructor
  · intro hx
    rw [Set.mem_singleton_iff.1 hx]
    exact Set.mem_inter (decide_eq_true (le_refl j)) (decide_eq_false (by omega))
  · rintro ⟨h₁, h₂⟩
    rcases eq_posInf_or_eq_negInf_or_exists_pt x with rfl | rfl | ⟨i, rfl⟩
    · exact absurd h₁ Bool.false_ne_true
    · exact absurd h₂ bool_true_ne_false
    · have hij : i ≤ j := of_decide_eq_true h₁
      have hij' : ¬ i ≤ j - 1 := of_decide_eq_false h₂
      rw [le_antisymm hij (by omega)]
      exact Set.mem_singleton _

/-- Each integer is an isolated point. -/
theorem isOpen_singleton_pt (j : ℤ) : IsOpen ({pt j} : Set TwoEnded) := by
  rw [singleton_pt_eq]
  exact (isClopen_setOf_apply_eq j true).isOpen.inter
    (isClopen_setOf_apply_eq (j - 1) false).isOpen

theorem isClosed_setOf_monotone : IsClosed {f : ℤ → Bool | Monotone f} := by
  have h : {f : ℤ → Bool | Monotone f} = ⋂ a : ℤ, ⋂ b : ℤ, ⋂ (_ : a ≤ b), {f | f a ≤ f b} := by
    ext f
    simp only [Set.mem_setOf_eq, Set.mem_iInter]
    rfl
  rw [h]
  refine isClosed_iInter fun a => isClosed_iInter fun b => isClosed_iInter fun _ => ?_
  have hc : IsClosed ((fun g : ℤ → Bool => (g a, g b)) ⁻¹' {p : Bool × Bool | p.1 ≤ p.2}) :=
    IsClosed.preimage ((continuous_apply a).prodMk (continuous_apply b))
      (isClopen_discrete {p : Bool × Bool | p.1 ≤ p.2}).isClosed
  exact hc

instance : CompactSpace TwoEnded := by
  exact isCompact_iff_compactSpace.1 isClosed_setOf_monotone.isCompact

instance : TopologicalSpace.MetrizableSpace TwoEnded := by
  haveI : TopologicalSpace.MetrizableSpace Bool := inferInstance
  haveI : TopologicalSpace.MetrizableSpace (ℤ → Bool) := TopologicalSpace.metrizableSpace_pi
  exact TopologicalSpace.MetrizableSpace.subtype {f : ℤ → Bool | Monotone f}

theorem tendsto_pt_atTop : Tendsto pt atTop (𝓝 posInf) := by
  rw [tendsto_subtype_rng, tendsto_pi_nhds]
  intro i
  refine tendsto_const_nhds.congr' ?_
  filter_upwards [eventually_gt_atTop i] with j hj
  exact (decide_eq_false (by omega)).symm

theorem tendsto_pt_atBot : Tendsto pt atBot (𝓝 negInf) := by
  rw [tendsto_subtype_rng, tendsto_pi_nhds]
  intro i
  refine tendsto_const_nhds.congr' ?_
  filter_upwards [eventually_le_atBot i] with j hj
  exact (decide_eq_true hj).symm

/-- `X` is the two-ended compactification of `ℤ`, with integers `ι j` and endpoints `top`
(`+∞`) and `bot` (`−∞`): the points are the integers and the two distinct endpoints, each integer
is isolated, `j → +∞` converges to `top`, `j → −∞` converges to `bot`, and `X` is compact
Hausdorff.  These properties determine `X` up to homeomorphism. -/
structure IsTwoEndedCompactification (X : Type*) [TopologicalSpace X] (ι : ℤ → X)
    (top bot : X) : Prop where
  injective : Function.Injective ι
  isOpen_singleton : ∀ j, IsOpen ({ι j} : Set X)
  cover : ∀ x, x = top ∨ x = bot ∨ ∃ j, x = ι j
  top_ne_bot : top ≠ bot
  ne_top : ∀ j, ι j ≠ top
  ne_bot : ∀ j, ι j ≠ bot
  tendsto_atTop : Tendsto ι atTop (𝓝 top)
  tendsto_atBot : Tendsto ι atBot (𝓝 bot)
  compact : CompactSpace X
  t2 : T2Space X

theorem isTwoEndedCompactification : IsTwoEndedCompactification TwoEnded pt posInf negInf where
  injective := pt_injective
  isOpen_singleton := isOpen_singleton_pt
  cover := eq_posInf_or_eq_negInf_or_exists_pt
  top_ne_bot := posInf_ne_negInf
  ne_top := pt_ne_posInf
  ne_bot := pt_ne_negInf
  tendsto_atTop := tendsto_pt_atTop
  tendsto_atBot := tendsto_pt_atBot
  compact := inferInstance
  t2 := inferInstance

/-- **The shift** `T(j) = j + 1` of `ℤ ∪ {−∞, +∞}`, with both endpoints fixed. -/
def shift : TwoEnded ≃ₜ TwoEnded where
  toFun x := ⟨fun i => x.1 (i - 1), fun a b hab => x.2 (by omega)⟩
  invFun x := ⟨fun i => x.1 (i + 1), fun a b hab => x.2 (by omega)⟩
  left_inv x := Subtype.ext (funext fun i => by simp)
  right_inv x := Subtype.ext (funext fun i => by simp)
  continuous_toFun := by
    refine Continuous.subtype_mk (continuous_pi fun i => ?_) _
    exact (continuous_apply (i - 1)).comp continuous_subtype_val
  continuous_invFun := by
    refine Continuous.subtype_mk (continuous_pi fun i => ?_) _
    exact (continuous_apply (i + 1)).comp continuous_subtype_val

theorem shift_pt (j : ℤ) : shift (pt j) = pt (j + 1) := by
  refine Subtype.ext (funext fun i => ?_)
  show decide (j ≤ i - 1) = decide (j + 1 ≤ i)
  exact decide_eq_decide.2 (by omega)

theorem shift_posInf : shift posInf = posInf := rfl

theorem shift_negInf : shift negInf = negInf := rfl

/-! ### The core -/

theorem isChainRecurrent_of_apply_eq {T : TwoEnded → TwoEnded} {x : TwoEnded} (hx : T x = x) :
    Dynamics.IsChainRecurrent T x := fun V hV =>
  Relation.TransGen.single (by
    show (T x, x) ∈ V
    rw [hx]
    exact refl_mem_uniformity hV)

/-- The integers above `j`, together with `+∞`. -/
def above (j : ℤ) : Set TwoEnded := {x | x.1 j = false}

theorem shift_mem_above {j : ℤ} {x : TwoEnded} (hx : x ∈ above j) : shift x ∈ above j :=
  val_eq_false_of_le (x := x) (by omega : j - 1 ≤ j) hx

theorem shift_pt_mem_above (j : ℤ) : shift (pt j) ∈ above j := by
  rw [shift_pt]
  exact decide_eq_false (by omega)

theorem mem_uniformity_sameSide {Q : Set TwoEnded} (hQ : IsClopen Q) :
    {p : TwoEnded × TwoEnded | p.1 ∈ Q ↔ p.2 ∈ Q} ∈ 𝓤 TwoEnded := by
  rw [← nhdsSet_diagonal_eq_uniformity]
  have hopen : IsOpen {p : TwoEnded × TwoEnded | p.1 ∈ Q ↔ p.2 ∈ Q} := by
    have h : {p : TwoEnded × TwoEnded | p.1 ∈ Q ↔ p.2 ∈ Q} = Q ×ˢ Q ∪ Qᶜ ×ˢ Qᶜ := by
      ext p
      simp only [Set.mem_setOf_eq, Set.mem_union, Set.mem_prod, Set.mem_compl_iff]
      tauto
    rw [h]
    exact (hQ.isOpen.prod hQ.isOpen).union (hQ.compl.isOpen.prod hQ.compl.isOpen)
  refine hopen.mem_nhdsSet.2 fun p hp => ?_
  have hp' : p.1 = p.2 := hp
  show p.1 ∈ Q ↔ p.2 ∈ Q
  rw [hp']

/-- No integer is chain recurrent: a chain whose steps never cross the clopen set `above j`
cannot return from `j`. -/
theorem not_isChainRecurrent_pt (j : ℤ) : ¬ Dynamics.IsChainRecurrent shift (pt j) := by
  intro h
  have hQ : IsClopen (above j) := isClopen_setOf_apply_eq j false
  have hstay : ∀ b, Relation.TransGen (Dynamics.ChainStep shift
      {p : TwoEnded × TwoEnded | p.1 ∈ above j ↔ p.2 ∈ above j}) (pt j) b → b ∈ above j := by
    intro b hb
    induction hb with
    | single hstep => exact Iff.mp hstep (shift_pt_mem_above j)
    | tail _ hstep ih => exact Iff.mp hstep (shift_mem_above ih)
  have hnot : pt j ∉ above j := fun hmem =>
    bool_true_ne_false ((decide_eq_true (le_refl j)).symm.trans hmem)
  exact hnot (hstay _ (h _ (mem_uniformity_sameSide hQ)))

/-- **The core** (tex 1771–1772): "the core consists of those endpoints". -/
theorem chainRecurrentSet_shift : Dynamics.chainRecurrentSet shift = {negInf, posInf} := by
  ext x
  rw [Dynamics.mem_chainRecurrentSet]
  constructor
  · intro hx
    rcases eq_posInf_or_eq_negInf_or_exists_pt x with rfl | rfl | ⟨j, rfl⟩
    · exact Set.mem_insert_of_mem _ (Set.mem_singleton _)
    · exact Set.mem_insert _ _
    · exact absurd hx (not_isChainRecurrent_pt j)
  · rintro (rfl | rfl)
    · exact isChainRecurrent_of_apply_eq shift_negInf
    · exact isChainRecurrent_of_apply_eq shift_posInf

/-! ### The clopen defect -/

/-- `P = {0, 1, …} ∪ {+∞}` (tex 1773). -/
def P : Set TwoEnded := {x | x.1 (-1) = false}

theorem P_eq : P = Set.range (fun n : ℕ => pt n) ∪ {posInf} := by
  ext x
  constructor
  · intro hx
    rcases eq_posInf_or_eq_negInf_or_exists_pt x with rfl | rfl | ⟨j, rfl⟩
    · exact Set.mem_union_right _ (Set.mem_singleton _)
    · exact absurd hx bool_true_ne_false
    · have hj : ¬ j ≤ -1 := of_decide_eq_false hx
      refine Set.mem_union_left _ (Set.mem_range.2 ⟨j.toNat, ?_⟩)
      show pt (j.toNat : ℤ) = pt j
      rw [Int.toNat_of_nonneg (by omega)]
  · rintro (⟨n, rfl⟩ | rfl)
    · exact decide_eq_false (by omega)
    · exact rfl

theorem isClopen_P : IsClopen P := isClopen_setOf_apply_eq (-1) false

theorem shift_image_P : shift '' P = {x : TwoEnded | x.1 0 = false} := by
  ext x
  constructor
  · rintro ⟨y, hy, rfl⟩
    have hy' : y.1 (-1) = false := hy
    show y.1 (0 - 1) = false
    rw [zero_sub]
    exact hy'
  · intro hx
    have hx' : x.1 0 = false := hx
    refine ⟨shift.symm x, ?_, shift.apply_symm_apply x⟩
    show x.1 (-1 + 1) = false
    rw [neg_add_cancel]
    exact hx'

/-- `T(P) ⊆ P`. -/
theorem shift_image_P_subset : shift '' P ⊆ P := by
  rw [shift_image_P]
  intro x hx
  exact val_eq_false_of_le (by omega : (-1 : ℤ) ≤ 0) hx

/-- "`P` has defect `{0}`": `P ∖ T(P) = {0}`. -/
theorem P_diff_shift_image_P : P \ shift '' P = {pt 0} := by
  rw [shift_image_P, singleton_pt_eq]
  ext x
  constructor
  · rintro ⟨h₁, h₂⟩
    have h₁' : x.1 (-1) = false := h₁
    have h₂' : ¬ x.1 0 = false := h₂
    refine Set.mem_inter ?_ ?_
    · cases hx0 : x.1 0
      · exact absurd hx0 h₂'
      · exact hx0
    · show x.1 (0 - 1) = false
      rw [zero_sub]
      exact h₁'
  · rintro ⟨h₁, h₂⟩
    have h₁' : x.1 0 = true := h₁
    have h₂' : x.1 (0 - 1) = false := h₂
    rw [zero_sub] at h₂'
    refine Set.mem_sdiff_of_mem h₂' ?_
    show ¬ x.1 0 = false
    rw [h₁']
    exact bool_true_ne_false

theorem coeff_injective {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type*) [Ring k] :
    Function.Injective (ClopenCrossedProduct.coeff T k) := by
  intro f g hfg
  have h : SkewMonoidAlgebra.coeff (ClopenCrossedProduct.coeff T k f) 1 =
      SkewMonoidAlgebra.coeff (ClopenCrossedProduct.coeff T k g) 1 := by
    rw [hfg]
  rw [ClopenCrossedProduct.coeff_apply, ClopenCrossedProduct.coeff_apply,
    Pestov91.CrossedProduct.C_apply, Pestov91.CrossedProduct.C_apply,
    SkewMonoidAlgebra.coeff_single, SkewMonoidAlgebra.coeff_single, Finsupp.single_eq_same,
    Finsupp.single_eq_same] at h
  exact (ClopenCoeff.of T k).injective h

theorem defectD_ne_zero : ClopenCrossedProduct.defectD shift (ZMod 2) isClopen_P ≠ 0 := by
  intro hd
  have hd' : ClopenCrossedProduct.coeff shift (ZMod 2) (LocallyConstant.charFn (ZMod 2)
      (isClopen_P.diff (ClopenCrossedProduct.isClopen_image shift isClopen_P))) = 0 := hd
  have hf := coeff_injective shift (ZMod 2)
    (hd'.trans (map_zero (ClopenCrossedProduct.coeff shift (ZMod 2))).symm)
  have hmem : pt 0 ∈ P \ shift '' P := by
    rw [P_diff_shift_image_P]
    exact Set.mem_singleton _
  have h := DFunLike.congr_fun hf (pt 0)
  have h0 : (0 : LocallyConstant TwoEnded (ZMod 2)) (pt 0) = 0 := rfl
  rw [h0, LocallyConstant.coe_charFn, Set.indicator_of_mem hmem, Pi.one_apply] at h
  exact one_ne_zero h

/-- **tex 1772–1773**: "Over $\F_2$ the ring is not directly finite". -/
theorem not_isDedekindFiniteMonoid :
    ¬ IsDedekindFiniteMonoid (ClopenCrossedProduct shift (ZMod 2)) := by
  intro hfin
  have hts := ClopenCrossedProduct.defectT_mul_defectS shift (ZMod 2) isClopen_P
    shift_image_P_subset
  have hst := ClopenCrossedProduct.defectS_mul_defectT shift (ZMod 2) isClopen_P
    shift_image_P_subset
  rw [mul_eq_one_symm hts] at hst
  exact defectD_ne_zero (sub_eq_self.1 hst.symm)

/-- **tex 1769–1773**, the first two sentences of the remark after `cor:dynamic-rank-budget`:
"For the shift on the two-ended compactification $X=\mathbb Z\cup\{-\infty,+\infty\}$, with its
endpoints fixed, the core consists of those endpoints.  Over $\F_2$ the ring is not directly
finite: $P=\{0,1,\dots\}\cup\{+\infty\}$ has defect $\{0\}$."

`X` is the two-ended compactification, compact, metrizable and totally disconnected; `T` is
`j ↦ j + 1` fixing both endpoints; `CR(T) = {−∞, +∞}`; `P` is clopen, `T(P) ⊆ P` and
`P ∖ T(P) = {0}`; and `R_X = LC(X, 𝔽₂) ⋊_T ℤ` is not directly finite. -/
def PrintedTwoEndedCoreAndDefect : Prop :=
  IsTwoEndedCompactification TwoEnded pt posInf negInf ∧
    TopologicalSpace.MetrizableSpace TwoEnded ∧ TotallyDisconnectedSpace TwoEnded ∧
    (∀ j : ℤ, shift (pt j) = pt (j + 1)) ∧ shift posInf = posInf ∧ shift negInf = negInf ∧
    Dynamics.chainRecurrentSet shift = {negInf, posInf} ∧
    IsClopen P ∧ P = Set.range (fun n : ℕ => pt n) ∪ {posInf} ∧ shift '' P ⊆ P ∧
    P \ shift '' P = {pt 0} ∧
    ¬ IsDedekindFiniteMonoid (ClopenCrossedProduct shift (ZMod 2))

theorem printedTwoEndedCoreAndDefect : PrintedTwoEndedCoreAndDefect :=
  ⟨isTwoEndedCompactification, inferInstance, inferInstance, shift_pt, shift_posInf,
    shift_negInf, chainRecurrentSet_shift, isClopen_P, P_eq, shift_image_P_subset,
    P_diff_shift_image_P, not_isDedekindFiniteMonoid⟩

end TwoEndedShift
