/-!
# The corners of a vertex touch are distinct G-faces

Osin (math/0411039v3, §9), Lemma 9.4, Case 1, across one relator cell `Π`: at a vertex touch of the
pocket walk, the vertex split cuts at `alpha in₁` and `alpha in₂` and merges `faceOf in₁` with
`faceOf in₂`.  `OsinPocketTouchCornerStatement` asks for an O-equivalent copy in which these are
distinct G-faces off the exterior, with the walk data transported.

The proof doubles the edge of each corner face along its in-dart (`exists_doubling`), first at `in₁`
and then at the image of `in₂`.  Each in-dart then lies on a G-cell digon.  The second doubling
moves the first digon by an injective face map that never meets the second digon, so the two
corner faces are distinct, and neither is the exterior or a relator face.

## Manuscript status

Infrastructure for `thm:hull` (tex 2121, Hull's small cancellation theorem, through Osin's
Lemma 9.4, Case 1).  It certifies no printed sentence on its own.
-/

namespace GroupApproximation.GGT.VanKampen.PocketTouchCorner

open EdgeInsertion SimpleClosedWalkSides Embedded
open scoped Classical

universe u w v

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}

/-- **One doubling at a walk dart `d₀` that is not a loop.**  An O-equivalent labelled copy, a dart
embedding keeping reversal, labels and vertices, the rotation away from `d₀`, and an injective face
map `ψ` such that: every other dart lands on the image of its face; `e d₀` lies on a face off the
exterior, off every relator face and off the image of `ψ`; the exterior and the relator faces are
images; and the exterior and cell `i` stay off the side, with the carrier of `i` mapped by `e`. -/
theorem exists_doubling (D : RelGenSet G Lambda) (X : DiscDiagram.{u, w, v} W)
    (hlabel : ∀ d, (symmetricLabelAlphabet D).IsLetter (X.label d))
    (walk : List X.toCombMap.Dart) (d₀ : X.toCombMap.Dart) (hd₀ : d₀ ∈ walk)
    (hloop : X.toCombMap.vertexOf d₀ ≠ X.toCombMap.vertexOf (X.toCombMap.alpha d₀))
    (hout : X.outerFace ∉ sideFaces X.toCombMap walk)
    (i : Fin X.rCellCount) (hi : (cell X i).face ∉ sideFaces X.toCombMap walk) :
    ∃ (X' : DiscDiagram.{u, w, v} W) (e : X.toCombMap.Dart ↪ X'.toCombMap.Dart)
      (i' : Fin X'.rCellCount),
      Nonempty (OEquivalentDiscDiagram X X') ∧
      (∀ d, (symmetricLabelAlphabet D).IsLetter (X'.label d)) ∧
      (∀ d, X'.toCombMap.alpha (e d) = e (X.toCombMap.alpha d)) ∧
      (∀ d, X'.label (e d) = X.label d) ∧
      (∀ d d', X'.toCombMap.vertexOf (e d) = X'.toCombMap.vertexOf (e d') ↔
        X.toCombMap.vertexOf d = X.toCombMap.vertexOf d') ∧
      (∀ z, X.toCombMap.sigma z ≠ d₀ → z ≠ X.toCombMap.alpha d₀ →
        X'.toCombMap.sigma (e z) = e (X.toCombMap.sigma z)) ∧
      ∃ ψ : X.toCombMap.Face → X'.toCombMap.Face, Function.Injective ψ ∧
        (∀ z, z ≠ d₀ → X'.toCombMap.faceOf (e z) = ψ (X.toCombMap.faceOf z)) ∧
        X'.toCombMap.faceOf (e d₀) ≠ X'.outerFace ∧
        (∀ C ∈ X'.relatorCells, C.face ≠ X'.toCombMap.faceOf (e d₀)) ∧
        (∀ g, ψ g ≠ X'.toCombMap.faceOf (e d₀)) ∧
        X'.outerFace = ψ X.outerFace ∧
        (∀ C ∈ X'.relatorCells, ∃ C₀ ∈ X.relatorCells, C.face = ψ C₀.face) ∧
        X'.outerFace ∉ sideFaces X'.toCombMap (walk.map e) ∧
        (cell X' i').face ∉ sideFaces X'.toCombMap (walk.map e) ∧
        cellDarts X' i' = (cellDarts X i).map e := by
  obtain ⟨j, hj⟩ := exists_dart_eq X d₀
  have hlen := one_lt_faceBoundary_length_of_not_loop hloop
  have hf : X.toCombMap.faceOf d₀ ≠ X.outerFace := faceOf_ne_of_not_mem hd₀ hout
  have hne : (cell X i).face ≠ X.toCombMap.faceOf d₀ := fun h => faceOf_ne_of_not_mem hd₀ hi h.symm
  have hdigon := faceOf_embedding_doubled X _ j hlen hf hj
  refine ⟨FaceEdgeDoubling.diagram X _ j hlen hf, (FaceEdgeDoubling.embedding X _ j hlen hf).darts,
    (FaceEdgeDoubling.cellMap X _ j hlen hf).indexEquiv i,
    ⟨FaceEdgeDoubling.oEquivalent X _ j hlen hf⟩, FaceEdgeDoubling.label_letter X _ j hlen hf hlabel,
    fun d => (FaceEdgeDoubling.embedding X _ j hlen hf).alpha d,
    fun d => (FaceEdgeDoubling.embedding X _ j hlen hf).label d,
    FaceEdgeDoubling.vertexOf_embedding_eq_iff X _ j hlen hf, ?_,
    FaceEdgeDoubling.faceImage X _ j hlen, FaceEdgeDoubling.faceImage_injective X _ j hlen, ?_, ?_,
    ?_, ?_, outerFace_eq_faceImage X _ j hlen hf, ?_, outerFace_not_mem_sideFaces X _ j hlen hf hout,
    cell_not_mem_sideFaces X _ j hlen hf i hi hne, cellDarts_indexEquiv X _ j hlen hf i hne⟩
  · intro z hz hz'
    refine sigma_embedding_of_ne X _ j hlen hf (by rw [hj]; exact hloop) z (by rw [hj]; exact hz)
      (by rw [hj]; exact hz')
  · intro z hz
    exact CellEdgeThickening.faceOf_embed_eq_faceImage X _ j hlen (by rw [hj]; exact hz)
  · rw [hdigon]
    exact digon_ne_outerFace X _ j hlen hf
  · intro C hC
    rw [hdigon]
    exact digon_not_cell X _ j hlen hf C hC
  · intro g
    rw [hdigon]
    exact FaceEdgeDoubling.faceImage_ne_digon X _ j hlen g
  · intro C hC
    obtain ⟨C₀, hC₀, rfl⟩ := List.mem_map.mp hC
    exact ⟨C₀, hC₀, rfl⟩

/-- **`OsinPocketTouchCornerStatement`, proved.**  Double along `in₁`, then along the image of
`in₂`. -/
theorem osinPocketTouchCorner : OsinPocketTouchCornerStatement.{u, w, v} := by
  intro G _ Lambda W D X hlabel walk _hwalk in₁ h₁ in₂ h₂ hne _hv hloop₁ hloop₂ hout i hi
  obtain ⟨X₁, e₁, i₁, ⟨eq₁⟩, hlabel₁, halpha₁, hlab₁, hvert₁, hsig₁, ψ₁, _hψ₁, _hface₁, hout₁',
    hcell₁', _hψne₁, _houter₁, _hcells₁, hout₁, hi₁, hcd₁⟩ :=
    exists_doubling D X hlabel walk in₁ h₁ hloop₁ hout i hi
  have h₂' : e₁ in₂ ∈ walk.map e₁ := List.mem_map_of_mem h₂
  have hloop₂' : X₁.toCombMap.vertexOf (e₁ in₂) ≠
      X₁.toCombMap.vertexOf (X₁.toCombMap.alpha (e₁ in₂)) := by
    rw [halpha₁, Ne, hvert₁]
    exact hloop₂
  obtain ⟨X₂, e₂, i₂, ⟨eq₂⟩, hlabel₂, halpha₂, hlab₂, hvert₂, hsig₂, ψ₂, hψ₂, hface₂, hout₂',
    hcell₂', hψne₂, houter₂, hcells₂, hout₂, hi₂, hcd₂⟩ :=
    exists_doubling D X₁ hlabel₁ (walk.map e₁) (e₁ in₂) h₂' hloop₂' hout₁ i₁ hi₁
  have hne₁ : e₁ in₁ ≠ e₁ in₂ := fun h => hne (e₁.injective h)
  have hw : walk.map (fun d => e₂ (e₁ d)) = (walk.map e₁).map e₂ := by
    rw [List.map_map]
    rfl
  refine ⟨X₂, fun d => e₂ (e₁ d), i₂, ⟨eq₁.trans eq₂⟩, hlabel₂, e₂.injective.comp e₁.injective,
    ?_, fun d => (hlab₂ _).trans (hlab₁ d), fun d d' => (hvert₂ _ _).trans (hvert₁ d d'), ?_, ?_,
    ?_, hout₂', ?_, hcell₂', by rw [hw]; exact hout₂, by rw [hw]; exact hi₂, ?_⟩
  · intro d
    rw [halpha₂, halpha₁]
  · intro d hd₁ hd₂ hs₁ hs₂
    have h₃ : X₁.toCombMap.sigma (e₁ d) ≠ e₁ in₂ := by
      rw [hsig₁ d hs₁ hd₁]
      exact fun h => hs₂ (e₁.injective h)
    have h₄ : e₁ d ≠ X₁.toCombMap.alpha (e₁ in₂) := by
      rw [halpha₁]
      exact fun h => hd₂ (e₁.injective h)
    rw [hsig₂ (e₁ d) h₃ h₄, hsig₁ d hs₁ hd₁]
  · rw [hface₂ (e₁ in₁) hne₁]
    exact hψne₂ _
  · rw [hface₂ (e₁ in₁) hne₁, houter₂]
    exact fun h => hout₁' (hψ₂ h)
  · intro C hC
    obtain ⟨C₀, hC₀, hCf⟩ := hcells₂ C hC
    rw [hCf, hface₂ (e₁ in₁) hne₁]
    exact fun h => hcell₁' C₀ hC₀ (hψ₂ h)
  · rw [hcd₂, hcd₁, List.map_map]
    rfl

end GroupApproximation.GGT.VanKampen.PocketTouchCorner
