/-!
# One doubling at a pocket corner

Osin (math/0411039v3, §9), Lemma 9.4, Case 1, a vertex touch of the pocket walk.  This module collects
what one doubling along a walk dart `d₀ = FaceEdgeDoubling.dart Δ f j` gives, for the corner statement
`OsinPocketTouchCornerStatement`:

* `facePerm_ne_self_of_not_loop` and `sigma_embedding_of_ne`: the rotation of an old dart `z` is
  kept unless `σ z = d₀` or `z = alpha d₀`.
* `faceOf_ne_outerFace_of_mem`: the face of a walk dart is not the exterior when the exterior is
  off the side of the walk.
* `outerFace_not_mem_sideFaces` and `cell_not_mem_sideFaces`, with `cellDarts_indexEquiv`: the
  exterior and the carrier of a cell off the side stay off the side of the embedded walk, and the
  carrier maps by the embedding.

## Manuscript status

Infrastructure for `thm:hull` (tex 2121, Hull's small cancellation theorem, through Osin's
Lemma 9.4, Case 1).  It certifies no printed sentence on its own.
-/

universe u w v

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}

/-- A dart that is not a loop is not fixed by face rotation. -/
theorem facePerm_ne_self_of_not_loop {M : CombMap.{v}} {d : M.Dart}
    (hloop : M.vertexOf d ≠ M.vertexOf (M.alpha d)) : M.facePerm d ≠ d := by
  intro hfix
  have hsig : M.sigma (M.alpha d) = d := by
    unfold CombMap.facePerm at hfix
    rwa [Equiv.Perm.mul_apply] at hfix
  exact hloop ((M.vertexOf_eq_iff _ _).mpr ⟨1, by rw [zpow_one]; exact hsig⟩).symm

/-- The face of a walk dart lies on the side of the walk. -/
theorem faceOf_mem_sideFaces {M : CombMap.{v}} {walk : List M.Dart} {d : M.Dart} (hd : d ∈ walk) :
    M.faceOf d ∈ sideFaces M walk :=
  (mem_sideFaces_iff M walk d).mpr ⟨d, hd, .refl _⟩

/-- The face of a walk dart is not a face off the side. -/
theorem faceOf_ne_of_not_mem {M : CombMap.{v}} {walk : List M.Dart} {d : M.Dart} (hd : d ∈ walk)
    {g : M.Face} (hg : g ∉ sideFaces M walk) : M.faceOf d ≠ g :=
  fun h => hg (h ▸ faceOf_mem_sideFaces hd)

section Step

variable (Delta : DiscDiagram.{u, w, v} W) (f : Delta.toCombMap.Face)
  (j : Fin (Delta.faceBoundary f).darts.length)
  (hlen : 1 < (Delta.faceBoundary f).darts.length) (hf : f ≠ Delta.outerFace)

/-- **The rotation of an old dart is kept** unless its successor is the doubled dart or it is the
reverse of the doubled dart. -/
theorem sigma_embedding_of_ne
    (hloop : Delta.toCombMap.vertexOf (FaceEdgeDoubling.dart Delta f j) ≠
      Delta.toCombMap.vertexOf (Delta.toCombMap.alpha (FaceEdgeDoubling.dart Delta f j)))
    (z : Delta.toCombMap.Dart) (hz : Delta.toCombMap.sigma z ≠ FaceEdgeDoubling.dart Delta f j)
    (hz' : z ≠ Delta.toCombMap.alpha (FaceEdgeDoubling.dart Delta f j)) :
    (FaceEdgeDoubling.diagram Delta f j hlen hf).toCombMap.sigma
        ((FaceEdgeDoubling.embedding Delta f j hlen hf).darts z) =
      (FaceEdgeDoubling.embedding Delta f j hlen hf).darts (Delta.toCombMap.sigma z) := by
  have hsecond := FaceEdgeDoubling.Holding.secondCorner_eq_facePerm Delta f j hlen
  have hab : firstCorner Delta.toCombMap (FaceEdgeDoubling.rebased Delta f j) ≠
      secondCorner Delta.toCombMap (FaceEdgeDoubling.rebased Delta f j)
        (FaceEdgeDoubling.second Delta f j hlen) := by
    rw [hsecond]
    exact (facePerm_ne_self_of_not_loop hloop).symm
  change (EdgeInsertion.toCombMap Delta.toCombMap
      (firstCorner Delta.toCombMap (FaceEdgeDoubling.rebased Delta f j))
      (secondCorner Delta.toCombMap (FaceEdgeDoubling.rebased Delta f j)
        (FaceEdgeDoubling.second Delta f j hlen))).sigma (embed Delta.toCombMap z) =
    embed Delta.toCombMap (Delta.toCombMap.sigma z)
  rw [EdgeInsertion.sigma_embed_apply hab z, if_neg hz, if_neg]
  rw [hsecond]
  intro h
  unfold CombMap.facePerm at h
  rw [Equiv.Perm.mul_apply] at h
  exact hz' (Delta.toCombMap.sigma.injective h)

/-- The exterior of the doubled map is the image of the old exterior. -/
theorem outerFace_eq_faceImage :
    (FaceEdgeDoubling.diagram Delta f j hlen hf).outerFace =
      FaceEdgeDoubling.faceImage Delta f j hlen Delta.outerFace :=
  (FaceEdgeDoubling.faceImage_of_ne Delta f j hlen hf.symm).symm

/-- **The exterior stays off the side.** -/
theorem outerFace_not_mem_sideFaces {walk : List Delta.toCombMap.Dart}
    (hout : Delta.outerFace ∉ sideFaces Delta.toCombMap walk) :
    (FaceEdgeDoubling.diagram Delta f j hlen hf).outerFace ∉
      sideFaces (FaceEdgeDoubling.diagram Delta f j hlen hf).toCombMap
        (walk.map (FaceEdgeDoubling.embedding Delta f j hlen hf).darts) := by
  rw [outerFace_eq_faceImage Delta f j hlen hf]
  exact not_mem_sideFaces_faceImage Delta f j hlen hf hf.symm hout

/-- The carrier face of a transported cell is the image of the old carrier face. -/
theorem cell_indexEquiv_face (i : Fin Delta.rCellCount) :
    (cell (FaceEdgeDoubling.diagram Delta f j hlen hf)
      ((FaceEdgeDoubling.cellMap Delta f j hlen hf).indexEquiv i)).face =
      FaceEdgeDoubling.faceImage Delta f j hlen (cell Delta i).face := by
  rw [(FaceEdgeDoubling.cellMap Delta f j hlen hf).indexed_cell i,
    (FaceEdgeDoubling.cellMap Delta f j hlen hf).face_eq]

/-- **The carrier of a cell off the side stays off the side.** -/
theorem cell_not_mem_sideFaces {walk : List Delta.toCombMap.Dart} (i : Fin Delta.rCellCount)
    (hi : (cell Delta i).face ∉ sideFaces Delta.toCombMap walk) (hne : (cell Delta i).face ≠ f) :
    (cell (FaceEdgeDoubling.diagram Delta f j hlen hf)
      ((FaceEdgeDoubling.cellMap Delta f j hlen hf).indexEquiv i)).face ∉
      sideFaces (FaceEdgeDoubling.diagram Delta f j hlen hf).toCombMap
        (walk.map (FaceEdgeDoubling.embedding Delta f j hlen hf).darts) := by
  rw [cell_indexEquiv_face Delta f j hlen hf i]
  exact not_mem_sideFaces_faceImage Delta f j hlen hf hne hi

/-- **The carrier of a cell other than the doubled face maps by the embedding.** -/
theorem cellDarts_indexEquiv (i : Fin Delta.rCellCount) (hne : (cell Delta i).face ≠ f) :
    cellDarts (FaceEdgeDoubling.diagram Delta f j hlen hf)
        ((FaceEdgeDoubling.cellMap Delta f j hlen hf).indexEquiv i) =
      (cellDarts Delta i).map (FaceEdgeDoubling.embedding Delta f j hlen hf).darts := by
  rw [FaceEdgeDoubling.cellDarts_eq Delta f j hlen hf i,
    FaceEdgeDoubling.carrierImage_of_ne Delta f j hlen hf hne]
  rfl

end Step
