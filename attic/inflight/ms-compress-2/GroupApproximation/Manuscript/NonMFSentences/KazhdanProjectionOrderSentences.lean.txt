/-!
# `lem:kazhdan-projection-order`: the faithful representation and the moved projection

`non_mf_groups_exist.tex`, proof of Lemma `lem:kazhdan-projection-order` (tex lines 478–488 at
68481e4d7):

> Fix a faithful nondegenerate representation of `B` on a Hilbert space `𝓗`, and let
> `Fix π(L)` be the space of vectors fixed by `π(L)`.  Then `P` is the orthogonal projection onto
> `Fix π(L)`.  [...]  Since `U*PU` is the orthogonal projection onto `U* Fix π(L)`, it follows
> that `U*PU ≤ P`.

Two sentences of this proof had carriers that were narrower than the printed text.

* `PrintedFaithfulNondegenerateRepresentationEveryUnital` is the first sentence for every unital
  C*-algebra `B`, in every universe.  The earlier carrier
  `OneSidedMFRadical.PrintedFaithfulNondegenerateRepresentation` binds `B : Type` and
  `[Nontrivial B]`.  A nontrivial `B` acts on the Hilbert sum of the GNS spaces of its states, as
  before.  A trivial `B` acts on the zero Hilbert space, which is the only space a unital
  representation of the zero algebra can live on.  The two cases are exhaustive
  (`subsingleton_or_nontrivial`), and the statement records that as its third clause.
* `PrintedConjugatedProjectionOrder` is the last sentence on the Hilbert space itself: `U*PU` is an
  orthogonal projection, its fixed vectors are exactly `U* Fix π(L)`, and `U*PU ≤ P` in the Loewner
  order of `E →L[ℂ] E` (Mathlib's `ContinuousLinearMap.instLoewnerPartialOrder`).  The earlier
  carriers state only the absorption identity `P (U*PU) = U*PU`.
-/

namespace GroupApproximation

namespace NonMFSentences

open GroupApproximation.Manuscript.OneSidedMFRadical

universe u v

/-! ## A faithful nondegenerate representation of every unital C*-algebra -/

/-- The zero Hilbert space. -/
abbrev ZeroHilbertSpace : Type := EuclideanSpace ℂ (Fin 0)

instance : Subsingleton ZeroHilbertSpace :=
  ⟨fun x y ↦ WithLp.ext_iff.mpr (funext fun i ↦ i.elim0)⟩

/-- The unital `⋆`-homomorphism of any unital C*-algebra into the operators on the zero Hilbert
space.  It is faithful exactly when the algebra is trivial. -/
noncomputable def zeroSpaceRep (B : Type u) [CStarAlgebra B] :
    B →⋆ₐ[ℂ] (ZeroHilbertSpace →L[ℂ] ZeroHilbertSpace) where
  toFun _ := 0
  map_one' := ContinuousLinearMap.ext fun _ ↦ Subsingleton.elim _ _
  map_mul' _ _ := ContinuousLinearMap.ext fun _ ↦ Subsingleton.elim _ _
  map_zero' := rfl
  map_add' _ _ := ContinuousLinearMap.ext fun _ ↦ Subsingleton.elim _ _
  commutes' _ := ContinuousLinearMap.ext fun _ ↦ Subsingleton.elim _ _
  map_star' _ := ContinuousLinearMap.ext fun _ ↦ Subsingleton.elim _ _

/-- **"Fix a faithful nondegenerate representation of `B` on a Hilbert space `𝓗`."**

For every unital C*-algebra `B` in every universe, a unital, injective and nondegenerate
`⋆`-representation on a Hilbert space: the universal GNS representation when `B` is nontrivial,
the zero representation when `B` is trivial, and one of the two cases always holds.
Nondegeneracy is spelled as the printed condition that the vectors `π(b)η` exhaust the space. -/
def PrintedFaithfulNondegenerateRepresentationEveryUnital : Prop :=
  ∀ (B : Type u) [CStarAlgebra B],
    (∀ [Nontrivial B],
      CompleteSpace (CStarState.UniversalGNSSpace B) ∧
        Function.Injective (CStarState.universalGNSStarAlgHom (A := B)) ∧
        CStarState.universalGNSStarAlgHom (A := B) 1 = 1 ∧
        ∀ xi : CStarState.UniversalGNSSpace B,
          ∃ (b : B) (eta : CStarState.UniversalGNSSpace B),
            CStarState.universalGNSStarAlgHom b eta = xi) ∧
    (Subsingleton B →
      Function.Injective (zeroSpaceRep B) ∧ zeroSpaceRep B 1 = 1 ∧
        ∀ xi : ZeroHilbertSpace, ∃ (b : B) (eta : ZeroHilbertSpace), zeroSpaceRep B b eta = xi) ∧
    (Subsingleton B ∨ Nontrivial B)

/-- Closed proof that every unital C*-algebra has a faithful nondegenerate Hilbert-space
representation. -/
theorem manuscriptPrintedFaithfulNondegenerateRepresentationEveryUnital :
    PrintedFaithfulNondegenerateRepresentationEveryUnital.{u} := by
  intro B _
  refine ⟨?_, ?_, subsingleton_or_nontrivial B⟩
  · intro _
    refine ⟨inferInstance, CStarState.universalGNSStarAlgHom_injective,
      map_one (CStarState.universalGNSStarAlgHom (A := B)), ?_⟩
    intro xi
    refine ⟨1, xi, ?_⟩
    calc (CStarState.universalGNSStarAlgHom (A := B)) 1 xi
        = (1 : CStarState.UniversalGNSSpace B →L[ℂ] CStarState.UniversalGNSSpace B) xi := by
          rw [map_one (CStarState.universalGNSStarAlgHom (A := B))]
      _ = xi := rfl
  · intro _
    exact ⟨fun a b _ ↦ Subsingleton.elim a b, map_one (zeroSpaceRep B),
      fun xi ↦ ⟨0, xi, Subsingleton.elim _ _⟩⟩

/-! ## The moved projection and the operator order -/

/-- **"Since `U*PU` is the orthogonal projection onto `U* Fix π(L)`, it follows that
`U*PU ≤ P`."**

On a complex Hilbert space `E`, let `π : L → U(B(E))` be a homomorphism, let `P` be the orthogonal
projection whose fixed vectors are `Fix π(L)`, and let `U` be a unitary with
`U π(L) U* ⊆ π(L)`.  Then `U*PU` is an orthogonal projection, its fixed vectors are the `x` with
`Ux ∈ Fix π(L)`, that is `U* Fix π(L)`, and `U*PU ≤ P` in the Loewner order of `E →L[ℂ] E`. -/
def PrintedConjugatedProjectionOrder : Prop :=
  ∀ (L : Type u) [Group L] (E : Type v) [NormedAddCommGroup E] [InnerProductSpace ℂ E]
    [CompleteSpace E] (pi : L →* unitary (E →L[ℂ] E)) (U : unitary (E →L[ℂ] E))
    (P : E →L[ℂ] E),
    IsStarProjection P →
    (∀ x : E, P x = x ↔ ∀ l : L, ((pi l : unitary (E →L[ℂ] E)) : E →L[ℂ] E) x = x) →
    (∀ l : L, ∃ l' : L,
      ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) * ((pi l : unitary (E →L[ℂ] E)) : E →L[ℂ] E) *
          star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E)
        = ((pi l' : unitary (E →L[ℂ] E)) : E →L[ℂ] E)) →
    IsStarProjection (star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) * P *
        ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E)) ∧
      (∀ x : E,
        (star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) * P *
            ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E)) x = x ↔
          ∀ l : L, ((pi l : unitary (E →L[ℂ] E)) : E →L[ℂ] E)
              (((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) x)
            = ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) x) ∧
      star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) * P *
          ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) ≤ P

/-- Closed proof of the last sentence of the proof of `lem:kazhdan-projection-order`. -/
theorem manuscriptPrintedConjugatedProjectionOrder :
    PrintedConjugatedProjectionOrder.{u, v} := by
  intro L _ E _ _ _ pi U P hP hfix hcomp
  have hVV : ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) *
      star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) = 1 :=
    Unitary.mul_star_self_of_mem U.2
  have hVV' : star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) *
      ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) = 1 :=
    Unitary.star_mul_self_of_mem U.2
  have hQ : IsStarProjection (star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) * P *
      ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E)) :=
    isStarProjection_star_conj hP hVV
  have hmoved := manuscriptSentence_movedFixedSpaceInclusionAndProjectionOrder pi U hcomp P
    hP.isIdempotentElem.eq hfix
  refine ⟨hQ, ?_, ?_⟩
  · intro x
    have hVx : ∀ y : E, ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E)
        (star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) y) = y :=
      fun y ↦ congrArg (fun T : E →L[ℂ] E ↦ T y) hVV
    have hVx' : ∀ y : E, star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E)
        (((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) y) = y :=
      fun y ↦ congrArg (fun T : E →L[ℂ] E ↦ T y) hVV'
    constructor
    · intro h
      have h2 : ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E)
          (star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E)
            (P (((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) x)))
          = ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) x :=
        congrArg (fun y : E ↦ ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) y) h
      rw [hVx] at h2
      exact (hfix _).mp h2
    · intro hl
      have h1 : P (((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) x)
          = ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) x := (hfix _).mpr hl
      show star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E)
          (P (((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) x)) = x
      rw [h1, hVx']
  · have habsP : P * (star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) * P *
        ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E))
        = star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) * P *
          ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) :=
      ContinuousLinearMap.ext fun x ↦ hmoved.2 x
    have habs : (star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) * P *
        ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E)) * P
        = star ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) * P *
          ((U : unitary (E →L[ℂ] E)) : E →L[ℂ] E) := by
      have h := congrArg star habsP
      rwa [star_mul, hQ.isSelfAdjoint.star_eq, hP.isSelfAdjoint.star_eq] at h
    exact (hQ.le_iff_mul_eq_left hP).mpr habs
