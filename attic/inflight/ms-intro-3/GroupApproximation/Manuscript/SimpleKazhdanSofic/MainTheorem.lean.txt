/-!
# `thm:main` of `simple_kazhdan_sofic_group.tex`

The note at origin/main bf961c128, Theorem `thm:main` (tex l.44–53):

> Let `X ⊆ A^ℤ` be an infinite minimal subshift over a finite alphabet, with shift `T`.  Then
> `G_X = EL₃(LC(X, F₂) ⋊_T ℤ)` is an infinite finitely generated simple group with Kazhdan's
> property (T) that is locally embeddable into finite groups (LEF).  In particular, it is sofic
> and hyperlinear.

`PrintedSimpleKazhdanSoficMain` states it over the setting of `SimpleKazhdanSofic.Setting`.  The
proof sections of the note supply five facts about the ring `R = LC(X, F₂) ⋊_T ℤ`, stated here as
named propositions at the printed generality, with their owners:

* `PrintedSubshiftShiftFree`: `T` has no periodic points (tex l.78; ct-involution);
* `PrintedMinimalSubshiftDenseOrbits`: every orbit is dense (tex l.45, minimality; ms-intro-3);
* `PrintedSubshiftRingFinitelyGenerated`: `R` is generated by `u^{±1}` and the letter indicators
  (tex l.84–86; ct-involution);
* `PrintedSubshiftRingCentre`: `Z(R) = F₂` (tex l.110–113; ct-involution);
* `PrintedMinimalSubshiftRingLEF`: `R` is LEF (tex l.118–149; ms-units).

`printedSimpleKazhdanSoficMain_of_pieces` assembles the theorem from them and the general results
of `GroupApproximation/Pestov91`:
* property (T) of `EL₃` over a finitely generated ring (Ershov–Jaikin-Zapirain, proved in this
  repository, `Pestov91.elementary_hasKazhdanPropertyT`), hence finite generation;
* simplicity of `EL₃(R)/Z` for a free minimal action (`Pestov91.isSimpleGroup_elementaryModCentre_of_minimal_free`),
  and `Z(EL₃(R)) = 1` when `Z(R) = {0, 1}` (`Pestov91.elementaryModCentreEquiv`);
* LEF of `EL₃` over an LEF ring, then soficity and hyperlinearity.

## Manuscript status

Endpoint of `thm:main` over the five named propositions above; closes as they are proved.
-/

namespace GroupApproximation
namespace SimpleKazhdanSofic

open SymbolicDynamics.FullShift Set

/-- **Theorem `thm:main` as printed** (tex l.44–53). -/
def PrintedSimpleKazhdanSoficMain : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Finite A] (S : Subshift A ℤ),
    Infinite S.carrier → IsMinimal S →
      Infinite (G S) ∧ Group.FG (G S) ∧ IsSimpleGroup (G S) ∧
        HasKazhdanPropertyT.{0, 0} (G S) ∧ IsLEF (G S) ∧ IsSofic (G S) ∧ IsHyperlinear (G S)

/-- **No periodic points** (tex l.78): "Minimality and infiniteness imply that `T` has no periodic
points." -/
def PrintedSubshiftShiftFree : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Finite A] (S : Subshift A ℤ),
    Infinite S.carrier → IsMinimal S →
      ∀ j : ℤ, j ≠ 0 → ∀ x : S.carrier, (subshiftHomeo S ^ j) x ≠ x

/-- **Dense orbits** (tex l.45): in a minimal subshift every orbit of the shift is dense. -/
def PrintedMinimalSubshiftDenseOrbits : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Finite A] (S : Subshift A ℤ),
    IsMinimal S → ∀ x : S.carrier, Dense (Set.range fun j : ℤ => (subshiftHomeo S ^ j) x)

/-- **Finite generation of the ring** (tex l.84–86): `R` is generated by `u`, `u⁻¹` and the letter
indicators, so it is a finitely generated ring. -/
def PrintedSubshiftRingFinitelyGenerated : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Finite A] (S : Subshift A ℤ),
    IsFinitelyGeneratedRing (R S)

/-- **The centre of the ring** (tex l.102, l.110–113): `Z(R) = F₂`. -/
def PrintedSubshiftRingCentre : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Finite A] (S : Subshift A ℤ),
    Infinite S.carrier → IsMinimal S →
      ∀ c : R S, (∀ r : R S, r * c = c * r) → c = 0 ∨ c = 1

/-- **LEF of the ring** (tex l.118–149): the finite periodic models make `R` an LEF ring. -/
def PrintedMinimalSubshiftRingLEF : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Finite A] (S : Subshift A ℤ),
    Infinite S.carrier → IsMinimal S → Pestov91.IsLEFRing (R S)

/-- **`thm:main` from the five named facts about the ring.** -/
theorem printedSimpleKazhdanSoficMain_of_pieces
    (hfree : PrintedSubshiftShiftFree) (hdense : PrintedMinimalSubshiftDenseOrbits)
    (hfg : PrintedSubshiftRingFinitelyGenerated) (hcentre : PrintedSubshiftRingCentre)
    (hlef : PrintedMinimalSubshiftRingLEF) : PrintedSimpleKazhdanSoficMain := by
  intro A _ _ _ S hinf hmin
  haveI : CompactSpace S.carrier := isCompact_iff_compactSpace.mp S.isClosed.isCompact
  haveI : Nontrivial (ZMod 2) := nontrivial_of_ne (0 : ZMod 2) 1 (by decide)
  haveI : Nontrivial (LocallyConstant S.carrier (ZMod 2)) := Pestov91.locallyConstant_nontrivial
  haveI : Nontrivial (ClopenCoeff (subshiftHomeo S) (ZMod 2)) :=
    inferInstanceAs (Nontrivial (LocallyConstant S.carrier (ZMod 2)))
  haveI : Infinite (R S) := Pestov91.skewMonoidAlgebra_infinite
  set T := subshiftHomeo S with hTdef
  -- Property (T), finite generation, infinitude.
  have hT : HasKazhdanPropertyT.{0, 0} (G S) :=
    Pestov91.elementary_hasKazhdanPropertyT (hfg A S) (le_refl 3)
  have hFG : Group.FG (G S) := KazhdanFiniteGeneration.fg_of_hasKazhdanPropertyT (G := G S) hT
  have hInf : Infinite (G S) :=
    elementaryGroup_infinite (ι := Fin 3) (R := R S) 0 1 (by decide)
  -- LEF, soficity, hyperlinearity.
  have hLEF : IsLEF (G S) := Pestov91.isLEF_elementaryGroup (ι := Fin 3) (hlef A S hinf hmin)
  -- Simplicity of `EL₃(R) / Z` for the free minimal action, then `Z = 1`.
  have hfree' : ∀ j : ℤ, j ≠ 0 → ∀ x : S.carrier, (T ^ (-j)) x ≠ x :=
    fun j hj x => hfree A S hinf hmin (-j) (neg_ne_zero.mpr hj) x
  have hmin' : ∀ x : S.carrier, Dense (Set.range fun j : ℤ => (T ^ (-j)) x) := by
    intro x
    refine (hdense A S hmin x).mono ?_
    rintro _ ⟨j, rfl⟩
    exact ⟨-j, by simp only [neg_neg]⟩
  have hconj : ∀ (j : ℤ) (a : LocallyConstant S.carrier (ZMod 2)),
      ((ClopenCrossedProduct.unit T (ZMod 2) ^ j : (R S)ˣ) : R S) *
          ClopenCrossedProduct.coeff T (ZMod 2) a =
        ClopenCrossedProduct.coeff T (ZMod 2)
            (LocallyConstant.comap ⟨⇑(T ^ (-j)), (T ^ (-j)).continuous⟩ a) *
          ((ClopenCrossedProduct.unit T (ZMod 2) ^ j : (R S)ˣ) : R S) := by
    intro j a
    have h := Pestov91.CrossedProduct.unit_zpow_mul_C (A := ClopenCoeff T (ZMod 2)) j
      (ClopenCoeff.of T (ZMod 2) a)
    rw [ClopenCoeff.smul_of] at h
    exact h
  have hsimpleQ :=
    Pestov91.isSimpleGroup_elementaryModCentre_of_minimal_free (R := R S) (X := S.carrier)
      (K := ZMod 2) (ClopenCrossedProduct.coeff T (ZMod 2)) (ClopenCrossedProduct.unit T (ZMod 2))
      (fun j => ⇑(T ^ (-j))) (fun j => (T ^ (-j)).continuous) hfree' hmin'
      (fun j a => LocallyConstant.comap ⟨⇑(T ^ (-j)), (T ^ (-j)).continuous⟩ a)
      (fun _ _ _ => rfl) hconj (ClopenCrossedProduct.exists_sum_coeff_mul_unit_zpow T (ZMod 2))
  haveI := hsimpleQ
  have hsimple : IsSimpleGroup (G S) :=
    (Pestov91.elementaryModCentreEquiv (ι := Fin 3) (hcentre A S hinf hmin)).symm.isSimpleGroup
  exact ⟨hInf, hFG, hsimple, hT, hLEF, isSofic_of_isLEF hLEF, Pestov91.isHyperlinear_of_isLEF hLEF⟩
