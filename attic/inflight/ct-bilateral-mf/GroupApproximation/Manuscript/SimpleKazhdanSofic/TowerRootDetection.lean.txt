/-!
# A root that does not commute with a nontrivial element of a normal subgroup

`simple_kazhdan_sofic_group.tex` at bf961c128, "A finite tower detects every normal subgroup",
tex l.175–187 and l.194:

> Let $1\ne N\trianglelefteq G$, and choose $1\ne g\in N$.
> Let $w\ge0$ bound the absolute values of the exponents in all entries of $g$ and $g^{-1}$.
> [...] Some matrix $h=e_{ij}(s)$, $s\in\{e_C,e_Cu,e_Cu^{-1}:C\in\mathcal P\}$, does not commute
> with $g$. Otherwise, by~\eqref{eq:elementary}, the coefficients $s$ for which $g$ commutes with
> every $e_{ij}(s)$ form a subring containing all ring generators, so $g\in Z(G)=\{1\}$.
> [...] Set $k=[g,h]\in N\setminus\{1\}$ [...]

The printed argument uses only that the coefficients commuting with `g` are closed under the ring
operations (`Pestov91.SimpleModCentre.commutingSubring`, from the elementary identities) and that
the finite set of coefficients generates the ring.  This module proves the step for any
coefficient ring and any finite index type with at least three elements:

* `exists_elGen_not_commute_of_closure_eq_top`: a non-central `g` fails to commute with some root
  whose coefficient lies in a set `D` that generates `R` as a non-unital ring;
* `commutatorElement_mem_ne_one`: `k = [g, h] ∈ N ∖ {1}` for `g ∈ N` not commuting with `h`;
* `exists_root_commutator_mem_ne_one`: the combination, for a group with trivial centre;
* `exists_exponent_bound`, `exists_matrix_exponent_bound`: the bound `w` (tex l.176–177).
-/

section Roots

variable {ι R : Type*} [Fintype ι] [DecidableEq ι] [Ring R]

/-- **Some root does not commute with `g`** (tex l.180–187): if `g` is not central and `D`
generates `R` as a non-unital ring, some `e_{ij}(s)` with `s ∈ D` does not commute with `g`.
Otherwise every `s ∈ D` lies in the non-unital subring of coefficients commuting with `g`, so all
of `R` does, and `g` commutes with every root, hence is central. -/
theorem exists_elGen_not_commute_of_closure_eq_top (hcard : 3 ≤ Fintype.card ι)
    {g : elementaryGroup ι R} (hg : g ∉ Subgroup.center (elementaryGroup ι R)) {D : Set R}
    (hD : NonUnitalSubring.closure D = ⊤) :
    ∃ s ∈ D, ∃ (i j : ι) (hij : i ≠ j), ¬ Commute g (elGen i j hij s) := by
  by_contra h
  apply hg
  have hsub : D ⊆ (SimpleModCentre.commutingSubring hcard g : Set R) := by
    intro s hs
    show ∀ (i j : ι) (hij : i ≠ j), Commute g (elGen i j hij s)
    intro i j hij
    by_contra hc
    exact h ⟨s, hs, i, j, hij, hc⟩
  have hle : NonUnitalSubring.closure D ≤ SimpleModCentre.commutingSubring hcard g :=
    NonUnitalSubring.closure_le.mpr hsub
  rw [hD] at hle
  refine SimpleModCentre.mem_center_of_commute_elGen g fun i j hij s => ?_
  have hmem : s ∈ SimpleModCentre.commutingSubring hcard g := hle (NonUnitalSubring.mem_top s)
  exact hmem i j hij

end Roots

/-- **`k = [g, h] ∈ N ∖ {1}`** (tex l.194): a commutator with its first entry in a normal
subgroup lies there, and it is `1` exactly when the two commute. -/
theorem commutatorElement_mem_ne_one {G : Type*} [Group G] (N : Subgroup G) [N.Normal]
    {g h : G} (hg : g ∈ N) (hgh : ¬ Commute g h) : ⁅g, h⁆ ∈ N ∧ ⁅g, h⁆ ≠ 1 := by
  refine ⟨?_, fun h1 => hgh (commutatorElement_eq_one_iff_commute.mp h1)⟩
  rw [commutatorElement_def]
  exact SimpleModCentre.normal_comm_mem_left N hg h

/-- **tex l.175–187 and l.194 together**: in `EL_ι(R)` with trivial centre, for `1 ≠ g ∈ N ⊴ EL_ι(R)`
and a non-unitally generating set `D` of coefficients, some root `h = e_{ij}(s)`, `s ∈ D`, gives
`k = [g, h] ∈ N ∖ {1}`. -/
theorem exists_root_commutator_mem_ne_one {ι R : Type*} [Fintype ι] [DecidableEq ι] [Ring R]
    (hcard : 3 ≤ Fintype.card ι) (hcenter : Subgroup.center (elementaryGroup ι R) = ⊥)
    (N : Subgroup (elementaryGroup ι R)) [N.Normal] {g : elementaryGroup ι R} (hgN : g ∈ N)
    (hg1 : g ≠ 1) {D : Set R} (hD : NonUnitalSubring.closure D = ⊤) :
    ∃ s ∈ D, ∃ (i j : ι) (hij : i ≠ j),
      ⁅g, elGen i j hij s⁆ ∈ N ∧ ⁅g, elGen i j hij s⁆ ≠ 1 := by
  have hgc : g ∉ Subgroup.center (elementaryGroup ι R) := by
    rw [hcenter, Subgroup.mem_bot]
    exact hg1
  obtain ⟨s, hs, i, j, hij, hc⟩ := exists_elGen_not_commute_of_closure_eq_top hcard hgc hD
  exact ⟨s, hs, i, j, hij, commutatorElement_mem_ne_one N hgN hc⟩

section Exponents

/-- **A common exponent bound** for finitely many Laurent polynomials (tex l.176–177). -/
theorem exists_exponent_bound {A : Type*} [Ring A]
    (s : Finset (SkewMonoidAlgebra A (Multiplicative ℤ))) :
    ∃ w : ℕ, ∀ x ∈ s, ∀ j ∈ x.support, |Multiplicative.toAdd j| ≤ (w : ℤ) := by
  refine ⟨s.sup fun x => x.support.sup fun j => (Multiplicative.toAdd j).natAbs,
    fun x hx j hj => ?_⟩
  rw [Int.abs_eq_natAbs]
  exact_mod_cast (Finset.le_sup (f := fun j => (Multiplicative.toAdd j).natAbs) hj).trans
    (Finset.le_sup (f := fun x => x.support.sup fun j => (Multiplicative.toAdd j).natAbs) hx)

/-- **tex l.176–177**: `w ≥ 0` bounds the absolute values of the exponents in all entries of `g`
and `g⁻¹`. -/
theorem exists_matrix_exponent_bound {ι A : Type*} [Fintype ι] [DecidableEq ι] [Ring A]
    [MulSemiringAction (Multiplicative ℤ) A]
    (g : (Matrix ι ι (SkewMonoidAlgebra A (Multiplicative ℤ)))ˣ) :
    ∃ w : ℕ, ∀ (p q : ι) (j : Multiplicative ℤ),
      (j ∈ ((g : Matrix ι ι _) p q).support ∨ j ∈ ((↑g⁻¹ : Matrix ι ι _) p q).support) →
        |Multiplicative.toAdd j| ≤ (w : ℤ) := by
  classical
  obtain ⟨w, hw⟩ := exists_exponent_bound
    ((Finset.univ : Finset (ι × ι)).image (fun pq => (g : Matrix ι ι _) pq.1 pq.2) ∪
      (Finset.univ : Finset (ι × ι)).image (fun pq => (↑g⁻¹ : Matrix ι ι _) pq.1 pq.2))
  refine ⟨w, fun p q j hj => ?_⟩
  rcases hj with hj | hj
  · exact hw _ (Finset.mem_union_left _
      (Finset.mem_image.mpr ⟨(p, q), Finset.mem_univ _, rfl⟩)) j hj
  · exact hw _ (Finset.mem_union_right _
      (Finset.mem_image.mpr ⟨(p, q), Finset.mem_univ _, rfl⟩)) j hj

end Exponents
