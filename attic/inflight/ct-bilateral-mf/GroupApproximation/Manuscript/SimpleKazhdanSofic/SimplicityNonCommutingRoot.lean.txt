/-!
# A root `e_{ij}(e_V)` that does not commute with `g`

`simple_kazhdan_sofic_group.tex` at e80dcf20a, subsection "Simplicity", tex l.189–196:

> Some $h=e_{ij}(e_V)$ with $V$ small does not commute with $g$.
> Otherwise $g$ commutes with $e_{ij}(e_V)$ for every clopen $V$, by~\eqref{eq:elementary}.
> Taking $V=X$ gives $g=cI_3$ with $c=\sum_jc_ju^j\in R$. Then $e_Vc-ce_V=\sum_jc_j(e_V-e_{T^jV})u^j$
> vanishes for every clopen $V$. Choosing $V$ to contain $z$ but not $T^{-j}z$, we get $c_j(z)=0$
> for all $z\in X$ and $j\ne0$. So $c\in\LC(X,\F_2)$, and comparing constant coefficients in
> $cc^{-1}=1$ gives $c=1$ and $g=1$.

* `scalar_of_commute_elGen_one`: commuting with every `e_{ij}(1)` makes `g = cI₃` (tex l.191);
* `coeff_eq_zero_of_commute_charFn`: `e_V c = c e_V` for every clopen `V` gives `c_j = 0`, `j ≠ 0`
  (tex l.192–194);
* `eq_one_of_commute_elGen_charFn`: if `g` commutes with every `e_{ij}(e_V)`, then `g = 1`
  (tex l.190–196);
* `exists_not_commute_elGen_charFn_of_decomposition`: for `1 ≠ g` and a class of "small" clopen sets
  of which every clopen set is a finite disjoint union, some `e_{ij}(e_V)` with `V` small does not
  commute with `g` (tex l.189–190).
-/

section Matrices

variable {ι R : Type*} [Fintype ι] [DecidableEq ι] [Ring R]

/-- A root commuting with `g`, as a matrix identity. -/
theorem single_mul_eq_mul_single_of_commute_elGen {g : elementaryGroup ι R} {i j : ι}
    (hij : i ≠ j) {a : R} (hc : Commute g (elGen i j hij a)) :
    Matrix.single i j a * ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) =
      ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) * Matrix.single i j a := by
  have hm : ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) * (1 + Matrix.single i j a) =
      (1 + Matrix.single i j a) * ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) :=
    congrArg (fun z : elementaryGroup ι R => ((z : (Matrix ι ι R)ˣ) : Matrix ι ι R)) hc.eq
  rw [add_mul, mul_add, one_mul, mul_one] at hm
  exact (add_left_cancel hm).symm

/-- **tex l.191**: commuting with every `e_{ij}(1)` makes `g` a scalar matrix. -/
theorem scalar_of_commute_elGen_one {g : elementaryGroup ι R}
    (hg : ∀ (i j : ι) (hij : i ≠ j), Commute g (elGen i j hij 1)) :
    ∃ c : R, ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) = Matrix.scalar ι c := by
  obtain ⟨c, hc⟩ := Matrix.mem_range_scalar_of_commute_single
    (M := ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R)) fun i j hij =>
      single_mul_eq_mul_single_of_commute_elGen hij (hg i j hij)
  exact ⟨c, hc.symm⟩

/-- A scalar matrix `cI` commuting with `E_{ij}(a)` gives `a c = c a`. -/
theorem mul_comm_of_single_mul_scalar {i j : ι} {a c : R}
    (h : Matrix.single i j a * Matrix.scalar ι c = Matrix.scalar ι c * Matrix.single i j a) :
    a * c = c * a := by
  simpa [Matrix.scalar_apply, Matrix.mul_diagonal, Matrix.diagonal_mul] using
    congrFun (congrFun h i) j

end Matrices

section Coefficients

variable {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type*) [Ring k]

/-- **tex l.192–194**: if `e_V c = c e_V` for every clopen `V`, the Laurent coefficients `c_j`,
`j ≠ 0`, vanish. -/
theorem coeff_eq_zero_of_commute_charFn [TotallySeparatedSpace X]
    (hfree : ∀ j : ℤ, j ≠ 0 → ∀ x : X, (T ^ j) x ≠ x) {c : ClopenCrossedProduct T k}
    (hc : ∀ (V : Set X) (hV : IsClopen V),
      coeff T k (LocallyConstant.charFn k hV) * c = c * coeff T k (LocallyConstant.charFn k hV))
    {g : Multiplicative ℤ} (hg : g ≠ 1) : SkewMonoidAlgebra.coeff c g = 0 := by
  refine (ClopenCoeff.of T k).symm.injective (LocallyConstant.ext fun x => ?_)
  rw [map_zero, LocallyConstant.zero_apply]
  have hj : -toAdd g ≠ 0 := neg_ne_zero.2 fun h => hg (by rw [← ofAdd_toAdd g, h, ofAdd_zero])
  obtain ⟨W, hW, hxW, hTxW⟩ := exists_isClopen_of_totally_separated (hfree (-toAdd g) hj x).symm
  have hL : (ClopenCoeff.of T k).symm
      (SkewMonoidAlgebra.coeff (coeff T k (LocallyConstant.charFn k hW) * c) g) x =
      (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff c g) x := by
    rw [coeff_apply, Pestov91.CrossedProduct.C_apply, SkewMonoidAlgebra.coeff_single_one_mul,
      map_mul, LocallyConstant.mul_apply, RingEquiv.symm_apply_apply, LocallyConstant.coe_charFn,
      Set.indicator_of_mem hxW, Pi.one_apply, one_mul]
  have hR : (ClopenCoeff.of T k).symm
      (SkewMonoidAlgebra.coeff (c * coeff T k (LocallyConstant.charFn k hW)) g) x = 0 := by
    rw [coeff_apply, Pestov91.CrossedProduct.C_apply, SkewMonoidAlgebra.coeff_mul_single_one,
      map_mul, LocallyConstant.mul_apply, of_symm_smul_apply, RingEquiv.symm_apply_apply,
      LocallyConstant.coe_charFn, Set.indicator_of_notMem hTxW, mul_zero]
  rw [hc] at hL
  exact hL.symm.trans hR

/-- `c = f₀` when every coefficient away from `u⁰` vanishes. -/
theorem eq_coeff_of_coeff_eq_zero {c : ClopenCrossedProduct T k}
    (hzero : ∀ g : Multiplicative ℤ, g ≠ 1 → SkewMonoidAlgebra.coeff c g = 0) :
    c = coeff T k ((ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff c 1)) := by
  classical
  refine SkewMonoidAlgebra.ext fun g => ?_
  rw [coeff_apply, RingEquiv.apply_symm_apply, Pestov91.CrossedProduct.C_apply,
    SkewMonoidAlgebra.coeff_single_apply]
  by_cases hg : (1 : Multiplicative ℤ) = g
  · rw [if_pos hg, hg]
  · rw [if_neg hg, hzero g fun h => hg h.symm]

end Coefficients

section Group

variable {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X)

/-- The matrix of `g` commutes with `E_{ij}(e_V)` whenever `g` commutes with `e_{ij}(e_V)`; with
`g = cI`, `e_V c = c e_V`. -/
theorem commute_charFn_of_scalar {g : elementaryGroup (Fin 3) (ClopenCrossedProduct T (ZMod 2))}
    {c : ClopenCrossedProduct T (ZMod 2)}
    (hM : ((g : (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2)))ˣ) :
      Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2))) = Matrix.scalar (Fin 3) c)
    (hg : ∀ (V : Set X) (hV : IsClopen V) (i j : Fin 3) (hij : i ≠ j),
      Commute g (elGen i j hij (coeff T (ZMod 2) (LocallyConstant.charFn (ZMod 2) hV))))
    (V : Set X) (hV : IsClopen V) :
    coeff T (ZMod 2) (LocallyConstant.charFn (ZMod 2) hV) * c =
      c * coeff T (ZMod 2) (LocallyConstant.charFn (ZMod 2) hV) := by
  have h := single_mul_eq_mul_single_of_commute_elGen (show (0 : Fin 3) ≠ 1 by decide)
    (hg V hV 0 1 (by decide))
  rw [hM] at h
  exact mul_comm_of_single_mul_scalar h

/-- **tex l.190–196**: if `g ∈ EL₃(R_X)` commutes with `e_{ij}(e_V)` for every clopen `V` and
`i ≠ j`, then `g = 1`.  Taking `V = X` makes `g = cI₃`; commuting with the indicators kills the
nonzero Laurent coefficients of `c`, so `c ∈ LC(X, F₂)`; the same holds for `g⁻¹ = dI₃`, and the
constant coefficient of `cd = 1` gives `c = 1`. -/
theorem eq_one_of_commute_elGen_charFn [TotallySeparatedSpace X]
    (hfree : ∀ j : ℤ, j ≠ 0 → ∀ x : X, (T ^ j) x ≠ x)
    {g : elementaryGroup (Fin 3) (ClopenCrossedProduct T (ZMod 2))}
    (hg : ∀ (V : Set X) (hV : IsClopen V) (i j : Fin 3) (hij : i ≠ j),
      Commute g (elGen i j hij (coeff T (ZMod 2) (LocallyConstant.charFn (ZMod 2) hV)))) :
    g = 1 := by
  have hone : coeff T (ZMod 2) (LocallyConstant.charFn (ZMod 2) isClopen_univ) = 1 := by
    rw [← map_one (coeff T (ZMod 2))]
    congr 1
    ext x
    simp [LocallyConstant.coe_charFn]
  -- `g` and `g⁻¹` are scalar
  have hginv : ∀ (V : Set X) (hV : IsClopen V) (i j : Fin 3) (hij : i ≠ j),
      Commute g⁻¹ (elGen i j hij (coeff T (ZMod 2) (LocallyConstant.charFn (ZMod 2) hV))) :=
    fun V hV i j hij => (hg V hV i j hij).inv_left
  obtain ⟨c, hc⟩ := scalar_of_commute_elGen_one fun i j hij => by
    simpa [hone] using hg Set.univ isClopen_univ i j hij
  obtain ⟨d, hd⟩ := scalar_of_commute_elGen_one fun i j hij => by
    simpa [hone] using hginv Set.univ isClopen_univ i j hij
  -- `c, d ∈ LC(X, F₂)`
  have hc0 := eq_coeff_of_coeff_eq_zero T (ZMod 2) fun g' hg' =>
    coeff_eq_zero_of_commute_charFn T (ZMod 2) hfree (commute_charFn_of_scalar T hc hg) hg'
  have hd0 := eq_coeff_of_coeff_eq_zero T (ZMod 2) fun g' hg' =>
    coeff_eq_zero_of_commute_charFn T (ZMod 2) hfree (commute_charFn_of_scalar T hd hginv) hg'
  set f := (ClopenCoeff.of T (ZMod 2)).symm (SkewMonoidAlgebra.coeff c 1) with hf
  set f' := (ClopenCoeff.of T (ZMod 2)).symm (SkewMonoidAlgebra.coeff d 1) with hf'
  -- `cd = 1`
  have hcd : c * d = 1 := by
    have h1 : ((g : (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2)))ˣ) :
          Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2))) *
        (((g⁻¹ : elementaryGroup (Fin 3) (ClopenCrossedProduct T (ZMod 2))) :
          (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2)))ˣ) :
          Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2))) = 1 := by
      rw [Subgroup.coe_inv, Units.mul_inv]
    rw [hc, hd] at h1
    simpa [Matrix.scalar_apply] using congrFun (congrFun h1 0) 0
  -- `f f' = 1` in `LC(X, F₂)`, so `f = 1`
  have hff : f * f' = 1 := by
    have h := hcd
    rw [hc0, hd0, ← map_mul] at h
    exact coeff_injective T (ZMod 2) (h.trans (map_one _).symm)
  have hf1 : f = 1 := by
    ext x
    have hx := LocallyConstant.congr_fun hff x
    rw [LocallyConstant.mul_apply, LocallyConstant.one_apply] at hx
    rw [LocallyConstant.one_apply]
    have hvals : ∀ a b : ZMod 2, a * b = 1 → a = 1 := by decide
    exact hvals _ _ hx
  have hc1 : c = 1 := by
    rw [hc0, ← hf, hf1, map_one]
  apply Subtype.ext
  apply Units.ext
  rw [hc, hc1]
  simp

end Group
