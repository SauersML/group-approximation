import Mathlib.Topology.Sets.Closeds

/-!
# The coefficients `e_C, e_C u, e_C u⁻¹` generate the ring

`simple_kazhdan_sofic_group.tex` at bf961c128, tex l.178–187:

> Choose a finite clopen partition $\mathcal P$ refining the letter partition, with
> $C\cap T^jC=\varnothing$ for $C\in\mathcal P$ and $0<|j|\le2w+3$. Some matrix $h=e_{ij}(s)$,
> $s\in\{e_C,e_Cu,e_Cu^{-1}:C\in\mathcal P\}$, does not commute with $g$. Otherwise, by
> \eqref{eq:elementary}, the coefficients $s$ for which $g$ commutes with every $e_{ij}(s)$ form a
> subring containing all ring generators, so $g\in Z(G)=\{1\}$.

"Containing all ring generators": the generators are $u$, $u^{-1}$ and the letter indicators
(tex l.84–86).  Over a finite clopen partition refining the letters,
$1=\sum_C e_C$, $u=\sum_C e_Cu$, $u^{-1}=\sum_C e_Cu^{-1}$, and $e_a=\sum_{C\subseteq a}e_C$.
So the non-unital ring generated by the finite coefficient set contains $1$ and every generator,
and is everything once the generators generate the ring.

* `charFn_eq_sum_of_refines`: $e_a=\sum_{C\in\mathcal P,\,C\subseteq a}e_C$;
* `partitionRootCoefficients T k 𝒫`: the set $\{e_C,e_Cu,e_Cu^{-1}:C\in\mathcal P\}$;
* `nonUnitalSubring_closure_partitionRootCoefficients_eq_top`: it generates $R$ as a non-unital
  ring when $u$, $u^{-1}$ and the letters generate $R$.
-/

namespace GroupApproximation
namespace Manuscript
namespace SimpleKazhdanSofic

open GroupApproximation.ClopenCrossedProduct TopologicalSpace

variable {X : Type*} [TopologicalSpace X]

open Classical in
/-- **`e_a = ∑_{C ∈ 𝒫, C ⊆ a} e_C`** for a finite clopen partition `𝒫` refining the clopen set
`a`. -/
theorem charFn_eq_sum_of_refines (k : Type*) [Ring k] (𝒫 : Finset (Clopens X))
    (hpart : ∀ x : X, ∃! C, C ∈ 𝒫 ∧ x ∈ (C : Set X)) {a : Set X} (ha : IsClopen a)
    (hrefine : ∀ C ∈ 𝒫, (C : Set X) ⊆ a ∨ Disjoint (C : Set X) a) :
    LocallyConstant.charFn k ha =
      ∑ C ∈ 𝒫.filter (fun C => (C : Set X) ⊆ a), LocallyConstant.charFn k C.isClopen := by
  ext x
  have hsum : ∀ (s : Finset (Clopens X)) (f : Clopens X → LocallyConstant X k),
      (∑ C ∈ s, f C) x = ∑ C ∈ s, f C x := fun s f => by
    have h := congrFun (map_sum (LocallyConstant.coeFnAddMonoidHom (X := X) (Y := k)) f s) x
    rw [Finset.sum_apply] at h
    exact h
  simp only [hsum, LocallyConstant.coe_charFn]
  obtain ⟨C₀, ⟨hC₀, hxC₀⟩, huniq⟩ := hpart x
  by_cases hxa : x ∈ a
  · have hC₀a : (C₀ : Set X) ⊆ a :=
      (hrefine C₀ hC₀).resolve_right fun h => Set.disjoint_left.mp h hxC₀ hxa
    rw [Set.indicator_of_mem hxa,
      Finset.sum_eq_single_of_mem C₀ (Finset.mem_filter.mpr ⟨hC₀, hC₀a⟩)
        fun C hC hne => Set.indicator_of_notMem
          (fun hxC => hne (huniq C ⟨(Finset.mem_filter.mp hC).1, hxC⟩)) _,
      Set.indicator_of_mem hxC₀]
  · rw [Set.indicator_of_notMem hxa,
      Finset.sum_eq_zero fun C hC => Set.indicator_of_notMem
        (fun hxC => hxa ((Finset.mem_filter.mp hC).2 hxC)) _]

/-- **The coefficient set of tex l.183**: `{e_C, e_C u, e_C u⁻¹ : C ∈ 𝒫}`. -/
def partitionRootCoefficients (T : X ≃ₜ X) (k : Type*) [Ring k] (𝒫 : Finset (Clopens X)) :
    Set (ClopenCrossedProduct T k) :=
  {s | ∃ C ∈ 𝒫, s = coeff T k (LocallyConstant.charFn k C.isClopen) ∨
      s = coeff T k (LocallyConstant.charFn k C.isClopen) * (unit T k : ClopenCrossedProduct T k) ∨
      s = coeff T k (LocallyConstant.charFn k C.isClopen) * ↑(unit T k)⁻¹}

/-- **"A subring containing all ring generators"** (tex l.185–187): if `u`, `u⁻¹` and the letter
indicators generate `R = LC(X, k) ⋊_T ℤ` (tex l.84–86) and `𝒫` is a finite clopen partition
refining the letters, then `{e_C, e_C u, e_C u⁻¹ : C ∈ 𝒫}` generates `R` as a non-unital ring. -/
theorem nonUnitalSubring_closure_partitionRootCoefficients_eq_top (T : X ≃ₜ X) (k : Type*)
    [Ring k] (𝒫 : Finset (Clopens X)) (hpart : ∀ x : X, ∃! C, C ∈ 𝒫 ∧ x ∈ (C : Set X))
    (L : Set (Set X)) (hL : ∀ a ∈ L, IsClopen a)
    (hrefine : ∀ a ∈ L, ∀ C ∈ 𝒫, (C : Set X) ⊆ a ∨ Disjoint (C : Set X) a)
    (hgen : Subring.closure ({(unit T k : ClopenCrossedProduct T k), ↑(unit T k)⁻¹} ∪
        {r | ∃ (a : Set X) (ha : a ∈ L), r = coeff T k (LocallyConstant.charFn k (hL a ha))}) = ⊤) :
    NonUnitalSubring.closure (partitionRootCoefficients T k 𝒫) = ⊤ := by
  classical
  set S := NonUnitalSubring.closure (partitionRootCoefficients T k 𝒫) with hS
  have hmem : ∀ C ∈ 𝒫, coeff T k (LocallyConstant.charFn k C.isClopen) ∈ S ∧
      coeff T k (LocallyConstant.charFn k C.isClopen) * (unit T k : ClopenCrossedProduct T k) ∈ S ∧
      coeff T k (LocallyConstant.charFn k C.isClopen) * ↑(unit T k)⁻¹ ∈ S := fun C hC =>
    ⟨NonUnitalSubring.subset_closure ⟨C, hC, Or.inl rfl⟩,
      NonUnitalSubring.subset_closure ⟨C, hC, Or.inr (Or.inl rfl)⟩,
      NonUnitalSubring.subset_closure ⟨C, hC, Or.inr (Or.inr rfl)⟩⟩
  have hone : (1 : LocallyConstant X k) = ∑ C ∈ 𝒫, LocallyConstant.charFn k C.isClopen := by
    have h := charFn_eq_sum_of_refines k 𝒫 hpart isClopen_univ
      (fun C _ => Or.inl (Set.subset_univ _))
    rw [Finset.filter_true_of_mem (fun C _ => Set.subset_univ _)] at h
    rw [← h]
    ext x
    simp [LocallyConstant.coe_charFn]
  have hone' : (1 : ClopenCrossedProduct T k) =
      ∑ C ∈ 𝒫, coeff T k (LocallyConstant.charFn k C.isClopen) := by
    rw [← map_sum, ← hone, map_one]
  have h1 : (1 : ClopenCrossedProduct T k) ∈ S := by
    rw [hone']
    exact sum_mem fun C hC => (hmem C hC).1
  have hu : (unit T k : ClopenCrossedProduct T k) ∈ S := by
    have h : (unit T k : ClopenCrossedProduct T k) =
        ∑ C ∈ 𝒫, coeff T k (LocallyConstant.charFn k C.isClopen) *
          (unit T k : ClopenCrossedProduct T k) := by
      rw [← Finset.sum_mul, ← hone', one_mul]
    rw [h]
    exact sum_mem fun C hC => (hmem C hC).2.1
  have hui : (↑(unit T k)⁻¹ : ClopenCrossedProduct T k) ∈ S := by
    have h : (↑(unit T k)⁻¹ : ClopenCrossedProduct T k) =
        ∑ C ∈ 𝒫, coeff T k (LocallyConstant.charFn k C.isClopen) * ↑(unit T k)⁻¹ := by
      rw [← Finset.sum_mul, ← hone', one_mul]
    rw [h]
    exact sum_mem fun C hC => (hmem C hC).2.2
  have hletter : ∀ (a : Set X) (ha : a ∈ L),
      coeff T k (LocallyConstant.charFn k (hL a ha)) ∈ S := fun a ha => by
    rw [charFn_eq_sum_of_refines k 𝒫 hpart (hL a ha) (hrefine a ha), map_sum]
    exact sum_mem fun C hC => (hmem C (Finset.mem_filter.mp hC).1).1
  let S' : Subring (ClopenCrossedProduct T k) :=
    { carrier := S
      mul_mem' := S.mul_mem
      one_mem' := h1
      add_mem' := S.add_mem
      zero_mem' := S.zero_mem
      neg_mem' := S.neg_mem }
  have hle : Subring.closure ({(unit T k : ClopenCrossedProduct T k), ↑(unit T k)⁻¹} ∪
      {r | ∃ (a : Set X) (ha : a ∈ L), r = coeff T k (LocallyConstant.charFn k (hL a ha))}) ≤ S' := by
    rw [Subring.closure_le]
    intro r hr
    simp only [Set.mem_union, Set.mem_insert_iff, Set.mem_singleton_iff, Set.mem_setOf_eq] at hr
    show r ∈ S
    rcases hr with (rfl | rfl) | ⟨a, ha, rfl⟩
    · exact hu
    · exact hui
    · exact hletter a ha
  rw [hgen] at hle
  exact eq_top_iff.mpr fun r _ => hle (Subring.mem_top r)
