/-!
# `prop:bilateral-three` over `R_Y` LEF

`non_mf_groups_exist.tex`, Lemma `lem:transient-matrices` (tex 1476–1478) and Theorem
`thm:core-ring-reflection` (tex 1516–1535):

> The ideal $I$ is locally matricial and is generated by the indicators $1_{P\setminus T(P)}$ of the
> defects in Lemma~\ref{lem:chain-core-models}.
>
> A unital homomorphism to a directly finite ring kills every $d$ in
> \eqref{eq:clopen-defect-pair}, hence all of $I$ by Lemma~\ref{lem:transient-matrices}.  Since $R_Y$
> itself is LEF and therefore stably finite, all three universal properties follow.

With the generation clause on main (`transientIdeal_chainRecurrentSet_eq_span`,
`chainRecurrentSet_eq_univ_of_forall_defectD_eq_zero`, `chainRecurrentSet_eq_univ_of_isDedekindFiniteMonoid`,
lane ms-core-4), the two remaining pieces of `prop:bilateral-three` follow from one input,
"`R_Y` is LEF" (Lemma `lem:chain-core-models`, `DynamicRankBudget.CoreModelsLEFStatement`):

* if every point is chain recurrent, restriction `R_X → R_Y` is injective, so `R_X` is LEF
  (`isLEFRing_of_chainRecurrentSet_eq_univ`);
* directly finite ⇒ every point chain recurrent ⇒ LEF (`clopenCrossedProductDirectlyFiniteLEFStatement_of_coreModels`);
* all defects zero ⇒ every point chain recurrent ⇒ LEF ⇒ directly finite, the contrapositive of
  "some defect is nonzero" (`clopenCrossedProductNonzeroDefectStatement_of_coreModels`).

The printed setting is a compact metrizable space; `CoreModelsLEFStatement` fixes a compatible metric,
so the metrizable wrappers choose one (`TopologicalSpace.metrizableSpaceMetric`, whose topology is the
given one by definition) and substitute it for the topology.

Also closed here: `CoreTransientIdealLeDefectIdealStatement` (lane ms-core-5), `I ≤ ⟨defects⟩` over
`F₂`, from `defectIdeal_eq_transientIdeal_chainRecurrentSet`.
-/

/-- **`I ≤ ⟨defects⟩` over `F₂`** (tex 1476–1478), closed: the transient ideal of the chain-recurrent
set is the ideal generated by the compressed-clopen defects. -/
theorem coreTransientIdealLeDefectIdealStatement_holds :
    GroupApproximation.ChainCore.CoreMFRadicalGLKill.CoreTransientIdealLeDefectIdealStatement := by
  intro X _ _ _ _ T
  exact (GroupApproximation.defectIdeal_eq_transientIdeal_chainRecurrentSet T (ZMod 2)).symm.le

/-! ## Over a compatible metric -/

/-- **If every point is chain recurrent, `R_X` is LEF**: restriction to `Y = CR(T) = X` is
injective and `R_Y` is LEF. -/
theorem isLEFRing_of_chainRecurrentSet_eq_univ (h : CoreModelsLEFStatement) {X : Type}
    [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X] (T : X ≃ₜ X)
    (k : Type) [Field k] [Finite k] (hY : Dynamics.chainRecurrentSet T = Set.univ) :
    Pestov91.IsLEFRing (ClopenCrossedProduct T k) :=
  Pestov91.IsLEFRing.of_injective (h X T k) (ClopenCrossedProduct.restrict T k (image_core T))
    (ClopenCrossedProduct.restrict_injective_of_eq_univ T k (image_core T) hY)

/-- **Directly finite ⇒ LEF** (tex 1522–1523), over `R_Y` LEF. -/
theorem isLEFRing_of_isDedekindFiniteMonoid_of_coreModels (h : CoreModelsLEFStatement)
    {X : Type} [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X) (k : Type) [Field k] [Finite k]
    (hdf : IsDedekindFiniteMonoid (ClopenCrossedProduct T k)) :
    Pestov91.IsLEFRing (ClopenCrossedProduct T k) := by
  haveI := hdf
  exact isLEFRing_of_chainRecurrentSet_eq_univ h T k
    (GroupApproximation.chainRecurrentSet_eq_univ_of_isDedekindFiniteMonoid T k)

/-- **Not directly finite ⇒ some defect is nonzero** (tex 1599–1600), over `R_Y` LEF. -/
theorem exists_defect_ne_zero_of_coreModels (h : CoreModelsLEFStatement) {X : Type}
    [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X] (T : X ≃ₜ X)
    (k : Type) [Field k] [Finite k]
    (hdf : ¬ IsDedekindFiniteMonoid (ClopenCrossedProduct T k)) :
    ∃ d ∈ clopenDefectSet T k, d ≠ 0 := by
  by_contra hne
  refine hdf (isLEFRing_of_chainRecurrentSet_eq_univ h T k ?_).isDedekindFiniteMonoid
  refine ClopenCrossedProduct.chainRecurrentSet_eq_univ_of_forall_defectD_eq_zero T k
    fun P hP hTP => ?_
  by_contra hd
  exact hne ⟨_, ⟨P, hP, hTP, rfl⟩, hd⟩

/-! ## The printed setting: a compact metrizable space -/

/-- `ClopenCrossedProductDirectlyFiniteLEFStatement` over `R_Y` LEF. -/
theorem clopenCrossedProductDirectlyFiniteLEFStatement_of_coreModels
    (h : CoreModelsLEFStatement) : ClopenCrossedProductDirectlyFiniteLEFStatement := by
  intro X t _ _ _ _ T k _ _ hdf
  obtain ⟨m, hm⟩ : ∃ m : MetricSpace X, m.toUniformSpace.toTopologicalSpace = t :=
    ⟨TopologicalSpace.metrizableSpaceMetric X, rfl⟩
  subst hm
  exact isLEFRing_of_isDedekindFiniteMonoid_of_coreModels h T k hdf

/-- `ClopenCrossedProductNonzeroDefectStatement` over `R_Y` LEF. -/
theorem clopenCrossedProductNonzeroDefectStatement_of_coreModels
    (h : CoreModelsLEFStatement) : ClopenCrossedProductNonzeroDefectStatement := by
  intro X t _ _ _ _ T k _ _ hdf
  obtain ⟨m, hm⟩ : ∃ m : MetricSpace X, m.toUniformSpace.toTopologicalSpace = t :=
    ⟨TopologicalSpace.metrizableSpaceMetric X, rfl⟩
  subst hm
  exact exists_defect_ne_zero_of_coreModels h T k hdf

/-- **`prop:bilateral-three` over "`R_Y` is LEF"** (Lemma `lem:chain-core-models`). -/
theorem printedBilateralThree_of_coreModels (h : CoreModelsLEFStatement) : PrintedBilateralThree :=
  printedBilateralThree_of_reflection (clopenCrossedProductDirectlyFiniteLEFStatement_of_coreModels h)
    (clopenCrossedProductNonzeroDefectStatement_of_coreModels h)

/-- The Introduction clause "`R_X` is directly finite exactly when `GL₃(R_X)` is MF" (tex 253–255)
over "`R_Y` is LEF". -/
theorem printedIntroDirectlyFiniteIffGLThreeMF_of_coreModels (h : CoreModelsLEFStatement) :
    PrintedIntroDirectlyFiniteIffGLThreeMF :=
  printedIntroDirectlyFiniteIffGLThreeMF_of_printedBilateralThree (printedBilateralThree_of_coreModels h)

/-- The transfer piece of `cor:dynamic-rank-budget` over "`R_Y` is LEF". -/
theorem bilateralThreeTransferStatement_of_coreModels (h : CoreModelsLEFStatement) :
    BilateralThreeTransferStatement :=
  bilateralThreeTransferStatement_of_nonzeroDefect
    (clopenCrossedProductNonzeroDefectStatement_of_coreModels h)
