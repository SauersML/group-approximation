/-!
# `prop:bilateral-three`: the printed statement and its assembly

`non_mf_groups_exist.tex`, Proposition `prop:bilateral-three` (tex 1541–1547):

> For every finite field `k` and `n ≥ 3`, the group `GL_n(R_X)` is MF if and only if it is
> LEF, if and only if `R_X` is directly finite.  Every homomorphism from `GL_n(R_X)` to an MF
> group kills `EL_n(R_X, I)`.

The printed proof (tex 1549–1603) has four inputs besides the group-level kill of
`BilateralThreeGroupKill`:

* the three-coordinate cell: `S`, `L = EL₃(S)` with property `(T)` (Ershov--Jaikin-Zapirain),
  `W`, `c`, `ℓ`, `y`, `z` and the displayed identities (`BilateralThreeCellStatement`);
* `R_X` is countable, so the countable-LEF-is-MF step applies
  (`ClopenCrossedProductCountableStatement`);
* Theorem `thm:core-ring-reflection`: a directly finite `R_X` is LEF
  (`ClopenCrossedProductDirectlyFiniteLEFStatement`);
* "If `R_X` is not directly finite, some defect is nonzero", from Lemma
  `lem:transient-matrices` and Theorem `thm:core-ring-reflection`
  (`ClopenCrossedProductNonzeroDefectStatement`).

This module states them as named Props, proves the rest of the printed route over them
("Normal root commutators and root addition kill all roots in the ideal generated by `d`",
"Extending by identity gives the same conclusion in rank `n ≥ 3`", "If `R_X` is not directly
finite, some defect is nonzero, so `GL_n(R_X)` is not MF", "Otherwise
Theorem~`thm:core-ring-reflection` makes `R_X`, and hence its general linear groups, LEF"),
and assembles `PrintedBilateralThree` in `printedBilateralThree_of_pieces`.

## The ideal `I`

The printed proof uses `I` only through "The defects generate `I` by
Lemma~`lem:transient-matrices`".  Here `I` is spelled as that generated ideal,
`defectIdeal T k`; its identification with `LC_c(X∖Y,k) ⋊_T ℤ` is Lemma
`lem:transient-matrices`, proved by its own lane.
-/

/-- The printed index pair `(2,1)` of `e₂₁(d)`, inside `Fin 3`. -/
theorem idxOneNeZero : (1 : Fin 3) ≠ 0 := by decide

/-! ## The printed objects -/

section Objects

variable {X : Type} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type) [Field k]

/-- The compressed-clopen defects `d = 1_{P∖T(P)}` of `eq:clopen-defect-pair`, one for every
clopen `P` with `T(P) ⊆ P`. -/
def clopenDefectSet : Set (ClopenCrossedProduct T k) :=
  {d | ∃ (P : Set X) (hP : IsClopen P), T '' P ⊆ P ∧ ClopenCrossedProduct.defectD T k hP = d}

/-- The two-sided ideal generated by the compressed-clopen defects.  By
Lemma~`lem:transient-matrices` it is the transient ideal `I`. -/
def defectIdeal : TwoSidedIdeal (ClopenCrossedProduct T k) :=
  TwoSidedIdeal.span (clopenDefectSet T k)

/-- "Every homomorphism from `GL_n(R_X)` to an MF group kills `EL_n(R_X, I)`." -/
def MFTargetsKillRelativeElementary (n : ℕ) (I : TwoSidedIdeal (ClopenCrossedProduct T k)) :
    Prop :=
  ∀ (M : Type) [Group M], IsOperatorMF M →
    ∀ f : (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ →* M,
      ∀ g ∈ relativeElementary (Fin n) I,
        f (g : (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ) = 1

end Objects

/-- **`prop:bilateral-three`, as printed** (tex 1541–1547).

Standing hypotheses of `sec:chain-core` (tex 1345–1350): `X` is a nonempty compact metrizable
zero-dimensional space (for compact Hausdorff spaces, zero-dimensional is totally
disconnected), `T` a homeomorphism, `k` a finite field, `R_X = LC(X,k) ⋊_T ℤ`
(`ClopenCrossedProduct T k`).  `GL_n(R_X)` is `(Matrix (Fin n) (Fin n) R_X)ˣ`; "MF" is
`IsOperatorMF`, "LEF" is `IsLEF`, "directly finite" is `IsDedekindFiniteMonoid`, and
`EL_n(R_X, I)` is `relativeElementary (Fin n) I` with `I = defectIdeal T k` (see the module
docstring). -/
def PrintedBilateralThree : Prop :=
  ∀ (X : Type) [TopologicalSpace X] [CompactSpace X] [TopologicalSpace.MetrizableSpace X]
    [TotallyDisconnectedSpace X] [Nonempty X] (T : X ≃ₜ X) (k : Type) [Field k] [Finite k]
    (n : ℕ), 3 ≤ n →
      (IsOperatorMF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ ↔
          IsLEF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ) ∧
        (IsLEF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ ↔
          IsDedekindFiniteMonoid (ClopenCrossedProduct T k)) ∧
        MFTargetsKillRelativeElementary T k n (defectIdeal T k)

/-! ## The pieces -/

/-- **Piece: the three-coordinate cell** (tex 1550–1593; lane ct-bilateral-cell for the
identities, with property `(T)` of `L = EL₃(S)` from Ershov--Jaikin-Zapirain).

For clopen `P` with `T(P) ⊆ P` there are, inside `GL₃(R_X)`, a property-`(T)` subgroup `L`
(the corner copy of `EL₃(S)`), the compressor `W`, the centralizing element `c`, `ℓ ∈ L`,
`y`, the finite-order element `z` with `[y, [WcW⁻¹, ℓ]] = z⁻¹` commuting with `L`, `W`, `c`,
`y`, and `g₁ = e₃₁(f_{-2,0})`, `g₂ = e₂₃(f_{0,-1})` with `[g₂, [z, g₁]] = e₂₁(d)`. -/
def BilateralThreeCellStatement : Prop :=
  ∀ (X : Type) [TopologicalSpace X] (T : X ≃ₜ X) (k : Type) [Field k] [Finite k]
    (P : Set X) (hP : IsClopen P), T '' P ⊆ P →
      ∃ (L : Subgroup (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T k))ˣ)
        (W c y ℓ z g₁ g₂ : (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T k))ˣ),
        HasKazhdanPropertyT.{0, 0} ↥L ∧
          (∀ δ ∈ L, W * δ * W⁻¹ ∈ L) ∧ (∀ δ ∈ L, Commute c δ) ∧ ℓ ∈ L ∧
          ⁅y, ⁅W * c * W⁻¹, ℓ⁆⁆ = z⁻¹ ∧
          (∀ δ ∈ L, Commute z δ) ∧ Commute z W ∧ Commute z c ∧ Commute z y ∧
          IsOfFinOrder z ∧
          ⁅g₂, ⁅z, g₁⁆⁆ = elementaryUnit (1 : Fin 3) 0 idxOneNeZero
            (ClopenCrossedProduct.defectD T k hP)

/-- **Piece: `R_X` is countable** ("Its countability hypothesis holds because `X` has a
countable clopen basis", tex 1647–1649; chain-core S1). -/
def ClopenCrossedProductCountableStatement : Prop :=
  ∀ (X : Type) [TopologicalSpace X] [CompactSpace X] [TopologicalSpace.MetrizableSpace X]
    [TotallyDisconnectedSpace X] (T : X ≃ₜ X) (k : Type) [Field k] [Finite k],
      Countable (ClopenCrossedProduct T k)

/-- **Piece: Theorem `thm:core-ring-reflection`, "`R_X` is directly finite if and only if it
is LEF"**, the direction the printed proof uses (chain-reflection). -/
def ClopenCrossedProductDirectlyFiniteLEFStatement : Prop :=
  ∀ (X : Type) [TopologicalSpace X] [CompactSpace X] [TopologicalSpace.MetrizableSpace X]
    [TotallyDisconnectedSpace X] [Nonempty X] (T : X ≃ₜ X) (k : Type) [Field k] [Finite k],
      IsDedekindFiniteMonoid (ClopenCrossedProduct T k) →
        Pestov91.IsLEFRing (ClopenCrossedProduct T k)

/-- **Piece: "If `R_X` is not directly finite, some defect is nonzero"** (tex 1599–1600), from
Lemma `lem:transient-matrices` and Theorem `thm:core-ring-reflection` (chain-matricial,
chain-reflection). -/
def ClopenCrossedProductNonzeroDefectStatement : Prop :=
  ∀ (X : Type) [TopologicalSpace X] [CompactSpace X] [TopologicalSpace.MetrizableSpace X]
    [TotallyDisconnectedSpace X] [Nonempty X] (T : X ≃ₜ X) (k : Type) [Field k] [Finite k],
      ¬ IsDedekindFiniteMonoid (ClopenCrossedProduct T k) →
        ∃ d ∈ clopenDefectSet T k, d ≠ 0

/-! ## Extension by identity -/

section Rank

variable {R : Type*} [Ring R]

/-- The printed "Extending by identity" reindexing `Fin 3 ⊕ Fin (n-3) ≃ Fin n`. -/
def threeBlockIndexEquiv {n : ℕ} (hn : 3 ≤ n) : Fin 3 ⊕ Fin (n - 3) ≃ Fin n :=
  finSumFinEquiv.trans (finCongr (by omega))

/-- The three printed coordinates inside `Fin n`. -/
def threeIdx {n : ℕ} (hn : 3 ≤ n) (i : Fin 3) : Fin n :=
  threeBlockIndexEquiv hn (Sum.inl i)

theorem threeIdx_injective {n : ℕ} (hn : 3 ≤ n) : Function.Injective (threeIdx hn) :=
  fun _ _ h ↦ Sum.inl_injective ((threeBlockIndexEquiv hn).injective h)

/-- Extension by identity, `GL₃(R) → GL_n(R)`. -/
noncomputable def glThreeBlockEmbedding {n : ℕ} (hn : 3 ≤ n) :
    (Matrix (Fin 3) (Fin 3) R)ˣ →* (Matrix (Fin n) (Fin n) R)ˣ :=
  (elementaryReindexUnitEquiv (R := R) (threeBlockIndexEquiv hn)).toMonoidHom.comp
    (stabilizeUnit (ι := Fin 3) (κ := Fin (n - 3)) (R := R))

theorem glThreeBlockEmbedding_elementaryUnit {n : ℕ} (hn : 3 ≤ n) (i j : Fin 3)
    (hij : i ≠ j) (a : R) :
    glThreeBlockEmbedding (R := R) hn (elementaryUnit i j hij a) =
      elementaryUnit (threeIdx hn i) (threeIdx hn j) ((threeIdx_injective hn).ne hij) a := by
  change elementaryReindexUnitEquiv (R := R) (threeBlockIndexEquiv hn)
      (stabilizeUnit (κ := Fin (n - 3)) (elementaryUnit i j hij a)) =
    elementaryUnit (threeIdx hn i) (threeIdx hn j) ((threeIdx_injective hn).ne hij) a
  rw [stabilizeUnit_elementaryUnit, elementaryReindexUnitEquiv_elementaryUnit]
  rfl

/-- **Printed (tex 1594–1595):** "Normal root commutators and root addition kill all roots in
the ideal generated by `d`", for a set of generators: a normal subgroup of `EL_ι(R)`
containing one elementary matrix with coefficient `d`, for every `d ∈ D`, contains
`EL_ι(R, ⟨D⟩)`.  Three coordinates suffice. -/
theorem relativeElementary_span_le_of_forall {ι : Type*} [Fintype ι] [DecidableEq ι]
    (N : Subgroup (elementaryGroup ι R)) [N.Normal]
    (hthree : ∀ i j : ι, ∃ k : ι, k ≠ i ∧ k ≠ j) {D : Set R}
    (hD : ∀ d ∈ D, ∃ (p q : ι) (hpq : p ≠ q), elGen p q hpq d ∈ N) :
    relativeElementary ι (TwoSidedIdeal.span D) ≤ N := by
  have hsub : D ⊆ (TorsionComplementaryIdempotents.normalCoefficientIdeal N hthree : Set R) := by
    intro d hd
    obtain ⟨p, q, hpq, h⟩ := hD d hd
    show d ∈ TorsionComplementaryIdempotents.normalCoefficientIdeal N hthree
    rw [TorsionComplementaryIdempotents.mem_normalCoefficientIdeal]
    exact TorsionComplementaryIdempotents.elGen_mem_of_one_position hthree hpq h
  have hspan : TwoSidedIdeal.span D ≤
      TorsionComplementaryIdempotents.normalCoefficientIdeal N hthree :=
    TwoSidedIdeal.span_le.mpr hsub
  refine Subgroup.normalClosure_le_normal ?_
  rintro g ⟨i, j, hij, b, hb, rfl⟩
  have hb' := hspan hb
  rw [TorsionComplementaryIdempotents.mem_normalCoefficientIdeal] at hb'
  exact hb' i j hij

/-- **Printed (tex 1594–1596):** if every MF target from `GL₃(R)` kills `e₂₁(d)` for every
`d ∈ D`, then for every `n ≥ 3` every MF target from `GL_n(R)` kills `EL_n(R, ⟨D⟩)`. -/
theorem relativeElementary_killed_of_rootThree_killed {D : Set R}
    (hroot : ∀ d ∈ D, ∀ (M : Type) [Group M], IsOperatorMF M →
      ∀ f : (Matrix (Fin 3) (Fin 3) R)ˣ →* M, f (elementaryUnit 1 0 idxOneNeZero d) = 1)
    {n : ℕ} (hn : 3 ≤ n) {M : Type} [Group M] (hM : IsOperatorMF M)
    (f : (Matrix (Fin n) (Fin n) R)ˣ →* M) :
    ∀ g ∈ relativeElementary (Fin n) (TwoSidedIdeal.span D),
      f (g : (Matrix (Fin n) (Fin n) R)ˣ) = 1 := by
  have hle : relativeElementary (Fin n) (TwoSidedIdeal.span D) ≤
      (f.comp (elementaryGroup (Fin n) R).subtype).ker := by
    refine relativeElementary_span_le_of_forall _
      (TorsionComplementaryIdempotents.exists_third_index hn) ?_
    intro d hd
    refine ⟨threeIdx hn 1, threeIdx hn 0, (threeIdx_injective hn).ne idxOneNeZero, ?_⟩
    rw [MonoidHom.mem_ker]
    have h := hroot d hd M hM (f.comp (glThreeBlockEmbedding (R := R) hn))
    rwa [MonoidHom.comp_apply, glThreeBlockEmbedding_elementaryUnit] at h
  intro g hg
  exact MonoidHom.mem_ker.mp (hle hg)

end Rank

/-! ## The two directions -/

section Directions

variable {R : Type} [Ring R]

theorem countable_units_matrix [Countable R] (n : ℕ) :
    Countable (Matrix (Fin n) (Fin n) R)ˣ := by
  haveI : Countable (Matrix (Fin n) (Fin n) R) :=
    inferInstanceAs (Countable (Fin n → Fin n → R))
  exact Units.val_injective.countable

/-- "`R_X`, and hence its general linear groups, LEF": units of matrices over an LEF ring are
LEF ("Entrywise finite ring models, including the entries of inverse matrices", tex 1359). -/
theorem isLEF_units_matrix_of_isLEFRing (hR : Pestov91.IsLEFRing R) (n : ℕ) :
    IsLEF (Matrix (Fin n) (Fin n) R)ˣ :=
  (hR.matrix (Fin n)).isLEF_units

/-- "Some defect is nonzero, so `GL_n(R_X)` is not MF": a nonzero elementary matrix killed by
every MF target, the identity included, rules out MF. -/
theorem not_isOperatorMF_of_root_killed {n : ℕ} {i j : Fin n} (hij : i ≠ j) {d : R}
    (hd : d ≠ 0)
    (hkill : ∀ (M : Type) [Group M], IsOperatorMF M →
      ∀ f : (Matrix (Fin n) (Fin n) R)ˣ →* M, f (elementaryUnit i j hij d) = 1) :
    ¬ IsOperatorMF (Matrix (Fin n) (Fin n) R)ˣ := by
  intro hMF
  have h := hkill _ hMF (MonoidHom.id _)
  rw [MonoidHom.id_apply] at h
  exact hd (elementaryUnit_injective i j hij (h.trans (elementaryUnit_zero (R := R) i j hij).symm))

end Directions

/-! ## The assembly -/

/-- The cell and the group-level kill give the kill of every defect root in `GL₃(R_X)`. -/
theorem rootThree_killed_of_cell (hcell : BilateralThreeCellStatement)
    {X : Type} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type) [Field k] [Finite k]
    [Countable (ClopenCrossedProduct T k)] :
    ∀ d ∈ clopenDefectSet T k, ∀ (M : Type) [Group M], IsOperatorMF M →
      ∀ f : (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T k))ˣ →* M,
        f (elementaryUnit 1 0 idxOneNeZero d) = 1 := by
  rintro d ⟨P, hP, hTP, rfl⟩ M _ hM f
  obtain ⟨L, W, c, y, ℓ, z, g₁, g₂, hL, hW, hc, hℓ, hyz, hzL, hzW, hzc, hzy, hfin, hroot⟩ :=
    hcell X T k P hP hTP
  haveI : Countable (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T k))ˣ :=
    countable_units_matrix 3
  exact root_killed_of_finite_central_defect L W c y hL hW hc hℓ hyz hzL hzW hzc hzy hfin
    rfl hroot hM f

/-- **`prop:bilateral-three` over its four pieces**, along the printed route. -/
theorem printedBilateralThree_of_pieces
    (hcell : BilateralThreeCellStatement)
    (hcount : ClopenCrossedProductCountableStatement)
    (hlef : ClopenCrossedProductDirectlyFiniteLEFStatement)
    (hdef : ClopenCrossedProductNonzeroDefectStatement) :
    PrintedBilateralThree := by
  intro X _ _ _ _ _ T k _ _ n hn
  haveI : Countable (ClopenCrossedProduct T k) := hcount X T k
  haveI : Countable (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ :=
    countable_units_matrix n
  have hkill : MFTargetsKillRelativeElementary T k n (defectIdeal T k) := by
    intro M _ hM f
    exact relativeElementary_killed_of_rootThree_killed (rootThree_killed_of_cell hcell T k)
      hn hM f
  have hMFdf : IsOperatorMF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ →
      IsDedekindFiniteMonoid (ClopenCrossedProduct T k) := by
    intro hMF
    by_contra hdf
    obtain ⟨d, hd, hne⟩ := hdef X T k hdf
    refine not_isOperatorMF_of_root_killed ((threeIdx_injective hn).ne idxOneNeZero) hne ?_ hMF
    intro M _ hM f
    have h := rootThree_killed_of_cell hcell T k d hd M hM
      (f.comp (glThreeBlockEmbedding (R := ClopenCrossedProduct T k) hn))
    rwa [MonoidHom.comp_apply, glThreeBlockEmbedding_elementaryUnit] at h
  have hdfLEF : IsDedekindFiniteMonoid (ClopenCrossedProduct T k) →
      IsLEF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ := fun hdf ↦
    isLEF_units_matrix_of_isLEFRing (hlef X T k hdf) n
  have hLEFMF : IsLEF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ →
      IsOperatorMF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ :=
    isOperatorMF_of_isLEF
  exact ⟨⟨fun h ↦ hdfLEF (hMFdf h), hLEFMF⟩, ⟨fun h ↦ hMFdf (hLEFMF h), hdfLEF⟩, hkill⟩

end BilateralThree
