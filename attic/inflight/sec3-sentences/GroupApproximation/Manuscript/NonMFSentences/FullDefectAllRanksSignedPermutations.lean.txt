/-!
# `thm:full-defect-ring`, "All ranks", along the printed signed-permutation route

`non_mf_groups_exist.tex`, the proof of Theorem `thm:full-defect-ring`, tex lines
999-1001 at origin/main `73c867c5b`:

> \emph{All ranks.}  For `n ≥ 3`, signed permutation matrices conjugate `e₁₂(r)`
> to `e_{ij}(±r)` for all `i ≠ j`, so the copy of `EL₂(R)` on coordinates `1,2`
> normally generates `EL_n(R)`.

`FullDefectRankTwo.hom_eq_one_of_rankFour` reaches the theorem's conclusion at
`n ≥ 3` by a different route (a unit-sandwiched root and the Steinberg relations,
`elementaryGroup_normal_eq_top_of_elementaryRoot_mem`), and the census row for this
sentence was graded `partial` for that reason.  This module carries the printed
route itself:

* `IsSignedPermutationMatrix` is the printed notion: a matrix whose `j`-th column
  is `±1` times the `σ(j)`-th standard basis vector for a permutation `σ`;
* each Weyl element `w_{uv} = e_{uv}(1)e_{vu}(-1)e_{uv}(1)` is a signed permutation
  matrix (`weyl_isSignedPermutationMatrix`), and signed permutation matrices are
  closed under products;
* `exists_signedPermutation_conj`: for any two ordered pairs of distinct indices
  `(a,b)` and `(i,j)`, a product of at most two Weyl elements — a signed
  permutation matrix in `EL(R)` — conjugates `e_{ab}(r)` to `e_{ij}(r)` or
  `e_{ij}(-r)`, for every `r`, with one and the same conjugating element;
* `manuscriptSentence_allRanksSignedPermutations` is the printed sentence at every
  `n = 2 + k ≥ 3`, with the copy of `EL₂(R)` on the first two coordinates being
  `FullDefectRankTwo.blockEmbed R k`.
-/

section SignedPermutation

variable {ι S : Type*} [Fintype ι] [DecidableEq ι] [Ring S]

/-- A **signed permutation matrix**: for some permutation `σ` and signs `ε j = ±1`,
the `j`-th column is `ε j` times the standard basis vector at `σ j`. -/
def IsSignedPermutationMatrix (M : Matrix ι ι S) : Prop :=
  ∃ (σ : Equiv.Perm ι) (ε : ι → S), (∀ j, ε j = 1 ∨ ε j = -1) ∧
    ∀ i j, M i j = if σ j = i then ε j else 0

theorem isSignedPermutationMatrix_one :
    IsSignedPermutationMatrix (1 : Matrix ι ι S) := by
  refine ⟨1, fun _ ↦ 1, fun _ ↦ Or.inl rfl, fun i j ↦ ?_⟩
  simp only [Equiv.Perm.coe_one, id_eq, Matrix.one_apply]
  by_cases h : j = i
  · subst h; simp
  · simp [h, Ne.symm h]

theorem isSignedPermutationMatrix_mul {M N : Matrix ι ι S}
    (hM : IsSignedPermutationMatrix M) (hN : IsSignedPermutationMatrix N) :
    IsSignedPermutationMatrix (M * N) := by
  obtain ⟨σ, ε, hε, hMe⟩ := hM
  obtain ⟨τ, δ, hδ, hNe⟩ := hN
  refine ⟨σ * τ, fun j ↦ ε (τ j) * δ j, fun j ↦ ?_, fun i j ↦ ?_⟩
  · rcases hε (τ j) with h | h <;> rcases hδ j with h' | h' <;> simp [h, h']
  · rw [Matrix.mul_apply]
    simp only [hMe, hNe]
    rw [Finset.sum_eq_single (τ j)]
    · simp only [eq_self_iff_true, if_true, Equiv.Perm.mul_apply]
      by_cases h : σ (τ j) = i
      · simp [h]
      · simp [h]
    · intro k _ hk
      have hk' : τ j ≠ k := Ne.symm hk
      simp [hk']
    · intro h
      exact absurd (Finset.mem_univ _) h

/-- **The Weyl element is a signed permutation matrix**: `σ = (u v)`, sign `-1` on
column `u` and `+1` elsewhere. -/
theorem weylUnit_isSignedPermutationMatrix (u v : ι) (huv : u ≠ v) :
    IsSignedPermutationMatrix (weylUnit (S := S) u v huv : Matrix ι ι S) := by
  refine ⟨Equiv.swap u v, fun j ↦ if j = u then -1 else 1, fun j ↦ ?_, fun i j ↦ ?_⟩
  · by_cases h : j = u
    · simp [h]
    · simp [h]
  · rw [weylUnit_val]
    simp only [Matrix.sub_apply, Matrix.add_apply, Matrix.one_apply, Matrix.single_apply]
    by_cases hju : j = u
    · rw [hju, Equiv.swap_apply_left]
      split_ifs <;> simp_all
    · by_cases hjv : j = v
      · rw [hjv, Equiv.swap_apply_right]
        split_ifs <;> simp_all
      · rw [Equiv.swap_apply_of_ne_of_ne hju hjv]
        split_ifs <;> simp_all

theorem weyl_isSignedPermutationMatrix (u v : ι) (huv : u ≠ v) :
    IsSignedPermutationMatrix
      (((weyl (S := S) u v huv : elementaryGroup ι S) : (Matrix ι ι S)ˣ) : Matrix ι ι S) := by
  rw [weyl_val]
  exact weylUnit_isSignedPermutationMatrix u v huv

/-- `g` carries `e_{ab}(r)` to `e_{ij}(±r)` for every `r` (one sign per `r`). -/
def ConjToUpToSign {a b : ι} (hab : a ≠ b) (g : elementaryGroup ι S)
    {i j : ι} (hij : i ≠ j) : Prop :=
  ∀ r : S, g * elGen a b hab r * g⁻¹ = elGen i j hij r ∨
    g * elGen a b hab r * g⁻¹ = elGen i j hij (-r)

theorem conjToUpToSign_one {a b : ι} (hab : a ≠ b) :
    ConjToUpToSign (S := S) hab 1 hab := fun r ↦ Or.inl (by group)

/-- One more Weyl conjugation moves the target pair by the transposition `(u v)`. -/
theorem conjToUpToSign_weyl_mul {a b : ι} (hab : a ≠ b) {g : elementaryGroup ι S}
    {p q : ι} (hpq : p ≠ q) (hg : ConjToUpToSign hab g hpq)
    (u v : ι) (huv : u ≠ v) {p' q' : ι} (hp : Equiv.swap u v p = p')
    (hq : Equiv.swap u v q = q') (hp'q' : p' ≠ q') :
    ConjToUpToSign hab (weyl (S := S) u v huv * g) hp'q' := by
  intro r
  have hassoc : weyl (S := S) u v huv * g * elGen a b hab r * (weyl (S := S) u v huv * g)⁻¹
      = weyl (S := S) u v huv * (g * elGen a b hab r * g⁻¹) * (weyl (S := S) u v huv)⁻¹ := by
    group
  rw [hassoc]
  rcases hg r with h | h
  · rw [h]
    exact weyl_conj_elGen u v huv hpq hp hq hp'q' r
  · rw [h]
    rcases weyl_conj_elGen u v huv hpq hp hq hp'q' (-r) with h' | h'
    · exact Or.inr h'
    · left
      rw [h', neg_neg]

/-- **Signed permutation matrices conjugate `e_{ab}(r)` to `e_{ij}(±r)`**, for any
two ordered pairs of distinct indices, by a product of at most two Weyl elements. -/
theorem exists_signedPermutation_conj {a b : ι} (hab : a ≠ b) {i j : ι} (hij : i ≠ j) :
    ∃ g : elementaryGroup ι S,
      IsSignedPermutationMatrix (((g : elementaryGroup ι S) : (Matrix ι ι S)ˣ) : Matrix ι ι S) ∧
      ConjToUpToSign hab g hij := by
  by_cases hia : i = a
  · subst hia
    by_cases hjb : j = b
    · subst hjb
      exact ⟨1, by simpa using isSignedPermutationMatrix_one (ι := ι) (S := S),
        conjToUpToSign_one hab⟩
    · have hbj : b ≠ j := Ne.symm hjb
      refine ⟨weyl (S := S) b j hbj * 1, ?_, ?_⟩
      · simpa using weyl_isSignedPermutationMatrix (S := S) b j hbj
      · exact conjToUpToSign_weyl_mul hab hab (conjToUpToSign_one hab) b j hbj
          (Equiv.swap_apply_of_ne_of_ne hab hij) (Equiv.swap_apply_left b j) hij
  · by_cases hib : i = b
    · subst hib
      by_cases hja : j = a
      · subst hja
        refine ⟨weyl (S := S) j i hab * 1, ?_, ?_⟩
        · simpa using weyl_isSignedPermutationMatrix (S := S) j i hab
        · exact conjToUpToSign_weyl_mul hab hab (conjToUpToSign_one hab) j i hab
            (Equiv.swap_apply_left j i) (Equiv.swap_apply_right j i) hij
      · have haj : a ≠ j := Ne.symm hja
        have h1 : ConjToUpToSign hab (weyl (S := S) a i hab * 1) (Ne.symm hab) :=
          conjToUpToSign_weyl_mul hab hab (conjToUpToSign_one hab) a i hab
            (Equiv.swap_apply_left a i) (Equiv.swap_apply_right a i) (Ne.symm hab)
        refine ⟨weyl (S := S) a j haj * (weyl (S := S) a i hab * 1), ?_, ?_⟩
        · simpa using isSignedPermutationMatrix_mul
            (weyl_isSignedPermutationMatrix (S := S) a j haj)
            (weyl_isSignedPermutationMatrix (S := S) a i hab)
        · exact conjToUpToSign_weyl_mul hab (Ne.symm hab) h1 a j haj
            (Equiv.swap_apply_of_ne_of_ne hia hij) (Equiv.swap_apply_left a j) hij
    · have hai : a ≠ i := Ne.symm hia
      have hbi : b ≠ i := Ne.symm hib
      have h1 : ConjToUpToSign hab (weyl (S := S) a i hai * 1) (Ne.symm hbi) :=
        conjToUpToSign_weyl_mul hab hab (conjToUpToSign_one hab) a i hai
          (Equiv.swap_apply_left a i) (Equiv.swap_apply_of_ne_of_ne (Ne.symm hab) hbi)
          (Ne.symm hbi)
      by_cases hjb : j = b
      · subst hjb
        refine ⟨weyl (S := S) a i hai * 1, ?_, h1⟩
        simpa using weyl_isSignedPermutationMatrix (S := S) a i hai
      · have hbj : b ≠ j := Ne.symm hjb
        refine ⟨weyl (S := S) b j hbj * (weyl (S := S) a i hai * 1), ?_, ?_⟩
        · simpa using isSignedPermutationMatrix_mul
            (weyl_isSignedPermutationMatrix (S := S) b j hbj)
            (weyl_isSignedPermutationMatrix (S := S) a i hai)
        · exact conjToUpToSign_weyl_mul hab (Ne.symm hbi) h1 b j hbj
            (Equiv.swap_apply_of_ne_of_ne hib hij) (Equiv.swap_apply_left b j) hij

/-- **The normal closure of all `e_{ab}(r)` contains every root**, by the
conjugations of `exists_signedPermutation_conj`, the sign absorbed by inverses. -/
theorem elGen_mem_of_forall_mem (N : Subgroup (elementaryGroup ι S)) [hN : N.Normal]
    {a b : ι} (hab : a ≠ b) (hmem : ∀ r : S, elGen a b hab r ∈ N)
    {i j : ι} (hij : i ≠ j) (r : S) : elGen i j hij r ∈ N := by
  obtain ⟨g, _, hg⟩ := exists_signedPermutation_conj (S := S) hab hij
  have hconj : ∀ x : S, g * elGen a b hab x * g⁻¹ ∈ N :=
    fun x ↦ hN.conj_mem _ (hmem x) g
  rcases hg r with h | h
  · rw [← h]; exact hconj r
  · have hneg : elGen i j hij (-r) ∈ N := by rw [← h]; exact hconj r
    rw [elGen_neg_eq_inv] at hneg
    simpa using N.inv_mem hneg

/-- A normal subgroup containing every `e_{ab}(r)` is the whole elementary group. -/
theorem normal_eq_top_of_forall_mem (N : Subgroup (elementaryGroup ι S)) [N.Normal]
    {a b : ι} (hab : a ≠ b) (hmem : ∀ r : S, elGen a b hab r ∈ N) : N = ⊤ := by
  have hle : Subgroup.closure (elementaryRootSet ι S) ≤ N := by
    rw [Subgroup.closure_le]
    rintro x ⟨i, j, hij, hx⟩
    obtain ⟨r, rfl⟩ := (mem_elementaryRootSubgroup_iff i j hij x).mp hx
    exact elGen_mem_of_forall_mem N hab hmem hij r
  apply top_unique
  rw [← elementaryRootSet_generate (I := ι) (R := S)]
  exact hle

end SignedPermutation

/-- **Printed (tex 999-1001).**  "\emph{All ranks.}  For `n ≥ 3`, signed permutation
matrices conjugate `e₁₂(r)` to `e_{ij}(±r)` for all `i ≠ j`, so the copy of
`EL₂(R)` on coordinates `1,2` normally generates `EL_n(R)`."

Stated at every `n = 2 + k` with `k ≥ 1`; coordinates `1,2` are the images of the
two coordinates of `Fin 2` under `finSumFinEquiv`, which carry the image of
`blockEmbed R k`. -/
theorem manuscriptSentence_allRanksSignedPermutations :
    ∀ (R : Type) [Ring R] (k : ℕ), 1 ≤ k →
      (∀ (i j : Fin (2 + k)) (hij : i ≠ j), ∃ g : elementaryGroup (Fin (2 + k)) R,
          IsSignedPermutationMatrix
            (((g : elementaryGroup (Fin (2 + k)) R) : (Matrix (Fin (2 + k)) (Fin (2 + k)) R)ˣ) :
              Matrix (Fin (2 + k)) (Fin (2 + k)) R) ∧
          ∀ r : R,
            g * elementaryRoot (finSumFinEquiv (Sum.inl (0 : Fin 2)))
                (finSumFinEquiv (Sum.inl (1 : Fin 2))) (blockEmbed_ne k) r * g⁻¹
              = elementaryRoot i j hij r ∨
            g * elementaryRoot (finSumFinEquiv (Sum.inl (0 : Fin 2)))
                (finSumFinEquiv (Sum.inl (1 : Fin 2))) (blockEmbed_ne k) r * g⁻¹
              = elementaryRoot i j hij (-r)) ∧
      Subgroup.normalClosure (Set.range (blockEmbed R k)) = ⊤ := by
  intro R _ k _hk
  refine ⟨fun i j hij ↦ exists_signedPermutation_conj (S := R) (blockEmbed_ne k) hij, ?_⟩
  haveI : (Subgroup.normalClosure (Set.range (blockEmbed R k))).Normal :=
    Subgroup.normalClosure_normal
  refine normal_eq_top_of_forall_mem _ (blockEmbed_ne k) fun r ↦ ?_
  have hrange : blockEmbed R k (upperRoot r) ∈ Set.range (blockEmbed R k) := ⟨_, rfl⟩
  rw [blockEmbed_upperRoot] at hrange
  exact Subgroup.subset_normalClosure hrange

end FullDefectAllRanksSignedPermutations
