/-!
# Degree bounds in ℤ-crossed products

`non_mf_groups_exist.tex`, proof of Corollary `cor:dynamic-rank-budget`
(`\label{cor:dynamic-rank-budget}`), tex 1723–1726:

> The unit $v=\sum_h u^h1_{C_h}$ implements $S_C$, and
> $P_C=\operatorname{LC}(C,k)\rtimes_{S_C}\mathbb Z$ [...]

The map `LC(C, k) ⋊_{S_C} ℤ → P_C` is injective because `v = ∑_h u^h 1_{C_h}` involves only the
positive powers `u^h`, `h ≥ 1`, and its inverse only negative powers, so the degree-zero Laurent
coefficient of the image of `∑_n f_n u_S^n` is the extension of `f_0`.  This module supplies the
degree bookkeeping for the algebraic crossed products `A ⋊ ℤ = SkewMonoidAlgebra A (Multiplicative ℤ)`
of `Pestov91.CrossedProduct`:

* `DegreesGE n x`: `x` has no coefficients in degrees `< n`; `DegreesLE n x`: none in degrees `> n`;
* closure under `0`, addition, finite sums, products (`DegreesGE.mul`: degrees add) and powers
  (`DegreesGE.pow_of_one`, `DegreesLE.pow_of_neg_one`);
* the monomials: `degreesGE_single`, `degreesLE_single`, `degreesGE_C`, `degreesLE_C`,
  `degreesGE_unit_zpow`, `degreesLE_unit_zpow`;
* reading degree zero: `coeff_one_of_degreesGE_one`, `coeff_one_of_degreesLE_neg_one`,
  `coeff_one_C`.
-/

namespace GroupApproximation
namespace Pestov91
namespace CrossedProduct

open SkewMonoidAlgebra (single)
open Multiplicative (ofAdd toAdd)

variable {A : Type*} [Ring A] [MulSemiringAction (Multiplicative ℤ) A]

/-- `x` has no coefficients in degrees `< n`. -/
def DegreesGE (n : ℤ) (x : SkewMonoidAlgebra A (Multiplicative ℤ)) : Prop :=
  ∀ g : Multiplicative ℤ, toAdd g < n → x.coeff g = 0

/-- `x` has no coefficients in degrees `> n`. -/
def DegreesLE (n : ℤ) (x : SkewMonoidAlgebra A (Multiplicative ℤ)) : Prop :=
  ∀ g : Multiplicative ℤ, n < toAdd g → x.coeff g = 0

section Closure

variable {m n : ℤ} {x y : SkewMonoidAlgebra A (Multiplicative ℤ)}

theorem degreesGE_zero (n : ℤ) : DegreesGE n (0 : SkewMonoidAlgebra A (Multiplicative ℤ)) :=
  fun g _ => by rw [SkewMonoidAlgebra.coeff_zero, Finsupp.coe_zero, Pi.zero_apply]

theorem degreesLE_zero (n : ℤ) : DegreesLE n (0 : SkewMonoidAlgebra A (Multiplicative ℤ)) :=
  fun g _ => by rw [SkewMonoidAlgebra.coeff_zero, Finsupp.coe_zero, Pi.zero_apply]

theorem DegreesGE.add (hx : DegreesGE n x) (hy : DegreesGE n y) : DegreesGE n (x + y) :=
  fun g hg => by rw [SkewMonoidAlgebra.coeff_add, Finsupp.add_apply, hx g hg, hy g hg, add_zero]

theorem DegreesLE.add (hx : DegreesLE n x) (hy : DegreesLE n y) : DegreesLE n (x + y) :=
  fun g hg => by rw [SkewMonoidAlgebra.coeff_add, Finsupp.add_apply, hx g hg, hy g hg, add_zero]

theorem degreesGE_sum {ι : Type*} (s : Finset ι) (f : ι → SkewMonoidAlgebra A (Multiplicative ℤ))
    (h : ∀ i ∈ s, DegreesGE n (f i)) : DegreesGE n (∑ i ∈ s, f i) :=
  Finset.sum_induction _ (DegreesGE n) (fun _ _ ha hb => ha.add hb) (degreesGE_zero n) h

theorem degreesLE_sum {ι : Type*} (s : Finset ι) (f : ι → SkewMonoidAlgebra A (Multiplicative ℤ))
    (h : ∀ i ∈ s, DegreesLE n (f i)) : DegreesLE n (∑ i ∈ s, f i) :=
  Finset.sum_induction _ (DegreesLE n) (fun _ _ ha hb => ha.add hb) (degreesLE_zero n) h

/-- **Degrees add under multiplication.** -/
theorem DegreesGE.mul (hx : DegreesGE m x) (hy : DegreesGE n y) : DegreesGE (m + n) (x * y) := by
  classical
  intro g hg
  rw [SkewMonoidAlgebra.coeff_mul, SkewMonoidAlgebra.sum_def']
  refine Finset.sum_eq_zero fun a₁ ha₁ => ?_
  rw [SkewMonoidAlgebra.sum_def']
  refine Finset.sum_eq_zero fun a₂ ha₂ => ?_
  have h₁ : m ≤ toAdd a₁ :=
    not_lt.1 fun h => (SkewMonoidAlgebra.mem_support_iff.1 ha₁) (hx a₁ h)
  have h₂ : n ≤ toAdd a₂ :=
    not_lt.1 fun h => (SkewMonoidAlgebra.mem_support_iff.1 ha₂) (hy a₂ h)
  rw [if_neg]
  intro heq
  have hsum := congrArg toAdd heq
  rw [toAdd_mul] at hsum
  omega

/-- **Degrees add under multiplication.** -/
theorem DegreesLE.mul (hx : DegreesLE m x) (hy : DegreesLE n y) : DegreesLE (m + n) (x * y) := by
  classical
  intro g hg
  rw [SkewMonoidAlgebra.coeff_mul, SkewMonoidAlgebra.sum_def']
  refine Finset.sum_eq_zero fun a₁ ha₁ => ?_
  rw [SkewMonoidAlgebra.sum_def']
  refine Finset.sum_eq_zero fun a₂ ha₂ => ?_
  have h₁ : toAdd a₁ ≤ m :=
    not_lt.1 fun h => (SkewMonoidAlgebra.mem_support_iff.1 ha₁) (hx a₁ h)
  have h₂ : toAdd a₂ ≤ n :=
    not_lt.1 fun h => (SkewMonoidAlgebra.mem_support_iff.1 ha₂) (hy a₂ h)
  rw [if_neg]
  intro heq
  have hsum := congrArg toAdd heq
  rw [toAdd_mul] at hsum
  omega

end Closure

section Monomials

theorem degreesGE_single {n : ℤ} {g : Multiplicative ℤ} (hg : n ≤ toAdd g) (a : A) :
    DegreesGE n (single g a : SkewMonoidAlgebra A (Multiplicative ℤ)) := by
  classical
  intro g' hg'
  rw [SkewMonoidAlgebra.coeff_single_apply, if_neg]
  rintro rfl
  omega

theorem degreesLE_single {n : ℤ} {g : Multiplicative ℤ} (hg : toAdd g ≤ n) (a : A) :
    DegreesLE n (single g a : SkewMonoidAlgebra A (Multiplicative ℤ)) := by
  classical
  intro g' hg'
  rw [SkewMonoidAlgebra.coeff_single_apply, if_neg]
  rintro rfl
  omega

theorem degreesGE_C (a : A) : DegreesGE 0 (C a : SkewMonoidAlgebra A (Multiplicative ℤ)) := by
  rw [C_apply]
  exact degreesGE_single (by simp) a

theorem degreesLE_C (a : A) : DegreesLE 0 (C a : SkewMonoidAlgebra A (Multiplicative ℤ)) := by
  rw [C_apply]
  exact degreesLE_single (by simp) a

theorem degreesGE_unit_zpow (j : ℤ) :
    DegreesGE j ((unit ^ j : (SkewMonoidAlgebra A (Multiplicative ℤ))ˣ) :
      SkewMonoidAlgebra A (Multiplicative ℤ)) := by
  rw [val_unit_zpow]
  exact degreesGE_single (by simp) 1

theorem degreesLE_unit_zpow (j : ℤ) :
    DegreesLE j ((unit ^ j : (SkewMonoidAlgebra A (Multiplicative ℤ))ˣ) :
      SkewMonoidAlgebra A (Multiplicative ℤ)) := by
  rw [val_unit_zpow]
  exact degreesLE_single (by simp) 1

/-- The degree-zero coefficient of `C a` is `a`. -/
theorem coeff_one_C (a : A) : (C a : SkewMonoidAlgebra A (Multiplicative ℤ)).coeff 1 = a := by
  classical
  rw [C_apply, SkewMonoidAlgebra.coeff_single_apply, if_pos rfl]

end Monomials

section Powers

variable {x : SkewMonoidAlgebra A (Multiplicative ℤ)}

theorem DegreesGE.pow_of_one (hx : DegreesGE 1 x) (j : ℕ) : DegreesGE (j : ℤ) (x ^ j) := by
  induction j with
  | zero =>
    rw [pow_zero, Nat.cast_zero, SkewMonoidAlgebra.one_def]
    exact degreesGE_single (by simp) 1
  | succ j ih =>
    rw [pow_succ, Nat.cast_succ]
    exact ih.mul hx

theorem DegreesLE.pow_of_neg_one (hx : DegreesLE (-1) x) (j : ℕ) :
    DegreesLE (-(j : ℤ)) (x ^ j) := by
  induction j with
  | zero =>
    rw [pow_zero, Nat.cast_zero, neg_zero, SkewMonoidAlgebra.one_def]
    exact degreesLE_single (by simp) 1
  | succ j ih =>
    rw [pow_succ, Nat.cast_succ, neg_add]
    exact ih.mul hx

/-- An element with no coefficients below degree `1` has degree-zero coefficient `0`. -/
theorem coeff_one_of_degreesGE_one (hx : DegreesGE 1 x) : x.coeff 1 = 0 :=
  hx 1 (by simp)

/-- An element with no coefficients above degree `-1` has degree-zero coefficient `0`. -/
theorem coeff_one_of_degreesLE_neg_one (hx : DegreesLE (-1) x) : x.coeff 1 = 0 :=
  hx 1 (by simp)

end Powers
