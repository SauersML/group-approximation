/-!
# The return ring map is injective

`non_mf_groups_exist.tex`, proof of Corollary `cor:dynamic-rank-budget`
(`\label{cor:dynamic-rank-budget}`), tex 1723–1726:

> The unit $v=\sum_h u^h1_{C_h}$ implements $S_C$, and
> $P_C=\operatorname{LC}(C,k)\rtimes_{S_C}\mathbb Z$ [...]

The injectivity half of the printed identification.

* `cellCoeff_zero`: `C_0 = ∅`, so `1_{C_0} = 0`;
* `degreesGE_one_returnV`: `v = ∑_h u^h 1_{C_h}` has no coefficients below degree `1`;
  `degreesLE_neg_one_returnW`: `w = ∑_h 1_{C_h} u^{-h}` has none above degree `-1`;
* `coeff_one_coeff_mul_coe_returnUnit_zpow`: the degree-zero coefficient of `f · v^n` is `f` for
  `n = 0` and `0` otherwise;
* `coeff_one_coe_returnRingHom`: the degree-zero coefficient of `θ z` is the extension by zero of
  the degree-zero coefficient of `z`;
* `returnRingHom_injective`: `θ : LC(C, k) ⋊_{S_C} ℤ →+* P_C` is injective, by reading every
  coefficient of `z` as the degree-zero coefficient of `z · u_S^{-n}`.

Laurent coefficients are written `SkewMonoidAlgebra.coeff x g`: on `ClopenCrossedProduct T k` the
field notation `x.coeff` resolves to the coefficient embedding `ClopenCrossedProduct.coeff`.
-/

namespace GroupApproximation.ChainCore

open Set MFQuotientUnits
open SkewMonoidAlgebra (single)
open Multiplicative (ofAdd toAdd)
open Pestov91.CrossedProduct (DegreesGE DegreesLE)

variable {X : Type*} [TopologicalSpace X] {T : X ≃ₜ X} {r : ℕ} {C : Set X}

/-- Positive powers in a corner are computed in the ambient ring. -/
theorem coe_corner_pow_succ {R : Type*} [Ring R] {e : R} {he : IsIdempotentElem e}
    (a : Corner R e he) (n : ℕ) : ((a ^ (n + 1) : Corner R e he) : R) = (a : R) ^ (n + 1) := by
  induction n with
  | zero => rw [zero_add, pow_one, pow_one]
  | succ n ih => rw [pow_succ, coe_corner_mul, ih, ← pow_succ]

section Injective

variable (k : Type*) [Ring k] (hC : IsReturnSection T r C)

/-- `C_0 = ∅`, so `1_{C_0} = 0`. -/
theorem cellCoeff_zero : cellCoeff k hC 0 = 0 := by
  have h : LocallyConstant.charFn k (isClopen_returnCell hC 0) = 0 := by
    ext x
    simp only [LocallyConstant.coe_charFn, returnCell_zero hC, Set.indicator_empty,
      LocallyConstant.coe_zero, Pi.zero_apply]
  rw [cellCoeff, h, map_zero]

/-- `v = ∑_h u^h 1_{C_h}` has no coefficients below degree `1`. -/
theorem degreesGE_one_returnV : DegreesGE 1 (returnV k hC) := by
  refine Pestov91.CrossedProduct.degreesGE_sum _ _ fun h _ => ?_
  rcases Nat.eq_zero_or_pos h with rfl | hpos
  · rw [cellCoeff_zero, mul_zero]
    exact Pestov91.CrossedProduct.degreesGE_zero 1
  · have h1 := (Pestov91.CrossedProduct.degreesGE_unit_zpow (A := ClopenCoeff T k) (h : ℤ)).mul
      (Pestov91.CrossedProduct.degreesGE_C
        (ClopenCoeff.of T k (LocallyConstant.charFn k (isClopen_returnCell hC h))))
    rw [add_zero] at h1
    exact h1.mono (by omega)

/-- `w = ∑_h 1_{C_h} u^{-h}` has no coefficients above degree `-1`. -/
theorem degreesLE_neg_one_returnW : DegreesLE (-1) (returnW k hC) := by
  refine Pestov91.CrossedProduct.degreesLE_sum _ _ fun h _ => ?_
  rcases Nat.eq_zero_or_pos h with rfl | hpos
  · rw [cellCoeff_zero, zero_mul]
    exact Pestov91.CrossedProduct.degreesLE_zero (-1)
  · have h1 := (Pestov91.CrossedProduct.degreesLE_C
        (ClopenCoeff.of T k (LocallyConstant.charFn k (isClopen_returnCell hC h)))).mul
      (Pestov91.CrossedProduct.degreesLE_unit_zpow (A := ClopenCoeff T k) (-(h : ℤ)))
    rw [zero_add] at h1
    exact h1.mono (by omega)

/-- **Degree zero of `f · v^n`**: `f` when `n = 0`, and `0` otherwise. -/
theorem coeff_one_coeff_mul_coe_returnUnit_zpow (f : LocallyConstant C k) (n : ℤ) :
    SkewMonoidAlgebra.coeff
        (ClopenCrossedProduct.coeff T k (clopenExtendByZero k hC.isClopen f) *
          (((returnUnit k hC ^ n : (ReturnCorner k hC)ˣ) : ReturnCorner k hC) :
            ClopenCrossedProduct T k)) 1 =
      if n = 0 then ClopenCoeff.of T k (clopenExtendByZero k hC.isClopen f) else 0 := by
  rcases lt_trichotomy n 0 with hn | rfl | hn
  · obtain ⟨m, rfl⟩ : ∃ m : ℕ, n = -((m + 1 : ℕ) : ℤ) := ⟨(-n).toNat - 1, by omega⟩
    rw [if_neg (show -((m + 1 : ℕ) : ℤ) ≠ 0 by omega), zpow_neg, zpow_natCast, ← inv_pow,
      Units.val_pow_eq_pow_val, coe_corner_pow_succ, coe_returnUnit_inv]
    have h1 := (Pestov91.CrossedProduct.degreesLE_C
      (ClopenCoeff.of T k (clopenExtendByZero k hC.isClopen f))).mul
        ((degreesLE_neg_one_returnW k hC).pow_of_neg_one (m + 1))
    exact Pestov91.CrossedProduct.coeff_one_of_degreesLE_neg_one (h1.mono (by omega))
  · rw [if_pos rfl, zpow_zero, Units.val_one, coe_cornerOne, returnP, ← map_mul,
      clopenExtendByZero_mul_charFn, ClopenCrossedProduct.coeff_apply,
      Pestov91.CrossedProduct.coeff_one_C]
  · obtain ⟨m, rfl⟩ : ∃ m : ℕ, n = ((m + 1 : ℕ) : ℤ) := ⟨n.toNat - 1, by omega⟩
    rw [if_neg (show ((m + 1 : ℕ) : ℤ) ≠ 0 by omega), zpow_natCast, Units.val_pow_eq_pow_val,
      coe_corner_pow_succ, coe_returnUnit]
    have h1 := (Pestov91.CrossedProduct.degreesGE_C
      (ClopenCoeff.of T k (clopenExtendByZero k hC.isClopen f))).mul
        ((degreesGE_one_returnV k hC).pow_of_one (m + 1))
    exact Pestov91.CrossedProduct.coeff_one_of_degreesGE_one (h1.mono (by omega))

/-- **Degree zero of `θ z`** is the extension by zero of the degree-zero coefficient of `z`. -/
theorem coeff_one_coe_returnRingHom (z : ClopenCrossedProduct (firstReturnHomeomorph hC) k) :
    SkewMonoidAlgebra.coeff
        ((returnRingHom k hC z : ReturnCorner k hC) : ClopenCrossedProduct T k) 1 =
      ClopenCoeff.of T k (clopenExtendByZero k hC.isClopen
        ((ClopenCoeff.of (firstReturnHomeomorph hC) k).symm (SkewMonoidAlgebra.coeff z 1))) := by
  classical
  induction z using SkewMonoidAlgebra.induction_on with
  | zero =>
    simp only [map_zero, coe_corner_zero, SkewMonoidAlgebra.coeff_zero, Finsupp.coe_zero,
      Pi.zero_apply, clopenExtendByZero_zero]
  | add f g hf hg =>
    simp only [map_add, coe_corner_add, SkewMonoidAlgebra.coeff_add, Finsupp.add_apply, hf, hg,
      clopenExtendByZero_add]
  | single g a =>
    obtain ⟨f, rfl⟩ := (ClopenCoeff.of (firstReturnHomeomorph hC) k).surjective a
    have hs : (single g (ClopenCoeff.of (firstReturnHomeomorph hC) k f) :
        ClopenCrossedProduct (firstReturnHomeomorph hC) k) =
        ClopenCrossedProduct.coeff (firstReturnHomeomorph hC) k f *
          ((ClopenCrossedProduct.unit (firstReturnHomeomorph hC) k ^ toAdd g :
            (ClopenCrossedProduct (firstReturnHomeomorph hC) k)ˣ) :
              ClopenCrossedProduct (firstReturnHomeomorph hC) k) := by
      rw [Pestov91.CrossedProduct.val_unit_zpow, ofAdd_toAdd, ClopenCrossedProduct.coeff_apply,
        Pestov91.CrossedProduct.single_eq_C_mul]
    rw [hs, map_mul, coe_corner_mul, coe_returnRingHom_coeff, coe_returnRingHom_unit_zpow,
      coeff_one_coeff_mul_coe_returnUnit_zpow, SkewMonoidAlgebra.coeff_single_apply]
    by_cases hg : g = 1
    · subst hg
      rw [if_pos toAdd_one, if_pos rfl, RingEquiv.symm_apply_apply]
    · have hg' : toAdd g ≠ 0 := fun h => hg (by rw [← ofAdd_toAdd g, h, ofAdd_zero])
      rw [if_neg hg', if_neg hg, map_zero, clopenExtendByZero_zero, map_zero]

/-- **The return ring map is injective.** -/
theorem returnRingHom_injective : Function.Injective (returnRingHom k hC) := by
  refine (injective_iff_map_eq_zero (returnRingHom k hC)).2 fun z hz => ?_
  refine SkewMonoidAlgebra.ext fun g => ?_
  have h := coeff_one_coe_returnRingHom k hC
    (z * single g⁻¹ (1 : ClopenCoeff (firstReturnHomeomorph hC) k))
  rw [map_mul, hz, zero_mul, coe_corner_zero, SkewMonoidAlgebra.coeff_zero, Finsupp.coe_zero,
    Pi.zero_apply] at h
  have hc : SkewMonoidAlgebra.coeff
      (z * single g⁻¹ (1 : ClopenCoeff (firstReturnHomeomorph hC) k)) 1 =
        SkewMonoidAlgebra.coeff z g := by
    have h2 := SkewMonoidAlgebra.coeff_mul_single z
      (1 : ClopenCoeff (firstReturnHomeomorph hC) k) g⁻¹ 1
    rwa [one_mul, inv_inv, smul_one, mul_one] at h2
  rw [hc] at h
  have h3 := h.symm
  rw [map_eq_zero_iff _ (ClopenCoeff.of T k).injective,
    ← clopenExtendByZero_zero k hC.isClopen] at h3
  have h4 := (map_eq_zero_iff _ (ClopenCoeff.of (firstReturnHomeomorph hC) k).symm.injective).1
    (clopenExtendByZero_injective k hC.isClopen h3)
  rw [h4, SkewMonoidAlgebra.coeff_zero, Finsupp.coe_zero, Pi.zero_apply]

end Injective
