/-!
# `P_C = LC(C, k) ⋊_{S_C} ℤ`: the closed return ring statement

`non_mf_groups_exist.tex`, proof of Corollary `cor:dynamic-rank-budget`
(`\label{cor:dynamic-rank-budget}`), tex 1721–1726:

> Let $C_h$ be the clopen first-return-time cells and $S_C$ the
> first-return homeomorphism.  Put $p_C=1_C$ and $P_C=p_CR_Xp_C$.
> The unit $v=\sum_h u^h1_{C_h}$ implements $S_C$, and
> $P_C=\operatorname{LC}(C,k)\rtimes_{S_C}\mathbb Z$: split any
> partial shift with endpoints in $C$ according to its successive
> returns.  This is a finite splitting for each fixed shift exponent.

`returnRingCrossedProductStatement_holds : ReturnRingCrossedProductStatement` closes the interface
Prop of `Dynamics/ReturnTowerStatements`.
- `S_C` is `firstReturnHomeomorph hC`, and the isomorphism is the return ring map
  `returnRingHom k hC` (`Dynamics/ReturnRingLift`).
- Injectivity is `returnRingHom_injective` (`Dynamics/ReturnRingInjective`).
- Surjectivity is ct-return-tower's `surjective_of_generators` (`Dynamics/ReturnCornerGeneration`):
  the partial shifts with endpoints in `C` split along successive returns, so `P_C` is generated by
  the extended coefficients, `v` and `w`, all of which lie in the image.
- Coefficients go to their extensions by zero, and the implementing unit goes to
  `v = ∑_h u^h 1_{C_h}`.
-/

namespace GroupApproximation.ChainCore

open Set MFQuotientUnits

variable {X : Type*} [TopologicalSpace X] {T : X ≃ₜ X} {r : ℕ} {C : Set X}

theorem returnRingHom_surjective (k : Type*) [Ring k] (hC : IsReturnSection T r C) :
    Function.Surjective (returnRingHom k hC) :=
  surjective_of_generators k hC (returnRingHom k hC)
    (fun f => ⟨ClopenCrossedProduct.coeff (firstReturnHomeomorph hC) k f,
      coe_returnRingHom_coeff k hC f⟩)
    ⟨(ClopenCrossedProduct.unit (firstReturnHomeomorph hC) k :
        ClopenCrossedProduct (firstReturnHomeomorph hC) k),
      corner_ext (by rw [coe_returnRingHom_unit, coe_returnUnit])⟩
    ⟨(((ClopenCrossedProduct.unit (firstReturnHomeomorph hC) k)⁻¹ :
        (ClopenCrossedProduct (firstReturnHomeomorph hC) k)ˣ) :
          ClopenCrossedProduct (firstReturnHomeomorph hC) k),
      corner_ext (by rw [coe_returnRingHom_unit_inv, coe_returnUnit_inv])⟩

/-- **The return ring isomorphism** `LC(C, k) ⋊_{S_C} ℤ ≃+* P_C`. -/
noncomputable def returnRingEquiv (k : Type*) [Ring k] (hC : IsReturnSection T r C) :
    ClopenCrossedProduct (firstReturnHomeomorph hC) k ≃+* ReturnCorner k hC :=
  RingEquiv.ofBijective (returnRingHom k hC)
    ⟨returnRingHom_injective k hC, returnRingHom_surjective k hC⟩

theorem returnRingEquiv_apply (k : Type*) [Ring k] (hC : IsReturnSection T r C)
    (z : ClopenCrossedProduct (firstReturnHomeomorph hC) k) :
    returnRingEquiv k hC z = returnRingHom k hC z :=
  rfl

/-- **The return ring** (tex 1723–1726): `P_C = p_C R_X p_C ≅ LC(C, k) ⋊_{S_C} ℤ`, with `S_C` the
first-return homeomorphism, coefficients sent to their extensions by zero and the implementing unit
sent to `v = ∑_h u^h 1_{C_h}`. -/
theorem returnRingCrossedProductStatement_holds : ReturnRingCrossedProductStatement := by
  intro X _ T k _ r C hC _
  refine ⟨firstReturnHomeomorph hC, isIdempotentElem_returnP k hC, returnRingEquiv k hC,
    firstReturnHomeomorph_apply hC, ?_, ?_⟩
  · intro f g hin hout
    rw [returnRingEquiv_apply, coe_returnRingHom_coeff,
      clopenExtendByZero_eq_of k hC.isClopen f g hin hout]
  · refine ⟨fun h => LocallyConstant.charFn k (isClopen_returnCell hC h), fun h x => ?_, ?_⟩
    · show (LocallyConstant.charFn k (isClopen_returnCell hC h) : X → k) x =
        Set.indicator (returnCell T C h) 1 x
      rw [LocallyConstant.coe_charFn]
    · rw [returnRingEquiv_apply, coe_returnRingHom_unit]
      rfl

end GroupApproximation.ChainCore

open GroupApproximation.ChainCore

#audit_closed_axioms returnRingCrossedProductStatement_holds
