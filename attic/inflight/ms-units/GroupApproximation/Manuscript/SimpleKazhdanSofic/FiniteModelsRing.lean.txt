/-!
# The finite ring models along the periodic word

`simple_kazhdan_sofic_group.tex` at bf961c128, "Finite models", tex 131–144:

> Let $P\delta_n=\delta_{n+1}$ on $\F_2^{\Z/N\Z}$. If $f$ depends on a window $[-\rho,\rho]$,
> evaluate its local table along $y$ to obtain a diagonal matrix $D_y(f)$. Send
> $\sum_j f_ju^j \mapsto \sum_j D_y(f_j)P^j$. For any fixed finite list of sums and products in $R$,
> choose $k$ larger than all windows involved, including those shifted in the products. The
> relation $P^iD_y(f)P^{-i}=D_y(f\circ T^{-i})$ then preserves every listed operation. Nonzero
> coefficients remain nonzero, since every allowed window occurs in $y$. Taking $N$ larger than
> twice the largest absolute exponent makes the corresponding cyclic diagonals distinct; hence
> nonzero differences also remain nonzero.

The map `∑_j f_j u^j ↦ ∑_j D_y(f_j) P^j` is Pestov91's `periodicModel D`
(`single g a ↦ diagonal (D a) * shiftMatrix K N (toAdd g)`), reindexed from `ZMod N` to `Fin N`.
For a finite list `t ⊆ R`, with degree bound `B` and the coefficients and coefficient differences
`S` of `t`:

* `D` evaluates a coefficient at points of `X` realizing the windows of the periodic word `y` of
  `MinimalSubshiftPeriodicWord`, taken with `k` at least every radius (`WordGraph.exists_radius`)
  plus `B`, and `N ≥ 2B + 1`;
* covariance `D(f ∘ T^{-i})(n) = D(f)(n - i)` for `|i| ≤ B` (`WordGraph.zpow_apply_coord`) gives
  `periodicModel_mul`: every listed product is preserved;
* a nonzero coefficient has a witness point whose window occurs in `y`, so `D` keeps it nonzero;
* `eq_of_periodicModel_eq` with `2B < N`: distinct exponents give distinct cyclic diagonals, so the
  model is injective on `t`.

* `exists_matrixModel_clopenCrossedProduct`: every finite subset of `R_X = LC(X, k) ⋊_T ℤ` has an
  injective model into `M_N(k)` preserving `0`, `1` and the sums and products of pairs, the form of
  ms-core-3's `IsMatricialLEFRing`.
* `isLEFRing_clopenCrossedProduct_of_minimal`: `R_X` is an LEF ring over every finite ring `k`.
* `PrintedFiniteModelsRing`, `printedFiniteModelsRing`: the models of the note over `F₂`, closed.
-/

/-- **The ring models of the note** (tex 131–144): for a minimal subshift over a finite alphabet, every
finite subset of `LC(X, k) ⋊_T ℤ` has an injective model into `M_N(k)` preserving `0`, `1` and the
sums and products of pairs from the subset. -/
theorem exists_matrixModel_clopenCrossedProduct {A : Type} [TopologicalSpace A]
    [DiscreteTopology A] [Finite A] {X : Set (ℤ → A)} (hXc : IsClosed X)
    (hX : ∀ n : ℤ, Set.MapsTo (shift n) X X) (hne : X.Nonempty)
    (hmin : ∀ C : Set (ℤ → A), IsClosed C → C ⊆ X → (∀ g : ℤ, Set.MapsTo (shift g) C C) →
      C = ∅ ∨ C = X)
    (T : ↥X ≃ₜ ↥X) (hT : ∀ x : ↥X, (T x).1 = shift 1 x.1) (k : Type) [Ring k]
    (t : Finset (ClopenCrossedProduct T k)) :
    ∃ (N : ℕ) (φ : ClopenCrossedProduct T k → Matrix (Fin N) (Fin N) k),
      Set.InjOn φ (t : Set (ClopenCrossedProduct T k)) ∧ φ 0 = 0 ∧ φ 1 = 1 ∧
        ∀ a ∈ t, ∀ b ∈ t, φ (a + b) = φ a + φ b ∧ φ (a * b) = φ a * φ b := by
  classical
  obtain ⟨B, hbd⟩ : ∃ B : ℕ, ∀ x ∈ t, ∀ g ∈ x.support, |toAdd g| ≤ B :=
    ⟨(t.biUnion SkewMonoidAlgebra.support).sup fun g ↦ (toAdd g).natAbs,
      fun x hx g hg ↦ by
        rw [Int.abs_eq_natAbs]
        exact_mod_cast Finset.le_sup (f := fun g : Multiplicative ℤ ↦ (toAdd g).natAbs)
          (Finset.mem_biUnion.mpr ⟨x, hx, hg⟩)⟩
  obtain ⟨S, hSdiff, hScoeff⟩ : ∃ S : Finset (ClopenCoeff T k),
      (∀ x ∈ t, ∀ y ∈ t, ∀ g ∈ x.support ∪ y.support, x.coeff g - y.coeff g ∈ S) ∧
      (∀ x ∈ t, ∀ g ∈ x.support, x.coeff g ∈ S) :=
    ⟨(t ×ˢ t).biUnion (fun p ↦ (p.1.support ∪ p.2.support).image fun g ↦ p.1.coeff g - p.2.coeff g)
        ∪ t.biUnion (fun x ↦ x.support.image x.coeff),
      fun x hx y hy g hg ↦ Finset.mem_union_left _ (Finset.mem_biUnion.mpr
        ⟨(x, y), Finset.mem_product.mpr ⟨hx, hy⟩, Finset.mem_image.mpr ⟨g, hg, rfl⟩⟩),
      fun x hx g hg ↦ Finset.mem_union_right _ (Finset.mem_biUnion.mpr
        ⟨x, hx, Finset.mem_image.mpr ⟨g, hg, rfl⟩⟩)⟩
  choose r hr using fun a : ClopenCoeff T k => exists_radius hXc ((ClopenCoeff.of T k).symm a)
  obtain ⟨ρ, hρr⟩ : ∃ ρ : ℕ, ∀ a ∈ S, r a + B ≤ ρ :=
    ⟨S.sup r + B, fun a ha => by have h := Finset.le_sup (f := r) ha; omega⟩
  obtain ⟨N, hNz, y, hN0, hyL, hyall⟩ :=
    exists_periodic_sameWords hXc hX hne hmin ρ (2 * B + 1)
  choose z hzX hzw using hyL
  obtain ⟨w, hw⟩ : ∃ w : ZMod N → ↥X, ∀ (n : ZMod N) (i : ℤ), |i| ≤ ρ →
      (w n).1 i = y (n + ((ρ + i : ℤ) : ZMod N)) := by
    refine ⟨fun n => ⟨shift (ρ : ℤ) (z n), hX ρ (hzX n)⟩, fun n i hi => ?_⟩
    obtain ⟨hi1, hi2⟩ := abs_le.mp hi
    obtain ⟨j, hj⟩ : ∃ j : ℕ, (j : ℤ) = ρ + i := ⟨(ρ + i).toNat, Int.toNat_of_nonneg (by omega)⟩
    have h := congrFun (hzw n) ⟨j, by omega⟩
    simp only [word_apply, zero_add] at h
    show z n (ρ + i) = _
    rw [← hj, Int.cast_natCast]
    exact h
  let D : ClopenCoeff T k →+* (ZMod N → k) :=
    { toFun := fun a n => (ClopenCoeff.of T k).symm a (w n)
      map_one' := rfl
      map_mul' := fun _ _ => rfl
      map_zero' := rfl
      map_add' := fun _ _ => rfl }
  have hnz : ∀ a ∈ S, a ≠ 0 → D a ≠ 0 := by
    intro a ha hne0 hD
    obtain ⟨x0, hx0⟩ : ∃ x0 : ↥X, (ClopenCoeff.of T k).symm a x0 ≠ 0 := by
      by_contra h
      push Not at h
      exact hne0 ((ClopenCoeff.of T k).symm.map_eq_zero_iff.mp
        (LocallyConstant.ext h : (ClopenCoeff.of T k).symm a = 0))
    obtain ⟨n0, hn0⟩ := hyall _ (word_mem_language hX x0.2 (-(ρ : ℤ)) (2 * ρ + 1))
    refine hx0 ?_
    have h0 : (ClopenCoeff.of T k).symm a (w n0) = 0 := congrFun hD n0
    rw [← h0]
    refine hr a _ _ fun i hi => ?_
    have hra := hρr a ha
    obtain ⟨hi1, hi2⟩ := abs_le.mp hi
    obtain ⟨j, hj⟩ : ∃ j : ℕ, (j : ℤ) = ρ + i := ⟨(ρ + i).toNat, Int.toNat_of_nonneg (by omega)⟩
    rw [hw n0 i (abs_le.mpr ⟨by omega, by omega⟩), ← hj, Int.cast_natCast]
    have h := congrFun hn0 ⟨j, by omega⟩
    simp only [word_apply] at h
    rw [h]
    exact congrArg _ (by omega)
  have hcov : ∀ a ∈ S, ∀ g : Multiplicative ℤ, |toAdd g| ≤ B → ∀ n : ZMod N,
      D (g • a) n = D a (n - ((toAdd g : ℤ) : ZMod N)) := by
    intro a ha g hg n
    show (ClopenCoeff.of T k).symm a ((T ^ (-toAdd g)) (w n)) =
      (ClopenCoeff.of T k).symm a (w (n - ((toAdd g : ℤ) : ZMod N)))
    refine hr a _ _ fun i hi => ?_
    have hra := hρr a ha
    obtain ⟨hg1, hg2⟩ := abs_le.mp hg
    obtain ⟨hi1, hi2⟩ := abs_le.mp hi
    rw [zpow_apply_coord hT, hw n (-toAdd g + i) (abs_le.mpr ⟨by omega, by omega⟩),
      hw (n - ((toAdd g : ℤ) : ZMod N)) i (abs_le.mpr ⟨by omega, by omega⟩)]
    congr 1
    push_cast
    ring
  let e : ZMod N ≃ Fin N := (ZMod.finEquiv N).symm.toEquiv
  refine ⟨N, fun x => Matrix.reindexRingEquiv k e (Pestov91.periodicModel D x), ?_, ?_, ?_,
    fun x hx y' hy' => ⟨?_, ?_⟩⟩
  · intro x hx y' hy' hxy
    refine Pestov91.eq_of_periodicModel_eq D (by omega) (hbd x hx) (hbd y' hy') (fun g hg h0 => ?_)
      ((Matrix.reindexRingEquiv k e).injective hxy)
    by_contra hne'
    exact hnz _ (hSdiff x hx y' hy' g hg) (sub_ne_zero.mpr hne') h0
  · simp only [map_zero]
  · simp only [Pestov91.periodicModel_one, map_one]
  · simp only [map_add]
  · simp only [Pestov91.periodicModel_mul D x y' fun g hg h hh n =>
      hcov _ (hScoeff y' hy' h hh) g (hbd x hx g hg) n, map_mul]

/-- **`R_X` is an LEF ring** over every finite ring, for a minimal subshift (tex 118–144). -/
theorem isLEFRing_clopenCrossedProduct_of_minimal {A : Type} [TopologicalSpace A]
    [DiscreteTopology A] [Finite A] {X : Set (ℤ → A)} (hXc : IsClosed X)
    (hX : ∀ n : ℤ, Set.MapsTo (shift n) X X) (hne : X.Nonempty)
    (hmin : ∀ C : Set (ℤ → A), IsClosed C → C ⊆ X → (∀ g : ℤ, Set.MapsTo (shift g) C C) →
      C = ∅ ∨ C = X)
    (T : ↥X ≃ₜ ↥X) (hT : ∀ x : ↥X, (T x).1 = shift 1 x.1) (k : Type) [Ring k] [Finite k] :
    Pestov91.IsLEFRing (ClopenCrossedProduct T k) := fun t => by
  obtain ⟨N, φ, hinj, h0, h1, hφ⟩ :=
    exists_matrixModel_clopenCrossedProduct hXc hX hne hmin T hT k t
  exact ⟨Matrix (Fin N) (Fin N) k, inferInstance,
    inferInstanceAs (Finite (Fin N → Fin N → k)), φ, hinj, h0, h1, hφ⟩

/-- **The finite ring models of the note** (tex 118–144): for an infinite minimal subshift `X` over a
finite alphabet with shift `T`, every finite subset of `R = LC(X, F₂) ⋊_T ℤ` has an injective model
into `M_N(F₂)` preserving `0`, `1` and the sums and products of pairs from the subset. -/
def PrintedFiniteModelsRing : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Finite A] (X : Set (ℤ → A)),
    IsClosed X → (∀ n : ℤ, Set.MapsTo (shift n) X X) → X.Infinite →
    (∀ C : Set (ℤ → A), IsClosed C → C ⊆ X → (∀ g : ℤ, Set.MapsTo (shift g) C C) →
      C = ∅ ∨ C = X) →
    ∀ (T : ↥X ≃ₜ ↥X), (∀ x : ↥X, (T x).1 = shift 1 x.1) →
    ∀ t : Finset (ClopenCrossedProduct T (ZMod 2)),
      ∃ (N : ℕ) (φ : ClopenCrossedProduct T (ZMod 2) → Matrix (Fin N) (Fin N) (ZMod 2)),
        Set.InjOn φ (t : Set (ClopenCrossedProduct T (ZMod 2))) ∧ φ 0 = 0 ∧ φ 1 = 1 ∧
          ∀ a ∈ t, ∀ b ∈ t, φ (a + b) = φ a + φ b ∧ φ (a * b) = φ a * φ b

theorem printedFiniteModelsRing : PrintedFiniteModelsRing :=
  fun _ _ _ _ _ hXc hX hinf hmin T hT t =>
    exists_matrixModel_clopenCrossedProduct hXc hX hinf.nonempty hmin T hT (ZMod 2) t

end SimpleKazhdanSofic
