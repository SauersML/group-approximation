/-!
# Periodic words with the same words as a minimal subshift

`simple_kazhdan_sofic_group.tex` at bf961c128, "Finite models", tex 123–129:

> For each $k$, there is a periodic sequence $y=w^\infty$ with exactly the same words of length
> $2k+1$ as $X$. Indeed, minimality implies uniform recurrence. In a point of $X$, choose two
> occurrences of the same word of length $2k$, separated by a segment $w$ long enough to contain
> every word of length $2k+1$. Repeating this segment introduces no new word of that length at the
> join. Its length $N$ can be chosen arbitrarily large.

Setting: `A` a finite discrete alphabet and `X ⊆ A^ℤ` closed and shift-invariant (the `mapsTo`
field of a Mathlib `Subshift`), with chain-words' `word` and `language`.  Minimality is taken in the
shape of the note's setting (`SimpleKazhdanSofic.IsMinimal`, tex l.45): every nonempty closed
shift-invariant subset of `X` is `X`.

* `subset_closure_orbit_of_minimal`: every orbit is dense in `X`.
* `IsUniformlyRecurrentAt X n`: some length `ℓ` such that every word of length `n` of `X` occurs in
  every window of length `ℓ` of every point of `X`.
* `isUniformlyRecurrentAt_of_minimal`: "minimality implies uniform recurrence".
* `exists_occurrences`: two occurrences of the same word of length `2k`, at `0` and at `N`, with the
  segment `w = x[0, N)` containing every word of length `2k + 1`, and `N` as large as prescribed.
* `periodicWord_window_eq`, `periodicWord_window_mem_language`: every window of `y = w^∞` is read
  straight from `x` ("no new word at the join").
* `periodicWord_exists_window_eq`: every word of length `2k + 1` of `X` occurs in `y`.
* `exists_periodic_sameWords`: the periodic sequence with exactly the same words.
-/

namespace GroupApproximation
namespace SimpleKazhdanSofic

open SymbolicDynamics.FullShift WordGraph

variable {A : Type*}

/-- **Every orbit of a minimal subshift is dense.**  The closure of an orbit is a nonempty closed
shift-invariant subset of `X`. -/
theorem subset_closure_orbit_of_minimal [TopologicalSpace A] {X : Set (ℤ → A)} (hXc : IsClosed X)
    (hX : ∀ m : ℤ, Set.MapsTo (shift m) X X)
    (hmin : ∀ C : Set (ℤ → A), IsClosed C → C ⊆ X → (∀ g : ℤ, Set.MapsTo (shift g) C C) →
      C = ∅ ∨ C = X)
    {x : ℤ → A} (hx : x ∈ X) : X ⊆ closure (Set.range fun n : ℤ => shift n x) := by
  have hsub : closure (Set.range fun n : ℤ => shift n x) ⊆ X :=
    hXc.closure_subset_iff.2 (Set.range_subset_iff.2 fun n => hX n hx)
  have hinv : ∀ g : ℤ, Set.MapsTo (shift g) (closure (Set.range fun n : ℤ => shift n x))
      (closure (Set.range fun n : ℤ => shift n x)) := by
    intro g
    refine Set.MapsTo.closure ?_ (continuous_shift g)
    rintro _ ⟨n, rfl⟩
    exact ⟨n + g, shift_add n g x⟩
  rcases hmin _ isClosed_closure hsub hinv with h | h
  · exact absurd h (Set.Nonempty.ne_empty ⟨x, subset_closure ⟨0, shift_zero x⟩⟩)
  · exact h.symm.le

/-- **Uniform recurrence** at length `n`: every word of length `n` of `X` occurs in every window of
length `ℓ` of every point of `X`. -/
def IsUniformlyRecurrentAt (X : Set (ℤ → A)) (n : ℕ) : Prop :=
  ∃ ℓ : ℕ, ∀ x ∈ X, ∀ i : ℤ, ∀ u ∈ language X n, ∃ j : ℕ, j + n ≤ ℓ ∧ word x (i + j) n = u

/-- **Minimality implies uniform recurrence** (tex 124–125).  The cylinders of a word, translated
along the orbit, cover `X`; compactness keeps finitely many translates. -/
theorem isUniformlyRecurrentAt_of_minimal [TopologicalSpace A] [DiscreteTopology A] [Finite A]
    {X : Set (ℤ → A)} (hXc : IsClosed X) (hX : ∀ m : ℤ, Set.MapsTo (shift m) X X)
    (hmin : ∀ C : Set (ℤ → A), IsClosed C → C ⊆ X → (∀ g : ℤ, Set.MapsTo (shift g) C C) →
      C = ∅ ∨ C = X) (n : ℕ) : IsUniformlyRecurrentAt X n := by
  classical
  have hone : ∀ u ∈ language X n, ∃ B : ℕ, ∀ x ∈ X, ∃ m : ℤ, |m| ≤ B ∧ word x m n = u := by
    intro u hu
    obtain ⟨z, hz, hzu⟩ := hu
    have hUo : ∀ m : ℤ, IsOpen ((fun y : ℤ → A => word y m n) ⁻¹' {u}) := fun m =>
      (continuous_pi fun j : Fin n => continuous_apply (m + (j : ℕ))).isOpen_preimage {u}
        (isOpen_discrete _)
    have hcov : X ⊆ ⋃ m : ℤ, (fun y : ℤ → A => word y m n) ⁻¹' {u} := by
      intro x hx
      obtain ⟨y, hyU, m, rfl⟩ :=
        mem_closure_iff.1 (subset_closure_orbit_of_minimal hXc hX hmin hx hz) _ (hUo 0) hzu
      refine Set.mem_iUnion.2 ⟨m, ?_⟩
      have h : word (shift m x) 0 n = u := hyU
      rw [word_shift, add_zero] at h
      exact h
    obtain ⟨t, ht⟩ := hXc.isCompact.elim_finite_subcover _ hUo hcov
    refine ⟨t.sup Int.natAbs, fun x hx => ?_⟩
    obtain ⟨m, hmt, hm⟩ := Set.mem_iUnion₂.1 (ht hx)
    refine ⟨m, ?_, hm⟩
    have hle := Finset.le_sup (f := Int.natAbs) hmt
    rw [Int.abs_eq_natAbs]
    exact_mod_cast hle
  haveI : Fintype A := Fintype.ofFinite A
  choose B hB using hone
  obtain ⟨Bmax, hBmax⟩ : ∃ Bmax : ℕ, ∀ u (hu : u ∈ language X n), B u hu ≤ Bmax :=
    ⟨Finset.univ.sup fun u : Fin n → A => if h : u ∈ language X n then B u h else 0,
      fun u hu => by
        have hle := Finset.le_sup
          (f := fun u : Fin n → A => if h : u ∈ language X n then B u h else 0) (Finset.mem_univ u)
        simpa only [dif_pos hu] using hle⟩
  refine ⟨2 * Bmax + n, fun x hx i u hu => ?_⟩
  obtain ⟨m, hm, hmw⟩ := hB u hu (shift (i + Bmax) x) (hX _ hx)
  rw [word_shift] at hmw
  have hBu := hBmax u hu
  rw [abs_le] at hm
  refine ⟨(Bmax + m).toNat, by omega, ?_⟩
  rw [show i + (((Bmax + m).toNat : ℕ) : ℤ) = i + Bmax + m by omega]
  exact hmw

/-- **Two occurrences of the same word, far apart** (tex 125–129): in a point `x` of `X`, the word of
length `2k` at `0` occurs again at `N`, the segment `x[0, N)` contains every word of length
`2k + 1` of `X`, and `N` is at least any prescribed `N0`. -/
theorem exists_occurrences [TopologicalSpace A] [DiscreteTopology A] [Finite A]
    {X : Set (ℤ → A)} (hXc : IsClosed X) (hX : ∀ m : ℤ, Set.MapsTo (shift m) X X)
    (hne : X.Nonempty)
    (hmin : ∀ C : Set (ℤ → A), IsClosed C → C ⊆ X → (∀ g : ℤ, Set.MapsTo (shift g) C C) →
      C = ∅ ∨ C = X) (k N0 : ℕ) :
    ∃ (x : ℤ → A) (N : ℕ), x ∈ X ∧ N0 ≤ N ∧ 2 * k + 1 ≤ N ∧
      word x 0 (2 * k) = word x N (2 * k) ∧
      ∀ u ∈ language X (2 * k + 1), ∃ i : ℕ, i + (2 * k + 1) ≤ N ∧ word x i (2 * k + 1) = u := by
  obtain ⟨ℓ, hℓ⟩ := isUniformlyRecurrentAt_of_minimal hXc hX hmin (2 * k + 1)
  obtain ⟨ℓ', hℓ'⟩ := isUniformlyRecurrentAt_of_minimal hXc hX hmin (2 * k)
  obtain ⟨x, hx⟩ := hne
  obtain ⟨j, -, hjv⟩ := hℓ' x hx ((ℓ + N0 + 2 * k + 1 : ℕ) : ℤ) (word x 0 (2 * k))
    (word_mem_language hX hx 0 (2 * k))
  refine ⟨x, ℓ + N0 + 2 * k + 1 + j, hx, by omega, by omega, ?_, fun u hu => ?_⟩
  · rw [← hjv]
    push_cast
    rfl
  · obtain ⟨i, hiℓ, hiu⟩ := hℓ x hx 0 u hu
    refine ⟨i, by omega, ?_⟩
    rw [← hiu, zero_add]

variable {x : ℤ → A} {k N : ℕ}

/-- The periodic sequence `y = w^∞` of the segment `w = x[0, N)`. -/
def periodicWord (x : ℤ → A) (N : ℕ) (n : ZMod N) : A :=
  x (n.val : ℤ)

/-- `y` agrees with `x` on `[0, N + 2k)`, using the repeated word of length `2k` at `N`. -/
theorem periodicWord_natCast_eq [NeZero N] (hrep : word x 0 (2 * k) = word x N (2 * k))
    (hNk : 2 * k + 1 ≤ N) {i : ℕ} (hi : i < N + 2 * k) : periodicWord x N (i : ZMod N) = x i := by
  by_cases hiN : i < N
  · rw [periodicWord, ZMod.val_natCast, Nat.mod_eq_of_lt hiN]
  · obtain ⟨t, rfl⟩ : ∃ t : ℕ, i = N + t := ⟨i - N, by omega⟩
    have ht : t < 2 * k := by omega
    have h1 : x (0 + (t : ℤ)) = x ((N : ℤ) + (t : ℤ)) := congrFun hrep ⟨t, ht⟩
    rw [periodicWord, ZMod.val_natCast, Nat.add_mod_left, Nat.mod_eq_of_lt (by omega : t < N),
      Nat.cast_add]
    rw [zero_add] at h1
    exact h1

/-- **Every window of `y` is read straight from `x`** (tex 127–128, "no new word at the join"). -/
theorem periodicWord_window_eq [NeZero N] (hrep : word x 0 (2 * k) = word x N (2 * k))
    (hNk : 2 * k + 1 ≤ N) (n : ZMod N) :
    (fun j : Fin (2 * k + 1) => periodicWord x N (n + (j : ℕ))) = word x (n.val : ℤ) (2 * k + 1) := by
  funext j
  have hj := j.2
  have hn := ZMod.val_lt n
  have heq : n + ((j : ℕ) : ZMod N) = ((n.val + (j : ℕ) : ℕ) : ZMod N) := by
    rw [Nat.cast_add, ZMod.natCast_zmod_val]
  rw [heq, periodicWord_natCast_eq hrep hNk (by omega), word_apply]
  push_cast
  rfl

theorem periodicWord_window_mem_language {X : Set (ℤ → A)} (hX : ∀ m : ℤ, Set.MapsTo (shift m) X X)
    (hx : x ∈ X) [NeZero N] (hrep : word x 0 (2 * k) = word x N (2 * k)) (hNk : 2 * k + 1 ≤ N)
    (n : ZMod N) :
    (fun j : Fin (2 * k + 1) => periodicWord x N (n + (j : ℕ))) ∈ language X (2 * k + 1) := by
  rw [periodicWord_window_eq hrep hNk]
  exact word_mem_language hX hx _ _

/-- **Every word of length `2k + 1` of `X` occurs in `y`**, since it occurs in the segment `w`. -/
theorem periodicWord_exists_window_eq {X : Set (ℤ → A)} [NeZero N]
    (hrep : word x 0 (2 * k) = word x N (2 * k)) (hNk : 2 * k + 1 ≤ N)
    (hseg : ∀ u ∈ language X (2 * k + 1), ∃ i : ℕ, i + (2 * k + 1) ≤ N ∧ word x i (2 * k + 1) = u)
    {u : Fin (2 * k + 1) → A} (hu : u ∈ language X (2 * k + 1)) :
    ∃ n : ZMod N, (fun j : Fin (2 * k + 1) => periodicWord x N (n + (j : ℕ))) = u := by
  obtain ⟨i, hiN, hiu⟩ := hseg u hu
  refine ⟨(i : ZMod N), ?_⟩
  rw [periodicWord_window_eq hrep hNk, ZMod.val_natCast, Nat.mod_eq_of_lt (by omega), ← hiu]

/-- **The periodic sequence of the note** (tex 123–129): for every `k` and every `N0` there is a
periodic sequence `y = w^∞` of period `N ≥ N0` with exactly the same words of length `2k + 1` as
`X`. -/
theorem exists_periodic_sameWords [TopologicalSpace A] [DiscreteTopology A] [Finite A]
    {X : Set (ℤ → A)} (hXc : IsClosed X) (hX : ∀ m : ℤ, Set.MapsTo (shift m) X X)
    (hne : X.Nonempty)
    (hmin : ∀ C : Set (ℤ → A), IsClosed C → C ⊆ X → (∀ g : ℤ, Set.MapsTo (shift g) C C) →
      C = ∅ ∨ C = X) (k N0 : ℕ) :
    ∃ (N : ℕ) (_ : NeZero N) (y : ZMod N → A), N0 ≤ N ∧
      (∀ n : ZMod N, (fun j : Fin (2 * k + 1) => y (n + (j : ℕ))) ∈ language X (2 * k + 1)) ∧
      ∀ u ∈ language X (2 * k + 1), ∃ n : ZMod N, (fun j : Fin (2 * k + 1) => y (n + (j : ℕ))) = u := by
  obtain ⟨x, N, hx, hN0, hNk, hrep, hseg⟩ := exists_occurrences hXc hX hne hmin k N0
  haveI : NeZero N := ⟨by omega⟩
  exact ⟨N, inferInstance, periodicWord x N, hN0, periodicWord_window_mem_language hX hx hrep hNk,
    fun u hu => periodicWord_exists_window_eq hrep hNk hseg hu⟩
