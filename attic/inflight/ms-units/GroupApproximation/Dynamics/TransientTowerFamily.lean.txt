/-!
# Tower levels over clopen cells give matrix units in the transient ideal

`non_mf_groups_exist.tex`, `\label{lem:transient-matrices}`, proof, tex 1508–1512:

> Refine it according to the occupied atoms, the shift exponents to each level, and all original
> coefficient values.  On each cell the classes have a fixed size $h$ and continuously enumerated
> clopen levels.  The characteristic arrows between these levels form a copy of $M_h(k)$.

Setting: `T : X ≃ₜ X`, `C ⊆ X` with the orbit bound `ExponentBound T C m`, and finitely many
pairwise disjoint clopen cells `cell a` inside the representative sets `baseCell T C E m (S a)`.
The levels of cell `a` are the translates `T^d(cell a)` for `d ∈ S a`, enumerated increasingly.

* `transientTowerExp`, `transientTowerLevel`, `transientTowerCoeff`: the exponents, the level sets
  and their indicators.
* `transientTowerCoeff_cov`, `transientTowerCoeff_mul_self`, `transientTowerCoeff_mul_of_disjoint`,
  `disjoint_transientTowerLevel`: the hypotheses of chain-matricial's
  `isMatrixUnitFamily_towerUnit`.
* `isMatrixUnitFamily_transientTower`: the characteristic arrows form a matrix unit family.
* `transientTowerUnit_mem_transientIdeal`: when `C` avoids `Y`, the arrows lie in the transient
  ideal.
* `transientTowerUnit_self_ne_zero`: nonempty cells give nonzero diagonal arrows.
-/

namespace GroupApproximation
namespace ClopenCrossedProduct

open ChainCore
open Multiplicative (ofAdd)

section Tower

variable {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type*) [Ring k]
  {C : Set X} {E : Finset ℤ} {m M : ℕ} (cell : Fin M → Set X) (S : Fin M → Finset ℤ)

/-- The exponent of level `i` of cell `a`: the `i`-th element of `S a`, in increasing order. -/
noncomputable def transientTowerExp (a : Fin M) (i : Fin (S a).card) : ℤ :=
  ((S a).orderIsoOfFin rfl i : ℤ)

theorem transientTowerExp_mem (a : Fin M) (i : Fin (S a).card) : transientTowerExp S a i ∈ S a :=
  ((S a).orderIsoOfFin rfl i).2

theorem transientTowerExp_injective (a : Fin M) : Function.Injective (transientTowerExp S a) :=
  fun _ _ hij => ((S a).orderIsoOfFin rfl).injective (Subtype.ext hij)

theorem exists_transientTowerExp_eq {a : Fin M} {d : ℤ} (hd : d ∈ S a) :
    ∃ i, transientTowerExp S a i = d :=
  ⟨((S a).orderIsoOfFin rfl).symm ⟨d, hd⟩,
    congrArg Subtype.val (((S a).orderIsoOfFin rfl).apply_symm_apply ⟨d, hd⟩)⟩

/-- Level `i` of cell `a`: the translate `T^{d_i}(cell a)`. -/
def transientTowerLevel (a : Fin M) (i : Fin (S a).card) : Set X :=
  ⇑(T ^ transientTowerExp S a i) '' cell a

theorem isClopen_transientTowerLevel (hcell : ∀ a, IsClopen (cell a)) (a : Fin M)
    (i : Fin (S a).card) : IsClopen (transientTowerLevel T cell S a i) :=
  Dynamics.isClopen_zpow_image T _ (hcell a)

variable {cell} in
theorem mem_transientTowerLevel_iff {a : Fin M} {i : Fin (S a).card} {z : X} :
    z ∈ transientTowerLevel T cell S a i ↔ ∃ y ∈ cell a, (T ^ transientTowerExp S a i) y = z :=
  Iff.rfl

/-- The indicator of a level, as a coefficient of `R_X`. -/
noncomputable def transientTowerCoeff (hcell : ∀ a, IsClopen (cell a)) (a : Fin M)
    (i : Fin (S a).card) : ClopenCoeff T k :=
  ClopenCoeff.of T k (LocallyConstant.charFn k (isClopen_transientTowerLevel T cell S hcell a i))

theorem transientCharFn_congr {U V : Set X} (hU : IsClopen U) (hV : IsClopen V) (h : U = V) :
    LocallyConstant.charFn k hU = LocallyConstant.charFn k hV := by
  subst h
  rfl

variable {cell S}
variable (hcell : ∀ a, IsClopen (cell a))

theorem image_transientTowerLevel (a : Fin M) (i j : Fin (S a).card) :
    ⇑(T ^ (transientTowerExp S a i - transientTowerExp S a j)) '' transientTowerLevel T cell S a j =
      transientTowerLevel T cell S a i := by
  rw [transientTowerLevel, transientTowerLevel, Set.image_image]
  refine Set.image_congr fun x _ => ?_
  rw [Dynamics.zpow_apply_zpow_apply, sub_add_cancel]

/-- **Covariance**: level `i` is the translate of level `j`. -/
theorem transientTowerCoeff_cov (a : Fin M) (i j : Fin (S a).card) :
    ofAdd (transientTowerExp S a i - transientTowerExp S a j) •
        transientTowerCoeff T k cell S hcell a j =
      transientTowerCoeff T k cell S hcell a i := by
  rw [transientTowerCoeff, ClopenCoeff.smul_of, comap_zpow_charFn, transientTowerCoeff]
  exact congrArg (ClopenCoeff.of T k)
    (transientCharFn_congr k _ _ (image_transientTowerLevel T a i j))

theorem transientTowerCoeff_mul_self (a : Fin M) (i : Fin (S a).card) :
    transientTowerCoeff T k cell S hcell a i * transientTowerCoeff T k cell S hcell a i =
      transientTowerCoeff T k cell S hcell a i := by
  rw [transientTowerCoeff, ← map_mul, charFn_mul_charFn_of_subset k _ _ subset_rfl]

theorem transientTowerCoeff_mul_of_disjoint {a b : Fin M} {i : Fin (S a).card}
    {j : Fin (S b).card}
    (h : Disjoint (transientTowerLevel T cell S a i) (transientTowerLevel T cell S b j)) :
    transientTowerCoeff T k cell S hcell a i * transientTowerCoeff T k cell S hcell b j = 0 := by
  rw [transientTowerCoeff, transientTowerCoeff, ← map_mul, charFn_mul_charFn_of_disjoint k _ _ h,
    map_zero]

variable {T}

theorem transientTowerLevel_subset_level (hsub : ∀ a, cell a ⊆ baseCell T C E m (S a))
    (a : Fin M) (i : Fin (S a).card) :
    transientTowerLevel T cell S a i ⊆ level T C E m (S a) (transientTowerExp S a i) := by
  rw [level, ← image_zpow_eq_preimage_neg]
  exact Set.image_mono (hsub a)

/-- **Distinct levels are disjoint**, within a cell and across cells. -/
theorem disjoint_transientTowerLevel (hm : ExponentBound T C m)
    (hsub : ∀ a, cell a ⊆ baseCell T C E m (S a))
    (hdisj : ∀ a b, a ≠ b → Disjoint (cell a) (cell b)) {a b : Fin M} (i : Fin (S a).card)
    (j : Fin (S b).card) (hab : a ≠ b ∨ transientTowerExp S a i ≠ transientTowerExp S b j) :
    Disjoint (transientTowerLevel T cell S a i) (transientTowerLevel T cell S b j) := by
  by_cases hsame : (S a, transientTowerExp S a i) = (S b, transientTowerExp S b j)
  · obtain ⟨-, hd⟩ := Prod.mk.inj_iff.1 hsame
    rcases hab with hab | hne
    · rw [transientTowerLevel, transientTowerLevel, hd]
      exact (Set.disjoint_image_iff (T ^ transientTowerExp S b j).injective).2 (hdisj a b hab)
    · exact absurd hd hne
  · exact (disjoint_level hm (transientTowerExp_mem S a i) (transientTowerExp_mem S b j)
      hsame).mono (transientTowerLevel_subset_level hsub a i)
      (transientTowerLevel_subset_level hsub b j)

/-- **The characteristic arrows between the levels form a matrix unit family** (tex 1511–1512). -/
theorem isMatrixUnitFamily_transientTower (hm : ExponentBound T C m)
    (hsub : ∀ a, cell a ⊆ baseCell T C E m (S a))
    (hdisj : ∀ a b, a ≠ b → Disjoint (cell a) (cell b)) :
    IsMatrixUnitFamily
      (Pestov91.CrossedProduct.towerUnit (transientTowerCoeff T k cell S hcell)
        (transientTowerExp S)) :=
  Pestov91.CrossedProduct.isMatrixUnitFamily_towerUnit (transientTowerCoeff_cov T k hcell)
    (transientTowerCoeff_mul_self T k hcell)
    (fun a i j hij => transientTowerCoeff_mul_of_disjoint T k hcell
      (disjoint_transientTowerLevel hm hsub hdisj i j
        (Or.inr fun h => hij (transientTowerExp_injective S a h))))
    (fun a b hab i j => transientTowerCoeff_mul_of_disjoint T k hcell
      (disjoint_transientTowerLevel hm hsub hdisj i j (Or.inl hab)))

/-- **The arrows lie in the transient ideal** when `C` avoids the invariant set `Y`. -/
theorem transientTowerUnit_mem_transientIdeal {Y : Set X} (hY : T '' Y = Y)
    (hCY : Disjoint C Y) (hsub : ∀ a, cell a ⊆ baseCell T C E m (S a)) (a : Fin M)
    (i j : Fin (S a).card) :
    Pestov91.CrossedProduct.towerUnit (transientTowerCoeff T k cell S hcell)
        (transientTowerExp S) a i j ∈ transientIdeal T k hY := by
  rw [mem_transientIdeal_iff_forall_coeff]
  intro g y hy
  rw [Pestov91.CrossedProduct.towerUnit, SkewMonoidAlgebra.coeff_single, Finsupp.single_apply]
  split_ifs with hg
  · show LocallyConstant.charFn k (isClopen_transientTowerLevel T cell S hcell a i) y = 0
    exact charFn_apply_eq_zero k _ fun hyL => Set.disjoint_left.1 hCY
      (level_subset (transientTowerExp_mem S a i) (transientTowerLevel_subset_level hsub a i hyL))
      hy
  · rw [map_zero]
    rfl

/-- **Nonempty cells give nonzero diagonal arrows.** -/
theorem transientTowerUnit_self_ne_zero [Nontrivial k] (hne : ∀ a, (cell a).Nonempty)
    (a : Fin M) (i : Fin (S a).card) :
    Pestov91.CrossedProduct.towerUnit (transientTowerCoeff T k cell S hcell)
        (transientTowerExp S) a i i ≠ 0 := by
  refine Pestov91.CrossedProduct.towerUnit_self_ne_zero ?_
  intro h0
  obtain ⟨y, hy⟩ := hne a
  have hmem : (T ^ transientTowerExp S a i) y ∈ transientTowerLevel T cell S a i := ⟨y, hy, rfl⟩
  have h1 : LocallyConstant.charFn k (isClopen_transientTowerLevel T cell S hcell a i)
      ((T ^ transientTowerExp S a i) y) = 1 := by
    rw [LocallyConstant.coe_charFn, Set.indicator_of_mem hmem, Pi.one_apply]
  have h3 : LocallyConstant.charFn k (isClopen_transientTowerLevel T cell S hcell a i) = 0 := by
    have := congrArg (ClopenCoeff.of T k).symm h0
    rwa [transientTowerCoeff, RingEquiv.symm_apply_apply, map_zero] at this
  have h4 := LocallyConstant.congr_fun h3 ((T ^ transientTowerExp S a i) y)
  rw [h1] at h4
  exact one_ne_zero h4

end Tower
