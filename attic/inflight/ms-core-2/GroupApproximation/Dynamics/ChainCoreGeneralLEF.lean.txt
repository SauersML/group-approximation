/-- **Input: the cycle condition makes the crossed product LEF** (tex 1386: "A subshift satisfying the
cycle condition has an LEF crossed product").  Discharged by
`WordGraph.isLEFRing_clopenCrossedProduct_of_cycleCondition`. -/
def SubshiftCycleConditionLEFInput : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Finite A] (Z : Set (ℤ → A)),
    IsClosed Z → (∀ n : ℤ, MapsTo (shift n) Z Z) → ∀ (TZ : Z ≃ₜ Z),
      (∀ z : Z, (TZ z : ℤ → A) = shift 1 (z : ℤ → A)) → CycleCondition Z →

section Partition

variable {X : Type*} [MetricSpace X] {S : ℕ → DiscreteQuotient X}

/-- On a closed subset of a compact metrizable space, a locally constant function is constant on the
atoms of every sufficiently fine partition. -/
theorem exists_forall_proj_eq_imp_of_locallyConstant_subtype [CompactSpace X]
    (hS : IsRefiningClopenPartitions S) {Y : Set X} (hY : IsClosed Y) {β : Type*}
    (f : LocallyConstant Y β) :
    ∃ m, ∀ l, m ≤ l → ∀ x y : Y, (S l).proj x = (S l).proj y → f x = f y := by
  haveI : CompactSpace Y := isCompact_iff_compactSpace.1 hY.isCompact
  obtain ⟨δ, hδ, hball⟩ := lebesgue_number_lemma_of_metric (s := (univ : Set Y))
    (c := fun i : β ↦ {x : Y | f x = i}) isCompact_univ
    (fun i ↦ f.isLocallyConstant.isOpen_fiber i) fun x _ ↦ mem_iUnion.2 ⟨f x, rfl⟩
  obtain ⟨m, hm⟩ := hS.mesh_lt δ hδ
  refine ⟨m, fun l hml x y hxy ↦ ?_⟩
  have hxy' : dist x y < δ := hm x y (proj_eq_of_le (hS.antitone hml) hxy)
  obtain ⟨i, hi⟩ := hball x (mem_univ x)
  have hx : f x = i := hi (Metric.mem_ball_self hδ)
  have hy' : y ∈ Metric.ball x δ := by
    rw [Metric.mem_ball, dist_comm]
    exact hxy'
  have hy : f y = i := hi hy'
  exact hx.trans hy.symm

end Partition

section Factor

variable {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X) (S : ℕ → DiscreteQuotient X)

theorem itinerary_apply_zero (S' : DiscreteQuotient X) (x : X) :
    itinerary T S' x 0 = S'.proj x := by
  rw [itinerary_apply, zpow_zero]
  rfl

/-- **A locally constant function constant on the atoms of `S l` factors through `π_l`.** -/
theorem exists_comap_coreFactorMap_eq {β : Type*} [Zero β] (f : LocallyConstant (generalCore T S) β)
    (l : ℕ) (hf : ∀ x y : generalCore T S, (S l).proj x = (S l).proj y → f x = f y) :
    ∃ g : LocallyConstant (coreFactor T S l) β,
      LocallyConstant.comap (coreFactorMap T S l) g = f := by
  classical
  refine ⟨⟨fun z ↦ if hs : ∃ y : generalCore T S, (S l).proj y = (z : ℤ → S l) 0
      then f hs.choose else 0,
    (IsLocallyConstant.of_discrete fun s : S l ↦
      if hs : ∃ y : generalCore T S, (S l).proj y = s then f hs.choose else 0).comp_continuous
      ((continuous_apply 0).comp continuous_subtype_val)⟩, ?_⟩
  refine LocallyConstant.ext fun y ↦ ?_
  have hs : ∃ y' : generalCore T S, (S l).proj y' = itinerary T (S l) y 0 :=
    ⟨y, (itinerary_apply_zero T (S l) y).symm⟩
  change (if hs : ∃ y' : generalCore T S, (S l).proj y' = itinerary T (S l) y 0
    then f hs.choose else 0) = f y
  rw [dif_pos hs]
  exact hf _ _ (hs.choose_spec.trans (itinerary_apply_zero T (S l) y))

end Factor

section Cover

variable {X : Type} [MetricSpace X] [CompactSpace X] (T : X ≃ₜ X) {S : ℕ → DiscreteQuotient X}

/-- **Every element of `R_{Y_*}` is pulled back from some `π_m(Y_*)`**: its finitely many Laurent
coefficients are constant on the atoms of one fine partition. -/
theorem exists_mem_range_comap_coreFactorMap (k : Type) [Ring k] (hS : IsRefiningClopenPartitions S)
    (a : ClopenCrossedProduct (ClopenCrossedProduct.restrictHomeo T (image_generalCore T S)) k) :
    ∃ m, a ∈
      (ClopenCrossedProduct.comap k (coreFactorMap T S m) (coreFactorMap_semiconj T S m)).range := by
  obtain ⟨J, f, rfl⟩ := ClopenCrossedProduct.exists_sum_coeff_mul_unit_zpow _ k a
  choose m hm using fun j ↦
    exists_forall_proj_eq_imp_of_locallyConstant_subtype hS (isClosed_generalCore T S) (f j)
  have hfac : ∀ j ∈ J, ∃ g : LocallyConstant (coreFactor T S (J.sup m)) k,
      LocallyConstant.comap (coreFactorMap T S (J.sup m)) g = f j := fun j hj ↦
    exists_comap_coreFactorMap_eq T S (f j) (J.sup m) (hm j (J.sup m) (Finset.le_sup hj))
  choose! g hg using hfac
  refine ⟨J.sup m, RingHom.mem_range.2
    ⟨∑ j ∈ J, ClopenCrossedProduct.coeff (coreFactorShift T S (J.sup m)) k (g j) *
      ((ClopenCrossedProduct.unit (coreFactorShift T S (J.sup m)) k ^ j :
        (ClopenCrossedProduct (coreFactorShift T S (J.sup m)) k)ˣ) :
        ClopenCrossedProduct (coreFactorShift T S (J.sup m)) k), ?_⟩⟩
  rw [comap_sum_coeff_mul_unit_zpow]
  exact Finset.sum_congr rfl fun j hj ↦ by rw [hg j hj]

/-- **`R_{Y_*}` is LEF** (tex 1444–1445): an increasing union of the pulled-back rings
`R_{π_m(Y_*)}`, each LEF because `π_m(Y_*)` satisfies the cycle condition (tex 1439–1441). -/
theorem isLEFRing_clopenCrossedProduct_generalCore (hS : IsRefiningClopenPartitions S)
    (hLEF : SubshiftCycleConditionLEFInput)
    (hcore : ∀ l, CycleCondition (itineraryCore T (S l))) (k : Type) [Ring k] [Finite k] :
    Pestov91.IsLEFRing
      (ClopenCrossedProduct (ClopenCrossedProduct.restrictHomeo T (image_generalCore T S)) k) :=
  Pestov91.isLEFRing_of_monotone_ringHom_range
    (fun m ↦ ClopenCrossedProduct (coreFactorShift T S m) k)
    (fun m ↦ ClopenCrossedProduct.comap k (coreFactorMap T S m) (coreFactorMap_semiconj T S m))
    (fun m ↦ comap_injective k (coreFactorMap T S m) (coreFactorMap_semiconj T S m)
      (coreFactorMap_surjective T S m))
    (fun _ _ h ↦ range_comap_coreFactorMap_mono (T := T) (k := k) hS.antitone h)
    (exists_mem_range_comap_coreFactorMap T k hS)
    (fun m ↦ hLEF (S m) (coreFactor T S m) (isClosed_coreFactor T S m)
      (mapsTo_shift_coreFactor T S m) (coreFactorShift T S m) (coreFactorShift_apply T S m)
      (cycleCondition_itinerary_image_generalCore T hS.antitone hcore m) k)

end Cover
