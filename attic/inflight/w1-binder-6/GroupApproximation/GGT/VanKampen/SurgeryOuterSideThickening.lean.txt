/-!
# Thickening the edges between region sides and the exterior

Osin, arXiv:math/0411039v3, §9, proof of Lemma 9.7(b).  The pocket walk of two exterior regions
`x`, `y` of a cell has no repeated dart only if no dart of the left side of `y` or the right side
of `x` has the exterior across it (the field `side_outer` of `PocketWalk.CopyClean`,
`Estimating/OsinPocketCopyWalk`).  A side can run along the boundary of the diagram.

Let `w_j` be a dart of the exterior whose reverse is a side dart of a selected region.  The
doubling of `FaceEdgeDoubling` along `w_j` puts a G-cell digon between the side and the exterior.
The owning region does not use `alpha w_j` in its target arc, and every other region avoids its
face, so the whole family transports (`OuterDoublingTransport`).  The sides become the images of
the old sides, and the new exterior dart has its reverse on the digon, so the number of outer
side darts drops by one (`outerSideCount_lt`).  Outer spurs, outer cell darts and relator words
of value one do not appear.

* `IsOuterSideDart`: a dart of the exterior whose reverse is a side dart of a region of the family.
* `exists_sideFree`: induction on the number of outer side darts.
* `outerSideThickening`: `OuterSideThickeningStatement`, after `outerCellThickening`.
* `side_outer_of_noOuterSideDart`: the edge condition.

## Manuscript status

Infrastructure for `thm:hull` (tex 2121, "Hull's small cancellation theorem", through Osin's
Lemma 9.7(b)); certifies no printed sentence on its own.
-/

section Basics

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {eps : ℕ}

/-- An **outer side dart** of a family: a dart on the exterior whose reverse is a left or right
side dart of a region of the family. -/
def IsOuterSideDart (Delta : DiscDiagram.{u, w, v} W)
    (family : Finset (RegionCandidate D eps Delta)) (x : Delta.toCombMap.Dart) : Prop :=
  Delta.toCombMap.faceOf x = Delta.outerFace ∧
    ∃ a ∈ family, Delta.toCombMap.alpha x ∈ a.2.rightSide ++ a.2.leftSide

/-- The number of outer side darts. -/
noncomputable def outerSideCount (Delta : DiscDiagram.{u, w, v} W)
    (family : Finset (RegionCandidate D eps Delta)) : ℕ :=
  (Finset.univ.filter (IsOuterSideDart Delta family)).card

/-- Without outer side darts, no side dart of a region has the exterior across it
(`CopyClean.side_outer`). -/
theorem side_outer_of_noOuterSideDart {Delta : DiscDiagram.{u, w, v} W}
    {family : Finset (RegionCandidate D eps Delta)} (hno : ∀ x, ¬ IsOuterSideDart Delta family x)
    {x y : RegionCandidate D eps Delta} (hx : x ∈ family) (hy : y ∈ family) :
    ∀ d ∈ y.2.leftSide ++ x.2.rightSide,
      Delta.toCombMap.faceOf (Delta.toCombMap.alpha d) ≠ Delta.outerFace := by
  intro d hd hout
  apply hno (Delta.toCombMap.alpha d)
  refine ⟨hout, ?_⟩
  rw [Delta.toCombMap.alpha_involutive d]
  rcases List.mem_append.mp hd with hl | hr
  · exact ⟨y, hy, List.mem_append_right _ hl⟩
  · exact ⟨x, hx, List.mem_append_left _ hr⟩

/-- A side dart of a region lies on a face of the region. -/
theorem faceOf_mem_of_side {Delta : DiscDiagram.{u, w, v} W} {s : Finset Delta.toCombMap.Face}
    (H : ContiguityGeometry D eps Delta s) {d : Delta.toCombMap.Dart}
    (hd : d ∈ H.rightSide ++ H.leftSide) : Delta.toCombMap.faceOf d ∈ s := by
  have hcycle : d ∈ H.boundary.cycle := by
    rw [H.boundary_decomposition]
    rcases List.mem_append.mp hd with hr | hl
    · exact List.mem_append_left _ (List.mem_append_left _ (List.mem_append_right _ hr))
    · exact List.mem_append_right _ hl
  exact ((H.boundary.cycle_mem_iff d).mp hcycle).1

end Basics

section Thickening

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
variable (Delta : DiscDiagram.{u, w, v} W)
  (j : Fin (Delta.faceBoundary Delta.outerFace).darts.length)
  (hlen : 1 < (Delta.faceBoundary Delta.outerFace).darts.length)

/-- **At an outer side dart, no region uses the reverse of the doubled dart in its target arc.**
The owning region has it as a side dart; every other region avoids its face. -/
theorem not_mem_targetArc_of_outerSide {D : RelGenSet G Lambda} {eps : ℕ}
    (family : Finset (RegionCandidate D eps Delta))
    (hcompat : EstimatingSelection.PairwiseCompatible RegionCandidate.Compatible family)
    (hside : IsOuterSideDart Delta family (FaceEdgeDoubling.dart Delta Delta.outerFace j)) :
    ∀ a ∈ family,
      Delta.toCombMap.alpha (FaceEdgeDoubling.dart Delta Delta.outerFace j) ∉ a.2.targetArc.darts := by
  obtain ⟨-, b, hb, hbside⟩ := hside
  intro a ha
  by_cases hab : a = b
  · subst hab
    exact OuterDoublingTransport.alpha_dart_not_mem_targetArc_of_side Delta j a.2 hbside
  · have hface : Delta.toCombMap.faceOf
        (Delta.toCombMap.alpha (FaceEdgeDoubling.dart Delta Delta.outerFace j)) ∉ a.1 := by
      intro hmem
      exact Finset.disjoint_left.mp (hcompat a ha b hb hab) hmem (faceOf_mem_of_side b.2 hbside)
    exact OuterSpurThickening.alpha_dart_not_mem_targetArc Delta j a.2 hface

/-- **An outer side dart after the doubling is an old outer side dart other than `w_j`.** -/
theorem exists_of_isOuterSideDart {D : RelGenSet G Lambda} {eps : ℕ}
    {family : Finset (RegionCandidate D eps Delta)}
    {htgt : ∀ a ∈ family,
      Delta.toCombMap.alpha (FaceEdgeDoubling.dart Delta Delta.outerFace j) ∉ a.2.targetArc.darts}
    {x : (OuterSpurThickening.diagram Delta j hlen).toCombMap.Dart}
    (hx : IsOuterSideDart (OuterSpurThickening.diagram Delta j hlen)
      (OuterDoublingTransport.regionFamily Delta j hlen family htgt) x) :
    ∃ d, embed Delta.toCombMap d = x ∧ IsOuterSideDart Delta family d ∧
      d ≠ FaceEdgeDoubling.dart Delta Delta.outerFace j := by
  obtain ⟨d, hd, hdx⟩ := (OuterSpurThickening.faceOf_eq_outer_iff Delta j hlen x).mp hx.1
  obtain ⟨a, ha, hside⟩ := hx.2
  obtain ⟨b, rfl⟩ := OuterDoublingTransport.mem_regionFamily ha
  rw [OuterDoublingTransport.regionCandidate_sides] at hside
  obtain ⟨e, he, hex⟩ := List.mem_map.mp hside
  by_cases hdd : d = FaceEdgeDoubling.dart Delta Delta.outerFace j
  · rw [FaceEdgeDoubling.dartImage, if_pos hdd] at hdx
    subst hdx
    change some (some e) = none at hex
    exact Option.noConfusion hex
  · rw [FaceEdgeDoubling.dartImage, if_neg hdd] at hdx
    subst hdx
    change embed Delta.toCombMap e = embed Delta.toCombMap (Delta.toCombMap.alpha d) at hex
    have hea : e = Delta.toCombMap.alpha d := embed_injective Delta.toCombMap hex
    subst hea
    exact ⟨d, rfl, ⟨hd, b.val, b.property, he⟩, hdd⟩

/-- **Thickening an outer side dart removes it and creates none.** -/
theorem outerSideCount_lt {D : RelGenSet G Lambda} {eps : ℕ}
    (family : Finset (RegionCandidate D eps Delta))
    (htgt : ∀ a ∈ family,
      Delta.toCombMap.alpha (FaceEdgeDoubling.dart Delta Delta.outerFace j) ∉ a.2.targetArc.darts)
    (hside : IsOuterSideDart Delta family (FaceEdgeDoubling.dart Delta Delta.outerFace j)) :
    outerSideCount (OuterSpurThickening.diagram Delta j hlen)
        (OuterDoublingTransport.regionFamily Delta j hlen family htgt) <
      outerSideCount Delta family := by
  have hmem : FaceEdgeDoubling.dart Delta Delta.outerFace j ∈
      Finset.univ.filter (IsOuterSideDart Delta family) :=
    Finset.mem_filter.mpr ⟨Finset.mem_univ _, hside⟩
  unfold outerSideCount
  refine lt_of_le_of_lt ?_ (Finset.card_erase_lt_of_mem hmem)
  refine Finset.card_le_card_of_injOn
    (fun x : (OuterSpurThickening.diagram Delta j hlen).toCombMap.Dart =>
      (Option.bind x id).getD (FaceEdgeDoubling.dart Delta Delta.outerFace j)) ?_ ?_
  · intro x hx
    obtain ⟨d, rfl, hd, hne⟩ :=
      exists_of_isOuterSideDart Delta j hlen (Finset.mem_filter.mp (Finset.mem_coe.mp hx)).2
    exact Finset.mem_coe.mpr
      (Finset.mem_erase.mpr ⟨hne, Finset.mem_filter.mpr ⟨Finset.mem_univ _, hd⟩⟩)
  · intro x hx y hy hxy
    obtain ⟨d, rfl, -, -⟩ :=
      exists_of_isOuterSideDart Delta j hlen (Finset.mem_filter.mp (Finset.mem_coe.mp hx)).2
    obtain ⟨e, rfl, -, -⟩ :=
      exists_of_isOuterSideDart Delta j hlen (Finset.mem_filter.mp (Finset.mem_coe.mp hy)).2
    exact congrArg (embed Delta.toCombMap) hxy

/-- An outer doubling creates no outer cell dart. -/
theorem noOuterCellDart (hfree : ∀ x, ¬ OuterCellThickening.IsOuterCellDart Delta x) :
    ∀ x, ¬ OuterCellThickening.IsOuterCellDart (OuterSpurThickening.diagram Delta j hlen) x := by
  intro x hx
  obtain ⟨d, -, hd, -⟩ := OuterCellThickening.exists_of_isOuterCellDart Delta j hlen hx
  exact hfree d hd

end Thickening

section Family

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {lambda c : ℝ} {eps : ℕ} {Delta : DiscDiagram.{u, w, v} W}
  {cuts : SectionCuts D lambda c Delta.boundaryWord}

/-- **A distinguished section family across an outer doubling** whose target arcs avoid the
reverse of the doubled dart. -/
noncomputable def sectionFamilyOfTransport
    (S : GloballyDistinguishedSectionFamily D lambda c eps Delta cuts)
    (j : Fin (S.diagram.faceBoundary S.diagram.outerFace).darts.length)
    (hlen : 1 < (S.diagram.faceBoundary S.diagram.outerFace).darts.length)
    (htgt : ∀ a ∈ S.family, S.diagram.toCombMap.alpha
      (FaceEdgeDoubling.dart S.diagram S.diagram.outerFace j) ∉ a.2.targetArc.darts) :
    GloballyDistinguishedSectionFamily D lambda c eps Delta cuts where
  diagram := OuterSpurThickening.diagram S.diagram j hlen
  equiv := S.equiv.trans (OuterSpurThickening.oEquivalent S.diagram j hlen)
  reduced := OuterSpurThickening.reduced S.diagram j hlen S.reduced
  family := OuterDoublingTransport.regionFamily S.diagram j hlen S.family htgt
  pairwise :=
    OuterDoublingTransport.regionFamily_pairwise S.diagram j hlen S.family htgt S.pairwise
  respects := by
    intro a ha
    obtain ⟨b, hb, hab⟩ :=
      OuterDoublingTransport.regionFamily_profile S.diagram j hlen S.family htgt ha
    exact RegionCandidate.respectsSections_of_sameTargetProfile cuts hab
      (OuterDoublingTransport.regionFamily_noLoop S.diagram j hlen S.family htgt
        (fun x hx => (S.respects x hx).1) ha)
      (S.respects b hb)
  nondegenerate := by
    intro a ha
    obtain ⟨b, hb, hab⟩ :=
      OuterDoublingTransport.regionFamily_profile S.diagram j hlen S.family htgt ha
    obtain ⟨hs, ht⟩ := S.nondegenerate b hb
    exact ⟨lt_of_lt_of_eq hs hab.2.2.2.symm, lt_of_lt_of_eq ht hab.2.2.1.symm⟩
  label_admissible :=
    OuterSpurThickening.label_admissible S.diagram j hlen (symmetricLabelAlphabet D)
      (symmetricLabelAlphabet.symmetric D) S.label_admissible
  weight_maximal other hother :=
    (S.weight_maximal other hother).trans_eq
      (OuterDoublingTransport.regionFamily_weight S.diagram j hlen S.family htgt).symm
  card_minimal other hother hweight :=
    (OuterDoublingTransport.regionFamily_card S.diagram j hlen S.family htgt).trans_le
      (S.card_minimal other hother (hweight.trans
        (OuterDoublingTransport.regionFamily_weight S.diagram j hlen S.family htgt)))

/-- **Induction on the number of outer side darts**, keeping the absence of outer spurs, outer
cell darts and relator words of value one. -/
theorem exists_sideFree (S : GloballyDistinguishedSectionFamily D lambda c eps Delta cuts)
    (hlen0 : 1 < Delta.boundaryWord.length)
    (hvalue : ∀ C ∈ S.diagram.relatorCells, RelLetter.listVal C.word ≠ 1)
    (hfree : ∀ x, ¬ OuterSpurThickening.IsOuterSpur S.diagram x)
    (hcell : ∀ x, ¬ OuterCellThickening.IsOuterCellDart S.diagram x) :
    ∃ (S' : GloballyDistinguishedSectionFamily D lambda c eps Delta cuts)
      (e : S.family ≃ S'.family),
      Nonempty (OEquivalentDiscDiagram S.diagram S'.diagram) ∧
        (∀ x, ¬ OuterSpurThickening.IsOuterSpur S'.diagram x) ∧
          (∀ x, ¬ OuterCellThickening.IsOuterCellDart S'.diagram x) ∧
            (∀ x, ¬ IsOuterSideDart S'.diagram S'.family x) ∧
              (∀ C ∈ S'.diagram.relatorCells, RelLetter.listVal C.word ≠ 1) ∧
                S'.toRealizedSectionFamily.weight = S.toRealizedSectionFamily.weight ∧
                  ∀ a, RegionCandidate.SameTargetProfile (e a).1 a.1 ∧
                    (e a).1.2.source.val = a.1.2.source.val := by
  revert hvalue hfree hcell
  generalize hn : outerSideCount S.diagram S.family = n
  induction n using Nat.strong_induction_on generalizing S with
  | _ n ih =>
    intro hvalue hfree hcell
    by_cases hside : ∃ x, IsOuterSideDart S.diagram S.family x
    · obtain ⟨x, hx⟩ := hside
      have hbw : S.diagram.boundaryWord = Delta.boundaryWord := S.equiv.boundaryWord_eq
      have hlen : 1 < (S.diagram.faceBoundary S.diagram.outerFace).darts.length :=
        OuterCellThickening.one_lt_outer_length (by rw [hbw]; exact hlen0)
      obtain ⟨j, hj⟩ := List.get_of_mem
        (((S.diagram.faceBoundary S.diagram.outerFace).mem_iff x).mpr hx.1)
      have hside2 : IsOuterSideDart S.diagram S.family
          (FaceEdgeDoubling.dart S.diagram S.diagram.outerFace j) := by
        rw [FaceEdgeDoubling.dart_eq_get, hj]
        exact hx
      have htgt := not_mem_targetArc_of_outerSide S.diagram j S.family S.pairwise hside2
      obtain ⟨S', e, ⟨equiv'⟩, hspur', hcell', hside', hvalue', hweight, hprofile⟩ :=
        ih _ (lt_of_lt_of_eq (outerSideCount_lt S.diagram j hlen S.family htgt hside2) hn)
          (sectionFamilyOfTransport S j hlen htgt) rfl
          (OuterCellThickening.relatorValue_ne_one S.diagram j hlen hvalue)
          (OuterCellThickening.noOuterSpur S.diagram j hlen hfree)
          (noOuterCellDart S.diagram j hlen hcell)
      refine ⟨S', (OuterDoublingTransport.regionFamilyEquiv S.diagram j hlen S.family htgt).trans e,
        ⟨(OuterSpurThickening.oEquivalent S.diagram j hlen).trans equiv'⟩, hspur', hcell', hside',
        hvalue',
        hweight.trans (OuterDoublingTransport.regionFamily_weight S.diagram j hlen S.family htgt),
        fun a => ?_⟩
      obtain ⟨hprof, hsource⟩ :=
        hprofile (OuterDoublingTransport.regionFamilyEquiv S.diagram j hlen S.family htgt a)
      exact And.intro
        (hprof.trans
          (OuterDoublingTransport.regionFamilyEquiv_profile S.diagram j hlen S.family htgt a))
        (hsource.trans
          (OuterDoublingTransport.regionFamilyEquiv_source S.diagram j hlen S.family htgt a))
    · exact ⟨S, Equiv.refl _, ⟨OEquivalentDiscDiagram.refl _⟩, hfree, hcell,
        fun x hx => hside ⟨x, hx⟩, hvalue, rfl,
        fun _ => And.intro (And.intro Iff.rfl (And.intro rfl (And.intro rfl rfl))) rfl⟩

end Family

/-- **Outer side thickening.**  A distinguished section family over a boundary word of length at
least two, with no relator word of value one, has a distinguished section family over the same
cuts whose diagram is O-equivalent to the old one, with no outer spur, no edge between a relator
cell and the exterior, and no edge between a side of a selected region and the exterior.  No
relator word of the new diagram has value one.  Its regions correspond bijectively to the old
ones, with the same total weight, and every region keeps its target profile and the index of its
source cell. -/
def OuterSideThickeningStatement : Prop :=
  ∀ {G : Type u} [Group G] {Lambda : Type w} (D : RelGenSet G Lambda) (lambda c : ℝ) (eps : ℕ)
    (W : Set (List (RelLetter G Lambda))) (Delta : DiscDiagram.{u, w, v} W)
    (cuts : SectionCuts D lambda c Delta.boundaryWord)
    (S : GloballyDistinguishedSectionFamily D lambda c eps Delta cuts),
    1 < Delta.boundaryWord.length →
      (∀ C ∈ S.diagram.relatorCells, RelLetter.listVal C.word ≠ 1) →
        ∃ (S' : GloballyDistinguishedSectionFamily D lambda c eps Delta cuts)
          (e : S.family ≃ S'.family),
          Nonempty (OEquivalentDiscDiagram S.diagram S'.diagram) ∧
            (∀ x, ¬ OuterSpurThickening.IsOuterSpur S'.diagram x) ∧
              (∀ x, ¬ OuterCellThickening.IsOuterCellDart S'.diagram x) ∧
                (∀ x, ¬ IsOuterSideDart S'.diagram S'.family x) ∧
                  (∀ C ∈ S'.diagram.relatorCells, RelLetter.listVal C.word ≠ 1) ∧
                    S'.toRealizedSectionFamily.weight = S.toRealizedSectionFamily.weight ∧
                      ∀ a, RegionCandidate.SameTargetProfile (e a).1 a.1 ∧
                        (e a).1.2.source.val = a.1.2.source.val

theorem outerSideThickening : OuterSideThickeningStatement.{u, w, v} := by
  unfold OuterSideThickeningStatement
  intro _ _ _ D lambda c eps W Delta cuts S hlen0 hvalue
  obtain ⟨S₁, e₁, ⟨equiv₁⟩, hfree₁, hcell₁, hvalue₁, hweight₁, hprofile₁⟩ :=
    OuterCellThickening.outerCellThickening D lambda c eps W Delta cuts S hlen0 hvalue
  obtain ⟨S₂, e₂, ⟨equiv₂⟩, hfree₂, hcell₂, hside₂, hvalue₂, hweight₂, hprofile₂⟩ :=
    exists_sideFree S₁ hlen0 hvalue₁ hfree₁ hcell₁
  refine ⟨S₂, e₁.trans e₂, ⟨equiv₁.trans equiv₂⟩, hfree₂, hcell₂, hside₂, hvalue₂,
    hweight₂.trans hweight₁, fun a => ?_⟩
  obtain ⟨hprof₂, hsource₂⟩ := hprofile₂ (e₁ a)
  obtain ⟨hprof₁, hsource₁⟩ := hprofile₁ a
  exact And.intro (hprof₂.trans hprof₁) (hsource₂.trans hsource₁)

end GroupApproximation.GGT.VanKampen.OuterSideThickening
