/-- **tex 1594–1596**, "Normal root commutators and root addition kill all roots in the ideal generated by $d$.  Extending
by identity gives the same conclusion in rank $n\ge3$": for every compact metrizable zero-dimensional `X`, homeomorphism
`T`, finite field `k` and `n ≥ 3`, every homomorphism from `GL_n(R_X)` to an MF group kills `EL_n(R_X, I)`, where `I` is
generated by the defects. -/
def PrintedBilateralThreeRootAddition : Prop :=
  ∀ (X : Type) [TopologicalSpace X] [CompactSpace X] [TopologicalSpace.MetrizableSpace X]
    [TotallyDisconnectedSpace X] (T : X ≃ₜ X) (k : Type) [Field k] [Finite k] (n : ℕ), 3 ≤ n →
      MFTargetsKillRelativeElementary T k n (defectIdeal T k)

/-- The cell kills `e₂₁(d)` in `GL₃(R_X)` for every defect `d`, and root addition with extension by identity carries the
kill to `EL_n(R_X, ⟨defects⟩)`. -/
theorem printedBilateralThreeRootAddition_closed : PrintedBilateralThreeRootAddition := by
  intro X _ _ _ _ T k _ _ n hn
  haveI : Countable (ClopenCrossedProduct T k) := ClopenCrossedProduct.countable T k
  intro M _ hM f
  exact relativeElementary_killed_of_rootThree_killed
    (rootThree_killed_of_cell bilateralThreeCellStatement T k) hn hM f

/-- **tex 1599–1600**, "If $R_X$ is not directly finite, some defect is nonzero, so $\GL_n(R_X)$ is not MF": for every
nonempty compact metrizable zero-dimensional `X`, homeomorphism `T`, finite field `k` and `n ≥ 3`, if `R_X` is not
directly finite then some compressed-clopen defect is nonzero and `GL_n(R_X)` is not MF. -/
def PrintedNotDirectlyFiniteGLNotMF : Prop :=
  ∀ (X : Type) [TopologicalSpace X] [CompactSpace X] [TopologicalSpace.MetrizableSpace X]
    [TotallyDisconnectedSpace X] [Nonempty X] (T : X ≃ₜ X) (k : Type) [Field k] [Finite k] (n : ℕ), 3 ≤ n →
      ¬ IsDedekindFiniteMonoid (ClopenCrossedProduct T k) →
        (∃ d ∈ clopenDefectSet T k, d ≠ 0) ∧
          ¬ IsOperatorMF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ

/-- A nonzero defect `d` gives the nonzero root `e_{ij}(d)` of `GL_n(R_X)`; every MF target kills it (the cell in `GL₃`,
extended by identity), the identity homomorphism included, so `GL_n(R_X)` is not MF. -/
theorem printedNotDirectlyFiniteGLNotMF_closed : PrintedNotDirectlyFiniteGLNotMF := by
  intro X _ _ _ _ _ T k _ _ n hn hdf
  haveI : Countable (ClopenCrossedProduct T k) := ClopenCrossedProduct.countable T k
  obtain ⟨d, hd, hne⟩ :=
    clopenCrossedProductNonzeroDefectStatement_of_coreModels coreModelsLEFStatement X T k hdf
  refine ⟨⟨d, hd, hne⟩, not_isOperatorMF_of_root_killed ((threeIdx_injective hn).ne idxOneNeZero) hne ?_⟩
  intro M _ hM f
  have h := rootThree_killed_of_cell bilateralThreeCellStatement T k d hd M hM
    (f.comp (glThreeBlockEmbedding (R := ClopenCrossedProduct T k) hn))
  rwa [MonoidHom.comp_apply, glThreeBlockEmbedding_elementaryUnit] at h

/-- **tex 1643–1644**, "Proposition~\ref{prop:bilateral-three} kills $\EL_n(R_X,I)$ in every general linear MF image for
$n\ge3$", closed: over `F₂`, `K_n ≤ Rad_MF(GL_n(R_X))` for `n ≥ 3`, from prop:bilateral-three at the closed LEF core and
`K_n ≤ EL_n(R_X, I)`. -/
theorem coreMFRadicalKillGLStatement_holds : CoreMFRadicalGLKill.CoreMFRadicalKillGLStatement :=
