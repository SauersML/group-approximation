/-!
# The exact MF radical

`non_mf_groups_exist.tex`, `thm:core-mf-radical`, tex 1609–1656:

> Take $k=\F_2$ and write $K_n=\ker(\GL_n(R_X)\to\GL_n(R_Y))$.  For $n\ge2$, $K_n=\EL_n(R_X,I)$ is
> locally finite.  Moreover, $\Rad_{\mathrm{MF}}(\GL_n(R_X))=K_n$ ($n\ge3$) and
> $\Rad_{\mathrm{MF}}(\EL_n(R_X))=K_n$ ($n\ge4$).  The elementary restriction is surjective with LEF
> quotient $\EL_n(R_Y)$.  The general linear quotient is the LEF restriction image.

* `coreKernelLeRelativeElementaryStatement_holds`, `coreKernelLocallyFiniteStatement_holds`: the
  kernel identification and local finiteness (tex 1626–1641), from chain-radical's congruence kernel
  lemmas over ct-involution's closed local matriciality and ct-bilateral-mf's closed
  `coreTransientIdealLeDefectIdealStatement_holds`.
* `relativeElementary_defectIdeal_le_mfHomKernel`: for `n ≥ 4`, prop:torsion-defect-ring kills
  `EL_n(R_X, R d R)` for every defect `d` (additive order two, `R_X` countable), so every defect lies
  in the level ideal of the MF radical, and the level ideal absorbs `EL_n(R_X, I)` (tex 1643–1649).
* `coreMFRadicalKillELStatement_holds`: `EL_n(R_X) ∩ K_n ≤ Rad_MF(EL_n(R_X))` for `n ≥ 4`, closed.
* `coreMFRadicalKillStatement_of_coreModels`: the three kill clauses; the general linear clause
  consumes prop:bilateral-three through `printedBilateralThree_of_coreModels`.
* Closed endpoint `printedCoreKernelAndELKill`: the kernel clauses, the elementary kill and
  surjectivity of the elementary restriction.
* `printedCoreMFRadical_of_coreModels : CoreModelsLEFStatement → PrintedCoreMFRadical`: the printed
  theorem.  "$R_Y$ is LEF" (lem:chain-core-models) is the only premise: it makes `GL_n(R_Y)` MF, so
  the restriction bounds both radicals from above, and it makes the quotients LEF.
-/

/-! ### Kernel identification and local finiteness -/

/-- **`K_n ≤ EL_n(R_X, I)` over `F₂` for `n ≥ 2`** (tex 1626–1639), closed. -/
theorem coreKernelLeRelativeElementaryStatement_holds : CoreKernelLeRelativeElementaryStatement :=
  coreKernelLeRelativeElementary_of_pieces transientIdealLocallyMatricialFTwoStatement_holds
    coreTransientIdealLeDefectIdealStatement_holds

/-- **`K_n` is locally finite over `F₂` for `n ≥ 2`** (tex 1640–1641), closed. -/
theorem coreKernelLocallyFiniteStatement_holds : CoreKernelLocallyFiniteStatement := by
  intro X _ _ _ _ T n _
  letI := zmodTwoAlgebra T
  exact ChainRadical.isLocallyFiniteGroup_ker_elementaryMatrixUnitMap (ι := Fin n)
    (coreRestrict T (ZMod 2))
    (isLocallyMatricialIn_coreTransientIdeal transientIdealLocallyMatricialFTwoStatement_holds T)

/-! ### The elementary kill -/

/-- **prop:torsion-defect-ring on the defects** (tex 1644–1649): for `n ≥ 4`, every homomorphism
from `EL_n(R_X)` to an MF group kills `EL_n(R_X, I)`, where `I` is generated by the defects. -/
theorem relativeElementary_defectIdeal_le_mfHomKernel {X : Type} [TopologicalSpace X]
    [CompactSpace X] [SecondCountableTopology X] (T : X ≃ₜ X) {n : ℕ} (hn : 4 ≤ n) :
    relativeElementary (Fin n) (defectIdeal T (ZMod 2)) ≤
      mfHomKernel ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))) := by
  haveI : Countable (ClopenCrossedProduct T (ZMod 2)) := ClopenCrossedProduct.countable T (ZMod 2)
  have hcard : 3 ≤ Fintype.card (Fin n) := by
    rw [Fintype.card_fin]
    omega
  refine (relativeElementary_mono ?_).trans (relativeElementary_normalLevel_le hcard _)
  rw [defectIdeal, TwoSidedIdeal.span_le]
  rintro _ ⟨P, hP, hTP, rfl⟩
  have hst : 1 - defectS T (ZMod 2) hP * defectT T (ZMod 2) hP = defectD T (ZMod 2) hP := by
    rw [defectS_mul_defectT T (ZMod 2) hP hTP, sub_sub_cancel]
  have hme : 2 • (1 - defectS T (ZMod 2) hP * defectT T (ZMod 2) hP) = 0 := by
    rw [hst]
    exact two_nsmul_defectD T (ZMod 2) (by decide) hP
  have hkill :=
    Manuscript.OneSidedMFRadical.TorsionComplementaryIdempotents.relativeElementary_le_mfHomKernel
      (defectS T (ZMod 2) hP) (defectT T (ZMod 2) hP) (defectT_mul_defectS T (ZMod 2) hP hTP)
      (m := 2) two_pos hme hn
  refine (mem_normalLevel_iff hcard _).2 fun i j hij ↦
    hkill (elGen_mem_relativeElementary i j hij ?_)
  rw [hst]
  exact TwoSidedIdeal.subset_span (Set.mem_singleton _)

/-- The elementary clause of the kill from the kernel identification. -/
theorem coreMFRadicalKillEL_of_relativeElementary (hid : CoreKernelLeRelativeElementaryStatement) :
    CoreMFRadicalKillELStatement := by
  intro X _ _ _ _ T n hn x hx
  have hgl : (x : (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T (ZMod 2)))ˣ) ∈
      glCoreKernel T (ZMod 2) n := MonoidHom.mem_ker.2 (MonoidHom.mem_ker.1 hx)
  obtain ⟨g, hg, hgx⟩ := Subgroup.mem_map.1 (hid X T n (by omega) hgl)
  have hxg : x = g := Subtype.ext hgx.symm
  rw [hxg]
  exact relativeElementary_defectIdeal_le_mfHomKernel T hn hg

/-- **`EL_n(R_X) ∩ K_n ≤ Rad_MF(EL_n(R_X))` over `F₂` for `n ≥ 4`** (tex 1644–1649), closed. -/
theorem coreMFRadicalKillELStatement_holds : CoreMFRadicalKillELStatement :=
  coreMFRadicalKillEL_of_relativeElementary coreKernelLeRelativeElementaryStatement_holds

/-- The general linear clause, over "`R_Y` is LEF" (prop:bilateral-three). -/
theorem coreMFRadicalKillGL_of_coreModels (hlef : CoreModelsLEFStatement) :
    CoreMFRadicalKillGLStatement :=
  coreMFRadicalKillGL_of_pieces (printedBilateralThree_of_coreModels hlef)
    coreKernelLeRelativeElementaryStatement_holds

/-- **`DynamicRankBudget.CoreMFRadicalKillStatement`**, over "`R_Y` is LEF". -/
theorem coreMFRadicalKillStatement_of_coreModels (hlef : CoreModelsLEFStatement) :
    CoreMFRadicalKillStatement :=
  coreMFRadicalKillStatement_of_halves (coreMFRadicalKillGL_of_coreModels hlef)
    coreMFRadicalKillELStatement_holds coreKernelLocallyFiniteStatement_holds

/-! ### The printed theorem -/

section Assembly

variable {X : Type} [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] (T : X ≃ₜ X)

/-- **The elementary restriction is surjective** (tex 1651–1652, "Lifting elementary coefficients
makes the elementary restriction surjective"). -/
theorem elementaryGroupMap_coreRestrict_surjective (n : ℕ) :
    Function.Surjective (elementaryGroupMap (ι := Fin n) (coreRestrict T (ZMod 2))) := by
  refine elementaryGroupMap_surjective_of_surjective _ ?_
  have hYc : IsClosed (Dynamics.chainRecurrentSet ⇑T) :=
    Dynamics.isClosed_chainRecurrentSet T.continuous
  show Function.Surjective (restrict T (ZMod 2) (image_core T))
  unfold restrict comap
  refine Pestov91.CrossedProduct.coeffMap_surjective _ _ fun f ↦ ?_
  obtain ⟨g, hg⟩ := LocallyConstant.exists_comap_val_eq_of_isClosed hYc
    ((ClopenCoeff.of (restrictHomeo T (image_core T)) (ZMod 2)).symm f)
  exact ⟨ClopenCoeff.of T (ZMod 2) g, by rw [coeffComap_of, hg, RingEquiv.apply_symm_apply]⟩

/-- `K_n = EL_n(R_X, I)` in both forms, for `n ≥ 2`. -/
theorem coreKernel_eq_relativeElementary {n : ℕ} (hn : 2 ≤ n) :
    glCoreKernel T (ZMod 2) n =
        (relativeElementary (Fin n) (coreTransientIdeal T (ZMod 2))).map
          (elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))).subtype ∧
      elCoreKernel T (ZMod 2) n = relativeElementary (Fin n) (coreTransientIdeal T (ZMod 2)) := by
  letI := zmodTwoAlgebra T
  haveI : Nontrivial (Fin n) := Fin.nontrivial_iff_two_le.2 hn
  have hI :=
    isLocallyMatricialIn_coreTransientIdeal transientIdealLocallyMatricialFTwoStatement_holds T
  refine ⟨ChainRadical.ker_elementaryMatrixUnitMap_eq (ι := Fin n) (coreRestrict T (ZMod 2)) hI, ?_⟩
  have h := ChainRadical.ker_elementaryGroupMap_eq_relativeElementary (ι := Fin n)
    (coreRestrict T (ZMod 2)) hI
  have hc := ChainRadical.ker_elementaryGroupMap_eq_comap (ι := Fin n) (coreRestrict T (ZMod 2))
  rw [MonoidHom.comap_ker] at hc
  exact hc.symm.trans h

end Assembly

/-- **tex 1614–1621 and 1644–1652**, the clauses of `thm:core-mf-radical` that need no LEF model:
over `F₂`, `K_n = EL_n(R_X, I)` and `K_n` is locally finite for `n ≥ 2`, every homomorphism from
`EL_n(R_X)` to an MF group kills `EL_n(R_X) ∩ K_n` for `n ≥ 4`, and the elementary restriction is
surjective. -/
def PrintedCoreKernelAndELKill : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X),
    (∀ n : ℕ, 2 ≤ n →
      glCoreKernel T (ZMod 2) n =
          (relativeElementary (Fin n) (coreTransientIdeal T (ZMod 2))).map
            (elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))).subtype ∧
        elCoreKernel T (ZMod 2) n = relativeElementary (Fin n) (coreTransientIdeal T (ZMod 2)) ∧
        IsLocallyFiniteGroup (glCoreKernel T (ZMod 2) n)) ∧
      (∀ n : ℕ, 4 ≤ n → elCoreKernel T (ZMod 2) n ≤
        mfHomKernel ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2)))) ∧
      ∀ n : ℕ, Function.Surjective (elementaryGroupMap (ι := Fin n) (coreRestrict T (ZMod 2)))

theorem printedCoreKernelAndELKill : PrintedCoreKernelAndELKill := by
  intro X _ _ _ _ T
  refine ⟨fun n hn ↦ ?_, fun n hn ↦ coreMFRadicalKillELStatement_holds X T n hn,
    fun n ↦ elementaryGroupMap_coreRestrict_surjective T n⟩
  obtain ⟨hgl, hel⟩ := coreKernel_eq_relativeElementary T hn
  exact ⟨hgl, hel, coreKernelLocallyFiniteStatement_holds X T n hn⟩

/-- **`thm:core-mf-radical`**, as printed (tex 1609–1623). -/
def PrintedCoreMFRadical : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X),
    (∀ n : ℕ, 2 ≤ n →
      glCoreKernel T (ZMod 2) n =
          (relativeElementary (Fin n) (coreTransientIdeal T (ZMod 2))).map
            (elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))).subtype ∧
        elCoreKernel T (ZMod 2) n = relativeElementary (Fin n) (coreTransientIdeal T (ZMod 2)) ∧
        IsLocallyFiniteGroup (glCoreKernel T (ZMod 2) n)) ∧
      (∀ n : ℕ, 3 ≤ n →
        mfHomKernel (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T (ZMod 2)))ˣ =
          glCoreKernel T (ZMod 2) n) ∧
      (∀ n : ℕ, 4 ≤ n →
        mfHomKernel ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))) =
          elCoreKernel T (ZMod 2) n) ∧
      (∀ n : ℕ, Function.Surjective (elementaryGroupMap (ι := Fin n) (coreRestrict T (ZMod 2))) ∧
        IsLEF ↥(elementaryGroup (Fin n)
          (ClopenCrossedProduct (restrictHomeo T (image_core T)) (ZMod 2)))) ∧
      ∀ n : ℕ, IsLEF (glCoreRestrict T (ZMod 2) n).range

/-- **`thm:core-mf-radical`** over "`R_Y` is LEF" (lem:chain-core-models). -/
theorem printedCoreMFRadical_of_coreModels (hlef : CoreModelsLEFStatement) :
    PrintedCoreMFRadical := by
  intro X _ _ _ _ T
  have hR : Pestov91.IsLEFRing
      (ClopenCrossedProduct (restrictHomeo T (image_core T)) (ZMod 2)) := hlef X T (ZMod 2)
  haveI : CompactSpace ↥(Dynamics.chainRecurrentSet ⇑T) :=
    isCompact_iff_compactSpace.1 (Dynamics.isClosed_chainRecurrentSet T.continuous).isCompact
  haveI : Countable (ClopenCrossedProduct (restrictHomeo T (image_core T)) (ZMod 2)) :=
    ClopenCrossedProduct.countable (restrictHomeo T (image_core T)) (ZMod 2)
  have hM : ∀ n : ℕ, IsOperatorMF
      (Matrix (Fin n) (Fin n) (ClopenCrossedProduct (restrictHomeo T (image_core T)) (ZMod 2)))ˣ :=
    fun n ↦ hR.isOperatorMF_generalLinear (Fin n)
  have hkill := coreMFRadicalKillStatement_of_coreModels hlef X T
  obtain ⟨hker, -, hsurj⟩ := printedCoreKernelAndELKill X T
  refine ⟨hker, fun n hn ↦ ?_, fun n hn ↦ ?_, fun n ↦ ⟨hsurj n, Pestov91.isLEF_elementaryGroup hR⟩,
    fun n ↦ MFRadicalTransport.isLEF_range (hR.isLEF_generalLinear (Fin n))
      (glCoreRestrict T (ZMod 2) n)⟩
  · exact MFRadicalTransport.mfHomKernel_eq_ker_of_le (hM n) (glCoreRestrict T (ZMod 2) n)
      (hkill.1 n hn)
  · exact MFRadicalTransport.mfHomKernel_eq_ker_of_le (hM n) (elCoreRestrict T (ZMod 2) n)
      (hkill.2.1 n hn)

end CoreMFRadical
