/-!
# `simple_kazhdan_sofic_group.tex`, marked limits: the free algebra has Kazhdan `EL₃`

`simple_kazhdan_sofic_group.tex` at e80dcf20a ("Finite models"), tex 147 and 163–164:

> Let $F=\F_2\langle t_+,t_-,t_a:a\in A\rangle$ be the free algebra, [...]
> Since $\EL_3(F)$ has property~\textup{(T)}, their Cayley graphs form a family of
> expanders~\cite{Kassabov}.

This module proves the property (T) input of the expander sentence:

* `isFinitelyGeneratedRing_freeAlgebra`: for a finite index set `ι`, the free algebra
  `F₂⟨t_i : i ∈ ι⟩` is a finitely generated ring, generated by its letters.  Scalars `r ∈ F₂` are
  the natural casts `r.val`, which lie in every subring.
* `hasKazhdanPropertyT_elementaryGroup_freeAlgebra`: `EL₃(F₂⟨t_i⟩)` has property (T), by the proved
  Ershov–Jaikin-Zapirain theorem `FullDefectRingEJZUnconditional.ejz`.
* `printedFreeAlgebraElementaryKazhdan`: the closed endpoint at the printed letters
  `t_+, t_-, t_a` (`a ∈ A`, `A` finite), indexed by `Bool ⊕ A`.

The expander conclusion itself is not carried here.
-/

namespace GroupApproximation

namespace SimpleKazhdanSofic

open Manuscript.OneSidedMFRadical

/-- tex 147: the free algebra `F₂⟨t_i : i ∈ ι⟩` over a finite index set is a finitely generated
ring, generated by its letters. -/
theorem isFinitelyGeneratedRing_freeAlgebra (ι : Type*) [Finite ι] :
    IsFinitelyGeneratedRing (FreeAlgebra (ZMod 2) ι) := by
  classical
  refine ⟨(Set.finite_range (FreeAlgebra.ι (ZMod 2) : ι → FreeAlgebra (ZMod 2) ι)).toFinset, ?_⟩
  refine eq_top_iff.2 fun x _ => ?_
  induction x using FreeAlgebra.induction with
  | grade0 r =>
      rw [← ZMod.natCast_zmod_val r, map_natCast]
      exact natCast_mem _ _
  | grade1 g =>
      exact Subring.subset_closure (by simp)
  | mul a b ha hb => exact Subring.mul_mem _ ha hb
  | add a b ha hb => exact Subring.add_mem _ ha hb

/-- tex 163: "$\EL_3(F)$ has property~\textup{(T)}", for the free algebra on any finite set of
letters, by the Ershov–Jaikin-Zapirain theorem. -/
theorem hasKazhdanPropertyT_elementaryGroup_freeAlgebra (ι : Type) [Finite ι] :
    HasKazhdanPropertyT.{0, 0} (elementaryGroup (Fin 3) (FreeAlgebra (ZMod 2) ι)) :=
  FullDefectRingEJZUnconditional.ejz (FreeAlgebra (ZMod 2) ι)
    (isFinitelyGeneratedRing_freeAlgebra ι) 3 le_rfl

/-- **tex 147 and 163**, "Let $F=\F_2\langle t_+,t_-,t_a:a\in A\rangle$ be the free algebra [...]
Since $\EL_3(F)$ has property~\textup{(T)}": for every finite alphabet `A`, the elementary group
`EL₃(F₂⟨t_+, t_-, t_a : a ∈ A⟩)` has property (T).  The letters `t_+`, `t_-` are
`Sum.inl true`, `Sum.inl false`, and `t_a` is `Sum.inr a`. -/
def PrintedFreeAlgebraElementaryKazhdan : Prop :=
  ∀ (A : Type) [Finite A],
    HasKazhdanPropertyT.{0, 0} (elementaryGroup (Fin 3) (FreeAlgebra (ZMod 2) (Bool ⊕ A)))

theorem printedFreeAlgebraElementaryKazhdan : PrintedFreeAlgebraElementaryKazhdan :=
  fun A _ => hasKazhdanPropertyT_elementaryGroup_freeAlgebra (Bool ⊕ A)

end SimpleKazhdanSofic

end GroupApproximation
