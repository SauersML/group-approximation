/-!
# `lem:involution-localization`: the `F₂` matrix clause from the ring clause

`non_mf_groups_exist.tex`, proof of `lem:involution-localization` (tex 1689–1693):

> For kernel matrices, include all entries of their deviations and inverse deviations in $F$.
> Scalar conjugation then gives the unitized corner kernels.  Over $\F_2$, the finite matrix
> argument in Theorem~\ref{thm:core-mf-radical} puts $wI_n$ in $\EL_n(R_X)$ for $n\ge2$; it also
> proves local finiteness of $K_1(I)$.

This module derives the `F₂` clause `InvolutionLocalizationMatrixClause` of the lemma from its
finite-field clause `InvolutionLocalizationRingClause` and the finite matrix argument
`CoreKernelElementaryStatement` (chain-radical), along the printed route:

* `map_entry_sub_one_eq_zero`, `deviation_mem_coreTransientIdeal`: the entries of the deviation
  `G - 1` of a matrix of `K_n(I)` lie in `I`;
* `exists_finset_deviations`: all entries of the deviations `G - 1` and inverse deviations
  `G⁻¹ - 1`, `G ∈ F`, form one finite subset of `I`;
* `exists_involution_zmodTwo`: the finite-field clause at `F₂`;
* `scalar_conj_sub_one_apply`: scalar conjugation acts entrywise on deviations,
  `((w I_n) G (w I_n)⁻¹ - 1)_{ij} = w (G - 1)_{ij} w⁻¹`;
* `restrict_eq_one_of_sub_one_mem`, `scalar_mem_ker_of_map_eq_one`: `w - 1 ∈ I` puts `w I_n` in
  `K_n(I)`, hence in `EL_n(R_X)` for `n ≥ 2` by the finite matrix argument;
* `involutionLocalizationMatrixClause_of_ringClause`: the assembly.

The local finiteness of `K_1(I)` in the last printed sentence belongs to the finite matrix argument
of `thm:core-mf-radical` (chain-radical) and is not part of this clause.
-/

namespace GroupApproximation

namespace ClopenCrossedProduct

section Scalar

variable {R : Type*} [Ring R] {n : ℕ}

/-- "Scalar conjugation then gives the unitized corner kernels" (tex 1690): conjugation by the
scalar matrix `w I_n` acts entrywise, `((w I_n) G (w I_n)⁻¹)_{ij} = w G_{ij} w⁻¹`. -/
theorem scalar_conj_apply (w : Rˣ) (G : (Matrix (Fin n) (Fin n) R)ˣ) (i j : Fin n) :
    ((Units.map (Matrix.scalar (Fin n)).toMonoidHom w * G *
        (Units.map (Matrix.scalar (Fin n)).toMonoidHom w)⁻¹ : (Matrix (Fin n) (Fin n) R)ˣ) :
        Matrix (Fin n) (Fin n) R) i j =
      (w : R) * (G : Matrix (Fin n) (Fin n) R) i j * ((w⁻¹ : Rˣ) : R) := by
  show (Matrix.scalar (Fin n) (w : R) * (G : Matrix (Fin n) (Fin n) R) *
      Matrix.scalar (Fin n) ((w⁻¹ : Rˣ) : R)) i j = _
  simp only [Matrix.scalar_apply, Matrix.mul_diagonal, Matrix.diagonal_mul]

/-- Scalar conjugation on deviations: `((w I_n) G (w I_n)⁻¹ - 1)_{ij} = w (G - 1)_{ij} w⁻¹`. -/
theorem scalar_conj_sub_one_apply (w : Rˣ) (G : (Matrix (Fin n) (Fin n) R)ˣ) (i j : Fin n) :
    ((Units.map (Matrix.scalar (Fin n)).toMonoidHom w * G *
        (Units.map (Matrix.scalar (Fin n)).toMonoidHom w)⁻¹ : (Matrix (Fin n) (Fin n) R)ˣ) :
        Matrix (Fin n) (Fin n) R) i j - (1 : Matrix (Fin n) (Fin n) R) i j =
      (w : R) * ((G : Matrix (Fin n) (Fin n) R) i j - (1 : Matrix (Fin n) (Fin n) R) i j) *
        ((w⁻¹ : Rˣ) : R) := by
  rw [scalar_conj_apply, mul_sub, sub_mul, Matrix.one_apply]
  split_ifs
  · rw [mul_one, Units.mul_inv]
  · rw [mul_zero, zero_mul]

end Scalar

section Kernel

variable {R S : Type*} [Ring R] [Ring S] {n : ℕ} (f : R →+* S)

/-- An entry of the deviation `G - 1` of a matrix in the kernel of `GL_n(f)` is killed by `f`. -/
theorem map_entry_sub_one_eq_zero {G : (Matrix (Fin n) (Fin n) R)ˣ}
    (hG : G ∈ (Units.map (RingHom.mapMatrix f).toMonoidHom).ker) (i j : Fin n) :
    f ((G : Matrix (Fin n) (Fin n) R) i j - (1 : Matrix (Fin n) (Fin n) R) i j) = 0 := by
  have h : RingHom.mapMatrix f (G : Matrix (Fin n) (Fin n) R) = 1 :=
    congrArg Units.val (MonoidHom.mem_ker.1 hG)
  have hij := congrArg (fun M : Matrix (Fin n) (Fin n) S => M i j) h
  simp only [RingHom.mapMatrix_apply, Matrix.map_apply] at hij
  rw [map_sub, hij, Matrix.one_apply, Matrix.one_apply]
  split_ifs
  · rw [map_one, sub_self]
  · rw [map_zero, sub_self]

/-- `f w = 1` puts the scalar matrix `w I_n` in the kernel of `GL_n(f)`. -/
theorem scalar_mem_ker_of_map_eq_one {w : Rˣ} (hw : f (w : R) = 1) :
    Units.map (Matrix.scalar (Fin n)).toMonoidHom w ∈
      (Units.map (RingHom.mapMatrix f).toMonoidHom).ker := by
  rw [MonoidHom.mem_ker]
  ext i j
  show f ((Matrix.scalar (Fin n) (w : R)) i j) = (1 : Matrix (Fin n) (Fin n) S) i j
  rw [Matrix.scalar_apply, Matrix.diagonal_apply, Matrix.one_apply]
  split_ifs
  · exact hw
  · exact map_zero f

end Kernel

section Core

variable {X : Type} [MetricSpace X] [CompactSpace X]

/-- "For kernel matrices, include all entries of their deviations ... in $F$" (tex 1689): the
entries of the deviation `G - 1` of a matrix of `K_n(I)` lie in `I`. -/
theorem deviation_mem_coreTransientIdeal {n : ℕ} (k : Type) [Ring k] (T : X ≃ₜ X)
    {G : (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ}
    (hG : G ∈ (coreRestrictMatrixUnits T k n).ker) (i j : Fin n) :
    (G : Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k)) i j -
        (1 : Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k)) i j ∈ coreTransientIdeal T k :=
  (mem_transientIdeal T k (Dynamics.image_chainRecurrentSet T)).2
    (map_entry_sub_one_eq_zero (restrict T k (Dynamics.image_chainRecurrentSet T)) hG i j)

/-- `w - 1 ∈ I` means that `w` restricts to `1` on the chain-recurrent core. -/
theorem restrict_eq_one_of_sub_one_mem (k : Type) [Ring k] (T : X ≃ₜ X)
    {w : ClopenCrossedProduct T k} (hw : w - 1 ∈ coreTransientIdeal T k) :
    restrict T k (Dynamics.image_chainRecurrentSet T) w = 1 := by
  have h := (mem_transientIdeal T k (Dynamics.image_chainRecurrentSet T)).1 hw
  rwa [map_sub, map_one, sub_eq_zero] at h

/-- "For kernel matrices, include all entries of their deviations and inverse deviations in $F$"
(tex 1689): for a finite subset `F` of `K_n(I)`, all entries of the deviations `G - 1` and
`G⁻¹ - 1`, `G ∈ F`, form one finite subset of `I`. -/
theorem exists_finset_deviations {n : ℕ} (k : Type) [Ring k] (T : X ≃ₜ X)
    (F : Finset (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ)
    (hF : ∀ G ∈ F, G ∈ (coreRestrictMatrixUnits T k n).ker) :
    ∃ E : Finset (ClopenCrossedProduct T k), (∀ x ∈ E, x ∈ coreTransientIdeal T k) ∧
      ∀ G ∈ F, ∀ i j : Fin n,
        (G : Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k)) i j -
            (1 : Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k)) i j ∈ E ∧
          ((G⁻¹ : (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ) :
              Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k)) i j -
            (1 : Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k)) i j ∈ E := by
  classical
  refine ⟨F.biUnion fun G =>
      (Finset.univ : Finset (Fin n × Fin n)).image
          (fun p => (G : Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k)) p.1 p.2 -
            (1 : Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k)) p.1 p.2) ∪
        (Finset.univ : Finset (Fin n × Fin n)).image
          (fun p => ((G⁻¹ : (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ) :
              Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k)) p.1 p.2 -
            (1 : Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k)) p.1 p.2), ?_, ?_⟩
  · intro x hx
    obtain ⟨G, hG, hx⟩ := Finset.mem_biUnion.1 hx
    rcases Finset.mem_union.1 hx with hx | hx
    · obtain ⟨p, _, rfl⟩ := Finset.mem_image.1 hx
      exact deviation_mem_coreTransientIdeal k T (hF G hG) p.1 p.2
    · obtain ⟨p, _, rfl⟩ := Finset.mem_image.1 hx
      exact deviation_mem_coreTransientIdeal k T (inv_mem (hF G hG)) p.1 p.2
  · intro G hG i j
    exact ⟨Finset.mem_biUnion.2 ⟨G, hG, Finset.mem_union.2 (Or.inl
        (Finset.mem_image.2 ⟨(i, j), Finset.mem_univ _, rfl⟩))⟩,
      Finset.mem_biUnion.2 ⟨G, hG, Finset.mem_union.2 (Or.inr
        (Finset.mem_image.2 ⟨(i, j), Finset.mem_univ _, rfl⟩))⟩⟩

end Core

/-- The finite-field clause of `lem:involution-localization` at `F₂` (tex 1665–1667), with the
coefficient ring spelled as in the `F₂` clause. -/
theorem exists_involution_zmodTwo (hring : InvolutionLocalizationRingClause) {X : Type}
    [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X] (T : X ≃ₜ X)
    {C : Set X} (hC : IsClopen C) (hfull : ∀ x : X, ∃ j : ℤ, (T ^ j) x ∈ C)
    (hret : ∃ H : ℕ, ∀ x ∈ C, ∃ h : ℕ, 0 < h ∧ h ≤ H ∧ (T ^ (h : ℤ)) x ∈ C)
    (E : Finset (ClopenCrossedProduct T (ZMod 2)))
    (hE : ∀ x ∈ E, x ∈ coreTransientIdeal T (ZMod 2)) :
    ∃ w : (ClopenCrossedProduct T (ZMod 2))ˣ,
      (w : ClopenCrossedProduct T (ZMod 2)) - 1 ∈ coreTransientIdeal T (ZMod 2) ∧
      (w : ClopenCrossedProduct T (ZMod 2)) * w = 1 ∧
      ∀ x ∈ E, ∃ a ∈ coreTransientIdeal T (ZMod 2),
        (w : ClopenCrossedProduct T (ZMod 2)) * x * ↑w⁻¹ =
          coeff T (ZMod 2) (LocallyConstant.charFn (ZMod 2) hC) * a *
            coeff T (ZMod 2) (LocallyConstant.charFn (ZMod 2) hC) :=
  hring X T C hC hfull hret (ZMod 2) E hE

/-- **lem:involution-localization, `F₂` clause, from the finite-field clause** (tex 1689–1693):
"For kernel matrices, include all entries of their deviations and inverse deviations in $F$.
Scalar conjugation then gives the unitized corner kernels.  Over $\F_2$, the finite matrix
argument in Theorem~\ref{thm:core-mf-radical} puts $wI_n$ in $\EL_n(R_X)$ for $n\ge2$."

The finite matrix argument is the explicit hypothesis `CoreKernelElementaryStatement`
(chain-radical). -/
theorem involutionLocalizationMatrixClause_of_ringClause
    (hring : InvolutionLocalizationRingClause) (hel : CoreKernelElementaryStatement) :
    InvolutionLocalizationMatrixClause := by
  intro X _ _ _ _ T C hC hfull hret n F hF
  obtain ⟨E, hE, hFE⟩ := exists_finset_deviations (ZMod 2) T F hF
  obtain ⟨w, hw1, hww, hconj⟩ := exists_involution_zmodTwo hring T hC hfull hret E hE
  refine ⟨w, hw1, hww, fun G hG i j => ?_, fun hn => hel X T n hn
    (scalar_mem_ker_of_map_eq_one (restrict T (ZMod 2) (Dynamics.image_chainRecurrentSet T))
      (restrict_eq_one_of_sub_one_mem (ZMod 2) T hw1))⟩
  obtain ⟨a, ha, hwa⟩ := hconj _ (hFE G hG i j).1
  exact ⟨a, ha, (scalar_conj_sub_one_apply w G i j).trans hwa⟩

end ClopenCrossedProduct
