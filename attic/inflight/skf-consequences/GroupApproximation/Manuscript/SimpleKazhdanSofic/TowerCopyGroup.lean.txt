/-!
# The copy `H_W` of `GL_d(F_2)` on the tower over `W`

`simple_kazhdan_sofic_group.tex` at bf961c128, tex 205–218:

> Let $H_W$ be the copy of $\GL_d(\F_2)$ acting on the tower over $W$ and as the identity elsewhere.
> It lies in $G$.  Indeed, index its coordinates by $(p,a)$, where $1\le p\le3$ and $|a|\le m$.
> A transvection between $(p,a)$ and $(q,b)$ with $p\ne q$ is $e_{pq}(E_{ab}(W))$.  For $p=q$ and
> $a\ne b$, take $q\ne p$ and use
> $[e_{pq}(E_{ab}(W)),e_{qp}(E_{bb}(W))]=I_3+E_{ab}(W)E_{pp}$.
> This follows from $E_{ab}E_{bb}=E_{ab}$ and $E_{bb}E_{ab}=0$.
> These transvections generate $H_W$.
> Over $\F_2$, the group $\GL_d(\F_2)=\SL_d(\F_2)=\PSL_d(\F_2)$ is simple and centerless for $d\ge3$.

* `towerCopy hlev hWU hW : GL_{3(2m+1)}(F_2) →* GL_3(R_X)`: flatten `(p,a)` coordinates
  (`Matrix.compRingEquiv`), then apply the corner map `h ↦ (1 - e) I_3 + h` of `towerHom`;
* `towerCopy_elementaryUnit`: a transvection between `(p,a)` and `(q,b)`, `p ≠ q`, goes to
  `e_{pq}(E_{ab}(W))`;
* `elementaryUnit_commutator_eq_one_add_single`: the printed commutator identity, for `x y` with
  `x y = x`, `y x = 0`, `x x = 0`;
* `towerCopy_mem_elementaryGroup`: `H_W ≤ G`;
* `towerCopy_injective` for nonempty `W`;
* `center_units_matrix_zmodTwo_eq_bot`: `GL_d(F_2)` is centerless for `d ≥ 3`.
-/

namespace GroupApproximation
namespace SimpleKazhdanSofic

open ClopenCrossedProduct Matrix ChainRadical

variable {X : Type*} [TopologicalSpace X] {T : X ≃ₜ X} {m : ℕ} {U W : Set X}

/-- Flattening block coordinates: `GL_{3(2m+1)}(F_2) ≃* GL_3(M_{2m+1}(F_2))`. -/
noncomputable def flatEquiv (m : ℕ) :
    (Matrix (Fin 3 × Fin (2 * m + 1)) (Fin 3 × Fin (2 * m + 1)) (ZMod 2))ˣ ≃*
      (Matrix (Fin 3) (Fin 3) (Matrix (Fin (2 * m + 1)) (Fin (2 * m + 1)) (ZMod 2)))ˣ :=
  Units.mapEquiv (Matrix.compRingEquiv (Fin 3) (Fin (2 * m + 1)) (ZMod 2)).symm.toMulEquiv

/-- **The copy `H_W`** of `GL_{3(2m+1)}(F_2)` acting on the tower over `W`. -/
noncomputable def towerCopy (hlev : LevelsDisjoint T m U) (hWU : W ⊆ U) (hW : IsClopen W) :
    (Matrix (Fin 3 × Fin (2 * m + 1)) (Fin 3 × Fin (2 * m + 1)) (ZMod 2))ˣ →*
      (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2)))ˣ :=
  (cornerUnitHom (ι := Fin 3) (towerHom hlev hWU hW)).comp (flatEquiv m).toMonoidHom

theorem flatEquiv_elementaryUnit {p q : Fin 3} (hpq : p ≠ q) (a b : Fin (2 * m + 1)) (c : ZMod 2) :
    flatEquiv m (elementaryUnit (p, a) (q, b) (by simp [hpq]) c) =
      elementaryUnit p q hpq (Matrix.single a b c) := by
  apply Units.ext
  change (Matrix.compRingEquiv (Fin 3) (Fin (2 * m + 1)) (ZMod 2)).symm
      (1 + Matrix.single (p, a) (q, b) c) = 1 + Matrix.single p q (Matrix.single a b c)
  rw [map_add, map_one, ← Matrix.comp_single_single, Matrix.compRingEquiv_symm_apply,
    Equiv.symm_apply_apply]

/-- A transvection between `(p,a)` and `(q,b)` with `p ≠ q` is `e_{pq}(E_{ab}(W))`. -/
theorem towerCopy_elementaryUnit (hlev : LevelsDisjoint T m U) (hWU : W ⊆ U) (hW : IsClopen W)
    {p q : Fin 3} (hpq : p ≠ q) (a b : Fin (2 * m + 1)) :
    towerCopy hlev hWU hW (elementaryUnit (p, a) (q, b) (by simp [hpq]) 1) =
      elementaryUnit p q hpq (towerE T (ZMod 2) m hW a b) := by
  rw [towerCopy, MonoidHom.comp_apply, MulEquiv.coe_toMonoidHom, flatEquiv_elementaryUnit hpq,
    cornerUnitHom_elementaryUnit]
  congr 1
  rw [towerHom_apply]
  simp only [Matrix.single_apply, apply_ite ZMod.val, ZMod.val_one, ZMod.val_zero, ite_nsmul,
    one_nsmul, zero_nsmul]
  simp [Finset.sum_ite, Finset.filter_eq', Finset.filter_eq]

/-- **The printed commutator identity** (tex 211–213):
`[e_{pq}(x), e_{qp}(y)] = I + x E_{pp}` when `x y = x`, `y x = 0` and `x x = 0`. -/
theorem elementaryUnit_commutator_eq_one_add_single {S : Type*} [Ring S] {ι : Type*} [Fintype ι]
    [DecidableEq ι] {p q : ι} (hpq : p ≠ q) {x y : S} (hxy : x * y = x) (hyx : y * x = 0)
    (hxx : x * x = 0) :
    ((elementaryUnit p q hpq x * elementaryUnit q p hpq.symm y * (elementaryUnit p q hpq x)⁻¹ *
        (elementaryUnit q p hpq.symm y)⁻¹ : (Matrix ι ι S)ˣ) : Matrix ι ι S) =
      1 + Matrix.single p p x := by
  have hyy : y * x * y = 0 := by rw [hyx, zero_mul]
  simp only [Units.val_mul, elementaryUnit, Units.inv_mk]
  simp only [add_mul, mul_add, one_mul, mul_one, sub_eq_add_neg, neg_mul, mul_neg,
    Matrix.single_mul_single_same, Matrix.single_mul_single_of_ne _ _ _ hpq,
    Matrix.single_mul_single_of_ne _ _ _ hpq.symm, hxy, hyx, hxx, hyy, Matrix.single_zero,
    neg_zero, add_zero, zero_add, mul_assoc]
  abel

/-- **`H_W` lies in `G = EL_3(R_X)`.** -/
theorem towerCopy_mem_elementaryGroup (hlev : LevelsDisjoint T m U) (hWU : W ⊆ U) (hW : IsClopen W)
    (x : (Matrix (Fin 3 × Fin (2 * m + 1)) (Fin 3 × Fin (2 * m + 1)) (ZMod 2))ˣ) :
    towerCopy hlev hWU hW x ∈ elementaryGroup (Fin 3) (ClopenCrossedProduct T (ZMod 2)) := by
  have hmap : (elementaryGroup (Fin 3) (Matrix (Fin (2 * m + 1)) (Fin (2 * m + 1)) (ZMod 2))).map
      (cornerUnitHom (ι := Fin 3) (towerHom hlev hWU hW)) ≤
        elementaryGroup (Fin 3) (ClopenCrossedProduct T (ZMod 2)) := by
    refine Subgroup.map_le_iff_le_comap.2 ((Subgroup.closure_le (K := _)).2 ?_)
    rintro _ ⟨i, j, hij, c, rfl⟩
    refine Subgroup.mem_comap.2 ?_
    rw [cornerUnitHom_elementaryUnit]
    exact elementaryUnit_mem _ _ _ _
  refine hmap ⟨flatEquiv m x, ?_, rfl⟩
  rw [elementaryGroup_matrix_zmodTwo_eq_top (ι := Fin 3) (Fin (2 * m + 1))]
  exact Subgroup.mem_top _

theorem towerCopy_injective (hlev : LevelsDisjoint T m U) (hWU : W ⊆ U) (hW : IsClopen W)
    (hne : W.Nonempty) : Function.Injective (towerCopy hlev hWU hW) :=
  (cornerUnitHom_injective _ (towerHom_injective hlev hWU hW hne)).comp (flatEquiv m).injective

/-- **`GL_d(F_2)` is centerless** for `d ≥ 3`: it is simple and not commutative. -/
theorem center_units_matrix_zmodTwo_eq_bot (ι : Type*) [Fintype ι] [DecidableEq ι]
    (hcard : 3 ≤ Fintype.card ι) : Subgroup.center (Matrix ι ι (ZMod 2))ˣ = ⊥ := by
  haveI := FinitaryLinear.isSimpleGroup_units_matrix_zmodTwo ι hcard
  refine (IsSimpleGroup.eq_bot_or_eq_top_of_normal _ inferInstance).resolve_right fun htop => ?_
  obtain ⟨i, j, hij⟩ : ∃ i j : ι, i ≠ j :=
    Fintype.one_lt_card_iff_nontrivial.1 (by omega) |>.exists_pair_ne
  have hc : elementaryUnit i j hij (1 : ZMod 2) ∈ Subgroup.center (Matrix ι ι (ZMod 2))ˣ := by
    rw [htop]
    exact Subgroup.mem_top _
  have hcomm := (Subgroup.mem_center_iff.1 hc (elementaryUnit j i hij.symm 1)).symm
  have hii := congrArg (fun u : (Matrix ι ι (ZMod 2))ˣ => (u : Matrix ι ι (ZMod 2)) i i) hcomm
  simp [elementaryUnit, Matrix.mul_apply, Matrix.single_apply, Matrix.one_apply, hij,
    hij.symm] at hii

end SimpleKazhdanSofic
