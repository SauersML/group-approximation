/-!
# A finite tower detects every normal subgroup: statements

`simple_kazhdan_sofic_group.tex` at bf961c128, subsection "A finite tower detects every normal
subgroup", tex 154–228.

> For $|a|,|b|\le m$ and clopen $W\subseteq U$, put $E_{ab}(W)=e_{T^aW}u^{a-b}$.  These satisfy
> $E_{ab}(W)E_{cd}(W')=\delta_{bc}E_{ad}(W\cap W')$.  Their span is a subring
> $B_m(U)\cong M_{2m+1}(\LC(U,\F_2))$.

This module fixes the spellings shared by the two halves of the printed proof:

* `towerE T k m hW a b` is the printed `E_{ab}(W)`, with `a, b : Fin (2m+1)` standing for
  `a - m, b - m`: `coeff (1_{T^{a-m}(W)}) * u^{a-b}`;
* `towerSpan T k m U` is the additive span of the `E_{ab}(W')` over clopen `W' ⊆ U` (the printed
  `B_m(U)`; over `F_2` the additive span is the linear span);
* `LevelsDisjoint T m U`: the levels `T^a U`, `|a| ≤ m`, are pairwise disjoint;
* `TowerCornerWitness T`: what tex 174–204 produce for a nontrivial normal subgroup `N` of
  `G = EL_3(R_X)` over `F_2`.  There is a tower over clopen `U ⊇ W`, `W` nonempty, and
  `1 ≠ k ∈ N` with `k - I_3, k⁻¹ - I_3 ∈ M_3(B_m(U))`.  The coordinates of `k` over `W` are a
  constant matrix `κ₀ ≠ I_d`, `d = 3(2m+1)`: `E_{aa}(W) k_{pq} E_{bb}(W) = κ₀((p,a),(q,b)) E_{ab}(W)`.

The coefficient `κ₀((p,a),(q,b)) ∈ F_2` acts through `ZMod.val`, so this module needs no
`F_2`-algebra structure on `R_X`.  Tex 205–228 derive `N = G` from `TowerCornerWitness T`
(`Manuscript/SimpleKazhdanSofic/TowerNormalSubgroup`).
-/

namespace GroupApproximation
namespace SimpleKazhdanSofic

open ClopenCrossedProduct

variable {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type*) [Ring k]

/-- The printed `E_{ab}(W) = e_{T^a W} u^{a-b}`, indices `a, b : Fin (2m+1)` shifted by `m`. -/
noncomputable def towerE (m : ℕ) {W : Set X} (hW : IsClopen W) (a b : Fin (2 * m + 1)) :
    ClopenCrossedProduct T k :=
  coeff T k (LocallyConstant.charFn k (isClopen_image (T ^ (((a : ℕ) : ℤ) - m)) hW)) *
    ((unit T k ^ (((a : ℕ) : ℤ) - ((b : ℕ) : ℤ)) : (ClopenCrossedProduct T k)ˣ) :
      ClopenCrossedProduct T k)

/-- The printed `B_m(U)`: the additive span of the `E_{ab}(W')` over clopen `W' ⊆ U`. -/
def towerSpan (m : ℕ) (U : Set X) : AddSubgroup (ClopenCrossedProduct T k) :=
  AddSubgroup.closure
    {x | ∃ (W' : Set X) (hW' : IsClopen W'), W' ⊆ U ∧
      ∃ a b : Fin (2 * m + 1), x = towerE T k m hW' a b}

/-- The levels `T^a U`, `|a| ≤ m`, are pairwise disjoint. -/
def LevelsDisjoint (m : ℕ) (U : Set X) : Prop :=
  ∀ a b : Fin (2 * m + 1), a ≠ b →
    Disjoint ((T ^ (((a : ℕ) : ℤ) - m)) '' U) ((T ^ (((b : ℕ) : ℤ) - m)) '' U)

/-- **The tower witness of tex 174–204.**  Every nontrivial normal subgroup `N` of `EL_3(R_X)` over
`F_2` contains `1 ≠ k` supported on a disjoint tower over clopen `U`.  Over a nonempty clopen
`W ⊆ U` its coordinates are a constant matrix `κ₀ ≠ I_{3(2m+1)}`. -/
def TowerCornerWitness : Prop :=
  ∀ (N : Subgroup (elementaryGroup (Fin 3) (ClopenCrossedProduct T (ZMod 2)))), N.Normal → N ≠ ⊥ →
    ∃ (U W : Set X) (hU : IsClopen U) (hW : IsClopen W) (m : ℕ)
      (g : elementaryGroup (Fin 3) (ClopenCrossedProduct T (ZMod 2))),
      g ∈ N ∧ W ⊆ U ∧ W.Nonempty ∧ LevelsDisjoint T m U ∧ g ≠ 1 ∧
      (∀ p q : Fin 3,
        (((g : (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2)))ˣ) :
            Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2))) - 1) p q ∈
          towerSpan T (ZMod 2) m U ∧
        ((((g⁻¹ : elementaryGroup (Fin 3) (ClopenCrossedProduct T (ZMod 2))) :
            (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2)))ˣ) :
            Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2))) - 1) p q ∈
          towerSpan T (ZMod 2) m U) ∧
      ∃ κ₀ : Matrix (Fin 3 × Fin (2 * m + 1)) (Fin 3 × Fin (2 * m + 1)) (ZMod 2), κ₀ ≠ 1 ∧
        ∀ (p q : Fin 3) (a b : Fin (2 * m + 1)),
          towerE T (ZMod 2) m hW a a *
              ((g : (Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2)))ˣ) :
                Matrix (Fin 3) (Fin 3) (ClopenCrossedProduct T (ZMod 2))) p q *
            towerE T (ZMod 2) m hW b b =
          (κ₀ (p, a) (q, b)).val • towerE T (ZMod 2) m hW a b

end SimpleKazhdanSofic
end GroupApproximation
