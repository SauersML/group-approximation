/-!
# Words in `Λ`: the normal form `(c, δ)`

`simple_kazhdan_sofic_group.tex` at 696c4b602, section "LEF groups", l.445–447:

> An element $x\mapsto\delta x+c$ of $\Lambda$ is given by a word for $\delta$ and words for the lamps
> of $c$, so the word problem of $\Delta$ decides equality in $\Lambda$ and computes the values of $c$.

`Λ = LampAffine Δ` is generated by the generators `t` of `Δ` and the map `z` adding `1` at `e`. A word
in these letters has two parts:

* its `Δ`-letters `deltaWord w`, a word for `δ`;
* the `Δ`-prefixes `lampPrefixes w` in front of its `z`-letters, words for the lamps of `c`.

* `wordValue_lampGen`: the value of `w` is `inl c · inr δ`, where `c = Σ_p 1_{value p}` and
  `δ = value (deltaWord w)`, using `z⁻¹ = z` and `inr δ · inl F = inl (δ·F) · inr δ`;
* `wordValue_lampGen_eq_one_iff`: `w` is trivial exactly when `deltaWord w` is trivial in `Δ` and
  every lamp position occurs an even number of times among the prefixes.
-/

namespace GroupApproximation
namespace SimpleKazhdanSofic
namespace Lamplighter

open Multiplicative

variable {Δ : Type} [Group Δ] {ι : Type*}

/-- The generators of `Λ`: `none ↦ z`, `some i ↦ t i`. -/
noncomputable def lampGen (t : ι → Δ) : Option ι → LampAffine Δ
  | none => lampAdd Δ
  | some i => SemidirectProduct.inr (t i)

/-- The `Δ`-letters of a word in the generators of `Λ`. -/
def deltaWord : List (Option ι × Bool) → List (ι × Bool)
  | [] => []
  | (none, _) :: w => deltaWord w
  | (some i, b) :: w => (i, b) :: deltaWord w

/-- The `Δ`-prefixes in front of the `z`-letters of a word. -/
def lampPrefixes : List (Option ι × Bool) → List (List (ι × Bool))
  | [] => []
  | (none, _) :: w => [] :: lampPrefixes w
  | (some i, b) :: w => (lampPrefixes w).map fun p => (i, b) :: p

/-- The lamp configuration of a word: `Σ_p 1_{value p}` over its prefixes. -/
noncomputable def lampSum (t : ι → Δ) (w : List (Option ι × Bool)) : Δ →₀ ZMod 2 :=
  ((lampPrefixes w).map fun p => Finsupp.single (wordValue t p) (1 : ZMod 2)).sum

variable (Δ) in
/-- `z⁻¹ = z`. -/
theorem lampAdd_inv : (lampAdd Δ)⁻¹ = lampAdd Δ := by
  unfold lampAdd
  rw [← map_inv, ← ofAdd_neg, ← Finsupp.single_neg, show (-1 : ZMod 2) = 1 from by decide]

/-- `inr δ · inl F = inl (δ·F) · inr δ`. -/
theorem inr_mul_inl (δ : Δ) (F : Δ →₀ ZMod 2) :
    (SemidirectProduct.inr δ : LampAffine Δ) * SemidirectProduct.inl (ofAdd F) =
      SemidirectProduct.inl (ofAdd (translate Δ δ F)) * SemidirectProduct.inr δ := by
  have h := SemidirectProduct.inl_aut (φ := lampAut Δ) δ (ofAdd F)
  have h2 : lampAut Δ δ (ofAdd F) = ofAdd (translate Δ δ F) := rfl
  rw [h2] at h
  rw [h, mul_assoc, ← map_mul, inv_mul_cancel, map_one, mul_one]

theorem lampSum_nil (t : ι → Δ) : lampSum t [] = 0 := by
  simp [lampSum, lampPrefixes]

theorem lampSum_cons_none (t : ι → Δ) (b : Bool) (w : List (Option ι × Bool)) :
    lampSum t ((none, b) :: w) = Finsupp.single 1 1 + lampSum t w := by
  simp only [lampSum, lampPrefixes, List.map_cons, List.sum_cons, wordValue_nil]

theorem lampSum_cons_some (t : ι → Δ) (i : ι) (b : Bool) (w : List (Option ι × Bool)) :
    lampSum t ((some i, b) :: w) =
      translate Δ (if b then t i else (t i)⁻¹) (lampSum t w) := by
  simp only [lampSum, lampPrefixes, List.map_map]
  rw [map_list_sum (translate Δ (if b then t i else (t i)⁻¹)), List.map_map]
  congr 1
  refine List.map_congr_left fun p _ => ?_
  simp only [Function.comp_apply, translate_single, wordValue_cons]

theorem deltaWord_cons_none (b : Bool) (w : List (Option ι × Bool)) :
    deltaWord ((none, b) :: w) = deltaWord w :=
  rfl

theorem deltaWord_cons_some (i : ι) (b : Bool) (w : List (Option ι × Bool)) :
    deltaWord ((some i, b) :: w) = (i, b) :: deltaWord w :=
  rfl

/-- **The normal form of a word** (l.445): the value of `w` is `inl c · inr δ`, with the lamps `c`
read off the prefixes and `δ` the value of the `Δ`-letters. -/
theorem wordValue_lampGen (t : ι → Δ) (w : List (Option ι × Bool)) :
    wordValue (lampGen t) w =
      SemidirectProduct.inl (ofAdd (lampSum t w)) *
        SemidirectProduct.inr (wordValue t (deltaWord w)) := by
  induction w with
  | nil =>
    rw [wordValue_nil, lampSum_nil, deltaWord, wordValue_nil, ofAdd_zero, map_one, map_one,
      mul_one]
  | cons x w ih =>
    obtain ⟨o, b⟩ := x
    cases o with
    | none =>
      have hz : (if b then lampGen t none else (lampGen t none)⁻¹) = lampAdd Δ := by
        cases b
        · show (lampAdd Δ)⁻¹ = lampAdd Δ
          exact lampAdd_inv Δ
        · rfl
      rw [wordValue_cons, hz, ih, lampSum_cons_none, deltaWord_cons_none, ← mul_assoc]
      congr 1
      rw [lampAdd, ← map_mul, ← ofAdd_add]
    | some i =>
      have hl : (if b then lampGen t (some i) else (lampGen t (some i))⁻¹) =
          SemidirectProduct.inr (if b then t i else (t i)⁻¹) := by
        cases b
        · show (SemidirectProduct.inr (t i) : LampAffine Δ)⁻¹ = SemidirectProduct.inr (t i)⁻¹
          rw [map_inv]
        · rfl
      rw [wordValue_cons, hl, ih, ← mul_assoc, inr_mul_inl, mul_assoc, ← map_mul,
        lampSum_cons_some, deltaWord_cons_some, wordValue_cons]

/-- `inl c · inr δ = 1` exactly when `c = 0` and `δ = 1`. -/
theorem inl_mul_inr_eq_one_iff (F : Δ →₀ ZMod 2) (δ : Δ) :
    (SemidirectProduct.inl (ofAdd F) * SemidirectProduct.inr δ : LampAffine Δ) = 1 ↔
      δ = 1 ∧ F = 0 := by
  constructor
  · intro h
    have hr := congrArg SemidirectProduct.right h
    have hl := congrArg SemidirectProduct.left h
    rw [SemidirectProduct.mul_right, SemidirectProduct.right_inl, SemidirectProduct.right_inr,
      one_mul, SemidirectProduct.one_right] at hr
    rw [SemidirectProduct.mul_left, SemidirectProduct.left_inl, SemidirectProduct.left_inr,
      SemidirectProduct.right_inl, map_one, mul_one, SemidirectProduct.one_left] at hl
    exact ⟨hr, by simpa using hl⟩
  · rintro ⟨rfl, rfl⟩
    rw [ofAdd_zero, map_one, map_one, mul_one]

/-- The lamp at `a` is the parity of the prefixes with value `a`. -/
theorem lampSum_apply [DecidableEq Δ] (t : ι → Δ) (w : List (Option ι × Bool)) (a : Δ) :
    lampSum t w a = (((lampPrefixes w).filter fun p => wordValue t p = a).length : ZMod 2) := by
  unfold lampSum
  induction lampPrefixes w with
  | nil => simp
  | cons p l ih =>
    rw [List.map_cons, List.sum_cons, Finsupp.add_apply, ih]
    by_cases hp : wordValue t p = a
    · rw [List.filter_cons_of_pos (by simpa using hp), List.length_cons, Nat.cast_succ, hp,
        Finsupp.single_eq_same]
      ring
    · rw [List.filter_cons_of_neg (by simpa using hp), Finsupp.single_eq_of_ne (Ne.symm hp),
        zero_add]

/-- **The lamps vanish** exactly when every lamp position occurs an even number of times. -/
theorem lampSum_eq_zero_iff [DecidableEq Δ] (t : ι → Δ) (w : List (Option ι × Bool)) :
    lampSum t w = 0 ↔ ∀ p ∈ lampPrefixes w,
      ((lampPrefixes w).filter fun q => wordValue t q = wordValue t p).length % 2 = 0 := by
  constructor
  · intro h p _
    have h1 := lampSum_apply t w (wordValue t p)
    rw [h, Finsupp.zero_apply] at h1
    exact Nat.mod_eq_zero_of_dvd ((ZMod.natCast_eq_zero_iff _ 2).1 h1.symm)
  · intro h
    refine Finsupp.ext fun a => ?_
    rw [lampSum_apply, Finsupp.zero_apply]
    by_cases ha : ∃ p ∈ lampPrefixes w, wordValue t p = a
    · obtain ⟨p, hp, rfl⟩ := ha
      exact (ZMod.natCast_eq_zero_iff _ 2).2 (Nat.dvd_of_mod_eq_zero (h p hp))
    · have hnil : ((lampPrefixes w).filter fun q => wordValue t q = a) = [] := by
        refine List.filter_eq_nil_iff.2 fun q hq => ?_
        simpa using fun hqa => ha ⟨q, hq, hqa⟩
      rw [hnil, List.length_nil, Nat.cast_zero]

/-- **`w` is trivial in `Λ`** exactly when its `Δ`-letters are trivial and every lamp position occurs
an even number of times (l.445–447). -/
theorem wordValue_lampGen_eq_one_iff [DecidableEq Δ] (t : ι → Δ) (w : List (Option ι × Bool)) :
    wordValue (lampGen t) w = 1 ↔
      wordValue t (deltaWord w) = 1 ∧ ∀ p ∈ lampPrefixes w,
        ((lampPrefixes w).filter fun q => wordValue t q = wordValue t p).length % 2 = 0 := by
  rw [wordValue_lampGen, inl_mul_inr_eq_one_iff, lampSum_eq_zero_iff]
