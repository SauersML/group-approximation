/-!
# Model test: doubling a relator-face edge whose reverse lies on a relator face

A model test for the cell-edge thickening (`SurgeryCellEdgeThickening.lean`), run before its
proof.  The edge conditions `CellPocketWalk.CopyClean` (`Estimating/OsinPocketCellCopyWalk.lean`)
ask that no edge have `Π_i` on both sides, `Π_j` on both sides, or `Π_i` and `Π_j` on its two
sides.  Configuration (b′) of the multiple-edge pocket is a spur on `Π_i` between the arcs of the
two regions; locally it is the first map below.  Doubling the edge inside the relator face
(`FaceEdgeDoubling.diagram`) puts a G-cell digon on one side of it.  The combinatorial map is
given by the edge involution `alpha` and the face permutation, with dart indices in `Fin n`, and
the vertex permutation is `face ∘ alpha`.  Every claim below is checked by `decide`.

A **cell-edge dart** is a dart on a relator face whose reverse lies on a relator face.

## A spur on a relator face

Before the doubling there are four darts: a loop `{0,1}` at `v` and a spur `{2,3}` from `v` to a
pendant vertex `p`.  The relator face `Π = [0, 2, 3]` (class `0`) contains the spur, and the
exterior is `[1]` (class `1`).  The vertex count is `2 - 2 + 2 = 2`.  The darts `2` and `3` are
cell-edge darts.

After doubling `2` inside `Π` there are six darts.  The new dart `4` stands in `Π` in the place of
`2`, and the new dart `5 = ᾱ 4` and `2` form the digon `[2, 5]` (class `2`).  The vertex count is
`2 - 3 + 3 = 2`.  No cell-edge dart remains.

## An edge between two relator faces

Before the doubling there are six darts on three edges `{0,1}`, `{2,3}`, `{4,5}` from `u` to `v`.
The relator faces are `Π = [0, 3]` (class `0`) and `Σ = [4, 1]` (class `1`), and the exterior is
`[2, 5]` (class `2`).  The vertex count is `2 - 3 + 3 = 2`.  The darts `0` and `1` of the shared
edge are cell-edge darts.

After doubling `0` inside `Π` there are eight darts.  The new dart `6` stands in `Π` in the place
of `0`, and `0` and `7 = ᾱ 6` form the digon `[0, 7]` (class `3`).  The vertex count is
`2 - 4 + 4 = 2`.  No cell-edge dart remains.

* `spur_map_before`, `spur_cycles_before`, `spur_cellEdge_before`;
  `spur_map_after`, `spur_cycles_after`, `spur_cellEdge_after`, `spur_retained`.
* `shared_map_before`, `shared_cycles_before`, `shared_cellEdge_before`;
  `shared_map_after`, `shared_cycles_after`, `shared_cellEdge_after`, `shared_retained`.

-/

/-! ## A spur on a relator face, before the doubling -/

/-- The edge involution before doubling the spur. -/
def spurAlphaB : Fin 4 → Fin 4 := ![1, 0, 3, 2]

/-- The face permutation before doubling the spur. -/
def spurFaceB : Fin 4 → Fin 4 := ![2, 1, 3, 0]

/-- The vertex permutation `spurFaceB ∘ spurAlphaB` before doubling the spur. -/
def spurSigmaB : Fin 4 → Fin 4 := ![1, 2, 0, 3]

/-- The faces: `Π = 0`, exterior `1`. -/
def spurFaceClassB : Fin 4 → Fin 2 := ![0, 1, 0, 0]

/-- The vertices: `v = 0`, `p = 1`. -/
def spurVertexClassB : Fin 4 → Fin 2 := ![0, 0, 0, 1]

theorem spur_map_before :
    (∀ x, spurAlphaB (spurAlphaB x) = x) ∧ (∀ x, spurAlphaB x ≠ x) ∧
      (∀ x, spurSigmaB x = spurFaceB (spurAlphaB x)) ∧
      (∀ x y, spurFaceB x = spurFaceB y → x = y) ∧
      (∀ x, spurFaceClassB (spurFaceB x) = spurFaceClassB x) ∧
      (∀ x, spurVertexClassB (spurSigmaB x) = spurVertexClassB x) := by
  decide

/-- The two face cycles and two vertex cycles, listed. -/
theorem spur_cycles_before :
    [spurFaceB 0, spurFaceB 2, spurFaceB 3] = [2, 3, 0] ∧ [spurFaceB 1] = [1] ∧
      [spurSigmaB 0, spurSigmaB 1, spurSigmaB 2] = [1, 2, 0] ∧ [spurSigmaB 3] = [3] := by
  decide

/-- **Two cell-edge darts before the doubling**, the spur darts `2` and `3`, and `Π` has three
darts. -/
theorem spur_cellEdge_before :
    (Finset.univ.filter fun x : Fin 4 =>
        spurFaceClassB x = 0 ∧ spurFaceClassB (spurAlphaB x) = 0).card = 2 ∧
      spurFaceClassB 2 = 0 ∧ spurFaceClassB (spurAlphaB 2) = 0 ∧
      (Finset.univ.filter fun x : Fin 4 => spurFaceClassB x = 0).card = 3 := by
  decide

/-! ## A spur on a relator face, after the doubling -/

/-- The edge involution after doubling the spur. -/
def spurAlphaA : Fin 6 → Fin 6 := ![1, 0, 3, 2, 5, 4]

/-- The face permutation after doubling the spur. -/
def spurFaceA : Fin 6 → Fin 6 := ![4, 1, 5, 0, 3, 2]

/-- The vertex permutation `spurFaceA ∘ spurAlphaA` after doubling the spur. -/
def spurSigmaA : Fin 6 → Fin 6 := ![1, 4, 0, 5, 2, 3]

/-- The faces: `Π = 0`, exterior `1`, digon `2`. -/
def spurFaceClassA : Fin 6 → Fin 3 := ![0, 1, 2, 0, 0, 2]

/-- The vertices: `v = 0`, `p = 1`. -/
def spurVertexClassA : Fin 6 → Fin 2 := ![0, 0, 0, 1, 0, 1]

theorem spur_map_after :
    (∀ x, spurAlphaA (spurAlphaA x) = x) ∧ (∀ x, spurAlphaA x ≠ x) ∧
      (∀ x, spurSigmaA x = spurFaceA (spurAlphaA x)) ∧
      (∀ x y, spurFaceA x = spurFaceA y → x = y) ∧
      (∀ x, spurFaceClassA (spurFaceA x) = spurFaceClassA x) ∧
      (∀ x, spurVertexClassA (spurSigmaA x) = spurVertexClassA x) := by
  decide

/-- The three face cycles and two vertex cycles, listed. -/
theorem spur_cycles_after :
    [spurFaceA 0, spurFaceA 4, spurFaceA 3] = [4, 3, 0] ∧ [spurFaceA 1] = [1] ∧
      [spurFaceA 2, spurFaceA 5] = [5, 2] ∧
      [spurSigmaA 0, spurSigmaA 1, spurSigmaA 4, spurSigmaA 2] = [1, 4, 2, 0] ∧
      [spurSigmaA 3, spurSigmaA 5] = [5, 3] := by
  decide

/-- **No cell-edge dart after the doubling**: the new dart `4` of `Π` and the old dart `3` both
have the digon across them. -/
theorem spur_cellEdge_after :
    (Finset.univ.filter fun x : Fin 6 =>
        spurFaceClassA x = 0 ∧ spurFaceClassA (spurAlphaA x) = 0).card = 0 ∧
      spurFaceClassA 4 = 0 ∧ spurFaceClassA (spurAlphaA 4) = 2 ∧
      spurFaceClassA 3 = 0 ∧ spurFaceClassA (spurAlphaA 3) = 2 := by
  decide

/-- The old darts other than `2` keep their faces, every old dart keeps its reverse, and `Π`
still has three darts. -/
theorem spur_retained :
    (∀ x : Fin 4, x ≠ 2 →
      (spurFaceClassA (Fin.castLE (by norm_num) x)).val = (spurFaceClassB x).val) ∧
      (∀ x : Fin 4,
        spurAlphaA (Fin.castLE (by norm_num) x) = Fin.castLE (by norm_num) (spurAlphaB x)) ∧
      (Finset.univ.filter fun x : Fin 6 => spurFaceClassA x = 0).card = 3 := by
  decide

/-! ## An edge between two relator faces, before the doubling -/

/-- The edge involution before doubling the shared edge. -/
def sharedAlphaB : Fin 6 → Fin 6 := ![1, 0, 3, 2, 5, 4]

/-- The face permutation before doubling the shared edge. -/
def sharedFaceB : Fin 6 → Fin 6 := ![3, 4, 5, 0, 1, 2]

/-- The vertex permutation `sharedFaceB ∘ sharedAlphaB` before doubling the shared edge. -/
def sharedSigmaB : Fin 6 → Fin 6 := ![4, 3, 0, 5, 2, 1]

/-- The faces: `Π = 0`, `Σ = 1`, exterior `2`. -/
def sharedFaceClassB : Fin 6 → Fin 3 := ![0, 1, 2, 0, 1, 2]

/-- The vertices: `u = 0`, `v = 1`. -/
def sharedVertexClassB : Fin 6 → Fin 2 := ![0, 1, 0, 1, 0, 1]

      (∀ x, sharedSigmaB x = sharedFaceB (sharedAlphaB x)) ∧
      (∀ x y, sharedFaceB x = sharedFaceB y → x = y) ∧
      (∀ x, sharedFaceClassB (sharedFaceB x) = sharedFaceClassB x) ∧
      (∀ x, sharedVertexClassB (sharedSigmaB x) = sharedVertexClassB x) := by
  decide

/-- The three face cycles and two vertex cycles, listed. -/
theorem shared_cycles_before :
    [sharedFaceB 0, sharedFaceB 3] = [3, 0] ∧ [sharedFaceB 4, sharedFaceB 1] = [1, 4] ∧
      [sharedFaceB 2, sharedFaceB 5] = [5, 2] ∧
      [sharedSigmaB 0, sharedSigmaB 4, sharedSigmaB 2] = [4, 2, 0] ∧
      [sharedSigmaB 1, sharedSigmaB 3, sharedSigmaB 5] = [3, 5, 1] := by
  decide

/-- **Two cell-edge darts before the doubling**, the darts `0` and `1` of the shared edge, and `Π`
has two darts. -/
theorem shared_cellEdge_before :
    (Finset.univ.filter fun x : Fin 6 =>
        sharedFaceClassB x ≠ 2 ∧ sharedFaceClassB (sharedAlphaB x) ≠ 2).card = 2 ∧
      sharedFaceClassB 0 = 0 ∧ sharedFaceClassB (sharedAlphaB 0) = 1 ∧
      (Finset.univ.filter fun x : Fin 6 => sharedFaceClassB x = 0).card = 2 := by
  decide

/-! ## An edge between two relator faces, after the doubling -/

/-- The edge involution after doubling the shared edge. -/
def sharedAlphaA : Fin 8 → Fin 8 := ![1, 0, 3, 2, 5, 4, 7, 6]

/-- The face permutation after doubling the shared edge. -/
def sharedFaceA : Fin 8 → Fin 8 := ![7, 4, 5, 6, 1, 2, 3, 0]

/-- The vertex permutation `sharedFaceA ∘ sharedAlphaA` after doubling the shared edge. -/
def sharedSigmaA : Fin 8 → Fin 8 := ![4, 7, 6, 5, 2, 1, 0, 3]

/-- The faces: `Π = 0`, `Σ = 1`, exterior `2`, digon `3`. -/
def sharedFaceClassA : Fin 8 → Fin 4 := ![3, 1, 2, 0, 1, 2, 0, 3]

/-- The vertices: `u = 0`, `v = 1`. -/
def sharedVertexClassA : Fin 8 → Fin 2 := ![0, 1, 0, 1, 0, 1, 0, 1]

theorem shared_map_after :
    (∀ x, sharedAlphaA (sharedAlphaA x) = x) ∧ (∀ x, sharedAlphaA x ≠ x) ∧
      (∀ x, sharedSigmaA x = sharedFaceA (sharedAlphaA x)) ∧
      (∀ x y, sharedFaceA x = sharedFaceA y → x = y) ∧
      (∀ x, sharedFaceClassA (sharedFaceA x) = sharedFaceClassA x) ∧
      (∀ x, sharedVertexClassA (sharedSigmaA x) = sharedVertexClassA x) := by
  decide

/-- The four face cycles and two vertex cycles, listed. -/
theorem shared_cycles_after :
    [sharedFaceA 6, sharedFaceA 3] = [3, 6] ∧ [sharedFaceA 4, sharedFaceA 1] = [1, 4] ∧
      [sharedFaceA 2, sharedFaceA 5] = [5, 2] ∧ [sharedFaceA 0, sharedFaceA 7] = [7, 0] ∧
      [sharedSigmaA 0, sharedSigmaA 4, sharedSigmaA 2, sharedSigmaA 6] = [4, 2, 6, 0] ∧
      [sharedSigmaA 1, sharedSigmaA 7, sharedSigmaA 3, sharedSigmaA 5] = [7, 3, 5, 1] := by
  decide

/-- **No cell-edge dart after the doubling**: the new dart `6` of `Π` and the old dart `1` of `Σ`
both have the digon across them. -/
theorem shared_cellEdge_after :
    (Finset.univ.filter fun x : Fin 8 =>
        (sharedFaceClassA x).val ≤ 1 ∧ (sharedFaceClassA (sharedAlphaA x)).val ≤ 1).card = 0 ∧
      sharedFaceClassA 6 = 0 ∧ sharedFaceClassA (sharedAlphaA 6) = 3 ∧
      sharedFaceClassA 1 = 1 ∧ sharedFaceClassA (sharedAlphaA 1) = 3 := by
  decide

/-- The old darts other than `0` keep their faces, every old dart keeps its reverse, and `Π`
still has two darts. -/
theorem shared_retained :
    (∀ x : Fin 6, x ≠ 0 →
      (sharedFaceClassA (Fin.castLE (by norm_num) x)).val = (sharedFaceClassB x).val) ∧
      (∀ x : Fin 6,
        sharedAlphaA (Fin.castLE (by norm_num) x) = Fin.castLE (by norm_num) (sharedAlphaB x)) ∧
      (Finset.univ.filter fun x : Fin 8 => sharedFaceClassA x = 0).card = 2 := by
  decide
