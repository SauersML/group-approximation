/-!
# Transient coefficients, finite clopen supports and uniform returns

Inputs of the proof of `lem:involution-localization` (`non_mf_groups_exist.tex`, tex 1672–1676):

> Let $K\subset X\setminus Y$ contain the sources and ranges of a finite partial-shift description of
> $F$, and cover $K$ by $m$ wandering clopen sets.  [...]  If $H$ bounds the return gaps, [...]

and of its statement (tex 1664): "Let $C$ be a full clopen section with bounded positive return
times".

* `CoeffVanishOn T k Y x`: every Laurent coefficient of `x ∈ R_X` vanishes on `Y`.  For
  `Y = CR(T)` this is membership in `I = LC_c(X∖Y, k) ⋊_T ℤ`.  It is closed under sums and under
  products on either side when `Y` is invariant (`CoeffVanishOn.mul_left`, `CoeffVanishOn.mul_right`),
  and contains the partial shifts `f u^j` with `f` vanishing on `Y`;
* `charFn_mul_eq_self`, `mul_charFn_eq_self`: `1_K x = x` and `x 1_K = x` from coefficient supports;
* `exists_clopen_support`: a finite `F ⊆ I` lies in `1_K R_X 1_K` for a clopen `K ⊆ X ∖ Y`;
* `exists_wandering_clopen_cover`: from the defect cover of lem:chain-core-models, a clopen
  `K ⊆ X ∖ Y` is covered by finitely many wandering clopen sets;
* `exists_uniform_return`: on compact `X`, a full clopen section with bounded positive return times
  is reached from every point within a bounded number of positive steps.
-/

namespace GroupApproximation

namespace ClopenCrossedProduct

open Multiplicative (ofAdd toAdd)

section Support

variable {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type*) [Ring k] (Y : Set X)

/-- Every Laurent coefficient of `x` vanishes on `Y`. -/
def CoeffVanishOn (x : ClopenCrossedProduct T k) : Prop :=
  ∀ (g : Multiplicative ℤ), ∀ y ∈ Y, (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g) y = 0

theorem of_symm_smul_apply (a : Multiplicative ℤ) (c : ClopenCoeff T k) (y : X) :
    (ClopenCoeff.of T k).symm (a • c) y = (ClopenCoeff.of T k).symm c ((T ^ (-toAdd a)) y) :=
  rfl

variable {T k Y}

theorem coeffVanishOn_zero : CoeffVanishOn T k Y 0 := fun g y _ => by
  rw [SkewMonoidAlgebra.coeff_zero, Finsupp.zero_apply, map_zero, LocallyConstant.zero_apply]

theorem CoeffVanishOn.add {x x' : ClopenCrossedProduct T k} (hx : CoeffVanishOn T k Y x)
    (hx' : CoeffVanishOn T k Y x') : CoeffVanishOn T k Y (x + x') := fun g y hy => by
  rw [SkewMonoidAlgebra.coeff_add, Finsupp.add_apply, map_add, LocallyConstant.add_apply,
    hx g y hy, hx' g y hy, add_zero]

theorem CoeffVanishOn.neg {x : ClopenCrossedProduct T k} (hx : CoeffVanishOn T k Y x) :
    CoeffVanishOn T k Y (-x) := fun g y hy => by
  rw [SkewMonoidAlgebra.coeff_neg, Finsupp.neg_apply, map_neg, LocallyConstant.neg_apply,
    hx g y hy, neg_zero]

theorem CoeffVanishOn.sub {x x' : ClopenCrossedProduct T k} (hx : CoeffVanishOn T k Y x)
    (hx' : CoeffVanishOn T k Y x') : CoeffVanishOn T k Y (x - x') := by
  rw [sub_eq_add_neg]
  exact hx.add hx'.neg

theorem coeffVanishOn_sum {ι : Type*} (s : Finset ι) (f : ι → ClopenCrossedProduct T k)
    (hf : ∀ i ∈ s, CoeffVanishOn T k Y (f i)) : CoeffVanishOn T k Y (∑ i ∈ s, f i) :=
  Finset.sum_induction f (CoeffVanishOn T k Y) (fun _ _ ha hb => ha.add hb) coeffVanishOn_zero hf

/-- Left products stay in `I` when `Y` is invariant. -/
theorem CoeffVanishOn.mul_left (hY : ∀ j : ℤ, ∀ y ∈ Y, (T ^ j) y ∈ Y)
    {x : ClopenCrossedProduct T k} (hx : CoeffVanishOn T k Y x) (z : ClopenCrossedProduct T k) :
    CoeffVanishOn T k Y (z * x) := fun g y hy => by
  rw [SkewMonoidAlgebra.coeff_mul_left, SkewMonoidAlgebra.sum_def']
  refine Finset.sum_induction _ (fun c : ClopenCoeff T k => (ClopenCoeff.of T k).symm c y = 0)
    (fun a b ha hb => by rw [map_add, LocallyConstant.add_apply, ha, hb, add_zero])
    (by rw [map_zero, LocallyConstant.zero_apply]) fun a _ => ?_
  dsimp only
  rw [map_mul, LocallyConstant.mul_apply, of_symm_smul_apply,
    hx _ _ (hY (-toAdd a) y hy), mul_zero]

/-- Right products stay in `I`. -/
theorem CoeffVanishOn.mul_right {x : ClopenCrossedProduct T k} (hx : CoeffVanishOn T k Y x)
    (z : ClopenCrossedProduct T k) : CoeffVanishOn T k Y (x * z) := fun g y hy => by
  rw [SkewMonoidAlgebra.coeff_mul_right, SkewMonoidAlgebra.sum_def']
  refine Finset.sum_induction _ (fun c : ClopenCoeff T k => (ClopenCoeff.of T k).symm c y = 0)
    (fun a b ha hb => by rw [map_add, LocallyConstant.add_apply, ha, hb, add_zero])
    (by rw [map_zero, LocallyConstant.zero_apply]) fun a _ => ?_
  dsimp only
  rw [map_mul, LocallyConstant.mul_apply, hx _ y hy, zero_mul]

/-- The partial shift `f u^j` lies in `I` when `f` vanishes on `Y`. -/
theorem coeffVanishOn_coeff_mul_unit_zpow {f : LocallyConstant X k} (hf : ∀ y ∈ Y, f y = 0)
    (j : ℤ) : CoeffVanishOn T k Y (coeff T k f * ↑(unit T k ^ j)) := fun g y hy => by
  classical
  rw [Pestov91.CrossedProduct.val_unit_zpow, coeff_apply, Pestov91.CrossedProduct.C_mul_single,
    mul_one, SkewMonoidAlgebra.coeff_single_apply]
  split_ifs
  · exact hf y hy
  · rw [map_zero, LocallyConstant.zero_apply]

theorem charFn_apply_eq_zero {A : Set X} (hA : IsClopen A) {y : X} (hy : y ∉ A) :
    LocallyConstant.charFn k hA y = 0 := by
  rw [LocallyConstant.coe_charFn, Set.indicator_of_notMem hy]

/-- `1_A ∈ I` for a clopen `A` disjoint from `Y`. -/
theorem coeffVanishOn_charFn {A : Set X} (hA : IsClopen A) (hAY : Disjoint A Y) :
    CoeffVanishOn T k Y (coeff T k (LocallyConstant.charFn k hA)) := by
  have h := coeffVanishOn_coeff_mul_unit_zpow (T := T) (Y := Y)
    (fun y hy => charFn_apply_eq_zero k hA (Set.disjoint_right.1 hAY hy)) 0
  rwa [zpow_zero, Units.val_one, mul_one] at h

variable (T k)

/-- `1_K x = x` when all coefficients of `x` vanish off `K`. -/
theorem charFn_mul_eq_self {K : Set X} (hK : IsClopen K) {x : ClopenCrossedProduct T k}
    (hx : ∀ g, ∀ y, y ∉ K → (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g) y = 0) :
    coeff T k (LocallyConstant.charFn k hK) * x = x := by
  refine SkewMonoidAlgebra.ext fun g => ?_
  rw [coeff_apply, Pestov91.CrossedProduct.C_apply, SkewMonoidAlgebra.coeff_single_one_mul]
  apply (ClopenCoeff.of T k).symm.injective
  ext y
  rw [map_mul, LocallyConstant.mul_apply, RingEquiv.symm_apply_apply, LocallyConstant.coe_charFn]
  by_cases hy : y ∈ K
  · rw [Set.indicator_of_mem hy, Pi.one_apply, one_mul]
  · rw [Set.indicator_of_notMem hy, zero_mul, hx g y hy]

/-- `x 1_K = x` when the coefficient of `u^j` vanishes at every `y` with `T^{-j} y ∉ K`. -/
theorem mul_charFn_eq_self {K : Set X} (hK : IsClopen K) {x : ClopenCrossedProduct T k}
    (hx : ∀ g, ∀ y, (T ^ (-toAdd g)) y ∉ K → (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g) y = 0) :
    x * coeff T k (LocallyConstant.charFn k hK) = x := by
  refine SkewMonoidAlgebra.ext fun g => ?_
  rw [coeff_apply, Pestov91.CrossedProduct.C_apply, SkewMonoidAlgebra.coeff_mul_single_one]
  apply (ClopenCoeff.of T k).symm.injective
  ext y
  rw [map_mul, LocallyConstant.mul_apply, of_symm_smul_apply, RingEquiv.symm_apply_apply,
    LocallyConstant.coe_charFn]
  by_cases hy : (T ^ (-toAdd g)) y ∈ K
  · rw [Set.indicator_of_mem hy, Pi.one_apply, mul_one]
  · rw [Set.indicator_of_notMem hy, mul_zero, hx g y hy]

/-- **Finite support** (tex 1673): a finite `F ⊆ I` lies in `1_K R_X 1_K` for a clopen `K` disjoint
from the invariant set `Y`. -/
theorem exists_clopen_support (hY : ∀ j : ℤ, ∀ y ∈ Y, (T ^ j) y ∈ Y)
    (F : Finset (ClopenCrossedProduct T k)) (hF : ∀ x ∈ F, CoeffVanishOn T k Y x) :
    ∃ (K : Set X) (hK : IsClopen K), Disjoint K Y ∧ ∀ x ∈ F,
      coeff T k (LocallyConstant.charFn k hK) * x = x ∧
        x * coeff T k (LocallyConstant.charFn k hK) = x := by
  classical
  obtain ⟨S, hSdef⟩ : ∃ S : ClopenCrossedProduct T k → Multiplicative ℤ → Set X,
      ∀ x g, S x g = {y | (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g) y ≠ 0} := ⟨_, fun _ _ => rfl⟩
  have hS : ∀ x g, IsClopen (S x g) := fun x g => by
    rw [hSdef]
    exact (((ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g)).isLocallyConstant.isClopen_fiber 0).compl
  have hmemS : ∀ x g y, (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g) y ≠ 0 → g ∈ SkewMonoidAlgebra.support x ∧ y ∈ S x g :=
    fun x g y hy => by
      refine ⟨SkewMonoidAlgebra.mem_support_iff.2 fun h0 => hy ?_, by rw [hSdef]; exact hy⟩
      rw [h0, map_zero, LocallyConstant.zero_apply]
  obtain ⟨K, hKdef⟩ : ∃ K : Set X,
      K = ⋃ x ∈ F, ⋃ g ∈ SkewMonoidAlgebra.support x, (S x g ∪ ⇑(T ^ (-toAdd g)) '' S x g) := ⟨_, rfl⟩
  have hK : IsClopen K := by
    rw [hKdef]
    exact isClopen_biUnion_finset fun x _ => isClopen_biUnion_finset fun g _ =>
      (hS x g).union (Dynamics.isClopen_zpow_image T _ (hS x g))
  have hsub : ∀ x ∈ F, ∀ g ∈ SkewMonoidAlgebra.support x, S x g ∪ ⇑(T ^ (-toAdd g)) '' S x g ⊆ K := by
    intro x hx g hg y hy
    rw [hKdef]
    exact Set.mem_iUnion₂.2 ⟨x, hx, Set.mem_iUnion₂.2 ⟨g, hg, hy⟩⟩
  refine ⟨K, hK, ?_, fun x hx => ⟨charFn_mul_eq_self T k hK fun g y hyK => ?_,
    mul_charFn_eq_self T k hK fun g y hyK => ?_⟩⟩
  · rw [Set.disjoint_left]
    intro y hyK hyY
    rw [hKdef] at hyK
    obtain ⟨x, hx, hy⟩ := Set.mem_iUnion₂.1 hyK
    obtain ⟨g, _, hy⟩ := Set.mem_iUnion₂.1 hy
    rcases hy with hy | ⟨q, hq, hqy⟩
    · rw [hSdef] at hy
      exact hy (hF x hx g y hyY)
    · have hqeq : q = (T ^ (toAdd g)) y := by
        rw [← hqy, Dynamics.zpow_apply_zpow_apply, add_neg_cancel, zpow_zero, Homeomorph.one_apply]
      rw [hSdef, hqeq] at hq
      exact hq (hF x hx g _ (hY _ y hyY))
  · by_contra hne
    obtain ⟨hg, hyS⟩ := hmemS x g y hne
    exact hyK (hsub x hx g hg (Or.inl hyS))
  · by_contra hne
    obtain ⟨hg, hyS⟩ := hmemS x g y hne
    exact hyK (hsub x hx g hg (Or.inr ⟨y, hyS, rfl⟩))

theorem isWandering_zpow_image {W : Set X} (hW : Dynamics.IsWandering T.toEquiv W) (j : ℤ) :
    Dynamics.IsWandering T.toEquiv (⇑(T ^ j) '' W) := by
  intro a b hab
  have himg : ∀ c : ℤ, ⇑(T.toEquiv ^ c) '' (⇑(T ^ j) '' W) = ⇑(T.toEquiv ^ (c + j)) '' W :=
    fun c => by
      rw [Set.image_image]
      refine Set.image_congr fun y _ => ?_
      rw [Dynamics.toEquiv_zpow_apply, Dynamics.toEquiv_zpow_apply, Dynamics.zpow_apply_zpow_apply]
  rw [himg, himg]
  exact hW (by omega : a + j ≠ b + j)

theorem isWandering_defect {P : Set X} (hTP : T '' P ⊆ P) :
    Dynamics.IsWandering T.toEquiv (P \ T '' P) := by
  have h := Dynamics.isWandering_diff_image (T := T.toEquiv) (P := P)
    (by rwa [Homeomorph.coe_toEquiv])
  rwa [Homeomorph.coe_toEquiv] at h

/-- **Wandering clopen cover** (tex 1673–1674): if the defects of compressed clopen sets cover
`X ∖ Y` by translates, a clopen `K ⊆ X ∖ Y` of compact `X` is covered by finitely many wandering
clopen sets. -/
theorem exists_wandering_clopen_cover [CompactSpace X]
    (hcover : ∀ x ∉ Y, ∃ (P : Set X) (_ : IsClopen P) (j : ℤ),
      T '' P ⊆ P ∧ x ∈ ⇑(T ^ j) '' (P \ T '' P))
    {K : Set X} (hK : IsClopen K) (hKY : Disjoint K Y) :
    ∃ (m : ℕ) (W : Fin m → Set X), (∀ i, IsClopen (W i)) ∧
      (∀ i, Dynamics.IsWandering T.toEquiv (W i)) ∧ K ⊆ ⋃ i, W i := by
  classical
  have hsel : ∀ x : K, ∃ V : Set X, IsClopen V ∧ Dynamics.IsWandering T.toEquiv V ∧ (x : X) ∈ V :=
    fun x => by
      obtain ⟨P, hP, j, hTP, hx⟩ := hcover x (Set.disjoint_left.1 hKY x.2)
      exact ⟨_, Dynamics.isClopen_zpow_image T j (hP.diff (isClopen_image T hP)),
        isWandering_zpow_image T (isWandering_defect T hTP) j, hx⟩
  choose V hVc hVw hxV using hsel
  obtain ⟨t, ht⟩ := hK.isClosed.isCompact.elim_finite_subcover V (fun x => (hVc x).isOpen)
    (fun x hx => Set.mem_iUnion.2 ⟨⟨x, hx⟩, hxV ⟨x, hx⟩⟩)
  refine ⟨t.card, fun i => V (t.equivFin.symm i).1, fun i => hVc _, fun i => hVw _,
    fun x hx => ?_⟩
  obtain ⟨p, hpt, hxp⟩ := Set.mem_iUnion₂.1 (ht hx)
  refine Set.mem_iUnion.2 ⟨t.equivFin ⟨p, hpt⟩, ?_⟩
  simpa using hxp

variable {T}

/-- A visit to `C` at a nonpositive time gives a positive return within `H` steps. -/
theorem exists_pos_return_of_visit {C : Set X} {H : ℕ}
    (hH : ∀ x ∈ C, ∃ h : ℕ, 0 < h ∧ h ≤ H ∧ (T ^ (h : ℤ)) x ∈ C) :
    ∀ (d : ℕ) (x : X), (T ^ (-(d : ℤ))) x ∈ C → ∃ h : ℕ, 0 < h ∧ h ≤ H ∧ (T ^ (h : ℤ)) x ∈ C := by
  intro d
  induction d using Nat.strong_induction_on with
  | _ d ih =>
    intro x hx
    obtain ⟨h₁, h1pos, h1le, h1C⟩ := hH _ hx
    rw [Dynamics.zpow_apply_zpow_apply] at h1C
    by_cases hd : d < h₁
    · refine ⟨h₁ - d, by omega, by omega, ?_⟩
      have hcast : (((h₁ - d : ℕ) : ℤ)) = (h₁ : ℤ) + -(d : ℤ) := by omega
      rw [hcast]
      exact h1C
    · refine ih (d - h₁) (by omega) x ?_
      have hcast : (-(((d - h₁ : ℕ) : ℤ))) = (h₁ : ℤ) + -(d : ℤ) := by omega
      rw [hcast]
      exact h1C

/-- **Uniform returns** (tex 1664, 1676): on compact `X`, a full clopen section with bounded
positive return times is reached from every point within `L` positive steps. -/
theorem exists_uniform_return [CompactSpace X] {C : Set X} (hC : IsClopen C)
    (hfull : ∀ x : X, ∃ j : ℤ, (T ^ j) x ∈ C) {H : ℕ}
    (hH : ∀ x ∈ C, ∃ h : ℕ, 0 < h ∧ h ≤ H ∧ (T ^ (h : ℤ)) x ∈ C) :
    ∃ L : ℕ, ∀ x : X, ∃ h : ℕ, 0 < h ∧ h ≤ L ∧ (T ^ (h : ℤ)) x ∈ C := by
  obtain ⟨t, ht⟩ := isCompact_univ.elim_finite_subcover (fun j : ℤ => ⇑(T ^ j) ⁻¹' C)
    (fun j => hC.isOpen.preimage (T ^ j).continuous)
    (fun x _ => Set.mem_iUnion.2 (hfull x))
  refine ⟨t.sup Int.toNat + H, fun x => ?_⟩
  obtain ⟨j, hjt, hj⟩ := Set.mem_iUnion₂.1 (ht (Set.mem_univ x))
  have hjle : j.toNat ≤ t.sup Int.toNat := Finset.le_sup hjt
  rcases lt_or_ge 0 j with hjpos | hjnon
  · have hcast : ((j.toNat : ℕ) : ℤ) = j := Int.toNat_of_nonneg hjpos.le
    refine ⟨j.toNat, by omega, by omega, ?_⟩
    rw [hcast]
    exact hj
  · obtain ⟨h, hpos, hle, hC'⟩ := exists_pos_return_of_visit hH (-j).toNat x (by
      have hcast : (-(((-j).toNat : ℕ) : ℤ)) = j := by omega
      rw [hcast]
      exact hj)
    exact ⟨h, hpos, by omega, hC'⟩
