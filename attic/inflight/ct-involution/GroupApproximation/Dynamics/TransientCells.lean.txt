/-!
# The cells of `lem:transient-matrices`

`non_mf_groups_exist.tex`, `\label{lem:transient-matrices}`, proof, tex 1492–1511:

> Express a prescribed finite subset of $I$ as linear combinations of characteristic functions of
> compact clopen partial shift graphs.  Include inverse partial shifts, and let $C$ be the union of
> their sources and ranges.  [...]  Choose a finite clopen partition of $C$ separating distinct
> points in each class [...].  Refine it according to the occupied atoms, the shift exponents to
> each level, and all original coefficient values.  On each cell the classes have a fixed size $h$
> and continuously enumerated clopen levels.

For a finite `F ⊆ I` this module produces the data consumed by ms-units' tower family and expansion
(`Dynamics/TransientTowerFamily`, `Dynamics/TransientTowerExpansion`):

* a clopen `C` off `Y = CR(T)` containing the sources and ranges of the coefficients of `F`
  (`exists_clopen_support`, `mem_of_charFn_mul_eq_self`, `zpow_mem_of_mul_charFn_eq_self`);
* the exponents `E` of `F` and the orbit bound `ExponentBound T C m`, from the wandering cover
  (`exists_wandering_clopen_cover`, `exponentBound_of_wanderingCover`);
* finitely many pairwise disjoint clopen cells inside representative cells, covering the
  representatives, with a point in each, on which every coefficient of `F` is constant at every
  level (`exists_cells`).

`exists_transientCells` packages them.

Construction note: the printed proof first separates the points of each class by a finite clopen
partition and orders its atoms; here the representative is the point of least exponent
(`baseSet`, theoremc-retire), which needs no separating partition.  The refinement by the class
and by the coefficient values at the levels is as printed.
-/

section Support

variable {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type*) [Ring k]

/-- From `1_K x = x`: a nonzero coefficient of `x` is supported in `K` ("sources"). -/
theorem mem_of_charFn_mul_eq_self {K : Set X} (hK : IsClopen K) {x : ClopenCrossedProduct T k}
    (h : coeff T k (LocallyConstant.charFn k hK) * x = x) (g : Multiplicative ℤ) (z : X)
    (hz : (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g) z ≠ 0) : z ∈ K := by
  by_contra hzK
  apply hz
  rw [← h, coeff_apply, Pestov91.CrossedProduct.C_apply, SkewMonoidAlgebra.coeff_single_one_mul,
    map_mul, LocallyConstant.mul_apply, RingEquiv.symm_apply_apply, LocallyConstant.coe_charFn,
    Set.indicator_of_notMem hzK, zero_mul]

/-- From `x 1_K = x`: a nonzero coefficient of `u^g` at `z` has `T^{-g} z ∈ K` ("ranges"). -/
theorem zpow_mem_of_mul_charFn_eq_self {K : Set X} (hK : IsClopen K)
    {x : ClopenCrossedProduct T k} (h : x * coeff T k (LocallyConstant.charFn k hK) = x)
    (g : Multiplicative ℤ) (z : X)
    (hz : (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g) z ≠ 0) :
    (T ^ (-toAdd g)) z ∈ K := by
  by_contra hzK
  apply hz
  rw [← h, coeff_apply, Pestov91.CrossedProduct.C_apply, SkewMonoidAlgebra.coeff_mul_single_one,
    map_mul, LocallyConstant.mul_apply, of_symm_smul_apply, RingEquiv.symm_apply_apply,
    LocallyConstant.coe_charFn, Set.indicator_of_notMem hzK, mul_zero]

end Support

section Cells

variable {X : Type*} [TopologicalSpace X]

/-- Finitely many locally constant functions are jointly locally constant. -/
theorem isLocallyConstant_pi {ι Z : Type*} [Finite ι] {f : ι → X → Z}
    (hf : ∀ i, IsLocallyConstant (f i)) : IsLocallyConstant fun y i => f i y :=
  (IsLocallyConstant.iff_eventually_eq _).2 fun x =>
    (Filter.eventually_all.2 fun i => (hf i).eventually_eq x).mono fun _ hy => funext hy

variable [CompactSpace X] {T : X ≃ₜ X} {C : Set X} {E : Finset ℤ} {m : ℕ}

/-- The finitely many exponents occurring in some class. -/
theorem exists_finset_classExponents (hC : IsClopen C) :
    ∃ D : Finset ℤ, ∀ y : X, reachableExponents T C E (m - 1) y ⊆ D := by
  classical
  have hfin := (isLocallyConstant_reachableExponents (T := T) (E := E) hC (m - 1)).range_finite
  exact ⟨hfin.toFinset.biUnion id, fun y d hd => Finset.mem_biUnion.2
    ⟨reachableExponents T C E (m - 1) y, hfin.mem_toFinset.2 ⟨y, rfl⟩, hd⟩⟩

/-- **The cells** (tex 1508–1510): refining the representative set by the class and by finitely
many locally constant values gives finitely many pairwise disjoint nonempty clopen cells inside
representative cells, covering the representatives, on which the values are constant. -/
theorem exists_cells (hC : IsClopen C) {ι Z : Type*} [Finite ι] (g : ι → X → Z)
    (hg : ∀ i, IsLocallyConstant (g i)) :
    ∃ (M : ℕ) (cell : Fin M → Set X) (S : Fin M → Finset ℤ),
      (∀ a, IsClopen (cell a)) ∧ (∀ a, cell a ⊆ baseCell T C E m (S a)) ∧
      (∀ a b, a ≠ b → Disjoint (cell a) (cell b)) ∧ (∀ a, (cell a).Nonempty) ∧
      (∀ y ∈ baseSet T C E m, ∃ a, y ∈ cell a) ∧
      ∀ a, ∀ y ∈ cell a, ∀ y' ∈ cell a, ∀ i, g i y = g i y' := by
  classical
  obtain ⟨φ, hφdef⟩ : ∃ φ : X → Finset ℤ × (ι → Z),
      φ = fun y => (reachableExponents T C E (m - 1) y, fun i => g i y) := ⟨_, rfl⟩
  have hφ : IsLocallyConstant φ := by
    rw [hφdef]
    exact (isLocallyConstant_reachableExponents hC (m - 1)).prodMk (isLocallyConstant_pi hg)
  have hfin := hφ.range_finite
  obtain ⟨V, hVdef⟩ : ∃ V : Finset (Finset ℤ × (ι → Z)),
      V = hfin.toFinset.filter fun v => (baseSet T C E m ∩ {y | φ y = v}).Nonempty := ⟨_, rfl⟩
  have hcellsub : ∀ v : Finset ℤ × (ι → Z),
      baseSet T C E m ∩ {y | φ y = v} ⊆ baseCell T C E m v.1 := by
    rintro v y ⟨hyB, hyv⟩
    refine ⟨hyB, ?_⟩
    have h1 : φ y = v := hyv
    rw [hφdef] at h1
    exact congrArg Prod.fst h1
  refine ⟨V.card, fun a => baseSet T C E m ∩ {y | φ y = (V.equivFin.symm a).1},
    fun a => (V.equivFin.symm a).1.1, fun a => (isClopen_baseSet hC).inter (hφ.isClopen_fiber _),
    fun a => hcellsub _, fun a b hab => ?_, fun a => ?_, fun y hy => ?_,
    fun a y hy y' hy' i => ?_⟩
  · rw [Set.disjoint_left]
    rintro y ⟨-, hya⟩ ⟨-, hyb⟩
    have hya' : φ y = (V.equivFin.symm a).1 := hya
    have hyb' : φ y = (V.equivFin.symm b).1 := hyb
    exact hab (V.equivFin.symm.injective (Subtype.ext (hya'.symm.trans hyb')))
  · have hmem := (V.equivFin.symm a).2
    rw [hVdef, Finset.mem_filter] at hmem
    exact hmem.2
  · have hv : φ y ∈ V := by
      rw [hVdef, Finset.mem_filter]
      exact ⟨hfin.mem_toFinset.2 ⟨y, rfl⟩, ⟨y, hy, rfl⟩⟩
    refine ⟨V.equivFin ⟨φ y, hv⟩, hy, ?_⟩
    show φ y = (V.equivFin.symm (V.equivFin ⟨φ y, hv⟩)).1
    rw [Equiv.symm_apply_apply]
  · have h1 : φ y = (V.equivFin.symm a).1 := hy.2
    have h2 : φ y' = (V.equivFin.symm a).1 := hy'.2
    have h3 := h1.trans h2.symm
    rw [hφdef] at h3
    exact congrFun (congrArg Prod.snd h3) i

end Cells

/-- **The cells of lem:transient-matrices** (tex 1492–1511) for a finite subset `F` of the transient
ideal, over the defect cover of lem:chain-core-models: the section `C`, the exponents `E`, the orbit
bound, the cells with representatives, the support of the coefficients of `F` in `C`, and the
constancy of every coefficient of `F` at every level of every cell. -/
theorem exists_transientCells (hcover : ChainCoreDefectCoverStatement) (X : Type) [MetricSpace X]
    [CompactSpace X] [TotallyDisconnectedSpace X] (T : X ≃ₜ X) (k : Type) [Ring k]
    (F : Finset (ClopenCrossedProduct T k))
    (hF : ∀ x ∈ F, x ∈ transientIdeal T k (Dynamics.image_chainRecurrentSet T)) :
    ∃ (C : Set X) (E : Finset ℤ) (m M : ℕ) (cell : Fin M → Set X) (S : Fin M → Finset ℤ)
      (hcell : ∀ a, IsClopen (cell a)) (rep : Fin M → X),
      IsClopen C ∧ Disjoint C (Dynamics.chainRecurrentSet ⇑T) ∧ ExponentBound T C m ∧
      (∀ a, cell a ⊆ baseCell T C E m (S a)) ∧
      (∀ a b, a ≠ b → Disjoint (cell a) (cell b)) ∧
      (∀ y ∈ baseSet T C E m, ∃ a, y ∈ cell a) ∧
      (∀ a, rep a ∈ cell a) ∧
      (∀ x ∈ F, ∀ g ∈ SkewMonoidAlgebra.support x, toAdd g ∈ E ∧
        ∀ z, (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g) z ≠ 0 →
          z ∈ C ∧ (T ^ (-toAdd g)) z ∈ C) ∧
      ∀ x ∈ F, ∀ g ∈ SkewMonoidAlgebra.support x, ∀ a, ∀ d ∈ S a, ∀ y ∈ cell a, ∀ y' ∈ cell a,
        (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g) ((T ^ d) y) =
          (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff x g) ((T ^ d) y') := by
  classical
  obtain ⟨K, hK, hKY, hKF⟩ := exists_clopen_support T k (fun j => zpow_mem_chainRecurrentSet T j) F
    fun x hx => (mem_coreTransientIdeal_iff T k x).1 (hF x hx)
  obtain ⟨m, W, -, hWw, hKW⟩ := exists_wandering_clopen_cover T
    (fun x hx => haveI : Nonempty X := ⟨x⟩; hcover X T x hx) hK hKY
  have hm : ExponentBound T K m := exponentBound_of_wanderingCover hWw hKW
  obtain ⟨E, hEdef⟩ : ∃ E : Finset ℤ,
      E = F.biUnion fun x => (SkewMonoidAlgebra.support x).image toAdd := ⟨_, rfl⟩
  obtain ⟨G, hGdef⟩ : ∃ G : Finset (Multiplicative ℤ),
      G = F.biUnion fun x => SkewMonoidAlgebra.support x := ⟨_, rfl⟩
  obtain ⟨D, hD⟩ := exists_finset_classExponents (T := T) (E := E) (m := m) hK
  obtain ⟨M, cell, S, hcell, hsub, hdisj, hne, hcov, hconst⟩ :=
    exists_cells (T := T) (E := E) (m := m) hK
      (ι := {x // x ∈ F} × {g // g ∈ G} × {d // d ∈ D})
      (fun p y => (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff p.1.1 p.2.1.1)
        ((T ^ p.2.2.1) y))
      fun p => ((ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff p.1.1 p.2.1.1)).isLocallyConstant
        |>.comp_continuous (T ^ p.2.2.1).continuous
  refine ⟨K, E, m, M, cell, S, hcell, fun a => (hne a).some, hK, hKY, hm, hsub, hdisj, hcov,
    fun a => (hne a).some_mem, fun x hx g hg => ⟨?_, fun z hz =>
      ⟨mem_of_charFn_mul_eq_self T k hK (hKF x hx).1 g z hz,
        zpow_mem_of_mul_charFn_eq_self T k hK (hKF x hx).2 g z hz⟩⟩,
    fun x hx g hg a d hd y hy y' hy' => ?_⟩
  · rw [hEdef]
    exact Finset.mem_biUnion.2 ⟨x, hx, Finset.mem_image_of_mem _ hg⟩
  · have hyS : reachableExponents T K E (m - 1) y = S a := (hsub a hy).2
    have hdD : d ∈ D := hD y (by rw [hyS]; exact hd)
    have hgG : g ∈ G := by
      rw [hGdef]
      exact Finset.mem_biUnion.2 ⟨x, hx, hg⟩
    exact hconst a y hy y' hy' (⟨x, hx⟩, ⟨g, hgG⟩, ⟨d, hdD⟩)

end ClopenCrossedProduct
