/-!
# The instances in the last statement of `cor:host`

`simple_kazhdan_sofic_group.tex` at origin/main 696c4b602, `cor:host` (tex l.324–327):

> So one such group contains every recursively presented finitely generated LEF group, for instance
> every finitely presented residually finite group and every `G_Y` whose language is recursive.

* `PrintedFinitelyPresentedRFInstance`: a finitely presented residually finite group is finitely
  generated, recursively presented and LEF. Finite generation comes from the surjection from a free
  group of finite rank, and LEF from `isLEF_of_residuallyFinite`. The recursive presentation is the
  piece `FinitelyPresentedRecursivePresentationStatement`.
* `PrintedRecursiveLanguageInstance`: for an infinite minimal subshift `Y` with recursive language,
  `G_Y` is finitely generated, recursively presented and LEF. Finite generation and LEF come from
  `thm:main`. `WP(G_Y) ≤_T L(Y)` (skf-degrees' `PrintedWordProblemReducesToLanguage`) and the
  generation of `G_Y` by the `e_ij(s)` (the piece `GenValueGeneratesStatement`) give a solvable word
  problem, and `nonempty_recursivePresentation_of_hasSolvableWordProblem` turns it into a recursive
  presentation.
* `PrintedRecursivelyPresentedLEFHostInstances`: the whole sentence, one host containing all three
  kinds of group, assembled over `PrintedRecursivelyPresentedLEFHost` and the two instances.
-/

namespace GroupApproximation
namespace SimpleKazhdanSofic

/-! ## Statements -/

/-- **Finitely presented residually finite groups** are finitely generated, recursively presented
and LEF (tex l.325–326). -/
def PrintedFinitelyPresentedRFInstance : Prop :=
  ∀ (Γ : Type) [Group Γ], Group.IsFinitelyPresented Γ → Group.ResiduallyFinite Γ →
    Group.FG Γ ∧ Nonempty (Higman.RecursivePresentation Γ) ∧ IsLEF Γ

/-- **`G_Y` with recursive language** is finitely generated, recursively presented and LEF
(tex l.326–327). -/
def PrintedRecursiveLanguageInstance : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Fintype A] [DecidableEq A]
    [Primcodable A] (S : Subshift A ℤ), Infinite S.carrier → IsMinimal S →
      Partrec (languageOracle S) →
        Group.FG (G S) ∧ Nonempty (Higman.RecursivePresentation (G S)) ∧ IsLEF (G S)

/-- **The last statement of `cor:host`, with its instances** (tex l.324–327). -/
def PrintedRecursivelyPresentedLEFHostInstances : Prop :=
  ∃ (E : Type) (_ : Group E), IsLEFHost E ∧
    (∀ (Γ : Type) [Group Γ], Group.FG Γ → Nonempty (Higman.RecursivePresentation Γ) →
      IsLEF Γ → ∃ f : Γ →* E, Function.Injective f) ∧
    (∀ (Γ : Type) [Group Γ], Group.IsFinitelyPresented Γ → Group.ResiduallyFinite Γ →
      ∃ f : Γ →* E, Function.Injective f) ∧
    (∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Fintype A] [DecidableEq A]
      [Primcodable A] (S : Subshift A ℤ), Infinite S.carrier → IsMinimal S →
        Partrec (languageOracle S) → ∃ f : G S →* E, Function.Injective f)

/-! ## Pieces -/

/-- **Finitely presented groups are recursively presented.** -/
def FinitelyPresentedRecursivePresentationStatement : Prop :=
  ∀ (Γ : Type) [Group Γ], Group.IsFinitelyPresented Γ →
    Nonempty (Higman.RecursivePresentation Γ)

/-- **`G_X` is generated by the `e_ij(s)`** with `s ∈ {u, u⁻¹} ∪ {e_a}` (tex l.97–99 and
eq:elementary). -/
def GenValueGeneratesStatement : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Fintype A] [DecidableEq A]
    (S : Subshift A ℤ), Subgroup.closure (Set.range (genValue S)) = ⊤

/-! ## Assemblies -/

/-- **The finitely presented residually finite instance** from its piece. -/
theorem printedFinitelyPresentedRFInstance_of_pieces
    (hfp : FinitelyPresentedRecursivePresentationStatement) : PrintedFinitelyPresentedRFInstance := by
  intro Γ _ hpres hrf
  obtain ⟨n, φ, hsurj, -⟩ := hpres.out
  haveI : Group.ResiduallyFinite Γ := hrf
  exact ⟨Group.fg_of_surjective hsurj, hfp Γ hpres, isLEF_of_residuallyFinite⟩

/-- A recursive language gives `G_Y` a solvable word problem. -/
theorem hasSolvableWordProblem_G_of_recursiveLanguage (hwp : PrintedWordProblemReducesToLanguage)
    (hgen : GenValueGeneratesStatement) {A : Type} [TopologicalSpace A] [DiscreteTopology A]
    [Fintype A] [DecidableEq A] [Primcodable A] (S : Subshift A ℤ)
    (hlang : Partrec (languageOracle S)) : HasSolvableWordProblem (G S) := by
  haveI : Finite (Coeff A) := Finite.of_equiv _ (Coeff.equivOption A).symm
  refine ⟨Gen A, inferInstance, inferInstance, genValue S, hgen A S, ?_⟩
  exact partrec_iff_forall_turingReducible.mpr fun o =>
    TuringReducible.trans (hwp A S) (partrec_iff_forall_turingReducible.mp hlang o)

/-- **The recursive-language instance** from `thm:main` and its pieces. -/
theorem printedRecursiveLanguageInstance_of_pieces (hmain : PrintedSimpleKazhdanSoficMain)
    (hwp : PrintedWordProblemReducesToLanguage) (hgen : GenValueGeneratesStatement) :
    PrintedRecursiveLanguageInstance := by
  intro A _ _ _ _ _ S hinf hmin hlang
  obtain ⟨-, hfg, -, -, -, hlef, -, -⟩ := hmain A S hinf hmin 3 le_rfl
  exact ⟨hfg, nonempty_recursivePresentation_of_hasSolvableWordProblem
    (hasSolvableWordProblem_G_of_recursiveLanguage hwp hgen S hlang), hlef⟩

/-- **The last statement of `cor:host`, with its instances**, from the host and the two
instances. -/
theorem printedRecursivelyPresentedLEFHostInstances_of_pieces
    (hhost : PrintedRecursivelyPresentedLEFHost) (hfpi : PrintedFinitelyPresentedRFInstance)
    (hlang : PrintedRecursiveLanguageInstance) : PrintedRecursivelyPresentedLEFHostInstances := by
  obtain ⟨E, _, hE, hemb⟩ := hhost
  refine ⟨E, inferInstance, hE, hemb, fun Γ _ hpres hrf => ?_,
    fun A _ _ _ _ _ S hinf hmin hl => ?_⟩
  · obtain ⟨hfg, hrec, hlef⟩ := hfpi Γ hpres hrf
    exact hemb Γ hfg hrec hlef
  · obtain ⟨hfg, hrec, hlef⟩ := hlang A S hinf hmin hl
    exact hemb (G S) hfg hrec hlef

/-- **The last statement of `cor:host`, with its instances**, over the lamplighter piece and the
instance pieces. -/
theorem printedRecursivelyPresentedLEFHostInstances_of_lamplighter
    (hlamp : LEFLamplighterStatement) (hfp : FinitelyPresentedRecursivePresentationStatement)
    (hmain : PrintedSimpleKazhdanSoficMain) (hwp : PrintedWordProblemReducesToLanguage)
    (hgen : GenValueGeneratesStatement) : PrintedRecursivelyPresentedLEFHostInstances :=
  printedRecursivelyPresentedLEFHostInstances_of_pieces
    (printedRecursivelyPresentedLEFHost_of_lamplighter hlamp)
    (printedFinitelyPresentedRFInstance_of_pieces hfp)
    (printedRecursiveLanguageInstance_of_pieces hmain hwp hgen)
