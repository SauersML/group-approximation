/-!
# Laurent form and products in `R` (simple_kazhdan_sofic_group.tex, "The ring and property (T)")

`simple_kazhdan_sofic_group.tex` at e80dcf20a, "The ring and property (T)":

> Write $e_U$ for the indicator of a clopen set $U\subseteq X$. Our conventions are $(Tx)_t=x_{t+1}$
> and $R=\LC(X,\F_2)\rtimes_T\Z=\bigoplus_{j\in\Z}\LC(X,\F_2)\,u^j$, $ufu^{-1}=f\circ T^{-1}$.
> So every element of $R$ is uniquely a finite sum $\sum_jf_ju^j$, the product is
> $(fu^i)(f'u^j)=f\,(f'\circ T^{-i})\,u^{i+j}$, and $ue_Uu^{-1}=e_{TU}$.

`R` is chain-core's `ClopenCrossedProduct T k` (a skew monoid algebra, `⊕_j LC(X,k) u^j`), with
`ufu⁻¹ = f ∘ T⁻¹` its `unit_mul_coeff_mul_inv`, and `ue_Uu⁻¹ = e_{TU}` is `unit_mul_eU_mul_inv`
(`RingPartitions`). For every homeomorphism `T` and ring `k`:

* `exists_sum_coeff_mul_unit_zpow` (chain-core): existence of the finite sum;
* `coeff_sum_coeff_mul_unit_zpow`, `eq_of_sum_coeff_mul_unit_zpow_eq`: uniqueness, since the
  coefficient of `u^j` in `∑_{i ∈ S} f_i u^i` is `f_j` for `j ∈ S`;
* `coeff_mul_unit_zpow_mul_coeff_mul_unit_zpow`: `(fu^i)(f'u^j) = f (f' ∘ T^{-i}) u^{i+j}`.
-/

namespace GroupApproximation

namespace SimpleKazhdanSofic

open ClopenCrossedProduct
open Multiplicative (ofAdd toAdd)

variable {Y : Type*} [TopologicalSpace Y] (T : Y ≃ₜ Y) (k : Type*) [Ring k]

theorem coeff_mul_unit_zpow_eq_single (f : LocallyConstant Y k) (j : ℤ) :
    coeff T k f * ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) =
      SkewMonoidAlgebra.single (ofAdd j) (ClopenCoeff.of T k f) := by
  rw [Pestov91.CrossedProduct.val_unit_zpow, coeff_apply, Pestov91.CrossedProduct.C_mul_single,
    mul_one]

/-- The coefficient of `u^j` in a finite Laurent sum `∑_{i ∈ S} f_i u^i` is `f_j`, for `j ∈ S`. -/
theorem coeff_sum_coeff_mul_unit_zpow (S : Finset ℤ) (f : ℤ → LocallyConstant Y k) {j : ℤ}
    (hj : j ∈ S) :
    SkewMonoidAlgebra.coeff (∑ i ∈ S, coeff T k (f i) *
        ((unit T k ^ i : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k)) (ofAdd j) =
      ClopenCoeff.of T k (f j) := by
  classical
  simp only [coeff_mul_unit_zpow_eq_single, SkewMonoidAlgebra.coeff_sum,
    SkewMonoidAlgebra.coeff_single_apply]
  rw [Finset.sum_eq_single_of_mem j hj fun i _ hij => by rw [if_neg]; exact fun h => hij (ofAdd.injective h)]
  rw [if_pos rfl]

/-- **Uniqueness of the Laurent form** (tex 93): two finite sums over the same exponents agree only
if their coefficients agree. -/
theorem eq_of_sum_coeff_mul_unit_zpow_eq (S : Finset ℤ) (f g : ℤ → LocallyConstant Y k)
    (h : ∑ i ∈ S, coeff T k (f i) *
        ((unit T k ^ i : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) =
      ∑ i ∈ S, coeff T k (g i) *
        ((unit T k ^ i : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k)) :
    ∀ j ∈ S, f j = g j := fun j hj => by
  have hc := congrArg (fun r => SkewMonoidAlgebra.coeff r (ofAdd j)) h
  simp only at hc
  rw [coeff_sum_coeff_mul_unit_zpow T k S f hj, coeff_sum_coeff_mul_unit_zpow T k S g hj] at hc
  exact (ClopenCoeff.of T k).injective hc

/-- **The product formula** (tex 94): `(fu^i)(f'u^j) = f (f' ∘ T^{-i}) u^{i+j}`. -/
theorem coeff_mul_unit_zpow_mul_coeff_mul_unit_zpow (f f' : LocallyConstant Y k) (i j : ℤ) :
    coeff T k f * ((unit T k ^ i : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) *
        (coeff T k f' * ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k)) =
      coeff T k (f * LocallyConstant.comap ⟨⇑(T ^ (-i)), (T ^ (-i)).continuous⟩ f') *
        ((unit T k ^ (i + j) : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) := by
  have hu := Pestov91.CrossedProduct.unit_zpow_mul_C (A := ClopenCoeff T k) i (ClopenCoeff.of T k f')
  rw [ClopenCoeff.smul_of] at hu
  calc coeff T k f * ((unit T k ^ i : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) *
        (coeff T k f' * ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k))
      = coeff T k f * (((unit T k ^ i : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) *
          coeff T k f') *
          ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) := by
        simp only [mul_assoc]
    _ = coeff T k f * (coeff T k
          (LocallyConstant.comap ⟨⇑(T ^ (-i)), (T ^ (-i)).continuous⟩ f') *
          ((unit T k ^ i : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k)) *
          ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) := by
        rw [coeff_apply, coeff_apply]
        exact congrArg (fun z => coeff T k f * z *
          ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k)) hu
    _ = _ := by
        rw [← mul_assoc, ← map_mul, mul_assoc, ← Units.val_mul, ← zpow_add]

end SimpleKazhdanSofic

end GroupApproximation
