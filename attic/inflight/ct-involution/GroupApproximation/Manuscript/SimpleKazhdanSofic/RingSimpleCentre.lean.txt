/-!
# `R` is simple and `Z(R) = F₂` (simple_kazhdan_sofic_group.tex, tex 102–113)

`simple_kazhdan_sofic_group.tex` at bf961c128, tex 102–113:

> We will also need that $R$ is simple and $Z(R)=\F_2$. These are standard groupoid-algebra facts
> [BCFS, ClarkEdie, Steinberg]; here is the direct argument. A nonzero two-sided ideal contains an
> element $r=\sum_j f_ju^j$ with $f_0\ne0$, after multiplication by a power of $u$. Choose a nonempty
> clopen $U$ on which $f_0=1$, disjoint from $T^jU$ for every nonzero exponent occurring in $r$.
> Then $e_Ure_U=e_U$. Finitely many translates of $U$ cover $X$, so the ideal contains
> $1=1-\prod_i(1-e_{T^{n_i}U})$. If $r$ is central, commuting with all clopen indicators forces
> $f_j=0$ for $j\ne0$: otherwise a point in the support of $f_j$ and its $T^{-j}$-translate can be
> separated by a clopen set. Commuting with $u$ then makes $f_0$ invariant, hence constant by
> minimality.

* `isSimpleRing`: the diagonal cut is `Pestov91.crossedProduct_one_mem` and
  `Pestov91.locallyConstant_cut_key`, instantiated by `Pestov91.isSimpleRing_of_minimal_free` with
  `τ j = T^{-j}`. It follows the printed route: multiply by a power of `u` to get a nonzero `f_0`, cut
  with the indicator of a clopen set on which `f_0` is a nonzero constant and whose translates are
  disjoint, then cover `X` by translates;
* `coeff_const_mem_center`, `mem_center_iff`: the central elements are the constants, along the
  printed route (clopen separation of `x` and `T^{-j}x`, then invariance and minimality);
* `center_eq_zero_one`: over `F₂`, `Z(R) = {0, 1}`.
-/

open ClopenCrossedProduct
open Multiplicative (ofAdd toAdd)

variable {Y : Type*} [TopologicalSpace Y] (T : Y ≃ₜ Y)

theorem dense_range_zpow_neg {y : Y} (h : Dense (Set.range fun j : ℤ => (T ^ j) y)) :
    Dense (Set.range fun j : ℤ => (T ^ (-j)) y) := by
  have hset : (Set.range fun j : ℤ => (T ^ (-j)) y) = Set.range fun j : ℤ => (T ^ j) y :=
    Set.ext fun z => ⟨fun ⟨j, hj⟩ => ⟨-j, hj⟩, fun ⟨j, hj⟩ => ⟨-j, by rw [neg_neg]; exact hj⟩⟩
  rw [hset]
  exact h

/-- **`R` is a simple ring** (tex 102–109), for a minimal system with no periodic points on a
compact totally separated nonempty space, over any field. -/
theorem isSimpleRing [CompactSpace Y] [TotallySeparatedSpace Y] [Nonempty Y]
    (hmin : IsMinimalSystem T) (hfree : ∀ j : ℤ, j ≠ 0 → ∀ y : Y, (T ^ j) y ≠ y)
    (k : Type*) [Field k] : IsSimpleRing (ClopenCrossedProduct T k) := by
  haveI : Nontrivial (ClopenCrossedProduct T k) := by
    refine ⟨⟨0, 1, fun h => ?_⟩⟩
    have h1 : (0 : LocallyConstant Y k) = 1 := coeff_injective T k (by rw [map_zero, map_one, h])
    obtain ⟨y⟩ := ‹Nonempty Y›
    exact zero_ne_one (LocallyConstant.congr_fun h1 y)
  refine Pestov91.isSimpleRing_of_minimal_free (coeff T k) (unit T k) (fun j => ⇑(T ^ (-j)))
    (fun j => (T ^ (-j)).continuous) (fun j hj y => hfree (-j) (neg_ne_zero.2 hj) y)
    (fun y => dense_range_zpow_neg T (hmin y))
    (fun j f => (ClopenCoeff.of T k).symm (ofAdd j • ClopenCoeff.of T k f)) (fun _ _ _ => rfl)
    (fun j a => ?_) (exists_sum_coeff_mul_unit_zpow T k)
  show ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) *
      Pestov91.CrossedProduct.C (ClopenCoeff.of T k a) =
    Pestov91.CrossedProduct.C (ClopenCoeff.of T k
      ((ClopenCoeff.of T k).symm (ofAdd j • ClopenCoeff.of T k a))) *
      ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k)
  rw [RingEquiv.apply_symm_apply]
  exact Pestov91.CrossedProduct.unit_zpow_mul_C j (ClopenCoeff.of T k a)

variable (k : Type*) [CommRing k]

/-- Constants are central. -/
theorem coeff_const_mem_center (c : k) :
    coeff T k (LocallyConstant.const Y c) ∈ Set.center (ClopenCrossedProduct T k) := by
  refine Semigroup.mem_center_iff.2 fun g => ?_
  obtain ⟨S, f, rfl⟩ := exists_sum_coeff_mul_unit_zpow T k g
  rw [Finset.sum_mul, Finset.mul_sum]
  refine Finset.sum_congr rfl fun j _ => ?_
  have hinv : ofAdd j • ClopenCoeff.of T k (LocallyConstant.const Y c) =
      ClopenCoeff.of T k (LocallyConstant.const Y c) := LocallyConstant.ext fun _ => rfl
  have hu : ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) *
      coeff T k (LocallyConstant.const Y c) =
      coeff T k (LocallyConstant.const Y c) *
        ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) := by
    have h := Pestov91.CrossedProduct.unit_zpow_mul_C j (ClopenCoeff.of T k (LocallyConstant.const Y c))
    rw [hinv] at h
    exact h
  calc coeff T k (f j) * ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) *
        coeff T k (LocallyConstant.const Y c)
      = coeff T k (f j) * coeff T k (LocallyConstant.const Y c) *
          ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k) := by
        rw [mul_assoc, hu, ← mul_assoc]
    _ = coeff T k (LocallyConstant.const Y c) * (coeff T k (f j) *
          ((unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k)) := by
        rw [← map_mul, mul_comm (f j), map_mul, mul_assoc]

/-- **The central elements of `R` are the constants** (tex 110–113). -/
theorem mem_center_iff [TotallySeparatedSpace Y] [Nonempty Y] (hmin : IsMinimalSystem T)
    (hfree : ∀ j : ℤ, j ≠ 0 → ∀ y : Y, (T ^ j) y ≠ y) (r : ClopenCrossedProduct T k) :
    r ∈ Set.center (ClopenCrossedProduct T k) ↔ ∃ c : k, r = coeff T k (LocallyConstant.const Y c) := by
  refine ⟨fun hr => ?_, fun ⟨c, hc⟩ => hc ▸ coeff_const_mem_center T k c⟩
  classical
  have hcomm : ∀ g : ClopenCrossedProduct T k, g * r = r * g := Semigroup.mem_center_iff.1 hr
  -- commuting with all clopen indicators forces `f_j = 0` for `j ≠ 0` (tex 110–112)
  have hzero : ∀ g : Multiplicative ℤ, g ≠ 1 → SkewMonoidAlgebra.coeff r g = 0 := by
    intro g hg
    refine (ClopenCoeff.of T k).symm.injective (LocallyConstant.ext fun x => ?_)
    rw [map_zero, LocallyConstant.zero_apply]
    have hj : -toAdd g ≠ 0 := neg_ne_zero.2 fun h => hg (by rw [← ofAdd_toAdd g, h, ofAdd_zero])
    obtain ⟨W, hW, hxW, hTxW⟩ := exists_isClopen_of_totally_separated (hfree (-toAdd g) hj x).symm
    have hL : (ClopenCoeff.of T k).symm
        (SkewMonoidAlgebra.coeff (coeff T k (LocallyConstant.charFn k hW) * r) g) x =
        (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff r g) x := by
      rw [coeff_apply, Pestov91.CrossedProduct.C_apply, SkewMonoidAlgebra.coeff_single_one_mul,
        map_mul, LocallyConstant.mul_apply, RingEquiv.symm_apply_apply, LocallyConstant.coe_charFn,
        Set.indicator_of_mem hxW, Pi.one_apply, one_mul]
    have hR : (ClopenCoeff.of T k).symm
        (SkewMonoidAlgebra.coeff (r * coeff T k (LocallyConstant.charFn k hW)) g) x = 0 := by
      rw [coeff_apply, Pestov91.CrossedProduct.C_apply, SkewMonoidAlgebra.coeff_mul_single_one,
        map_mul, LocallyConstant.mul_apply, of_symm_smul_apply, RingEquiv.symm_apply_apply,
        LocallyConstant.coe_charFn, Set.indicator_of_notMem hTxW, mul_zero]
    rw [hcomm] at hL
    exact hL.symm.trans hR
  -- so `r = f_0`
  have hr0 : r = coeff T k ((ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff r 1)) := by
    refine SkewMonoidAlgebra.ext fun g => ?_
    rw [coeff_apply, RingEquiv.apply_symm_apply, Pestov91.CrossedProduct.C_apply,
      SkewMonoidAlgebra.coeff_single_apply]
    by_cases hg : (1 : Multiplicative ℤ) = g
    · rw [if_pos hg, hg]
    · rw [if_neg hg, hzero g fun h => hg h.symm]
  obtain ⟨f0, hf0⟩ : ∃ f0 : LocallyConstant Y k,
      f0 = (ClopenCoeff.of T k).symm (SkewMonoidAlgebra.coeff r 1) := ⟨_, rfl⟩
  rw [← hf0] at hr0
  -- commuting with `u` makes `f_0` invariant (tex 112)
  have hu := hcomm (unit T k)
  rw [hr0, unit_mul_coeff, Units.mul_left_inj] at hu
  have hinv : ∀ y, f0 (T.symm y) = f0 y := fun y => LocallyConstant.congr_fun (coeff_injective T k hu) y
  have horbit : ∀ (j : ℤ) (y : Y), f0 ((T ^ j) y) = f0 y := by
    intro j
    induction j using Int.induction_on with
    | zero => intro y; rw [zpow_zero, Homeomorph.one_apply]
    | succ i ih =>
      intro y
      rw [zpow_add_one, Homeomorph.mul_apply, ih, ← hinv (T y), Homeomorph.symm_apply_apply]
    | pred i ih =>
      intro y
      rw [zpow_sub_one, Homeomorph.mul_apply, ih, Homeomorph.inv_apply, hinv]
  -- hence constant by minimality (tex 113)
  obtain ⟨y0⟩ := ‹Nonempty Y›
  refine ⟨f0 y0, hr0.trans (congrArg (coeff T k) (LocallyConstant.ext fun z => ?_))⟩
  have hsub : Set.range (fun j : ℤ => (T ^ j) y0) ⊆ {z | f0 z = f0 y0} := by
    rintro _ ⟨j, rfl⟩
    exact horbit j y0
  have huniv : (Set.univ : Set Y) ⊆ {z | f0 z = f0 y0} := by
    rw [← (hmin y0).closure_eq]
    exact closure_minimal hsub (f0.isLocallyConstant.isClopen_fiber (f0 y0)).isClosed
  exact huniv (Set.mem_univ z)

/-- **`Z(R) = F₂`** (tex 102, 110–113): over `F₂` the central elements of `R` are `0` and `1`. -/
theorem mem_center_iff_eq_zero_or_one [TotallySeparatedSpace Y] [Nonempty Y]
    (hmin : IsMinimalSystem T) (hfree : ∀ j : ℤ, j ≠ 0 → ∀ y : Y, (T ^ j) y ≠ y)
    (r : ClopenCrossedProduct T (ZMod 2)) :
    r ∈ Set.center (ClopenCrossedProduct T (ZMod 2)) ↔ r = 0 ∨ r = 1 := by
  rw [mem_center_iff T (ZMod 2) hmin hfree]
  constructor
  · rintro ⟨c, rfl⟩
    have hc : c = 0 ∨ c = 1 := by fin_cases c <;> simp
    rcases hc with rfl | rfl
    · exact Or.inl (by rw [show LocallyConstant.const Y (0 : ZMod 2) = 0 from rfl, map_zero])
    · exact Or.inr (by rw [show LocallyConstant.const Y (1 : ZMod 2) = 1 from rfl, map_one])
  · rintro (rfl | rfl)
    · exact ⟨0, by rw [show LocallyConstant.const Y (0 : ZMod 2) = 0 from rfl, map_zero]⟩
    · exact ⟨1, by rw [show LocallyConstant.const Y (1 : ZMod 2) = 1 from rfl, map_one]⟩
