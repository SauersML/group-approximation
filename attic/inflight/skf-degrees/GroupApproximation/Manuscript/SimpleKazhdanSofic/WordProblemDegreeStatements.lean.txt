open SymbolicDynamics.FullShift Set

section Subshift

variable {A : Type} [TopologicalSpace A]

/-- The shift homeomorphism of a subshift, `(T x)_n = x_{n+1}` (tex l.71). -/
def subshiftHomeo (S : Subshift A ℤ) : S.carrier ≃ₜ S.carrier where
  toFun x := ⟨shift 1 x.1, S.mapsTo 1 x.2⟩
  invFun x := ⟨shift (-1) x.1, S.mapsTo (-1) x.2⟩
  left_inv x := by
    apply Subtype.ext
    funext n
    show shift (-1) (shift 1 x.1) n = x.1 n
    simp only [shift_apply]
    congr 1
    omega
  right_inv x := by
    apply Subtype.ext
    funext n
    show shift 1 (shift (-1) x.1) n = x.1 n
    simp only [shift_apply]
    congr 1
    omega
  continuous_toFun :=
    ((continuous_shift 1 : Continuous (shift (1 : ℤ) : (ℤ → A) → ℤ → A)).comp
      continuous_subtype_val).subtype_mk fun x => S.mapsTo 1 x.2
  continuous_invFun :=
    ((continuous_shift (-1) : Continuous (shift (-1 : ℤ) : (ℤ → A) → ℤ → A)).comp
      continuous_subtype_val).subtype_mk fun x => S.mapsTo (-1) x.2

/-- A subshift is minimal if it has no nonempty proper closed shift-invariant subset (tex l.45). -/
def IsMinimal (S : Subshift A ℤ) : Prop :=
  ∀ C : Set (ℤ → A), IsClosed C → C ⊆ S.carrier → (∀ g : ℤ, MapsTo (shift g) C C) →
    C = ∅ ∨ C = S.carrier

/-- The language `L(X)`: the finite words occurring in `X` (tex l.239). -/
def language (S : Subshift A ℤ) : Set (List A) :=
  {v | ∃ x ∈ S.carrier, ∀ t : Fin v.length, x ((t : ℕ) : ℤ) = v.get t}

/-- The letter indicator `e_a` of the cylinder `{x : x_0 = a}` (tex l.84). -/
noncomputable def letterIndicator [DiscreteTopology A] [DecidableEq A] (S : Subshift A ℤ) (a : A) :
    LocallyConstant S.carrier (ZMod 2) where
  toFun x := if x.1 0 = a then 1 else 0
  isLocallyConstant :=
    ((IsLocallyConstant.iff_continuous (fun x : S.carrier => x.1 0)).2
      ((continuous_apply (0 : ℤ)).comp continuous_subtype_val)).comp
      fun b => if b = a then (1 : ZMod 2) else 0

end Subshift

/-- The coefficient symbols `{1, u, u⁻¹} ∪ {e_a : a ∈ A}` of the generating set (tex l.94–95). -/
inductive Coeff (A : Type) where
  | one
  | u
  | uInv
  | letter (a : A)

/-- `Coeff A` as a nested option, for the encoding of words. -/
def Coeff.equivOption (A : Type) : Coeff A ≃ Option (Option (Option A)) where
  toFun
    | .one => none
    | .u => some none
    | .uInv => some (some none)
    | .letter a => some (some (some a))
  invFun
    | none => .one
    | some none => .u
    | some (some none) => .uInv
    | some (some (some a)) => .letter a
  left_inv c := by cases c <;> rfl
  right_inv o := by
    match o with
    | none => rfl
    | some none => rfl
    | some (some none) => rfl
    | some (some (some _)) => rfl

instance (A : Type) [Encodable A] : Encodable (Coeff A) :=
  Encodable.ofEquiv _ (Coeff.equivOption A)

/-- The generator symbols: an off-diagonal position of `M₃` and a coefficient symbol. -/
abbrev Gen (A : Type) : Type := {p : Fin 3 × Fin 3 // p.1 ≠ p.2} × Coeff A

section Group

variable {A : Type} [TopologicalSpace A] [DiscreteTopology A] [DecidableEq A]

/-- The ring `R = LC(X, F₂) ⋊_T ℤ` (tex l.73). -/
abbrev R (S : Subshift A ℤ) : Type := ClopenCrossedProduct (subshiftHomeo S) (ZMod 2)

/-- The group `G_X = EL₃(R)` (tex l.46 and l.86). -/
abbrev G (S : Subshift A ℤ) : Type := ↥(elementaryGroup (Fin 3) (R S))

/-- The value of a coefficient symbol in `R`. -/
noncomputable def coeffValue (S : Subshift A ℤ) : Coeff A → R S
  | .one => 1
  | .u => (ClopenCrossedProduct.unit (subshiftHomeo S) (ZMod 2) : R S)
  | .uInv => ((ClopenCrossedProduct.unit (subshiftHomeo S) (ZMod 2))⁻¹ : (R S)ˣ)
  | .letter a => ClopenCrossedProduct.coeff (subshiftHomeo S) (ZMod 2) (letterIndicator S a)

/-- The generating family `e_ij(s)` of `G_X` (tex l.94–95). -/
noncomputable def genValue (S : Subshift A ℤ) (g : Gen A) : G S :=
  ⟨elementaryUnit g.1.1.1 g.1.1.2 g.1.2 (coeffValue S g.2), elementaryUnit_mem _ _ _ _⟩

end Group

section Oracles

/-- The value of a word `w` in the generators `s`; a letter `(i, false)` stands for `(s i)⁻¹`. -/
def wordValue {ι H : Type*} [Group H] (s : ι → H) : List (ι × Bool) → H
  | [] => 1
  | (i, b) :: w => (if b then s i else (s i)⁻¹) * wordValue s w

open Classical in
/-- The characteristic function of the word problem in the generators `s`, on codes of words. -/
noncomputable def wordProblemOracle {ι H : Type*} [Group H] [Encodable ι] (s : ι → H) :
    ℕ →. ℕ :=
  fun n => Part.some
    (if ∃ w : List (ι × Bool), Encodable.decode n = some w ∧ wordValue s w = 1 then 1 else 0)

open Classical in
/-- The characteristic function of a set of naturals. -/
noncomputable def charOracle (B : Set ℕ) : ℕ →. ℕ :=
  fun n => Part.some (if n ∈ B then 1 else 0)

open Classical in
/-- The characteristic function of the language `L(X)`, on codes of words. -/
noncomputable def languageOracle {A : Type} [TopologicalSpace A] [Encodable A]
    (S : Subshift A ℤ) : ℕ →. ℕ :=
  fun n => Part.some
    (if ∃ v : List A, Encodable.decode n = some v ∧ v ∈ language S then 1 else 0)

/-- The binary digits of a real `α`, the oracle through which `α` computes and is computed. -/
noncomputable def slopeOracle (α : ℝ) : ℕ →. ℕ :=
  fun n => Part.some (Nat.floor (2 ^ n * α) % 2)

end Oracles

/-- **tex l.238–239.**  For every infinite minimal subshift, the word problem of `G_X` in the
generators `e_ij(s)` has the Turing degree of the language `L(X)`. -/
def PrintedWordProblemDegreeEqLanguage : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Fintype A] [DecidableEq A]
    [Encodable A] (S : Subshift A ℤ), Infinite S.carrier → IsMinimal S →
      TuringEquivalent (wordProblemOracle (genValue S)) (languageOracle S)

/-- **tex l.259–261.**  The word-problem degree of a finitely generated group does not depend on
the finite generating family, and is an isomorphism invariant. -/
def PrintedWordProblemDegreeIsomorphismInvariant : Prop :=
  ∀ (H K : Type) [Group H] [Group K] (ι κ : Type) [Encodable ι] [Encodable κ] [Finite ι]
    [Finite κ] (s : ι → H) (t : κ → K), Subgroup.closure (range s) = ⊤ →
      Subgroup.closure (range t) = ⊤ → Nonempty (H ≃* K) →
        TuringEquivalent (wordProblemOracle s) (wordProblemOracle t)

/-- **tex l.256–258.**  Every Turing degree of a set of naturals has an irrational representative
in `(0, 1)`. -/
def PrintedIrrationalRepresentative : Prop :=
  ∀ B : Set ℕ, ∃ α : ℝ, Irrational α ∧ α ∈ Ioo (0 : ℝ) 1 ∧
    TuringEquivalent (slopeOracle α) (charOracle B)

/-- **tex l.259.**  There are continuum many Turing degrees of sets of naturals. -/
def PrintedContinuumManyDegrees : Prop :=
  ∃ F : Set (Set ℕ), Cardinal.mk F = Cardinal.continuum ∧
    ∀ B ∈ F, ∀ C ∈ F, TuringEquivalent (charOracle B) (charOracle C) → B = C

/-- **The corollary, tex l.231–235.**  Every Turing degree of a set of naturals is the
word-problem degree of `G_X` for an infinite minimal subshift `X` over `{0, 1}`, and `G_X` is an
infinite simple Kazhdan LEF group; there are continuum many isomorphism classes of these groups. -/
def PrintedEveryWordProblemDegree : Prop :=
  (∀ B : Set ℕ, ∃ S : Subshift Bool ℤ, Infinite S.carrier ∧ IsMinimal S ∧
      Infinite (G S) ∧ IsSimpleGroup (G S) ∧ HasKazhdanPropertyT.{0, 0} (G S) ∧ IsLEF (G S) ∧
      TuringEquivalent (wordProblemOracle (genValue S)) (charOracle B)) ∧
    ∃ F : Set (Subshift Bool ℤ), Cardinal.mk F = Cardinal.continuum ∧
      (∀ S ∈ F, Infinite (G S) ∧ IsSimpleGroup (G S) ∧ HasKazhdanPropertyT.{0, 0} (G S) ∧
        IsLEF (G S)) ∧
      ∀ S ∈ F, ∀ S' ∈ F, Nonempty (G S ≃* G S') → S = S'

end SimpleKazhdanSofic
