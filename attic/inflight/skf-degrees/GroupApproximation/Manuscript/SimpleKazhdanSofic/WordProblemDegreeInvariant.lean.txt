/-!
# The word-problem degree is an isomorphism invariant

Section "Word problems" of `simple_kazhdan_sofic_group.tex` (origin tip e80dcf20a), proof, last
sentence:

> ... and the degree of the word problem is an isomorphism invariant.

This module proves `PrintedWordProblemDegreeIsomorphismInvariant`. Each generator of the second
family is the value of a word in the first, since the first family generates; substituting these
words is a primitive recursive translation of words, so one word problem reduces to the other. An
isomorphism transports a generating family without changing its word problem.
-/

namespace GroupApproximation
namespace SimpleKazhdanSofic

open Encodable Set

variable {ι κ H : Type*} [Group H]

/-- Every element of a group generated by `s` is the value of a word in `s`. -/
theorem exists_wordValue_eq {s : ι → H} (hs : Subgroup.closure (range s) = ⊤) (x : H) :
    ∃ w : List (ι × Bool), wordValue s w = x := by
  have hx : x ∈ Subgroup.closure (range s) := by
    rw [hs]
    exact Subgroup.mem_top x
  induction hx using Subgroup.closure_induction with
  | mem x hx =>
    obtain ⟨i, rfl⟩ := hx
    exact ⟨[(i, true)], by simp [wordValue_cons, wordValue_nil]⟩
  | one => exact ⟨[], rfl⟩
  | mul x y _ _ hx hy =>
    obtain ⟨a, rfl⟩ := hx
    obtain ⟨b, rfl⟩ := hy
    exact ⟨a ++ b, wordValue_append s a b⟩
  | inv x _ hx =>
    obtain ⟨a, rfl⟩ := hx
    exact ⟨wordInv a, wordValue_wordInv s a⟩

/-- Substitution of a table of words for the generators. -/
def translate (table : κ → List (ι × Bool)) (w : List (κ × Bool)) : List (ι × Bool) :=
  w.flatMap fun x => if x.2 then table x.1 else wordInv (table x.1)

theorem wordValue_translate (s : ι → H) (t : κ → H) (table : κ → List (ι × Bool))
    (htable : ∀ k, wordValue s (table k) = t k) (w : List (κ × Bool)) :
    wordValue s (translate table w) = wordValue t w := by
  induction w with
  | nil => rfl
  | cons x w ih =>
    obtain ⟨k, b⟩ := x
    simp only [translate, List.flatMap_cons] at ih ⊢
    rw [wordValue_append, ih, wordValue_cons]
    cases b <;> simp [htable, wordValue_wordInv]

/-- **Change of generators.**  If `s` generates `H` and `κ` is finite, the word problem of any
family `t : κ → H` reduces to the word problem of `s`. -/
theorem turingReducible_wordProblemOracle_of_generates [Primcodable ι] [Primcodable κ] [Finite κ]
    {s : ι → H} (hs : Subgroup.closure (range s) = ⊤) (t : κ → H) :
    TuringReducible (wordProblemOracle t) (wordProblemOracle s) := by
  classical
  choose table htable using fun k => exists_wordValue_eq hs (t k)
  have htr : Primrec (translate table) := by
    have hg : Primrec₂ fun (_ : List (κ × Bool)) (x : κ × Bool) =>
        if x.2 then table x.1 else wordInv (table x.1) :=
      (Primrec.dom_finite fun x : κ × Bool =>
        if x.2 then table x.1 else wordInv (table x.1)).comp Primrec.snd
    exact Primrec.list_flatMap Primrec.id hg
  refine turingReducible_of_query
    (fun n => if ∃ w : List (ι × Bool), decode n = some w ∧ wordValue s w = 1 then 1 else 0)
    (fun _ => rfl)
    (fun n => encode (translate table ((decode (α := List (κ × Bool)) n).getD [])))
    (fun n m => cond (decode (α := List (κ × Bool)) n).isSome m 0)
    (Primrec.encode.comp (htr.comp (Primrec.option_getD.comp Primrec.decode (Primrec.const []))))
    (Primrec.cond (Primrec.option_isSome.comp (Primrec.decode.comp Primrec.fst)) Primrec.snd
      (Primrec.const 0))
    fun n => ?_
  cases hd : (decode n : Option (List (κ × Bool))) with
  | none => simp [wordProblemOracle, hd]
  | some w =>
    by_cases hw : wordValue t w = 1
    · simp [wordProblemOracle, hd, wordValue_translate s t table htable, hw]
    · simp [wordProblemOracle, hd, wordValue_translate s t table htable, hw]

/-- **Proof, last sentence.**  The word-problem degree of a finitely generated group does not
depend on the finite generating family, and is an isomorphism invariant. -/
theorem printedWordProblemDegreeIsomorphismInvariant :
    PrintedWordProblemDegreeIsomorphismInvariant := by
  intro H K _ _ ι κ _ _ _ _ s t hs ht ⟨φ⟩
  have e₁ : wordProblemOracle t = wordProblemOracle (φ.symm.toMonoidHom ∘ t) :=
    (wordProblemOracle_comp_injective t φ.symm.toMonoidHom φ.symm.injective).symm
  have e₂ : wordProblemOracle s = wordProblemOracle (φ.toMonoidHom ∘ s) :=
    (wordProblemOracle_comp_injective s φ.toMonoidHom φ.injective).symm
  refine ⟨?_, ?_⟩
  · rw [e₂]
    exact turingReducible_wordProblemOracle_of_generates ht _
  · rw [e₁]
    exact turingReducible_wordProblemOracle_of_generates hs _

#audit_closed_axioms printedWordProblemDegreeIsomorphismInvariant

end SimpleKazhdanSofic
end GroupApproximation
