/-!
# Rank two: why the compression criterion gives no obstruction

`non_mf_groups_exist.tex`, the remark after `prop:torsion-defect-ring`
(tex lines 1146--1154):

> Every countable subgroup of `GL_2` over a field has the Haagerup
> property~[GHW, Theorem 4], and the kernel of
> `GL_2(J) → GL_2(F_2[z,z^{-1}])` is locally finite, so a Kazhdan subgroup of
> `GL_2(J)` has finite image in `GL_2(F_2(z))` and a locally finite subgroup of
> finite index, and is finite.  A finite subgroup `L` of a group `G` admits no
> proper compression, since `uLu^{-1} ≤ L` forces `uLu^{-1} = L`; then
> `uC_G(L)u^{-1} = C_G(L)` and `𝔇_G(L) = 1`, so
> Theorem~`thm:compression-criterion` gives no obstruction at rank two.

This module proves the group theory of that argument at the printed
generality, independently of the ring `J` and of the Guentner--Higson--Weinberger
theorem:

* a finitely generated group is countable, and a finitely generated locally
  finite group is finite;
* a group with property `(T)` and a locally finite subgroup of finite index is
  finite;
* **the printed deduction.**  If `φ : G → K` has locally finite kernel and every
  countable subgroup of `K` has the Haagerup property, then every Kazhdan
  subgroup `L ≤ G` has finite image `φ(L)` and is finite: `φ(L)` is Kazhdan and
  Haagerup, hence finite (`Kazhdan/HaagerupKazhdanFinite`, through Delorme), so
  the kernel of `L → φ(L)` has finite index in `L` and is locally finite;
* the printed finite-subgroup sentences: a one-sided compressor of a finite
  subgroup is two-sided, it conjugates the centralizer onto itself, and the
  printed defect is trivial.
-/

namespace GroupApproximation

/-! ## Finitely generated groups -/

/-- A finitely generated group is countable: it is a quotient of the free group
on a finite generating set. -/
theorem countable_of_groupFG {G : Type*} [Group G] (h : Group.FG G) : Countable G := by
  classical
  obtain ⟨S, hS⟩ := Group.fg_def.mp h
  have hsurj : Function.Surjective (FreeGroup.lift (fun x : (S : Set G) ↦ (x : G))) := by
    rw [← MonoidHom.range_eq_top, FreeGroup.range_lift_eq_closure, Subtype.range_coe]
    exact hS
  haveI : Countable (FreeGroup (S : Set G)) := Quotient.countable
  exact hsurj.countable

/-- A finitely generated locally finite group is finite. -/
theorem finite_of_groupFG_of_isLocallyFiniteGroup {G : Type*} [Group G] (h : Group.FG G)
    (hlf : IsLocallyFiniteGroup G) : Finite G := by
  obtain ⟨S, hS⟩ := Group.fg_def.mp h
  obtain ⟨H, _, hSH⟩ := hlf S
  have hle : Subgroup.closure (S : Set G) ≤ H :=
    (Subgroup.closure_le H).mpr fun x hx ↦ hSH x hx
  have htop : ∀ g : G, g ∈ H := by
    intro g
    apply hle
    rw [hS]
    exact Subgroup.mem_top g
  exact Finite.of_surjective (fun x : H ↦ (x : G)) fun g ↦ ⟨⟨g, htop g⟩, rfl⟩

/-- **A Kazhdan group with a locally finite subgroup of finite index is
finite.**  Property `(T)` makes the group finitely generated, Schreier's lemma
makes the subgroup finitely generated, local finiteness makes it finite, and
finite index does the rest. -/
theorem finite_of_hasKazhdanPropertyT_of_finiteIndex_locallyFinite {G : Type u} [Group G]
    (hT : HasKazhdanPropertyT.{u, u} G) (N : Subgroup G) [N.FiniteIndex]
    (hN : IsLocallyFiniteGroup N) : Finite G := by
  haveI : Group.FG G := KazhdanFiniteGeneration.fg_of_hasKazhdanPropertyT G hT
  haveI : Group.FG N := inferInstance
  haveI : Finite N := finite_of_groupFG_of_isLocallyFiniteGroup inferInstance hN
  have hcard : Nat.card G ≠ 0 := by
    rw [← N.card_mul_index]
    exact mul_ne_zero Nat.card_pos.ne' Subgroup.FiniteIndex.index_ne_zero
  exact Nat.finite_of_card_ne_zero hcard

/-! ## The printed deduction -/

/-- The kernel of `L → φ(L)` embeds in the kernel of `φ`. -/
def kerSubgroupMapToKer {G K : Type*} [Group G] [Group K] (φ : G →* K) (L : Subgroup G) :
    (φ.subgroupMap L).ker →* φ.ker where
  toFun x := ⟨((x : L) : G), by
    have hx : φ.subgroupMap L (x : L) = 1 := MonoidHom.mem_ker.mp x.2
    have hval := congrArg (fun y : L.map φ ↦ (y : K)) hx
    exact MonoidHom.mem_ker.mpr hval⟩
  map_one' := rfl
  map_mul' _ _ := rfl

theorem kerSubgroupMapToKer_injective {G K : Type*} [Group G] [Group K] (φ : G →* K)
    (L : Subgroup G) : Function.Injective (kerSubgroupMapToKer φ L) :=
  fun _ _ h ↦ Subtype.ext (Subtype.ext (by simpa using congrArg Subtype.val h))

/-- **The printed deduction at rank two, in its group-theoretic form.**  Let
`φ : G → K` have locally finite kernel, and let every countable subgroup of `K`
have the Haagerup property.  Then every Kazhdan subgroup `L ≤ G` has finite
image `φ(L)` and a locally finite subgroup of finite index, and is finite. -/
theorem finite_image_and_finite_of_kazhdan {G K : Type u} [Group G] [Group K]
    (φ : G →* K) (hker : IsLocallyFiniteGroup φ.ker)
    (hHaag : ∀ M : Subgroup K, Countable M → HasHaagerupProperty.{u, u} M)
    (L : Subgroup G) (hL : HasKazhdanPropertyT.{u, u} L) :
    Finite (L.map φ) ∧ (φ.subgroupMap L).ker.FiniteIndex ∧
      IsLocallyFiniteGroup (φ.subgroupMap L).ker ∧ Finite L := by
  have hTM : HasKazhdanPropertyT.{u, u} (L.map φ) :=
    HasKazhdanPropertyT.of_surjective (φ.subgroupMap L) (φ.subgroupMap_surjective L) hL
  haveI : Group.FG (L.map φ) := KazhdanFiniteGeneration.fg_of_hasKazhdanPropertyT _ hTM
  haveI hfinM : Finite (L.map φ) :=
    finite_of_hasKazhdanPropertyT_of_hasHaagerupProperty hTM
      (hHaag _ (countable_of_groupFG inferInstance))
  haveI : Finite (φ.subgroupMap L).range :=
    Finite.of_injective (fun x : (φ.subgroupMap L).range ↦ (x : L.map φ)) Subtype.val_injective
  haveI hfi : (φ.subgroupMap L).ker.FiniteIndex := inferInstance
  have hlf : IsLocallyFiniteGroup (φ.subgroupMap L).ker :=
    hker.comap (kerSubgroupMapToKer φ L) (kerSubgroupMapToKer_injective φ L)
  exact ⟨hfinM, hfi, hlf,
    finite_of_hasKazhdanPropertyT_of_finiteIndex_locallyFinite hL _ hlf⟩

/-! ## The printed finite-subgroup sentences -/

section FiniteSubgroup

variable {G : Type*} [Group G]

/-- A compressor of a finite subgroup conjugates the centralizer onto itself:
`u C_G(L) u⁻¹ = C_G(L)`. -/
theorem conj_centralizer_eq_of_mem_compressionSet_of_finite (L : Subgroup G) [Finite L]
    {u : G} (hu : u ∈ compressionSet L) :
    (fun x ↦ u * x * u⁻¹) '' (Subgroup.centralizer (L : Set G) : Set G) =
      (Subgroup.centralizer (L : Set G) : Set G) := by
  ext y
  constructor
  · rintro ⟨c, hc, rfl⟩
    rw [SetLike.mem_coe, Subgroup.mem_centralizer_iff]
    intro ell hell
    have hc' : ∀ delta ∈ L, Commute c delta := fun delta hdelta ↦
      ((Subgroup.mem_centralizer_iff.mp hc) delta hdelta).symm
    exact (commute_conj_of_mem_compressionSet_of_finite hu hc' hell).eq.symm
  · intro hy
    refine ⟨u⁻¹ * y * u, ?_, by group⟩
    rw [SetLike.mem_coe, Subgroup.mem_centralizer_iff]
    intro ell hell
    have h1 : u * ell * u⁻¹ ∈ L := hu ell hell
    have h2 : u * ell * u⁻¹ * y = y * (u * ell * u⁻¹) :=
      (Subgroup.mem_centralizer_iff.mp hy) _ h1
    calc ell * (u⁻¹ * y * u) = u⁻¹ * ((u * ell * u⁻¹) * y) * u := by group
      _ = u⁻¹ * (y * (u * ell * u⁻¹)) * u := by rw [h2]
      _ = u⁻¹ * y * u * ell := by group

end FiniteSubgroup

/-- **Printed (tex 1151--1153).**  "A finite subgroup `L` of a group `G` admits
no proper compression, since `uLu^{-1} ≤ L` forces `uLu^{-1} = L`; then
`uC_G(L)u^{-1} = C_G(L)` and `𝔇_G(L) = 1`." -/
theorem manuscriptSentence_finiteSubgroupAdmitsNoProperCompression :
    ∀ (G : Type u) [Group G] (L : Subgroup G) [Finite L],
      (∀ u ∈ compressionSet L,
        (fun x ↦ u * x * u⁻¹) '' (L : Set G) = (L : Set G) ∧
          (fun x ↦ u * x * u⁻¹) '' (Subgroup.centralizer (L : Set G) : Set G) =
            (Subgroup.centralizer (L : Set G) : Set G)) ∧
        printedDefect L = ⊥ := by
  intro G _ L _
  exact ⟨fun u hu ↦ ⟨conj_image_eq_of_mem_compressionSet_of_finite hu,
      conj_centralizer_eq_of_mem_compressionSet_of_finite L hu⟩,
    printedDefect_eq_bot_of_finite L⟩

/-- **Printed (tex 1146--1151), group-theoretic form, closed.**  For every
homomorphism with locally finite kernel into a group whose countable subgroups
all have the Haagerup property, every Kazhdan subgroup of the source has finite
image, a locally finite subgroup of finite index, and is finite. -/
theorem manuscriptSentence_kazhdanSubgroupFinite :
    ∀ (G K : Type u) [Group G] [Group K] (φ : G →* K),
      IsLocallyFiniteGroup φ.ker →
      (∀ M : Subgroup K, Countable M → HasHaagerupProperty.{u, u} M) →
      ∀ L : Subgroup G, HasKazhdanPropertyT.{u, u} L →
        Finite (L.map φ) ∧ (φ.subgroupMap L).ker.FiniteIndex ∧
          IsLocallyFiniteGroup (φ.subgroupMap L).ker ∧ Finite L :=
  fun _ _ _ _ φ hker hHaag L hL ↦ finite_image_and_finite_of_kazhdan φ hker hHaag L hL
