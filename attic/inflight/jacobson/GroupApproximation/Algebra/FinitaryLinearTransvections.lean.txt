import GroupApproximation.Algebra.FinitaryLinearBinaryPower

/-!
# Over `F_2` every finitary automorphism is a product of transvections

`non_mf_groups_exist.tex`, the remark after `prop:torsion-defect-ring`
(tex lines 1134--1136):

> Over `F_2` every element of `GL_fs(V^n)` is a product of transvections …

For a basis `B` of an `F_2`-vector space, the *basis transvection* `τ_{a,b}`
(`a ≠ b`) is `x ↦ x + (coefficient of B b in x) • B a`; it is an involution.
The block homomorphism sends the elementary matrix `e_{ab}(1)` to `τ_{a,b}`
(`blockHom_elementaryUnit_one`), every invertible matrix over `F_2` is
elementary, and every finitary automorphism is a block matrix, so the finitary
group is generated by the basis transvections
(`finitary_eq_closure_basisTransvections`).
-/

namespace GroupApproximation
namespace FinitaryLinear

open Matrix

universe v w

variable {M : Type v} [AddCommGroup M] [Module (ZMod 2) M] {β : Type w}
  (B : Module.Basis β (ZMod 2) M)

/-- The basis transvection `τ_{a,b} : x ↦ x + (coefficient of B b in x) • B a`. -/
noncomputable def basisTransvection {a b : β} (h : a ≠ b) : M ≃ₗ[ZMod 2] M :=
  binaryTransvectionEquiv B (b₀ := b) (b := a) h

theorem basisTransvection_apply {a b : β} (h : a ≠ b) (x : M) :
    basisTransvection B h x = x + B.coord b x • B a := rfl

/-- A basis transvection is an involution. -/
theorem basisTransvection_mul_self {a b : β} (h : a ≠ b) :
    basisTransvection B h * basisTransvection B h = 1 := by
  refine LinearEquiv.ext fun x ↦ ?_
  have hid := congrArg (fun f : M →ₗ[ZMod 2] M ↦ f x) (binaryTransvection_mul_self B (b₀ := b) h)
  simpa [basisTransvection, binaryTransvectionEquiv] using hid

/-- The set of basis transvections. -/
def basisTransvections : Set (M ≃ₗ[ZMod 2] M) :=
  {g | ∃ (a b : β) (h : a ≠ b), basisTransvection B h = g}

theorem basisTransvection_mem_finitary {a b : β} (h : a ≠ b) :
    basisTransvection B h ∈ finitary B :=
  binaryTransvectionEquiv_mem_finitary B h

section Blocks

variable [DecidableEq β] (T : Finset β)

/-- **The block image of `e_{ab}(1)` is the basis transvection `τ_{a,b}`.** -/
theorem blockHom_elementaryUnit_one {i j : T} (hij : i ≠ j) :
    blockHom B T (elementaryUnit i j hij 1) =
      basisTransvection B (Subtype.coe_injective.ne hij) := by
  classical
  refine LinearEquiv.ext fun x ↦ ?_
  have key : blockLin B T ((elementaryUnit i j hij (1 : ZMod 2) : (Matrix T T (ZMod 2))ˣ) :
      Matrix T T (ZMod 2)) = (basisTransvection B (Subtype.coe_injective.ne hij) : M →ₗ[ZMod 2] M) := by
    refine B.ext fun c ↦ ?_
    rw [LinearEquiv.coe_coe, basisTransvection_apply]
    by_cases hc : c ∈ T
    · rw [blockLin_basis_mem B T _ hc]
      show ∑ a : T, (1 + Matrix.single i j (1 : ZMod 2)) a ⟨c, hc⟩ • B a = _
      simp only [Matrix.add_apply, add_smul, Finset.sum_add_distrib]
      rw [Finset.sum_eq_single ⟨c, hc⟩ (fun a _ ha ↦ by rw [Matrix.one_apply_ne ha, zero_smul])
        (by simp), Matrix.one_apply_eq, one_smul]
      congr 1
      rw [Finset.sum_eq_single i (fun a _ ha ↦ by simp [Matrix.single_apply, Ne.symm ha])
        (by simp)]
      by_cases hjc : j = ⟨c, hc⟩
      · subst hjc
        rw [Matrix.single_apply_same, one_smul, coord_basis_self, one_smul]
      · rw [coord_basis_of_ne B (b := c) (c := (j : β)) (fun h ↦ hjc (Subtype.ext h.symm))]
        simp [Matrix.single_apply, hjc]
    · rw [blockLin_basis_not_mem B T _ hc,
        coord_basis_of_ne B (b := c) (c := (j : β)) (fun h ↦ hc (h ▸ j.2)), zero_smul, add_zero]
  exact congrArg (fun f : M →ₗ[ZMod 2] M ↦ f x) key

/-- Every block matrix over `F_2` lies in the subgroup generated by the basis
transvections. -/
theorem blockHom_mem_closure (A : (Matrix T T (ZMod 2))ˣ) :
    blockHom B T A ∈ Subgroup.closure (basisTransvections B) := by
  have hA : A ∈ elementaryGroup T (ZMod 2) := by
    rw [elementaryGroup_zmodTwo_eq_top]
    exact Subgroup.mem_top A
  have hle : elementaryGroup T (ZMod 2) ≤
      (Subgroup.closure (basisTransvections B)).comap (blockHom B T) := by
    refine (Subgroup.closure_le _).mpr ?_
    rintro _ ⟨i, j, hij, c, rfl⟩
    have hc : c = 0 ∨ c = 1 := by
      have : ∀ x : ZMod 2, x = 0 ∨ x = 1 := by decide
      exact this c
    rcases hc with rfl | rfl
    · rw [elementaryUnit_zero]
      exact Subgroup.one_mem _
    · show blockHom B T (elementaryUnit i j hij 1) ∈ Subgroup.closure (basisTransvections B)
      rw [blockHom_elementaryUnit_one B T hij]
      exact Subgroup.subset_closure ⟨i, j, _, rfl⟩
  exact hle hA

end Blocks

/-- **Over `F_2` the finitary group is generated by the basis transvections.** -/
theorem finitary_eq_closure_basisTransvections :
    finitary B = Subgroup.closure (basisTransvections B) := by
  classical
  refine le_antisymm (fun g hg ↦ ?_) ((Subgroup.closure_le _).mpr ?_)
  · obtain ⟨T, hT⟩ := (mem_finitary B).mp hg
    obtain ⟨A, rfl⟩ := supportedOn_le_range B T hT
    exact blockHom_mem_closure B T A
  · rintro _ ⟨a, b, h, rfl⟩
    exact basisTransvection_mem_finitary B h

/-- **Printed (tex 1134--1136).**  Over `F_2` every element of `GL_fs(V^n)` is a
product of transvections: `GL_fs(V^n)` is the subgroup generated by the basis
transvections of `V^n`, each of which is an involution. -/
theorem manuscriptSentence_binaryGLfsProductOfTransvections :
    ∀ n : ℕ, binaryGLfs n = Subgroup.closure (basisTransvections (binaryPowerBasis n)) ∧
      ∀ (a b : Σ _ : Fin n, ℕ) (h : a ≠ b),
        basisTransvection (binaryPowerBasis n) h * basisTransvection (binaryPowerBasis n) h = 1 :=
  fun _ ↦ ⟨finitary_eq_closure_basisTransvections _,
    fun _ _ h ↦ basisTransvection_mul_self _ h⟩

end FinitaryLinear
end GroupApproximation
