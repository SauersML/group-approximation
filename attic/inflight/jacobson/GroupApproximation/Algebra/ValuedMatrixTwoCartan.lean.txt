import GroupApproximation.Algebra.ValuedMatrixTwo

/-!
# Cartan decomposition and the unique lower neighbour

Continuation of `Algebra/ValuedMatrixTwo` (the Bruhat--Tits tree step of the
rank-two remark after `prop:torsion-defect-ring`, tex lines 1146--1150).

Fix a uniformizer `π` (`v π = 1`) and put `D n = diag(1, π^n)`, `P = diag(π, 1)`.

* `dispInt (D n) = n`, `D 0 = 1`, `D (n+1) = D n · D 1`.
* **Cartan decomposition.**  Every `g ∈ GL_2(K)` is `s₁ · D n · s₂` with
  `s₁, s₂ ∈ S` and `n = dispInt g`: move an entry of least valuation to the corner,
  scale it to `1`, clear its row and column with integral elementary matrices, and
  split the remaining determinant as a unit times `π^n`.
* **Unique lower neighbour.**  If `dispInt a = 1`, `n ≥ 1` and
  `dispInt (D n · a) + 1 = n`, then `a = P · b` with `b ∈ S`.
-/

namespace GroupApproximation
namespace ValuedMatrixTwo

open Matrix
open Classical

universe u

variable {K : Type u} [Field K] (v : AddValuation K (WithTop ℤ))

/-- The matrix of `mkOfDetNeZero A h` is `A`. -/
theorem coe_mkOfDetNeZero (A : Matrix (Fin 2) (Fin 2) K) (h : A.det ≠ 0) :
    ((Matrix.GeneralLinearGroup.mkOfDetNeZero A h : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) = A :=
  rfl

/-- A matrix with integral entries and determinant of valuation zero lies in `S`. -/
theorem mem_stab_of {g : GL (Fin 2) K} (hg : EntriesGe v (g : Matrix (Fin 2) (Fin 2) K) 0)
    (hdet : vz v (g : Matrix (Fin 2) (Fin 2) K).det = 0) : g ∈ stab v := by
  have h1 := le_entryMin v hg
  have h2 := dispInt_nonneg v g
  rw [mem_stab]
  unfold dispInt at h2 ⊢
  omega

theorem vz_pow {x : K} (hx : x ≠ 0) (n : ℕ) : vz v (x ^ n) = n * vz v x := by
  induction n with
  | zero => rw [pow_zero, vz_one]; simp
  | succ n ih =>
      rw [pow_succ, vz_mul v (pow_ne_zero n hx) hx, ih]
      push_cast
      ring

/-- The entry `(0,0)` of value `vz v x` bounds `μ` from above. -/
theorem entryMin_le_of_entry {g : GL (Fin 2) K} {i j : Fin 2}
    (hij : (g : Matrix (Fin 2) (Fin 2) K) i j ≠ 0) :
    entryMin v g ≤ vz v ((g : Matrix (Fin 2) (Fin 2) K) i j) :=
  Finset.min'_le _ _
    (Finset.mem_image.mpr ⟨(i, j), Finset.mem_filter.mpr ⟨Finset.mem_univ _, hij⟩, rfl⟩)

/-- A copy of a lemma of the base module under a new name: the `(T)`-free
entry bound read from `S`-membership. -/
theorem vz_det_eq_of_mem_stab {s : GL (Fin 2) K} (hs : s ∈ stab v) :
    vz v (s : Matrix (Fin 2) (Fin 2) K).det = 2 * entryMin v s := by
  have hz := (mem_stab v).mp hs
  unfold dispInt at hz
  omega

/-- A permuted copy of `g` whose corner entry has least valuation, by integral
permutation matrices. -/
theorem exists_corner_min (g : GL (Fin 2) K) :
    ∃ a ∈ stab v, EntriesGe v (a : Matrix (Fin 2) (Fin 2) K) 0 ∧
      ∃ b ∈ stab v, EntriesGe v (b : Matrix (Fin 2) (Fin 2) K) 0 ∧
        ((a * g * b : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 0 ≠ 0 ∧
        vz v (((a * g * b : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 0) = entryMin v g := by
  classical
  obtain ⟨p, hp, hpv⟩ :=
    Finset.mem_image.mp (Finset.min'_mem (entryVals v g) (entryVals_nonempty v g))
  have hpne : (g : Matrix (Fin 2) (Fin 2) K) p.1 p.2 ≠ 0 := (Finset.mem_filter.mp hp).2
  let w : GL (Fin 2) K := Matrix.GeneralLinearGroup.mkOfDetNeZero !![0, 1; 1, 0]
    (by simp [Matrix.det_fin_two])
  have hwI : EntriesGe v (w : Matrix (Fin 2) (Fin 2) K) 0 := by
    intro i j hij
    fin_cases i <;> fin_cases j <;> simp [w] at hij ⊢ <;> rw [vz_one]
  have hw : w ∈ stab v := mem_stab_of v hwI (by
    simp only [w, coe_mkOfDetNeZero, Matrix.det_fin_two_of, mul_zero,
      one_mul, zero_sub]
    rw [vz_neg v one_ne_zero, vz_one])
  have h1I : EntriesGe v ((1 : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 := by
    rw [Units.val_one]
    exact entriesGe_one
  have h1 := (stab v).one_mem
  have key : ∀ a b : GL (Fin 2) K, a ∈ stab v → EntriesGe v (a : Matrix (Fin 2) (Fin 2) K) 0 →
      b ∈ stab v → EntriesGe v (b : Matrix (Fin 2) (Fin 2) K) 0 →
      ((a * g * b : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 0 =
        (g : Matrix (Fin 2) (Fin 2) K) p.1 p.2 →
      ∃ a ∈ stab v, EntriesGe v (a : Matrix (Fin 2) (Fin 2) K) 0 ∧
        ∃ b ∈ stab v, EntriesGe v (b : Matrix (Fin 2) (Fin 2) K) 0 ∧
          ((a * g * b : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 0 ≠ 0 ∧
          vz v (((a * g * b : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 0) = entryMin v g :=
    fun a b ha haI hb hbI hab ↦ ⟨a, ha, haI, b, hb, hbI, by rw [hab]; exact hpne,
      by rw [hab]; exact hpv⟩
  obtain ⟨i, j⟩ := p
  fin_cases i <;> fin_cases j
  · exact key 1 1 h1 h1I h1 h1I (by simp)
  · exact key 1 w h1 h1I hw hwI (by simp [w, Matrix.mul_apply, Fin.sum_univ_two])
  · exact key w 1 hw hwI h1 h1I (by simp [w, Matrix.mul_apply, Fin.sum_univ_two])
  · exact key w w hw hwI hw hwI
      (by simp [w, Matrix.mul_apply, Fin.sum_univ_two, Matrix.vecMul, dotProduct])

variable {π : K} (hπ : v π = 1)

include hπ in
theorem pi_ne_zero : π ≠ 0 := by
  intro h
  rw [h, AddValuation.map_zero] at hπ
  simp at hπ

include hπ in
theorem vz_pi : vz v π = 1 := by
  have h := coe_vz v (pi_ne_zero v hπ)
  rw [hπ] at h
  exact_mod_cast h

/-- `D n = diag(1, π^n)`. -/
noncomputable def diagPow (n : ℕ) : GL (Fin 2) K :=
  Matrix.GeneralLinearGroup.mkOfDetNeZero !![1, 0; 0, π ^ n]
    (by simp [Matrix.det_fin_two, pow_ne_zero n (pi_ne_zero v hπ)])

theorem diagPow_val (n : ℕ) :
    ((diagPow v hπ n : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) = !![1, 0; 0, π ^ n] := rfl

theorem vz_det_diagPow (n : ℕ) :
    vz v ((diagPow v hπ n : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K).det = n := by
  rw [diagPow_val, Matrix.det_fin_two_of, one_mul, mul_zero, sub_zero,
    vz_pow v (pi_ne_zero v hπ), vz_pi v hπ, mul_one]

theorem dispInt_diagPow (n : ℕ) : dispInt v (diagPow v hπ n) = n := by
  have hπ0 := pi_ne_zero v hπ
  have hge : EntriesGe v ((diagPow v hπ n : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 := by
    intro i j hij
    rw [diagPow_val] at hij ⊢
    fin_cases i <;> fin_cases j <;> simp at hij ⊢
    · rw [vz_one]
    · rw [vz_pow v hπ0, vz_pi v hπ]
      positivity
  have hle : entryMin v (diagPow v hπ n) ≤ 0 := by
    have h00 : ((diagPow v hπ n : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 0 ≠ 0 := by
      simp [diagPow_val]
    have := entryMin_le_of_entry v h00
    rwa [diagPow_val, show (!![1, 0; 0, π ^ n] : Matrix (Fin 2) (Fin 2) K) 0 0 = 1 by simp,
      vz_one] at this
  have hmin := le_entryMin v hge
  unfold dispInt
  rw [vz_det_diagPow]
  omega

theorem diagPow_zero : diagPow v hπ 0 = 1 := by
  refine Matrix.GeneralLinearGroup.ext fun i j ↦ ?_
  rw [diagPow_val, Units.val_one]
  fin_cases i <;> fin_cases j <;> simp

theorem diagPow_succ (n : ℕ) : diagPow v hπ (n + 1) = diagPow v hπ n * diagPow v hπ 1 := by
  refine Matrix.GeneralLinearGroup.ext fun i j ↦ ?_
  rw [Units.val_mul, diagPow_val, diagPow_val, diagPow_val]
  fin_cases i <;> fin_cases j <;> simp [Matrix.mul_apply, Fin.sum_univ_two, pow_succ]

/-- `P = diag(π, 1)`. -/
noncomputable def diagPiOne : GL (Fin 2) K :=
  Matrix.GeneralLinearGroup.mkOfDetNeZero !![π, 0; 0, 1]
    (by simp [Matrix.det_fin_two, pi_ne_zero v hπ])

/-! ## The Cartan decomposition -/

/-- **Cartan decomposition.**  `g = s₁ · D n · s₂` with `s₁, s₂ ∈ S` and
`n = dispInt g`. -/
theorem cartan (g : GL (Fin 2) K) :
    ∃ s₁ ∈ stab v, ∃ s₂ ∈ stab v, g = s₁ * diagPow v hπ (dispInt v g).toNat * s₂ := by
  obtain ⟨a, ha, haI, b, hb, hbI, hne, hmin⟩ := exists_corner_min v g
  have hπ0 := pi_ne_zero v hπ
  -- the corner-normalized copy
  have hge : EntriesGe v ((a * g * b : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) (entryMin v g) := by
    rw [Units.val_mul, Units.val_mul]
    exact ((haI.mul (entriesGe_entryMin v g)).mul hbI).mono (by omega)
  set c : K := (((a * g * b : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 0)⁻¹ with hc
  have hc0 : c ≠ 0 := inv_ne_zero hne
  let sc : GL (Fin 2) K := Matrix.GeneralLinearGroup.mkOfDetNeZero (c • 1)
    (by simp [hc0])
  have hsc : sc ∈ stab v := by
    have hgeS : EntriesGe v ((sc : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) (vz v c) := by
      have := (entriesGe_one (v := v)).smul hc0
      simpa [sc] using this
    have hminS := le_entryMin v hgeS
    have hnn := dispInt_nonneg v sc
    have hdetS : vz v ((sc : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K).det = 2 * vz v c := by
      simp only [sc, coe_mkOfDetNeZero, Matrix.det_smul, Fintype.card_fin,
        Matrix.det_one, mul_one]
      rw [vz_pow v hc0]
      push_cast
      ring
    rw [mem_stab]
    unfold dispInt at hnn ⊢
    rw [hdetS] at hnn ⊢
    omega
  set h' : GL (Fin 2) K := sc * (a * g * b) with hh'
  have hh'M : ((h' : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) =
      c • ((a * g * b : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) := by
    rw [hh', Units.val_mul]
    simp only [sc, coe_mkOfDetNeZero, Matrix.smul_mul, one_mul]
  have h'00 : ((h' : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 0 = 1 := by
    rw [hh'M, Matrix.smul_apply, smul_eq_mul, hc, inv_mul_cancel₀ hne]
  have hge' : EntriesGe v ((h' : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 := by
    rw [hh'M]
    refine (hge.smul hc0).mono ?_
    rw [hc, vz_inv v hne, hmin]
    omega
  have hdisp' : dispInt v h' = dispInt v g := by
    rw [hh', dispInt_stab_mul v hsc, dispInt_mul_stab v hb, dispInt_stab_mul v ha]
  have hmin' : entryMin v h' = 0 := by
    refine le_antisymm ?_ (le_entryMin v hge')
    have := entryMin_le_of_entry v (i := 0) (j := 0) (g := h') (by rw [h'00]; exact one_ne_zero)
    rwa [h'00, vz_one] at this
  set d : K := ((h' : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K).det with hd
  have hd0 : d ≠ 0 := det_ne_zero h'
  have hvd : vz v d = dispInt v g := by
    have h2 := hdisp'
    unfold dispInt at h2 ⊢
    rw [hmin'] at h2
    rw [hd]
    omega
  set n : ℕ := (dispInt v g).toNat with hn
  have hnz : (n : ℤ) = dispInt v g := Int.toNat_of_nonneg (dispInt_nonneg v g)
  -- the integral elementary matrices clearing the corner's row and column
  set p : K := ((h' : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 1 0 with hpdef
  set q : K := ((h' : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 1 with hqdef
  let e₁ : GL (Fin 2) K := Matrix.GeneralLinearGroup.mkOfDetNeZero !![1, 0; -p, 1]
    (by simp [Matrix.det_fin_two])
  let e₂ : GL (Fin 2) K := Matrix.GeneralLinearGroup.mkOfDetNeZero !![1, -q; 0, 1]
    (by simp [Matrix.det_fin_two])
  have he₁ : e₁ ∈ stab v := by
    refine mem_stab_of v ?_ ?_
    · intro i j hij
      fin_cases i <;> fin_cases j <;> simp [e₁] at hij ⊢
      · rw [vz_one]
      · rw [vz_neg v hij]
        exact hge' 1 0 hij
      · rw [vz_one]
    · simp [e₁, Matrix.det_fin_two, vz_one]
  have he₂ : e₂ ∈ stab v := by
    refine mem_stab_of v ?_ ?_
    · intro i j hij
      fin_cases i <;> fin_cases j <;> simp [e₂] at hij ⊢
      · rw [vz_one]
      · rw [vz_neg v hij]
        exact hge' 0 1 hij
      · rw [vz_one]
    · simp [e₂, Matrix.det_fin_two, vz_one]
  -- the diagonal factor
  set u : K := d * (π ^ n)⁻¹ with hu
  have hπn : π ^ n ≠ 0 := pow_ne_zero n hπ0
  have hu0 : u ≠ 0 := mul_ne_zero hd0 (inv_ne_zero hπn)
  have hvu : vz v u = 0 := by
    rw [hu, vz_mul v hd0 (inv_ne_zero hπn), vz_inv v hπn, vz_pow v hπ0, vz_pi v hπ, hvd, ← hnz]
    ring
  let du : GL (Fin 2) K := Matrix.GeneralLinearGroup.mkOfDetNeZero !![1, 0; 0, u]
    (by simp [Matrix.det_fin_two, hu0])
  have hdu : du ∈ stab v := by
    refine mem_stab_of v ?_ ?_
    · intro i j hij
      fin_cases i <;> fin_cases j <;> simp [du] at hij ⊢
      · rw [vz_one]
      · rw [hvu]
    · simp [du, Matrix.det_fin_two, hvu]
  have heq : e₁ * h' * e₂ = du * diagPow v hπ n := by
    refine Matrix.GeneralLinearGroup.ext fun i j ↦ ?_
    rw [Units.val_mul (e₁ * h') e₂, Units.val_mul e₁ h', Units.val_mul du (diagPow v hπ n),
      diagPow_val]
    have hdet2 : d = ((h' : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 1 1 - q * p := by
      rw [hd, Matrix.det_fin_two, h'00, one_mul]
    fin_cases i <;> fin_cases j <;>
      simp [e₁, e₂, du, Matrix.mul_apply, Fin.sum_univ_two, Matrix.vecMul, dotProduct, h'00,
        ← hpdef, ← hqdef, hu, inv_mul_cancel₀ hπn, mul_assoc] <;>
      first
      | ring
      | (rw [hdet2]; ring)
  refine ⟨a⁻¹ * sc⁻¹ * e₁⁻¹ * du, ?_, e₂⁻¹ * b⁻¹, ?_, ?_⟩
  · exact (stab v).mul_mem ((stab v).mul_mem ((stab v).mul_mem ((stab v).inv_mem ha)
      ((stab v).inv_mem hsc)) ((stab v).inv_mem he₁)) hdu
  · exact (stab v).mul_mem ((stab v).inv_mem he₂) ((stab v).inv_mem hb)
  · calc g = a⁻¹ * sc⁻¹ * e₁⁻¹ * (e₁ * (sc * (a * g * b)) * e₂) * e₂⁻¹ * b⁻¹ := by group
      _ = a⁻¹ * sc⁻¹ * e₁⁻¹ * (du * diagPow v hπ n) * e₂⁻¹ * b⁻¹ := by rw [← hh', heq]
      _ = a⁻¹ * sc⁻¹ * e₁⁻¹ * du * diagPow v hπ n * (e₂⁻¹ * b⁻¹) := by group

/-! ## The unique lower neighbour -/

/-- **Unique lower neighbour.**  If `dispInt a = 1`, `n ≥ 1` and
`dispInt (D n · a) + 1 = n`, then `a ∈ P · S`. -/
theorem lower_neighbour {a : GL (Fin 2) K} (ha : dispInt v a = 1) {n : ℕ}
    (hlow : dispInt v (diagPow v hπ n * a) + 1 = n) :
    ∃ b ∈ stab v, a = diagPiOne v hπ * b := by
  have hπ0 := pi_ne_zero v hπ
  have hdetDa : vz v ((diagPow v hπ n * a : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K).det =
      n + vz v (a : Matrix (Fin 2) (Fin 2) K).det := by
    rw [vz_det_mul, vz_det_diagPow]
  have hrow : ∀ j, ((diagPow v hπ n * a : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 j =
      (a : Matrix (Fin 2) (Fin 2) K) 0 j := by
    intro j
    rw [Units.val_mul, diagPow_val]
    simp [Matrix.mul_apply, Fin.sum_univ_two]
  have hnum : entryMin v (diagPow v hπ n * a) = entryMin v a + 1 := by
    unfold dispInt at ha hlow
    rw [hdetDa] at hlow
    omega
  let Pinv : GL (Fin 2) K := Matrix.GeneralLinearGroup.mkOfDetNeZero !![π⁻¹, 0; 0, 1]
    (by simp [Matrix.det_fin_two, hπ0])
  refine ⟨Pinv * a, ?_, ?_⟩
  · have hge : EntriesGe v ((Pinv * a : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K)
        (entryMin v a) := by
      intro i j hij
      fin_cases i
      · have hval : ((Pinv * a : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 0 j =
            π⁻¹ * (a : Matrix (Fin 2) (Fin 2) K) 0 j := by
          rw [Units.val_mul]
          simp [Pinv, Matrix.mul_apply, Fin.sum_univ_two]
        simp only [Fin.zero_eta] at hij ⊢
        rw [hval] at hij ⊢
        have hM : (a : Matrix (Fin 2) (Fin 2) K) 0 j ≠ 0 := right_ne_zero_of_mul hij
        rw [vz_mul v (inv_ne_zero hπ0) hM, vz_inv v hπ0, vz_pi v hπ]
        have hD := entriesGe_entryMin v (diagPow v hπ n * a) 0 j (by rw [hrow]; exact hM)
        rw [hrow, hnum] at hD
        omega
      · have hval : ((Pinv * a : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K) 1 j =
            (a : Matrix (Fin 2) (Fin 2) K) 1 j := by
          rw [Units.val_mul]
          simp [Pinv, Matrix.mul_apply, Fin.sum_univ_two]
        simp only [Fin.mk_one] at hij ⊢
        rw [hval] at hij ⊢
        exact entriesGe_entryMin v a 1 j hij
    have hdetP : vz v ((Pinv : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K).det = -1 := by
      simp only [Pinv, coe_mkOfDetNeZero, Matrix.det_fin_two_of, mul_one,
        mul_zero, sub_zero]
      rw [vz_inv v hπ0, vz_pi v hπ]
    have hdet : vz v ((Pinv * a : GL (Fin 2) K) : Matrix (Fin 2) (Fin 2) K).det =
        vz v (a : Matrix (Fin 2) (Fin 2) K).det - 1 := by
      rw [vz_det_mul, hdetP]
      ring
    have hmin := le_entryMin v hge
    have hnn := dispInt_nonneg v (Pinv * a)
    rw [mem_stab]
    unfold dispInt at ha hnn ⊢
    rw [hdet] at hnn ⊢
    omega
  · have hPP : diagPiOne v hπ * Pinv = 1 := by
      refine Matrix.GeneralLinearGroup.ext fun i j ↦ ?_
      rw [Units.val_mul, Units.val_one]
      fin_cases i <;> fin_cases j <;>
        simp [diagPiOne, Pinv, Matrix.mul_apply, Fin.sum_univ_two, hπ0]
    rw [← mul_assoc, hPP, one_mul]

end ValuedMatrixTwo
