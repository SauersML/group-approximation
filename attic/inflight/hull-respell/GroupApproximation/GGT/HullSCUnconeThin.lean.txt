/-!
# Un-coning cyclic peripherals, step 11: the guessed paths are thin, and the residue

Every point of the guessed path from `x` to `y` is within a fixed un-coned distance of the guessed
paths from `x` to `z` and from `z` to `y` (`thin`):

* a vertex of the geodesic word is `4δ`-close in `Γ(Q, S)` to a point between `x` and `z` or between
  `z` and `y` (slim triangles), hence `6δ`-close to a vertex of the other side (two geodesics with
  the same ends), and the corner walk from `x` or from `y` puts it near the other side
  (`vertex_near`);
* a point inside the expansion of a cyclic letter is near the other sides when another side meets its
  coset (Lemma A), and otherwise the letter is short and the point is near its vertex.

With the step and short-path bounds of `GGT/HullSCUnconeExpansion.lean`, Bowditch's guessing-geodesics
criterion proves the hyperbolicity residue `CyclicPeripheralRemovalHyperbolicityStatement`, and with it
the removal statement `CyclicPeripheralRemovalStatement`.
-/

variable {Q : Type u} [Group Q] {Lambda : Type w} {I : Type v}

/-- **A vertex of the side from `x` to `y` is near the other sides.** -/
theorem vertex_near (J : RelGenSet Q (Sum Lambda I)) (g : I → Q)
    (hg : ∀ i : I, J.fam (Sum.inr i) = Subgroup.zpowers (g i))
    (hbaseInv : ∀ x ∈ J.base, x⁻¹ ∈ J.base) (hinf : ∀ i : I, ¬ IsOfFinOrder (g i))
    {C Dc T δ : ℕ}
    (hbnd : ∀ (n : ℕ), n ≤ 6 → ∀ (v : Q) (u : List (RelLetter Q (Sum Lambda I))),
      IsQuasiGeodesicPolygon J 1 0 n v u →
      ∀ (nu : Sum Lambda I) (i k : ℕ), IsComp nu u i k → IsIsolated J.fam nu v u i →
        (vertex v u i)⁻¹ * vertex v u k ∈ J.relBall nu (C * n))
    (hDc : C * 4 ≤ Dc)
    (hT : ∀ (i : I) (e : Q), e ∈ J.relBall (Sum.inr i) Dc → (expo g i e).natAbs ≤ T)
    (hδ : Hyperbolic.IsFourPointHyperbolic J.alphabet.carrier δ) (x y z : Q) {k : ℕ}
    (hk : k ≤ (geodWord J x y).length) :
    ∃ p ∈ sidePoints J g x y z,
      wordDist (removeCyclicPeripherals J g hg hbaseInv).alphabet.carrier
        (vertex x (geodWord J x y) k) p ≤ 2 * T + (3 * T + 1) * (6 * δ) := by
  have hS := J.alphabet.symmetricGenerating
  have hw₁ := geodWord_spec J x y
  have hw₂ := geodWord_spec J x z
  have hw₃ := geodWord_spec J z y
  rcases Hyperbolic.isSlimTriangles_of_isFourPointHyperbolic hS hδ x y z _
      (isBetween_vertex_of_isGeodesicWord J hw₁ hk) with ⟨q, hq, hdq⟩ | ⟨q, hq, hdq⟩
  · -- a point between `x` and `z`: walk from the corner `x`
    have hm : wordDist J.alphabet.carrier x q ≤ (geodWord J x z).length := by
      have h1 : wordDist J.alphabet.carrier x q + wordDist J.alphabet.carrier q z
          = wordDist J.alphabet.carrier x z := hq
      have h2 := hw₂.2.2
      omega
    have hbet := isBetween_vertex_of_isGeodesicWord J hw₂ hm
    have hbig := OsinEnlargement.wordDist_le_of_isBetween_of_wordDist_eq hS hδ hq hbet
      (wordDist_vertex_eq_index J hw₂ hm).symm
    have hu := geodWord_spec J (vertex x (geodWord J x y) k)
      (vertex x (geodWord J x z) (wordDist J.alphabet.carrier x q))
    have hulen : (geodWord J (vertex x (geodWord J x y) k)
        (vertex x (geodWord J x z) (wordDist J.alphabet.carrier x q))).length ≤ 6 * δ := by
      have htri := wordDist_triangle hS (vertex x (geodWord J x y) k) q
        (vertex x (geodWord J x z) (wordDist J.alphabet.carrier x q))
      rw [hu.2.2]
      omega
    obtain ⟨p, hp, hd⟩ := corner_near J g hg hbaseInv hbnd hDc hT (sidePoints J g x y z) hw₁ hw₂
      hk hm hu (fun i _ => vertex_mem_sidePoints_left J g hg x y z i)
      (side_left J g hg hbaseInv hinf hbnd hDc hT x y z k)
    refine ⟨p, hp, le_trans hd ?_⟩
    have hmul := Nat.mul_le_mul_left (3 * T + 1) hulen
    omega
  · -- a point between `z` and `y`: walk from the corner `y` on the reversed words
    have hrev₁ := isGeodesicWord_revWord J hbaseInv hw₁
    have hrev₃ := isGeodesicWord_revWord J hbaseInv hw₃
    have hq' : Hyperbolic.IsBetween J.alphabet.carrier y q z := by
      have h1 : wordDist J.alphabet.carrier z q + wordDist J.alphabet.carrier q y
          = wordDist J.alphabet.carrier z y := hq
      show wordDist J.alphabet.carrier y q + wordDist J.alphabet.carrier q z
        = wordDist J.alphabet.carrier y z
      rw [wordDist_comm hS y q, wordDist_comm hS q z, wordDist_comm hS y z]
      omega
    have hm : wordDist J.alphabet.carrier y q ≤ (revWord (geodWord J z y)).length := by
      have h1 : wordDist J.alphabet.carrier y q + wordDist J.alphabet.carrier q z
          = wordDist J.alphabet.carrier y z := hq'
      have h2 := hrev₃.2.2
      omega
    have hbet := isBetween_vertex_of_isGeodesicWord J hrev₃ hm
    have hbig := OsinEnlargement.wordDist_le_of_isBetween_of_wordDist_eq hS hδ hq' hbet
      (wordDist_vertex_eq_index J hrev₃ hm).symm
    have hlen₁ : (revWord (geodWord J x y)).length = (geodWord J x y).length := by
      simp [revWord]
    have hk' : (geodWord J x y).length - k ≤ (revWord (geodWord J x y)).length := by omega
    have hp₀ : vertex y (revWord (geodWord J x y)) ((geodWord J x y).length - k)
        = vertex x (geodWord J x y) k := by
      have h := vertex_revWord_of_end (geodWord J x y) x k
      rwa [hw₁.2.1] at h
    have hu := geodWord_spec J
      (vertex y (revWord (geodWord J x y)) ((geodWord J x y).length - k))
      (vertex y (revWord (geodWord J z y)) (wordDist J.alphabet.carrier y q))
    have hulen : (geodWord J
        (vertex y (revWord (geodWord J x y)) ((geodWord J x y).length - k))
        (vertex y (revWord (geodWord J z y)) (wordDist J.alphabet.carrier y q))).length
          ≤ 6 * δ := by
      have htri := wordDist_triangle hS (vertex x (geodWord J x y) k) q
        (vertex y (revWord (geodWord J z y)) (wordDist J.alphabet.carrier y q))
      rw [hu.2.2, hp₀]
      omega
    obtain ⟨p, hp, hd⟩ := corner_near J g hg hbaseInv hbnd hDc hT (sidePoints J g x y z) hrev₁
      hrev₃ hk' hm hu (fun i hi => vertex_revWord_mem_sidePoints_right J g hg x y z hi)
      (side_right J g hg hbaseInv hinf hbnd hDc hT x y z _)
    rw [hp₀] at hd
    refine ⟨p, hp, le_trans hd ?_⟩
    have hmul := Nat.mul_le_mul_left (3 * T + 1) hulen
    omega

/-- **The guessed paths are thin.** -/
theorem thin (J : RelGenSet Q (Sum Lambda I)) (g : I → Q)
    (hg : ∀ i : I, J.fam (Sum.inr i) = Subgroup.zpowers (g i))
    (hbaseInv : ∀ x ∈ J.base, x⁻¹ ∈ J.base) (hinf : ∀ i : I, ¬ IsOfFinOrder (g i))
    {C Dc T δ : ℕ}
    (hbnd : ∀ (n : ℕ), n ≤ 6 → ∀ (v : Q) (u : List (RelLetter Q (Sum Lambda I))),
      IsQuasiGeodesicPolygon J 1 0 n v u →
      ∀ (nu : Sum Lambda I) (i k : ℕ), IsComp nu u i k → IsIsolated J.fam nu v u i →
        (vertex v u i)⁻¹ * vertex v u k ∈ J.relBall nu (C * n))
    (hDc : C * 4 ≤ Dc)
    (hT : ∀ (i : I) (e : Q), e ∈ J.relBall (Sum.inr i) Dc → (expo g i e).natAbs ≤ T)
    (hδ : Hyperbolic.IsFourPointHyperbolic J.alphabet.carrier δ) (x y z : Q) {i : ℕ}
    (hi : i ≤ guessLen J g x y) :
    ∃ p ∈ sidePoints J g x y z,
      wordDist (removeCyclicPeripherals J g hg hbaseInv).alphabet.carrier
        (guessPath J g x y i) p ≤ 2 * T + (3 * T + 1) * (6 * δ) + T := by
  classical
  have hS' := (removeCyclicPeripherals J g hg hbaseInv).alphabet.symmetricGenerating
  have hw₁ := geodWord_spec J x y
  rcases exists_letter_block g (geodWord J x y) i hi with ⟨k, hk, rfl⟩ | ⟨k, hk, s, hs0, hsl, rfl⟩
  · -- a vertex of the geodesic word
    obtain ⟨p, hp, hd⟩ := vertex_near J g hg hbaseInv hinf hbnd hDc hT hδ x y z hk
    refine ⟨p, hp, ?_⟩
    rw [guessPath_at_letter J g hg x y k]
    omega
  · -- a point strictly inside the expansion of a letter, which is then cyclic
    have hletter : ∃ (i₀ : I) (e : Q), (geodWord J x y)[k]'hk = RelLetter.comp (Sum.inr i₀) e := by
      by_contra hne
      push_neg at hne
      have h1 : (expLetters g ((geodWord J x y)[k]'hk)).length = 1 := by
        revert hne
        generalize (geodWord J x y)[k]'hk = a
        intro hne
        cases a with
        | base b => rfl
        | comp mu h =>
            cases mu with
            | inl lam => rfl
            | inr i₀ => exact absurd rfl (hne i₀ h)
      omega
    obtain ⟨i₀, e, he⟩ := hletter
    have hsl' : s < (expo g i₀ e).natAbs := by
      rw [he, length_expLetters_inr g i₀ e] at hsl
      exact hsl
    have hc : IsComp (Sum.inr i₀) (geodWord J x y) k (k + 1) :=
      isComp_succ_of_isCompOf_geodesic J hw₁ hk (by rw [he]; exact rfl)
    by_cases hconn : (∃ i₂ k₂ : ℕ, IsComp (Sum.inr i₀) (geodWord J x z) i₂ k₂ ∧
          (QuotientGroup.mk (vertex x (geodWord J x z) i₂) : Q ⧸ J.fam (Sum.inr i₀))
            = QuotientGroup.mk (vertex x (geodWord J x y) k)) ∨
        (∃ i₃ k₃ : ℕ, IsComp (Sum.inr i₀) (geodWord J z y) i₃ k₃ ∧
          (QuotientGroup.mk (vertex z (geodWord J z y) i₃) : Q ⧸ J.fam (Sum.inr i₀))
            = QuotientGroup.mk (vertex x (geodWord J x y) k))
    · rcases letter_point_near_sides J g hg hbaseInv hinf hbnd hDc hT x y z hk he hconn
          (le_of_lt hsl') with ⟨n, hn, hd⟩ | ⟨n, hn, hd⟩
      · exact ⟨guessPath J g x z n, Or.inl ⟨n, hn, rfl⟩, by omega⟩
      · exact ⟨guessPath J g z y n, Or.inr ⟨n, hn, rfl⟩, by omega⟩
    · have hno₂ : ∀ i' k' : ℕ, IsComp (Sum.inr i₀) (geodWord J x z) i' k' →
          (QuotientGroup.mk (vertex x (geodWord J x z) i') : Q ⧸ J.fam (Sum.inr i₀))
            ≠ QuotientGroup.mk (vertex x (geodWord J x y) k) :=
        fun i' k' hc' hcos => hconn (Or.inl ⟨i', k', hc', hcos⟩)
      have hno₃ : ∀ i' k' : ℕ, IsComp (Sum.inr i₀) (geodWord J z y) i' k' →
          (QuotientGroup.mk (vertex z (geodWord J z y) i') : Q ⧸ J.fam (Sum.inr i₀))
            ≠ QuotientGroup.mk (vertex x (geodWord J x y) k) :=
        fun i' k' hc' hcos => hconn (Or.inr ⟨i', k', hc', hcos⟩)
      have h0 := span_mem_relBall_of_noComp J (Sum.inr i₀) hbaseInv hbnd hDc hw₁
        (geodWord_spec J x z) (geodWord_spec J z y) hc hno₂ hno₃
      rw [vertex_succ (geodWord J x y) x k hk, he, inv_mul_cancel_left] at h0
      have hexp := hT i₀ e h0
      obtain ⟨p, hp, hd⟩ := vertex_near J g hg hbaseInv hinf hbnd hDc hT hδ x y z (le_of_lt hk)
      have hwalk := wordDist_guessPath_le J g hg hbaseInv x y
        (Nat.le_add_right (expWord g ((geodWord J x y).take k)).length s) hi
      rw [guessPath_at_letter J g hg x y k] at hwalk
      have htri := wordDist_triangle hS'
        (guessPath J g x y ((expWord g ((geodWord J x y).take k)).length + s))
        (vertex x (geodWord J x y) k) p
      rw [wordDist_comm hS'
        (guessPath J g x y ((expWord g ((geodWord J x y).take k)).length + s))
        (vertex x (geodWord J x y) k)] at htri
      exact ⟨p, hp, by omega⟩

end Thin

end Uncone

/-- **Removing infinite cyclic peripheral members keeps the Cayley graph hyperbolic.** -/
theorem cyclicPeripheralRemovalHyperbolicityStatement :
    CyclicPeripheralRemovalHyperbolicityStatement.{u, w, v} := by
  intro Q _ Lambda I _ J g hg hbaseInv hinf hJ
  obtain ⟨δ, hδ⟩ := exists_isFourPointHyperbolic_of_isHyperbolicallyEmbedded J hJ
  obtain ⟨C, hC, hbnd⟩ := sixBound_one_of_fourPointHyperbolic J hbaseInv hδ 0 le_rfl
  obtain ⟨T, hT⟩ := Uncone.exists_expo_bound J g hJ (C * 4)
  have hT1 : ∀ (i : I) (e : Q), e ∈ J.relBall (Sum.inr i) 1 → (Uncone.expo g i e).natAbs ≤ T :=
    fun i e he => hT i e (relBall_mono_radius J (Sum.inr i) (by omega) he)
  obtain ⟨δ', hδ'⟩ := OsinEnlargement.guessingGeodesics Q
    (removeCyclicPeripherals J g hg hbaseInv).alphabet.carrier
    (removeCyclicPeripherals J g hg hbaseInv).alphabet.symmetricGenerating
    (max (max T 1) (2 * T + (3 * T + 1) * (6 * δ) + T))
    (Uncone.guessLen J g) (Uncone.guessPath J g) (Uncone.guessPath_zero J g)
    (Uncone.guessPath_len J g hg)
    (fun x y _ hi => Uncone.wordDist_guessPath_succ J g hg hbaseInv x y hi)
    (fun x y hxy i hi j hj =>
      le_trans (Uncone.guessPath_short J g hg hbaseInv hinf hT1 x y hxy i hi j hj)
        (le_max_left _ _))
    (fun x y z i hi => by
      obtain ⟨p, hp, hd⟩ := Uncone.thin J g hg hbaseInv hinf hbnd le_rfl hT hδ x y z hi
      rcases hp with ⟨n, hn, rfl⟩ | ⟨n, hn, rfl⟩
      · exact ⟨n, Or.inl ⟨hn, le_trans hd (le_max_right _ _)⟩⟩
      · exact ⟨n, Or.inr ⟨hn, le_trans hd (le_max_right _ _)⟩⟩)
  exact ⟨(δ' : ℝ), isHyperbolicSpace_cayley_of_fourPoint _ hδ'⟩

/-- **Removing infinite cyclic peripheral members keeps the rest hyperbolically embedded.** -/
theorem cyclicPeripheralRemovalStatement : CyclicPeripheralRemovalStatement.{u, w, v} :=
  cyclicPeripheralRemovalStatement_of_hyperbolicity cyclicPeripheralRemovalHyperbolicityStatement
