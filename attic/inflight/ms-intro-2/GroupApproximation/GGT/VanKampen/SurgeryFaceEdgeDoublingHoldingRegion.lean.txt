/-!
# A face set holding the doubled face, through the doubling of an edge

`FaceEdgeDoubling.diagram` inserts a digon inside an inner face `f` along a dart `w_j`.
`SurgeryFaceEdgeDoublingRegions.lean` and `SurgeryFaceEdgeDoublingSideRegions.lean` carry face sets
avoiding `f`, and `SurgeryGeodesicCollarDouble.lean` carries a face set holding `f` together with the
digon.  This file carries a face set `s` holding `f` **without** the digon, when `w_j` lies on its
boundary: the face across `w_j` lies outside `s`.  This is the transport the region-pair thickening
needs (`SurgeryRegionPairThickening.lean`): the digon separates `s` from the face across `w_j`.

The face `f` becomes the retained face `cellFace`, and the boundary dart `w_j` is replaced by the
new dart (`FaceEdgeDoubling.dartImage`).  Every other old dart keeps its role.

* `facePerm_dartImage`: face rotation commutes with `dartImage`, for every old dart.  At the corner
  `w_j` the new dart follows the predecessor of `w_j` and precedes the successor of `w_j`.
* `faceOf_dartImage`, `alpha_dartImage_of_ne`: faces through `faceImage`; edges off `{w_j, α w_j}`.
* `holdingFaceSet`: the images of the faces of `s`, without the digon.
* `isBoundaryDart_dartImage_iff`, `boundary_dart_holding_iff`: the boundary darts are the images of
  the old ones.
* `internalMove_dartImage`, `boundaryStep_dartImage`: internal moves and boundary steps lift.
* `holdingBoundary`: the boundary cycle, mapped by `dartImage`.

## Manuscript status

Infrastructure for `thm:hull` (tex 2121, "Hull's small cancellation theorem", through Osin's
Lemma 9.7(b)); certifies no printed sentence on its own.
-/

namespace GroupApproximation.GGT.VanKampen.FaceEdgeDoubling.Holding

open HullSC EdgeInsertion Surgery Embedded
open scoped Classical

universe u w v

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
variable (Delta : DiscDiagram.{u, w, v} W) (f : Delta.toCombMap.Face)
  (j : Fin (Delta.faceBoundary f).darts.length)
  (hlen : 1 < (Delta.faceBoundary f).darts.length) (hf : f ≠ Delta.outerFace)

/-! ## The dart map -/

theorem dartImage_dart : dartImage Delta f j hlen (dart Delta f j) = some none := by
  rw [dartImage, if_pos rfl]

theorem dartImage_of_ne {d : Delta.toCombMap.Dart} (hd : d ≠ dart Delta f j) :
    dartImage Delta f j hlen d = embed Delta.toCombMap d := by
  rw [dartImage, if_neg hd]

theorem dartImage_injective : Function.Injective (dartImage Delta f j hlen) := by
  intro d e h
  by_cases hd : d = dart Delta f j
  · by_cases he : e = dart Delta f j
    · exact hd.trans he.symm
    · rw [hd, dartImage_dart, dartImage_of_ne Delta f j hlen he] at h
      cases h
  · by_cases he : e = dart Delta f j
    · rw [he, dartImage_dart, dartImage_of_ne Delta f j hlen hd] at h
      cases h
    · rw [dartImage_of_ne Delta f j hlen hd, dartImage_of_ne Delta f j hlen he] at h
      exact embed_injective Delta.toCombMap h

theorem dartImage_ne_none (d : Delta.toCombMap.Dart) : dartImage Delta f j hlen d ≠ none := by
  unfold dartImage
  split_ifs <;> exact Option.some_ne_none _

/-! ## Face rotation -/

/-- The corner after the doubled dart is the face successor of the doubled dart. -/
theorem secondCorner_eq_facePerm :
    secondCorner Delta.toCombMap (rebased Delta f j) (second Delta f j hlen) =
      Delta.toCombMap.facePerm (dart Delta f j) := by
  have hlt : 0 + 1 < (rebased Delta f j).darts.length := by
    rw [rebased_length]
    exact hlen
  have h := List.isChain_iff_getElem.mp (rebased Delta f j).chain 0 hlt
  change (rebased Delta f j).darts[1] = Delta.toCombMap.facePerm ((rebased Delta f j).darts[0])
  exact h.symm

include hlen in
/-- The doubled dart is not its own face successor. -/
theorem facePerm_dart_ne : Delta.toCombMap.facePerm (dart Delta f j) ≠ dart Delta f j := by
  rw [← secondCorner_eq_facePerm Delta f j hlen]
  exact (corners_ne Delta.toCombMap (rebased Delta f j) (second Delta f j hlen) Nat.one_pos).symm

/-- **Face rotation commutes with the dart map onto the face of `f`.** -/
theorem facePerm_dartImage (d : Delta.toCombMap.Dart) :
    (map Delta f j hlen).facePerm (dartImage Delta f j hlen d) =
      dartImage Delta f j hlen (Delta.toCombMap.facePerm d) := by
  have hb := secondCorner_eq_facePerm Delta f j hlen
  have hne := facePerm_dart_ne Delta f j hlen
  by_cases hd : d = dart Delta f j
  · rw [hd, dartImage_dart, dartImage_of_ne Delta f j hlen hne]
    change PermOrbitInsert.insertBefore
        (PermOrbitInsert.insertBefore Delta.toCombMap.sigma (dart Delta f j))
        (some (secondCorner Delta.toCombMap (rebased Delta f j) (second Delta f j hlen))) none =
      embed Delta.toCombMap (Delta.toCombMap.facePerm (dart Delta f j))
    rw [PermOrbitInsert.insertBefore_none, hb]
    rfl
  · rw [dartImage_of_ne Delta f j hlen hd]
    change PermOrbitInsert.insertBefore
        (PermOrbitInsert.insertBefore Delta.toCombMap.sigma (dart Delta f j))
        (some (secondCorner Delta.toCombMap (rebased Delta f j) (second Delta f j hlen)))
        (some (some (Delta.toCombMap.alpha d))) = dartImage Delta f j hlen (Delta.toCombMap.facePerm d)
    have e1 : Delta.toCombMap.sigma (Delta.toCombMap.alpha d) = Delta.toCombMap.facePerm d := rfl
    rw [hb, PermOrbitInsert.insertBefore_some, PermOrbitInsert.insertBefore_some, e1, dartImage]
    split_ifs with h1 h2
    · rfl
    · exact absurd (Delta.toCombMap.facePerm.injective (Option.some.inj h2)) hd
    · rfl

/-! ## Faces and edges -/

/-- Every old dart lies in the image of its face; `w_j` goes to the new dart of `f`. -/
theorem faceOf_dartImage (d : Delta.toCombMap.Dart) :
    (map Delta f j hlen).faceOf (dartImage Delta f j hlen d) =
      faceImage Delta f j hlen (Delta.toCombMap.faceOf d) := by
  by_cases hd : d = dart Delta f j
  · rw [hd, dartImage_dart, dart_face, faceImage_self]
  · rw [dartImage_of_ne Delta f j hlen hd]
    by_cases hg : Delta.toCombMap.faceOf d = f
    · rw [hg, faceImage_self]
      exact faceOf_embed_of_face_of_ne Delta f j hlen hg hd
    · exact faceOf_embed_of_face_ne Delta f j hlen hg

/-- Off `w_j` and `α w_j`, edges commute with the dart map. -/
theorem alpha_dartImage_of_ne {d : Delta.toCombMap.Dart} (hd : d ≠ dart Delta f j)
    (had : Delta.toCombMap.alpha d ≠ dart Delta f j) :
    (map Delta f j hlen).alpha (dartImage Delta f j hlen d) =
      dartImage Delta f j hlen (Delta.toCombMap.alpha d) := by
  rw [dartImage_of_ne Delta f j hlen hd, dartImage_of_ne Delta f j hlen had]
  rfl

/-! ## The face set holding `f` -/

/-- The images of the faces of `s`, without the digon. -/
noncomputable def holdingFaceSet (s : Finset Delta.toCombMap.Face) :
    Finset (diagram Delta f j hlen hf).toCombMap.Face :=
  s.map ⟨faceImage Delta f j hlen, faceImage_injective Delta f j hlen⟩

theorem mem_holdingFaceSet_faceImage (s : Finset Delta.toCombMap.Face) (g : Delta.toCombMap.Face) :
    faceImage Delta f j hlen g ∈ holdingFaceSet Delta f j hlen hf s ↔ g ∈ s :=
  Finset.mem_map' ⟨faceImage Delta f j hlen, faceImage_injective Delta f j hlen⟩

theorem digon_not_mem_holdingFaceSet (s : Finset Delta.toCombMap.Face) :
    digon Delta f j hlen ∉ holdingFaceSet Delta f j hlen hf s := by
  intro h
  obtain ⟨g, _, hg⟩ := Finset.mem_map.mp h
  exact faceImage_ne_digon Delta f j hlen g hg

theorem faceOf_none_not_mem (s : Finset Delta.toCombMap.Face) :
    (diagram Delta f j hlen hf).toCombMap.faceOf none ∉ holdingFaceSet Delta f j hlen hf s := by
  intro h
  obtain ⟨g, _, hg⟩ := Finset.mem_map.mp h
  exact faceImage_ne_digon Delta f j hlen g (hg.trans (faceOf_none_digon Delta f j hlen))

theorem faceOf_embed_dart_not_mem (s : Finset Delta.toCombMap.Face) :
    (diagram Delta f j hlen hf).toCombMap.faceOf (embed Delta.toCombMap (dart Delta f j)) ∉
      holdingFaceSet Delta f j hlen hf s := by
  intro h
  obtain ⟨g, _, hg⟩ := Finset.mem_map.mp h
  exact faceImage_ne_digon Delta f j hlen g (hg.trans (faceOf_embed_dart_digon Delta f j hlen))

theorem faceOf_dartImage_mem_iff (s : Finset Delta.toCombMap.Face) (d : Delta.toCombMap.Dart) :
    (diagram Delta f j hlen hf).toCombMap.faceOf (dartImage Delta f j hlen d) ∈
        holdingFaceSet Delta f j hlen hf s ↔ Delta.toCombMap.faceOf d ∈ s :=
  (congrArg (· ∈ holdingFaceSet Delta f j hlen hf s)
      (faceOf_dartImage Delta f j hlen d)).to_iff.trans
    (mem_holdingFaceSet_faceImage Delta f j hlen hf s _)

/-- **The boundary darts of the holding face set are the images of the old ones**, when `w_j` lies
on the old boundary. -/
theorem isBoundaryDart_dartImage_iff (s : Finset Delta.toCombMap.Face)
    (hw : Delta.toCombMap.faceOf (Delta.toCombMap.alpha (dart Delta f j)) ∉ s)
    (d : Delta.toCombMap.Dart) :
    IsBoundaryDart (diagram Delta f j hlen hf) (holdingFaceSet Delta f j hlen hf s)
        (dartImage Delta f j hlen d) ↔ IsBoundaryDart Delta s d := by
  unfold IsBoundaryDart
  rw [faceOf_dartImage_mem_iff Delta f j hlen hf s d]
  by_cases hd : d = dart Delta f j
  · have hnew : (diagram Delta f j hlen hf).toCombMap.faceOf
        ((diagram Delta f j hlen hf).toCombMap.alpha (dartImage Delta f j hlen d)) ∉
          holdingFaceSet Delta f j hlen hf s := by
      rw [hd, dartImage_dart]
      exact faceOf_none_not_mem Delta f j hlen hf s
    have hold : Delta.toCombMap.faceOf (Delta.toCombMap.alpha d) ∉ s := by
      rw [hd]
      exact hw
    exact ⟨fun h => ⟨h.1, hold⟩, fun h => ⟨h.1, hnew⟩⟩
  · by_cases had : Delta.toCombMap.alpha d = dart Delta f j
    · have hds : Delta.toCombMap.faceOf d ∉ s := by
        have hdeq : d = Delta.toCombMap.alpha (dart Delta f j) := by
          rw [← had, Delta.toCombMap.alpha_involutive d]
        rw [hdeq]
        exact hw
      exact ⟨fun h => absurd h.1 hds, fun h => absurd h.1 hds⟩
    · have halpha : (diagram Delta f j hlen hf).toCombMap.alpha (dartImage Delta f j hlen d) =
          dartImage Delta f j hlen (Delta.toCombMap.alpha d) :=
        alpha_dartImage_of_ne Delta f j hlen hd had
      rw [halpha, faceOf_dartImage_mem_iff Delta f j hlen hf s (Delta.toCombMap.alpha d)]

theorem boundary_dart_holding_iff (s : Finset Delta.toCombMap.Face)
    (hw : Delta.toCombMap.faceOf (Delta.toCombMap.alpha (dart Delta f j)) ∉ s)
    (x : (diagram Delta f j hlen hf).toCombMap.Dart) :
    IsBoundaryDart (diagram Delta f j hlen hf) (holdingFaceSet Delta f j hlen hf s) x ↔
      ∃ d, dartImage Delta f j hlen d = x ∧ IsBoundaryDart Delta s d := by
  constructor
  · intro hx
    rcases x with _ | (_ | e)
    · exact absurd hx.1 (faceOf_none_not_mem Delta f j hlen hf s)
    · refine ⟨dart Delta f j, dartImage_dart Delta f j hlen, ?_⟩
      refine (isBoundaryDart_dartImage_iff Delta f j hlen hf s hw (dart Delta f j)).mp ?_
      rw [dartImage_dart]
      exact hx
    · by_cases he : e = dart Delta f j
      · refine absurd hx.1 ?_
        rw [he]
        exact faceOf_embed_dart_not_mem Delta f j hlen hf s
      · refine ⟨e, dartImage_of_ne Delta f j hlen he, ?_⟩
        refine (isBoundaryDart_dartImage_iff Delta f j hlen hf s hw e).mp ?_
        rw [dartImage_of_ne Delta f j hlen he]
        exact hx
  · rintro ⟨d, rfl, hd⟩
    exact (isBoundaryDart_dartImage_iff Delta f j hlen hf s hw d).mpr hd

/-! ## Boundary steps and the boundary cycle -/

theorem internalMove_dartImage (s : Finset Delta.toCombMap.Face)
    (hw : Delta.toCombMap.faceOf (Delta.toCombMap.alpha (dart Delta f j)) ∉ s)
    {d e : Delta.toCombMap.Dart} (h : InternalBoundaryMove Delta s d e) :
    InternalBoundaryMove (diagram Delta f j hlen hf) (holdingFaceSet Delta f j hlen hf s)
      (dartImage Delta f j hlen d) (dartImage Delta f j hlen e) := by
  obtain ⟨hd, had, rfl⟩ := h
  have hne : d ≠ dart Delta f j := by
    intro heq
    rw [heq] at had
    exact hw had
  have hane : Delta.toCombMap.alpha d ≠ dart Delta f j := by
    intro heq
    have hdeq : d = Delta.toCombMap.alpha (dart Delta f j) := by
      rw [← heq, Delta.toCombMap.alpha_involutive d]
    rw [hdeq] at hd
    exact hw hd
  have halpha : (diagram Delta f j hlen hf).toCombMap.alpha (dartImage Delta f j hlen d) =
      dartImage Delta f j hlen (Delta.toCombMap.alpha d) :=
    alpha_dartImage_of_ne Delta f j hlen hne hane
  refine ⟨(faceOf_dartImage_mem_iff Delta f j hlen hf s d).mpr hd, ?_, ?_⟩
  · rw [halpha]
    exact (faceOf_dartImage_mem_iff Delta f j hlen hf s _).mpr had
  · rw [halpha]
    exact (facePerm_dartImage Delta f j hlen (Delta.toCombMap.alpha d)).symm

theorem boundaryStep_dartImage (s : Finset Delta.toCombMap.Face)
    (hw : Delta.toCombMap.faceOf (Delta.toCombMap.alpha (dart Delta f j)) ∉ s)
    {d e : Delta.toCombMap.Dart} (h : BoundaryStep Delta s d e) :
    BoundaryStep (diagram Delta f j hlen hf) (holdingFaceSet Delta f j hlen hf s)
      (dartImage Delta f j hlen d) (dartImage Delta f j hlen e) := by
  refine ⟨(isBoundaryDart_dartImage_iff Delta f j hlen hf s hw d).mpr h.1,
    (isBoundaryDart_dartImage_iff Delta f j hlen hf s hw e).mpr h.2.1, ?_⟩
  have hfp : (diagram Delta f j hlen hf).toCombMap.facePerm (dartImage Delta f j hlen d) =
      dartImage Delta f j hlen (Delta.toCombMap.facePerm d) :=
    facePerm_dartImage Delta f j hlen d
  rw [hfp]
  exact h.2.2.lift (dartImage Delta f j hlen)
    (fun _ _ hm => internalMove_dartImage Delta f j hlen hf s hw hm)

/-- **The boundary cycle of a face set holding `f`, after the doubling**: the old cycle mapped by
`dartImage`, when the face across `w_j` lies outside the face set. -/
noncomputable def holdingBoundary (s : Finset Delta.toCombMap.Face)
    (hw : Delta.toCombMap.faceOf (Delta.toCombMap.alpha (dart Delta f j)) ∉ s)
    (B : FaceSetBoundary Delta s) :
    FaceSetBoundary (diagram Delta f j hlen hf) (holdingFaceSet Delta f j hlen hf s) where
  faces_nonempty := by
    obtain ⟨g, hg⟩ := B.faces_nonempty
    exact ⟨faceImage Delta f j hlen g, (mem_holdingFaceSet_faceImage Delta f j hlen hf s g).mpr hg⟩
  all_gCells := by
    intro F hF
    obtain ⟨g, hg, rfl⟩ := Finset.mem_map.mp hF
    refine ⟨?_, ?_⟩
    · change faceImage Delta f j hlen g ≠ keep Delta f j hlen Delta.outerFace
      rw [← faceImage_of_ne Delta f j hlen hf.symm]
      exact fun heq => (B.all_gCells g hg).1 (faceImage_injective Delta f j hlen heq)
    · change RelLetter.listVal ((boundary Delta f j hlen (faceImage Delta f j hlen g)).darts.map
        (label Delta f j hlen)) = 1
      rw [boundary_faceImage_word]
      exact (B.all_gCells g hg).2
  cycle := B.cycle.map (dartImage Delta f j hlen)
  cycle_nonempty := fun h => B.cycle_nonempty (List.map_eq_nil_iff.mp h)
  cycle_nodup := B.cycle_nodup.map (dartImage_injective Delta f j hlen)
  cycle_mem_iff := by
    intro x
    rw [boundary_dart_holding_iff Delta f j hlen hf s hw x, List.mem_map]
    constructor
    · rintro ⟨e, he, hex⟩
      exact ⟨e, hex, (B.cycle_mem_iff e).mp he⟩
    · rintro ⟨e, hex, he⟩
      exact ⟨e, (B.cycle_mem_iff e).mpr he, hex⟩
  cycle_chain := by
    rw [List.isChain_map]
    exact B.cycle_chain.imp (fun _ _ h => boundaryStep_dartImage Delta f j hlen hf s hw h)
  cycle_closes := by
    rw [List.getLast_map, List.head_map]
    exact boundaryStep_dartImage Delta f j hlen hf s hw B.cycle_closes

theorem holdingBoundary_cycle (s : Finset Delta.toCombMap.Face)
    (hw : Delta.toCombMap.faceOf (Delta.toCombMap.alpha (dart Delta f j)) ∉ s)
    (B : FaceSetBoundary Delta s) :
    (holdingBoundary Delta f j hlen hf s hw B).cycle = B.cycle.map (dartImage Delta f j hlen) :=
  rfl
