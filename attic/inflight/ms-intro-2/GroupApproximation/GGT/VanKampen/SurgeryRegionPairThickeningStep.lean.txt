/-!
# One step of the region-pair thickening: the carried family

Osin, arXiv:math/0411039v3, §9, proof of Lemma 9.7(b).  An edge between faces of two selected
regions lies on a side of each.  Doubling it inside the face `f` of one of them
(`FaceEdgeDoubling.diagram`) puts a G-cell digon between the two regions.  The region holding `f`
is carried through `dartImage` (`FaceEdgeDoubling.Holding.holdingCandidate`); every other region
avoids `f` and is carried along its arcs (`FaceEdgeDoubling.regionFamilyOfArcs`).

* `StepData`: the region holding `f`, with `w_j` on a side, and the arcs of the other regions avoiding
  `w_j`.
* `stepFamily`, with `holding_not_mem_carried`, `stepFamily_card`, `stepFamily_weight`,
  `stepFamily_pairwise`, `stepFamily_profile`, `stepFamily_noLoop`.
* `stepEquiv`: the old regions correspond to the new ones, with the same target profile, source
  index and target index.
* `sectionFamily`: the distinguished section family after the step, with the same weight and card.

## Manuscript status

Infrastructure for `thm:hull` (tex 2121, "Hull's small cancellation theorem", through Osin's
Lemma 9.7(b)); certifies no printed sentence on its own.
-/

section Doubling

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {eps : ℕ}
  (Delta : DiscDiagram.{u, w, v} W) (f : Delta.toCombMap.Face)
  (j : Fin (Delta.faceBoundary f).darts.length)
  (hlen : 1 < (Delta.faceBoundary f).darts.length) (hf : f ≠ Delta.outerFace)

/-- **The data of one step**: the region holding `f`, whose side runs along `w_j`, and the arcs of
the other regions avoiding `w_j`. -/
structure StepData (family : Finset (RegionCandidate D eps Delta)) where
  holder : RegionCandidate D eps Delta
  holder_mem : holder ∈ family
  side : FaceEdgeDoubling.dart Delta f j ∈ holder.2.rightSide ++ holder.2.leftSide
  avoid : FaceEdgeDoubling.FamilyArcAvoid Delta f j (family.erase holder)

variable {family : Finset (RegionCandidate D eps Delta)}

/-- The holder holds `f`. -/
theorem f_mem_holder (T : StepData Delta f j family) : f ∈ T.holder.1 := by
  have hcyc : FaceEdgeDoubling.dart Delta f j ∈ T.holder.2.boundary.cycle := by
    rw [T.holder.2.boundary_decomposition]
    rcases List.mem_append.mp T.side with hr | hl
    · exact List.mem_append_left _ (List.mem_append_left _ (List.mem_append_right _ hr))
    · exact List.mem_append_right _ hl
  have h := ((T.holder.2.boundary.cycle_mem_iff _).mp hcyc).1
  rwa [FaceEdgeDoubling.dart_face] at h

/-- The holder, carried through `dartImage`. -/
noncomputable abbrev holding (T : StepData Delta f j family) :
    RegionCandidate D eps (FaceEdgeDoubling.diagram Delta f j hlen hf) :=
  FaceEdgeDoubling.Holding.holdingCandidate Delta f j hlen hf T.holder T.side

/-- The other regions, carried along their arcs. -/
noncomputable abbrev carried (T : StepData Delta f j family) :
    Finset (RegionCandidate D eps (FaceEdgeDoubling.diagram Delta f j hlen hf)) :=
  FaceEdgeDoubling.regionFamilyOfArcs Delta f j hlen hf (family.erase T.holder) T.avoid

/-- The family after the step. -/
noncomputable def stepFamily (T : StepData Delta f j family) :
    Finset (RegionCandidate D eps (FaceEdgeDoubling.diagram Delta f j hlen hf)) :=
  insert (holding Delta f j hlen hf T) (carried Delta f j hlen hf T)

theorem holding_not_mem_carried (T : StepData Delta f j family) :
    holding Delta f j hlen hf T ∉ carried Delta f j hlen hf T := by
  intro hmem
  obtain ⟨b, -, hb⟩ := Finset.mem_map.mp hmem
  have hfaces : (FaceEdgeDoubling.embedding Delta f j hlen hf).faceSet b.val.1 =
      FaceEdgeDoubling.Holding.holdingFaceSet Delta f j hlen hf T.holder.1 :=
    congrArg Sigma.fst hb
  have hcell : FaceEdgeDoubling.faceImage Delta f j hlen f ∈
      FaceEdgeDoubling.Holding.holdingFaceSet Delta f j hlen hf T.holder.1 :=
    (FaceEdgeDoubling.Holding.mem_holdingFaceSet_faceImage Delta f j hlen hf T.holder.1 f).mpr
      (f_mem_holder Delta f j T)
  have hcell' := (Finset.ext_iff.mp hfaces (FaceEdgeDoubling.faceImage Delta f j hlen f)).mpr hcell
  obtain ⟨g, hg, hgc⟩ := Finset.mem_map.mp hcell'
  exact FaceEdgeDoubling.keep_ne_cellFace Delta f j hlen
    (fun h => (T.avoid b.val b.property).1 (h ▸ hg))
    (hgc.trans (FaceEdgeDoubling.faceImage_self Delta f j hlen))

theorem stepFamily_card (T : StepData Delta f j family) :
    (stepFamily Delta f j hlen hf T).card = family.card := by
  rw [stepFamily, Finset.card_insert_of_notMem (holding_not_mem_carried Delta f j hlen hf T),
    FaceEdgeDoubling.regionFamilyOfArcs_card, Finset.card_erase_add_one T.holder_mem]

theorem stepFamily_weight (T : StepData Delta f j family) :
    EstimatingSelection.familyWeight RegionCandidate.weight (stepFamily Delta f j hlen hf T) =
      EstimatingSelection.familyWeight RegionCandidate.weight family := by
  have hc := FaceEdgeDoubling.regionFamilyOfArcs_weight Delta f j hlen hf (family.erase T.holder)
    T.avoid
  unfold EstimatingSelection.familyWeight at hc ⊢
  rw [stepFamily, Finset.sum_insert (holding_not_mem_carried Delta f j hlen hf T), hc,
    FaceEdgeDoubling.Holding.holdingCandidate_weight, Finset.add_sum_erase _ _ T.holder_mem]

theorem stepFamily_pairwise (T : StepData Delta f j family)
    (hcompat : EstimatingSelection.PairwiseCompatible RegionCandidate.Compatible family) :
    EstimatingSelection.PairwiseCompatible RegionCandidate.Compatible
      (stepFamily Delta f j hlen hf T) := by
  have hcarried := FaceEdgeDoubling.regionFamilyOfArcs_pairwise Delta f j hlen hf
    (family.erase T.holder) T.avoid
    (fun a ha b hb hab => hcompat a (Finset.mem_of_mem_erase ha) b (Finset.mem_of_mem_erase hb) hab)
  have hcross : ∀ b ∈ carried Delta f j hlen hf T,
      RegionCandidate.Compatible (holding Delta f j hlen hf T) b := by
    intro b hb
    obtain ⟨a, -, rfl⟩ := Finset.mem_map.mp hb
    have hne : a.val ≠ T.holder := Finset.ne_of_mem_erase a.property
    have hdisj : Disjoint T.holder.1 a.val.1 :=
      hcompat T.holder T.holder_mem a.val (Finset.mem_of_mem_erase a.property) hne.symm
    refine Finset.disjoint_left.mpr ?_
    intro F hF1 hF2
    obtain ⟨g, hg, rfl⟩ := Finset.mem_map.mp hF1
    obtain ⟨h, hh, hhg⟩ := Finset.mem_map.mp hF2
    have hhf : h ≠ f := fun heq => (T.avoid a.val a.property).1 (heq ▸ hh)
    have heq : h = g := FaceEdgeDoubling.faceImage_injective Delta f j hlen
      ((FaceEdgeDoubling.faceImage_of_ne Delta f j hlen hhf).trans hhg)
    exact Finset.disjoint_left.mp hdisj hg (heq ▸ hh)
  intro x hx y hy hxy
  rcases Finset.mem_insert.mp hx with rfl | hx'
  · rcases Finset.mem_insert.mp hy with rfl | hy'
    · exact absurd rfl hxy
    · exact hcross y hy'
  · rcases Finset.mem_insert.mp hy with rfl | hy'
    · exact Disjoint.symm (hcross x hx')
    · exact hcarried x hx' y hy' hxy

theorem stepFamily_profile (T : StepData Delta f j family)
    {a : RegionCandidate D eps (FaceEdgeDoubling.diagram Delta f j hlen hf)}
    (ha : a ∈ stepFamily Delta f j hlen hf T) :
    ∃ b ∈ family, RegionCandidate.SameTargetProfile a b := by
  rcases Finset.mem_insert.mp ha with rfl | ha'
  · exact ⟨T.holder, T.holder_mem,
      FaceEdgeDoubling.Holding.holdingCandidate_profile Delta f j hlen hf T.holder T.side⟩
  · obtain ⟨b, hb, hab⟩ :=
      FaceEdgeDoubling.regionFamilyOfArcs_profile Delta f j hlen hf _ T.avoid ha'
    exact ⟨b, Finset.mem_of_mem_erase hb, hab⟩

theorem stepFamily_noLoop (T : StepData Delta f j family)
    (hfamily : ∀ b ∈ family, b.2.target ≠ some b.2.source)
    {a : RegionCandidate D eps (FaceEdgeDoubling.diagram Delta f j hlen hf)}
    (ha : a ∈ stepFamily Delta f j hlen hf T) :
    a.2.target ≠ some a.2.source := by
  rcases Finset.mem_insert.mp ha with rfl | ha'
  · intro h
    change Option.map (FaceEdgeDoubling.cellMap Delta f j hlen hf).indexEquiv T.holder.2.target =
      some ((FaceEdgeDoubling.cellMap Delta f j hlen hf).indexEquiv T.holder.2.source) at h
    exact hfamily T.holder T.holder_mem
      (Option.map_injective (FaceEdgeDoubling.cellMap Delta f j hlen hf).indexEquiv.injective h)
  · exact FaceEdgeDoubling.regionFamilyOfArcs_noLoop Delta f j hlen hf _ T.avoid
      (fun b hb => hfamily b (Finset.mem_of_mem_erase hb)) ha'

/-! ## The correspondence of regions -/

/-- Where an old region goes. -/
noncomputable def stepMap (T : StepData Delta f j family) (a : family) :
    stepFamily Delta f j hlen hf T :=
  if h : a.val = T.holder then ⟨holding Delta f j hlen hf T, Finset.mem_insert_self _ _⟩
  else ⟨FaceEdgeDoubling.regionFamilyOfArcsEmbedding Delta f j hlen hf (family.erase T.holder)
      T.avoid ⟨a.val, Finset.mem_erase.mpr ⟨h, a.property⟩⟩,
    Finset.mem_insert_of_mem (Finset.mem_map_of_mem _ (Finset.mem_attach _ _))⟩

theorem stepMap_val_of_eq (T : StepData Delta f j family) {a : family} (h : a.val = T.holder) :
    (stepMap Delta f j hlen hf T a).val = holding Delta f j hlen hf T := by
  rw [stepMap, dif_pos h]

theorem stepMap_val_of_ne (T : StepData Delta f j family) {a : family} (h : a.val ≠ T.holder) :
    (stepMap Delta f j hlen hf T a).val =
      FaceEdgeDoubling.regionFamilyOfArcsEmbedding Delta f j hlen hf (family.erase T.holder) T.avoid
        ⟨a.val, Finset.mem_erase.mpr ⟨h, a.property⟩⟩ := by
  rw [stepMap, dif_neg h]

theorem stepMap_bijective (T : StepData Delta f j family) :
    Function.Bijective (stepMap Delta f j hlen hf T) := by
  refine ⟨fun a b hab => ?_, fun y => ?_⟩
  · have hv := congrArg Subtype.val hab
    by_cases ha : a.val = T.holder
    · by_cases hb : b.val = T.holder
      · exact Subtype.ext (ha.trans hb.symm)
      · rw [stepMap_val_of_eq Delta f j hlen hf T ha, stepMap_val_of_ne Delta f j hlen hf T hb] at hv
        have hmem : holding Delta f j hlen hf T ∈ carried Delta f j hlen hf T := by
          rw [hv]
          exact Finset.mem_map_of_mem _ (Finset.mem_attach _ _)
        exact absurd hmem (holding_not_mem_carried Delta f j hlen hf T)
    · by_cases hb : b.val = T.holder
      · rw [stepMap_val_of_ne Delta f j hlen hf T ha, stepMap_val_of_eq Delta f j hlen hf T hb] at hv
        have hmem : holding Delta f j hlen hf T ∈ carried Delta f j hlen hf T := by
          rw [← hv]
          exact Finset.mem_map_of_mem _ (Finset.mem_attach _ _)
        exact absurd hmem (holding_not_mem_carried Delta f j hlen hf T)
      · rw [stepMap_val_of_ne Delta f j hlen hf T ha, stepMap_val_of_ne Delta f j hlen hf T hb] at hv
        exact Subtype.ext (congrArg Subtype.val
          ((FaceEdgeDoubling.regionFamilyOfArcsEmbedding Delta f j hlen hf _ T.avoid).injective hv))
  · rcases Finset.mem_insert.mp y.property with hy | hy
    · refine ⟨⟨T.holder, T.holder_mem⟩, Subtype.ext ?_⟩
      rw [stepMap_val_of_eq Delta f j hlen hf T rfl]
      exact hy.symm
    · obtain ⟨b, -, hb⟩ := Finset.mem_map.mp hy
      refine ⟨⟨b.val, Finset.mem_of_mem_erase b.property⟩, Subtype.ext ?_⟩
      rw [stepMap_val_of_ne Delta f j hlen hf T (Finset.ne_of_mem_erase b.property)]
      exact hb

/-- **The old regions correspond to the new ones.** -/
noncomputable def stepEquiv (T : StepData Delta f j family) :
    family ≃ stepFamily Delta f j hlen hf T :=
  Equiv.ofBijective (stepMap Delta f j hlen hf T) (stepMap_bijective Delta f j hlen hf T)

theorem stepEquiv_profile (T : StepData Delta f j family) (a : family) :
    RegionCandidate.SameTargetProfile (stepEquiv Delta f j hlen hf T a).val a.val := by
  change RegionCandidate.SameTargetProfile (stepMap Delta f j hlen hf T a).val a.val
  by_cases ha : a.val = T.holder
  · rw [stepMap_val_of_eq Delta f j hlen hf T ha, ha]
    exact FaceEdgeDoubling.Holding.holdingCandidate_profile Delta f j hlen hf T.holder T.side
  · rw [stepMap_val_of_ne Delta f j hlen hf T ha]
    exact FaceEdgeDoubling.regionCandidateOfArcs_profile Delta f j hlen hf
      ⟨a.val, T.avoid a.val (Finset.mem_erase.mpr ⟨ha, a.property⟩)⟩

theorem stepEquiv_source (T : StepData Delta f j family) (a : family) :
    (stepEquiv Delta f j hlen hf T a).val.2.source.val = a.val.2.source.val := by
  change (stepMap Delta f j hlen hf T a).val.2.source.val = a.val.2.source.val
  by_cases ha : a.val = T.holder
  · rw [stepMap_val_of_eq Delta f j hlen hf T ha, ha]
    rfl
  · rw [stepMap_val_of_ne Delta f j hlen hf T ha]
    rfl

theorem stepEquiv_target (T : StepData Delta f j family) (a : family) :
    (stepEquiv Delta f j hlen hf T a).val.2.target.map Fin.val = a.val.2.target.map Fin.val := by
  change (stepMap Delta f j hlen hf T a).val.2.target.map Fin.val = a.val.2.target.map Fin.val
  by_cases ha : a.val = T.holder
  · rw [stepMap_val_of_eq Delta f j hlen hf T ha, ha]
    change Option.map Fin.val
        (Option.map (FaceEdgeDoubling.cellMap Delta f j hlen hf).indexEquiv T.holder.2.target) =
      Option.map Fin.val T.holder.2.target
    rw [Option.map_map]
    rfl
  · rw [stepMap_val_of_ne Delta f j hlen hf T ha]
    change Option.map Fin.val
        (Option.map (FaceEdgeDoubling.cellMap Delta f j hlen hf).indexEquiv a.val.2.target) =
      Option.map Fin.val a.val.2.target
    rw [Option.map_map]
    rfl

end Doubling

/-! ## The section family after the step -/

section Family

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {lambda c : ℝ} {eps : ℕ} {Delta : DiscDiagram.{u, w, v} W}
  {cuts : SectionCuts D lambda c Delta.boundaryWord}

/-- **A distinguished section family across the doubling of an edge between two regions.**  The
weight and the number of regions are unchanged and the labels stay legal, so the carried family is
again optimal. -/
noncomputable def sectionFamily
    (S : GloballyDistinguishedSectionFamily D lambda c eps Delta cuts)
    (f : S.diagram.toCombMap.Face) (j : Fin (S.diagram.faceBoundary f).darts.length)
    (hlen : 1 < (S.diagram.faceBoundary f).darts.length) (hf : f ≠ S.diagram.outerFace)
    (T : StepData S.diagram f j S.family) :
    GloballyDistinguishedSectionFamily D lambda c eps Delta cuts where
  diagram := FaceEdgeDoubling.diagram S.diagram f j hlen hf
  equiv := S.equiv.trans (FaceEdgeDoubling.oEquivalent S.diagram f j hlen hf)
  reduced := FaceEdgeDoubling.reduced S.diagram f j hlen hf S.reduced
  family := stepFamily S.diagram f j hlen hf T
  pairwise := stepFamily_pairwise S.diagram f j hlen hf T S.pairwise
  respects := by
    intro a ha
    obtain ⟨b, hb, hab⟩ := stepFamily_profile S.diagram f j hlen hf T ha
    exact RegionCandidate.respectsSections_of_sameTargetProfile cuts hab
      (stepFamily_noLoop S.diagram f j hlen hf T (fun x hx => (S.respects x hx).1) ha)
      (S.respects b hb)
  nondegenerate := by
    intro a ha
    obtain ⟨b, hb, hab⟩ := stepFamily_profile S.diagram f j hlen hf T ha
    obtain ⟨hs, ht⟩ := S.nondegenerate b hb
    exact ⟨lt_of_lt_of_eq hs hab.2.2.2.symm, lt_of_lt_of_eq ht hab.2.2.1.symm⟩
  label_admissible := FaceEdgeDoubling.label_admissible S.diagram f j hlen hf
    (symmetricLabelAlphabet D) (symmetricLabelAlphabet.symmetric D) S.label_admissible
  weight_maximal other hother :=
    (S.weight_maximal other hother).trans_eq (stepFamily_weight S.diagram f j hlen hf T).symm
  card_minimal other hother hweight :=
    (stepFamily_card S.diagram f j hlen hf T).trans_le
      (S.card_minimal other hother (hweight.trans (stepFamily_weight S.diagram f j hlen hf T)))

end Family
