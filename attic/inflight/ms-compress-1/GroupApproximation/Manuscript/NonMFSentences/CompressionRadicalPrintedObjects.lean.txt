variable {G : Type} [Group G]

/-! ## 1. The printed operator norm asymptotic representation -/

/-- **The printed operator norm asymptotic representation (tex 410–415).** The maps
`V_n : G → U(d_n)` are asymptotically multiplicative in operator norm and satisfy
`V_n(1) = 1`. -/
structure PrintedOpAsymptoticRepresentation (G : Type) [Group G]
    extends OpAlmostRepresentation G where
  /-- The printed normalization `V_n(1) = 1`. -/
  map_one : ∀ n, map n 1 = 1

open scoped Classical in
/-- Reset the value at `1` to the identity matrix. -/
def normalizeOneMap (B : OpAlmostRepresentation G) (n : ℕ) (g : G) :
    Matrix.unitaryGroup (B.model n) ℂ :=
  if g = 1 then 1 else B.map n g

theorem normalizeOneMap_one (B : OpAlmostRepresentation G) (n : ℕ) :
    normalizeOneMap B n 1 = 1 := by
  simp [normalizeOneMap]

theorem normalizeOneMap_of_ne (B : OpAlmostRepresentation G) (n : ℕ) {g : G}
    (hg : g ≠ 1) : normalizeOneMap B n g = B.map n g := by
  simp [normalizeOneMap, hg]

/-- The multiplicative defect of the normalized maps is at most the distance of `V_n(1)`
from `1` plus the original multiplicative defect. -/
theorem normalizeOne_defect_le (B : OpAlmostRepresentation G) (n : ℕ) (g h : G) :
    ‖(normalizeOneMap B n (g * h) : Matrix (B.model n) (B.model n) ℂ) -
        (normalizeOneMap B n g : Matrix (B.model n) (B.model n) ℂ) *
          (normalizeOneMap B n h : Matrix (B.model n) (B.model n) ℂ)‖ ≤
      ‖(B.map n 1 : Matrix (B.model n) (B.model n) ℂ) - 1‖ +
        ‖(B.map n (g * h) : Matrix (B.model n) (B.model n) ℂ) -
          (B.map n g : Matrix (B.model n) (B.model n) ℂ) *
            (B.map n h : Matrix (B.model n) (B.model n) ℂ)‖ := by
  by_cases hg : g = 1
  · subst hg
    simp only [one_mul, normalizeOneMap_one, OneMemClass.coe_one, sub_self, norm_zero]
    positivity
  · by_cases hh : h = 1
    · subst hh
      simp only [mul_one, normalizeOneMap_one, OneMemClass.coe_one, sub_self, norm_zero]
      positivity
    · by_cases hgh : g * h = 1
      · rw [hgh, normalizeOneMap_one, OneMemClass.coe_one, normalizeOneMap_of_ne B n hg,
          normalizeOneMap_of_ne B n hh]
        have htri := norm_sub_le_norm_sub_add_norm_sub
          (1 : Matrix (B.model n) (B.model n) ℂ)
          (B.map n 1 : Matrix (B.model n) (B.model n) ℂ)
          ((B.map n g : Matrix (B.model n) (B.model n) ℂ) *
            (B.map n h : Matrix (B.model n) (B.model n) ℂ))
        rwa [norm_sub_rev (1 : Matrix (B.model n) (B.model n) ℂ)
          (B.map n 1 : Matrix (B.model n) (B.model n) ℂ)] at htri
      · rw [normalizeOneMap_of_ne B n hgh, normalizeOneMap_of_ne B n hg,
          normalizeOneMap_of_ne B n hh]
        exact le_add_of_nonneg_left (norm_nonneg _)

/-- **Normalization at `1`.** An operator norm asymptotic representation with its value at
`1` reset to the identity is a printed one. -/
def normalizeOne (B : OpAlmostRepresentation G) : PrintedOpAsymptoticRepresentation G where
  model := B.model
  modelNonempty := B.modelNonempty
  map := normalizeOneMap B
  asymptoticallyMultiplicative g h ε hε := by
    have hlim := (opAlmost_tendsto_map_one_sub_one B).add (opAlmost_tendsto_mulDefect B g h)
    rw [add_zero] at hlim
    obtain ⟨N, hN⟩ := (Metric.tendsto_atTop.mp hlim) ε hε
    refine ⟨N, fun n hn ↦ ?_⟩
    have hb := hN n hn
    rw [Real.dist_eq, sub_zero] at hb
    exact (normalizeOne_defect_le B n g h).trans ((le_abs_self _).trans hb.le)
  map_one n := normalizeOneMap_one B n

/-- The normalized maps differ from the original ones by an operator norm null sequence. -/
theorem normalizeOne_map_sub_tendsto (B : OpAlmostRepresentation G) (g : G) :
    Tendsto (fun n ↦ ‖(normalizeOneMap B n g : Matrix (B.model n) (B.model n) ℂ) -
      (B.map n g : Matrix (B.model n) (B.model n) ℂ)‖) atTop (𝓝 0) := by
  by_cases hg : g = 1
  · subst hg
    have hfun : (fun n ↦ ‖(normalizeOneMap B n 1 : Matrix (B.model n) (B.model n) ℂ) -
        (B.map n 1 : Matrix (B.model n) (B.model n) ℂ)‖) =
        fun n ↦ ‖(B.map n 1 : Matrix (B.model n) (B.model n) ℂ) - 1‖ := by
      funext n
      rw [normalizeOneMap_one, OneMemClass.coe_one, norm_sub_rev]
    rw [hfun]
    exact opAlmost_tendsto_map_one_sub_one B
  · simp only [normalizeOneMap_of_ne B _ hg, sub_self, norm_zero]
    exact tendsto_const_nhds

/-- **The printed definition (tex 410–415) and its agreement with the repository notion.**
For every operator norm asymptotic representation `B`, `normalizeOne B` satisfies
`V_n(1) = 1` and has the same maps as `B` up to an operator norm null sequence. Conversely,
every printed representation is an `OpAlmostRepresentation` through the `extends`
projection. -/
def PrintedAsymptoticRepresentationNormalization : Prop :=
  ∀ (G : Type) [Group G] (B : OpAlmostRepresentation G),
    (∀ n, (normalizeOne B).map n 1 = 1) ∧
      ∀ g : G, Tendsto (fun n ↦
        ‖((normalizeOne B).map n g : Matrix (B.model n) (B.model n) ℂ) -
          (B.map n g : Matrix (B.model n) (B.model n) ℂ)‖) atTop (𝓝 0)

theorem printedAsymptoticRepresentationNormalization :
    PrintedAsymptoticRepresentationNormalization := by
  intro G _ B
  exact ⟨(normalizeOne B).map_one, normalizeOne_map_sub_tendsto B⟩

/-! ## 2. `Ad(V(g))` on the Hilbert--Schmidt bounded sequences -/

/-- Conjugation by a unitary preserves the normalized Hilbert--Schmidt norm coordinatewise. -/
theorem hsNorm_adSequence (B : OpAlmostRepresentation G) (g : G)
    (x : ∀ n, Matrix (B.model n) (B.model n) ℂ) (n : ℕ) :
    hsNorm (B.model n) (adSequence B g x n) = hsNorm (B.model n) (x n) := by
  have hV : (B.map n g : Matrix (B.model n) (B.model n) ℂ) ∈
      Matrix.unitaryGroup (B.model n) ℂ := (B.map n g).2
  have hVs : (B.map n g : Matrix (B.model n) (B.model n) ℂ)ᴴ ∈
      Matrix.unitaryGroup (B.model n) ℂ := by
    rw [← Matrix.star_eq_conjTranspose]
    exact Unitary.star_mem hV
  show hsNorm (B.model n)
      (applyOp (B.model n) (adMatrix (B.model n) (B.map n g)) (x n)) = _
  rw [applyOp_adMatrix]
  exact PrelimNotation.hsNorm_unitary_mul_mul (B.model n) hV hVs (x n)

/-- **The printed sentence (tex 434–436), at the printed domain.** `Ad(V(g))` maps the
Hilbert--Schmidt bounded matrix sequences bijectively onto themselves. -/
def PrintedAdBijectionOfHSBoundedSequences : Prop :=
  ∀ (G : Type) [Group G] (B : OpAlmostRepresentation G) (g : G),
    Set.BijOn (adSequence B g)
      {x | OneSidedMFRadical.TransportPrintedCommutant.IsHSBounded B x}
      {x | OneSidedMFRadical.TransportPrintedCommutant.IsHSBounded B x}

theorem printedAdBijectionOfHSBoundedSequences :
    PrintedAdBijectionOfHSBoundedSequences := by
  intro G _ B g
  have hbij := manuscriptSentence_adSequenceBijective G B g
  refine ⟨?_, fun _ _ _ _ hxy ↦ hbij.1 hxy, ?_⟩
  · rintro x ⟨C, hC⟩
    exact ⟨C, fun n ↦ (hsNorm_adSequence B g x n).trans_le (hC n)⟩
  · rintro y ⟨C, hC⟩
    obtain ⟨x, hx⟩ := hbij.2 y
    refine ⟨x, ⟨C, fun n ↦ ?_⟩, hx⟩
    rw [← hsNorm_adSequence B g x n, hx]
    exact hC n

/-! ## 3. `M_k(A)` is finite as the corona with matrix sizes `km_n` -/

/-- **Proof sentence, tex 455–456, second clause, along the printed route.** An isometry of
`M_k(A)` is unitary, because `M_k(A)` is the corona with matrix sizes `km_n`
(`matrixCoronaAmplificationEquiv`) and that corona is finite (`sentence_993df6a79c44`). -/
theorem printedMatrixCoronaFinite_viaAmplification (X : ℕ → FiniteModel)
    [∀ n, Nonempty (X n)] (k : ℕ) [Nonempty (Fin k)]
    (v : CStarMatrix (Fin k) (Fin k) (OneSidedMFRadical.stableFiniteCorona X))
    (hv : star v * v = 1) : v * star v = 1 := by
  let e : CStarMatrix (Fin k) (Fin k) (OneSidedMFRadical.stableFiniteCorona X) ≃⋆ₐ[ℂ]
      NormMatrixCStarCorona (fun n ↦ (Fin k × X n : Type)) :=
    MatrixCoronaAmplificationEmbedding.matrixCoronaAmplificationEquiv X
  have h1 : star (e v) * e v = 1 := by
    rw [← map_star, ← map_mul, hv, map_one]
  have h2 : e v * star (e v) = 1 := OneSidedMFRadical.sentence_993df6a79c44 X k (e v) h1
  apply e.injective
  rw [map_mul, map_star, h2, map_one]

/-- **The printed sentence (tex 455–456), both clauses.** The corona `A` is finite, and every
matrix algebra `M_k(A)` over it is finite, the latter through the amplification
isomorphism. -/
def PrintedCoronaStablyFiniteViaAmplification : Prop :=
  ∀ (X : ℕ → FiniteModel) [∀ n, Nonempty (X n)],
    IsCStarFinite (OneSidedMFRadical.stableFiniteCorona X) ∧
      IsCStarStablyFinite (OneSidedMFRadical.stableFiniteCorona X)

theorem printedCoronaStablyFiniteViaAmplification :
    PrintedCoronaStablyFiniteViaAmplification := by
  intro X _
  refine ⟨fun v hv ↦ OneSidedMFRadical.sentence_b21cf7f0f627 X hv, fun k hk v hv ↦ ?_⟩
  haveI : Nonempty (Fin k) := ⟨⟨0, hk⟩⟩
  let f : Matrix (Fin k) (Fin k) (OneSidedMFRadical.stableFiniteCorona X) ≃⋆ₐ[ℂ]
      CStarMatrix (Fin k) (Fin k) (OneSidedMFRadical.stableFiniteCorona X) :=
    CStarMatrix.ofMatrixStarAlgEquiv
  have h1 : star (f v) * f v = 1 := by
    rw [← map_star, ← map_mul, hv, map_one]
  have h2 := printedMatrixCoronaFinite_viaAmplification X k (f v) h1
  apply f.injective
  rw [map_mul, map_star, h2, map_one]

end
