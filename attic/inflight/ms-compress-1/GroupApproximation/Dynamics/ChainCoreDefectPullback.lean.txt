/-!
# Pulled-back symbolic defects cover `X ∖ Y`

`non_mf_groups_exist.tex`, Lemma `lem:chain-core-models` (tex 1373–1376) and its proof (tex 1456–1457):

> The complement $X\setminus Y$ is covered by translates of sets $P\setminus T(P)$, where $P$ is
> clopen and $T(P)\subseteq P$.

> Pull back the symbolic defects to cover $X\setminus Y$ by their translates.

The printed route. For a finite clopen partition `S` with itinerary map `π_S : X → X_S`, a
noncyclic edge `a → b` of a word graph of `X_S` gives chain-subshift's cylinder union
`P = reachableCylinderUnion X_S r b`, with `shift(P) ⊆ P` and `shift x ∈ P ∖ shift(P)` when `x`
starts with the edge. Its pullback `Q = π_S⁻¹(P)` is clopen, `T(Q) ⊆ Q` because `π_S ∘ T = shift ∘ π_S`,
and a noncyclic window of `π_S(x)` at position `i` puts `T^{i+1} x` in `Q ∖ T(Q)`. If `x ∉ Y_*`,
some itinerary `π_m(x)` leaves the core `Y_m`, so it has a noncyclic window
(`mem_coreSubshift_iff`). Finally `Y_* ⊆ Y` (ms-core-2's `generalCore_subset_chainRecurrentSet`),
so every `x ∉ Y` lies in a translate `T^{-(i+1)}(Q ∖ T(Q))`.

* `pulledBackDefect`, `isClopen_pulledBackDefect`, `mapsTo_pulledBackDefect`;
* `zpow_succ_mem_pulledBackDefect_diff`: the window step;
* `exists_mem_zpow_image_pulledBackDefect_diff`: `x ∉ Y_*` lies in a translate of a pulled-back defect;
* `compl_chainRecurrentSet_subset_pulledBackDefects`: `X ∖ Y` is covered (tex 1456);
* closed endpoints `printedPulledBackDefectsCover` (tex 1456, with clopenness, invariance and
  wandering of the pulled-back sets) and `printedChainCoreCoveringPrintedRoute` (tex 1373–1374, in
  the format of hull-euler's `PrintedChainCoreCovering`, proved through the pulled-back defects).
-/

section Pullback

variable {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X)

/-- Powers of `T.toEquiv` agree with powers of `T` pointwise. -/
theorem toEquiv_zpow_apply_pullback (j : ℤ) (x : X) : (T.toEquiv ^ j) x = (T ^ j) x := by
  let φ : (X ≃ₜ X) →* Equiv.Perm X :=
    { toFun := fun h ↦ h.toEquiv
      map_one' := rfl
      map_mul' := fun _ _ ↦ rfl }
  exact (congrArg (fun p : Equiv.Perm X ↦ p x) (map_zpow φ T j)).symm

theorem zpow_apply_zpow_apply_pullback (a b : ℤ) (x : X) :
    (T ^ a) ((T ^ b) x) = (T ^ (a + b)) x := by
  rw [zpow_add, Homeomorph.mul_apply]

/-- The pulled-back symbolic defect set `π_S⁻¹(P)`, for the cylinder union `P` of the vertices
reachable from `b` in the level-`r` word graph of the itinerary subshift `X_S`. -/
def pulledBackDefect (S : DiscreteQuotient X) (r : ℕ) (b : Fin r → S) : Set X :=
  itinerary T S ⁻¹' reachableCylinderUnion (itinerarySubshift T S) r b

/-- `π_S⁻¹(P)` is clopen. -/
theorem isClopen_pulledBackDefect (S : DiscreteQuotient X) (r : ℕ) (b : Fin r → S) :
    IsClopen (pulledBackDefect T S r b) := by
  have h : pulledBackDefect T S r b =
      itinerary T S ⁻¹' {y : ℤ → S | word y 0 r ∈
        {u | Relation.ReflTransGen (wordEdge (itinerarySubshift T S) r) b u}} := by
    ext x
    simp only [pulledBackDefect, reachableCylinderUnion, mem_preimage, mem_setOf_eq]
    exact and_iff_right (Set.mem_range_self (f := itinerary T S) x)
  rw [h]
  exact (isClopen_setOf_word_mem r _).preimage (continuous_itinerary T S)

/-- `T(π_S⁻¹(P)) ⊆ π_S⁻¹(P)`, since `π_S ∘ T = shift ∘ π_S` and `shift(P) ⊆ P`. -/
theorem mapsTo_pulledBackDefect (S : DiscreteQuotient X) (r : ℕ) (b : Fin r → S) :
    T '' pulledBackDefect T S r b ⊆ pulledBackDefect T S r b := by
  rintro _ ⟨x, hx, rfl⟩
  show itinerary T S (T x) ∈ reachableCylinderUnion (itinerarySubshift T S) r b
  rw [itinerary_apply_self]
  exact mapsTo_shift_reachableCylinderUnion (mapsTo_shift_itinerarySubshift T S) r b hx

/-- **The window step.** A noncyclic window of `π_S(x)` at position `i` puts `T^{i+1} x` in
`Q ∖ T(Q)` for the pulled-back defect `Q` of that edge. -/
theorem zpow_succ_mem_pulledBackDefect_diff (S : DiscreteQuotient X) {r : ℕ} {x : X} {i : ℤ}
    (hnc : ¬ IsCyclicEdge (itinerarySubshift T S) r (word (itinerary T S x) i (r + 1))) :
    (T ^ (i + 1)) x ∈
      pulledBackDefect T S r (Fin.tail (word (itinerary T S x) i (r + 1))) \
        T '' pulledBackDefect T S r (Fin.tail (word (itinerary T S x) i (r + 1))) := by
  have hy : itinerary T S ((T ^ i) x) ∈ itinerarySubshift T S :=
    Set.mem_range_self (f := itinerary T S) _
  have hyw : word (itinerary T S ((T ^ i) x)) 0 (r + 1) = word (itinerary T S x) i (r + 1) := by
    rw [itinerary_zpow_apply, word_shift, add_zero]
  have hwl : word (itinerary T S x) i (r + 1) ∈ language (itinerarySubshift T S) (r + 1) :=
    word_mem_language (mapsTo_shift_itinerarySubshift T S)
      (Set.mem_range_self (f := itinerary T S) x) i (r + 1)
  have hmem := shift_mem_diff_of_not_isCyclicEdge (mapsTo_shift_itinerarySubshift T S) hwl hnc
    hy hyw
  have hT : (T ^ (i + 1)) x = T ((T ^ i) x) := by
    rw [add_comm, ← zpow_apply_zpow_apply_pullback, zpow_one]
  refine ⟨?_, ?_⟩
  · show itinerary T S ((T ^ (i + 1)) x) ∈ _
    rw [hT, itinerary_apply_self]
    exact hmem.1
  · rintro ⟨z, hz, hzx⟩
    refine hmem.2 ⟨itinerary T S z, hz, ?_⟩
    have hz' : T z = T ((T ^ i) x) := hzx.trans hT
    rw [← itinerary_apply_self, hz', itinerary_apply_self]

/-- A point outside the inverse-limit core `Y_*` lies in a translate of a pulled-back defect. -/
theorem exists_mem_zpow_image_pulledBackDefect_diff {S : ℕ → DiscreteQuotient X} {x : X}
    (hx : x ∉ generalCore T S) :
    ∃ (m r : ℕ) (b : Fin r → S m) (n : ℤ),
      x ∈ ⇑(T.toEquiv ^ n) ''
        (pulledBackDefect T (S m) r b \ T '' pulledBackDefect T (S m) r b) := by
  obtain ⟨m, hm⟩ : ∃ m, itinerary T (S m) x ∉ itineraryCore T (S m) := by
    by_contra h
    push Not at h
    exact hx (show x ∈ inverseLimitCore T (fun m ↦ itineraryCore T (S m)) from
      (mem_inverseLimitCore T).2 h)
  obtain ⟨r, i, hnc⟩ : ∃ (r : ℕ) (i : ℤ),
      ¬ IsCyclicEdge (itinerarySubshift T (S m)) r (word (itinerary T (S m) x) i (r + 1)) := by
    by_contra h
    push Not at h
    exact hm (mem_coreSubshift_iff.2 h)
  refine ⟨m, r, _, -(i + 1), (T ^ (i + 1)) x, zpow_succ_mem_pulledBackDefect_diff T (S m) hnc, ?_⟩
  have h0 : -(i + 1) + (i + 1) = 0 := by omega
  rw [toEquiv_zpow_apply_pullback, zpow_apply_zpow_apply_pullback, h0, zpow_zero,
    Homeomorph.one_apply]

end Pullback

section Metric

variable {X : Type*} [MetricSpace X] [CompactSpace X] (T : X ≃ₜ X) {S : ℕ → DiscreteQuotient X}

/-- **tex 1456**, "Pull back the symbolic defects to cover $X\setminus Y$ by their translates":
for refining clopen partitions, `X ∖ CR(T)` is covered by translates of the pulled-back defects. -/
theorem compl_chainRecurrentSet_subset_pulledBackDefects (hS : IsRefiningClopenPartitions S) :
    (chainRecurrentSet T)ᶜ ⊆ ⋃ (m : ℕ) (r : ℕ) (b : Fin r → S m) (n : ℤ),
      ⇑(T.toEquiv ^ n) '' (pulledBackDefect T (S m) r b \ T '' pulledBackDefect T (S m) r b) := by
  intro x hx
  obtain ⟨m, r, b, n, h⟩ := exists_mem_zpow_image_pulledBackDefect_diff T
    (fun hg ↦ hx (generalCore_subset_chainRecurrentSet T hS hg))
  exact mem_iUnion.2 ⟨m, mem_iUnion.2 ⟨r, mem_iUnion.2 ⟨b, mem_iUnion.2 ⟨n, h⟩⟩⟩⟩

end Metric

/-- **The printed sentence (tex 1456), with the properties the lemma asserts of the sets used.**
For a compact metric space, refining finite clopen partitions `S`, and `T`, the translates of the
pulled-back symbolic defects `π_m⁻¹(P) ∖ T(π_m⁻¹(P))` cover `X ∖ CR(T)`; every pulled-back set is
clopen with `T(Q) ⊆ Q`, and `Q ∖ T(Q)` is wandering. -/
def PrintedPulledBackDefectsCover : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] (T : X ≃ₜ X) (S : ℕ → DiscreteQuotient X),
    IsRefiningClopenPartitions S →
      (chainRecurrentSet T)ᶜ ⊆ (⋃ (m : ℕ) (r : ℕ) (b : Fin r → S m) (n : ℤ),
          ⇑(T.toEquiv ^ n) '' (pulledBackDefect T (S m) r b \ T '' pulledBackDefect T (S m) r b)) ∧
        ∀ (m r : ℕ) (b : Fin r → S m),
          IsClopen (pulledBackDefect T (S m) r b) ∧
            T '' pulledBackDefect T (S m) r b ⊆ pulledBackDefect T (S m) r b ∧
              IsWandering T.toEquiv
                (pulledBackDefect T (S m) r b \ T '' pulledBackDefect T (S m) r b)

theorem printedPulledBackDefectsCover : PrintedPulledBackDefectsCover := by
  intro X _ _ T S hS
  refine ⟨compl_chainRecurrentSet_subset_pulledBackDefects T hS, fun m r b ↦ ?_⟩
  exact ⟨isClopen_pulledBackDefect T (S m) r b, mapsTo_pulledBackDefect T (S m) r b,
    isWandering_diff_image (mapsTo_pulledBackDefect T (S m) r b)⟩

/-- **lem:chain-core-models, tex 1373–1374, along the printed proof.** "The complement
$X\setminus Y$ is covered by translates of sets $P\setminus T(P)$, where $P$ is clopen and
$T(P)\subseteq P$", for a compact totally disconnected metric space, with the sets `P` the pulled-back
symbolic defects of refining clopen partitions. -/
def PrintedChainCoreCoveringPrintedRoute : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] (T : X ≃ₜ X),
    (chainRecurrentSet T)ᶜ ⊆ ⋃ P ∈ {P : Set X | IsClopen P ∧ T '' P ⊆ P},
      ⋃ n : ℤ, ⇑(T.toEquiv ^ n) '' (P \ T '' P)

theorem printedChainCoreCoveringPrintedRoute : PrintedChainCoreCoveringPrintedRoute := by
  intro X _ _ _ T x hx
  obtain ⟨S, hS⟩ := exists_refiningClopenPartitions (X := X)
  obtain ⟨m, r, b, n, h⟩ := exists_mem_zpow_image_pulledBackDefect_diff T
    (fun hg ↦ hx (generalCore_subset_chainRecurrentSet T hS hg))
  exact mem_iUnion₂.2 ⟨pulledBackDefect T (S m) r b,
    ⟨isClopen_pulledBackDefect T (S m) r b, mapsTo_pulledBackDefect T (S m) r b⟩,
    mem_iUnion.2 ⟨n, h⟩⟩

end ChainCore
