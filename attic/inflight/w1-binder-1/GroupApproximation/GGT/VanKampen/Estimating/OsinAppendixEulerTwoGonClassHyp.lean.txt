/-!
# The class and walk hypotheses of a two-gon pocket

Osin, arXiv:math/0411039v3, Appendix, proof of Lemma 9.3: "inside every 2-gon of `Φ'_M`, there is
a vertex of `Φ_M`".  `ExtPhiData.cell_face_not_mem_sideFaces_of_twoGon` (`hno`) and
`ExtPhiData.disjoint_sideFaces_of_twoGon` (`havoid`) take two hypotheses, `TwoGonClassHyp` on the
face class of the two-gon and `TwoGonWalkHyp` on a closed walk around its pocket.  This module
produces the first, and the fields of the second that depend only on the walk.

* `faceClassO_facePerm`: consecutive darts of a face of `phiMapO` lie in one face class.
* `twoGonClassHyp_of_outer`: `TwoGonClassHyp` at the outer cell dart `y` of a two-gon, its region
  `a`, the region `b` two steps on and the cell `i` of `a`, when the face holds no cell off the
  ends of `E` and `E` is closed under shared ends.
* `twoGonClassHyp_of_linkedComponent`: the same for the component of an exterior region, the
  setting of `TwoGonHoldsInput`.
* `twoGonWalk_left_side`, `twoGonWalk_right_side`, `twoGonWalk_base`: the fields `left_side`,
  `right_side` and `base` of `TwoGonWalkHyp` on the walk
  `source.reverseDarts ++ a.rightSide ++ target.darts ++ b.leftSide`, when the source arc spans
  the source arcs of `a` and `b` and the target arc spans their target arcs.

## Manuscript status

Infrastructure for `thm:hull` (tex 2121 at origin/main 68481e4d7, through Osin's Lemma 9.7(a));
certifies no printed sentence on its own.
-/

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}

  {family E : Finset (RegionCandidate D eps Delta)}

/-- **Consecutive darts of a face of `Φ'_M` lie in one face class.**  The face rotation of
`phiMapO` returns to a retained dart through darts that are not retained. -/
theorem faceClassO_facePerm (y : (phiMapO family E).Dart) :
    FaceClassO family E y.1 ((phiMapO family E).facePerm y).1 := by
  have hR := phiMapO_isRestriction family E
  obtain ⟨m, hm, hpm, havoid⟩ := hR.sigma_firstReturn ((phiMapO family E).alpha y)
  have halpha : (Function.Embedding.subtype (PhiKeepO family E)) ((phiMapO family E).alpha y) =
      (collapsedMap family).dual.alpha y.1 := hR.alpha_map y
  rw [halpha] at hpm havoid
  have h := SimpleClosedWalkSides.faceClass_run (collapsedMap family).dual (PhiKeepO family E)
    y.1 m (fun k hk0 hkm hk => havoid k hk0 hkm ⟨⟨_, hk⟩, rfl⟩) m hm le_rfl
  rw [hpm] at h
  exact h

/-- **L4: the class hypotheses of a two-gon.**  Let `y` be the outer cell dart of a two-gon of
`phiMapO`, with region `a`, and let `b` be the region two steps on, an exterior region of the cell
`i` of `a`.  If the face holds no cell off the ends of `E`, and every region of the family sharing
an end with a region of `E` is in `E`, then the face class of `y` satisfies `TwoGonClassHyp`. -/
theorem twoGonClassHyp_of_outer (P : ExtPhiData family E)
    (hlinked : ∀ a ∈ E, ∀ b ∈ E, Relation.ReflTransGen (SharesEndO E) a b)
    (hclosed : ∀ a ∈ E, ∀ r ∈ family, ∀ s t : Bool, sideCellO a s = sideCellO r t → r ∈ E)
    (hnondeg : ∀ r ∈ family, 0 < r.2.sourceArc.length ∧ 0 < r.2.targetArc.length)
    {f : (phiMapO family E).Face} (hcell : ¬ HoldsCellO family E f)
    {y : (phiMapO family E).Dart} (hy : phiCellSideO y = true)
    (hyf : (phiMapO family E).faceOf y = f)
    (hnone : sideCellO (phiRegionO y) (phiSideO y) = none)
    (h4 : (phiMapO family E).facePerm ((phiMapO family E).facePerm
      ((phiMapO family E).facePerm ((phiMapO family E).facePerm y))) = y)
    (hb : phiRegionO ((phiMapO family E).facePerm ((phiMapO family E).facePerm y)) ∈
      exteriorAt family (phiRegionO y).2.source) :
    TwoGonClassHyp family E y.1 (phiRegionO y)
      (phiRegionO ((phiMapO family E).facePerm ((phiMapO family E).facePerm y)))
      (phiRegionO y).2.source := by
  obtain ⟨hkind, _hbsrc⟩ := Finset.mem_filter.mp hb
  obtain ⟨_hbfam, hbt⟩ := Finset.mem_filter.mp hkind
  have ht2 : (phiRegionO ((phiMapO family E).facePerm
      ((phiMapO family E).facePerm y))).2.target = none := by
    cases hT : (phiRegionO ((phiMapO family E).facePerm
        ((phiMapO family E).facePerm y))).2.target with
    | none => rfl
    | some j =>
      rw [hT] at hbt
      simp at hbt
  have hside : phiSideO y = false := by
    cases hs : phiSideO y with
    | false => rfl
    | true =>
      rw [hs] at hnone
      simp [sideCellO] at hnone
  have ht : (phiRegionO y).2.target = none := by
    rw [hside] at hnone
    exact hnone
  have hy2 := sideCellO_facePerm_facePerm P y hy
  have hside2 : phiSideO ((phiMapO family E).facePerm ((phiMapO family E).facePerm y)) = true := by
    cases hs : phiSideO ((phiMapO family E).facePerm ((phiMapO family E).facePerm y)) with
    | true => rfl
    | false =>
      exfalso
      have h := hy2.2
      rw [hs, hside] at h
      have h' : (phiRegionO ((phiMapO family E).facePerm
          ((phiMapO family E).facePerm y))).2.target = some (phiRegionO y).2.source := h
      rw [ht2] at h'
      cases h'
  have hstart_left : (phiRegionO y).2.targetArc.darts.head? =
      some (Delta.toCombMap.alpha y.1.1) := by
    have hc := phiO_cross y
    rw [hside, hy, crossO_false_of_eq_none _ ht, Bool.not_true, cross_false,
      Option.map_eq_some_iff] at hc
    obtain ⟨p, hp, hpy⟩ := hc
    simp only [cross, if_true] at hp
    rw [hp, ← hpy, Delta.toCombMap.alpha_involutive p]
  have hstart_right :
      (phiRegionO ((phiMapO family E).facePerm
        ((phiMapO family E).facePerm y))).2.sourceArc.darts.head? =
        some ((phiMapO family E).facePerm ((phiMapO family E).facePerm y)).1.1 := by
    have hc := phiO_cross ((phiMapO family E).facePerm ((phiMapO family E).facePerm y))
    rw [hside2, hy2.1, crossO_true] at hc
    simpa only [cross, if_true] using hc
  exact
    { mem_left := P.subset (phiRegionO_mem y)
      mem_right := P.subset (phiRegionO_mem _)
      target_left := ht
      target_right := ht2
      kept_faces := fun u hxu hk =>
        P.faceOf_of_faceClassO_twoGon hlinked hy hnone h4 (z := ⟨u, hk⟩) hxu
      selected_of_source := fun u hxu r hr hface => by
        by_contra hrE
        apply hcell
        refine ⟨r.2.source, ?_, y, hyf, u, hxu, hface⟩
        intro hmem
        obtain ⟨a', ha', s, hs⟩ := mem_endCellsO.mp hmem
        exact hrE (hclosed a' ha' r hr s true hs)
      nondegenerate := hnondeg
      start_left := ⟨y.1, hstart_left, Relation.EqvGen.refl _⟩
      start_right := ⟨_, hstart_right,
        Relation.EqvGen.trans _ _ _ (faceClassO_facePerm y) (faceClassO_facePerm _)⟩ }

/-- **L4 at the component of an exterior region**, the setting of `TwoGonHoldsInput`.  The
hypotheses on `y` are those given by `exists_twoGonLabels`. -/
theorem twoGonClassHyp_of_linkedComponent {lambda c : ℝ} {Delta₀ : DiscDiagram.{u, w, v} W}
    {cuts : SectionCuts D lambda c Delta₀.boundaryWord}
    (S : GloballyDistinguishedSectionFamily D lambda c eps Delta₀ cuts)
    {a₀ : RegionCandidate D eps S.diagram}
    (P : ExtPhiData S.family (linkedComponentO S.family a₀))
    {f : (phiMapO S.family (linkedComponentO S.family a₀)).Face}
    (hcell : ¬ HoldsCellO S.family (linkedComponentO S.family a₀) f)
    {y : (phiMapO S.family (linkedComponentO S.family a₀)).Dart} (hy : phiCellSideO y = true)
    (hyf : (phiMapO S.family (linkedComponentO S.family a₀)).faceOf y = f)
    (hnone : sideCellO (phiRegionO y) (phiSideO y) = none)
    (h4 : (phiMapO S.family (linkedComponentO S.family a₀)).facePerm
      ((phiMapO S.family (linkedComponentO S.family a₀)).facePerm
        ((phiMapO S.family (linkedComponentO S.family a₀)).facePerm
          ((phiMapO S.family (linkedComponentO S.family a₀)).facePerm y))) = y)
    (hb : phiRegionO ((phiMapO S.family (linkedComponentO S.family a₀)).facePerm
        ((phiMapO S.family (linkedComponentO S.family a₀)).facePerm y)) ∈
      exteriorAt S.family (phiRegionO y).2.source) :
    TwoGonClassHyp S.family (linkedComponentO S.family a₀) y.1 (phiRegionO y)
      (phiRegionO ((phiMapO S.family (linkedComponentO S.family a₀)).facePerm
        ((phiMapO S.family (linkedComponentO S.family a₀)).facePerm y)))
      (phiRegionO y).2.source :=
  twoGonClassHyp_of_outer P (linked_linkedComponentO _ _)
    (fun _ ha _ hr _ _ hst => mem_linkedComponentO_of_sideCellO_eq ha hr hst)
    S.nondegenerate hcell hy hyf hnone h4 hb

/-- **L5a: off the walk, a boundary dart of `a` lies on its left side.**  The walk holds the
reversed source span, the right side of `a` and the target span, which hold the other three parts
of the boundary of `a`. -/
theorem twoGonWalk_left_side {a b : RegionCandidate D eps Delta} {i : Fin Delta.rCellCount}
    (source : CyclicArc (cellDarts Delta i)) (target : CyclicArc (targetDarts Delta none))
    (ht : a.2.target = none)
    (hsrc : ∀ d ∈ a.2.sourceArc.darts, d ∈ source.darts)
    (htgt : ∀ d ∈ a.2.targetArc.darts, d ∈ target.darts) :
    ∀ d ∈ a.2.boundary.cycle, ¬ walkKeep Delta.toCombMap
        (source.reverseDarts ++ a.2.rightSide ++ target.darts ++ b.2.leftSide) d →
      d ∈ a.2.leftSide := by
  intro d hd hkeep
  have hwalk : ∀ x, x ∈ source.reverseDarts ++ a.2.rightSide ++ target.darts ++ b.2.leftSide →
      walkKeep Delta.toCombMap
        (source.reverseDarts ++ a.2.rightSide ++ target.darts ++ b.2.leftSide) x :=
    fun _ hx => Or.inl hx
  rw [a.2.boundary_decomposition, targetBoundaryDarts_eq_darts_of_none Delta _ _ ht] at hd
  simp only [List.mem_append] at hd
  rcases hd with ((hs | hr) | htg) | hl
  · simp only [CyclicArc.reverseDarts, List.mem_map, List.mem_reverse] at hs
    obtain ⟨p, hp, rfl⟩ := hs
    refine absurd (hwalk _ ?_) hkeep
    simp only [List.mem_append, CyclicArc.reverseDarts, List.mem_map, List.mem_reverse]
    exact Or.inl (Or.inl (Or.inl ⟨p, hsrc p hp, rfl⟩))
  · refine absurd (hwalk _ ?_) hkeep
    simp only [List.mem_append]
    exact Or.inl (Or.inl (Or.inr hr))
  · refine absurd (hwalk _ ?_) hkeep
    simp only [List.mem_append]
    exact Or.inl (Or.inr (htgt d htg))
  · exact hl

/-- **L5a: off the walk, a boundary dart of `b` lies on its right side.** -/
theorem twoGonWalk_right_side {a b : RegionCandidate D eps Delta} {i : Fin Delta.rCellCount}
    (source : CyclicArc (cellDarts Delta i)) (target : CyclicArc (targetDarts Delta none))
    (ht : b.2.target = none)
    (hsrc : ∀ d ∈ b.2.sourceArc.darts, d ∈ source.darts)
    (htgt : ∀ d ∈ b.2.targetArc.darts, d ∈ target.darts) :
    ∀ d ∈ b.2.boundary.cycle, ¬ walkKeep Delta.toCombMap
        (source.reverseDarts ++ a.2.rightSide ++ target.darts ++ b.2.leftSide) d →
      d ∈ b.2.rightSide := by
  intro d hd hkeep
  have hwalk : ∀ x, x ∈ source.reverseDarts ++ a.2.rightSide ++ target.darts ++ b.2.leftSide →
      walkKeep Delta.toCombMap
        (source.reverseDarts ++ a.2.rightSide ++ target.darts ++ b.2.leftSide) x :=
    fun _ hx => Or.inl hx
  rw [b.2.boundary_decomposition, targetBoundaryDarts_eq_darts_of_none Delta _ _ ht] at hd
  simp only [List.mem_append] at hd
  rcases hd with ((hs | hr) | htg) | hl
  · simp only [CyclicArc.reverseDarts, List.mem_map, List.mem_reverse] at hs
    obtain ⟨p, hp, rfl⟩ := hs
    refine absurd (hwalk _ ?_) hkeep
    simp only [List.mem_append, CyclicArc.reverseDarts, List.mem_map, List.mem_reverse]
    exact Or.inl (Or.inl (Or.inl ⟨p, hsrc p hp, rfl⟩))
  · exact hr
  · refine absurd (hwalk _ ?_) hkeep
    simp only [List.mem_append]
    exact Or.inl (Or.inr (htgt d htg))
  · refine absurd (hwalk _ ?_) hkeep
    simp only [List.mem_append]
    exact Or.inr hl

/-- **L5a: the walk holds a dart of `a`**: the first target dart of `a`, in the target span. -/
theorem twoGonWalk_base {a b : RegionCandidate D eps Delta} {i : Fin Delta.rCellCount}
    (source : CyclicArc (cellDarts Delta i)) (target : CyclicArc (targetDarts Delta none))
    (ht : a.2.target = none)
    (hnondeg : 0 < a.2.sourceArc.length ∧ 0 < a.2.targetArc.length)
    (htgt : ∀ d ∈ a.2.targetArc.darts, d ∈ target.darts) :
    ∃ d ∈ source.reverseDarts ++ a.2.rightSide ++ target.darts ++ b.2.leftSide,
      Delta.toCombMap.faceOf d ∈ a.1 := by
  obtain ⟨d, hd⟩ := exists_cross a hnondeg false true
  simp only [cross, if_true] at hd
  have hmem : d ∈ a.2.targetArc.darts := List.mem_of_mem_head? hd
  refine ⟨d, ?_, ?_⟩
  · simp only [List.mem_append]
    exact Or.inl (Or.inr (htgt d hmem))
  · have hcyc : d ∈ a.2.boundary.cycle := by
      rw [a.2.boundary_decomposition]
      simp only [List.mem_append]
      exact Or.inl (Or.inr
        (mem_targetBoundaryDarts_of_eq_none Delta a.2.target a.2.targetArc ht hmem))
    exact ((a.2.boundary.cycle_mem_iff d).mp hcyc).1
