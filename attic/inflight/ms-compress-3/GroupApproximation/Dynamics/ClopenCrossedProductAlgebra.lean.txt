/-!
# `R_X` as a `k`-algebra, and the remaining kernel at rank two

`non_mf_groups_exist.tex`, `sec:chain-core`.

**Locally matricial algebras (tex 1465–1467).**

> Call an algebra *locally matricial* if every finite subset is contained in a finite product of
> full matrix algebras over its coefficient field; these subalgebras may have their own local
> identities.

`R_X = LC(X, k) ⋊_T ℤ` is an algebra over its coefficient field `k`, and `lem:transient-matrices`
(tex 1469–1479) asserts that the transient ideal `I` is locally matricial in this sense.  The
interface module `Dynamics/ClopenCrossedProduct` records `R_X` only as a ring:
`ClopenCoeff T k` is a type synonym carrying `Ring` and the action of `ℤ`, so no `k`-module
structure reached `R_X`, and `Algebra/LocallyMatricial`'s `IsLocallyMatricialIn k` could not be
stated for the transient ideal.  This module supplies the missing structure:

* `ClopenCoeff.instAlgebra`: `c • f = const c * f`, and the `ℤ`-action commutes with scalars
  (`ClopenCoeff.smul_algebraMap`, `ClopenCoeff.instSMulCommClass`).  Mathlib's instance on
  `SkewMonoidAlgebra` then makes `R_X` a `k`-algebra, with `IsScalarTower k R_X R_X` and
  `SMulCommClass k R_X R_X`, as `IsLocallyMatricialIn` and the matrix-unit criterion
  `isLocallyMatricialIn_of_forall_exists_family` require;
* `ClopenCrossedProduct.algebraMap_eq_coeff`: the scalars are the constant functions;
* `ClopenCrossedProduct.restrict_algebraMap`, `restrict_smul`, `restrictAlgHom`: restriction
  `R_X → R_Y` is a `k`-algebra map, and `smul_mem_transientIdeal`: `I` is a `k`-subspace.

**The remaining kernel at rank two (tex 1799).**

> The remaining kernel is locally finite, [...]

`ChainCore.TransientIdealLocallyMatricialStatement` is the locally matricial clause of
`lem:transient-matrices` (tex 1476) for `Y = CR(T)`, over a finite field.  Over it,
`RankTwoLimit.coreRestrictionLocallyFiniteKernel_of_transientLocallyMatricial` discharges the
residual step `CoreRestrictionLocallyFiniteKernelStatement` of the rank-two remark, since finitely
many matrix algebras over a finite field are finite (`hasLocallyFiniteKernel_of_isLocallyMatricialIn`).
The producer of the statement is `lem:transient-matrices` itself (chain-matricial,
`isLocallyMatricialIn_transientIdeal`).
-/

namespace GroupApproximation

open Multiplicative (ofAdd toAdd)

namespace ClopenCoeff

variable {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type*) [CommRing k]

/-- **`LC(X, k)` is a `k`-algebra**: a scalar `c` acts as the constant function `c`. -/
noncomputable instance instAlgebra : Algebra k (ClopenCoeff T k) :=
  RingHom.toAlgebra'
    ((ClopenCoeff.of T k).toRingHom.comp (algebraMap k (LocallyConstant X k)))
    (fun c x ↦ (ClopenCoeff.of T k).symm.injective (by rw [map_mul, map_mul, mul_comm]))

/-- The action of `ℤ` through `T` fixes the constant functions. -/
theorem smul_algebraMap (j : Multiplicative ℤ) (c : k) :
    j • algebraMap k (ClopenCoeff T k) c = algebraMap k (ClopenCoeff T k) c :=
  LocallyConstant.ext fun _ ↦ rfl

/-- The action of `ℤ` through `T` commutes with `k`-scalars. -/
instance instSMulCommClass : SMulCommClass (Multiplicative ℤ) k (ClopenCoeff T k) where
  smul_comm _ _ _ := LocallyConstant.ext fun _ ↦ rfl

end ClopenCoeff

namespace ClopenCrossedProduct

section Algebra

variable {X : Type*} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type*) [CommRing k]

example : Nonempty (Algebra k (ClopenCrossedProduct T k)) := ⟨inferInstance⟩

example : IsScalarTower k (ClopenCrossedProduct T k) (ClopenCrossedProduct T k) := inferInstance

example : SMulCommClass k (ClopenCrossedProduct T k) (ClopenCrossedProduct T k) := inferInstance

/-- The scalars of `R_X` are the constant functions `C(const c)`. -/
theorem algebraMap_eq_coeff (c : k) :
    algebraMap k (ClopenCrossedProduct T k) c = coeff T k (LocallyConstant.const X c) :=
  rfl

variable {Y : Set X}

/-- Restriction `R_X → R_Y` fixes the scalars. -/
theorem restrict_algebraMap (hY : T '' Y = Y) (c : k) :
    restrict T k hY (algebraMap k (ClopenCrossedProduct T k) c) =
      algebraMap k (ClopenCrossedProduct (restrictHomeo T hY) k) c :=
  (Pestov91.CrossedProduct.coeffMap_single _ _ (1 : Multiplicative ℤ) _).trans
    (congrArg (SkewMonoidAlgebra.single (1 : Multiplicative ℤ)) (LocallyConstant.ext fun _ ↦ rfl))

/-- Restriction `R_X → R_Y` is `k`-linear. -/
theorem restrict_smul (hY : T '' Y = Y) (c : k) (x : ClopenCrossedProduct T k) :
    restrict T k hY (c • x) = c • restrict T k hY x := by
  rw [Algebra.smul_def, Algebra.smul_def, map_mul, restrict_algebraMap]

/-- **Restriction as a `k`-algebra map** `R_X →ₐ[k] R_Y`. -/
noncomputable def restrictAlgHom (hY : T '' Y = Y) :
    ClopenCrossedProduct T k →ₐ[k] ClopenCrossedProduct (restrictHomeo T hY) k :=
  { restrict T k hY with commutes' := restrict_algebraMap T k hY }

theorem restrictAlgHom_apply (hY : T '' Y = Y) (x : ClopenCrossedProduct T k) :
    restrictAlgHom T k hY x = restrict T k hY x :=
  rfl

/-- The transient ideal is closed under `k`-scalars. -/
theorem smul_mem_transientIdeal (hY : T '' Y = Y) (c : k) {x : ClopenCrossedProduct T k}
    (hx : x ∈ transientIdeal T k hY) : c • x ∈ transientIdeal T k hY := by
  rw [mem_transientIdeal] at hx ⊢
  rw [restrict_smul, hx, smul_zero]

end Algebra

end ClopenCrossedProduct

namespace ChainCore

open Dynamics

/-- **`lem:transient-matrices`, the locally matricial clause (tex 1476).**  "The ideal $I$ is
locally matricial": for a compact metrizable zero-dimensional `X`, a homeomorphism `T` and a finite
field `k`, the kernel `I` of restriction to `Y = CR(T)` is locally matricial over `k`. -/
def TransientIdealLocallyMatricialStatement : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] (T : X ≃ₜ X)
    (k : Type) [Field k] [Finite k],
    IsLocallyMatricialIn k
      ((ClopenCrossedProduct.transientIdeal T k (image_chainRecurrentSet T) :
          TwoSidedIdeal (ClopenCrossedProduct T k)) : Set (ClopenCrossedProduct T k))

namespace RankTwoLimit

/-- **The remaining kernel (tex 1799)**, over `lem:transient-matrices`: the kernel of core
restriction is locally finite, since it is locally matricial over a finite field. -/
theorem coreRestrictionLocallyFiniteKernel_of_transientLocallyMatricial
    (h : TransientIdealLocallyMatricialStatement) : CoreRestrictionLocallyFiniteKernelStatement := by
  intro X _ _ _ T k _ _
  refine hasLocallyFiniteKernel_of_isLocallyMatricialIn (k := k) _ ?_
  have hset : {x : ClopenCrossedProduct T k |
      ClopenCrossedProduct.restrict T k (image_chainRecurrentSet T) x = 0} =
      ((ClopenCrossedProduct.transientIdeal T k (image_chainRecurrentSet T) :
          TwoSidedIdeal (ClopenCrossedProduct T k)) : Set (ClopenCrossedProduct T k)) := by
    ext x
    exact (ClopenCrossedProduct.mem_transientIdeal T k (image_chainRecurrentSet T)).symm
  rw [hset]
  exact h X T k

end RankTwoLimit

end ChainCore

end GroupApproximation
