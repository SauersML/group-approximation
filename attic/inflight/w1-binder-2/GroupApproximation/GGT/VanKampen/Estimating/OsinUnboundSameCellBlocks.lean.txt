/-!
# Same-cell unbound darts: the scan along one relator cell

Take a relator cell `i` of a disc diagram and a family of regions.  An unbound dart of the cell
whose reverse also lies on the cell sits at one end of a pocket.  This module applies the scan of
`OsinUnboundSameCellScanBound` to the cell word.
* `not_mem_boundDarts`, `mem_unboundDarts`: a dart of the cell with its own cell across faces no
  region, since a region's faces are `G`-faces and a relator word does not have value one.
* The base is an oriented same-cell dart with the longest pocket (`pocketLen`).  Blocks join an
  oriented same-cell dart to its reverse.  `AvoidEdgeStep.lt_of_oriented` puts every block in
  order, `AvoidEdgeStep.false_of_cross` and `not_oriented_both` nest them, and the pocket value
  clause (`PocketValue`) gives each block value one.
* `exists_base_scan`: `λ · #same ≤ (1 - λ) · #other + c · (1 + #run starts)` from some base.

## Manuscript status

Infrastructure for `thm:hull` ("Hull's small cancellation theorem", through Osin's Lemma 9.4);
certifies no printed sentence on its own.
-/

/-- **A dart with its own cell across is bound by no region.** -/
theorem not_mem_boundDarts {X : DiscDiagram.{u, w, v} W}
    (hvalue : ∀ C ∈ X.relatorCells, RelLetter.listVal C.word ≠ 1)
    (family : Finset (RegionCandidate D eps X)) {i : Fin X.rCellCount} {e : X.toCombMap.Dart}
    (he : X.toCombMap.faceOf (X.toCombMap.alpha e) = (cell X i).face) (interior : Bool) :
    e ∉ RegionCandidate.boundDarts family interior i := by
  intro h
  simp only [RegionCandidate.boundDarts, Finset.mem_biUnion] at h
  obtain ⟨a, -, ha⟩ := h
  have hmem := a.faceOf_alpha_mem_of_mem_cellArcDarts i e ha
  rw [he] at hmem
  exact hvalue (cell X i) (cell_mem X i) (by
    rw [← dartWord_cellDarts X i]
    exact (a.2.boundary.all_gCells _ hmem).2)

/-- **A dart of the cell with its own cell across is unbound.** -/
theorem mem_unboundDarts {X : DiscDiagram.{u, w, v} W}
    (hvalue : ∀ C ∈ X.relatorCells, RelLetter.listVal C.word ≠ 1)
    (family : Finset (RegionCandidate D eps X)) {i : Fin X.rCellCount} {e : X.toCombMap.Dart}
    (hmem : e ∈ cellDarts X i)
    (he : X.toCombMap.faceOf (X.toCombMap.alpha e) = (cell X i).face) :
    e ∈ RegionCandidate.unboundDarts family i := by
  simp only [RegionCandidate.unboundDarts, Finset.mem_sdiff, Finset.mem_union, List.mem_toFinset]
  exact ⟨hmem, fun h => h.elim (not_mem_boundDarts hvalue family he false)
    (not_mem_boundDarts hvalue family he true)⟩

/-- **The pocket value clause on one diagram**: the tail of `OsinLemma94SameCellPocketInput`. -/
def PocketValue (X : DiscDiagram.{u, w, v} W) : Prop :=
  ∀ (i : Fin X.rCellCount) (d : X.toCombMap.Dart),
    d ∈ cellDarts X i →
    X.toCombMap.faceOf (X.toCombMap.alpha d) = (cell X i).face →
    (∃ o : X.toCombMap.Dart, X.toCombMap.faceOf o = X.outerFace ∧
      Relation.EqvGen (AvoidEdgeStep X.toCombMap d) d o) →
    ∀ m : ℕ, m + 1 < (cellDarts X i).length →
      (X.toCombMap.facePerm ^ (m + 1)) d = X.toCombMap.alpha d →
        RelLetter.listVal (dartWord X
          ((List.range m).map fun t => (X.toCombMap.facePerm ^ (t + 1)) d)) = 1

/-- The number of face steps from a dart to its reverse, when there is one. -/
noncomputable def pocketLen (M : CombMap.{v}) (d : M.Dart) : ℕ :=
  if h : ∃ t, (M.facePerm ^ t) d = M.alpha d then Nat.find h else 0

/-- **Positions are unchanged by a congruent predicate.** -/
theorem positionCount_congr {p q : ℕ → Prop} (h : ∀ j, p j ↔ q j) (k n : ℕ) :
    positionCount p k n = positionCount q k n := by
  rw [show p = q from funext fun j => propext (h j)]

/-- **The scan along one relator cell, from a suitable base.**  From some base dart of cell `i`,
`λ · #same ≤ (1 - λ) · #other + c · (1 + #run starts)`.  Here `same` are the unbound darts of the
cell with the cell across, `other` the other unbound darts of the cell, and a run start is a
position after the first whose dart is bound while the previous dart is unbound. -/

          (cell X i).face = X.toCombMap.faceOf (X.toCombMap.alpha d)).card : ℝ) ≤

          ¬ (cell X i).face = X.toCombMap.faceOf (X.toCombMap.alpha d)).card : ℝ) +
          c * (1 + positionCount (fun j => 0 < j ∧
            (X.toCombMap.facePerm ^ j) (X.faceBoundary (cell X i).face).darts[k₀] ∉
              RegionCandidate.unboundDarts family i ∧

              RegionCandidate.unboundDarts family i) 0

  have hUsub : ∀ d ∈ U, d ∈ B.darts := by
    intro d hd
    rw [hU, RegionCandidate.unboundDarts, Finset.mem_sdiff, List.mem_toFinset] at hd

    intro x y hx hy
    have h1 : 0 ≤ (1 - lambda) * x := mul_nonneg (by linarith) hx

      AvoidEdgeStep.Oriented X.toCombMap X.outerFace d
  swap
  · have hempty : (U.filter fun d =>
        (cell X i).face = X.toCombMap.faceOf (X.toCombMap.alpha d)) = ∅ := by
      refine Finset.filter_eq_empty_iff.mpr fun d hdU hdf => ?_
      have hdmem := hUsub d hdU
      rcases AvoidEdgeStep.oriented_or hconn ho d with hor | hor
      · exact hT ⟨d, hdmem, hdf.symm, hor⟩
      · refine hT ⟨X.toCombMap.alpha d, (hface _).mpr hdf.symm, ?_, hor⟩

        exact (hface _).mp hdmem

  obtain ⟨x₀, hx₀, hmax⟩ := (B.darts.toFinset.filter fun d =>
      X.toCombMap.faceOf (X.toCombMap.alpha d) = (cell X i).face ∧
        AvoidEdgeStep.Oriented X.toCombMap X.outerFace d).exists_max_image
      (pocketLen X.toCombMap) (by
        obtain ⟨d, hd, hd'⟩ := hT
        exact ⟨d, Finset.mem_filter.mpr ⟨List.mem_toFinset.mpr hd, hd'⟩⟩)

      X.toCombMap.faceOf (X.toCombMap.alpha B.darts[q]) = (cell X i).face →

    have hdmem := OsinUnboundSameCellCycle.mem_of_pow B hk₀ q
    have hdface : X.toCombMap.faceOf (X.toCombMap.alpha
        ((X.toCombMap.facePerm ^ q) B.darts[k₀])) = (cell X i).face := by
      rw [hqs]
      exact hposface s
    have he : (X.toCombMap.facePerm ^ ((s + B.darts.length - q) % B.darts.length))
        ((X.toCombMap.facePerm ^ q) B.darts[k₀]) =
          X.toCombMap.alpha ((X.toCombMap.facePerm ^ q) B.darts[k₀]) := by
      rw [facePerm_pow_apply_pow, hqs]
      by_cases hsq : q ≤ s
      · have e : (s + B.darts.length - q) % B.darts.length + q = s := by
          rw [show s + B.darts.length - q = (s - q) + B.darts.length by omega, Nat.add_mod_right,
            Nat.mod_eq_of_lt (show s - q < B.darts.length by omega)]
          omega
        rw [e]
      · have e : (s + B.darts.length - q) % B.darts.length + q = s + B.darts.length := by
          rw [Nat.mod_eq_of_lt (show s + B.darts.length - q < B.darts.length by omega)]
          omega
        rw [e, ← facePerm_pow_apply_pow, hcyc]
    have hmaxq := hmax _ (Finset.mem_filter.mpr ⟨List.mem_toFinset.mpr hdmem, hdface, hor⟩)

      X.toCombMap.faceOf (X.toCombMap.alpha ((X.toCombMap.facePerm ^ j) B.darts[k₀])))
    (fun j => (X.toCombMap.facePerm ^ j) B.darts[k₀] ∉ U)
    (fun k s => k < s ∧ s < B.darts.length ∧
      AvoidEdgeStep.Oriented X.toCombMap X.outerFace ((X.toCombMap.facePerm ^ k) B.darts[k₀]) ∧
      X.toCombMap.alpha ((X.toCombMap.facePerm ^ k) B.darts[k₀]) =
        (X.toCombMap.facePerm ^ s) B.darts[k₀])
    (fun _ _ h => ⟨h.1, h.2.1⟩) hnest hone hcover hend (fun _ h hb => hb h.1)
  have hsameEq : positionCount (fun j => (X.toCombMap.facePerm ^ j) B.darts[k₀] ∈ U ∧
      (cell X i).face = X.toCombMap.faceOf
        (X.toCombMap.alpha ((X.toCombMap.facePerm ^ j) B.darts[k₀]))) 0 B.darts.length =
      ((U.filter fun d => (cell X i).face =
        X.toCombMap.faceOf (X.toCombMap.alpha d)).card : ℝ) := by
    rw [OsinUnboundSameCellCycle.positionCount_eq_card B hk₀
      (fun d => d ∈ U ∧ (cell X i).face = X.toCombMap.faceOf (X.toCombMap.alpha d))]
    congr 2
    ext d
    rw [Finset.mem_filter, Finset.mem_filter, List.mem_toFinset]
    exact ⟨fun h => h.2, fun h => ⟨hUsub d h.1, h⟩⟩
  have hotherEq : positionCount (fun j => ¬ ((X.toCombMap.facePerm ^ j) B.darts[k₀] ∈ U ∧
      (cell X i).face = X.toCombMap.faceOf
        (X.toCombMap.alpha ((X.toCombMap.facePerm ^ j) B.darts[k₀]))) ∧
      ¬ (X.toCombMap.facePerm ^ j) B.darts[k₀] ∉ U) 0 B.darts.length =
      ((U.filter fun d => ¬ (cell X i).face =
        X.toCombMap.faceOf (X.toCombMap.alpha d)).card : ℝ) := by
    rw [OsinUnboundSameCellCycle.positionCount_eq_card B hk₀
      (fun d => ¬ (d ∈ U ∧ (cell X i).face = X.toCombMap.faceOf (X.toCombMap.alpha d)) ∧
        ¬ d ∉ U)]
    congr 2
    ext d
    rw [Finset.mem_filter, Finset.mem_filter, List.mem_toFinset]
    constructor
    · rintro ⟨-, h1, h2⟩
      exact ⟨not_not.mp h2, fun hq => h1 ⟨not_not.mp h2, hq⟩⟩
    · rintro ⟨hdU, hq⟩
      exact ⟨hUsub d hdU, fun h => hq h.2, fun h => h hdU⟩
  refine ⟨k₀, hk₀, ?_⟩
  rw [← hsameEq, ← hotherEq]
  exact hscan
