/-!
# Trim, then double: a G-corner next to a cell

Osin, arXiv:math/0411039v3, §9, proof of Lemma 9.7(b): the subdiagram `Γ_1` with
`∂Γ_1 = s_1 t_1 s_2 t_2` between two regions joining the cells `Π_i` and `Π_j`.

The cell analogue of `PocketFaceSet.exists_trimDouble` (lane w1-binder-7,
`Estimating/OsinPocketTrimDouble.lean`), following its proof pattern.  A pinch split merges two
corners that are G-faces.  Where a reversed arc ends at a repeated vertex, the corner there lies in
that arc's cell, and the doubling of an edge cannot fill it while the doubled dart is on the arc.
When the side before that arc has room, `CellPocketFaceSet.trimFirstLast` (or `trimSecondLast`)
moves the reversal of the last arc dart into the side, and the doubling then applies to the edge
of that dart in its cell.  In the doubled diagram the moved side dart has the new digon across it.

* `CellPocketFaceSet.faceEdgeDoubling_firstArc_darts`, `_secondArc_darts`, `_firstSide`,
  `_secondSide`: across the doubling outside the face set, arcs avoiding the doubled dart and the
  sides map by the embedding.
* `CellPocketFaceSet.faceEdgeDoubling_firstTurnChain`: first-turn order across that doubling.
* `CellPocketFaceSet.exists_trimDoubleFirst`, `exists_trimDoubleSecond`: for distinct cells, an
  O-equivalent copy with letter labels and a cell pocket face set in walk order with the same
  repeated visits, the arc one dart shorter, and a side dart across which lies a G-face away from
  the exterior, from the relator cells and from the face set.

## Manuscript status

-/

namespace GroupApproximation.GGT.VanKampen

open Embedded

namespace CellPocketFaceSet

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {eps : ℕ} {X : DiscDiagram.{u, w, v} W} {i j : Fin X.rCellCount}

section Outside

variable (K : CellPocketFaceSet D eps X i j) (f : X.toCombMap.Face)
  (k : Fin (X.faceBoundary f).darts.length) (hlen : 1 < (X.faceBoundary f).darts.length)
  (hf : f ≠ X.outerFace) (hs : f ∉ K.faces)
  (hfirst : FaceEdgeDoubling.dart X f k ∉ K.firstArc.darts)
  (hsecond : FaceEdgeDoubling.dart X f k ∉ K.secondArc.darts)

theorem faceEdgeDoubling_firstArc_darts :
    (K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).firstArc.darts =
      K.firstArc.darts.map (FaceEdgeDoubling.embedding X f k hlen hf).darts :=
  FaceEdgeDoubling.mapTo_darts_of_avoid X f k hlen hf K.firstArc _
    (fun _ hd => FaceEdgeDoubling.carrierImage_of_ne_dart X f k hlen hf
      (Embedded.cell X i).face hd) _ hfirst

theorem faceEdgeDoubling_secondArc_darts :
    (K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).secondArc.darts =
      K.secondArc.darts.map (FaceEdgeDoubling.embedding X f k hlen hf).darts :=
  FaceEdgeDoubling.mapTo_darts_of_avoid X f k hlen hf K.secondArc _
    (fun _ hd => FaceEdgeDoubling.carrierImage_of_ne_dart X f k hlen hf
      (Embedded.cell X j).face hd) _ hsecond

theorem faceEdgeDoubling_firstSide :
    (K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).firstSide =
      K.firstSide.map (FaceEdgeDoubling.embedding X f k hlen hf).darts :=
  rfl

theorem faceEdgeDoubling_secondSide :
    (K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).secondSide =
      K.secondSide.map (FaceEdgeDoubling.embedding X f k hlen hf).darts :=
  rfl

/-- **The doubling outside the face set keeps first-turn order.** -/
theorem faceEdgeDoubling_firstTurnChain
    (hK : K.boundary.cycle.IsChain (fun d e => FirstTurn X.toCombMap
          (K.boundary.cycle.reverse.map X.toCombMap.alpha) (X.toCombMap.alpha e)
          (X.toCombMap.alpha d)) ∧
        FirstTurn X.toCombMap (K.boundary.cycle.reverse.map X.toCombMap.alpha)
          (X.toCombMap.alpha (K.boundary.cycle.head K.boundary.cycle_nonempty))
          (X.toCombMap.alpha (K.boundary.cycle.getLast K.boundary.cycle_nonempty))) :
    (K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).boundary.cycle.IsChain (fun d e =>
        FirstTurn (FaceEdgeDoubling.diagram X f k hlen hf).toCombMap
          ((K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).boundary.cycle.reverse.map
            (FaceEdgeDoubling.diagram X f k hlen hf).toCombMap.alpha)
          ((FaceEdgeDoubling.diagram X f k hlen hf).toCombMap.alpha e)
          ((FaceEdgeDoubling.diagram X f k hlen hf).toCombMap.alpha d)) ∧
      FirstTurn (FaceEdgeDoubling.diagram X f k hlen hf).toCombMap
        ((K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).boundary.cycle.reverse.map
          (FaceEdgeDoubling.diagram X f k hlen hf).toCombMap.alpha)
        ((FaceEdgeDoubling.diagram X f k hlen hf).toCombMap.alpha
          ((K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).boundary.cycle.head
            (K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).boundary.cycle_nonempty))
        ((FaceEdgeDoubling.diagram X f k hlen hf).toCombMap.alpha
          ((K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).boundary.cycle.getLast
            (K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).boundary.cycle_nonempty)) :=
  EdgeInsertion.firstTurnChain_map_embed
    (EdgeInsertion.corners_ne X.toCombMap (FaceEdgeDoubling.rebased X f k)
      (FaceEdgeDoubling.second X f k hlen) Nat.one_pos)
    K.boundary.cycle_nonempty
    (K.faceEdgeDoubling f k hlen hf hs hfirst hsecond).boundary.cycle_nonempty hK

end Outside

/-- **Trim the first arc, then double the trimmed dart.**  For a cell pocket face set in walk order
of a diagram with letter labels and distinct cells, whose first side is shorter than `ε` and whose
first arc has at least two darts, there is an O-equivalent copy with letter labels and a cell
pocket face set in walk order, with the same repeated visits and a first arc one dart shorter,
whose first side holds a dart across which lies a G-face outside the face set, away from the
exterior and the relator cells. -/
theorem exists_trimDoubleFirst (K : CellPocketFaceSet D eps X i j) (hij : i ≠ j)
    (hlabel : ∀ d, (symmetricLabelAlphabet D).IsLetter (X.label d))
    (hroom : K.firstSide.length < eps) (hlong : 1 < K.firstArc.length) (hK : K.ClosedWalk) :
    ∃ (X' : DiscDiagram.{u, w, v} W) (i' j' : Fin X'.rCellCount)
      (K' : CellPocketFaceSet D eps X' i' j'),
      Nonempty (OEquivalentDiscDiagram X X') ∧
        (∀ d, (symmetricLabelAlphabet D).IsLetter (X'.label d)) ∧
        K'.ClosedWalk ∧ K'.repeatedVisits = K.repeatedVisits ∧
        K'.firstArc.length = K.firstArc.length - 1 ∧
        ∃ x ∈ K'.firstSide,
          X'.toCombMap.faceOf (X'.toCombMap.alpha x) ≠ X'.outerFace ∧
          (∀ C ∈ X'.relatorCells, C.face ≠ X'.toCombMap.faceOf (X'.toCombMap.alpha x)) ∧
          X'.toCombMap.faceOf (X'.toCombMap.alpha x) ∉ K'.faces := by
  have hne := K.firstArc_darts_ne_nil
  have hmem : K.firstArc.darts.getLast hne ∈ (X.faceBoundary (cell X i).face).darts :=
    K.firstArc.mem_cycle_of_mem_darts (List.getLast_mem hne)
  obtain ⟨k, hk⟩ := List.get_of_mem hmem
  have hdart : FaceEdgeDoubling.dart X (cell X i).face k = K.firstArc.darts.getLast hne :=
    (FaceEdgeDoubling.dart_eq_get X (cell X i).face k).trans hk
  have hlen : 1 < (X.faceBoundary (cell X i).face).darts.length :=
    lt_of_lt_of_le hlong K.firstArc.length_le
  have hf : (cell X i).face ≠ X.outerFace := (cell X i).face_ne_outer
  have hs : (cell X i).face ∉ (K.trimFirstLast hlabel hroom hlong).faces := K.first_not_mem
  have hfirst : FaceEdgeDoubling.dart X (cell X i).face k ∉
      (K.trimFirstLast hlabel hroom hlong).firstArc.darts := by
    rw [hdart]
    show K.firstArc.darts.getLast hne ∉ K.firstArc.dropLastArc.darts
    rw [CyclicArc.dropLastArc_darts]
    intro h
    have hnd := K.firstArc.darts_nodup (cellDarts_nodup X i)
    rw [← List.dropLast_concat_getLast hne, List.nodup_append] at hnd
    exact hnd.2.2 _ h _ (List.mem_singleton.mpr rfl) rfl
  have hsecond : FaceEdgeDoubling.dart X (cell X i).face k ∉
      (K.trimFirstLast hlabel hroom hlong).secondArc.darts := by
    intro h
    have h1 : X.toCombMap.faceOf (FaceEdgeDoubling.dart X (cell X i).face k) = (cell X i).face :=
      FaceEdgeDoubling.dart_face X (cell X i).face k
    have h2 : X.toCombMap.faceOf (FaceEdgeDoubling.dart X (cell X i).face k) = (cell X j).face :=
      ((X.faceBoundary (cell X j).face).mem_iff _).mp (K.secondArc.mem_cycle_of_mem_darts h)
    exact Embedded.cell_face_ne hij (h1.symm.trans h2)
  refine ⟨FaceEdgeDoubling.diagram X (cell X i).face k hlen hf, _, _,
    (K.trimFirstLast hlabel hroom hlong).faceEdgeDoubling (cell X i).face k hlen hf hs hfirst
      hsecond,
    ⟨FaceEdgeDoubling.oEquivalent X (cell X i).face k hlen hf⟩,
    FaceEdgeDoubling.label_letter X (cell X i).face k hlen hf hlabel,
    faceEdgeDoubling_closedWalk (K.trimFirstLast hlabel hroom hlong) (cell X i).face k hlen hf hs
      hfirst hsecond (K.closedWalk_trimFirstLast hlabel hroom hlong hK),
    (faceEdgeDoubling_repeatedVisits (K.trimFirstLast hlabel hroom hlong) (cell X i).face k hlen
      hf hs hfirst hsecond).trans (K.repeatedVisits_trimFirstLast hlabel hroom hlong), ?_, ?_⟩
  · have h1 := faceEdgeDoubling_firstArc_darts (K.trimFirstLast hlabel hroom hlong)
      (cell X i).face k hlen hf hs hfirst hsecond
    have h2 := congrArg List.length h1
    rw [List.length_map, CyclicArc.darts_length, CyclicArc.darts_length] at h2
    exact h2
  · have hface : (FaceEdgeDoubling.diagram X (cell X i).face k hlen hf).toCombMap.faceOf
        ((FaceEdgeDoubling.diagram X (cell X i).face k hlen hf).toCombMap.alpha
          ((FaceEdgeDoubling.embedding X (cell X i).face k hlen hf).darts
            (X.toCombMap.alpha (K.firstArc.darts.getLast hne)))) =
        FaceEdgeDoubling.digon X (cell X i).face k hlen := by
      rw [(FaceEdgeDoubling.embedding X (cell X i).face k hlen hf).alpha,
        X.toCombMap.alpha_involutive, ← hdart]
      refine ((FaceEdgeDoubling.boundary X (cell X i).face k hlen
        (FaceEdgeDoubling.digon X (cell X i).face k hlen)).mem_iff _).mp ?_
      rw [FaceEdgeDoubling.boundary_digon]
      exact List.mem_cons_of_mem _ (List.mem_singleton.mpr rfl)
    refine ⟨(FaceEdgeDoubling.embedding X (cell X i).face k hlen hf).darts
      (X.toCombMap.alpha (K.firstArc.darts.getLast hne)), ?_, ?_, ?_, ?_⟩
    · rw [faceEdgeDoubling_firstSide]
      exact List.mem_map_of_mem (List.mem_append_right _ (List.mem_singleton.mpr rfl))
    · rw [hface]
      exact (FaceEdgeDoubling.keep_ne_digon X (cell X i).face k hlen
        (fun h => hf h.symm)).symm
    · intro C hC
      rw [hface]
      obtain ⟨C₀, -, rfl⟩ := List.mem_map.mp hC
      exact FaceEdgeDoubling.faceImage_ne_digon X (cell X i).face k hlen C₀.face
    · rw [hface]
      intro h
      obtain ⟨g, hg, hkeep⟩ := Finset.mem_map.mp h
      exact FaceEdgeDoubling.keep_ne_digon X (cell X i).face k hlen
        (fun heq => K.first_not_mem (heq ▸ hg)) hkeep

/-- **Trim the second arc, then double the trimmed dart.**  As `exists_trimDoubleFirst`, for the
second arc and the second side. -/
theorem exists_trimDoubleSecond (K : CellPocketFaceSet D eps X i j) (hij : i ≠ j)
    (hlabel : ∀ d, (symmetricLabelAlphabet D).IsLetter (X.label d))
    (hroom : K.secondSide.length < eps) (hlong : 1 < K.secondArc.length) (hK : K.ClosedWalk) :
    ∃ (X' : DiscDiagram.{u, w, v} W) (i' j' : Fin X'.rCellCount)
      (K' : CellPocketFaceSet D eps X' i' j'),
      Nonempty (OEquivalentDiscDiagram X X') ∧
        (∀ d, (symmetricLabelAlphabet D).IsLetter (X'.label d)) ∧
        K'.ClosedWalk ∧ K'.repeatedVisits = K.repeatedVisits ∧
        K'.secondArc.length = K.secondArc.length - 1 ∧
        ∃ x ∈ K'.secondSide,
          X'.toCombMap.faceOf (X'.toCombMap.alpha x) ≠ X'.outerFace ∧
          (∀ C ∈ X'.relatorCells, C.face ≠ X'.toCombMap.faceOf (X'.toCombMap.alpha x)) ∧
          X'.toCombMap.faceOf (X'.toCombMap.alpha x) ∉ K'.faces := by
  have hne := K.secondArc_darts_ne_nil
  have hmem : K.secondArc.darts.getLast hne ∈ (X.faceBoundary (cell X j).face).darts :=
    K.secondArc.mem_cycle_of_mem_darts (List.getLast_mem hne)
  obtain ⟨k, hk⟩ := List.get_of_mem hmem
  have hdart : FaceEdgeDoubling.dart X (cell X j).face k = K.secondArc.darts.getLast hne :=
    (FaceEdgeDoubling.dart_eq_get X (cell X j).face k).trans hk
  have hlen : 1 < (X.faceBoundary (cell X j).face).darts.length :=
    lt_of_lt_of_le hlong K.secondArc.length_le
  have hf : (cell X j).face ≠ X.outerFace := (cell X j).face_ne_outer
  have hs : (cell X j).face ∉ (K.trimSecondLast hlabel hroom hlong).faces := K.second_not_mem
  have hfirst : FaceEdgeDoubling.dart X (cell X j).face k ∉
      (K.trimSecondLast hlabel hroom hlong).firstArc.darts := by
    intro h
    have h1 : X.toCombMap.faceOf (FaceEdgeDoubling.dart X (cell X j).face k) = (cell X j).face :=
      FaceEdgeDoubling.dart_face X (cell X j).face k
    have h2 : X.toCombMap.faceOf (FaceEdgeDoubling.dart X (cell X j).face k) = (cell X i).face :=
      ((X.faceBoundary (cell X i).face).mem_iff _).mp (K.firstArc.mem_cycle_of_mem_darts h)
    exact Embedded.cell_face_ne hij (h2.symm.trans h1)
  have hsecond : FaceEdgeDoubling.dart X (cell X j).face k ∉
      (K.trimSecondLast hlabel hroom hlong).secondArc.darts := by
    rw [hdart]
    show K.secondArc.darts.getLast hne ∉ K.secondArc.dropLastArc.darts
    rw [CyclicArc.dropLastArc_darts]
    intro h
    have hnd := K.secondArc.darts_nodup (cellDarts_nodup X j)
    rw [← List.dropLast_concat_getLast hne, List.nodup_append] at hnd
    exact hnd.2.2 _ h _ (List.mem_singleton.mpr rfl) rfl
  refine ⟨FaceEdgeDoubling.diagram X (cell X j).face k hlen hf, _, _,
    (K.trimSecondLast hlabel hroom hlong).faceEdgeDoubling (cell X j).face k hlen hf hs hfirst
      hsecond,
    ⟨FaceEdgeDoubling.oEquivalent X (cell X j).face k hlen hf⟩,
    FaceEdgeDoubling.label_letter X (cell X j).face k hlen hf hlabel,
    faceEdgeDoubling_closedWalk (K.trimSecondLast hlabel hroom hlong) (cell X j).face k hlen hf hs
      hfirst hsecond (K.closedWalk_trimSecondLast hlabel hroom hlong hK),
    (faceEdgeDoubling_repeatedVisits (K.trimSecondLast hlabel hroom hlong) (cell X j).face k hlen
      hf hs hfirst hsecond).trans (K.repeatedVisits_trimSecondLast hlabel hroom hlong), ?_, ?_⟩
  · have h1 := faceEdgeDoubling_secondArc_darts (K.trimSecondLast hlabel hroom hlong)
      (cell X j).face k hlen hf hs hfirst hsecond
    have h2 := congrArg List.length h1
    rw [List.length_map, CyclicArc.darts_length, CyclicArc.darts_length] at h2
    exact h2
  · have hface : (FaceEdgeDoubling.diagram X (cell X j).face k hlen hf).toCombMap.faceOf
        ((FaceEdgeDoubling.diagram X (cell X j).face k hlen hf).toCombMap.alpha
          ((FaceEdgeDoubling.embedding X (cell X j).face k hlen hf).darts
            (X.toCombMap.alpha (K.secondArc.darts.getLast hne)))) =
        FaceEdgeDoubling.digon X (cell X j).face k hlen := by
      rw [(FaceEdgeDoubling.embedding X (cell X j).face k hlen hf).alpha,
        X.toCombMap.alpha_involutive, ← hdart]
      refine ((FaceEdgeDoubling.boundary X (cell X j).face k hlen
        (FaceEdgeDoubling.digon X (cell X j).face k hlen)).mem_iff _).mp ?_
      rw [FaceEdgeDoubling.boundary_digon]
      exact List.mem_cons_of_mem _ (List.mem_singleton.mpr rfl)
    refine ⟨(FaceEdgeDoubling.embedding X (cell X j).face k hlen hf).darts
      (X.toCombMap.alpha (K.secondArc.darts.getLast hne)), ?_, ?_, ?_, ?_⟩
    · rw [faceEdgeDoubling_secondSide]
      exact List.mem_map_of_mem (List.mem_append_right _ (List.mem_singleton.mpr rfl))
    · rw [hface]
      exact (FaceEdgeDoubling.keep_ne_digon X (cell X j).face k hlen
        (fun h => hf h.symm)).symm
    · intro C hC
      rw [hface]
      obtain ⟨C₀, -, rfl⟩ := List.mem_map.mp hC
      exact FaceEdgeDoubling.faceImage_ne_digon X (cell X j).face k hlen C₀.face
    · rw [hface]
      intro h
      obtain ⟨g, hg, hkeep⟩ := Finset.mem_map.mp h
      exact FaceEdgeDoubling.keep_ne_digon X (cell X j).face k hlen
        (fun heq => K.second_not_mem (heq ▸ hg)) hkeep
