/-!
# `thm:factorization-nonmf-trace`: the printed corona proof, sentence by sentence

`non_mf_groups_exist.tex`, proof of Theorem `thm:factorization-nonmf-trace` (tex 1844–1854 at
origin/main 68481e4d7):

> Suppose that `τ_G` is quasidiagonal, with u.c.p. maps `φ_n : C*_max(G) → M_{d_n}(ℂ)` as in the
> definition.  Asymptotic multiplicativity in operator norm makes `a ↦ [(φ_n(a))_n]` a unital
> `*`-homomorphism `Φ : C*_max(G) → 𝒬_𝐝`.  If `Φ(u_g) = 1`, then `‖φ_n(u_g) − 1‖ → 0`, so
> `tr_{d_n}(φ_n(u_g)) → 1`, while `tr_{d_n}(φ_n(u_g)) → τ_G(u_g)`; so `g = 1`.  Hence
> `g ↦ Φ(u_g)` is a corona homomorphism injective on `G`, and `G` is MF, contrary to the
> hypothesis.

`Manuscript/OneSidedMFRadical/FactorizationTraceSentences.lean` carries an earlier printed proof
(polar decomposition of `φ_n(u_g)`), which the manuscript no longer prints; its first sentence
carrier `manuscriptSentence_quasidiagonalModel` still serves the first sentence above.  This module
carries the other three along the printed route: `Φ` is built from the quasidiagonal models, and
injectivity is read off `Φ(u_g) = 1` through the operator-norm-null deviations and the normalized
traces.

Two conventions.  Forming `[(φ_n(a))_n]` in `∏_n M_{d_n}(ℂ)` uses that a u.c.p. matrix map is
contractive; `Quasidiagonal.QuasidiagonalTraceModel.toMFTraceModel` records this through the
theorem `Quasidiagonal.ucpContractive`.  The printed `M_{d_n}(ℂ)` with its normalized trace has
`d_n ≥ 1`, while the Lean models allow empty coordinates; `exists_models_card_pos` discards the
finitely many empty ones.
-/

namespace GroupApproximation
namespace FactorizationTraceCoronaProof

open Filter Matrix Quasidiagonal
open scoped Matrix.Norms.L2Operator ENNReal

noncomputable section

universe u

/-! ## Matrix sizes -/

/-- The normalized trace of the identity matrix is `1` on a nonempty coordinate space. -/
theorem normTrace_one_of_card_pos (Y : FiniteModel) (hY : 0 < Fintype.card Y) :
    normTrace Y (1 : Matrix Y Y ℂ) = 1 := by
  show Matrix.trace (1 : Matrix Y Y ℂ) / ((Fintype.card Y : ℕ) : ℂ) = 1
  rw [Matrix.trace_one]
  exact div_self (Nat.cast_ne_zero.mpr hY.ne')

variable {A : Type u} [CStarAlgebra A] {τ : A → ℂ}

/-- Discard the first `N` models. -/
def shiftModel (M : QuasidiagonalTraceModel τ) (N : ℕ) : QuasidiagonalTraceModel τ where
  space n := M.space (n + N)
  map n := M.map (n + N)
  map_one n := M.map_one (n + N)
  completelyPositive n := M.completelyPositive (n + N)
  tendsto_mul a b := (M.tendsto_mul a b).comp (tendsto_add_atTop_nat N)
  tendsto_trace a := (M.tendsto_trace a).comp (tendsto_add_atTop_nat N)

/-- **The printed `d_n ≥ 1`.**  The trace clause at `1` makes all but finitely many models
nonempty, so discarding an initial segment gives models of positive size. -/
theorem exists_models_card_pos (M : QuasidiagonalTraceModel τ) (hone : τ 1 = 1) :
    ∃ M' : QuasidiagonalTraceModel τ, ∀ n, 0 < Fintype.card (M'.space n) := by
  obtain ⟨N, hN⟩ := eventually_atTop.mp
    ((M.toMFTraceModel ucpContractive).eventually_card_pos hone)
  exact ⟨shiftModel M N, fun n ↦ hN (n + N) (Nat.le_add_left N n)⟩

/-! ## Sentence 2: the unital `*`-homomorphism `Φ` -/

/-- The bounded matrix sequence `(φ_n(a))_n` of the quasidiagonal models. -/
def modelSeq (M : QuasidiagonalTraceModel τ) (a : A) :
    BoundedMatrixSequence (fun n ↦ M.space n) :=
  (M.toMFTraceModel ucpContractive).seq a

@[simp] theorem modelSeq_apply (M : QuasidiagonalTraceModel τ) (a : A) (n : ℕ) :
    modelSeq M a n = M.map n a := rfl

/-- The printed map `Φ : a ↦ [(φ_n(a))_n]` into the operator-norm corona of the models. -/
def coronaMap (M : QuasidiagonalTraceModel τ) [∀ n, Nonempty (M.space n)] :
    A →⋆ₙₐ[ℂ] NormMatrixCStarCorona (fun n ↦ M.space n) :=
  letI : ∀ n, Nonempty ((M.toMFTraceModel ucpContractive).space n) :=
    fun n ↦ (inferInstance : Nonempty (M.space n))
  (M.toMFTraceModel ucpContractive).coronaHom

theorem coronaMap_apply (M : QuasidiagonalTraceModel τ) [∀ n, Nonempty (M.space n)] (a : A) :
    coronaMap M a = normMatrixCStarCoronaQuotient (fun n ↦ M.space n) (modelSeq M a) := rfl

/-- **`Φ` is unital**, because every `φ_n` is. -/
theorem coronaMap_one (M : QuasidiagonalTraceModel τ) [∀ n, Nonempty (M.space n)] :
    coronaMap M 1 = 1 := by
  have hseq : modelSeq M 1 = 1 := by
    apply lp.ext
    funext n
    exact M.map_one n
  rw [coronaMap_apply, hseq]
  exact map_one _

/-- `Φ(a) = 1` forces `‖φ_n(a) − 1‖ → 0`. -/
theorem tendsto_map_sub_one_of_coronaMap_eq_one (M : QuasidiagonalTraceModel τ)
    [∀ n, Nonempty (M.space n)] {a : A} (ha : coronaMap M a = 1) :
    Tendsto (fun n ↦ ‖M.map n a - 1‖) atTop (nhds 0) := by
  have hab : coronaMap M a = coronaMap M 1 := by rw [ha, coronaMap_one]
  have hseq : normMatrixCStarCoronaMk (fun n ↦ M.space n)
      (modelSeq M a - modelSeq M 1) = 0 := by
    rw [map_sub]
    exact sub_eq_zero_of_eq hab
  rw [normMatrixCStarCoronaMk_eq_zero_iff, IsNullMatrixSequence,
    Nat.cofinite_eq_atTop] at hseq
  refine hseq.congr fun n ↦ ?_
  show ‖M.map n a - M.map n 1‖ = ‖M.map n a - 1‖
  rw [M.map_one]

/-- `‖φ_n(a) − 1‖ → 0` forces `tr_{d_n}(φ_n(a)) → 1`: the normalized trace is contractive for the
operator norm. -/
theorem tendsto_normTrace_of_tendsto_map_sub_one (M : QuasidiagonalTraceModel τ)
    (hcard : ∀ n, 0 < Fintype.card (M.space n)) {a : A}
    (h : Tendsto (fun n ↦ ‖M.map n a - 1‖) atTop (nhds 0)) :
    Tendsto (fun n ↦ normTrace (M.space n) (M.map n a)) atTop (nhds 1) := by
  rw [tendsto_iff_norm_sub_tendsto_zero]
  refine squeeze_zero (fun n ↦ norm_nonneg _) (fun n ↦ ?_) h
  rw [← normTrace_one_of_card_pos (M.space n) (hcard n), ← normTrace_sub]
  exact PrintedPreliminaryEstimates.norm_normTrace_le_l2_opNorm (M.space n) (M.map n a - 1)

/-- **Printed sentence 2.**

> Asymptotic multiplicativity in operator norm makes `a ↦ [(φ_n(a))_n]` a unital
> `*`-homomorphism `Φ : C*_max(G) → 𝒬_𝐝`.

`Φ` is `coronaMap M`: its value at `a` is the corona class of `(φ_n(a))_n`, it is a
`*`-homomorphism, and it is unital. -/
theorem manuscriptSentence_unitalStarHom (G : Type*) [Group G]
    (M : QuasidiagonalTraceModel (fun a : MaximalGroupCStar G ↦ canonicalMaximalTrace G a))
    [∀ n, Nonempty (M.space n)] :
    (∀ a : MaximalGroupCStar G,
        coronaMap M a = normMatrixCStarCoronaQuotient (fun n ↦ M.space n) (modelSeq M a)) ∧
      (∀ a b : MaximalGroupCStar G, coronaMap M (a * b) = coronaMap M a * coronaMap M b) ∧
      (∀ a : MaximalGroupCStar G, coronaMap M (star a) = star (coronaMap M a)) ∧
      coronaMap M 1 = 1 :=
  ⟨coronaMap_apply M, fun a b ↦ map_mul (coronaMap M) a b,
    fun a ↦ map_star (coronaMap M) a, coronaMap_one M⟩

/-! ## Sentence 3: the kernel on the canonical unitaries -/

/-- **Printed sentence 3.**

> If `Φ(u_g) = 1`, then `‖φ_n(u_g) − 1‖ → 0`, so `tr_{d_n}(φ_n(u_g)) → 1`, while
> `tr_{d_n}(φ_n(u_g)) → τ_G(u_g)`; so `g = 1`.

The four clauses, in order.  The last one uses `τ_G(u_g) = 0` for `g ≠ 1`, through
`FactorizationTraceSentences.manuscriptSentence_traceTendsToZero`. -/
theorem manuscriptSentence_generatorKernel (G : Type*) [Group G]
    (M : QuasidiagonalTraceModel (fun a : MaximalGroupCStar G ↦ canonicalMaximalTrace G a))
    (hcard : ∀ n, 0 < Fintype.card (M.space n)) [∀ n, Nonempty (M.space n)] {g : G}
    (hg : coronaMap M (maximalGroupCStarGenerator G g) = 1) :
    Tendsto (fun n ↦ ‖M.map n (maximalGroupCStarGenerator G g) - 1‖) atTop (nhds 0) ∧
      Tendsto (fun n ↦ normTrace (M.space n) (M.map n (maximalGroupCStarGenerator G g)))
        atTop (nhds 1) ∧
      Tendsto (fun n ↦ ‖canonicalMaximalTrace G (maximalGroupCStarGenerator G g)
          - normTrace (M.space n) (M.map n (maximalGroupCStarGenerator G g))‖)
        atTop (nhds 0) ∧
      g = 1 := by
  have h1 := tendsto_map_sub_one_of_coronaMap_eq_one M hg
  have h2 := tendsto_normTrace_of_tendsto_map_sub_one M hcard h1
  have h3 := M.tendsto_trace (maximalGroupCStarGenerator G g)
  refine ⟨h1, h2, h3, ?_⟩
  by_contra hne
  have h0 := FactorizationTraceSentences.manuscriptSentence_traceTendsToZero G M hne
  have hn1 := h2.norm
  rw [norm_one] at hn1
  exact one_ne_zero (tendsto_nhds_unique hn1 h0)

/-! ## Sentence 4: the injective corona homomorphism -/

/-- The printed corona homomorphism `g ↦ Φ(u_g)`. -/
def generatorCoronaHom (G : Type*) [Group G]
    (M : QuasidiagonalTraceModel (fun a : MaximalGroupCStar G ↦ canonicalMaximalTrace G a))
    [∀ n, Nonempty (M.space n)] :
    G →* unitary (NormMatrixCStarCorona (fun n ↦ M.space n)) :=
  (nonUnitalStarAlgHomUnitaryMap (coronaMap M)).comp (maximalGroupCStarUnitaryHom G)

theorem generatorCoronaHom_apply (G : Type*) [Group G]
    (M : QuasidiagonalTraceModel (fun a : MaximalGroupCStar G ↦ canonicalMaximalTrace G a))
    [∀ n, Nonempty (M.space n)] (g : G) :
    (generatorCoronaHom G M g : NormMatrixCStarCorona (fun n ↦ M.space n))
      = coronaMap M (maximalGroupCStarGenerator G g) := by
  show coronaMap M (maximalGroupCStarGenerator G g) + (1 - coronaMap M 1) = _
  rw [coronaMap_one, sub_self, add_zero]

/-- **Printed sentence 4, first clause.**

> Hence `g ↦ Φ(u_g)` is a corona homomorphism injective on `G` … -/
theorem manuscriptSentence_injectiveCoronaHom (G : Type*) [Group G]
    (M : QuasidiagonalTraceModel (fun a : MaximalGroupCStar G ↦ canonicalMaximalTrace G a))
    (hcard : ∀ n, 0 < Fintype.card (M.space n)) [∀ n, Nonempty (M.space n)] :
    Function.Injective (generatorCoronaHom G M) := by
  rw [injective_iff_map_eq_one]
  intro g hg
  have h : coronaMap M (maximalGroupCStarGenerator G g) = 1 := by
    rw [← generatorCoronaHom_apply G M g, hg]
    rfl
  exact (manuscriptSentence_generatorKernel G M hcard h).2.2.2

/-- **Printed sentence 4, second clause.**

> … and `G` is MF …

The injective corona homomorphism, read in the unitary-sequence corona. -/
theorem manuscriptSentence_isOperatorMF (G : Type*) [Group G]
    (M : QuasidiagonalTraceModel (fun a : MaximalGroupCStar G ↦ canonicalMaximalTrace G a))
    (hcard : ∀ n, 0 < Fintype.card (M.space n)) : IsOperatorMF G := by
  letI : ∀ n, Nonempty (M.space n) := fun n ↦ Fintype.card_pos_iff.mp (hcard n)
  exact ⟨M.space, hcard,
    (normMatrixCoronaUnitaryEquiv M.space).symm.toMonoidHom.comp (generatorCoronaHom G M),
    (normMatrixCoronaUnitaryEquiv M.space).symm.injective.comp
      (manuscriptSentence_injectiveCoronaHom G M hcard)⟩

/-! ## The printed proof, assembled -/

/-- `thm:factorization-nonmf-trace` at the printed generality (a countable group), proved along
the printed corona route of this module. -/
def PrintedFactorizationTraceCoronaProof : Prop :=
  ∀ (G : Type) [Group G] [Countable G], ¬ IsOperatorMF G →
    ¬ IsQuasidiagonalTrace (fun a : MaximalGroupCStar G ↦ canonicalMaximalTrace G a)

/-- **Printed sentence 4, conclusion.**

> … contrary to the hypothesis.

A quasidiagonal model of `τ_G`, trimmed to positive sizes, makes `G` MF. -/
theorem printedFactorizationTraceCoronaProof : PrintedFactorizationTraceCoronaProof := by
  rintro G _ _ hnot ⟨M⟩
  obtain ⟨M', hcard⟩ := exists_models_card_pos M (canonicalMaximalTrace_one G)
  exact hnot (manuscriptSentence_isOperatorMF G M' hcard)

end
