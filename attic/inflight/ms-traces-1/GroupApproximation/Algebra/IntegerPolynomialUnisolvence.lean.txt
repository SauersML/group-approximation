/-!
# Integer polynomials with bounded values on a grid

Two inputs for the archimedean places in the char-0 case of GHW Theorem 4 (tex L1146–1147:
"Every countable subgroup of $\mathrm{GL}_2$ over a field has the Haagerup
property~\cite[Theorem~4]{GHW}").

* `exists_algebraicIndependent_nat`: there is a sequence of complex numbers algebraically
  independent over `ℤ` (a transcendence basis of `ℂ` has cardinality `𝔠`).
* `finite_bounded_on_grid`: if every `t i : ℕ → ℂ` is injective, the integer polynomials of total
  degree `≤ D` whose values at the grid points `(t i (j i))_i`, `j i ≤ D`, are bounded by `R`
  form a finite set. The evaluation matrix of the monomials on the grid is injective
  (Combinatorial Nullstellensatz), so the coefficients are bounded by a left inverse.
-/

namespace GroupApproximation
namespace IntegerPolynomialUnisolvence

open MvPolynomial
open scoped Cardinal Matrix

variable {d : ℕ}

/-- The exponent vector with entries `m i`. -/
noncomputable def expVec {D : ℕ} (m : Fin d → Fin (D + 1)) : Fin d →₀ ℕ :=
  Finsupp.equivFunOnFinite.symm fun i ↦ (m i : ℕ)

theorem expVec_injective {D : ℕ} : Function.Injective (expVec (d := d) (D := D)) := by
  intro m m' h
  funext i
  exact Fin.ext (congrArg (fun s : Fin d →₀ ℕ ↦ s i) h)

/-- Every monomial of a polynomial of total degree `≤ D` is `expVec m` for some `m`. -/
theorem exists_expVec_eq {D : ℕ} {G : MvPolynomial (Fin d) ℤ} (hG : G.totalDegree ≤ D)
    {s : Fin d →₀ ℕ} (hs : s ∈ G.support) : ∃ m : Fin d → Fin (D + 1), expVec m = s :=
  ⟨fun i ↦ ⟨s i, Nat.lt_succ_of_le ((monomial_le_degreeOf i hs).trans
    ((degreeOf_le_totalDegree G i).trans hG))⟩, by ext i; rfl⟩

/-- The values of the monomials `expVec m` at the grid points `(t i (g i))_i`. -/
noncomputable def gridMatrix (t : Fin d → ℕ → ℂ) (D : ℕ) :
    Matrix (Fin d → Fin (D + 1)) (Fin d → Fin (D + 1)) ℂ :=
  Matrix.of fun g m ↦ ∏ i, t i (g i) ^ (m i : ℕ)

/-- The complex polynomial with coefficient `c m` at the monomial `expVec m`. -/
noncomputable def polyOf {D : ℕ} (c : (Fin d → Fin (D + 1)) → ℂ) : MvPolynomial (Fin d) ℂ :=
  ∑ m, monomial (expVec m) (c m)

theorem eval_polyOf (t : Fin d → ℕ → ℂ) {D : ℕ} (c : (Fin d → Fin (D + 1)) → ℂ)
    (g : Fin d → Fin (D + 1)) :
    eval (fun i ↦ t i (g i)) (polyOf c) = (gridMatrix t D *ᵥ c) g := by
  simp only [polyOf, map_sum, eval_monomial, Matrix.mulVec, dotProduct, gridMatrix,
    Matrix.of_apply]
  refine Finset.sum_congr rfl fun m _ ↦ ?_
  rw [Finsupp.prod_fintype]
  · exact mul_comm _ _
  · intro i
    exact pow_zero _

theorem coeff_polyOf {D : ℕ} (c : (Fin d → Fin (D + 1)) → ℂ) (m : Fin d → Fin (D + 1)) :
    coeff (expVec m) (polyOf c) = c m := by
  classical
  rw [polyOf, coeff_sum, Finset.sum_eq_single m]
  · rw [coeff_monomial, if_pos rfl]
  · intro m' _ hm'
    rw [coeff_monomial, if_neg fun h ↦ hm' (expVec_injective h)]
  · intro h
    exact absurd (Finset.mem_univ m) h

theorem coeff_polyOf_eq_zero {D : ℕ} (c : (Fin d → Fin (D + 1)) → ℂ) {s : Fin d →₀ ℕ}
    (hs : ∀ m : Fin d → Fin (D + 1), expVec m ≠ s) : coeff s (polyOf c) = 0 := by
  classical
  rw [polyOf, coeff_sum]
  exact Finset.sum_eq_zero fun m _ ↦ by rw [coeff_monomial, if_neg (hs m)]

/-- The evaluation matrix of the monomials on the grid is injective. -/
theorem ker_gridMatrix (t : Fin d → ℕ → ℂ) (ht : ∀ i, Function.Injective (t i)) (D : ℕ) :
    LinearMap.ker (gridMatrix t D).mulVecLin = ⊥ := by
  classical
  rw [LinearMap.ker_eq_bot']
  intro c hc
  have hP : polyOf c = 0 := by
    refine eq_zero_of_eval_zero_at_prod_finset (polyOf c)
      (fun i ↦ (Finset.range (D + 1)).image (t i)) (fun i ↦ ?_) (fun x hx ↦ ?_)
    · show (polyOf c).degreeOf i < ((Finset.range (D + 1)).image (t i)).card
      rw [Finset.card_image_of_injective _ (ht i), Finset.card_range,
        degreeOf_lt_iff (show 0 < D + 1 by omega)]
      intro s hs
      rw [polyOf] at hs
      obtain ⟨m, -, hm⟩ := Finset.mem_biUnion.mp (support_sum hs)
      rw [Finset.mem_singleton.mp (support_monomial_subset hm)]
      exact (m i).isLt
    · choose j hj hjx using fun i ↦ Finset.mem_image.mp (hx i)
      let g : Fin d → Fin (D + 1) := fun i ↦ ⟨j i, Finset.mem_range.mp (hj i)⟩
      have hxg : x = fun i ↦ t i (g i) := funext fun i ↦ (hjx i).symm
      rw [hxg]
      exact (eval_polyOf t c g).trans (congrFun hc g)
  funext m
  simpa [hP] using (coeff_polyOf c m).symm

/-- Integer polynomials of total degree `≤ D` whose values at the grid points
`(t i (j i))_i`, `j i ≤ D`, are bounded by `R` form a finite set, when every `t i` is
injective. -/
theorem finite_bounded_on_grid (t : Fin d → ℕ → ℂ) (ht : ∀ i, Function.Injective (t i))
    (D : ℕ) (R : ℝ) :
    {G : MvPolynomial (Fin d) ℤ | G.totalDegree ≤ D ∧
      ∀ j : Fin d → ℕ, (∀ i, j i ≤ D) → ‖aeval (fun i ↦ t i (j i)) G‖ ≤ R}.Finite := by
  classical
  obtain ⟨g, hg⟩ := LinearMap.exists_leftInverse_of_injective _ (ker_gridMatrix t ht D)
  obtain ⟨T, hT⟩ : ∃ T : ((Fin d → Fin (D + 1)) → ℂ) →L[ℂ] ((Fin d → Fin (D + 1)) → ℂ),
      ∀ w, T w = g w := ⟨LinearMap.toContinuousLinearMap g, fun _ ↦ rfl⟩
  obtain ⟨c, hc⟩ : ∃ c : ℤ, ‖T‖ * max R 0 ≤ c := ⟨⌈‖T‖ * max R 0⌉, Int.le_ceil _⟩
  have hmap : ∀ G : MvPolynomial (Fin d) ℤ, G.totalDegree ≤ D →
      MvPolynomial.map (Int.castRingHom ℂ) G = polyOf fun m ↦ ((coeff (expVec m) G : ℤ) : ℂ) := by
    intro G hG
    ext s
    rw [coeff_map]
    by_cases h : ∃ m : Fin d → Fin (D + 1), expVec m = s
    · obtain ⟨m, rfl⟩ := h
      exact (coeff_polyOf (fun m ↦ ((coeff (expVec m) G : ℤ) : ℂ)) m).symm
    · have hs : s ∉ G.support := fun hs ↦ h (exists_expVec_eq hG hs)
      rw [coeff_polyOf_eq_zero _ fun m hm ↦ h ⟨m, hm⟩, notMem_support_iff.mp hs, map_zero]
  have hbound : ∀ G : MvPolynomial (Fin d) ℤ, G.totalDegree ≤ D →
      (∀ j : Fin d → ℕ, (∀ i, j i ≤ D) → ‖aeval (fun i ↦ t i (j i)) G‖ ≤ R) →
      ∀ m, coeff (expVec m) G ∈ Set.Icc (-c) c := by
    intro G hG hval m
    obtain ⟨v, hv⟩ : ∃ v : (Fin d → Fin (D + 1)) → ℂ, v = fun m ↦ ((coeff (expVec m) G : ℤ) : ℂ) :=
      ⟨_, rfl⟩
    have hMv : ‖gridMatrix t D *ᵥ v‖ ≤ max R 0 := by
      refine (pi_norm_le_iff_of_nonneg (le_max_right R 0)).mpr fun g' ↦ ?_
      rw [← eval_polyOf, hv, ← hmap G hG, eval_map, ← algebraMap_int_eq, ← aeval_def]
      exact (hval (fun i ↦ g' i) fun i ↦ Nat.le_of_lt_succ (g' i).isLt).trans (le_max_left R 0)
    have hvT : T (gridMatrix t D *ᵥ v) = v := by
      rw [hT]
      exact LinearMap.congr_fun hg v
    have hnorm : ‖((coeff (expVec m) G : ℤ) : ℂ)‖ ≤ c :=
      calc ‖((coeff (expVec m) G : ℤ) : ℂ)‖ = ‖v m‖ := by rw [hv]
        _ ≤ ‖v‖ := (pi_norm_le_iff_of_nonneg (norm_nonneg v)).mp le_rfl m
        _ = ‖T (gridMatrix t D *ᵥ v)‖ := by rw [hvT]
        _ ≤ ‖T‖ * ‖gridMatrix t D *ᵥ v‖ := T.le_opNorm _
        _ ≤ ‖T‖ * max R 0 := mul_le_mul_of_nonneg_left hMv (norm_nonneg T)
        _ ≤ c := hc
    rw [Complex.norm_intCast] at hnorm
    obtain ⟨h1, h2⟩ := abs_le.mp hnorm
    exact ⟨by exact_mod_cast h1, by exact_mod_cast h2⟩
  refine Set.Finite.of_finite_image
    (f := fun G (m : Fin d → Fin (D + 1)) ↦ coeff (expVec m) G) ?_ ?_
  · refine (Set.Finite.pi fun _ ↦ Set.finite_Icc (-c) c).subset ?_
    rintro _ ⟨G, ⟨hG, hval⟩, rfl⟩
    exact Set.mem_univ_pi.mpr fun m ↦ hbound G hG hval m
  · intro G hG G' hG' h
    ext s
    by_cases hs : ∃ m : Fin d → Fin (D + 1), expVec m = s
    · obtain ⟨m, rfl⟩ := hs
      exact congrFun h m
    · have h0 : ∀ H : MvPolynomial (Fin d) ℤ, H.totalDegree ≤ D → coeff s H = 0 :=
        fun H hH ↦ notMem_support_iff.mp fun hsH ↦ hs (exists_expVec_eq hH hsH)
      rw [h0 G hG.1, h0 G' hG'.1]

/-- There is a sequence of complex numbers algebraically independent over `ℤ`. -/
theorem exists_algebraicIndependent_nat : ∃ t : ℕ → ℂ, AlgebraicIndependent ℤ t := by
  obtain ⟨s, hs⟩ := exists_isTranscendenceBasis ℤ ℂ
  have hcard : #ℂ = #s :=
    IsAlgClosed.cardinal_eq_cardinal_transcendence_basis_of_aleph0_lt' _ hs Cardinal.mk_int.le
      (by rw [Cardinal.mk_complex]; exact Cardinal.aleph0_lt_continuum)
  have hinf : s.Infinite := by
    rw [← Set.infinite_coe_iff, Cardinal.infinite_iff, ← hcard, Cardinal.mk_complex]
    exact Cardinal.aleph0_lt_continuum.le
  exact ⟨_, hs.1.comp _ (hinf.natEmbedding s).injective⟩
