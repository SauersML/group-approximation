/-!
# The exterior face is off the pocket side of one of the two orders

Osin, arXiv:math/0411039v3, §9, proof of Lemma 9.7(b): two distinct selected regions `a`, `b`
joining the same two distinct cells `Π_i`, `Π_j` bound a subdiagram `Γ_1`.  Among the two pockets
between the regions, the one taken is a disc away from the exterior face.

The pocket walk of the order `(a, b)` is `w₁ = s_bj ++ inv(a_i G₁ b_i) ++ s_ai ++ inv(b_j G₂ a_j)`,
and that of the order `(b, a)` is `w₂ = s_aj ++ inv(b_i G₁' a_i) ++ s_bi ++ inv(a_j G₂' b_j)`.  Here
`s_ki` is the side of region `k` leaving `Π_i`, `k_i` is its arc on `Π_i`, and the `G` are gaps of
the carriers.  The side of `w₁` holds `a`, `b` and the pocket after `G₁`; the side of `w₂` holds `a`,
`b` and the pocket after `G₁'`.

The main theorem: when both walks are noncrossing, a face on both sides is a face of `a` or `b`.
The proof follows a face-class chain of `w₁` from a walk dart and carries the invariant "on the side
of `w₂` implies in `a` or `b`".
* Across an edge on neither walk, both memberships are unchanged.  Every boundary dart of a region
  lies in its two arcs and its two sides, and all four are on `w₁` or `w₂`.
* Across an edge of `w₂` only, the dart lies on a side of `w₂`, whose own face is in a region, or on
  a gap of `w₂`, whose other face is a cell.  The cells are off the side of `w₁`, so that step does
  not occur on the chain.
* At a gap dart of `w₁`, the cell across it is off the side of `w₂`.  So the face is off that side
  too, unless the dart is on `w₂`.  It can only be on a side of `w₂`, since the gaps of the two
  orders are disjoint (`CyclicArc.not_mem_of_complementary_spans`).
Faces of regions are never the exterior face (`FaceSetBoundary.all_gCells`).

* `CellPocketWalkOrder.mem_regions_of_mem_sideFaces_both`: the main theorem.
* `CellPocketWalkOrder.outerFace_not_mem_sideFaces_or`: the exterior face is off one of the sides.
* `CellPocketWalkOrder.outerFace_not_mem_sideFaces_or_of_walks`: the same from the data of
  `CellPocketWalk.exists_of_joinsCells` for both orders and the closed-walk facts, with noncrossing
  derived through `CellPocketWalkColour.walk_orient`.

No `CellPocketWalk.CopyClean` condition is used.

-/

section Generic

variable {M : CombMap.{v}}

/-- The face of a walk dart is on the side of the walk. -/
theorem faceOf_mem_sideFaces_of_mem {w : List M.Dart} {d : M.Dart} (hd : d ∈ w) :
    M.faceOf d ∈ sideFaces M w :=
  (mem_sideFaces_iff M w d).mpr ⟨d, hd, .refl _⟩

/-- The face across a dart of a noncrossing closed walk is off its side. -/
theorem faceOf_alpha_not_mem_sideFaces (hM : M.IsPlanar) {w : List M.Dart}
    (hw : IsNoncrossingClosedWalk M w) {d : M.Dart} (hd : d ∈ w) :
    M.faceOf (M.alpha d) ∉ sideFaces M w :=
  (show M.faceOf d ∈ sideFaces M w ∧ M.faceOf (M.alpha d) ∉ sideFaces M w from
    (hw.isBoundaryDart_sideFaces_iff hM d).mpr hd).2

end Generic

section Walk

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {eps : ℕ} {X : DiscDiagram.{u, w, v} W}

/-- A dart of `invDarts l` is the reversal of a dart of `l`. -/
theorem mem_invDarts_iff (l : List X.toCombMap.Dart) (d : X.toCombMap.Dart) :
    d ∈ invDarts X l ↔ X.toCombMap.alpha d ∈ l := by
  constructor
  · intro h
    change d ∈ l.reverse.map X.toCombMap.alpha at h
    obtain ⟨y, hy, rfl⟩ := List.mem_map.mp h
    rw [X.toCombMap.alpha_involutive y]
    exact List.mem_reverse.mp hy
  · intro h
    change d ∈ l.reverse.map X.toCombMap.alpha
    exact List.mem_map.mpr ⟨X.toCombMap.alpha d, List.mem_reverse.mpr h,
      X.toCombMap.alpha_involutive d⟩

/-- **The darts of a pocket walk**: the first side, the reversed first arc, the second side and the
reversed second arc. -/
theorem mem_walk_iff {i j : Fin X.rCellCount} (K : CellPocketWalk D eps X i j)
    (d : X.toCombMap.Dart) :
    d ∈ K.walk ↔ d ∈ K.firstSide ∨ X.toCombMap.alpha d ∈ K.firstArc.darts ∨
      d ∈ K.secondSide ∨ X.toCombMap.alpha d ∈ K.secondArc.darts := by
  show d ∈ K.firstSide ++ invDarts X K.firstArc.darts ++ K.secondSide ++
    invDarts X K.secondArc.darts ↔ _
  simp only [List.mem_append, mem_invDarts_iff, or_assoc]

/-- The darts of the first arc lie on the face of `Π_i`. -/
theorem faceOf_of_mem_firstArc {i j : Fin X.rCellCount} (K : CellPocketWalk D eps X i j)
    {d : X.toCombMap.Dart} (hd : d ∈ K.firstArc.darts) :
    X.toCombMap.faceOf d = (cell X i).face :=
  ((X.faceBoundary (cell X i).face).mem_iff d).mp (K.firstArc.mem_cycle_of_mem_darts hd)

/-- The darts of the second arc lie on the face of `Π_j`. -/
theorem faceOf_of_mem_secondArc {i j : Fin X.rCellCount} (K : CellPocketWalk D eps X i j)
    {d : X.toCombMap.Dart} (hd : d ∈ K.secondArc.darts) :
    X.toCombMap.faceOf d = (cell X j).face :=
  ((X.faceBoundary (cell X j).face).mem_iff d).mp (K.secondArc.mem_cycle_of_mem_darts hd)

/-- A dart of a side of a region joining two cells lies on a face of the region. -/
theorem faceOf_mem_of_mem_sideFrom {a : RegionCandidate D eps X} {i j : Fin X.rCellCount}
    (hij : i ≠ j) (hai : a.JoinsCells i j) {d : X.toCombMap.Dart}
    (hd : d ∈ a.sideFrom i ∨ d ∈ a.sideFrom j) : X.toCombMap.faceOf d ∈ a.1 := by
  obtain ⟨n, hn⟩ := RegionCandidate.boundary_cycle_rotate_of_joinsCells hij hai
  have hrot : d ∈ a.2.boundary.cycle.rotate n := by
    rw [hn]
    rcases hd with hd | hd <;> simp [hd]
  exact And.left ((a.2.boundary.cycle_mem_iff d).mp (List.mem_rotate.mp hrot))

/-- **The boundary darts of a region joining two cells** lie in its two arcs, read backwards, and
its two sides. -/
theorem mem_parts_of_boundary {a : RegionCandidate D eps X} {i j : Fin X.rCellCount}
    (hij : i ≠ j) (hai : a.JoinsCells i j) {x : X.toCombMap.Dart}
    (hx : X.toCombMap.faceOf x ∈ a.1) (hxα : X.toCombMap.faceOf (X.toCombMap.alpha x) ∉ a.1) :
    X.toCombMap.alpha x ∈ a.cellArcList i ∨ x ∈ a.sideFrom i ∨
      X.toCombMap.alpha x ∈ a.cellArcList j ∨ x ∈ a.sideFrom j := by
  obtain ⟨n, hn⟩ := RegionCandidate.boundary_cycle_rotate_of_joinsCells hij hai
  have hmem : x ∈ a.2.boundary.cycle := (a.2.boundary.cycle_mem_iff x).mpr (And.intro hx hxα)
  have hrot : x ∈ a.2.boundary.cycle.rotate n := List.mem_rotate.mpr hmem
  rw [hn] at hrot
  simp only [List.mem_append, mem_invDarts_iff, or_assoc] at hrot
  exact hrot

/-- **Region membership across an edge on neither walk.**  If the four boundary parts of a region
are on `w₁` or `w₂`, an edge on neither walk does not leave the region. -/
theorem faceOf_mem_iff_of_not_kept {a : RegionCandidate D eps X} {i j : Fin X.rCellCount}
    (hij : i ≠ j) (hai : a.JoinsCells i j) {w₁ w₂ : List X.toCombMap.Dart}
    (hparts : ∀ x, X.toCombMap.alpha x ∈ a.cellArcList i ∨ x ∈ a.sideFrom i ∨
      X.toCombMap.alpha x ∈ a.cellArcList j ∨ x ∈ a.sideFrom j →
        walkKeep X.toCombMap w₁ x ∨ walkKeep X.toCombMap w₂ x)
    {y : X.toCombMap.Dart} (hk₁ : ¬ walkKeep X.toCombMap w₁ y)
    (hk₂ : ¬ walkKeep X.toCombMap w₂ y) :
    X.toCombMap.faceOf y ∈ a.1 ↔ X.toCombMap.faceOf (X.toCombMap.alpha y) ∈ a.1 := by
  constructor
  · intro hy
    by_contra hα
    rcases hparts y (mem_parts_of_boundary hij hai hy hα) with h | h
    · exact hk₁ h
    · exact hk₂ h
  · intro hα
    by_contra hy
    have hy' : X.toCombMap.faceOf (X.toCombMap.alpha (X.toCombMap.alpha y)) ∉ a.1 := by
      rw [X.toCombMap.alpha_involutive y]
      exact hy
    rcases hparts (X.toCombMap.alpha y) (mem_parts_of_boundary hij hai hα hy') with h | h
    · exact hk₁ ((walkKeep_alpha X.toCombMap w₁ y).mp h)
    · exact hk₂ ((walkKeep_alpha X.toCombMap w₂ y).mp h)

end Walk

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {eps : ℕ} {lambda c : ℝ} {Delta : DiscDiagram.{u, w, v} W}
  {cuts : SectionCuts D lambda c Delta.boundaryWord}

/-- **A face on the sides of the pocket walks of both orders is a face of a region.**  The walk
`K₁` is spelled from the order `(a, b)` and `K₂` from the order `(b, a)`, both noncrossing. -/
theorem mem_regions_of_mem_sideFaces_both (S : RealizedSectionFamily D lambda c eps Delta cuts)
    {i j : Fin S.diagram.rCellCount} {a b : RegionCandidate D eps S.diagram}
    (ha : a ∈ S.family) (hb : b ∈ S.family) (hij : i ≠ j)
    (hai : a.JoinsCells i j) (hbi : b.JoinsCells i j)
    (K₁ K₂ : CellPocketWalk D eps S.diagram i j) {gi gj gi' gj' : List S.diagram.toCombMap.Dart}
    (hfirst₁ : K₁.firstSide = b.sideFrom j) (hsecond₁ : K₁.secondSide = a.sideFrom i)
    (harc₁ : K₁.firstArc.darts = a.cellArcList i ++ gi ++ b.cellArcList i)
    (harc₂ : K₁.secondArc.darts = b.cellArcList j ++ gj ++ a.cellArcList j)
    (hfirst₂ : K₂.firstSide = a.sideFrom j) (hsecond₂ : K₂.secondSide = b.sideFrom i)
    (harc₁' : K₂.firstArc.darts = b.cellArcList i ++ gi' ++ a.cellArcList i)
    (harc₂' : K₂.secondArc.darts = a.cellArcList j ++ gj' ++ b.cellArcList j)
    (hw₁ : IsNoncrossingClosedWalk S.diagram.toCombMap K₁.walk)
    (hw₂ : IsNoncrossingClosedWalk S.diagram.toCombMap K₂.walk)
    {F : S.diagram.toCombMap.Face}
    (hF₁ : F ∈ sideFaces S.diagram.toCombMap K₁.walk)
    (hF₂ : F ∈ sideFaces S.diagram.toCombMap K₂.walk) : F ∈ a.1 ∨ F ∈ b.1 := by
  -- The region arcs are nonempty.
  have hai_ne : a.cellArcList i ≠ [] := List.ne_nil_of_length_pos
    (RegionCandidate.cellArcList_length_pos (S.nondegenerate a ha) i)
  have hbi_ne : b.cellArcList i ≠ [] := List.ne_nil_of_length_pos
    (RegionCandidate.cellArcList_length_pos (S.nondegenerate b hb) i)
  have haj_ne : a.cellArcList j ≠ [] := List.ne_nil_of_length_pos
    (RegionCandidate.cellArcList_length_pos (S.nondegenerate a ha) j)
  have hbj_ne : b.cellArcList j ≠ [] := List.ne_nil_of_length_pos
    (RegionCandidate.cellArcList_length_pos (S.nondegenerate b hb) j)
  have hΠ : (cell S.diagram i).face ≠ (cell S.diagram j).face :=
    fun h => hij (cell_face_injective S.diagram h)
  -- Both cells are off both sides.
  obtain ⟨ci, hci⟩ := List.exists_mem_of_ne_nil _ hai_ne
  obtain ⟨cj, hcj⟩ := List.exists_mem_of_ne_nil _ haj_ne
  have hci₁ : ci ∈ K₁.firstArc.darts := by
    rw [harc₁]
    simp [hci]
  have hci₂ : ci ∈ K₂.firstArc.darts := by
    rw [harc₁']
    simp [hci]
  have hcj₁ : cj ∈ K₁.secondArc.darts := by
    rw [harc₂]
    simp [hcj]
  have hcj₂ : cj ∈ K₂.secondArc.darts := by
    rw [harc₂']
    simp [hcj]
  have hΠi₁ : (cell S.diagram i).face ∉ sideFaces S.diagram.toCombMap K₁.walk := by
    have hmem : S.diagram.toCombMap.alpha ci ∈ K₁.walk := (mem_walk_iff K₁ _).mpr
      (Or.inr (Or.inl (by rw [S.diagram.toCombMap.alpha_involutive]; exact hci₁)))
    have h := faceOf_alpha_not_mem_sideFaces S.diagram.planar hw₁ hmem
    rwa [S.diagram.toCombMap.alpha_involutive, faceOf_of_mem_firstArc K₁ hci₁] at h
  have hΠi₂ : (cell S.diagram i).face ∉ sideFaces S.diagram.toCombMap K₂.walk := by
    have hmem : S.diagram.toCombMap.alpha ci ∈ K₂.walk := (mem_walk_iff K₂ _).mpr
      (Or.inr (Or.inl (by rw [S.diagram.toCombMap.alpha_involutive]; exact hci₂)))
    have h := faceOf_alpha_not_mem_sideFaces S.diagram.planar hw₂ hmem
    rwa [S.diagram.toCombMap.alpha_involutive, faceOf_of_mem_firstArc K₂ hci₂] at h
  have hΠj₁ : (cell S.diagram j).face ∉ sideFaces S.diagram.toCombMap K₁.walk := by
    have hmem : S.diagram.toCombMap.alpha cj ∈ K₁.walk := (mem_walk_iff K₁ _).mpr
      (Or.inr (Or.inr (Or.inr (by rw [S.diagram.toCombMap.alpha_involutive]; exact hcj₁))))
    have h := faceOf_alpha_not_mem_sideFaces S.diagram.planar hw₁ hmem
    rwa [S.diagram.toCombMap.alpha_involutive, faceOf_of_mem_secondArc K₁ hcj₁] at h
  have hΠj₂ : (cell S.diagram j).face ∉ sideFaces S.diagram.toCombMap K₂.walk := by
    have hmem : S.diagram.toCombMap.alpha cj ∈ K₂.walk := (mem_walk_iff K₂ _).mpr
      (Or.inr (Or.inr (Or.inr (by rw [S.diagram.toCombMap.alpha_involutive]; exact hcj₂))))
    have h := faceOf_alpha_not_mem_sideFaces S.diagram.planar hw₂ hmem
    rwa [S.diagram.toCombMap.alpha_involutive, faceOf_of_mem_secondArc K₂ hcj₂] at h
  -- The arcs of `K₁` have no repeated dart.
  have hnd₁ : (a.cellArcList i ++ gi ++ b.cellArcList i).Nodup := by
    rw [← harc₁]
    exact K₁.firstArc.darts_nodup (cellDarts_nodup S.diagram i)
  have hnd₂ : (b.cellArcList j ++ gj ++ a.cellArcList j).Nodup := by
    rw [← harc₂]
    exact K₁.secondArc.darts_nodup (cellDarts_nodup S.diagram j)
  -- Darts whose reversal lies on an arc of `K₁` are kept by `K₁`.
  have hkeep₁ : ∀ x, S.diagram.toCombMap.alpha x ∈ K₁.firstArc.darts →
      walkKeep S.diagram.toCombMap K₁.walk x :=
    fun x hx => Or.inl ((mem_walk_iff K₁ x).mpr (Or.inr (Or.inl hx)))
  have hkeep₂ : ∀ x, S.diagram.toCombMap.alpha x ∈ K₁.secondArc.darts →
      walkKeep S.diagram.toCombMap K₁.walk x :=
    fun x hx => Or.inl ((mem_walk_iff K₁ x).mpr (Or.inr (Or.inr (Or.inr hx))))
  -- The four boundary parts of each region are on one of the walks.
  have hparts_a : ∀ x, S.diagram.toCombMap.alpha x ∈ a.cellArcList i ∨ x ∈ a.sideFrom i ∨
      S.diagram.toCombMap.alpha x ∈ a.cellArcList j ∨ x ∈ a.sideFrom j →
        walkKeep S.diagram.toCombMap K₁.walk x ∨ walkKeep S.diagram.toCombMap K₂.walk x := by
    intro x hx
    rcases hx with h | h | h | h
    · exact Or.inl (hkeep₁ x (by rw [harc₁]; simp [h]))
    · exact Or.inl (Or.inl ((mem_walk_iff K₁ x).mpr
        (Or.inr (Or.inr (Or.inl (by rw [hsecond₁]; exact h))))))
    · exact Or.inl (hkeep₂ x (by rw [harc₂]; simp [h]))
    · exact Or.inr (Or.inl ((mem_walk_iff K₂ x).mpr (Or.inl (by rw [hfirst₂]; exact h))))
  have hparts_b : ∀ x, S.diagram.toCombMap.alpha x ∈ b.cellArcList i ∨ x ∈ b.sideFrom i ∨
      S.diagram.toCombMap.alpha x ∈ b.cellArcList j ∨ x ∈ b.sideFrom j →
        walkKeep S.diagram.toCombMap K₁.walk x ∨ walkKeep S.diagram.toCombMap K₂.walk x := by
    intro x hx
    rcases hx with h | h | h | h
    · exact Or.inl (hkeep₁ x (by rw [harc₁]; simp [h]))
    · exact Or.inr (Or.inl ((mem_walk_iff K₂ x).mpr
        (Or.inr (Or.inr (Or.inl (by rw [hsecond₂]; exact h))))))
    · exact Or.inl (hkeep₂ x (by rw [harc₂]; simp [h]))
    · exact Or.inl (Or.inl ((mem_walk_iff K₁ x).mpr (Or.inl (by rw [hfirst₁]; exact h))))
  -- The invariant at the darts of `K₁`.
  have hbase : ∀ d ∈ K₁.walk,
      S.diagram.toCombMap.faceOf d ∈ sideFaces S.diagram.toCombMap K₂.walk →
        S.diagram.toCombMap.faceOf d ∈ a.1 ∨ S.diagram.toCombMap.faceOf d ∈ b.1 := by
    intro d hd hd₂
    rcases (mem_walk_iff K₁ d).mp hd with h | h | h | h
    · rw [hfirst₁] at h
      exact Or.inr (faceOf_mem_of_mem_sideFrom hij hbi (Or.inr h))
    · have hcell := faceOf_of_mem_firstArc K₁ h
      rw [harc₁] at h
      simp only [List.mem_append] at h
      rcases h with (h | h) | h
      · left
        have h' := RegionCandidate.faceOf_alpha_mem_of_mem_cellArcList hai h
        rwa [S.diagram.toCombMap.alpha_involutive] at h'
      · by_cases hk : walkKeep S.diagram.toCombMap K₂.walk d
        · rcases hk with hk | hk
          · rcases (mem_walk_iff K₂ d).mp hk with h' | h' | h' | h'
            · rw [hfirst₂] at h'
              exact Or.inl (faceOf_mem_of_mem_sideFrom hij hai (Or.inr h'))
            · rw [harc₁'] at h'
              simp only [List.mem_append] at h'
              rcases h' with (h' | h') | h'
              · exact ((List.nodup_append.mp hnd₁).2.2 _ (List.mem_append_right _ h) _ h'
                  rfl).elim
              · exact (CyclicArc.not_mem_of_complementary_spans (cellDarts_nodup S.diagram i)
                  K₁.firstArc K₂.firstArc hai_ne hbi_ne harc₁ harc₁' _ h h').elim
              · exact ((List.nodup_append.mp (List.nodup_append.mp hnd₁).1).2.2 _ h' _ h
                  rfl).elim
            · rw [hsecond₂] at h'
              exact Or.inr (faceOf_mem_of_mem_sideFrom hij hbi (Or.inl h'))
            · exact (hΠ (hcell.symm.trans (faceOf_of_mem_secondArc K₂ h'))).elim
          · exact (hΠi₂ (hcell ▸ faceOf_mem_sideFaces_of_mem hk)).elim
        · have h' := (mem_sideFaces_iff_of_not_walkKeep K₂.walk hk).mp hd₂
          rw [hcell] at h'
          exact (hΠi₂ h').elim
      · right
        have h' := RegionCandidate.faceOf_alpha_mem_of_mem_cellArcList hbi h
        rwa [S.diagram.toCombMap.alpha_involutive] at h'
    · rw [hsecond₁] at h
      exact Or.inl (faceOf_mem_of_mem_sideFrom hij hai (Or.inl h))
    · have hcell := faceOf_of_mem_secondArc K₁ h
      rw [harc₂] at h
      simp only [List.mem_append] at h
      rcases h with (h | h) | h
      · right
        have h' := RegionCandidate.faceOf_alpha_mem_of_mem_cellArcList hbi h
        rwa [S.diagram.toCombMap.alpha_involutive] at h'
      · by_cases hk : walkKeep S.diagram.toCombMap K₂.walk d
        · rcases hk with hk | hk
          · rcases (mem_walk_iff K₂ d).mp hk with h' | h' | h' | h'
            · rw [hfirst₂] at h'
              exact Or.inl (faceOf_mem_of_mem_sideFrom hij hai (Or.inr h'))
            · exact (hΠ ((faceOf_of_mem_firstArc K₂ h').symm.trans hcell)).elim
            · rw [hsecond₂] at h'
              exact Or.inr (faceOf_mem_of_mem_sideFrom hij hbi (Or.inl h'))
            · rw [harc₂'] at h'
              simp only [List.mem_append] at h'
              rcases h' with (h' | h') | h'
              · exact ((List.nodup_append.mp hnd₂).2.2 _ (List.mem_append_right _ h) _ h'
                  rfl).elim
              · exact (CyclicArc.not_mem_of_complementary_spans (cellDarts_nodup S.diagram j)
                  K₁.secondArc K₂.secondArc hbj_ne haj_ne harc₂ harc₂' _ h h').elim
              · exact ((List.nodup_append.mp (List.nodup_append.mp hnd₂).1).2.2 _ h' _ h
                  rfl).elim
          · exact (hΠj₂ (hcell ▸ faceOf_mem_sideFaces_of_mem hk)).elim
        · have h' := (mem_sideFaces_iff_of_not_walkKeep K₂.walk hk).mp hd₂
          rw [hcell] at h'
          exact (hΠj₂ h').elim
      · left
        have h' := RegionCandidate.faceOf_alpha_mem_of_mem_cellArcList hai h
        rwa [S.diagram.toCombMap.alpha_involutive] at h'
  -- One step of a face-class chain of `K₁` keeps the invariant.
  have hstep : ∀ y z, CombMap.FaceClassStep S.diagram.toCombMap
      (walkKeep S.diagram.toCombMap K₁.walk) y z →
        S.diagram.toCombMap.faceOf y ∈ sideFaces S.diagram.toCombMap K₁.walk →
          ((S.diagram.toCombMap.faceOf y ∈ sideFaces S.diagram.toCombMap K₂.walk →
              S.diagram.toCombMap.faceOf y ∈ a.1 ∨ S.diagram.toCombMap.faceOf y ∈ b.1) ↔
            (S.diagram.toCombMap.faceOf z ∈ sideFaces S.diagram.toCombMap K₂.walk →
              S.diagram.toCombMap.faceOf z ∈ a.1 ∨ S.diagram.toCombMap.faceOf z ∈ b.1)) := by
    intro y z hyz hy₁
    rcases hyz with rfl | ⟨hk₁, rfl⟩
    · rw [S.diagram.toCombMap.faceOf_facePerm]
    · have hα₁ := (mem_sideFaces_iff_of_not_walkKeep K₁.walk hk₁).mp hy₁
      by_cases hk₂ : walkKeep S.diagram.toCombMap K₂.walk y
      · rcases hk₂ with hy₂ | hy₂
        · have hU : S.diagram.toCombMap.faceOf y ∈ a.1 ∨ S.diagram.toCombMap.faceOf y ∈ b.1 := by
            rcases (mem_walk_iff K₂ y).mp hy₂ with h | h | h | h
            · rw [hfirst₂] at h
              exact Or.inl (faceOf_mem_of_mem_sideFrom hij hai (Or.inr h))
            · have hcell := faceOf_of_mem_firstArc K₂ h
              rw [harc₁'] at h
              simp only [List.mem_append] at h
              rcases h with (h | h) | h
              · exact (hk₁ (hkeep₁ y (by rw [harc₁]; simp [h]))).elim
              · rw [hcell] at hα₁
                exact (hΠi₁ hα₁).elim
              · exact (hk₁ (hkeep₁ y (by rw [harc₁]; simp [h]))).elim
            · rw [hsecond₂] at h
              exact Or.inr (faceOf_mem_of_mem_sideFrom hij hbi (Or.inl h))
            · have hcell := faceOf_of_mem_secondArc K₂ h
              rw [harc₂'] at h
              simp only [List.mem_append] at h
              rcases h with (h | h) | h
              · exact (hk₁ (hkeep₂ y (by rw [harc₂]; simp [h]))).elim
              · rw [hcell] at hα₁
                exact (hΠj₁ hα₁).elim
              · exact (hk₁ (hkeep₂ y (by rw [harc₂]; simp [h]))).elim
          constructor
          · intro _ h
            exact (faceOf_alpha_not_mem_sideFaces S.diagram.planar hw₂ hy₂ h).elim
          · intro _ _
            exact hU
        · have hU : S.diagram.toCombMap.faceOf (S.diagram.toCombMap.alpha y) ∈ a.1 ∨
              S.diagram.toCombMap.faceOf (S.diagram.toCombMap.alpha y) ∈ b.1 := by
            rcases (mem_walk_iff K₂ _).mp hy₂ with h | h | h | h
            · rw [hfirst₂] at h
              exact Or.inl (faceOf_mem_of_mem_sideFrom hij hai (Or.inr h))
            · rw [S.diagram.toCombMap.alpha_involutive] at h
              have hcell := faceOf_of_mem_firstArc K₂ h
              rw [harc₁'] at h
              simp only [List.mem_append] at h
              rcases h with (h | h) | h
              · exact (hk₁ (Or.inr ((mem_walk_iff K₁ _).mpr (Or.inr (Or.inl
                  (by rw [S.diagram.toCombMap.alpha_involutive, harc₁]; simp [h])))))).elim
              · rw [hcell] at hy₁
                exact (hΠi₁ hy₁).elim
              · exact (hk₁ (Or.inr ((mem_walk_iff K₁ _).mpr (Or.inr (Or.inl
                  (by rw [S.diagram.toCombMap.alpha_involutive, harc₁]; simp [h])))))).elim
            · rw [hsecond₂] at h
              exact Or.inr (faceOf_mem_of_mem_sideFrom hij hbi (Or.inl h))
            · rw [S.diagram.toCombMap.alpha_involutive] at h
              have hcell := faceOf_of_mem_secondArc K₂ h
              rw [harc₂'] at h
              simp only [List.mem_append] at h
              rcases h with (h | h) | h
              · exact (hk₁ (Or.inr ((mem_walk_iff K₁ _).mpr (Or.inr (Or.inr (Or.inr
                  (by rw [S.diagram.toCombMap.alpha_involutive, harc₂]; simp [h]))))))).elim
              · rw [hcell] at hy₁
                exact (hΠj₁ hy₁).elim
              · exact (hk₁ (Or.inr ((mem_walk_iff K₁ _).mpr (Or.inr (Or.inr (Or.inr
                  (by rw [S.diagram.toCombMap.alpha_involutive, harc₂]; simp [h]))))))).elim
          have hnot : S.diagram.toCombMap.faceOf y ∉ sideFaces S.diagram.toCombMap K₂.walk := by
            have h := faceOf_alpha_not_mem_sideFaces S.diagram.planar hw₂ hy₂
            rwa [S.diagram.toCombMap.alpha_involutive] at h
          constructor
          · intro _ _
            exact hU
          · intro _ h
            exact (hnot h).elim
      · have htr := mem_sideFaces_iff_of_not_walkKeep K₂.walk hk₂
        have ha' := faceOf_mem_iff_of_not_kept hij hai hparts_a hk₁ hk₂
        have hb' := faceOf_mem_iff_of_not_kept hij hbi hparts_b hk₁ hk₂
        exact imp_congr htr (or_congr ha' hb')
  -- The invariant along a face-class chain of `K₁`.
  have hprop : ∀ y z, Relation.EqvGen (CombMap.FaceClassStep S.diagram.toCombMap
      (walkKeep S.diagram.toCombMap K₁.walk)) y z →
        S.diagram.toCombMap.faceOf y ∈ sideFaces S.diagram.toCombMap K₁.walk →
          ((S.diagram.toCombMap.faceOf y ∈ sideFaces S.diagram.toCombMap K₂.walk →
              S.diagram.toCombMap.faceOf y ∈ a.1 ∨ S.diagram.toCombMap.faceOf y ∈ b.1) ↔
            (S.diagram.toCombMap.faceOf z ∈ sideFaces S.diagram.toCombMap K₂.walk →
              S.diagram.toCombMap.faceOf z ∈ a.1 ∨ S.diagram.toCombMap.faceOf z ∈ b.1)) := by
    intro y z h
    induction h with
    | rel y z hyz => exact hstep y z hyz
    | refl y =>
        intro _
        exact Iff.rfl
    | symm y z hyz ih =>
        intro hz
        exact (ih ((mem_sideFaces_iff_of_eqvGen K₁.walk hyz).mpr hz)).symm
    | trans y z t hyz _ ih₁ ih₂ =>
        intro hy
        exact (ih₁ hy).trans (ih₂ ((mem_sideFaces_iff_of_eqvGen K₁.walk hyz).mp hy))
  obtain ⟨x, hx⟩ := Quotient.exists_rep F
  have hx' : S.diagram.toCombMap.faceOf x = F := hx
  subst hx'
  obtain ⟨d, hd, hdx⟩ := (mem_sideFaces_iff S.diagram.toCombMap K₁.walk x).mp hF₁
  exact (hprop d x hdx (faceOf_mem_sideFaces_of_mem hd)).mp (hbase d hd) hF₂

/-- **The exterior face is off the side of one of the two orders.** -/
theorem outerFace_not_mem_sideFaces_or (S : RealizedSectionFamily D lambda c eps Delta cuts)
    {i j : Fin S.diagram.rCellCount} {a b : RegionCandidate D eps S.diagram}
    (ha : a ∈ S.family) (hb : b ∈ S.family) (hij : i ≠ j)
    (hai : a.JoinsCells i j) (hbi : b.JoinsCells i j)
    (K₁ K₂ : CellPocketWalk D eps S.diagram i j) {gi gj gi' gj' : List S.diagram.toCombMap.Dart}
    (hfirst₁ : K₁.firstSide = b.sideFrom j) (hsecond₁ : K₁.secondSide = a.sideFrom i)
    (harc₁ : K₁.firstArc.darts = a.cellArcList i ++ gi ++ b.cellArcList i)
    (harc₂ : K₁.secondArc.darts = b.cellArcList j ++ gj ++ a.cellArcList j)
    (hfirst₂ : K₂.firstSide = a.sideFrom j) (hsecond₂ : K₂.secondSide = b.sideFrom i)
    (harc₁' : K₂.firstArc.darts = b.cellArcList i ++ gi' ++ a.cellArcList i)
    (harc₂' : K₂.secondArc.darts = a.cellArcList j ++ gj' ++ b.cellArcList j)
    (hw₁ : IsNoncrossingClosedWalk S.diagram.toCombMap K₁.walk)
    (hw₂ : IsNoncrossingClosedWalk S.diagram.toCombMap K₂.walk) :
    S.diagram.outerFace ∉ sideFaces S.diagram.toCombMap K₁.walk ∨
      S.diagram.outerFace ∉ sideFaces S.diagram.toCombMap K₂.walk := by
  by_cases h₁ : S.diagram.outerFace ∈ sideFaces S.diagram.toCombMap K₁.walk
  · by_cases h₂ : S.diagram.outerFace ∈ sideFaces S.diagram.toCombMap K₂.walk
    · rcases mem_regions_of_mem_sideFaces_both S ha hb hij hai hbi K₁ K₂ hfirst₁ hsecond₁ harc₁
        harc₂ hfirst₂ hsecond₂ harc₁' harc₂' hw₁ hw₂ h₁ h₂ with h | h
      · exact ((a.2.boundary.all_gCells _ h).1 rfl).elim
      · exact ((b.2.boundary.all_gCells _ h).1 rfl).elim
    · exact Or.inr h₂
  · exact Or.inl h₁

/-- **The exterior face is off one of the two sides, from the walk data.**  `K₁` carries the data
of `CellPocketWalk.exists_of_joinsCells` for `(a, b)` and `K₂` for `(b, a)`, each with the
closed-walk facts.  Both walks are noncrossing by `CellPocketWalkColour.walk_orient`. -/
theorem outerFace_not_mem_sideFaces_or_of_walks
    (S : RealizedSectionFamily D lambda c eps Delta cuts)
    {i j : Fin S.diagram.rCellCount} {a b : RegionCandidate D eps S.diagram}
    (ha : a ∈ S.family) (hb : b ∈ S.family) (hab : a ≠ b) (hij : i ≠ j)
    (hai : a.JoinsCells i j) (hbi : b.JoinsCells i j) (K₁ K₂ : CellPocketWalk D eps S.diagram i j)
    (hfirst₁ : K₁.firstSide = b.sideFrom j) (hsecond₁ : K₁.secondSide = a.sideFrom i)
    (harc₁ : ∃ G₁ : CyclicArc (cellDarts S.diagram i),
      K₁.firstArc.darts = a.cellArcList i ++ G₁.darts ++ b.cellArcList i)
    (harc₂ : ∃ G₂ : CyclicArc (cellDarts S.diagram j),
      K₁.secondArc.darts = b.cellArcList j ++ G₂.darts ++ a.cellArcList j)
    (hne₁ : K₁.walk ≠ [])
    (hchain₁ : K₁.walk.IsChain fun d e => S.diagram.toCombMap.vertexOf
      (S.diagram.toCombMap.alpha d) = S.diagram.toCombMap.vertexOf e)
    (hcloses₁ : S.diagram.toCombMap.vertexOf (S.diagram.toCombMap.alpha (K₁.walk.getLast hne₁)) =
      S.diagram.toCombMap.vertexOf (K₁.walk.head hne₁))
    (hnodup₁ : K₁.walk.Nodup) (halpha₁ : ∀ d ∈ K₁.walk, S.diagram.toCombMap.alpha d ∉ K₁.walk)
    (hfirst₂ : K₂.firstSide = a.sideFrom j) (hsecond₂ : K₂.secondSide = b.sideFrom i)
    (harc₁' : ∃ G₁ : CyclicArc (cellDarts S.diagram i),
      K₂.firstArc.darts = b.cellArcList i ++ G₁.darts ++ a.cellArcList i)
    (harc₂' : ∃ G₂ : CyclicArc (cellDarts S.diagram j),
      K₂.secondArc.darts = a.cellArcList j ++ G₂.darts ++ b.cellArcList j)
    (hne₂ : K₂.walk ≠ [])
    (hchain₂ : K₂.walk.IsChain fun d e => S.diagram.toCombMap.vertexOf
      (S.diagram.toCombMap.alpha d) = S.diagram.toCombMap.vertexOf e)
    (hcloses₂ : S.diagram.toCombMap.vertexOf (S.diagram.toCombMap.alpha (K₂.walk.getLast hne₂)) =
      S.diagram.toCombMap.vertexOf (K₂.walk.head hne₂))
    (hnodup₂ : K₂.walk.Nodup) (halpha₂ : ∀ d ∈ K₂.walk, S.diagram.toCombMap.alpha d ∉ K₂.walk) :
    S.diagram.outerFace ∉ sideFaces S.diagram.toCombMap K₁.walk ∨
      S.diagram.outerFace ∉ sideFaces S.diagram.toCombMap K₂.walk := by
  have hw₁ := ClosedWalkFaceColouring.isNoncrossingClosedWalk_of_orient S.diagram.planar hne₁
    hnodup₁ halpha₁ hchain₁ hcloses₁ fun _ hcol =>
      CellPocketWalkColour.walk_orient S ha hb hab hai hbi K₁ hfirst₁ hsecond₁ harc₁ harc₂ hcol
  have hw₂ := ClosedWalkFaceColouring.isNoncrossingClosedWalk_of_orient S.diagram.planar hne₂
    hnodup₂ halpha₂ hchain₂ hcloses₂ fun _ hcol =>
      CellPocketWalkColour.walk_orient S hb ha hab.symm hbi hai K₂ hfirst₂ hsecond₂ harc₁' harc₂'
        hcol
  obtain ⟨G₁, hG₁⟩ := harc₁
  obtain ⟨G₂, hG₂⟩ := harc₂
  obtain ⟨G₁', hG₁'⟩ := harc₁'
  obtain ⟨G₂', hG₂'⟩ := harc₂'
  exact outerFace_not_mem_sideFaces_or S ha hb hij hai hbi K₁ K₂ hfirst₁ hsecond₁ hG₁ hG₂
    hfirst₂ hsecond₂ hG₁' hG₂' hw₁ hw₂

end CellPocketWalkOrder
