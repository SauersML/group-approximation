/-!
# Locally finite kernels give locally finite congruence kernels

`non_mf_groups_exist.tex`, `sec:chain-core`, the remark after `cor:dynamic-rank-budget`
(tex 1794–1805):

> If the core $Y$ consists of fixed points, every Kazhdan subgroup of $\GL_2(R_X)$ is finite.
> [...] The remaining kernel is locally finite, [...] The same reasoning applies to unit groups
> when every core point has period at most two [...]

The remaining kernel is the kernel of `GL_2(R_X) → GL_2(R_Y)` (resp. `R_X^× → R_Y^×`).  The
deviations `g - 1` of its elements have entries in the transient ideal `I`, which is locally
matricial over the finite field `k` (`lem:transient-matrices`), so every finite subset of `I` lies
in a finite subring of `I`.

* `HasLocallyFiniteKernel ρ`: every finite subset of `ker ρ` lies in a finite non-unital subring
  contained in `ker ρ`;
* `hasLocallyFiniteKernel_of_isLocallyMatricialIn`: a locally matricial kernel over a finite field
  is locally finite;
* `inv_mem_of_finite_of_mul_mem`: a finite subset of a group that contains `1` and is closed under
  multiplication is closed under inverses;
* `glMap n ρ : GL_n(R) → GL_n(S)` for rings that need not be commutative;
* `isLocallyFiniteGroup_ker_glMap`, `isLocallyFiniteGroup_ker_unitsMap`: the kernels of
  `GL_n(R) → GL_n(S)` and of `Rˣ → Sˣ` are locally finite groups.

The argument: the deviations of finitely many kernel elements lie in one finite subring `B`; the
units whose deviations lie in `B` form a finite set closed under multiplication, hence a finite
subgroup containing them.
-/

namespace GroupApproximation
namespace ChainCore

variable {R S : Type*} [Ring R] [Ring S]

/-- The kernel of `ρ` is **locally finite**: every finite subset of `ker ρ` lies in a finite
non-unital subring contained in `ker ρ`. -/
def HasLocallyFiniteKernel (ρ : R →+* S) : Prop :=
  ∀ F : Finset R, (∀ x ∈ F, ρ x = 0) →
    ∃ B : NonUnitalSubring R, (B : Set R).Finite ∧ (∀ x ∈ F, x ∈ B) ∧ ∀ x ∈ B, ρ x = 0

/-- A locally matricial kernel over a finite field is locally finite: the finitely many matrix
algebras containing a finite subset are finite. -/
theorem hasLocallyFiniteKernel_of_isLocallyMatricialIn {k : Type*} [Field k] [Finite k]
    [Module k R] (ρ : R →+* S) (h : IsLocallyMatricialIn k {x : R | ρ x = 0}) :
    HasLocallyFiniteKernel ρ := by
  intro F hF
  obtain ⟨m, hm, φ, -, hφS, hFφ, hfin⟩ := h.exists_finite F hF
  refine ⟨NonUnitalRingHom.range (φ : (∀ a : Fin m, Matrix (Fin (hm a)) (Fin (hm a)) k) →ₙ+* R),
    ?_, fun x hx ↦ ?_, fun x hx ↦ ?_⟩
  · rw [NonUnitalRingHom.coe_range]
    exact hfin
  · obtain ⟨y, hy⟩ := hFφ x hx
    exact NonUnitalRingHom.mem_range.mpr ⟨y, hy⟩
  · obtain ⟨y, rfl⟩ := NonUnitalRingHom.mem_range.mp hx
    exact hφS y

/-- In a group, a finite set containing `1` and closed under multiplication is closed under
inverses: its elements have finite order. -/
theorem inv_mem_of_finite_of_mul_mem {G : Type*} [Group G] {M : Set G} (hM : M.Finite)
    (h1 : (1 : G) ∈ M) (hmul : ∀ a ∈ M, ∀ b ∈ M, a * b ∈ M) {g : G} (hg : g ∈ M) :
    g⁻¹ ∈ M := by
  have hpow : ∀ n : ℕ, g ^ n ∈ M := by
    intro n
    induction n with
    | zero => simpa using h1
    | succ n ih =>
      rw [pow_succ]
      exact hmul _ ih _ hg
  have key : ∀ {a b : ℕ}, a < b → g ^ a = g ^ b → g⁻¹ ∈ M := by
    intro a b hlt he
    have hone : g ^ (b - a) = 1 := by
      have h2 : g ^ a * g ^ (b - a) = g ^ a * 1 := by
        rw [pow_mul_pow_sub g hlt.le, mul_one, he]
      exact mul_left_cancel h2
    have hinv : g⁻¹ = g ^ (b - a - 1) := by
      rw [inv_eq_iff_mul_eq_one, ← pow_succ', Nat.sub_add_cancel (by omega : 1 ≤ b - a), hone]
    rw [hinv]
    exact hpow _
  haveI : Finite M := hM.to_subtype
  obtain ⟨a, b, hab, heq⟩ :=
    Finite.exists_ne_map_eq_of_infinite fun n : ℕ ↦ (⟨g ^ n, hpow n⟩ : M)
  have heq' : g ^ a = g ^ b := congrArg Subtype.val heq
  rcases lt_or_gt_of_ne hab with h | h
  · exact key h heq'
  · exact key h heq'.symm

/-- `GL_n(ρ) : GL_n(R) → GL_n(S)`, for a ring hom between rings that need not be commutative. -/
def glMap (n : Type*) [Fintype n] [DecidableEq n] (ρ : R →+* S) : GL n R →* GL n S :=
  Units.map (RingHom.mapMatrix ρ : Matrix n n R →* Matrix n n S)

/-- **The general linear congruence kernel is locally finite.**  If every finite subset of
`ker ρ` lies in a finite subring of `ker ρ`, then the kernel of `GL_n(ρ)` is a locally finite
group. -/
theorem isLocallyFiniteGroup_ker_glMap {n : Type*} [Fintype n] [DecidableEq n] (ρ : R →+* S)
    (hρ : HasLocallyFiniteKernel ρ) : IsLocallyFiniteGroup (glMap n ρ).ker := by
  classical
  intro s
  let dev : GL n R → n → n → R := fun g i j ↦ (g : Matrix n n R) i j - (1 : Matrix n n R) i j
  let F : Finset R := s.biUnion fun g : (glMap n ρ).ker ↦
    Finset.univ.image fun p : n × n ↦ dev (g : GL n R) p.1 p.2
  have hF : ∀ x ∈ F, ρ x = 0 := by
    intro x hx
    simp only [F, Finset.mem_biUnion, Finset.mem_image, Finset.mem_univ, true_and] at hx
    obtain ⟨g, -, p, rfl⟩ := hx
    have hg : glMap n ρ (g : GL n R) = 1 := MonoidHom.mem_ker.mp g.2
    have he : ρ (((g : GL n R) : Matrix n n R) p.1 p.2) = (1 : Matrix n n S) p.1 p.2 :=
      congrArg (fun z : GL n S ↦ (z : Matrix n n S) p.1 p.2) hg
    show ρ (((g : GL n R) : Matrix n n R) p.1 p.2 - (1 : Matrix n n R) p.1 p.2) = 0
    rw [map_sub, sub_eq_zero, he, Matrix.one_apply, Matrix.one_apply]
    split_ifs <;> simp
  obtain ⟨B, hBfin, hFB, -⟩ := hρ F hF
  let M : Set (GL n R) := {g | ∀ i j, dev g i j ∈ B}
  have h1 : (1 : GL n R) ∈ M := fun i j ↦ by
    show ((1 : GL n R) : Matrix n n R) i j - (1 : Matrix n n R) i j ∈ B
    rw [Units.val_one, sub_self]
    exact zero_mem B
  have hmul : ∀ a ∈ M, ∀ b ∈ M, a * b ∈ M := by
    intro a ha b hb i j
    have hm : ((a * b : GL n R) : Matrix n n R) - 1 =
        ((a : Matrix n n R) - 1) * ((b : Matrix n n R) - 1) + ((a : Matrix n n R) - 1) +
          ((b : Matrix n n R) - 1) := by
      rw [Units.val_mul]
      noncomm_ring
    show ((a * b : GL n R) : Matrix n n R) i j - (1 : Matrix n n R) i j ∈ B
    rw [← Matrix.sub_apply, hm, Matrix.add_apply, Matrix.add_apply, Matrix.mul_apply]
    exact add_mem (add_mem (sum_mem fun l _ ↦ mul_mem (ha i l) (hb l j)) (ha i j)) (hb i j)
  have hMfin : M.Finite := by
    haveI : Finite B := hBfin.to_subtype
    have hinj : Function.Injective fun g : M ↦
        fun i j ↦ (⟨dev (g : GL n R) i j, g.2 i j⟩ : B) := by
      intro g h hgh
      refine Subtype.ext (Units.ext (Matrix.ext fun i j ↦ ?_))
      have := congrArg Subtype.val (congrFun (congrFun hgh i) j)
      exact sub_left_inj.mp this
    exact Set.finite_coe_iff.mp (Finite.of_injective _ hinj)
  let H0 : Subgroup (GL n R) :=
    { carrier := M
      one_mem' := h1
      mul_mem' := fun ha hb ↦ hmul _ ha _ hb
      inv_mem' := fun hg ↦ inv_mem_of_finite_of_mul_mem hMfin h1 hmul hg }
  refine ⟨H0.comap (glMap n ρ).ker.subtype, ?_, fun g hg ↦ ?_⟩
  · haveI : Finite H0 := hMfin.to_subtype
    refine Finite.of_injective
      (fun x : H0.comap (glMap n ρ).ker.subtype ↦
        (⟨((x : (glMap n ρ).ker) : GL n R), x.2⟩ : H0)) ?_
    intro x y hxy
    exact Subtype.ext (Subtype.ext (congrArg (fun z : H0 ↦ (z : GL n R)) hxy))
  · show ∀ i j, dev ((g : (glMap n ρ).ker) : GL n R) i j ∈ B
    intro i j
    refine hFB _ ?_
    simp only [F, Finset.mem_biUnion, Finset.mem_image, Finset.mem_univ, true_and]
    exact ⟨g, hg, (i, j), rfl⟩

/-- **The unit congruence kernel is locally finite.**  If every finite subset of `ker ρ` lies in a
finite subring of `ker ρ`, then the kernel of `Rˣ → Sˣ` is a locally finite group. -/
theorem isLocallyFiniteGroup_ker_unitsMap (ρ : R →+* S) (hρ : HasLocallyFiniteKernel ρ) :
    IsLocallyFiniteGroup (Units.map (ρ : R →* S)).ker := by
  classical
  intro s
  let F : Finset R := s.image fun u : (Units.map (ρ : R →* S)).ker ↦ ((u : Rˣ) : R) - 1
  have hF : ∀ x ∈ F, ρ x = 0 := by
    intro x hx
    simp only [F, Finset.mem_image] at hx
    obtain ⟨u, -, rfl⟩ := hx
    have hu : Units.map (ρ : R →* S) (u : Rˣ) = 1 := MonoidHom.mem_ker.mp u.2
    have he : ρ ((u : Rˣ) : R) = 1 := congrArg Units.val hu
    rw [map_sub, he, map_one, sub_self]
  obtain ⟨B, hBfin, hFB, -⟩ := hρ F hF
  let M : Set Rˣ := {u | (u : R) - 1 ∈ B}
  have h1 : (1 : Rˣ) ∈ M := by
    show ((1 : Rˣ) : R) - 1 ∈ B
    rw [Units.val_one, sub_self]
    exact zero_mem B
  have hmul : ∀ a ∈ M, ∀ b ∈ M, a * b ∈ M := by
    intro a ha b hb
    show ((a * b : Rˣ) : R) - 1 ∈ B
    have hm : ((a * b : Rˣ) : R) - 1 =
        ((a : R) - 1) * ((b : R) - 1) + ((a : R) - 1) + ((b : R) - 1) := by
      rw [Units.val_mul]
      noncomm_ring
    rw [hm]
    exact add_mem (add_mem (mul_mem ha hb) ha) hb
  have hMfin : M.Finite := by
    haveI : Finite B := hBfin.to_subtype
    have hinj : Function.Injective fun u : M ↦ (⟨((u : Rˣ) : R) - 1, u.2⟩ : B) := by
      intro u v huv
      exact Subtype.ext (Units.ext (sub_left_inj.mp (congrArg Subtype.val huv)))
    exact Set.finite_coe_iff.mp (Finite.of_injective _ hinj)
  let H0 : Subgroup Rˣ :=
    { carrier := M
      one_mem' := h1
      mul_mem' := fun ha hb ↦ hmul _ ha _ hb
      inv_mem' := fun hg ↦ inv_mem_of_finite_of_mul_mem hMfin h1 hmul hg }
  refine ⟨H0.comap (Units.map (ρ : R →* S)).ker.subtype, ?_, fun u hu ↦ ?_⟩
  · haveI : Finite H0 := hMfin.to_subtype
    refine Finite.of_injective
      (fun x : H0.comap (Units.map (ρ : R →* S)).ker.subtype ↦
        (⟨((x : (Units.map (ρ : R →* S)).ker) : Rˣ), x.2⟩ : H0)) ?_
    intro x y hxy
    exact Subtype.ext (Subtype.ext (congrArg (fun z : H0 ↦ (z : Rˣ)) hxy))
  · show (((u : (Units.map (ρ : R →* S)).ker) : Rˣ) : R) - 1 ∈ B
    refine hFB _ ?_
    simp only [F, Finset.mem_image]
    exact ⟨u, hu, rfl⟩
