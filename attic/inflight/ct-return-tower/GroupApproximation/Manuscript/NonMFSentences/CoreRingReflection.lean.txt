/-!
# `thm:core-ring-reflection`: the universal directly finite quotient

`non_mf_groups_exist.tex`, Theorem `thm:core-ring-reflection`
(`\label{thm:core-ring-reflection}`, tex 1516–1523) and its proof (tex 1526–1539):

> Every unital homomorphism from $R_X$ to a directly finite ring factors
> uniquely through restriction $R_X\to R_Y$.  This is also the universal
> stably finite and LEF ring quotient.  The same assertions hold for
> $M_m(R_X)\to M_m(R_Y)$ for every $m\ge1$.  In particular, $R_X$ is
> directly finite if and only if it is stably finite, if and only if it
> is LEF, if and only if every point of $X$ is chain recurrent.

The printed assembly.  It instantiates the abstract reflection of `Algebra/DirectlyFiniteReflection`
(`isUniversalQuotients_of_surjective`, `isUniversalQuotients_mapMatrix`,
`directlyFinite_iff_stablyFinite_iff_lef_iff_injective`, lane chain-reflection) at the core restriction
`coreRestrict T k = ClopenCrossedProduct.restrict T k (image_core T) : R_X →+* R_Y`, `Y = CR(T)`:

* `killsKernelOfDirectlyFinite_restrict` (tex 1527–1528, "A unital homomorphism to a directly finite
  ring kills every $d$ ..., hence all of $I$"): the kernel is the transient ideal, generated by the
  defects (`transientIdeal_chainRecurrentSet_eq_span`), and each defect dies (`map_defectD_eq_zero`);
* `restrict_surjective_of_isClosed`: restriction to a closed invariant set is onto, by extension of
  locally constant functions (`LocallyConstant.exists_comap_val_eq_of_isClosed`);
* "Since $R_Y$ itself is LEF and therefore stably finite" (tex 1529–1530): `R_Y` LEF is the named
  `DynamicRankBudget.CoreModelsLEFStatement` (Lemma `lem:chain-core-models`), and LEF ⇒ stably finite is
  `lefStablyFiniteInterface`;
* "Finally, $I=0$ exactly when $X=Y$" (tex 1538): `restrict_injective_iff`, `transientIdeal_eq_bot_iff`.

Endpoints:
* closed: `printedCoreKillsDefectsAndI : PrintedCoreKillsDefectsAndI` (tex 1527–1528) and
  `printedCoreIZeroIff : PrintedCoreIZeroIff` (tex 1538);
* over `CoreModelsLEFStatement`: `printedCoreRingReflection_of_coreModels : PrintedCoreRingReflection`
  (tex 1516–1523, one conjunct per printed sentence) and
  `coreRingReflectionStatement_of_coreModels : DynamicRankBudget.CoreRingReflectionStatement`.

The setting is that of `DynamicRankBudget`: `X` a nonempty compact totally disconnected metric space
(`CR(T)` does not depend on the compatible metric) and `k` a finite field.
-/

universe w

/-! ### Restriction to the chain-recurrent set -/

section Restriction

variable {X : Type*} [UniformSpace X] [CompactSpace X] [T2Space X] [TotallyDisconnectedSpace X]
  (T : X ≃ₜ X) (k : Type*) [Ring k]

/-- **Restriction to a closed invariant set is onto** (tex 1470–1471, 1482–1483: "Locally constant
functions on a closed subset extend by finite clopen partitions"). -/
theorem restrict_surjective_of_isClosed {Y : Set X} (hY : T '' Y = Y) (hYc : IsClosed Y) :
    Function.Surjective (ClopenCrossedProduct.restrict T k hY) := by
  unfold ClopenCrossedProduct.restrict ClopenCrossedProduct.comap
  refine Pestov91.CrossedProduct.coeffMap_surjective _ _ fun f => ?_
  obtain ⟨g, hg⟩ := GroupApproximation.ChainCore.LocallyConstant.exists_comap_val_eq_of_isClosed hYc
    ((ClopenCoeff.of (ClopenCrossedProduct.restrictHomeo T hY) k).symm f)
  exact ⟨ClopenCoeff.of T k g, by
    rw [ClopenCrossedProduct.coeffComap_of, hg, RingEquiv.apply_symm_apply]⟩

/-- **A unital homomorphism to a directly finite ring kills every defect, hence all of `I`**
(tex 1527–1528): the kernel of restriction to `CR(T)` is the transient ideal, which is generated by
the defects `d = 1_{P∖T(P)}`, and each defect dies under a unital map into a directly finite ring. -/
theorem killsKernelOfDirectlyFinite_restrict :
    KillsKernelOfDirectlyFinite.{w}
      (ClopenCrossedProduct.restrict T k (Dynamics.image_chainRecurrentSet T)) := by
  intro A _ _ ψ x hx
  have hmem : x ∈ ClopenCrossedProduct.transientIdeal T k (Dynamics.image_chainRecurrentSet T) :=
    (ClopenCrossedProduct.mem_transientIdeal T k _).2 hx
  rw [ClopenCrossedProduct.transientIdeal_chainRecurrentSet_eq_span T k] at hmem
  have hle : TwoSidedIdeal.span (ClopenCrossedProduct.compressedDefectSet T k) ≤
      TwoSidedIdeal.ker ψ := by
    rw [TwoSidedIdeal.span_le]
    rintro _ ⟨P, hP, hTP, rfl⟩
    exact (TwoSidedIdeal.mem_ker ψ).2 (ClopenCrossedProduct.map_defectD_eq_zero T k ψ hP hTP)
  exact (TwoSidedIdeal.mem_ker ψ).1 (hle hmem)

end Restriction

/-! ### The printed sentences -/

/-- **tex 1527–1528**, "A unital homomorphism to a directly finite ring kills every $d$ in
\eqref{eq:clopen-defect-pair}, hence all of $I$ by Lemma~\ref{lem:transient-matrices}." -/
def PrintedCoreKillsDefectsAndI : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X) (k : Type) [Field k] [Finite k] (A : Type w) [Ring A] [IsDedekindFiniteMonoid A]
    (ψ : ClopenCrossedProduct T k →+* A),
    (∀ (P : Set X) (hP : IsClopen P), T '' P ⊆ P → ψ (ClopenCrossedProduct.defectD T k hP) = 0) ∧
      ∀ x ∈ ClopenCrossedProduct.transientIdeal T k (image_core T), ψ x = 0

theorem printedCoreKillsDefectsAndI : PrintedCoreKillsDefectsAndI.{w} := by
  intro X _ _ _ _ T k _ _ A _ _ ψ
  exact ⟨fun P hP hTP => ClopenCrossedProduct.map_defectD_eq_zero T k ψ hP hTP,
    fun x hx => killsKernelOfDirectlyFinite_restrict T k A ψ x
      ((ClopenCrossedProduct.mem_transientIdeal T k _).1 hx)⟩

/-- **tex 1538**, "Finally, $I=0$ exactly when $X=Y$, giving the stated equivalences." -/
def PrintedCoreIZeroIff : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X) (k : Type) [Field k] [Finite k],
    ClopenCrossedProduct.transientIdeal T k (image_core T) = ⊥ ↔
      Dynamics.chainRecurrentSet T = Set.univ

theorem printedCoreIZeroIff : PrintedCoreIZeroIff := by
  intro X _ _ _ _ T k _ _
  exact ClopenCrossedProduct.transientIdeal_eq_bot_iff T k
    (Dynamics.isClosed_chainRecurrentSet T.continuous) (image_core T)

/-- **`thm:core-ring-reflection` as printed** (tex 1516–1523), one conjunct per sentence. -/
def PrintedCoreRingReflection : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X) (k : Type) [Field k] [Finite k],
    -- tex 1518–1519: every unital hom into a directly finite ring factors uniquely through `R_X → R_Y`
    IsUniversalDirectlyFiniteQuotient.{w} (coreRestrict T k) ∧
    -- tex 1519–1520: also the universal stably finite and LEF ring quotient
    (IsUniversalStablyFiniteQuotient.{w} (coreRestrict T k) ∧
      IsUniversalLEFQuotient.{w} (coreRestrict T k)) ∧
    -- tex 1520–1521: the same for `M_m(R_X) → M_m(R_Y)`, `m ≥ 1`
    (∀ m : ℕ, 1 ≤ m →
      IsUniversalDirectlyFiniteQuotient.{w}
          ((coreRestrict T k).mapMatrix : Matrix (Fin m) (Fin m) (ClopenCrossedProduct T k) →+* _) ∧
        IsUniversalStablyFiniteQuotient.{w}
          ((coreRestrict T k).mapMatrix : Matrix (Fin m) (Fin m) (ClopenCrossedProduct T k) →+* _) ∧
        IsUniversalLEFQuotient.{w}
          ((coreRestrict T k).mapMatrix : Matrix (Fin m) (Fin m) (ClopenCrossedProduct T k) →+* _)) ∧
    -- tex 1521–1523: DF ⇔ SF ⇔ LEF ⇔ every point chain recurrent
    ((IsDedekindFiniteMonoid (ClopenCrossedProduct T k) ↔
        IsStablyFiniteRing (ClopenCrossedProduct T k)) ∧
      (IsStablyFiniteRing (ClopenCrossedProduct T k) ↔
        Pestov91.IsLEFRing (ClopenCrossedProduct T k)) ∧
      (Pestov91.IsLEFRing (ClopenCrossedProduct T k) ↔ ∀ x : X, x ∈ Dynamics.chainRecurrentSet T))

/-! ### The assembly over "`R_Y` is LEF" -/

section Assembly

variable (h : CoreModelsLEFStatement) {X : Type} [MetricSpace X] [CompactSpace X]
  [TotallyDisconnectedSpace X] [Nonempty X] (T : X ≃ₜ X) (k : Type) [Field k] [Finite k]

omit [Nonempty X] [Finite k] in
theorem coreRestrict_surjective : Function.Surjective (coreRestrict T k) :=
  restrict_surjective_of_isClosed T k (image_core T)
    (Dynamics.isClosed_chainRecurrentSet T.continuous)

omit [Nonempty X] [Finite k] in
theorem killsKernelOfDirectlyFinite_coreRestrict :
    KillsKernelOfDirectlyFinite.{w} (coreRestrict T k) :=
  killsKernelOfDirectlyFinite_restrict T k

include h in
/-- **The three universal properties** (tex 1518–1520, 1527–1530). -/
theorem isUniversalQuotients_coreRestrict :
    IsUniversalDirectlyFiniteQuotient.{w} (coreRestrict T k) ∧
      IsUniversalStablyFiniteQuotient.{w} (coreRestrict T k) ∧
      IsUniversalLEFQuotient.{w} (coreRestrict T k) :=
  isUniversalQuotients_of_surjective lefStablyFiniteInterface lefStablyFiniteInterface
    (coreRestrict T k) (coreRestrict_surjective T k) (h X T k)
    (killsKernelOfDirectlyFinite_coreRestrict T k)

include h in
/-- **Amplification** (tex 1520–1521, 1531–1537). -/
theorem isUniversalQuotients_coreRestrict_mapMatrix (m : ℕ) :
    IsUniversalDirectlyFiniteQuotient.{w}
        ((coreRestrict T k).mapMatrix : Matrix (Fin m) (Fin m) (ClopenCrossedProduct T k) →+* _) ∧
      IsUniversalStablyFiniteQuotient.{w}
        ((coreRestrict T k).mapMatrix : Matrix (Fin m) (Fin m) (ClopenCrossedProduct T k) →+* _) ∧
      IsUniversalLEFQuotient.{w}
        ((coreRestrict T k).mapMatrix : Matrix (Fin m) (Fin m) (ClopenCrossedProduct T k) →+* _) :=
  isUniversalQuotients_mapMatrix lefStablyFiniteInterface lefStablyFiniteInterface
    (coreRestrict T k) (coreRestrict_surjective T k) (h X T k)
    (killsKernelOfDirectlyFinite_coreRestrict T k) m

include h in
/-- **The equivalences** (tex 1521–1523, 1538): directly finite ⇔ stably finite ⇔ LEF ⇔ restriction
injective, and restriction is injective exactly when every point is chain recurrent. -/
theorem directlyFinite_iff_of_coreModels :
    (IsDedekindFiniteMonoid (ClopenCrossedProduct T k) ↔
        IsStablyFiniteRing (ClopenCrossedProduct T k)) ∧
      (IsStablyFiniteRing (ClopenCrossedProduct T k) ↔
        Pestov91.IsLEFRing (ClopenCrossedProduct T k)) ∧
      (Pestov91.IsLEFRing (ClopenCrossedProduct T k) ↔ ∀ x : X, x ∈ Dynamics.chainRecurrentSet T) := by
  obtain ⟨h1, h2, h3⟩ := directlyFinite_iff_stablyFinite_iff_lef_iff_injective
    lefStablyFiniteInterface (coreRestrict T k) (h X T k) (killsKernelOfDirectlyFinite_coreRestrict T k)
  exact ⟨h1, h2, h3.trans ((ClopenCrossedProduct.restrict_injective_iff T k
    (Dynamics.isClosed_chainRecurrentSet T.continuous) (image_core T)).trans Set.eq_univ_iff_forall)⟩

end Assembly

/-- **`thm:core-ring-reflection`** (tex 1516–1523) over "`R_Y` is LEF" (`lem:chain-core-models`). -/
theorem printedCoreRingReflection_of_coreModels (h : CoreModelsLEFStatement) :
    PrintedCoreRingReflection.{w} := by
  intro X _ _ _ _ T k _ _
  obtain ⟨hdf, hsf, hlef⟩ := isUniversalQuotients_coreRestrict.{w} h T k
  exact ⟨hdf, ⟨hsf, hlef⟩, fun m _ => isUniversalQuotients_coreRestrict_mapMatrix h T k m,
    directlyFinite_iff_of_coreModels h T k⟩

/-- **ct-rank-budget's `CoreRingReflectionStatement`** (tex 1522–1523) over "`R_Y` is LEF". -/
theorem coreRingReflectionStatement_of_coreModels (h : CoreModelsLEFStatement) :
    CoreRingReflectionStatement := by
  intro X _ _ _ _ T k _ _
  exact directlyFinite_iff_of_coreModels h T k

end CoreRingReflection
