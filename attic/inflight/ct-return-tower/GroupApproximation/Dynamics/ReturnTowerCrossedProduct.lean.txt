/-!
# The return tower in `R_X`

`non_mf_groups_exist.tex`, proof of Corollary `cor:dynamic-rank-budget`
(`\label{cor:dynamic-rank-budget}`), tex 1727–1734:

> For $0\le i<2r-1$ put
> $D_i=\bigcup_{h>i}C_h$, $a_i=u^i1_{D_i}$, $b_i=1_{D_i}u^{-i}$.
> The tower partition gives $b_i a_j=0$ for $i\ne j$,
> $b_i a_i=1_{D_i}$ and $\sum_i a_i b_i=1$.
> For $i<r$ we have $D_i=C$.

For a return section `C` of spread `r ≥ 1`:

* `towerLevel T C i = {x ∈ C | i < returnTime T C x}` (printed `D_i = ⋃_{h>i} C_h`), clopen, and
  equal to `C` for `i < r`;
* the tower partition: every point is `T^i y` for exactly one `i < 2r - 1` and `y ∈ D_i`
  (`exists_mem_image_towerLevel`, `eq_of_mem_image_towerLevel`);
* `towerA`, `towerB` (printed `a_i`, `b_i`), with `towerB_mul_towerA_of_ne`,
  `towerB_mul_towerA_self`, `sum_towerA_mul_towerB`;
* `returnTowerUnits hC k hr : ReturnTowerUnits R_X (2r - 1) r` over `p_C = 1_C`, so
  `Algebra/ReturnTowerUnits` applies: `R_X` is a unital corner of `M_{2r-1}(P_C)`, `Φ` is injective,
  unitization embeds `GL_{nr}(P_C)` into `GL_n(R_X)`, and `EL_{nr}(P_C)` lands in `EL_n(R_X)`;
* the closed endpoint `returnTowerUnitsStatement_holds : ReturnTowerUnitsStatement`.
-/

namespace GroupApproximation.ChainCore

open Set MFQuotientUnits

variable {X : Type*} [TopologicalSpace X] {T : X ≃ₜ X} {r : ℕ} {C : Set X}

/-! ### The tower levels -/

/-- The tower level `D_i = ⋃_{h>i} C_h = {x ∈ C | i < returnTime x}`. -/
def towerLevel (T : X ≃ₜ X) (C : Set X) (i : ℕ) : Set X :=
  {x | x ∈ C ∧ i < returnTime T C x}

theorem towerLevel_subset {i : ℕ} : towerLevel T C i ⊆ C :=
  fun _ h => h.1

theorem towerLevel_eq (hC : IsReturnSection T r C) (i : ℕ) :
    towerLevel T C i = C ∩ ⋂ j ∈ Finset.Ioc 0 i, (T ^ (j : ℤ)) ⁻¹' Cᶜ := by
  ext x
  simp only [towerLevel, mem_setOf_eq, mem_inter_iff, mem_iInter₂, Finset.mem_Ioc, mem_preimage,
    mem_compl_iff]
  refine and_congr_right fun _ => ⟨fun hi j hj => returnTime_min hC x hj.1 (by omega), fun h => ?_⟩
  exact not_le.1 fun hle => h _ ⟨(returnTime_spec hC x).1, hle⟩ (returnTime_spec hC x).2

theorem isClopen_towerLevel (hC : IsReturnSection T r C) (i : ℕ) :
    IsClopen (towerLevel T C i) := by
  rw [towerLevel_eq hC i]
  exact hC.isClopen.inter
    (isClopen_biInter_finset fun j _ => hC.isClopen.compl.preimage (T ^ (j : ℤ)).continuous)

/-- **For `i < r` we have `D_i = C`** (tex 1734). -/
theorem towerLevel_of_lt (hC : IsReturnSection T r C) {i : ℕ} (hi : i < r) :
    towerLevel T C i = C :=
  Subset.antisymm towerLevel_subset fun x hx => ⟨hx, lt_of_lt_of_le hi (le_returnTime hC hx)⟩

/-- Every point visits `C` backwards within `2r - 2` steps. -/
theorem IsReturnSection.exists_backward (hC : IsReturnSection T r C) (x : X) :
    ∃ m : ℕ, m < 2 * r - 1 ∧ (T ^ (-(m : ℤ))) x ∈ C := by
  obtain ⟨j, hj, c, hc, hcx⟩ := hC.exists_mem_image ((T ^ (-((r : ℤ) - 1))) x)
  obtain ⟨h₁, h₂⟩ := abs_lt.1 hj
  have hc' : c = (T ^ (-((r : ℤ) - 1 + j))) x := by
    rw [← zpow_neg_apply_zpow_apply_homeomorph T j c, hcx, zpow_apply_zpow_apply_homeomorph,
      show -j + -((r : ℤ) - 1) = -((r : ℤ) - 1 + j) by ring]
  refine ⟨((r : ℤ) - 1 + j).toNat, by omega, ?_⟩
  rw [Int.toNat_of_nonneg (by omega), ← hc']
  exact hc

/-- **The tower covers `X`**: every point is `T^i y` with `y ∈ D_i`, `i < 2r - 1`. -/
theorem exists_mem_image_towerLevel (hC : IsReturnSection T r C) (x : X) :
    ∃ i : ℕ, i < 2 * r - 1 ∧ x ∈ (T ^ (i : ℤ)) '' towerLevel T C i := by
  classical
  have hex : ∃ m : ℕ, (T ^ (-(m : ℤ))) x ∈ C :=
    let ⟨m, _, hm⟩ := hC.exists_backward x
    ⟨m, hm⟩
  obtain ⟨m, hm, hmC⟩ := hC.exists_backward x
  refine ⟨Nat.find hex, lt_of_le_of_lt (Nat.find_le hmC) hm,
    (T ^ (-(Nat.find hex : ℤ))) x, ⟨Nat.find_spec hex, ?_⟩, ?_⟩
  · by_contra hle
    push Not at hle
    obtain ⟨h₁, h₃⟩ := returnTime_spec hC ((T ^ (-(Nat.find hex : ℤ))) x)
    refine Nat.find_min hex
      (m := Nat.find hex - returnTime T C ((T ^ (-(Nat.find hex : ℤ))) x)) (by omega) ?_
    rw [zpow_apply_zpow_apply_homeomorph] at h₃
    rwa [show -(((Nat.find hex - returnTime T C ((T ^ (-(Nat.find hex : ℤ))) x) : ℕ) : ℤ)) =
        (returnTime T C ((T ^ (-(Nat.find hex : ℤ))) x) : ℤ) + -(Nat.find hex : ℤ) by
      rw [Nat.cast_sub hle]
      ring]
  · rw [zpow_apply_zpow_apply_homeomorph, add_neg_cancel, zpow_zero, Homeomorph.one_apply]

/-- **The tower levels are disjoint**: `T^i D_i ∩ T^{i'} D_{i'} ≠ ∅` forces `i = i'`. -/
theorem eq_of_mem_image_towerLevel (hC : IsReturnSection T r C) {x : X} {i i' : ℕ}
    (hi : x ∈ (T ^ (i : ℤ)) '' towerLevel T C i)
    (hi' : x ∈ (T ^ (i' : ℤ)) '' towerLevel T C i') : i = i' := by
  obtain ⟨y, ⟨hyC, hyi⟩, rfl⟩ := hi
  obtain ⟨y', ⟨hy'C, hy'i⟩, hy'x⟩ := hi'
  rcases lt_trichotomy i i' with hlt | heq | hgt
  · exfalso
    refine returnTime_min hC y' (j := i' - i) (by omega) (by omega) ?_
    have hyy : y = (T ^ (((i' - i : ℕ)) : ℤ)) y' := by
      have h2 := congrArg (T ^ (-(i : ℤ))) hy'x
      rw [zpow_neg_apply_zpow_apply_homeomorph, zpow_apply_zpow_apply_homeomorph] at h2
      rw [← h2, Nat.cast_sub hlt.le, show -(i : ℤ) + i' = (i' : ℤ) - i by ring]
    rw [← hyy]
    exact hyC
  · exact heq
  · exfalso
    refine returnTime_min hC y (j := i - i') (by omega) (by omega) ?_
    have hyy : y' = (T ^ (((i - i' : ℕ)) : ℤ)) y := by
      have h2 := congrArg (T ^ (-(i' : ℤ))) hy'x
      rw [zpow_neg_apply_zpow_apply_homeomorph, zpow_apply_zpow_apply_homeomorph] at h2
      rw [h2, Nat.cast_sub hgt.le, show -(i' : ℤ) + i = (i : ℤ) - i' by ring]
    rw [← hyy]
    exact hy'C

theorem disjoint_towerLevel_image (hC : IsReturnSection T r C) {i j : ℕ} (hij : i ≠ j) :
    Disjoint (towerLevel T C i) ((T ^ (-(i : ℤ) + j)) '' towerLevel T C j) := by
  refine disjoint_left.2 fun x hx hximg => hij ?_
  obtain ⟨y, hy, rfl⟩ := hximg
  refine eq_of_mem_image_towerLevel hC ⟨_, hx, rfl⟩ ⟨y, hy, ?_⟩
  rw [zpow_apply_zpow_apply_homeomorph, show (i : ℤ) + (-(i : ℤ) + j) = j by ring]

theorem sum_indicator_image_towerLevel (hC : IsReturnSection T r C) (k : Type*) [Ring k]
    (x : X) :
    ∑ i : Fin (2 * r - 1),
        Set.indicator ((T ^ ((i : ℕ) : ℤ)) '' towerLevel T C i) (1 : X → k) x = 1 := by
  obtain ⟨i₀, hi₀, hx⟩ := exists_mem_image_towerLevel hC x
  rw [Finset.sum_eq_single ⟨i₀, hi₀⟩]
  · simp only [Set.indicator_of_mem hx, Pi.one_apply]
  · intro i _ hne
    exact Set.indicator_of_notMem
      (fun hmem => hne (Fin.ext (eq_of_mem_image_towerLevel hC hmem hx))) _
  · intro h
    exact absurd (Finset.mem_univ _) h

/-! ### Covariance in `R_X` -/

section Crossed

variable (k : Type*) [Ring k]

/-- `u^j` in `R_X`. -/
noncomputable abbrev unitPow (T : X ≃ₜ X) (k : Type*) [Ring k] (j : ℤ) :
    ClopenCrossedProduct T k :=
  ((ClopenCrossedProduct.unit T k ^ j : (ClopenCrossedProduct T k)ˣ) : ClopenCrossedProduct T k)

theorem unitPow_mul (a b : ℤ) : unitPow T k a * unitPow T k b = unitPow T k (a + b) := by
  rw [unitPow, unitPow, unitPow, ← Units.val_mul, ← zpow_add]

theorem unitPow_zero : unitPow T k 0 = 1 := by
  rw [unitPow, zpow_zero, Units.val_one]

/-- `u^j f = (f ∘ T^{-j}) u^j`. -/
theorem unitPow_mul_coeff (j : ℤ) (f : LocallyConstant X k) :
    unitPow T k j * ClopenCrossedProduct.coeff T k f =
      ClopenCrossedProduct.coeff T k
          (LocallyConstant.comap ⟨⇑(T ^ (-j)), (T ^ (-j)).continuous⟩ f) * unitPow T k j := by
  have h := Pestov91.CrossedProduct.unit_zpow_mul_C (A := ClopenCoeff T k) j (ClopenCoeff.of T k f)
  rw [ClopenCoeff.smul_of] at h
  exact h

/-- `1_D ∘ T^{-j} = 1_{T^j(D)}`. -/
theorem comap_zpow_neg_charFn (j : ℤ) {D : Set X} (hD : IsClopen D) :
    LocallyConstant.comap ⟨⇑(T ^ (-j)), (T ^ (-j)).continuous⟩ (LocallyConstant.charFn k hD) =
      LocallyConstant.charFn k (ClopenCrossedProduct.isClopen_image (T ^ j) hD) := by
  have hmem : ∀ x, x ∈ (T ^ j) '' D ↔ (T ^ (-j)) x ∈ D := fun x => by
    constructor
    · rintro ⟨y, hy, rfl⟩
      rwa [zpow_neg_apply_zpow_apply_homeomorph]
    · intro h
      exact ⟨(T ^ (-j)) x, h, by
        rw [zpow_apply_zpow_apply_homeomorph, add_neg_cancel, zpow_zero, Homeomorph.one_apply]⟩
  ext x
  rw [LocallyConstant.coe_comap_apply]
  simp only [LocallyConstant.coe_charFn, ContinuousMap.coe_mk]
  by_cases hx : (T ^ (-j)) x ∈ D
  · simp only [Set.indicator_of_mem hx, Set.indicator_of_mem ((hmem x).2 hx), Pi.one_apply]
  · simp only [Set.indicator_of_notMem hx, Set.indicator_of_notMem (mt (hmem x).1 hx)]

theorem charFn_mul_charFn_of_disjoint {U V : Set X} (hU : IsClopen U) (hV : IsClopen V)
    (h : Disjoint U V) :
    LocallyConstant.charFn k hU * LocallyConstant.charFn k hV = 0 := by
  ext x
  simp only [LocallyConstant.mul_apply, LocallyConstant.coe_charFn, LocallyConstant.zero_apply]
  by_cases hx : x ∈ U
  · simp only [Set.indicator_of_notMem (disjoint_left.1 h hx), mul_zero]
  · simp only [Set.indicator_of_notMem hx, zero_mul]

/-! ### The levels `a_i`, `b_i` -/

variable (hC : IsReturnSection T r C)

/-- `1_{D_i}` in `R_X`. -/
noncomputable def levelCoeff (i : ℕ) : ClopenCrossedProduct T k :=
  ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k (isClopen_towerLevel hC i))

/-- **`a_i = u^i 1_{D_i}`** (tex 1729). -/
noncomputable def towerA (i : ℕ) : ClopenCrossedProduct T k :=
  unitPow T k (i : ℤ) * levelCoeff k hC i

/-- **`b_i = 1_{D_i} u^{-i}`** (tex 1729). -/
noncomputable def towerB (i : ℕ) : ClopenCrossedProduct T k :=
  levelCoeff k hC i * unitPow T k (-(i : ℤ))

theorem levelCoeff_mul_levelCoeff_mul (i : ℕ) (y : ClopenCrossedProduct T k) :
    levelCoeff k hC i * (levelCoeff k hC i * y) = levelCoeff k hC i * y := by
  rw [← mul_assoc, levelCoeff, ← map_mul,
    ClopenCrossedProduct.charFn_mul_charFn_of_subset k _ _ subset_rfl]

theorem levelCoeff_mul_levelCoeff (i : ℕ) :
    levelCoeff k hC i * levelCoeff k hC i = levelCoeff k hC i := by
  simpa only [mul_one] using levelCoeff_mul_levelCoeff_mul k hC i 1

/-- `b_i a_j = 1_{D_i ∩ T^{j-i} D_j} u^{j-i}`. -/
theorem towerB_mul_towerA (i j : ℕ) :
    towerB k hC i * towerA k hC j =
      ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k (isClopen_towerLevel hC i) *
          LocallyConstant.charFn k
            (ClopenCrossedProduct.isClopen_image (T ^ (-(i : ℤ) + j)) (isClopen_towerLevel hC j))) *
        unitPow T k (-(i : ℤ) + j) := by
  rw [towerB, towerA, levelCoeff, levelCoeff, mul_assoc, ← mul_assoc (unitPow T k _), unitPow_mul,
    unitPow_mul_coeff, comap_zpow_neg_charFn, ← mul_assoc, ← map_mul]

/-- **`b_i a_j = 0` for `i ≠ j`** (tex 1730). -/
theorem towerB_mul_towerA_of_ne {i j : ℕ} (hij : i ≠ j) : towerB k hC i * towerA k hC j = 0 := by
  rw [towerB_mul_towerA, charFn_mul_charFn_of_disjoint k _ _ (disjoint_towerLevel_image hC hij),
    map_zero, zero_mul]

/-- **`b_i a_i = 1_{D_i}`** (tex 1730). -/
theorem towerB_mul_towerA_self (i : ℕ) : towerB k hC i * towerA k hC i = levelCoeff k hC i := by
  rw [towerB, towerA, mul_assoc, ← mul_assoc (unitPow T k _), unitPow_mul, neg_add_cancel,
    unitPow_zero, one_mul, levelCoeff_mul_levelCoeff]

/-- `a_i b_i = 1_{T^i(D_i)}`. -/
theorem towerA_mul_towerB (i : ℕ) :
    towerA k hC i * towerB k hC i =
      ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k
        (ClopenCrossedProduct.isClopen_image (T ^ (i : ℤ)) (isClopen_towerLevel hC i))) := by
  have h : towerA k hC i * towerB k hC i =
      unitPow T k (i : ℤ) * levelCoeff k hC i * unitPow T k (-(i : ℤ)) := by
    simp only [towerA, towerB, mul_assoc, levelCoeff_mul_levelCoeff_mul]
  rw [h, levelCoeff, unitPow_mul_coeff, comap_zpow_neg_charFn, mul_assoc, unitPow_mul,
    add_neg_cancel, unitPow_zero, mul_one]

/-- **`∑_i a_i b_i = 1`** (tex 1731): the tower partition. -/
theorem sum_towerA_mul_towerB :
    ∑ i : Fin (2 * r - 1), towerA k hC i * towerB k hC i = 1 := by
  simp only [towerA_mul_towerB]
  rw [← map_sum (ClopenCrossedProduct.coeff T k), ← map_one (ClopenCrossedProduct.coeff T k)]
  congr 1
  ext x
  have hev := map_sum (LocallyConstant.evalRingHom (Y := k) x)
    (fun i : Fin (2 * r - 1) => LocallyConstant.charFn k
      (ClopenCrossedProduct.isClopen_image (T ^ ((i : ℕ) : ℤ)) (isClopen_towerLevel hC i)))
    Finset.univ
  simp only [LocallyConstant.evalRingHom_apply] at hev
  rw [hev, LocallyConstant.one_apply]
  simp only [LocallyConstant.coe_charFn]
  exact sum_indicator_image_towerLevel hC k x

theorem charFn_towerLevel_of_lt {i : ℕ} (hi : i < r) :
    LocallyConstant.charFn k (isClopen_towerLevel hC i) = LocallyConstant.charFn k hC.isClopen := by
  ext x
  simp only [LocallyConstant.coe_charFn, towerLevel_of_lt hC hi]

/-- **The return tower of `R_X`** over `p_C = 1_C`, height `2r - 1`, full base rank `r`
(tex 1727–1734). -/
noncomputable def returnTowerUnits (hr : 1 ≤ r) :
    ReturnTowerUnits (ClopenCrossedProduct T k) (2 * r - 1) r where
  p := ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k hC.isClopen)
  a i := towerA k hC i
  b i := towerB k hC i
  le := by omega
  isIdempotentElem_p := by
    show ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k hC.isClopen) *
        ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k hC.isClopen) =
      ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k hC.isClopen)
    rw [← map_mul, ClopenCrossedProduct.charFn_mul_charFn_of_subset k _ _ subset_rfl]
  a_mul_p i := by
    rw [towerA, levelCoeff, mul_assoc, ← map_mul,
      ClopenCrossedProduct.charFn_mul_charFn_of_subset k _ _ towerLevel_subset]
  p_mul_b i := by
    rw [towerB, levelCoeff, ← mul_assoc, ← map_mul,
      ClopenCrossedProduct.charFn_mul_charFn_of_superset k _ _ towerLevel_subset]
  b_mul_a_of_ne i j hij := towerB_mul_towerA_of_ne k hC fun h => hij (Fin.ext h)
  sum_a_mul_b := sum_towerA_mul_towerB k hC
  b_mul_a_of_lt i hi := by
    rw [towerB_mul_towerA_self, levelCoeff, charFn_towerLevel_of_lt k hC hi]

theorem returnTowerUnits_p (hr : 1 ≤ r) :
    (returnTowerUnits k hC hr).p =
      ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k hC.isClopen) :=
  rfl

theorem returnTowerUnits_a (hr : 1 ≤ r) (i : Fin (2 * r - 1)) :
    (returnTowerUnits k hC hr).a i = towerA k hC i :=
  rfl

theorem returnTowerUnits_b (hr : 1 ≤ r) (i : Fin (2 * r - 1)) :
    (returnTowerUnits k hC hr).b i = towerB k hC i :=
  rfl

end Crossed

/-- **The return tower** (tex 1727–1734), closed. -/
theorem returnTowerUnitsStatement_holds : ReturnTowerUnitsStatement := by
  intro X _ T k _ r C hC hr
  exact ⟨returnTowerUnits k hC hr, rfl, fun i => ⟨isClopen_towerLevel hC i, rfl, rfl⟩⟩
