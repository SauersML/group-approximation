/-!
# The unit implementing the first return

`non_mf_groups_exist.tex`, proof of Corollary `cor:dynamic-rank-budget`
(`\label{cor:dynamic-rank-budget}`), tex 1722–1724:

> Put $p_C=1_C$ and $P_C=p_CR_Xp_C$.
> The unit $v=\sum_h u^h1_{C_h}$ implements $S_C$.

For a return section `C` of `T` (`IsReturnSection T r C`), in `R_X = ClopenCrossedProduct T k`:

* `returnP k hC = 1_C`, `cellCoeff k hC h = 1_{C_h}`, `backCellCoeff k hC h = 1_{C'_h}`
  (`C'_h = returnCell T⁻¹ C h`); orthogonality and partition of both families;
* `unitPow_mul_cellCoeff`: `u^h 1_{C_h} = 1_{C'_h} u^h`;
* `returnV k hC = ∑_{h<2r} u^h 1_{C_h}` (printed `v`), `returnW k hC = ∑_{h<2r} 1_{C_h} u^{-h}`,
  with `returnV_mul_returnW`, `returnW_mul_returnV` (both `= p_C`) and the corner identities;
* `returnUnit k hC : (P_C)ˣ`, the unit `v` of the corner with inverse `w`;
* `returnV_mul_coeff_extend`: `v · f = (f ∘ S_C⁻¹) · v` for `f ∈ LC(C, k)` extended by zero, where
  `S_C = firstReturnHomeomorph hC`: **`v` implements `S_C`**.
-/

namespace GroupApproximation.ChainCore

open Set MFQuotientUnits

variable {X : Type*} [TopologicalSpace X] {T : X ≃ₜ X} {r : ℕ} {C : Set X}

theorem mem_zpow_image_iff {S : Set X} {x : X} (j : ℤ) : x ∈ (T ^ j) '' S ↔ (T ^ (-j)) x ∈ S := by
  constructor
  · rintro ⟨y, hy, rfl⟩
    rwa [zpow_neg_apply_zpow_apply_homeomorph]
  · intro h
    exact ⟨(T ^ (-j)) x, h, by
      rw [zpow_apply_zpow_apply_homeomorph, add_neg_cancel, zpow_zero, Homeomorph.one_apply]⟩

section Unit

variable (k : Type*) [Ring k] (hC : IsReturnSection T r C)

theorem charFn_congr {U V : Set X} (hU : IsClopen U) (hV : IsClopen V) (h : U = V) :
    LocallyConstant.charFn k hU = LocallyConstant.charFn k hV := by
  subst h
  rfl

/-- `p_C = 1_C`. -/
noncomputable def returnP : ClopenCrossedProduct T k :=
  ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k hC.isClopen)

theorem isIdempotentElem_returnP : IsIdempotentElem (returnP k hC) := by
  show ClopenCrossedProduct.coeff T k _ * ClopenCrossedProduct.coeff T k _ =
    ClopenCrossedProduct.coeff T k _
  rw [← map_mul, ClopenCrossedProduct.charFn_mul_charFn_of_subset k _ _ subset_rfl]

/-- `1_{C_h}`. -/
noncomputable def cellCoeff (h : ℕ) : ClopenCrossedProduct T k :=
  ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k (isClopen_returnCell hC h))

/-- `1_{C'_h}`, the backward return cell. -/
noncomputable def backCellCoeff (h : ℕ) : ClopenCrossedProduct T k :=
  ClopenCrossedProduct.coeff T k (LocallyConstant.charFn k (isClopen_returnCell hC.inv h))

theorem cellCoeff_mul_self (h : ℕ) : cellCoeff k hC h * cellCoeff k hC h = cellCoeff k hC h := by
  rw [cellCoeff, ← map_mul, ClopenCrossedProduct.charFn_mul_charFn_of_subset k _ _ subset_rfl]

theorem cellCoeff_mul_of_ne {h h' : ℕ} (hne : h ≠ h') :
    cellCoeff k hC h * cellCoeff k hC h' = 0 := by
  rw [cellCoeff, cellCoeff, ← map_mul, charFn_mul_charFn_of_disjoint k _ _ (disjoint_returnCell hne),
    map_zero]

theorem backCellCoeff_mul_self (h : ℕ) :
    backCellCoeff k hC h * backCellCoeff k hC h = backCellCoeff k hC h := by
  rw [backCellCoeff, ← map_mul, ClopenCrossedProduct.charFn_mul_charFn_of_subset k _ _ subset_rfl]

theorem backCellCoeff_mul_of_ne {h h' : ℕ} (hne : h ≠ h') :
    backCellCoeff k hC h * backCellCoeff k hC h' = 0 := by
  rw [backCellCoeff, backCellCoeff, ← map_mul,
    charFn_mul_charFn_of_disjoint k _ _ (disjoint_returnCell hne), map_zero]

theorem returnP_mul_cellCoeff (h : ℕ) : returnP k hC * cellCoeff k hC h = cellCoeff k hC h := by
  rw [returnP, cellCoeff, ← map_mul,
    ClopenCrossedProduct.charFn_mul_charFn_of_superset k _ _ returnCell_subset]

theorem cellCoeff_mul_returnP (h : ℕ) : cellCoeff k hC h * returnP k hC = cellCoeff k hC h := by
  rw [returnP, cellCoeff, ← map_mul,
    ClopenCrossedProduct.charFn_mul_charFn_of_subset k _ _ returnCell_subset]

theorem returnP_mul_backCellCoeff (h : ℕ) :
    returnP k hC * backCellCoeff k hC h = backCellCoeff k hC h := by
  rw [returnP, backCellCoeff, ← map_mul,
    ClopenCrossedProduct.charFn_mul_charFn_of_superset k _ _ returnCell_subset]

theorem backCellCoeff_mul_returnP (h : ℕ) :
    backCellCoeff k hC h * returnP k hC = backCellCoeff k hC h := by
  rw [returnP, backCellCoeff, ← map_mul,
    ClopenCrossedProduct.charFn_mul_charFn_of_subset k _ _ returnCell_subset]

theorem sum_cellCoeff : ∑ h ∈ Finset.range (2 * r), cellCoeff k hC h = returnP k hC := by
  simp only [cellCoeff, returnP]
  rw [← map_sum]
  congr 1
  ext x
  have hev := map_sum (LocallyConstant.evalRingHom (Y := k) x)
    (fun h => LocallyConstant.charFn k (isClopen_returnCell hC h)) (Finset.range (2 * r))
  simp only [LocallyConstant.evalRingHom_apply] at hev
  rw [hev]
  simp only [LocallyConstant.coe_charFn]
  exact sum_indicator_returnCell hC k x

theorem sum_backCellCoeff : ∑ h ∈ Finset.range (2 * r), backCellCoeff k hC h = returnP k hC := by
  simp only [backCellCoeff, returnP]
  rw [← map_sum]
  congr 1
  ext x
  have hev := map_sum (LocallyConstant.evalRingHom (Y := k) x)
    (fun h => LocallyConstant.charFn k (isClopen_returnCell hC.inv h)) (Finset.range (2 * r))
  simp only [LocallyConstant.evalRingHom_apply] at hev
  rw [hev]
  simp only [LocallyConstant.coe_charFn]
  exact sum_indicator_returnCell hC.inv k x

/-- **`u^h 1_{C_h} = 1_{C'_h} u^h`**, since `T^h(C_h) = C'_h`. -/
theorem unitPow_mul_cellCoeff (h : ℕ) :
    unitPow T k (h : ℤ) * cellCoeff k hC h = backCellCoeff k hC h * unitPow T k (h : ℤ) := by
  rw [cellCoeff, unitPow_mul_coeff, comap_zpow_neg_charFn]
  exact congrArg (fun F => ClopenCrossedProduct.coeff T k F * unitPow T k (h : ℤ))
    (charFn_congr k _ _ (image_returnCell hC h))

theorem cellCoeff_mul_unitPow_neg (h : ℕ) :
    cellCoeff k hC h * unitPow T k (-(h : ℤ)) = unitPow T k (-(h : ℤ)) * backCellCoeff k hC h := by
  calc cellCoeff k hC h * unitPow T k (-(h : ℤ))
      = unitPow T k (-(h : ℤ)) * (unitPow T k (h : ℤ) * cellCoeff k hC h) *
          unitPow T k (-(h : ℤ)) := by
        rw [← mul_assoc (unitPow T k (-(h : ℤ))), unitPow_mul, neg_add_cancel, unitPow_zero,
          one_mul]
    _ = unitPow T k (-(h : ℤ)) * (backCellCoeff k hC h * unitPow T k (h : ℤ)) *
          unitPow T k (-(h : ℤ)) := by
        rw [unitPow_mul_cellCoeff]
    _ = unitPow T k (-(h : ℤ)) * backCellCoeff k hC h := by
        rw [mul_assoc, mul_assoc, unitPow_mul, add_neg_cancel, unitPow_zero, mul_one]

/-! ### `v` and `w` -/

/-- **`v = ∑_h u^h 1_{C_h}`** (tex 1723). -/
noncomputable def returnV : ClopenCrossedProduct T k :=
  ∑ h ∈ Finset.range (2 * r), unitPow T k (h : ℤ) * cellCoeff k hC h

/-- `w = ∑_h 1_{C_h} u^{-h}`, the inverse of `v` in `P_C`. -/
noncomputable def returnW : ClopenCrossedProduct T k :=
  ∑ h ∈ Finset.range (2 * r), cellCoeff k hC h * unitPow T k (-(h : ℤ))

theorem returnV_eq_sum_back :
    returnV k hC = ∑ h ∈ Finset.range (2 * r), backCellCoeff k hC h * unitPow T k (h : ℤ) :=
  Finset.sum_congr rfl fun h _ => unitPow_mul_cellCoeff k hC h

theorem returnW_eq_sum_back :
    returnW k hC = ∑ h ∈ Finset.range (2 * r), unitPow T k (-(h : ℤ)) * backCellCoeff k hC h :=
  Finset.sum_congr rfl fun h _ => cellCoeff_mul_unitPow_neg k hC h

theorem returnV_mul_returnW : returnV k hC * returnW k hC = returnP k hC := by
  rw [returnV, returnW, Finset.sum_mul, ← sum_backCellCoeff k hC]
  refine Finset.sum_congr rfl fun h hh => ?_
  rw [Finset.mul_sum, Finset.sum_eq_single h]
  · rw [mul_assoc, ← mul_assoc (cellCoeff k hC h), cellCoeff_mul_self, ← mul_assoc,
      unitPow_mul_cellCoeff, mul_assoc, unitPow_mul, add_neg_cancel, unitPow_zero, mul_one]
  · intro h' _ hne
    rw [mul_assoc, ← mul_assoc (cellCoeff k hC h), cellCoeff_mul_of_ne k hC (Ne.symm hne),
      zero_mul, mul_zero]
  · intro hn
    exact absurd hh hn

theorem returnW_mul_returnV : returnW k hC * returnV k hC = returnP k hC := by
  rw [returnW_eq_sum_back, returnV_eq_sum_back, Finset.sum_mul, ← sum_cellCoeff k hC]
  refine Finset.sum_congr rfl fun h hh => ?_
  rw [Finset.mul_sum, Finset.sum_eq_single h]
  · rw [mul_assoc, ← mul_assoc (backCellCoeff k hC h), backCellCoeff_mul_self, ← mul_assoc,
      ← cellCoeff_mul_unitPow_neg, mul_assoc, unitPow_mul, neg_add_cancel, unitPow_zero, mul_one]
  · intro h' _ hne
    rw [mul_assoc, ← mul_assoc (backCellCoeff k hC h), backCellCoeff_mul_of_ne k hC (Ne.symm hne),
      zero_mul, mul_zero]
  · intro hn
    exact absurd hh hn

theorem returnP_mul_returnV : returnP k hC * returnV k hC = returnV k hC := by
  rw [returnV_eq_sum_back, Finset.mul_sum]
  exact Finset.sum_congr rfl fun h _ => by rw [← mul_assoc, returnP_mul_backCellCoeff]

theorem returnV_mul_returnP : returnV k hC * returnP k hC = returnV k hC := by
  rw [returnV, Finset.sum_mul]
  exact Finset.sum_congr rfl fun h _ => by rw [mul_assoc, cellCoeff_mul_returnP]

theorem returnP_mul_returnW : returnP k hC * returnW k hC = returnW k hC := by
  rw [returnW, Finset.mul_sum]
  exact Finset.sum_congr rfl fun h _ => by rw [← mul_assoc, returnP_mul_cellCoeff]

theorem returnW_mul_returnP : returnW k hC * returnP k hC = returnW k hC := by
  rw [returnW_eq_sum_back, Finset.sum_mul]
  exact Finset.sum_congr rfl fun h _ => by rw [mul_assoc, backCellCoeff_mul_returnP]

/-- The corner `P_C = p_C R_X p_C`. -/
abbrev ReturnCorner : Type _ :=
  Corner (ClopenCrossedProduct T k) (returnP k hC) (isIdempotentElem_returnP k hC)

/-- **The unit `v` of `P_C`**, with inverse `w`. -/
noncomputable def returnUnit : (ReturnCorner k hC)ˣ where
  val := ⟨returnV k hC, returnP_mul_returnV k hC, returnV_mul_returnP k hC⟩
  inv := ⟨returnW k hC, returnP_mul_returnW k hC, returnW_mul_returnP k hC⟩
  val_inv := corner_ext (returnV_mul_returnW k hC)
  inv_val := corner_ext (returnW_mul_returnV k hC)

theorem coe_returnUnit : (((returnUnit k hC : (ReturnCorner k hC)ˣ) : ReturnCorner k hC) :
    ClopenCrossedProduct T k) = returnV k hC :=
  rfl

theorem coe_returnUnit_inv : ((((returnUnit k hC)⁻¹ : (ReturnCorner k hC)ˣ) : ReturnCorner k hC) :
    ClopenCrossedProduct T k) = returnW k hC :=
  rfl

/-! ### `v` implements `S_C` -/

/-- **`v · f = (f ∘ S_C⁻¹) · v`** for `f ∈ LC(C, k)` extended by zero (tex 1723). -/
theorem returnV_mul_coeff_extend (f : LocallyConstant C k) :
    returnV k hC * ClopenCrossedProduct.coeff T k (clopenExtendByZero k hC.isClopen f) =
      ClopenCrossedProduct.coeff T k (clopenExtendByZero k hC.isClopen
          (LocallyConstant.comap ⟨⇑(firstReturnHomeomorph hC).symm,
            (firstReturnHomeomorph hC).symm.continuous⟩ f)) * returnV k hC := by
  rw [returnV, Finset.sum_mul, Finset.mul_sum]
  refine Finset.sum_congr rfl fun h _ => ?_
  rw [mul_assoc, cellCoeff, ← map_mul, unitPow_mul_coeff, unitPow_mul_cellCoeff k hC h, ← mul_assoc,
    backCellCoeff, ← map_mul]
  refine congrArg (fun F => ClopenCrossedProduct.coeff T k F * unitPow T k (h : ℤ)) ?_
  ext x
  simp only [LocallyConstant.coe_comap_apply, LocallyConstant.mul_apply, ContinuousMap.coe_mk,
    LocallyConstant.coe_charFn]
  by_cases hx : x ∈ returnCell T⁻¹ C h
  · have hy : (T ^ (-(h : ℤ))) x ∈ returnCell T C h :=
      (mem_zpow_image_iff (h : ℤ)).1 (by rw [image_returnCell hC]; exact hx)
    rw [Set.indicator_of_mem hy, Set.indicator_of_mem hx, Pi.one_apply, Pi.one_apply, one_mul,
      mul_one, clopenExtendByZero_apply_of_mem k hC.isClopen f hy.1,
      clopenExtendByZero_apply_of_mem k hC.isClopen _ hx.1, LocallyConstant.coe_comap_apply,
      ContinuousMap.coe_mk]
    refine congrArg f (Subtype.ext ?_)
    show (T ^ (-(h : ℤ))) x = (T⁻¹ ^ (returnTime T⁻¹ C x : ℤ)) x
    rw [hx.2, inv_zpow']
  · have hy : (T ^ (-(h : ℤ))) x ∉ returnCell T C h := fun hy =>
      hx (by rw [← image_returnCell hC]; exact (mem_zpow_image_iff (h : ℤ)).2 hy)
    rw [Set.indicator_of_notMem hy, Set.indicator_of_notMem hx, zero_mul, mul_zero]

end Unit
