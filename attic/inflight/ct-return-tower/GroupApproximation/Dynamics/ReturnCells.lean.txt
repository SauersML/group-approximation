import GroupApproximation.Dynamics.FirstReturnMap

/-!
# Return cells and extension by zero

`non_mf_groups_exist.tex`, proof of Corollary `cor:dynamic-rank-budget`
(`\label{cor:dynamic-rank-budget}`), tex 1721–1726:

> Let $C_h$ be the clopen first-return-time cells and $S_C$ the
> first-return homeomorphism.  Put $p_C=1_C$ and $P_C=p_CR_Xp_C$.
> The unit $v=\sum_h u^h1_{C_h}$ implements $S_C$, and
> $P_C=\operatorname{LC}(C,k)\rtimes_{S_C}\mathbb Z$: split any
> partial shift with endpoints in $C$ according to its successive
> returns.

The cell combinatorics and the coefficient map behind the return ring, for a return section `C`
of `T` (`IsReturnSection T r C`):

* `image_returnCell`: `T^h(C_h) = C'_h`, where `C'_h = returnCell T⁻¹ C h` is the backward return
  cell of time `h`;
* `returnCell_eq_empty_of_le`: `C_h = ∅` for `h ≥ 2r`; `disjoint_returnCell`;
* `sum_indicator_returnCell`: the cells `C_h`, `h < 2r`, partition `C`;
* `clopenExtendByZero k hC f : LocallyConstant X k`, the extension of `f : LC(C, k)` by zero,
  with `clopenExtendByZero_mul`, `_one` (`= 1_C`), `_add`, `_zero`, `_injective`, `_eq_of`, and
  `clopenExtendByZero_restrict` (`= 1_C · g` for the restriction of `g`).
-/

namespace GroupApproximation.ChainCore

open Set

variable {X : Type*} [TopologicalSpace X] {T : X ≃ₜ X} {r : ℕ} {C : Set X}

/-! ### Cells -/

/-- **`T^h(C_h) = C'_h`**: pushing the forward return cell of time `h` forward by `T^h` gives the
backward return cell of time `h`. -/
theorem image_returnCell (hC : IsReturnSection T r C) (h : ℕ) :
    (T ^ (h : ℤ)) '' returnCell T C h = returnCell T⁻¹ C h := by
  ext x
  constructor
  · rintro ⟨y, ⟨hy, rfl⟩, rfl⟩
    exact ⟨(returnTime_spec hC y).2, returnTime_inv_zpow_returnTime hC hy⟩
  · rintro ⟨hx, rfl⟩
    refine ⟨(T⁻¹ ^ (returnTime T⁻¹ C x : ℤ)) x, ⟨(returnTime_spec hC.inv x).2, ?_⟩, ?_⟩
    · have h2 := returnTime_inv_zpow_returnTime hC.inv hx
      rwa [inv_inv] at h2
    · rw [inv_zpow', zpow_apply_zpow_apply_homeomorph, add_neg_cancel, zpow_zero,
        Homeomorph.one_apply]

theorem returnCell_eq_empty_of_le (hC : IsReturnSection T r C) {h : ℕ} (hh : 2 * r ≤ h) :
    returnCell T C h = ∅ :=
  eq_empty_of_forall_notMem fun x hx => by
    have h₁ := returnTime_le hC x
    have h₂ := (returnTime_spec hC x).1
    have h₃ := hx.2
    omega

theorem disjoint_returnCell {h h' : ℕ} (hne : h ≠ h') :
    Disjoint (returnCell T C h) (returnCell T C h') :=
  disjoint_left.2 fun _ hx hx' => hne (hx.2.symm.trans hx'.2)

theorem returnCell_subset {h : ℕ} : returnCell T C h ⊆ C :=
  fun _ hx => hx.1

theorem returnTime_lt_two_mul (hC : IsReturnSection T r C) (x : X) : returnTime T C x < 2 * r := by
  have h₁ := returnTime_le hC x
  have h₂ := (returnTime_spec hC x).1
  omega

/-- **The cells `C_h`, `h < 2r`, partition `C`.** -/
theorem sum_indicator_returnCell (hC : IsReturnSection T r C) (k : Type*) [Ring k] (x : X) :
    ∑ h ∈ Finset.range (2 * r), Set.indicator (returnCell T C h) (1 : X → k) x =
      Set.indicator C (1 : X → k) x := by
  by_cases hx : x ∈ C
  · rw [Finset.sum_eq_single (returnTime T C x)]
    · rw [Set.indicator_of_mem (mem_returnCell_returnTime hx), Set.indicator_of_mem hx]
    · intro h _ hne
      exact Set.indicator_of_notMem (fun hmem => hne hmem.2.symm) _
    · intro hn
      exact absurd (Finset.mem_range.2 (returnTime_lt_two_mul hC x)) hn
  · rw [Set.indicator_of_notMem hx]
    exact Finset.sum_eq_zero fun h _ => Set.indicator_of_notMem (fun hmem => hx hmem.1) _

/-! ### Extension by zero -/

section Extend

variable (k : Type*) [Ring k] (hC : IsClopen C)

open Classical in
/-- The extension of a locally constant function on the clopen set `C` by zero. -/
noncomputable def clopenExtendByZero (f : LocallyConstant C k) : LocallyConstant X k where
  toFun x := if hx : x ∈ C then f ⟨x, hx⟩ else 0
  isLocallyConstant :=
    (LocallyConstant.piecewise hC.isClosed hC.compl.isClosed (union_compl_self C) f 0
      (fun _ hx => absurd hx.1 hx.2)).isLocallyConstant

open Classical in
theorem clopenExtendByZero_apply_of_mem (f : LocallyConstant C k) {x : X} (hx : x ∈ C) :
    clopenExtendByZero k hC f x = f ⟨x, hx⟩ := by
  show (if h : x ∈ C then f ⟨x, h⟩ else 0) = f ⟨x, hx⟩
  rw [dif_pos hx]

open Classical in
theorem clopenExtendByZero_apply_of_notMem (f : LocallyConstant C k) {x : X} (hx : x ∉ C) :
    clopenExtendByZero k hC f x = 0 := by
  show (if h : x ∈ C then f ⟨x, h⟩ else 0) = 0
  rw [dif_neg hx]

theorem clopenExtendByZero_eq_of (f : LocallyConstant C k) (g : LocallyConstant X k)
    (hin : ∀ x : C, g x = f x) (hout : ∀ x : X, x ∉ C → g x = 0) :
    clopenExtendByZero k hC f = g := by
  ext x
  by_cases hx : x ∈ C
  · rw [clopenExtendByZero_apply_of_mem k hC f hx, hin ⟨x, hx⟩]
  · rw [clopenExtendByZero_apply_of_notMem k hC f hx, hout x hx]

theorem clopenExtendByZero_mul (f g : LocallyConstant C k) :
    clopenExtendByZero k hC (f * g) = clopenExtendByZero k hC f * clopenExtendByZero k hC g := by
  ext x
  by_cases hx : x ∈ C
  · simp only [LocallyConstant.mul_apply, clopenExtendByZero_apply_of_mem k hC _ hx]
  · simp only [LocallyConstant.mul_apply, clopenExtendByZero_apply_of_notMem k hC _ hx, mul_zero]

theorem clopenExtendByZero_add (f g : LocallyConstant C k) :
    clopenExtendByZero k hC (f + g) = clopenExtendByZero k hC f + clopenExtendByZero k hC g := by
  ext x
  by_cases hx : x ∈ C
  · simp only [LocallyConstant.add_apply, clopenExtendByZero_apply_of_mem k hC _ hx]
  · simp only [LocallyConstant.add_apply, clopenExtendByZero_apply_of_notMem k hC _ hx, add_zero]

theorem clopenExtendByZero_zero : clopenExtendByZero k hC (0 : LocallyConstant C k) = 0 := by
  ext x
  by_cases hx : x ∈ C
  · simp only [LocallyConstant.zero_apply, clopenExtendByZero_apply_of_mem k hC _ hx]
  · simp only [LocallyConstant.zero_apply, clopenExtendByZero_apply_of_notMem k hC _ hx]

/-- The extension of `1` is `1_C`. -/
theorem clopenExtendByZero_one :
    clopenExtendByZero k hC (1 : LocallyConstant C k) = LocallyConstant.charFn k hC := by
  ext x
  rw [LocallyConstant.coe_charFn]
  by_cases hx : x ∈ C
  · rw [clopenExtendByZero_apply_of_mem k hC _ hx, Set.indicator_of_mem hx,
      LocallyConstant.one_apply, Pi.one_apply]
  · rw [clopenExtendByZero_apply_of_notMem k hC _ hx, Set.indicator_of_notMem hx]

theorem clopenExtendByZero_injective : Function.Injective (clopenExtendByZero k hC) :=
  fun f g hfg => LocallyConstant.ext fun x => by
    have h := congrArg (fun F : LocallyConstant X k => F x) hfg
    simpa only [clopenExtendByZero_apply_of_mem k hC _ x.2, Subtype.coe_eta] using h

/-- The restriction of `g` to `C`. -/
def clopenRestrict (g : LocallyConstant X k) : LocallyConstant C k :=
  LocallyConstant.comap ⟨Subtype.val, continuous_subtype_val⟩ g

theorem clopenRestrict_apply (g : LocallyConstant X k) (x : C) : clopenRestrict k g x = g x :=
  rfl

/-- Extending the restriction of `g` by zero gives `1_C · g`. -/
theorem clopenExtendByZero_restrict (g : LocallyConstant X k) :
    clopenExtendByZero k hC (clopenRestrict k g) = LocallyConstant.charFn k hC * g := by
  ext x
  rw [LocallyConstant.mul_apply, LocallyConstant.coe_charFn]
  by_cases hx : x ∈ C
  · rw [clopenExtendByZero_apply_of_mem k hC _ hx, clopenRestrict_apply, Set.indicator_of_mem hx,
      Pi.one_apply, one_mul]
  · rw [clopenExtendByZero_apply_of_notMem k hC _ hx, Set.indicator_of_notMem hx, zero_mul]

theorem clopenExtendByZero_charFn_mul (f : LocallyConstant C k) :
    LocallyConstant.charFn k hC * clopenExtendByZero k hC f = clopenExtendByZero k hC f := by
  ext x
  rw [LocallyConstant.mul_apply, LocallyConstant.coe_charFn]
  by_cases hx : x ∈ C
  · rw [Set.indicator_of_mem hx, Pi.one_apply, one_mul]
  · rw [clopenExtendByZero_apply_of_notMem k hC _ hx, mul_zero]

theorem clopenExtendByZero_mul_charFn (f : LocallyConstant C k) :
    clopenExtendByZero k hC f * LocallyConstant.charFn k hC = clopenExtendByZero k hC f := by
  rw [mul_comm, clopenExtendByZero_charFn_mul]

end Extend
