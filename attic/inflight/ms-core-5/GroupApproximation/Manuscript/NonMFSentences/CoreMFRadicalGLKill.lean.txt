/-!
# `thm:core-mf-radical`: the general linear kill in rank `n ≥ 3`

`non_mf_groups_exist.tex`, Theorem `thm:core-mf-radical` (`\label{thm:core-mf-radical}`,
tex lines 1609–1623), the general linear line of the radical display,

> $\Rad_{\mathrm{MF}}(\GL_n(R_X))=K_n$ $(n\ge3)$,

in its inclusion `K_n ≤ Rad_MF(GL_n(R_X))`, along the printed proof (tex 1626–1644):

> Thus $g$ is generated by roots with coefficients in $I$.  This proves equality of the
> elementary and general linear restriction kernels with $\EL_n(R_X,I)$.  [...]
> Proposition~\ref{prop:bilateral-three} kills $\EL_n(R_X,I)$ in every general linear MF image
> for $n\ge3$.

Over `F_2`, `K_n` is `DynamicRankBudget.glCoreKernel T (ZMod 2) n`, the kernel of the core
restriction `GL_n(R_X) → GL_n(R_Y)`, and `Rad_MF` is `MFQuotientUnits.mfHomKernel`, the printed
intersection of the kernels of all homomorphisms to MF groups.  The two inputs are named Props:

* `CoreKernelLeRelativeElementaryStatement`: `K_n ≤ EL_n(R_X, I)` for `n ≥ 2`, with `I` spelled
  as the ideal generated by the defects, `BilateralThree.defectIdeal`.  This is the kernel
  identification of the proof: chain-radical's `ChainRadical.ker_elementaryMatrixUnitMap_eq` at
  the core restriction, over Lemma `lem:transient-matrices` (`I` is locally matricial over `F_2`
  and generated by the defects).
* `BilateralThree.PrintedBilateralThree`, its last clause: every homomorphism from `GL_n(R_X)` to
  an MF group kills `EL_n(R_X, I)`.

`coreMFRadicalKillGL_of_pieces` is the printed step.  `coreMFRadicalKillStatement_of_halves`
assembles `DynamicRankBudget.CoreMFRadicalKillStatement` from this clause and the elementary and
local-finiteness clauses, whose spellings are copied verbatim from it.
-/

namespace GroupApproximation
namespace ChainCore
namespace CoreMFRadicalGLKill

open MFQuotientUnits DynamicRankBudget GroupApproximation.Manuscript.ChainCore.BilateralThree

/-- **Piece: the general linear restriction kernel lies in `EL_n(R_X, I)`** (tex 1626–1641,
proof of `thm:core-mf-radical`: "Thus $g$ is generated by roots with coefficients in $I$.  This
proves equality of the elementary and general linear restriction kernels with
$\EL_n(R_X,I)$."), for `n ≥ 2` over `F_2`, with `I` the ideal generated by the defects (Lemma
`lem:transient-matrices`).  Owner chain-radical. -/
def CoreKernelLeRelativeElementaryStatement : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X) (n : ℕ), 2 ≤ n →
    glCoreKernel T (ZMod 2) n ≤
      (relativeElementary (Fin n) (defectIdeal T (ZMod 2))).map
        (elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))).subtype

/-- **The general linear clause of `DynamicRankBudget.CoreMFRadicalKillStatement`**, verbatim:
over `F_2`, `K_n ≤ Rad_MF(GL_n(R_X))` for `n ≥ 3` (tex 1617). -/
def CoreMFRadicalKillGLStatement : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X) (n : ℕ), 3 ≤ n → glCoreKernel T (ZMod 2) n ≤
      mfHomKernel (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T (ZMod 2)))ˣ

/-- **The elementary clause of `DynamicRankBudget.CoreMFRadicalKillStatement`**, verbatim:
over `F_2`, `EL_n(R_X) ∩ K_n ≤ Rad_MF(EL_n(R_X))` for `n ≥ 4` (tex 1618).  Owner chain-radical. -/
def CoreMFRadicalKillELStatement : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X) (n : ℕ), 4 ≤ n → elCoreKernel T (ZMod 2) n ≤
      mfHomKernel ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2)))

/-- **The local-finiteness clause of `DynamicRankBudget.CoreMFRadicalKillStatement`**, verbatim:
over `F_2`, `K_n` is locally finite for `n ≥ 2` (tex 1614).  Owner chain-radical. -/
def CoreKernelLocallyFiniteStatement : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X) (n : ℕ), 2 ≤ n → IsLocallyFiniteGroup (glCoreKernel T (ZMod 2) n)

/-- **`K_n ≤ Rad_MF(GL_n(R_X))` for `n ≥ 3`**, along the printed route (tex 1626–1644): the
restriction kernel lies in `EL_n(R_X, I)`, and Proposition `prop:bilateral-three` kills
`EL_n(R_X, I)` under every homomorphism from `GL_n(R_X)` to an MF group. -/
theorem coreMFRadicalKillGL_of_pieces (hbil : PrintedBilateralThree)
    (hid : CoreKernelLeRelativeElementaryStatement) : CoreMFRadicalKillGLStatement := by
  intro X _ _ _ _ T n hn x hx
  obtain ⟨g, hg, rfl⟩ := Subgroup.mem_map.1 (hid X T n (by omega) hx)
  exact (mem_mfHomKernel_iff _).2 fun M _ hM f ↦ (hbil X T (ZMod 2) n hn).2.2 M hM f g hg

/-- **Assembly of `DynamicRankBudget.CoreMFRadicalKillStatement`** from its general linear,
elementary and local-finiteness clauses. -/
theorem coreMFRadicalKillStatement_of_halves (hgl : CoreMFRadicalKillGLStatement)
    (hel : CoreMFRadicalKillELStatement) (hlf : CoreKernelLocallyFiniteStatement) :
    CoreMFRadicalKillStatement := by
  intro X _ _ _ _ T
  exact ⟨hgl X T, hel X T, hlf X T⟩

end CoreMFRadicalGLKill
end ChainCore
end GroupApproximation
