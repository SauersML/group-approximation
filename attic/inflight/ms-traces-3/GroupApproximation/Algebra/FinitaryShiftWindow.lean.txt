/-!
# Copying finitary operators into a finite cycle

`non_mf_groups_exist.tex`, the remark after `cor:dynamic-rank-budget` (tex 1786–1789):

> This semidirect product is LEF: copy any finite table's finitary
> supports and the translated supports used in its products into a
> sufficiently long finite cycle, replacing the shift by cyclic translation.
> Products agree without wraparound; ...

An operator `A` on `k^{(ℤ)}` supported in a finite window `W` is copied to the cycle `ℤ/N` as
`windowCopy N W A = cyclePush N ∘ A ∘ windowLift N W`: `windowLift N W` reads the coordinates of
`W` off a vector on the cycle, and `cyclePush N` reduces coordinates modulo `N`.  When reduction
modulo `N` is injective on `W` (no wraparound):

* `windowLift_cyclePush`: `windowLift N W ∘ cyclePush N` is the identity on vectors supported in `W`;
* `one_add_windowCopy_mul`: the copies multiply as the operators do;
* `cycleShift_zpow_mul_windowCopy`: the cyclic translation conjugates a copy as the shift conjugates
  the operator, provided the support and its translate lie in `W`;
* `windowCopy_single_of_mem`, `windowCopy_single_eq_zero`: the copy on basis vectors of the cycle.
-/

namespace GroupApproximation

namespace FinitaryGL

open Finsupp

variable {k : Type*} [CommRing k] (N : ℕ)

/-- Reduction of coordinates modulo `N`, `e_j ↦ e_{j mod N}`. -/
noncomputable def cyclePush : (ℤ →₀ k) →ₗ[k] (ZMod N →₀ k) :=
  Finsupp.lmapDomain k k (Int.cast : ℤ → ZMod N)

theorem cyclePush_single (j : ℤ) (a : k) : cyclePush N (single j a) = single (j : ZMod N) a :=
  Finsupp.mapDomain_single

/-- The coordinates of a finite window `W ⊆ ℤ`, read off a vector on the cycle `ℤ/N`. -/
noncomputable def windowLift (W : Finset ℤ) : (ZMod N →₀ k) →ₗ[k] (ℤ →₀ k) where
  toFun v := Finsupp.onFinset W (fun i => if i ∈ W then v (i : ZMod N) else 0) fun i hi => by
    by_contra h
    exact hi (if_neg h)
  map_add' v w := Finsupp.ext fun i => by
    show (if i ∈ W then (v + w) (i : ZMod N) else 0) =
      (if i ∈ W then v (i : ZMod N) else 0) + (if i ∈ W then w (i : ZMod N) else 0)
    split_ifs
    · rfl
    · exact (add_zero 0).symm
  map_smul' c v := Finsupp.ext fun i => by
    show (if i ∈ W then c • v (i : ZMod N) else 0) = c • (if i ∈ W then v (i : ZMod N) else 0)
    split_ifs
    · rfl
    · exact (smul_zero c).symm

theorem windowLift_apply (W : Finset ℤ) (v : ZMod N →₀ k) (i : ℤ) :
    windowLift N W v i = if i ∈ W then v (i : ZMod N) else 0 :=
  rfl

/-- The copy of an operator on `k^{(ℤ)}` into the cycle `ℤ/N`, read through the window `W`. -/
noncomputable def windowCopy (W : Finset ℤ) (A : Module.End k (ℤ →₀ k)) :
    Module.End k (ZMod N →₀ k) :=
  cyclePush N ∘ₗ A ∘ₗ windowLift N W

theorem windowCopy_apply (W : Finset ℤ) (A : Module.End k (ℤ →₀ k)) (v : ZMod N →₀ k) :
    windowCopy N W A v = cyclePush N (A (windowLift N W v)) :=
  rfl

/-- **No wraparound**: reading a window back after reduction recovers vectors supported in it. -/
theorem windowLift_cyclePush {W : Finset ℤ} (hinj : Set.InjOn (Int.cast : ℤ → ZMod N) (W : Set ℤ))
    {v : ℤ →₀ k} (hv : v ∈ supported k k (W : Set ℤ)) : windowLift N W (cyclePush N v) = v := by
  ext i
  rw [windowLift_apply]
  split_ifs with hi
  · exact Finsupp.mapDomain_apply' (W : Set ℤ) v hv hinj (Finset.mem_coe.2 hi)
  · exact ((mem_supported' k v).1 hv i fun h => hi (Finset.mem_coe.1 h)).symm

theorem cyclePush_injOn {W : Finset ℤ} (hinj : Set.InjOn (Int.cast : ℤ → ZMod N) (W : Set ℤ))
    {v w : ℤ →₀ k} (hv : v ∈ supported k k (W : Set ℤ)) (hw : w ∈ supported k k (W : Set ℤ))
    (h : cyclePush N v = cyclePush N w) : v = w := by
  rw [← windowLift_cyclePush N hinj hv, ← windowLift_cyclePush N hinj hw, h]

theorem windowCopy_add (W : Finset ℤ) (A B : Module.End k (ℤ →₀ k)) :
    windowCopy N W (A + B) = windowCopy N W A + windowCopy N W B := by
  refine LinearMap.ext fun v => ?_
  show cyclePush N ((A + B) (windowLift N W v)) =
    cyclePush N (A (windowLift N W v)) + cyclePush N (B (windowLift N W v))
  rw [LinearMap.add_apply, map_add]

theorem windowCopy_zero (W : Finset ℤ) : windowCopy N W (0 : Module.End k (ℤ →₀ k)) = 0 := by
  refine LinearMap.ext fun v => ?_
  show cyclePush N ((0 : Module.End k (ℤ →₀ k)) (windowLift N W v)) = 0
  rw [LinearMap.zero_apply, map_zero]

/-- **Products agree without wraparound.** -/
theorem windowCopy_mul {W : Finset ℤ} (hinj : Set.InjOn (Int.cast : ℤ → ZMod N) (W : Set ℤ))
    {A B : Module.End k (ℤ →₀ k)} (hB : SupportedIn k W B) :
    windowCopy N W A * windowCopy N W B = windowCopy N W (A * B) := by
  refine LinearMap.ext fun v => ?_
  show cyclePush N (A (windowLift N W (cyclePush N (B (windowLift N W v))))) =
    cyclePush N (A (B (windowLift N W v)))
  rw [windowLift_cyclePush N hinj (hB.1 _)]

theorem one_add_windowCopy_mul {W : Finset ℤ}
    (hinj : Set.InjOn (Int.cast : ℤ → ZMod N) (W : Set ℤ)) {A B : Module.End k (ℤ →₀ k)}
    (hB : SupportedIn k W B) :
    (1 + windowCopy N W A) * (1 + windowCopy N W B) = 1 + windowCopy N W (A * B + A + B) := by
  rw [windowCopy_add, windowCopy_add, ← windowCopy_mul N hinj hB]
  noncomm_ring

theorem windowLift_single_of_mem {W : Finset ℤ}
    (hinj : Set.InjOn (Int.cast : ℤ → ZMod N) (W : Set ℤ)) {j : ℤ} (hj : j ∈ W) (a : k) :
    windowLift N W (single (j : ZMod N) a) = single j a := by
  rw [← cyclePush_single]
  exact windowLift_cyclePush N hinj (single_mem_supported k a (Finset.mem_coe.2 hj))

theorem windowCopy_single_of_mem {W : Finset ℤ}
    (hinj : Set.InjOn (Int.cast : ℤ → ZMod N) (W : Set ℤ)) (A : Module.End k (ℤ →₀ k)) {j : ℤ}
    (hj : j ∈ W) (a : k) :
    windowCopy N W A (single (j : ZMod N) a) = cyclePush N (A (single j a)) := by
  rw [windowCopy_apply, windowLift_single_of_mem N hinj hj]

theorem windowLift_single_eq_zero {W : Finset ℤ} {c : ZMod N} (hc : ∀ j ∈ W, (j : ZMod N) ≠ c)
    (a : k) : windowLift N W (single c a) = 0 := by
  ext i
  rw [windowLift_apply, Finsupp.zero_apply]
  split_ifs with hi
  · exact Finsupp.single_eq_of_ne (hc i hi)
  · rfl

/-- A basis vector of the cycle far from the window is fixed by the deviation of every copy. -/
theorem windowCopy_single_eq_zero {W : Finset ℤ} {c : ZMod N} (hc : ∀ j ∈ W, (j : ZMod N) ≠ c)
    (A : Module.End k (ℤ →₀ k)) (a : k) : windowCopy N W A (single c a) = 0 := by
  rw [windowCopy_apply, windowLift_single_eq_zero N hc, map_zero, map_zero]

/-- The cyclic translation `e_c ↦ e_{c+1}` of `k^{(ℤ/N)}`. -/
noncomputable def cycleShift : LinearMap.GeneralLinearGroup k (ZMod N →₀ k) :=
  LinearMap.GeneralLinearGroup.ofLinearEquiv (Finsupp.domLCongr (Equiv.addRight (1 : ZMod N)))

theorem cycleShift_single (c : ZMod N) (a : k) :
    ((cycleShift N : LinearMap.GeneralLinearGroup k (ZMod N →₀ k)) : Module.End k (ZMod N →₀ k))
      (single c a) = single (c + 1) a :=
  Finsupp.domLCongr_single (Equiv.addRight (1 : ZMod N)) c a

theorem cycleShift_inv_single (c : ZMod N) (a : k) :
    (((cycleShift N)⁻¹ : LinearMap.GeneralLinearGroup k (ZMod N →₀ k)) :
      Module.End k (ZMod N →₀ k)) (single c a) = single (c - 1) a := by
  have h := cycleShift_single (k := k) N (c - 1) a
  rw [sub_add_cancel] at h
  rw [← h, ← Module.End.mul_apply, ← Units.val_mul, inv_mul_cancel, Units.val_one,
    Module.End.one_apply]

theorem cycleShift_zpow_single (n : ℤ) (c : ZMod N) (a : k) :
    ((cycleShift N ^ n : LinearMap.GeneralLinearGroup k (ZMod N →₀ k)) :
      Module.End k (ZMod N →₀ k)) (single c a) = single (c + n) a := by
  induction n using Int.induction_on generalizing c with
  | zero => rw [zpow_zero, Units.val_one, Module.End.one_apply, Int.cast_zero, add_zero]
  | succ n ih =>
      rw [zpow_add_one, Units.val_mul, Module.End.mul_apply, cycleShift_single, ih]
      congr 1
      push_cast
      ring
  | pred n ih =>
      rw [zpow_sub_one, Units.val_mul, Module.End.mul_apply, cycleShift_inv_single, ih]
      congr 1
      push_cast
      ring

/-- Reduction modulo `N` intertwines the shift with the cyclic translation. -/
theorem cyclePush_shiftGL_zpow (n : ℤ) (v : ℤ →₀ k) :
    cyclePush N (((shiftGL k ^ n : LinearMap.GeneralLinearGroup k (ℤ →₀ k)) :
      Module.End k (ℤ →₀ k)) v) =
      ((cycleShift N ^ n : LinearMap.GeneralLinearGroup k (ZMod N →₀ k)) :
        Module.End k (ZMod N →₀ k)) (cyclePush N v) := by
  have h : cyclePush N ∘ₗ ((shiftGL k ^ n : LinearMap.GeneralLinearGroup k (ℤ →₀ k)) :
      Module.End k (ℤ →₀ k)) =
      ((cycleShift N ^ n : LinearMap.GeneralLinearGroup k (ZMod N →₀ k)) :
        Module.End k (ZMod N →₀ k)) ∘ₗ cyclePush N := by
    refine Finsupp.lhom_ext fun j a => ?_
    rw [LinearMap.comp_apply, LinearMap.comp_apply, shiftGL_zpow_single, cyclePush_single,
      cyclePush_single, cycleShift_zpow_single, Int.cast_add]
  exact LinearMap.congr_fun h v

/-- An operator supported in `S` sees only the coordinates of `S`. -/
theorem SupportedIn.apply_eq_of_eqOn {S : Finset ℤ} {A : Module.End k (ℤ →₀ k)}
    (hA : SupportedIn k S A) {u u' : ℤ →₀ k} (h : ∀ i ∈ S, u i = u' i) : A u = A u' := by
  have hmem : u - u' ∈ supported k k ((S : Set ℤ)ᶜ) :=
    (mem_supported' k _).2 fun i hi => by
      have hiS : i ∈ S := Finset.mem_coe.1 (by
        by_contra h'
        exact hi ((Set.mem_compl_iff _ _).2 h'))
      rw [Finsupp.sub_apply, h i hiS, sub_self]
  have h0 := hA.2 _ hmem
  rwa [map_sub, sub_eq_zero] at h0

/-- Conjugating by a power of the shift translates the support. -/
theorem SupportedIn.conj {S S' : Finset ℤ} {A : Module.End k (ℤ →₀ k)} (hA : SupportedIn k S A)
    (n : ℤ) (hSS' : ∀ i ∈ S, i + n ∈ S') :
    SupportedIn k S' (((shiftGL k ^ n : LinearMap.GeneralLinearGroup k (ℤ →₀ k)) :
      Module.End k (ℤ →₀ k)) * A *
        ((shiftGL k ^ (-n) : LinearMap.GeneralLinearGroup k (ℤ →₀ k)) :
          Module.End k (ℤ →₀ k))) := by
  refine ⟨fun v => (mem_supported' k _).2 fun i hi => ?_, fun v hv => ?_⟩
  · rw [Module.End.mul_apply, Module.End.mul_apply, shiftGL_zpow_apply]
    refine (mem_supported' k _).1 (hA.1 _) (i - n) fun h => hi ?_
    have h' := hSS' _ (Finset.mem_coe.1 h)
    rw [sub_add_cancel] at h'
    exact Finset.mem_coe.2 h'
  · have hv' : ((shiftGL k ^ (-n) : LinearMap.GeneralLinearGroup k (ℤ →₀ k)) :
        Module.End k (ℤ →₀ k)) v ∈ supported k k ((S : Set ℤ)ᶜ) :=
      (mem_supported' k _).2 fun i hi => by
        have hiS : i ∈ S := Finset.mem_coe.1 (by
          by_contra h'
          exact hi ((Set.mem_compl_iff _ _).2 h'))
        rw [shiftGL_zpow_apply, sub_neg_eq_add]
        exact (mem_supported' k _).1 hv (i + n) fun h =>
          (Set.mem_compl_iff _ _).1 h (Finset.mem_coe.2 (hSS' i hiS))
    rw [Module.End.mul_apply, Module.End.mul_apply, hA.2 _ hv', map_zero]

/-- **Replacing the shift by cyclic translation**: the cyclic translation conjugates a copy as
the shift conjugates the operator, while the support and its translate lie in the window. -/
theorem cycleShift_zpow_mul_windowCopy {W S : Finset ℤ} {A : Module.End k (ℤ →₀ k)}
    (hA : SupportedIn k S A) (hSW : S ⊆ W) (n : ℤ) (hSn : ∀ i ∈ S, i + n ∈ W) :
    ((cycleShift N ^ n : LinearMap.GeneralLinearGroup k (ZMod N →₀ k)) :
      Module.End k (ZMod N →₀ k)) * windowCopy N W A =
      windowCopy N W (((shiftGL k ^ n : LinearMap.GeneralLinearGroup k (ℤ →₀ k)) :
        Module.End k (ℤ →₀ k)) * A *
          ((shiftGL k ^ (-n) : LinearMap.GeneralLinearGroup k (ℤ →₀ k)) :
            Module.End k (ℤ →₀ k))) *
        ((cycleShift N ^ n : LinearMap.GeneralLinearGroup k (ZMod N →₀ k)) :
          Module.End k (ZMod N →₀ k)) := by
  refine Finsupp.lhom_ext fun c a => ?_
  rw [Module.End.mul_apply, Module.End.mul_apply, windowCopy_apply, windowCopy_apply,
    ← cyclePush_shiftGL_zpow, cycleShift_zpow_single, Module.End.mul_apply, Module.End.mul_apply]
  refine congrArg (fun w => cyclePush N (((shiftGL k ^ n : LinearMap.GeneralLinearGroup k
    (ℤ →₀ k)) : Module.End k (ℤ →₀ k)) w)) (hA.apply_eq_of_eqOn fun i hi => ?_)
  rw [windowLift_apply, shiftGL_zpow_apply, sub_neg_eq_add, windowLift_apply, if_pos (hSW hi),
    if_pos (hSn i hi), Int.cast_add, Finsupp.single_apply, Finsupp.single_apply]
  simp only [add_left_inj]

/-- Reduction modulo `N` is injective on a symmetric window of radius `M` when `2M < N`. -/
theorem intCast_injOn_Icc {M : ℕ} (hM : 2 * M < N) :
    Set.InjOn (Int.cast : ℤ → ZMod N) ((Finset.Icc (-(M : ℤ)) M : Finset ℤ) : Set ℤ) := by
  intro a ha b hb hab
  rw [Finset.coe_Icc, Set.mem_Icc] at ha hb
  have h0 := Int.eq_zero_of_abs_lt_dvd ((ZMod.intCast_eq_intCast_iff_dvd_sub a b N).1 hab)
    (by rw [abs_lt]; constructor <;> omega)
  omega

/-- Distinct integers closer than `N` stay distinct modulo `N`. -/
theorem intCast_ne_of_ne {a b : ℤ} (hab : a ≠ b) (h : |b - a| < N) : (a : ZMod N) ≠ b :=
  fun he => hab (by
    have h0 := Int.eq_zero_of_abs_lt_dvd ((ZMod.intCast_eq_intCast_iff_dvd_sub a b N).1 he) h
    omega)
