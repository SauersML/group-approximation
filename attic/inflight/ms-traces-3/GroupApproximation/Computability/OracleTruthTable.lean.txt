import Mathlib.Computability.TuringDegree
import Mathlib.Computability.Primrec.List
import GroupApproximation.Meta.AxiomGuard

/-!
# Truth-table reductions are Turing reductions

The oracle-relative primitive layer for the word-problem reduction of
`simple_kazhdan_sofic_group.tex`, section "Word problems" ("so `L(X)` computes the word problem").

A total oracle `g` answers the finitely many queries `Q n`, computed primitive recursively from the
input, and a primitive recursive evaluation `D` turns the list of answers into the value `f n`.
Then `f` is Turing reducible to `g`.  Mathlib's `Nat.RecursiveIn` has no composition lemma for coded
types, so the answers are collected by one primitive recursion over the query index, calling the
oracle once at each step.

* `recursiveIn_of_primrec`: primitive recursive functions are recursive in every oracle;
* `rec_answers`: the primitive recursion collects the answers to the first `k` queries;
* `turingReducible_of_truthTable`: the truth-table reduction.
-/

namespace GroupApproximation
namespace OracleTruthTable

open Encodable

/-- Primitive recursive functions are recursive in every set of oracles. -/
theorem recursiveIn_of_primrec {O : Set (ℕ →. ℕ)} {c : ℕ → ℕ} (hc : Primrec c) :
    Nat.RecursiveIn O (fun n => Part.some (c n)) :=
  (Primrec.nat_iff.1 hc).recursiveIn

/-- The `y`-th query for the input `a`. -/
def query (Q : ℕ → List ℕ) (a y : ℕ) : ℕ :=
  (Q a).getD y 0

/-- The code of the answers of `gt` to the first `k` queries for the input `a`. -/
def answersCode (gt : ℕ → ℕ) (Q : ℕ → List ℕ) (a k : ℕ) : ℕ :=
  encode ((List.range k).map fun y => gt (query Q a y))

/-- The query read off `m = ⟪a, ⟪y, i⟫⟫`. -/
def queryAt (Q : ℕ → List ℕ) (m : ℕ) : ℕ :=
  query Q (Nat.unpair m).1 (Nat.unpair (Nat.unpair m).2).1

/-- Append the answer in `m' = ⟪m, answer⟫` to the coded list `i` in `m = ⟪a, ⟪y, i⟫⟫`. -/
def appendAnswer (m' : ℕ) : ℕ :=
  encode (((decode (Nat.unpair (Nat.unpair (Nat.unpair m').1).2).2 : Option (List ℕ)).getD []) ++
    [(Nat.unpair m').2])

/-- Evaluate `D` at `m = ⟪n, code of the answers⟫`. -/
def evalAt (D : ℕ → List ℕ → ℕ) (m : ℕ) : ℕ :=
  D (Nat.unpair m).1 ((decode (Nat.unpair m).2 : Option (List ℕ)).getD [])

theorem primrec_queryAt {Q : ℕ → List ℕ} (hQ : Primrec Q) : Primrec (queryAt Q) :=
  (Primrec.list_getD 0).comp (hQ.comp (Primrec.fst.comp Primrec.unpair))
    (Primrec.fst.comp (Primrec.unpair.comp (Primrec.snd.comp Primrec.unpair)))

theorem primrec_appendAnswer : Primrec appendAnswer :=
  Primrec.encode.comp (Primrec.list_append.comp
    (Primrec.option_getD.comp (Primrec.decode.comp (Primrec.snd.comp (Primrec.unpair.comp
      (Primrec.snd.comp (Primrec.unpair.comp (Primrec.fst.comp Primrec.unpair))))))
      (Primrec.const []))
    (Primrec.list_cons.comp (Primrec.snd.comp Primrec.unpair) (Primrec.const [])))

theorem primrec_evalAt {D : ℕ → List ℕ → ℕ} (hD : Primrec₂ D) : Primrec (evalAt D) :=
  hD.comp (Primrec.fst.comp Primrec.unpair)
    (Primrec.option_getD.comp (Primrec.decode.comp (Primrec.snd.comp Primrec.unpair))
      (Primrec.const []))

theorem map_range_getD (l : List ℕ) (gt : ℕ → ℕ) :
    (List.range l.length).map (fun y => gt (l.getD y 0)) = l.map gt := by
  refine List.ext_getElem (by simp) fun i h1 _ => ?_
  have hi : i < l.length := by simpa using h1
  simp only [List.getElem_map, List.getElem_range]
  rw [List.getD_eq_getElem?_getD, List.getElem?_eq_getElem hi, Option.getD_some]

/-- One step of the recursion, by definition. -/
theorem rec_succ (z : Part ℕ) (s : ℕ → Part ℕ → Part ℕ) (k : ℕ) :
    (Nat.rec (motive := fun _ => Part ℕ) z s (k + 1) : Part ℕ) = s k (Nat.rec z s k) :=
  rfl

/-- **The primitive recursion collects the answers.** -/
theorem rec_answers (gt : ℕ → ℕ) (Q : ℕ → List ℕ) (a k : ℕ) :
    (Nat.rec (motive := fun _ => Part ℕ) (Part.some (encode ([] : List ℕ))) (fun y IH => IH >>= fun i =>
        Part.some (appendAnswer (Nat.pair (Nat.pair a (Nat.pair y i))
          (gt (queryAt Q (Nat.pair a (Nat.pair y i))))))) k : Part ℕ) =
      Part.some (answersCode gt Q a k) := by
  induction k with
  | zero => simp [answersCode]
  | succ k ih =>
    rw [rec_succ, ih, Part.bind_eq_bind, Part.bind_some]
    simp [answersCode, appendAnswer, queryAt, List.range_succ]

/-- **Truth-table reductions are Turing reductions.**  If a total oracle `g = gt` answers the
queries `Q n` and a primitive recursive `D` computes `f n` from the answers, then `f ≤ᵀ g`. -/
theorem turingReducible_of_truthTable {f g : ℕ →. ℕ} {gt : ℕ → ℕ}
    (hg : ∀ n, g n = Part.some (gt n)) {Q : ℕ → List ℕ} (hQ : Primrec Q) {D : ℕ → List ℕ → ℕ}
    (hD : Primrec₂ D) (hf : ∀ n, f n = Part.some (D n ((Q n).map gt))) : TuringReducible f g := by
  refine RecursiveIn.iff_nat.2 ?_
  have hor : Nat.RecursiveIn {g} g := Nat.RecursiveIn.oracle g (Set.mem_singleton g)
  have hstep : Nat.RecursiveIn {g}
      (fun m => Part.some (appendAnswer (Nat.pair m (gt (queryAt Q m))))) := by
    refine Nat.RecursiveIn.of_eq
      (Nat.RecursiveIn.comp (recursiveIn_of_primrec primrec_appendAnswer)
        (Nat.RecursiveIn.pair (recursiveIn_of_primrec Primrec.id)
          (Nat.RecursiveIn.comp hor (recursiveIn_of_primrec (primrec_queryAt hQ))))) fun m => ?_
    simp [hg]
  have hans : Nat.RecursiveIn {g}
      (fun p => Part.some (answersCode gt Q (Nat.unpair p).1 (Nat.unpair p).2)) := by
    refine Nat.RecursiveIn.of_eq (Nat.RecursiveIn.prec
      (recursiveIn_of_primrec (Primrec.const (encode ([] : List ℕ)))) hstep) fun p => ?_
    rw [← Nat.pair_unpair p]
    simp only [Nat.unpair_pair]
    exact rec_answers gt Q _ _
  have hlen : Nat.RecursiveIn {g} (fun n => Part.some (Nat.pair n (Q n).length)) :=
    recursiveIn_of_primrec (Primrec₂.natPair.comp Primrec.id (Primrec.list_length.comp hQ))
  have hA : Nat.RecursiveIn {g} (fun n => Part.some (encode ((Q n).map gt))) := by
    refine Nat.RecursiveIn.of_eq (Nat.RecursiveIn.comp hans hlen) fun n => ?_
    simp only [Part.bind_eq_bind, Part.bind_some, Nat.unpair_pair, answersCode, query]
    rw [map_range_getD]
  refine Nat.RecursiveIn.of_eq (Nat.RecursiveIn.comp (recursiveIn_of_primrec (primrec_evalAt hD))
    (Nat.RecursiveIn.pair (recursiveIn_of_primrec Primrec.id) hA)) fun n => ?_
  simp [hf, evalAt]

end OracleTruthTable
end GroupApproximation
