/-!
# The cycle condition makes the crossed product LEF

`non_mf_groups_exist.tex`, proof of `lem:chain-core-models`, tex 1386–1400:

> A subshift satisfying the cycle condition has an LEF crossed product.  Indeed, fix a finite ring
> table, express its entries in Laurent form, and choose a window determining all coefficient
> functions and their translates used in that table.  At a sufficiently high word-graph level,
> choose cycles containing words witnessing every nonzero coefficient of every difference between
> distinct entries.  Reading these cycles gives finite periodic models whose relevant windows all
> belong to the subshift.  Repeat each cycle so its length exceeds twice the largest absolute
> Laurent degree appearing in the table. [...] The periodic models need not themselves belong to
> the subshift.

Let `X ⊆ A^ℤ` be closed and shift-invariant over a finite discrete alphabet, satisfying
`CycleCondition X`, let `T : X ≃ₜ X` act as the shift and let `k` be a finite ring.
`isLEFRing_clopenCrossedProduct_of_cycleCondition` proves that `ClopenCrossedProduct T k` is LEF
through `Pestov91.isLEFRing_skewMonoidAlgebra_of_periodic`.  For a finite set `S` of coefficients
and a degree bound `B`:

* `exists_radius` gives a window `[-ρ, ρ]` determining every coefficient in `S` and its translates
  by `|j| ≤ B`;
* each nonzero coefficient `s` has a witness point `x s`, whose word on `[-ρ, ρ]` is an edge of the
  word graph at level `2ρ`; the edge is cyclic, and `exists_periodic_of_isCyclicEdge` reads a
  periodic word `p s : ZMod (m s) → A` starting with it, all of whose windows are words of `X`;
* the cycles are repeated to a common length `N > 2B`, and the window at each `n : ZMod N` is
  realized by a point `w s n` of `X`;
* `evalFamily T k w` evaluates coefficients along these points, diagonally on `ZMod N → (F → k)`.

Model tests: `cycleCondition_univ` and `isLEFRing_clopenCrossedProduct_univ` (the full shift).
The printed statement is `PrintedSubshiftCycleConditionLEF`.
-/

namespace GroupApproximation
namespace WordGraph

/-- **The cycle condition makes the crossed product LEF** (tex 1386–1400). -/
theorem isLEFRing_clopenCrossedProduct_of_cycleCondition {A : Type} [TopologicalSpace A]
    [DiscreteTopology A] [Finite A] {X : Set (ℤ → A)} (hXc : IsClosed X)
    (hX : ∀ n : ℤ, Set.MapsTo (shift n) X X) (T : ↥X ≃ₜ ↥X) (hT : ∀ x : ↥X, (T x).1 = shift 1 x.1)
    (hcyc : CycleCondition X) (k : Type) [Ring k] [Finite k] :
    Pestov91.IsLEFRing (ClopenCrossedProduct T k) := by
  classical
  refine Pestov91.isLEFRing_skewMonoidAlgebra_of_periodic fun S B => ?_
  choose r hr using fun a : ClopenCoeff T k => exists_radius hXc ((ClopenCoeff.of T k).symm a)
  obtain ⟨ρ, hρr⟩ : ∃ ρ : ℕ, ∀ a ∈ S, r a + B ≤ ρ :=
    ⟨S.sup r + B, fun a ha => by have h := Finset.le_sup (f := r) ha; omega⟩
  obtain ⟨F, hF⟩ : ∃ F : Finset (ClopenCoeff T k), ∀ a, a ∈ F ↔ a ∈ S ∧ a ≠ 0 :=
    ⟨S.filter (· ≠ 0), fun a => Finset.mem_filter⟩
  have hwit : ∀ s : ↥F, ∃ x : ↥X, (ClopenCoeff.of T k).symm s.1 x ≠ 0 := by
    intro s
    by_contra h
    push Not at h
    exact ((hF s.1).mp s.2).2 ((ClopenCoeff.of T k).symm.map_eq_zero_iff.mp
      (LocallyConstant.ext h : (ClopenCoeff.of T k).symm s.1 = 0))
  choose x hx using hwit
  choose m hm p hp0 hpL using fun s : ↥F => exists_periodic_of_isCyclicEdge
    (hcyc (2 * ρ) _ (word_mem_language hX (x s).2 (-(ρ : ℤ)) (2 * ρ + 1)))
  obtain ⟨N, hN, hdvd⟩ : ∃ N : ℕ, 2 * B < N ∧ ∀ s : ↥F, m s ∣ N := by
    have hpos : 0 < ∏ s : ↥F, m s :=
      Finset.prod_pos fun s _ => Nat.pos_of_ne_zero (hm s).out
    exact ⟨(2 * B + 1) * ∏ s : ↥F, m s,
      by have := Nat.le_mul_of_pos_right (2 * B + 1) hpos; omega,
      fun s => dvd_mul_of_dvd_right (Finset.dvd_prod_of_mem m (Finset.mem_univ s)) _⟩
  obtain ⟨c⟩ : Nonempty (∀ s : ↥F, ZMod N →+* ZMod (m s)) :=
    ⟨fun s => ZMod.castHom (hdvd s) _⟩
  have hreal : ∀ (s : ↥F) (n : ZMod N), ∃ y ∈ X,
      word y 0 (2 * ρ + 1) = fun j : Fin (2 * ρ + 1) => p s (c s n + (j : ℕ)) :=
    fun s n => hpL s (c s n)
  choose y hyX hyw using hreal
  obtain ⟨w, hw⟩ : ∃ w : ↥F → ZMod N → ↥X, ∀ (s : ↥F) (n : ZMod N) (i : ℤ), |i| ≤ ρ →
      (w s n).1 i = p s (c s n + ((ρ + i : ℤ) : ZMod (m s))) := by
    refine ⟨fun s n => ⟨shift (ρ : ℤ) (y s n), hX ρ (hyX s n)⟩, fun s n i hi => ?_⟩
    obtain ⟨hi1, hi2⟩ := abs_le.mp hi
    obtain ⟨j, hj⟩ : ∃ j : ℕ, (j : ℤ) = ρ + i := ⟨(ρ + i).toNat, Int.toNat_of_nonneg (by omega)⟩
    have h := congrFun (hyw s n) ⟨j, by omega⟩
    simp only [word_apply, zero_add] at h
    show y s n (ρ + i) = _
    rw [← hj, Int.cast_natCast]
    exact h
  refine ⟨↥F → k, inferInstance, inferInstance, N, ⟨by omega⟩, evalFamily T k w, hN,
    fun a ha hne hD => ?_, fun a ha g hg n => ?_⟩
  · obtain ⟨s, hs⟩ : ∃ s : ↥F, s.1 = a := ⟨⟨a, (hF a).mpr ⟨ha, hne⟩⟩, rfl⟩
    refine hx s ?_
    have h0 : (ClopenCoeff.of T k).symm a (w s 0) = 0 := congrFun (congrFun hD 0) s
    rw [hs, ← h0]
    refine hr a _ _ fun i hi => ?_
    have hra := hρr a ha
    obtain ⟨hi1, hi2⟩ := abs_le.mp hi
    obtain ⟨j, hj⟩ : ∃ j : ℕ, (j : ℤ) = ρ + i := ⟨(ρ + i).toNat, Int.toNat_of_nonneg (by omega)⟩
    rw [hw s 0 i (abs_le.mpr ⟨by omega, by omega⟩), map_zero, zero_add, ← hj, Int.cast_natCast]
    have h := hp0 s ⟨j, by omega⟩
    simp only [word_apply] at h
    rw [h]
    congr 1
    omega
  · funext s
    show (ClopenCoeff.of T k).symm a ((T ^ (-toAdd g)) (w s n)) =
      (ClopenCoeff.of T k).symm a (w s (n - ((toAdd g : ℤ) : ZMod N)))
    refine hr a _ _ fun i hi => ?_
    have hra := hρr a ha
    obtain ⟨hg1, hg2⟩ := abs_le.mp hg
    obtain ⟨hi1, hi2⟩ := abs_le.mp hi
    rw [zpow_apply_coord hT, hw s n (-toAdd g + i) (abs_le.mpr ⟨by omega, by omega⟩),
      hw s (n - ((toAdd g : ℤ) : ZMod N)) i (abs_le.mpr ⟨by omega, by omega⟩), map_sub,
      map_intCast]
    congr 1
    push_cast
    ring

variable {A : Type*}

/-- Consecutive words of a point of `X` are joined by a directed path of the word graph. -/
theorem reflTransGen_word {X : Set (ℤ → A)} (hX : ∀ n : ℤ, Set.MapsTo (shift n) X X)
    {x : ℤ → A} (hx : x ∈ X) (i : ℤ) (r n : ℕ) :
    Relation.ReflTransGen (wordEdge X r) (word x i r) (word x (i + n) r) := by
  induction n with
  | zero => rw [show i + ((0 : ℕ) : ℤ) = i by omega]
  | succ n ih =>
    rw [show i + ((n + 1 : ℕ) : ℤ) = i + n + 1 by omega]
    exact ih.tail (wordEdge_word hX hx (i + n) r)

/-- **Model test.** The full shift satisfies the cycle condition: every word `w` of length `r + 1`
is read from a configuration showing `Fin.tail w` at `0` and `Fin.init w` at `r`. -/
theorem cycleCondition_univ : CycleCondition (Set.univ : Set (ℤ → A)) := by
  intro r w hw
  refine ⟨hw, ?_⟩
  obtain ⟨z, hz1, hz2⟩ : ∃ z : ℤ → A, word z 0 r = Fin.tail w ∧ word z r r = Fin.init w := by
    refine ⟨fun i => if h : i.toNat < r then Fin.tail w ⟨i.toNat, h⟩
      else if h' : i.toNat - r < r then Fin.init w ⟨i.toNat - r, h'⟩ else w 0,
      funext fun j => ?_, funext fun j => ?_⟩
    · have hj := j.2
      have h1 : ((0 : ℤ) + ((j : ℕ) : ℤ)).toNat < r := by omega
      simp only [word_apply]
      rw [dif_pos h1]
      exact congrArg (Fin.tail w) (Fin.ext (show ((0 : ℤ) + ((j : ℕ) : ℤ)).toNat = (j : ℕ) by omega))
    · have hj := j.2
      have h1 : ¬((r : ℤ) + ((j : ℕ) : ℤ)).toNat < r := by omega
      have h2 : ((r : ℤ) + ((j : ℕ) : ℤ)).toNat - r < r := by omega
      simp only [word_apply]
      rw [dif_neg h1, dif_pos h2]
      exact congrArg (Fin.init w)
        (Fin.ext (show ((r : ℤ) + ((j : ℕ) : ℤ)).toNat - r = (j : ℕ) by omega))
  have h := reflTransGen_word (fun n => Set.mapsTo_univ (shift n) _) (Set.mem_univ z) 0 r r
  rw [zero_add, hz1, hz2] at h
  exact h

/-- **Model test.** The full shift over a finite alphabet has an LEF crossed product. -/
theorem isLEFRing_clopenCrossedProduct_univ {A : Type} [TopologicalSpace A] [DiscreteTopology A]
    [Finite A] (k : Type) [Ring k] [Finite k] :
    Pestov91.IsLEFRing (ClopenCrossedProduct
      (shiftHomeomorph (X := (Set.univ : Set (ℤ → A))) fun n => Set.mapsTo_univ (shift n) _) k) :=
  isLEFRing_clopenCrossedProduct_of_cycleCondition isClosed_univ
    (fun n => Set.mapsTo_univ (shift n) _) _ (shiftHomeomorph_apply _) cycleCondition_univ k

/-- **tex 1386** (proof of `lem:chain-core-models`): "A subshift satisfying the cycle condition has
an LEF crossed product."  A subshift is a closed shift-invariant `X ⊆ A^ℤ` over a finite alphabet
(tex 1380); the cycle condition is `CycleCondition X` (tex 1380–1384); the crossed product is
`R_X = LC(X, k) ⋊_T ℤ` for the shift `T` and a finite field `k` (tex 1346–1348). -/
def PrintedSubshiftCycleConditionLEF : Prop :=
  ∀ (A : Type) [TopologicalSpace A] [DiscreteTopology A] [Finite A] (X : Set (ℤ → A)),
    IsClosed X → ∀ (hX : ∀ n : ℤ, Set.MapsTo (shift n) X X), CycleCondition X →
    ∀ (k : Type) [Field k] [Finite k],
      Pestov91.IsLEFRing (ClopenCrossedProduct (shiftHomeomorph hX) k)

theorem printedSubshiftCycleConditionLEF : PrintedSubshiftCycleConditionLEF :=
  fun _ _ _ _ _ hXc hX hcyc k _ _ =>
    isLEFRing_clopenCrossedProduct_of_cycleCondition hXc hX _ (shiftHomeomorph_apply hX) hcyc k
