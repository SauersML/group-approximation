/-!
# The pinch step of the collar strip

Stage 1 of the collar insertion (`GeodesicCollar.StripStatement`) follows from the pinch step
`GeodesicCollar.StripStepStatement` (`stripStatement_of_step`).  This module proves the pinch step
(`StripStep.stripStepStatement_holds`), so the collar insertion is closed
(`StripStep.geodesicCollarStatement_holds`, through `geodesicCollarStatement_of_strip`).

Let `X` be a strip along `p` whose remaining walk starts with `d`.

1. `d` is not in the strip face (`not_mem_face`, `faceOf_ne`).  It is not a loop, since the walk has
   at least two darts, so its face has at least two darts.
2. Double `d` in its face (`FaceEdgeDoubling`).  The strip lifts (`doubleStrip`), and `d` now spans
   the G-digon `[d, none]`, whose other dart crosses into the face set.
3. Pinch the strip face and the digon at the vertex where the walk leaves along `d`
   (`digonInput`, with `x = alpha (last p)` and `y = alpha none`).  The merged face is a rotation of
   `p ++ [d] ++ none :: q` (`rotate_merged`), the face set is its image, and the walk stays simple
   (`isSimpleClosedWalk_pinch`), so `p ++ [d]` spans a strip (`pinchStrip`).
4. The pinch keeps every dart, so it is a transport (`pinchTransport`).

## Manuscript status

Infrastructure for `thm:hull` (tex 1636, "Hull's small cancellation theorem", through Osin's
Lemma 9.7(b)); certifies no printed sentence on its own.
-/

universe u w v

namespace GroupApproximation.GGT.VanKampen.GeodesicCollar.StripStep

/-- The merged enumeration of the pinch, rotated to start at the head of the side. -/
theorem rotate_merged {α : Type*} (dl q : List α) (a b c : α) :
    ((a :: [b]) ++ (c :: (q ++ dl))).rotate (a :: [b] ++ c :: q).length =
      dl ++ [a] ++ [b] ++ c :: q := by
  rw [← List.cons_append c q dl, ← List.append_assoc, List.rotate_append_length_eq]
  simp

/-- A simple closed walk `p ++ d :: r` enters `d` from the end of the last dart of `p`. -/
theorem vertexOf_alpha_getLast {M : CombMap.{v}} {p r : List M.Dart} {d : M.Dart}
    (hw : IsSimpleClosedWalk M (p ++ d :: r)) (hp : p ≠ []) :
    M.vertexOf (M.alpha (p.getLast hp)) = M.vertexOf d :=
  (List.isChain_append.mp hw.chain).2.2 (p.getLast hp)
    (by rw [List.getLast?_eq_some_getLast hp, Option.mem_def]) d
    (by rw [List.head?_cons, Option.mem_def])

section Strip

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {Delta : DiscDiagram.{u, w, v} W} {S : Finset Delta.toCombMap.Face}
  {p r : List Delta.toCombMap.Dart} {d : Delta.toCombMap.Dart}

/-- The next dart of the walk is not in the strip face. -/
theorem not_mem_face (X : Strip Delta S p (d :: r)) : d ∉ p ++ X.q := by
  have hnodup : (p ++ d :: r).Nodup :=
    List.Nodup.of_map Delta.toCombMap.vertexOf X.walk.vertex_nodup
  have hbd := (X.boundary d).mpr (List.mem_append_right p List.mem_cons_self)
  intro h
  rcases List.mem_append.mp h with h | h
  · exact (List.nodup_append.mp hnodup).2.2 d h d List.mem_cons_self rfl
  · exact hbd.2 (X.alpha_mem d h)

/-- The last dart of the side lies in the strip face. -/
theorem faceOf_getLast (X : Strip Delta S p (d :: r)) :
    Delta.toCombMap.faceOf (p.getLast X.p_ne_nil) =
      Delta.toCombMap.faceOf (p.head X.p_ne_nil) := by
  rw [(X.cycle.mem_iff _).mp (List.mem_append_left _ (List.getLast_mem X.p_ne_nil)),
    List.head_append_of_ne_nil X.p_ne_nil]

/-- The face of the next dart is not the strip face. -/
theorem faceOf_ne (X : Strip Delta S p (d :: r)) :
    Delta.toCombMap.faceOf (p.getLast X.p_ne_nil) ≠ Delta.toCombMap.faceOf d := fun h =>
  not_mem_face X ((X.cycle.mem_iff d).mpr (by
    rw [List.head_append_of_ne_nil X.p_ne_nil, ← faceOf_getLast X, h]))

end Strip

section Pinch

open scoped Classical

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {Delta : DiscDiagram.{u, w, v} W} [DecidableEq Delta.toCombMap.Dart]
  (I : PinchSplit.Input Delta) {s : Finset Delta.toCombMap.Face}

/-- For a face set containing both pinch faces, a face lies in the image exactly when it lay in
the face set. -/
theorem kept_mem_faceSet_iff_of_mem (hl : I.leftFace ∈ s) (hr : I.rightFace ∈ s)
    (g : Delta.toCombMap.Face) : I.kept g ∈ I.faceSet s ↔ g ∈ s := by
  refine ⟨fun h => ?_, I.kept_mem_faceSet⟩
  obtain ⟨g', hg', heq⟩ := Finset.mem_image.mp h
  by_cases h₁ : g = I.leftFace
  · rw [h₁]
    exact hl
  · by_cases h₂ : g = I.rightFace
    · rw [h₂]
      exact hr
    · rw [I.kept_inj h₁ h₂ heq.symm]
      exact hg'

/-- **The boundary of a face set containing both pinch faces is unchanged.** -/
theorem isBoundaryDart_faceSet_iff_of_mem (hl : I.leftFace ∈ s) (hr : I.rightFace ∈ s)
    (e : Delta.toCombMap.Dart) :
    Surgery.MapCollapse.IsBoundaryDart I.diagram.toCombMap (I.faceSet s) e ↔
      Surgery.MapCollapse.IsBoundaryDart Delta.toCombMap s e :=
  and_congr (kept_mem_faceSet_iff_of_mem I hl hr (Delta.toCombMap.faceOf e))
    (not_congr (kept_mem_faceSet_iff_of_mem I hl hr
      (Delta.toCombMap.faceOf (Delta.toCombMap.alpha e))))

/-- **The transport of a pinch** from a face set containing both pinch faces: the identity on
darts. -/
noncomputable def pinchTransport (D : RelGenSet G Lambda)
    (hlabel : ∀ e, D.IsLetter (Delta.label e)) (hl : I.leftFace ∈ s) (hr : I.rightFace ∈ s) :
    Transport D s I.diagram (I.faceSet s) where
  equiv := I.oEquivalent
  embedding := Function.Embedding.refl Delta.toCombMap.Dart
  label_isLetter := I.label_isLetter D hlabel
  alpha_embedding _ := rfl
  label_embedding _ := rfl
  outerDarts_eq := I.outerDarts_eq
  cellDarts_eq i _ := I.cellDarts_eq i
  cell_mem_iff i := by
    have h : Embedded.cell I.diagram (I.oEquivalent.cellIndex i) =
        I.cell (Embedded.cell Delta i) :=
      I.cellMap.indexed_cell i
    rw [h]
    exact kept_mem_faceSet_iff_of_mem I hl hr (Embedded.cell Delta i).face

end Pinch

section Digon

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {Delta : DiscDiagram.{u, w, v} W} [DecidableEq Delta.toCombMap.Dart]
  {S : Finset Delta.toCombMap.Face} {p r : List Delta.toCombMap.Dart}
  {d n : Delta.toCombMap.Dart} (X : Strip Delta S p (d :: r))
  (hdigon : Delta.toCombMap.IsFaceCycle [d, n])

include hdigon in
/-- The digon returns to `d` around its vertex. -/
theorem sigma_alpha_of_digon : Delta.toCombMap.sigma (Delta.toCombMap.alpha n) = d :=
  hdigon.closes

include hdigon in
/-- The two darts of the digon lie on one face. -/
theorem faceOf_digon : Delta.toCombMap.faceOf n = Delta.toCombMap.faceOf d :=
  (hdigon.mem_iff n).mp (List.mem_cons_of_mem d (List.mem_singleton_self n))

include hdigon in
/-- The strip face and the digon, enumerated from their corners at the pinch. -/
noncomputable def digonCycles : PinchSplit.Cycles Delta.toCombMap
    (Delta.toCombMap.alpha (p.getLast X.p_ne_nil)) (Delta.toCombMap.alpha n) where
  xs := X.q ++ p.dropLast
  ys := [d]
  cycX := by
    rw [Delta.toCombMap.alpha_involutive (p.getLast X.p_ne_nil)]
    have e : p ++ X.q = p.dropLast ++ (p.getLast X.p_ne_nil :: X.q) := by
      rw [← List.singleton_append, ← List.append_assoc, List.dropLast_append_getLast]
    have h := X.cycle.rotate p.dropLast.length
    rw [e, List.rotate_append_length_eq, List.cons_append] at h
    exact h
  cycY := by
    rw [Delta.toCombMap.alpha_involutive n]
    exact hdigon.rotate 1
  face_ne := by
    rw [Delta.toCombMap.alpha_involutive (p.getLast X.p_ne_nil),
      Delta.toCombMap.alpha_involutive n, faceOf_digon hdigon]
    exact faceOf_ne X

include hdigon in
/-- **The pinch between the strip face and the digon**, at the vertex where the walk leaves
along `d`. -/
noncomputable def digonInput
    (hnot : ∀ C ∈ Delta.relatorCells, C.face ≠ Delta.toCombMap.faceOf d) :
    PinchSplit.Input Delta where
  x := Delta.toCombMap.alpha (p.getLast X.p_ne_nil)
  y := Delta.toCombMap.alpha n
  same_vertex := by
    have hxd := (Delta.toCombMap.vertexOf_eq_iff _ _).mp
      (vertexOf_alpha_getLast X.walk X.p_ne_nil)
    have hyd : Delta.toCombMap.sigma.SameCycle (Delta.toCombMap.alpha n) d := by
      rw [← sigma_alpha_of_digon hdigon]
      exact Equiv.Perm.SameCycle.rfl.apply_right
    exact hxd.trans hyd.symm
  cycles := digonCycles X hdigon
  left_ne_outer := by
    rw [Delta.toCombMap.alpha_involutive (p.getLast X.p_ne_nil)]
    intro h
    exact X.outerFace_not_mem
      (h ▸ ((X.boundary _).mpr (List.mem_append_left _ (List.getLast_mem X.p_ne_nil))).1)
  right_ne_outer := by
    rw [Delta.toCombMap.alpha_involutive n, faceOf_digon hdigon]
    intro h
    exact X.outerFace_not_mem
      (h ▸ ((X.boundary d).mpr (List.mem_append_right p List.mem_cons_self)).1)
  left_not_cell C hC := by
    rw [Delta.toCombMap.alpha_involutive (p.getLast X.p_ne_nil), faceOf_getLast X]
    exact X.not_cell C hC
  right_not_cell C hC := by
    rw [Delta.toCombMap.alpha_involutive n, faceOf_digon hdigon]
    exact hnot C hC

theorem leftFace_mem (hnot : ∀ C ∈ Delta.relatorCells, C.face ≠ Delta.toCombMap.faceOf d) :
    (digonInput X hdigon hnot).leftFace ∈ S := by
  show Delta.toCombMap.faceOf
    (Delta.toCombMap.alpha (Delta.toCombMap.alpha (p.getLast X.p_ne_nil))) ∈ S
  rw [Delta.toCombMap.alpha_involutive (p.getLast X.p_ne_nil)]
  exact ((X.boundary _).mpr (List.mem_append_left _ (List.getLast_mem X.p_ne_nil))).1

theorem rightFace_mem (hnot : ∀ C ∈ Delta.relatorCells, C.face ≠ Delta.toCombMap.faceOf d) :
    (digonInput X hdigon hnot).rightFace ∈ S := by
  show Delta.toCombMap.faceOf (Delta.toCombMap.alpha (Delta.toCombMap.alpha n)) ∈ S
  rw [Delta.toCombMap.alpha_involutive n, faceOf_digon hdigon]
  exact ((X.boundary d).mpr (List.mem_append_right p List.mem_cons_self)).1

/-- **The strip along `p ++ [d]`** after the pinch. -/
noncomputable def pinchStrip
    (hnot : ∀ C ∈ Delta.relatorCells, C.face ≠ Delta.toCombMap.faceOf d)
    (hn : Delta.toCombMap.faceOf (Delta.toCombMap.alpha n) ∈ S) :
    Strip (digonInput X hdigon hnot).diagram ((digonInput X hdigon hnot).faceSet S) (p ++ [d])
      r where
  q := n :: X.q
  p_ne_nil := by simp
  q_ne_nil := List.cons_ne_nil _ _
  cycle := by
    have h : (digonInput X hdigon hnot).diagram.toCombMap.IsFaceCycle
        ((Delta.toCombMap.alpha (Delta.toCombMap.alpha (p.getLast X.p_ne_nil)) :: [d] ++
          Delta.toCombMap.alpha (Delta.toCombMap.alpha n) :: (X.q ++ p.dropLast)).rotate
          (p.getLast X.p_ne_nil :: [d] ++ n :: X.q).length) :=
      (digonInput X hdigon hnot).cycles.isFaceCycle_merged.rotate _
    rw [Delta.toCombMap.alpha_involutive (p.getLast X.p_ne_nil),
      Delta.toCombMap.alpha_involutive n, rotate_merged,
      List.dropLast_append_getLast X.p_ne_nil] at h
    exact h
  not_cell C hC := by
    have hm : (digonInput X hdigon hnot).kept (Delta.toCombMap.faceOf (p.head X.p_ne_nil)) =
        (digonInput X hdigon hnot).merged := by
      rw [← faceOf_getLast X, ← Delta.toCombMap.alpha_involutive (p.getLast X.p_ne_nil)]
      exact (digonInput X hdigon hnot).kept_leftFace
    rw [List.head_append_of_ne_nil X.p_ne_nil, PinchSplit.Input.diagram_faceOf, hm]
    exact (digonInput X hdigon hnot).merged_not_relatorFace C hC
  alpha_mem e he := (digonInput X hdigon hnot).kept_mem_faceSet (by
    rcases List.mem_cons.mp he with rfl | he
    · exact hn
    · exact X.alpha_mem e he)
  walk := by
    rw [List.append_assoc, List.singleton_append]
    exact isSimpleClosedWalk_pinch (digonInput X hdigon hnot).same_vertex X.p_ne_nil X.walk rfl
      (sigma_alpha_of_digon hdigon)
  boundary e := by
    rw [List.append_assoc, List.singleton_append]
    exact (isBoundaryDart_faceSet_iff_of_mem _ (leftFace_mem X hdigon hnot)
      (rightFace_mem X hdigon hnot) e).trans (X.boundary e)
  outerFace_not_mem h := X.outerFace_not_mem
    ((kept_mem_faceSet_iff_of_mem _ (leftFace_mem X hdigon hnot) (rightFace_mem X hdigon hnot)
      Delta.outerFace).mp h)

include X hdigon in
/-- **The pinch step at a digon.**  A strip along `p` whose remaining walk starts with `d`, where
`d` spans a G-digon `[d, n]` whose other dart crosses into the face set, has a transport to a strip
along `p ++ [d]`. -/
theorem stripStep_of_digon (D : RelGenSet G Lambda) (hlabel : ∀ e, D.IsLetter (Delta.label e))
    (hnot : ∀ C ∈ Delta.relatorCells, C.face ≠ Delta.toCombMap.faceOf d)
    (hn : Delta.toCombMap.faceOf (Delta.toCombMap.alpha n) ∈ S) :
    ∃ (Delta' : DiscDiagram.{u, w, v} W) (S' : Finset Delta'.toCombMap.Face)
      (T : Transport D S Delta' S'),
      Nonempty (Strip Delta' S' ((p ++ [d]).map T.embedding) (r.map T.embedding)) := by
  refine ⟨(digonInput X hdigon hnot).diagram, (digonInput X hdigon hnot).faceSet S,
    pinchTransport (digonInput X hdigon hnot) D hlabel (leftFace_mem X hdigon hnot)
      (rightFace_mem X hdigon hnot), ⟨?_⟩⟩
  have e₁ : (p ++ [d]).map (pinchTransport (digonInput X hdigon hnot) D hlabel
      (leftFace_mem X hdigon hnot) (rightFace_mem X hdigon hnot)).embedding = p ++ [d] :=
    List.map_id _
  have e₂ : r.map (pinchTransport (digonInput X hdigon hnot) D hlabel
      (leftFace_mem X hdigon hnot) (rightFace_mem X hdigon hnot)).embedding = r :=
    List.map_id _
  rw [e₁, e₂]
  exact pinchStrip X hdigon hnot hn

end Digon

section Double

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  (Delta : DiscDiagram.{u, w, v} W) (f : Delta.toCombMap.Face)
  (j : Fin (Delta.faceBoundary f).darts.length)
  (hlen : 1 < (Delta.faceBoundary f).darts.length)

/-- A face cycle off the doubled face lifts to the doubled map. -/
theorem isFaceCycle_embed {l : List Delta.toCombMap.Dart} (hl : Delta.toCombMap.IsFaceCycle l)
    (hne : Delta.toCombMap.faceOf (l.head hl.ne_nil) ≠ f) :
    (FaceEdgeDoubling.map Delta f j hlen).IsFaceCycle
      (l.map (EdgeInsertion.embed Delta.toCombMap)) := by
  have hstep : ∀ e ∈ l, (FaceEdgeDoubling.map Delta f j hlen).facePerm
      (EdgeInsertion.embed Delta.toCombMap e) =
        EdgeInsertion.embed Delta.toCombMap (Delta.toCombMap.facePerm e) := fun e he =>
    EdgeInsertion.facePerm_embed_of_ne Delta.toCombMap _ _ e
      (EdgeInsertion.corners_face_eq Delta.toCombMap (FaceEdgeDoubling.rebased Delta f j)
        (FaceEdgeDoubling.second Delta f j hlen))
      (by
        rw [EdgeInsertion.firstCorner_face Delta.toCombMap (FaceEdgeDoubling.rebased Delta f j),
          (hl.mem_iff e).mp he]
        exact hne)
  refine ⟨by simpa using hl.ne_nil, hl.nodup.map (EdgeInsertion.embed_injective Delta.toCombMap),
    ?_, ?_⟩
  · exact (List.isChain_map _).mpr (hl.chain.imp_of_mem_imp fun a b ha _ h => by
      show (FaceEdgeDoubling.map Delta f j hlen).facePerm (EdgeInsertion.embed Delta.toCombMap a) =
        EdgeInsertion.embed Delta.toCombMap b
      rw [hstep a ha, h])
  · rw [List.getLast_map, List.head_map, hstep _ (List.getLast_mem hl.ne_nil), hl.closes]

variable {Delta f j} {S : Finset Delta.toCombMap.Face} {p r : List Delta.toCombMap.Dart}

/-- The strip face is not the face of the doubled dart. -/
theorem faceOf_head_ne (X : Strip Delta S p (FaceEdgeDoubling.dart Delta f j :: r)) :
    Delta.toCombMap.faceOf (p.head X.p_ne_nil) ≠ f := fun h =>
  faceOf_ne X (by rw [faceOf_getLast X, h, FaceEdgeDoubling.dart_face Delta f j])

variable (hf : f ≠ Delta.outerFace)

/-- **The strip after doubling the next dart**, along the images of the old darts. -/
noncomputable def doubleStrip (X : Strip Delta S p (FaceEdgeDoubling.dart Delta f j :: r))
    (hS : f ∈ S) :
    Strip (FaceEdgeDoubling.diagram Delta f j hlen hf)
      (FaceEdgeDoubling.collarFaceSet Delta f j hlen S)
      (p.map (EdgeInsertion.embed Delta.toCombMap))
      (EdgeInsertion.embed Delta.toCombMap (FaceEdgeDoubling.dart Delta f j) ::
        r.map (EdgeInsertion.embed Delta.toCombMap)) where
  q := X.q.map (EdgeInsertion.embed Delta.toCombMap)
  p_ne_nil := by simpa using X.p_ne_nil
  q_ne_nil := by simpa using X.q_ne_nil
  cycle := by
    rw [← List.map_append]
    refine isFaceCycle_embed Delta f j hlen X.cycle fun h =>
      not_mem_face X ((X.cycle.mem_iff _).mpr ?_)
    rw [FaceEdgeDoubling.dart_face Delta f j, h]
  not_cell C hC := by
    change C ∈ Delta.relatorCells.map (FaceEdgeDoubling.cell Delta f j hlen hf) at hC
    obtain ⟨C₀, hC₀, rfl⟩ := List.mem_map.mp hC
    rw [List.head_map, FaceEdgeDoubling.faceOf_embed_of_face_ne Delta f j hlen (faceOf_head_ne X)]
    exact fun h => X.not_cell C₀ hC₀ (FaceEdgeDoubling.faceImage_injective Delta f j hlen h)
  alpha_mem e he := by
    obtain ⟨e₀, he₀, rfl⟩ := List.mem_map.mp he
    exact (FaceEdgeDoubling.faceOf_embed_mem_collarFaceSet_iff Delta f j hlen hS
      (Delta.toCombMap.alpha e₀)).mpr (X.alpha_mem e₀ he₀)
  walk := by
    have h := FaceEdgeDoubling.isSimpleClosedWalk_embed Delta f j hlen X.walk
    rwa [List.map_append, List.map_cons] at h
  boundary x := by
    have h := FaceEdgeDoubling.isBoundaryDart_collarFaceSet_iff Delta f j hlen hS X.boundary x
    rwa [List.map_append, List.map_cons] at h
  outerFace_not_mem :=
    FaceEdgeDoubling.outerFace_not_mem_collarFaceSet Delta f j hlen hf X.outerFace_not_mem

end Double

/-- **The pinch step.**  Double the next dart `d` in its face, then pinch the digon into the strip
face. -/
theorem stripStepStatement_holds : StripStepStatement.{u, w, v} := by
  intro G _ Lambda W D hsymm Delta hlabel S p d r X
  classical
  have hdmem : d ∈ p ++ d :: r := List.mem_append_right p List.mem_cons_self
  have hS : Delta.toCombMap.faceOf d ∈ S := ((X.boundary d).mpr hdmem).1
  have hlen2 : 2 ≤ (p ++ d :: r).length := by
    have := List.length_pos_iff.mpr X.p_ne_nil
    rw [List.length_append, List.length_cons]
    omega
  have hmono : (Delta.faceBoundary (Delta.toCombMap.faceOf d)).darts.length ≠ 1 := fun h =>
    vertexOf_alpha_ne_of_mem X.walk hdmem hlen2 (vertexOf_alpha_of_monogon _ h rfl)
  have hpos := List.length_pos_of_mem
    (((Delta.faceBoundary (Delta.toCombMap.faceOf d)).mem_iff d).mpr rfl)
  have hlen : 1 < (Delta.faceBoundary (Delta.toCombMap.faceOf d)).darts.length := by omega
  have hf : Delta.toCombMap.faceOf d ≠ Delta.outerFace := fun h =>
    X.outerFace_not_mem (by rw [← h]; exact hS)
  generalize hfd : Delta.toCombMap.faceOf d = f at hS hlen hf
  obtain ⟨j, hj⟩ := List.mem_iff_get.mp (((Delta.faceBoundary f).mem_iff d).mpr hfd)
  rw [← FaceEdgeDoubling.dart_eq_get Delta f j] at hj
  subst hj
  obtain ⟨Delta₂, S₂, T₂, ⟨X₂⟩⟩ := stripStep_of_digon (doubleStrip hlen hf X hS)
    (FaceEdgeDoubling.isFaceCycle_collarDigon Delta f j hlen) D
    (FaceEdgeDoubling.label_admissible Delta f j hlen hf D hsymm hlabel)
    (fun C hC h => FaceEdgeDoubling.digon_not_collarCell Delta f j hlen hf C hC
      (h.trans (FaceEdgeDoubling.faceOf_embed_dart_digon Delta f j hlen)))
    (by
      change (FaceEdgeDoubling.map Delta f j hlen).faceOf (some none) ∈
        FaceEdgeDoubling.collarFaceSet Delta f j hlen S
      rw [FaceEdgeDoubling.faceOf_some_none_cellFace]
      exact FaceEdgeDoubling.cellFace_mem_collarFaceSet Delta f j hlen hS)
  let T₁ := FaceEdgeDoubling.collarTransport Delta f j hlen hf D hsymm hlabel hS
  refine ⟨Delta₂, S₂, T₁.trans T₂, ⟨?_⟩⟩
  have e₁ : (p.map (EdgeInsertion.embed Delta.toCombMap) ++
      [EdgeInsertion.embed Delta.toCombMap (FaceEdgeDoubling.dart Delta f j)]).map T₂.embedding =
      (p ++ [FaceEdgeDoubling.dart Delta f j]).map (T₁.trans T₂).embedding := by
    simp only [List.map_append, List.map_map, List.map_cons, List.map_nil]
    rfl
  have e₂ : (r.map (EdgeInsertion.embed Delta.toCombMap)).map T₂.embedding =
      r.map (T₁.trans T₂).embedding := by
    rw [List.map_map]
    rfl
  rw [← e₁, ← e₂]
  exact X₂

/-- **Stage 1 of the collar insertion**, the strip, from the pinch step. -/
theorem stripStatement_holds : StripStatement.{u, w, v} :=
  stripStatement_of_step stripStepStatement_holds

/-- **The collar insertion** (`GeodesicCollarStatement`, binder 8 of the W1 waist), closed. -/
theorem geodesicCollarStatement_holds : GeodesicCollarStatement.{u, w, v} :=
