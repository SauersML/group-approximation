/-!
# The Case 1 one-cell pocket walks are noncrossing

Osin (math/0411039v3, §9), Lemma 9.4, Case 1, for a backwards connector pair whose source and
target sides lie across one relator cell `Π` (`OsinLemma94CaseOneSameCellStatement`).  Some
rotation of the face walk of `f` reads `X q⁻¹ Y p⁻¹`, and some rotation of the carrier of `Π` reads
`q B p A`, with `f ≠ Π`.  The pocket walk `invDarts X ++ invDarts B` is the reversed walk of the
complement spelling `c = B ++ X`, and `c` is a first-turn walk (`FirstTurn`):

* inside `B` and inside `X` consecutive darts are face steps;
* from the last dart of `B` the rotation passes the first dart of `p` and reaches the first dart of
  `X`, and from the last dart of `X` it passes the reversal of the last dart of `q` and reaches the
  first dart of `B`; both darts passed lie off the edges of `c`
  (`not_walkKeep_head_p`, `not_walkKeep_alpha_getLast_q`);
* when `X` or `B` is empty the two junctions merge into one turn of three steps
  (`firstTurn_of_three`).

So a pocket walk with no dart meeting its reversal is noncrossing, its outer cycle follows its
boundary, and the reclosed map of its side has the Euler characteristic of the map
(`firstTurnWalkPocketInputs`).

## Manuscript status

Infrastructure for `thm:hull` (tex 2121, Hull's small cancellation theorem, through Osin's
Lemma 9.4, Case 1); certifies no printed sentence on its own.
-/

/-- **A junction of a face cycle.**  If a rotation of a face cycle reads `a ++ (b ++ rest)` with
`a` and `b` nonempty, the face permutation takes the last dart of `a` to the first dart of `b`. -/
theorem facePerm_getLast_of_rotate_eq {M : CombMap.{v}} {L : List M.Dart} (hL : M.IsFaceCycle L)
    {m : ℕ} {a b rest : List M.Dart} (hrot : L.rotate m = a ++ (b ++ rest)) (ha : a ≠ [])
    (hb : b ≠ []) : M.facePerm (a.getLast ha) = b.head hb := by
  have hchain := (hL.rotate m).chain
  rw [hrot] at hchain
  refine (List.isChain_append.mp hchain).2.2 _ ?_ _ ?_
  · rw [List.getLast?_eq_some_getLast ha, Option.mem_def]
  · rw [List.head?_append, List.head?_eq_some_head hb, Option.some_or, Option.mem_def]

/-- **Three rotation steps past two darts off the edges of `c` are a first turn.** -/
theorem firstTurn_of_three {M : CombMap.{v}} {c : List M.Dart} {x y z₁ z₂ : M.Dart}
    (h₁ : M.sigma (M.alpha x) = z₁) (hz₁ : ¬ walkKeep M c z₁) (h₂ : M.sigma z₁ = z₂)
    (hz₂ : ¬ walkKeep M c z₂) (h₃ : M.sigma z₂ = y) : FirstTurn M c x y := by
  refine ⟨3, by norm_num, ?_, fun t ht htk => ?_⟩
  · rw [pow_three, Perm.mul_apply, Perm.mul_apply, h₁, h₂, h₃]
  · interval_cases t
    · rw [pow_one, h₁]
      exact hz₁
    · rw [pow_two, Perm.mul_apply, h₁, h₂]
      exact hz₂

theorem reverseDarts_ne_nil {M : CombMap.{v}} {cycle : List M.Dart} (arc : CyclicArc cycle)
    (h : 0 < arc.length) : arc.reverseDarts ≠ [] := by
  simpa [CyclicArc.reverseDarts] using arc.darts_ne_nil_of_length_pos h

theorem reverseDarts_head {M : CombMap.{v}} {cycle : List M.Dart} (arc : CyclicArc cycle)
    (h : arc.reverseDarts ≠ []) (hd : arc.darts ≠ []) :
    arc.reverseDarts.head h = M.alpha (arc.darts.getLast hd) := by
  simp only [CyclicArc.reverseDarts, List.head_map, List.head_reverse]

theorem reverseDarts_getLast {M : CombMap.{v}} {cycle : List M.Dart} (arc : CyclicArc cycle)
    (h : arc.reverseDarts ≠ []) (hd : arc.darts ≠ []) :
    arc.reverseDarts.getLast h = M.alpha (arc.darts.head hd) := by
  simp only [CyclicArc.reverseDarts, List.getLast_map, List.getLast_reverse]

end Lists

variable {G : Type u} [Group G] {Lambda : Type w} {W : Set (List (RelLetter G Lambda))}
  {Delta : DiscDiagram.{u, w, v} W} {f : Delta.toCombMap.Face} {j : Fin Delta.rCellCount}
  {X Y B A : List Delta.toCombMap.Dart} {r n : ℕ}

/-- The face walk after `X` starts with the reversal of the last dart of `q`. -/
theorem facePerm_getLast_X (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (htrav : (Delta.faceBoundary f).darts.rotate r =
      X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts)
    (ht : 0 < targetArc.length) (hX : X ≠ []) :
    Delta.toCombMap.facePerm (X.getLast hX) =
      Delta.toCombMap.alpha (targetArc.darts.getLast (targetArc.darts_ne_nil_of_length_pos ht)) := by
  have hrot : (Delta.faceBoundary f).darts.rotate r =
      X ++ (targetArc.reverseDarts ++ (Y ++ sourceArc.reverseDarts)) := by
    rw [htrav]
    simp only [List.append_assoc]
  rw [facePerm_getLast_of_rotate_eq (Delta.faceBoundary f).isFaceCycle hrot hX
    (reverseDarts_ne_nil targetArc ht)]
  exact reverseDarts_head targetArc _ _

/-- The face walk after the reversal of `p` starts with `X`. -/
theorem facePerm_alpha_head_p (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (htrav : (Delta.faceBoundary f).darts.rotate r =
      X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts)
    (hs : 0 < sourceArc.length) (hX : X ≠ []) :
    Delta.toCombMap.facePerm
        (Delta.toCombMap.alpha (sourceArc.darts.head (sourceArc.darts_ne_nil_of_length_pos hs))) =
      X.head hX := by
  have hrot : (Delta.faceBoundary f).darts.rotate (r + (X ++ targetArc.reverseDarts ++ Y).length) =
      sourceArc.reverseDarts ++ (X ++ (targetArc.reverseDarts ++ Y)) := by
    rw [← List.rotate_rotate, htrav, List.rotate_append_length_eq]
    simp only [List.append_assoc]
  have h := facePerm_getLast_of_rotate_eq (Delta.faceBoundary f).isFaceCycle hrot
    (reverseDarts_ne_nil sourceArc hs) hX
  rwa [reverseDarts_getLast sourceArc _ (sourceArc.darts_ne_nil_of_length_pos hs)] at h

/-- With `X` empty, the face walk after the reversal of `p` starts with the reversal of `q`. -/
theorem facePerm_alpha_head_p_of_nil (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (htrav : (Delta.faceBoundary f).darts.rotate r =
      X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts)
    (hs : 0 < sourceArc.length) (ht : 0 < targetArc.length) (hX : X = []) :
    Delta.toCombMap.facePerm
        (Delta.toCombMap.alpha (sourceArc.darts.head (sourceArc.darts_ne_nil_of_length_pos hs))) =
      Delta.toCombMap.alpha (targetArc.darts.getLast (targetArc.darts_ne_nil_of_length_pos ht)) := by
  subst hX
  have hrot : (Delta.faceBoundary f).darts.rotate (r + (targetArc.reverseDarts ++ Y).length) =
      sourceArc.reverseDarts ++ (targetArc.reverseDarts ++ Y) := by
    rw [← List.rotate_rotate, htrav, List.nil_append, List.rotate_append_length_eq]
  have h := facePerm_getLast_of_rotate_eq (Delta.faceBoundary f).isFaceCycle hrot
    (reverseDarts_ne_nil sourceArc hs) (reverseDarts_ne_nil targetArc ht)
  rwa [reverseDarts_getLast sourceArc _ (sourceArc.darts_ne_nil_of_length_pos hs),
    reverseDarts_head targetArc _ (targetArc.darts_ne_nil_of_length_pos ht)] at h

/-- The carrier after `q` starts with `B`. -/
theorem facePerm_getLast_q (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (hPi : (cellDarts Delta j).rotate n = targetArc.darts ++ B ++ sourceArc.darts ++ A)
    (ht : 0 < targetArc.length) (hB : B ≠ []) :
    Delta.toCombMap.facePerm (targetArc.darts.getLast (targetArc.darts_ne_nil_of_length_pos ht)) =
      B.head hB := by
  have hrot : (cellDarts Delta j).rotate n = targetArc.darts ++ (B ++ (sourceArc.darts ++ A)) := by
    rw [hPi]
    simp only [List.append_assoc]
  exact facePerm_getLast_of_rotate_eq (Delta.faceBoundary (cell Delta j).face).isFaceCycle hrot
    (targetArc.darts_ne_nil_of_length_pos ht) hB

/-- With `B` empty, the carrier after `q` starts with `p`. -/
theorem facePerm_getLast_q_of_nil (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (hPi : (cellDarts Delta j).rotate n = targetArc.darts ++ B ++ sourceArc.darts ++ A)
    (hs : 0 < sourceArc.length) (ht : 0 < targetArc.length) (hB : B = []) :
    Delta.toCombMap.facePerm (targetArc.darts.getLast (targetArc.darts_ne_nil_of_length_pos ht)) =
      sourceArc.darts.head (sourceArc.darts_ne_nil_of_length_pos hs) := by
  subst hB
  have hrot : (cellDarts Delta j).rotate n = targetArc.darts ++ (sourceArc.darts ++ A) := by
    rw [hPi]
    simp only [List.append_nil, List.append_assoc]
  exact facePerm_getLast_of_rotate_eq (Delta.faceBoundary (cell Delta j).face).isFaceCycle hrot
    (targetArc.darts_ne_nil_of_length_pos ht) (sourceArc.darts_ne_nil_of_length_pos hs)

/-- The carrier after `B` starts with `p`. -/
theorem facePerm_getLast_B (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (hPi : (cellDarts Delta j).rotate n = targetArc.darts ++ B ++ sourceArc.darts ++ A)
    (hs : 0 < sourceArc.length) (hB : B ≠ []) :
    Delta.toCombMap.facePerm (B.getLast hB) =
      sourceArc.darts.head (sourceArc.darts_ne_nil_of_length_pos hs) := by
  have hrot : (cellDarts Delta j).rotate (n + targetArc.darts.length) =
      B ++ (sourceArc.darts ++ (A ++ targetArc.darts)) := by
    rw [← List.rotate_rotate, hPi]
    simp only [List.append_assoc]
    rw [List.rotate_append_length_eq]
    simp only [List.append_assoc]
  exact facePerm_getLast_of_rotate_eq (Delta.faceBoundary (cell Delta j).face).isFaceCycle hrot hB
    (sourceArc.darts_ne_nil_of_length_pos hs)

/-- **The first dart of `p` is off the edges of `B ++ X`.** -/
theorem not_walkKeep_head_p (hf : (cell Delta j).face ≠ f)
    (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (htrav : (Delta.faceBoundary f).darts.rotate r =
      X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts)
    (hPi : (cellDarts Delta j).rotate n = targetArc.darts ++ B ++ sourceArc.darts ++ A)
    (hs : 0 < sourceArc.length) :
    ¬ walkKeep Delta.toCombMap (B ++ X)
      (sourceArc.darts.head (sourceArc.darts_ne_nil_of_length_pos hs)) := by
  have hne := sourceArc.darts_ne_nil_of_length_pos hs
  have hwalk : (X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts).Nodup := by
    rw [← htrav]
    exact List.nodup_rotate.mpr (Delta.faceBoundary f).nodup
  have hcarrier : (targetArc.darts ++ B ++ sourceArc.darts ++ A).Nodup := by
    rw [← hPi]
    exact List.nodup_rotate.mpr (Delta.faceBoundary (cell Delta j).face).nodup
  have hp : sourceArc.darts.head hne ∈ sourceArc.darts := List.head_mem hne
  have hpface : Delta.toCombMap.faceOf (sourceArc.darts.head hne) = (cell Delta j).face :=
    SameCellPocketSides.faceOf_of_mem_cellDarts_rotate hPi
      (List.mem_append_left _ (List.mem_append_right _ hp))
  have hαp : Delta.toCombMap.alpha (sourceArc.darts.head hne) ∈ sourceArc.reverseDarts :=
    List.mem_map.mpr ⟨_, List.mem_reverse.mpr hp, rfl⟩
  have hαface : Delta.toCombMap.faceOf (Delta.toCombMap.alpha (sourceArc.darts.head hne)) = f :=
    SameCellPocketSides.faceOf_of_mem_rotate htrav (List.mem_append_right _ hαp)
  rintro (h | h)
  · rcases List.mem_append.mp h with hB | hX
    · exact (SameCellPocketSides.nodup_append_four hcarrier).1 _ (List.mem_append_right _ hB) _
        hp rfl
    · exact hf (hpface.symm.trans (SameCellPocketSides.faceOf_of_mem_rotate htrav
        (List.mem_append_left _ (List.mem_append_left _ (List.mem_append_left _ hX)))))
  · rcases List.mem_append.mp h with hB | hX
    · exact hf ((SameCellPocketSides.faceOf_of_mem_cellDarts_rotate hPi
        (List.mem_append_left _ (List.mem_append_left _ (List.mem_append_right _ hB)))).symm.trans
        hαface)
    · exact (SameCellPocketSides.nodup_append_four hwalk).2 _
        (List.mem_append_left _ (List.mem_append_left _ (List.mem_append_left _ hX))) _ hαp rfl

/-- **The reversal of the last dart of `q` is off the edges of `B ++ X`.** -/
theorem not_walkKeep_alpha_getLast_q (hf : (cell Delta j).face ≠ f)
    (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (htrav : (Delta.faceBoundary f).darts.rotate r =
      X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts)
    (hPi : (cellDarts Delta j).rotate n = targetArc.darts ++ B ++ sourceArc.darts ++ A)
    (ht : 0 < targetArc.length) :
    ¬ walkKeep Delta.toCombMap (B ++ X)
      (Delta.toCombMap.alpha (targetArc.darts.getLast (targetArc.darts_ne_nil_of_length_pos ht))) := by
  have hne := targetArc.darts_ne_nil_of_length_pos ht
  have hwalk : (X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts).Nodup := by
    rw [← htrav]
    exact List.nodup_rotate.mpr (Delta.faceBoundary f).nodup
  have hcarrier : (targetArc.darts ++ B ++ sourceArc.darts ++ A).Nodup := by
    rw [← hPi]
    exact List.nodup_rotate.mpr (Delta.faceBoundary (cell Delta j).face).nodup
  have hq : targetArc.darts.getLast hne ∈ targetArc.darts := List.getLast_mem hne
  have hqface : Delta.toCombMap.faceOf (targetArc.darts.getLast hne) = (cell Delta j).face :=
    SameCellPocketSides.faceOf_of_mem_cellDarts_rotate hPi
      (List.mem_append_left _ (List.mem_append_left _ (List.mem_append_left _ hq)))
  have hαq : Delta.toCombMap.alpha (targetArc.darts.getLast hne) ∈ targetArc.reverseDarts :=
    List.mem_map.mpr ⟨_, List.mem_reverse.mpr hq, rfl⟩
  have hαface : Delta.toCombMap.faceOf (Delta.toCombMap.alpha (targetArc.darts.getLast hne)) = f :=
    SameCellPocketSides.faceOf_of_mem_rotate htrav
      (List.mem_append_left _ (List.mem_append_left _ (List.mem_append_right _ hαq)))
  rintro (h | h)
  · rcases List.mem_append.mp h with hB | hX
    · exact hf ((SameCellPocketSides.faceOf_of_mem_cellDarts_rotate hPi
        (List.mem_append_left _ (List.mem_append_left _ (List.mem_append_right _ hB)))).symm.trans
        hαface)
    · have hXq : (X ++ targetArc.reverseDarts).Nodup :=
        (List.nodup_append.mp (List.nodup_append.mp hwalk).1).1
      exact (List.nodup_append.mp hXq).2.2 _ hX _ hαq rfl
  · rw [Delta.toCombMap.alpha_involutive (targetArc.darts.getLast hne)] at h
    rcases List.mem_append.mp h with hB | hX
    · have hqB : (targetArc.darts ++ B).Nodup :=
        (List.nodup_append.mp (List.nodup_append.mp hcarrier).1).1
      exact (List.nodup_append.mp hqB).2.2 _ hq _ hB rfl
    · exact hf (hqface.symm.trans (SameCellPocketSides.faceOf_of_mem_rotate htrav
        (List.mem_append_left _ (List.mem_append_left _ (List.mem_append_left _ hX)))))

/-- **The complement spelling `B ++ X` is chained by first turns.** -/
theorem firstTurn_chain_X (hf : (cell Delta j).face ≠ f)
    (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (htrav : (Delta.faceBoundary f).darts.rotate r =
      X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts)
    (hPi : (cellDarts Delta j).rotate n = targetArc.darts ++ B ++ sourceArc.darts ++ A)
    (hs : 0 < sourceArc.length) :
    (B ++ X).IsChain (FirstTurn Delta.toCombMap (B ++ X)) := by
  have hwalkChain := ((Delta.faceBoundary f).isFaceCycle.rotate r).chain
  have hcarrierChain := ((Delta.faceBoundary (cell Delta j).face).isFaceCycle.rotate n).chain
  change ((cellDarts Delta j).rotate n).IsChain _ at hcarrierChain
  rw [htrav] at hwalkChain
  rw [hPi] at hcarrierChain
  have hB : B.IsChain (FirstTurn Delta.toCombMap (B ++ X)) :=
    (hcarrierChain.infix ⟨targetArc.darts, sourceArc.darts ++ A, by
      simp only [List.append_assoc]⟩).imp fun _ _ h => FirstTurn.of_facePerm h
  have hX : X.IsChain (FirstTurn Delta.toCombMap (B ++ X)) :=
    (hwalkChain.infix ⟨[], targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts, by
      simp only [List.nil_append, List.append_assoc]⟩).imp fun _ _ h => FirstTurn.of_facePerm h
  refine List.isChain_append.mpr ⟨hB, hX, fun a ha b hb => ?_⟩
  have hBne : B ≠ [] := fun h => by simp [h] at ha
  have hXne : X ≠ [] := fun h => by simp [h] at hb
  rw [List.getLast?_eq_some_getLast hBne, Option.mem_def, Option.some.injEq] at ha
  rw [List.head?_eq_some_head hXne, Option.mem_def, Option.some.injEq] at hb
  subst ha hb
  refine FirstTurn.of_sigma_sigma_alpha ?_ ?_
  · show ¬ walkKeep Delta.toCombMap (B ++ X) (Delta.toCombMap.facePerm (B.getLast hBne))
    rw [facePerm_getLast_B sourceArc targetArc hPi hs hBne]
    exact not_walkKeep_head_p hf sourceArc targetArc htrav hPi hs
  · show Delta.toCombMap.sigma (Delta.toCombMap.facePerm (B.getLast hBne)) = X.head hXne
    rw [facePerm_getLast_B sourceArc targetArc hPi hs hBne]
    have h := facePerm_alpha_head_p sourceArc targetArc htrav hs hXne
    rwa [CombMap.facePerm, Perm.mul_apply,
      Delta.toCombMap.alpha_involutive (sourceArc.darts.head (sourceArc.darts_ne_nil_of_length_pos hs))] at h

/-- **The last dart of `B ++ X` turns first to its first dart.** -/
theorem firstTurn_close_X (hf : (cell Delta j).face ≠ f)
    (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (htrav : (Delta.faceBoundary f).darts.rotate r =
      X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts)
    (hPi : (cellDarts Delta j).rotate n = targetArc.darts ++ B ++ sourceArc.darts ++ A)
    (hs : 0 < sourceArc.length) (ht : 0 < targetArc.length) (hne : B ++ X ≠ []) :
    FirstTurn Delta.toCombMap (B ++ X) ((B ++ X).getLast hne) ((B ++ X).head hne) := by
  have hp := sourceArc.darts_ne_nil_of_length_pos hs
  have hq := targetArc.darts_ne_nil_of_length_pos ht
  by_cases hXne : X = []
  · subst hXne
    have hBne : B ≠ [] := by simpa using hne
    have e₁ : (B ++ []).getLast hne = B.getLast hBne := by simp
    have e₂ : (B ++ []).head hne = B.head hBne := by simp
    rw [e₁, e₂]
    have h₂ := facePerm_alpha_head_p_of_nil sourceArc targetArc htrav hs ht rfl
    rw [CombMap.facePerm, Perm.mul_apply, Delta.toCombMap.alpha_involutive (sourceArc.darts.head hp)] at h₂
    exact firstTurn_of_three (facePerm_getLast_B sourceArc targetArc hPi hs hBne)
      (not_walkKeep_head_p hf sourceArc targetArc htrav hPi hs) h₂
      (not_walkKeep_alpha_getLast_q hf sourceArc targetArc htrav hPi ht)
      (facePerm_getLast_q sourceArc targetArc hPi ht hBne)
  by_cases hBne : B = []
  · subst hBne
    have e₁ : ([] ++ X).getLast hne = X.getLast hXne := by simp
    have e₂ : ([] ++ X).head hne = X.head hXne := by simp
    rw [e₁, e₂]
    have h₃ := facePerm_alpha_head_p sourceArc targetArc htrav hs hXne
    rw [CombMap.facePerm, Perm.mul_apply, Delta.toCombMap.alpha_involutive (sourceArc.darts.head hp)] at h₃
    exact firstTurn_of_three (facePerm_getLast_X sourceArc targetArc htrav ht hXne)
      (not_walkKeep_alpha_getLast_q hf sourceArc targetArc htrav hPi ht)
      (facePerm_getLast_q_of_nil sourceArc targetArc hPi hs ht rfl)
      (not_walkKeep_head_p hf sourceArc targetArc htrav hPi hs) h₃
  rw [List.getLast_append_of_ne_nil hne hXne, List.head_append_of_ne_nil hBne]
  refine FirstTurn.of_sigma_sigma_alpha ?_ ?_
  · show ¬ walkKeep Delta.toCombMap (B ++ X) (Delta.toCombMap.facePerm (X.getLast hXne))
    rw [facePerm_getLast_X sourceArc targetArc htrav ht hXne]
    exact not_walkKeep_alpha_getLast_q hf sourceArc targetArc htrav hPi ht
  · show Delta.toCombMap.sigma (Delta.toCombMap.facePerm (X.getLast hXne)) = B.head hBne
    rw [facePerm_getLast_X sourceArc targetArc htrav ht hXne]
    exact facePerm_getLast_q sourceArc targetArc hPi ht hBne

/-- **The X-pocket walk without spur is noncrossing, its outer cycle follows its boundary, and the
reclosed map of its side has the Euler characteristic of the map.** -/
theorem pocketInputs_X (hf : (cell Delta j).face ≠ f)
    (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (htrav : (Delta.faceBoundary f).darts.rotate r =
      X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts)
    (hPi : (cellDarts Delta j).rotate n = targetArc.darts ++ B ++ sourceArc.darts ++ A)
    (hs : 0 < sourceArc.length) (ht : 0 < targetArc.length) (hne : X ++ B ≠ [])
    (hnospur : ∀ d ∈ invDarts Delta X ++ invDarts Delta B,
      Delta.toCombMap.alpha d ∉ invDarts Delta X ++ invDarts Delta B) :
    ∃ hw : IsNoncrossingClosedWalk Delta.toCombMap (invDarts Delta X ++ invDarts Delta B),
      (hw.outerCycle Delta.planar).FollowsBoundary ∧
        (reclosedMap Delta.toCombMap
          (sideFaces Delta.toCombMap (invDarts Delta X ++ invDarts Delta B))
          (hw.innerCycle Delta.planar)).eulerCharacteristic = Delta.toCombMap.eulerCharacteristic := by
  have hc : B ++ X ≠ [] := fun h => hne (by
    obtain ⟨h₁, h₂⟩ := List.append_eq_nil_iff.mp h
    rw [h₁, h₂]
    rfl)
  have hwalk : (X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts).Nodup := by
    rw [← htrav]
    exact List.nodup_rotate.mpr (Delta.faceBoundary f).nodup
  have hcarrier : (targetArc.darts ++ B ++ sourceArc.darts ++ A).Nodup := by
    rw [← hPi]
    exact List.nodup_rotate.mpr (Delta.faceBoundary (cell Delta j).face).nodup
  have hXnd : X.Nodup :=
    (List.nodup_append.mp (List.nodup_append.mp (List.nodup_append.mp hwalk).1).1).1
  have hBnd : B.Nodup :=
    (List.nodup_append.mp (List.nodup_append.mp (List.nodup_append.mp hcarrier).1).1).2.1
  have hnodup : (B ++ X).Nodup := List.nodup_append.mpr ⟨hBnd, hXnd, fun a ha b hb hab => by
    subst hab
    exact hf ((SameCellPocketSides.faceOf_of_mem_cellDarts_rotate hPi
      (List.mem_append_left _ (List.mem_append_left _ (List.mem_append_right _ ha)))).symm.trans
      (SameCellPocketSides.faceOf_of_mem_rotate htrav
        (List.mem_append_left _ (List.mem_append_left _ (List.mem_append_left _ hb)))))⟩
  have e : invDarts Delta (B ++ X) = invDarts Delta X ++ invDarts Delta B := invDarts_append B X
  have hmem : ∀ z ∈ B ++ X,
      Delta.toCombMap.alpha z ∈ invDarts Delta X ++ invDarts Delta B := fun z hz => by
    rw [← e]
    exact List.mem_map.mpr ⟨z, List.mem_reverse.mpr hz, rfl⟩
  have halpha : ∀ x ∈ B ++ X, Delta.toCombMap.alpha x ∉ B ++ X := by
    intro x hx hax
    refine hnospur (Delta.toCombMap.alpha x) (hmem x hx) ?_
    have h := hmem _ hax
    rwa [Delta.toCombMap.alpha_involutive x] at h ⊢
  have h := firstTurnWalkPocketInputs Delta.toCombMap Delta.planar (B ++ X) hc hnodup halpha
    (firstTurn_chain_X hf sourceArc targetArc htrav hPi hs)
    (firstTurn_close_X hf sourceArc targetArc htrav hPi hs ht hc)
  have e' : (B ++ X).reverse.map Delta.toCombMap.alpha =
      invDarts Delta X ++ invDarts Delta B := e
  rw [e'] at h
  obtain ⟨hw, -, hfollows, heuler⟩ := h
  exact ⟨hw, hfollows, heuler⟩

/-- **The Y-pocket walk without spur**: the same three facts, from the X-pocket at the rotated face
walk `Y p⁻¹ X q⁻¹` and carrier `p A q B`. -/
theorem pocketInputs_Y (hf : (cell Delta j).face ≠ f)
    (sourceArc targetArc : CyclicArc (cellDarts Delta j))
    (htrav : (Delta.faceBoundary f).darts.rotate r =
      X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts)
    (hPi : (cellDarts Delta j).rotate n = targetArc.darts ++ B ++ sourceArc.darts ++ A)
    (hs : 0 < sourceArc.length) (ht : 0 < targetArc.length) (hne : Y ++ A ≠ [])
    (hnospur : ∀ d ∈ invDarts Delta Y ++ invDarts Delta A,
      Delta.toCombMap.alpha d ∉ invDarts Delta Y ++ invDarts Delta A) :
    ∃ hw : IsNoncrossingClosedWalk Delta.toCombMap (invDarts Delta Y ++ invDarts Delta A),
      (hw.outerCycle Delta.planar).FollowsBoundary ∧
        (reclosedMap Delta.toCombMap
          (sideFaces Delta.toCombMap (invDarts Delta Y ++ invDarts Delta A))
          (hw.innerCycle Delta.planar)).eulerCharacteristic = Delta.toCombMap.eulerCharacteristic := by
  have htrav' : (Delta.faceBoundary f).darts.rotate (r + (X ++ targetArc.reverseDarts).length) =
      Y ++ sourceArc.reverseDarts ++ X ++ targetArc.reverseDarts := by
    rw [← List.rotate_rotate, htrav, show X ++ targetArc.reverseDarts ++ Y ++ sourceArc.reverseDarts
      = (X ++ targetArc.reverseDarts) ++ (Y ++ sourceArc.reverseDarts) by
        simp only [List.append_assoc], List.rotate_append_length_eq]
    simp only [List.append_assoc]
  have hPi' : (cellDarts Delta j).rotate (n + (targetArc.darts ++ B).length) =
      sourceArc.darts ++ A ++ targetArc.darts ++ B := by
    rw [← List.rotate_rotate, hPi, show targetArc.darts ++ B ++ sourceArc.darts ++ A
      = (targetArc.darts ++ B) ++ (sourceArc.darts ++ A) by
        simp only [List.append_assoc], List.rotate_append_length_eq]
    simp only [List.append_assoc]
  exact pocketInputs_X hf targetArc sourceArc htrav' hPi' ht hs hne hnospur
