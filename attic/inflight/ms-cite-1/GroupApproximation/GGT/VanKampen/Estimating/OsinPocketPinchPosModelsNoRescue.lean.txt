/-!
# No rescue relation in the group of the ε = 1 counterexample

A simple pocket face set of an O-equivalent copy of the counterexample diagram, with the exterior off
its cycle and a proper arc `t₁`, gives a relation `α · p⁻¹ · β = g · r^{±1} · g⁻¹` in `G`, where `α`,
`β` have word norm at most one, `p` is the word of a proper cyclic arc of the source word and `r` is
the kept word.  With two relator cells the kept word is the other one.  These theorems refute every
such relation.

* `noRescueKeptQ`: kept word `Q`, source word `P`.
* `noRescueKeptP`: kept word `P`, source word `Q`.

In each, a case with no `c` among `α, β` is refuted by exponent sums (`Q` has `d, e` exponents `2, 1`,
two letters give at most one each), opposite signs of `c` by the same exponent sums (the `c`s cancel),
and every other case by the kill map into `Perm (Fin 4)`, where the kept word dies.

## Manuscript status

Model test for infrastructure of `thm:hull` (Hull's small cancellation theorem, through Osin's
Lemma 9.7(b)); certifies no printed sentence on its own.
-/

namespace GroupApproximation.GGT.VanKampen.OsinPocketPinchPosModels

open GGT

/-- **No relation for the kept word `Q` against an arc of `P`.** -/
theorem noRescueKeptQ {α β g : G} (hα : α ∈ T) (hβ : β ∈ T) {m n : ℕ} (hn : n < 3) (s : Bool)
    (h : α * (RelLetter.listVal ((pWord.rotate m).take n))⁻¹ * β =
      g * (if s then gQ else gQ⁻¹) * g⁻¹) : False := by
  obtain ⟨hk, hd, he⟩ := killP_listVal_ab _ (mem_pWord_arc (m := m) (n := n))
  rw [length_pWord_arc hn] at hk
  have hkill := congrArg killP h
  have hD := congrArg (fun z => Multiplicative.toAdd (expD z)) h
  have hE := congrArg (fun z => Multiplicative.toAdd (expE z)) h
  simp only [expD_conj, expE_conj] at hD hE
  cases s <;>
    simp only [Bool.false_eq_true, if_false, if_true, map_mul, map_inv, hk, hd, he, killP_gQ,
      toAdd_mul, toAdd_inv, toAdd_one, neg_zero, add_zero, inv_one, mul_one, mul_inv_cancel,
      expD_gQ, expE_gQ] at hkill hD hE <;>
    rcases letter_casesP hα with rfl | rfl | ⟨hαk, hαe⟩ <;>
    rcases letter_casesP hβ with rfl | rfl | ⟨hβk, hβe⟩
  all_goals first
    | (simp only [toAdd_inv, add_neg_cancel, neg_add_cancel] at hD; omega)
    | (simp only [killP_gc, map_inv] at hkill
       interval_cases n <;> exact absurd hkill (by decide))
    | (rcases hβk with hβk | hβk | hβk | hβk <;>
       simp only [killP_gc, map_inv, hβk] at hkill <;>
       interval_cases n <;> exact absurd hkill (by decide))
    | (rcases hαk with hαk | hαk | hαk | hαk <;>
       simp only [killP_gc, map_inv, hαk] at hkill <;>
       interval_cases n <;> exact absurd hkill (by decide))
    | (rcases hαe with ⟨h1, h2⟩ | ⟨h1, h2⟩ | ⟨h1, h2⟩ | ⟨h1, h2⟩ | ⟨h1, h2⟩ <;>
       rcases hβe with ⟨h3, h4⟩ | ⟨h3, h4⟩ | ⟨h3, h4⟩ | ⟨h3, h4⟩ | ⟨h3, h4⟩ <;> omega)

/-- **No relation for the kept word `P` against an arc of `Q`.** -/
theorem noRescueKeptP {α β g : G} (hα : α ∈ T) (hβ : β ∈ T) {m n : ℕ} (hn : n < 3) (s : Bool)
    (h : α * (RelLetter.listVal ((qWord.rotate m).take n))⁻¹ * β =
      g * (if s then gP else gP⁻¹) * g⁻¹) : False := by
  obtain ⟨hk, hd, he⟩ := killQ_listVal_de _ (mem_qWord_arc (m := m) (n := n))
  rw [length_qWord_arc hn] at hk
  have hkill := congrArg killQ h
  have hD := congrArg (fun z => Multiplicative.toAdd (expA z)) h
  have hE := congrArg (fun z => Multiplicative.toAdd (expB z)) h
  simp only [expA_conj, expB_conj] at hD hE
  cases s <;>
    simp only [Bool.false_eq_true, if_false, if_true, map_mul, map_inv, hk, hd, he, killQ_gP,
      toAdd_mul, toAdd_inv, toAdd_one, neg_zero, add_zero, inv_one, mul_one, mul_inv_cancel,
      expA_gP, expB_gP] at hkill hD hE <;>
    rcases letter_casesQ hα with rfl | rfl | ⟨hαk, hαe⟩ <;>
    rcases letter_casesQ hβ with rfl | rfl | ⟨hβk, hβe⟩
  all_goals first
    | (simp only [toAdd_inv, add_neg_cancel, neg_add_cancel] at hD; omega)
    | (simp only [killQ_gc, map_inv] at hkill
       interval_cases n <;> exact absurd hkill (by decide))
    | (rcases hβk with hβk | hβk | hβk | hβk <;>
       simp only [killQ_gc, map_inv, hβk] at hkill <;>
       interval_cases n <;> exact absurd hkill (by decide))
    | (rcases hαk with hαk | hαk | hαk | hαk <;>
       simp only [killQ_gc, map_inv, hαk] at hkill <;>
       interval_cases n <;> exact absurd hkill (by decide))
    | (rcases hαe with ⟨h1, h2⟩ | ⟨h1, h2⟩ | ⟨h1, h2⟩ | ⟨h1, h2⟩ | ⟨h1, h2⟩ <;>
       rcases hβe with ⟨h3, h4⟩ | ⟨h3, h4⟩ | ⟨h3, h4⟩ | ⟨h3, h4⟩ | ⟨h3, h4⟩ <;> omega)

end GroupApproximation.GGT.VanKampen.OsinPocketPinchPosModels
