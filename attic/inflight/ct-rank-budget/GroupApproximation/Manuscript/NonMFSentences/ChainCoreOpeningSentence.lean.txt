/-!
# `sec:chain-core`: the opening sentence

`non_mf_groups_exist.tex`, Section `sec:chain-core` (tex 1341–1343):

> The torsion-defect argument can identify the whole MF radical even
> when the defect ideal is proper.  We give such a computation for
> algebraic crossed products arising from zero-dimensional dynamics.

`printedChainCoreOpening_closed` states the first sentence for the section's rings `R_X`.

* Over every finite field, the ideal generated by the torsion defects `d = 1_{P∖T(P)}` is the transient
  ideal `I` (`defectIdeal_eq_transientIdeal_chainRecurrentSet`), and it is proper: `Y = CR(T)` is nonempty
  (`chainRecurrentSet_nonempty`), so `1` does not restrict to `0` in `R_Y`.
* Over `F₂`, the torsion-defect argument (`printedCoreMFRadical_closed`, which runs through
  Proposition `prop:bilateral-three`) identifies the whole MF radical:
  - `Rad_MF(GL_n(R_X)) = K_n = EL_n(R_X, I)` for `n ≥ 3`;
  - `Rad_MF(EL_n(R_X)) = EL_n(R_X, I)` for `n ≥ 4`.
-/

namespace GroupApproximation.ChainCore.DynamicRankBudget

open MFQuotientUnits ClopenCrossedProduct

/-- **tex 1341**, "The torsion-defect argument can identify the whole MF radical even when the defect
ideal is proper": for every nonempty compact metrizable zero-dimensional `X` and homeomorphism `T`, over
every finite field the defect ideal is the transient ideal and is proper, and over `F₂` the MF radicals
of `GL_n(R_X)` (`n ≥ 3`) and `EL_n(R_X)` (`n ≥ 4`) are the relative elementary groups over it. -/
def PrintedChainCoreOpening : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X),
    (∀ (k : Type) [Field k] [Finite k],
      GroupApproximation.Manuscript.ChainCore.BilateralThree.defectIdeal T k ≠ ⊤ ∧
        GroupApproximation.Manuscript.ChainCore.BilateralThree.defectIdeal T k =
          coreTransientIdeal T k) ∧
      (∀ n : ℕ, 3 ≤ n →
        mfHomKernel (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T (ZMod 2)))ˣ =
            glCoreKernel T (ZMod 2) n ∧
          glCoreKernel T (ZMod 2) n =
            (relativeElementary (Fin n) (coreTransientIdeal T (ZMod 2))).map
              (elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))).subtype) ∧
      (∀ n : ℕ, 4 ≤ n →
        mfHomKernel ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))) =
          relativeElementary (Fin n) (coreTransientIdeal T (ZMod 2)))

theorem printedChainCoreOpening_closed : PrintedChainCoreOpening := by
  intro X _ _ _ _ T
  refine ⟨fun k _ _ => ?_, ?_⟩
  · have heq : GroupApproximation.Manuscript.ChainCore.BilateralThree.defectIdeal T k =
        coreTransientIdeal T k :=
      GroupApproximation.defectIdeal_eq_transientIdeal_chainRecurrentSet T k
    refine ⟨?_, heq⟩
    rw [heq]
    intro htop
    obtain ⟨y, hy⟩ := GroupApproximation.ChainCore.chainRecurrentSet_nonempty T
    have h1 : (1 : ClopenCrossedProduct T k) ∈ coreTransientIdeal T k :=
      (TwoSidedIdeal.one_mem_iff _).2 htop
    have h0 := (mem_transientIdeal T k (Dynamics.image_chainRecurrentSet T)).1 h1
    rw [map_one] at h0
    have hc := coeff_injective (restrictHomeo T (Dynamics.image_chainRecurrentSet T)) k
      ((map_one (coeff (restrictHomeo T (Dynamics.image_chainRecurrentSet T)) k)).trans
        (h0.trans (map_zero (coeff (restrictHomeo T (Dynamics.image_chainRecurrentSet T)) k)).symm))
    have hk : (1 : k) = 0 := by
      simpa using congrArg
        (fun f : LocallyConstant (Dynamics.chainRecurrentSet T) k => f ⟨y, hy⟩) hc
    exact one_ne_zero hk
  · obtain ⟨hK, hGL, hEL, -, -⟩ := GroupApproximation.ChainCore.ChainCoreClosures.printedCoreMFRadical_closed X T
    exact ⟨fun n hn => ⟨hGL n hn, (hK n (by omega)).1⟩,
      fun n hn => (hEL n hn).trans (hK n (by omega)).2.1⟩

end GroupApproximation.ChainCore.DynamicRankBudget
