> If $T$ has no fixed points, then over any finite field the groups
> $\GL_n(R_X)$ and $\EL_n(R_X)$, $n\ge2$, are MF if and only if they
> are LEF, if and only if $R_X$ is directly finite.
> If $T$ has no points of periods one or two, the same equivalence holds
> for the unit group $R_X^\times$.  Over $\F_2$, all these MF radicals
> are exactly the locally finite core restriction kernels.
> In particular, for every aperiodic system the maximal MF quotient of
> $R_X^\times$ is its LEF restriction image on the chain-recurrent core.

section Kernels

variable {X : Type} [MetricSpace X] [CompactSpace X]

/-- `Y = CR(T)` is invariant: `T(Y) = Y` (`lem:chain-core-models`). -/
theorem image_core (T : X ≃ₜ X) :
    T '' Dynamics.chainRecurrentSet T = Dynamics.chainRecurrentSet T :=
  Dynamics.image_chainRecurrentSet T

/-- The core restriction `R_X → R_Y`. -/
noncomputable abbrev coreRestrict (T : X ≃ₜ X) (k : Type) [Ring k] :
    ClopenCrossedProduct T k →+* ClopenCrossedProduct (restrictHomeo T (image_core T)) k :=
  restrict T k (image_core T)

/-- General linear core restriction `GL_n(R_X) → GL_n(R_Y)`. -/
noncomputable abbrev glCoreRestrict (T : X ≃ₜ X) (k : Type) [Ring k] (n : ℕ) :
    (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ →*
      (Matrix (Fin n) (Fin n) (ClopenCrossedProduct (restrictHomeo T (image_core T)) k))ˣ :=
  Units.map (RingHom.mapMatrix (coreRestrict T k)).toMonoidHom

/-- The general linear core restriction kernel `K_n(I)`. -/
noncomputable abbrev glCoreKernel (T : X ≃ₜ X) (k : Type) [Ring k] (n : ℕ) :
    Subgroup (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ :=
  (glCoreRestrict T k n).ker

/-- Elementary core restriction `EL_n(R_X) → GL_n(R_Y)`. -/
noncomputable abbrev elCoreRestrict (T : X ≃ₜ X) (k : Type) [Ring k] (n : ℕ) :
    ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T k)) →*
      (Matrix (Fin n) (Fin n) (ClopenCrossedProduct (restrictHomeo T (image_core T)) k))ˣ :=
  (glCoreRestrict T k n).comp (elementaryGroup (Fin n) (ClopenCrossedProduct T k)).subtype

/-- The elementary core restriction kernel `EL_n(R_X) ∩ K_n(I)`. -/
noncomputable abbrev elCoreKernel (T : X ≃ₜ X) (k : Type) [Ring k] (n : ℕ) :
    Subgroup ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T k)) :=
  (elCoreRestrict T k n).ker

/-- Unit-group core restriction `R_X^× → R_Y^×`. -/
noncomputable abbrev unitCoreRestrict (T : X ≃ₜ X) (k : Type) [Ring k] :
    (ClopenCrossedProduct T k)ˣ →* (ClopenCrossedProduct (restrictHomeo T (image_core T)) k)ˣ :=
  Units.map (coreRestrict T k).toMonoidHom

/-- The unit-group core restriction kernel `K_1(I)`. -/
noncomputable abbrev unitCoreKernel (T : X ≃ₜ X) (k : Type) [Ring k] :
    Subgroup (ClopenCrossedProduct T k)ˣ :=
  (unitCoreRestrict T k).ker

/-- The elementary kernel sits in the general linear kernel, so it inherits local finiteness. -/
theorem isLocallyFiniteGroup_elCoreKernel (T : X ≃ₜ X) (k : Type) [Ring k] (n : ℕ)
    (h : IsLocallyFiniteGroup (glCoreKernel T k n)) :
    IsLocallyFiniteGroup (elCoreKernel T k n) :=
  h.comap
    (((elementaryGroup (Fin n) (ClopenCrossedProduct T k)).subtype.comp
        (elCoreKernel T k n).subtype).codRestrict (glCoreKernel T k n)
      fun x ↦ MonoidHom.mem_ker.mpr (MonoidHom.mem_ker.mp x.2))
    fun _ _ hab ↦ Subtype.ext (Subtype.ext (congrArg Subtype.val hab))

end Kernels

theorem countable_units_matrix (R : Type) [Ring R] [Countable R] (n : ℕ) :
    Countable (Matrix (Fin n) (Fin n) R)ˣ := by
  haveI : Countable (Matrix (Fin n) (Fin n) R) := (inferInstance : Countable (Fin n → Fin n → R))
  exact Units.val_injective.countable

theorem countable_units (R : Type) [Ring R] [Countable R] : Countable Rˣ :=
  Units.val_injective.countable

/-- A nontrivial element of an MF radical pushed along an injective homomorphism. -/
theorem exists_mem_ne_one_of_injective {G H : Type} [Group G] [Group H] (E : H →* G)
    (hE : Function.Injective E) (h : ∃ x ∈ mfHomKernel H, x ≠ 1) :
    ∃ x ∈ mfHomKernel G, x ≠ 1 := by
  obtain ⟨x, hx, hx1⟩ := h
  exact ⟨E x, map_mfHomKernel_le E (Subgroup.mem_map_of_mem E hx),
    fun h1 ↦ hx1 (hE (h1.trans (map_one E).symm))⟩

/-- tex 1752–1758 for one group family: the restriction `ψ` into an LEF group, with its kernel
conjugated into the image of an embedded MF radical, has kernel equal to the MF radical. -/
theorem mfHomKernel_eq_ker_of_conj {G H M : Type} [Group G] [Group H] [Group M] [Countable G]
    (ψ : G →* M) (hLEF : IsLEF M) (E : H →* G) {L : Subgroup H} (hL : L ≤ mfHomKernel H)
    (hconj : ∀ x ∈ ψ.ker, ∃ w : G, w * x * w⁻¹ ∈ L.map E) : mfHomKernel G = ψ.ker := by
  have h := mfHomKernel_eq_ker_of_conj_into_embedded (isOperatorMF_range_of_isLEF hLEF ψ)
    ψ.rangeRestrict E hL (by rw [MonoidHom.ker_rangeRestrict]; exact hconj)
  rwa [MonoidHom.ker_rangeRestrict] at h

/-- **`cor:dynamic-rank-budget` as printed** (tex 1696–1705). -/
def PrintedDynamicRankBudget : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X),
    ((∀ x : X, T x ≠ x) → ∀ (k : Type) [Field k] [Finite k] (n : ℕ), 2 ≤ n →
      ((IsOperatorMF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ ↔
          IsLEF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ) ∧
        (IsLEF (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ ↔
          IsDedekindFiniteMonoid (ClopenCrossedProduct T k))) ∧
      ((IsOperatorMF ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T k)) ↔
          IsLEF ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T k))) ∧
        (IsLEF ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T k)) ↔
          IsDedekindFiniteMonoid (ClopenCrossedProduct T k)))) ∧
    ((∀ x : X, T x ≠ x ∧ T (T x) ≠ x) → ∀ (k : Type) [Field k] [Finite k],
      (IsOperatorMF (ClopenCrossedProduct T k)ˣ ↔ IsLEF (ClopenCrossedProduct T k)ˣ) ∧
        (IsLEF (ClopenCrossedProduct T k)ˣ ↔ IsDedekindFiniteMonoid (ClopenCrossedProduct T k))) ∧
    ((∀ x : X, T x ≠ x) → ∀ n : ℕ, 2 ≤ n →
      (mfHomKernel (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T (ZMod 2)))ˣ =
          glCoreKernel T (ZMod 2) n ∧ IsLocallyFiniteGroup (glCoreKernel T (ZMod 2) n)) ∧
        (mfHomKernel ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))) =
          elCoreKernel T (ZMod 2) n ∧ IsLocallyFiniteGroup (elCoreKernel T (ZMod 2) n))) ∧
    ((∀ x : X, T x ≠ x ∧ T (T x) ≠ x) →
      mfHomKernel (ClopenCrossedProduct T (ZMod 2))ˣ = unitCoreKernel T (ZMod 2) ∧
        IsLocallyFiniteGroup (unitCoreKernel T (ZMod 2))) ∧
    ((∀ (x : X) (j : ℕ), 0 < j → (⇑T)^[j] x ≠ x) →
      Nonempty ((ClopenCrossedProduct T (ZMod 2))ˣ ⧸ mfHomKernel (ClopenCrossedProduct T (ZMod 2))ˣ ≃*
          (unitCoreRestrict T (ZMod 2)).range) ∧
        IsLEF (unitCoreRestrict T (ZMod 2)).range)

/-- **The Introduction's aperiodic sentence** (tex 254–259): over `F_2`, for aperiodic `T`, the
MF radical of the unit group and of every `EL_n(R_X)`, `n ≥ 2`, is exactly the locally finite
restriction kernel, with LEF quotients. -/
def PrintedIntroductionAperiodicSentence : Prop :=
  ∀ (X : Type) [MetricSpace X] [CompactSpace X] [TotallyDisconnectedSpace X] [Nonempty X]
    (T : X ≃ₜ X), (∀ (x : X) (j : ℕ), 0 < j → (⇑T)^[j] x ≠ x) →
    (mfHomKernel (ClopenCrossedProduct T (ZMod 2))ˣ = unitCoreKernel T (ZMod 2) ∧
      IsLocallyFiniteGroup (unitCoreKernel T (ZMod 2)) ∧
      Nonempty ((ClopenCrossedProduct T (ZMod 2))ˣ ⧸ mfHomKernel (ClopenCrossedProduct T (ZMod 2))ˣ ≃*
        (unitCoreRestrict T (ZMod 2)).range) ∧
      IsLEF (unitCoreRestrict T (ZMod 2)).range) ∧
    ∀ n : ℕ, 2 ≤ n →
      mfHomKernel ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))) =
          elCoreKernel T (ZMod 2) n ∧
        IsLocallyFiniteGroup (elCoreKernel T (ZMod 2) n) ∧
        Nonempty (↥(elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))) ⧸
            mfHomKernel ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))) ≃*
          (elCoreRestrict T (ZMod 2) n).range) ∧
        IsLEF (elCoreRestrict T (ZMod 2) n).range

/-- **Piece: Theorem `thm:core-ring-reflection`, its last sentence** (tex 1522–1523; owner

/-- **Piece: Theorem `thm:core-mf-radical`, the kill half and local finiteness** (tex 1614–1618,

/-- **Piece: the return embeddings** (tex 1707–1757).  For a system with no points of period
below `r`, there is a return system `(C, S)` (the full clopen section `C`, its first-return map
`S = S_C`, `P_C = LC(C,k) ⋊_{S_C} ℤ`) such that, over every finite field,

* `R_X` embeds non-unitally and injectively in a matrix ring over `R_C` (the full tower corner,

* unitization of `Φ` embeds `GL_{nr}(R_C)` in `GL_n(R_X)` (tex 1739), `EL_{nr}(R_C)` in

and over `F_2` every element of the core restriction kernel of `GL_n(R_X)`, `EL_n(R_X)` (`n ≥ 2`)
or `R_X^×` has a conjugate in the image of the core restriction kernel of the embedded group
(tex 1743–1757: the induced core is `C ∩ Y`, `J = p_C I p_C`, the phase-zero placement, and

Owners: ct-return-tower (the tower and embeddings), ct-rank-budget (induced core, `J`, placement),

def ReturnEmbeddingStatement : Prop :=

          (∀ n : ℕ, 1 ≤ n → ∃ E : (Matrix (Fin (n * r)) (Fin (n * r)) (ClopenCrossedProduct S k))ˣ →*
            (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T k))ˣ, Function.Injective E) ∧
          (∀ n : ℕ, 2 ≤ n → ∃ E : ↥(elementaryGroup (Fin (n * r)) (ClopenCrossedProduct S k)) →*
            ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T k)), Function.Injective E) ∧
          (∃ E : (Matrix (Fin r) (Fin r) (ClopenCrossedProduct S k))ˣ →* (ClopenCrossedProduct T k)ˣ,
            Function.Injective E)) ∧
      (∀ n : ℕ, 1 ≤ n →
        ∃ E : (Matrix (Fin (n * r)) (Fin (n * r)) (ClopenCrossedProduct S (ZMod 2)))ˣ →*
            (Matrix (Fin n) (Fin n) (ClopenCrossedProduct T (ZMod 2)))ˣ,
          ∀ x ∈ glCoreKernel T (ZMod 2) n, ∃ w, w * x * w⁻¹ ∈ (glCoreKernel S (ZMod 2) (n * r)).map E) ∧
      (∀ n : ℕ, 2 ≤ n →
        ∃ E : ↥(elementaryGroup (Fin (n * r)) (ClopenCrossedProduct S (ZMod 2))) →*
            ↥(elementaryGroup (Fin n) (ClopenCrossedProduct T (ZMod 2))),
          ∀ x ∈ elCoreKernel T (ZMod 2) n, ∃ w, w * x * w⁻¹ ∈ (elCoreKernel S (ZMod 2) (n * r)).map E) ∧
      (∃ E : (Matrix (Fin r) (Fin r) (ClopenCrossedProduct S (ZMod 2)))ˣ →*
          (ClopenCrossedProduct T (ZMod 2))ˣ,
        ∀ x ∈ unitCoreKernel T (ZMod 2), ∃ w, w * x * w⁻¹ ∈ (glCoreKernel S (ZMod 2) r).map E)

/-! ## The printed assembly -/

/-- **The proof of `cor:dynamic-rank-budget`, over its pieces** (tex 1707–1767).  Use `r = 2` for
the matrix groups and `r = 3` for the unit group.  Over `F_2` the kernel of the embedded group is
killed by `thm:core-mf-radical`, the involution conjugates the whole kernel into its image, and the
LEF restriction image gives equality of radicals.  Over other finite fields a nonzero invisible
root of the return ring is transferred along the embedding, and the LEF ring models give the
positive implications. -/
theorem printedDynamicRankBudget_of_pieces (hRefl : CoreRingReflectionStatement)
    (hModels : CoreModelsLEFStatement) (hCount : CountableCrossedProductStatement)
    (hRad : CoreMFRadicalKillStatement) (hBil : BilateralThreeTransferStatement)
    (hTors : TorsionDefectTransferStatement) (hUnitLF : UnitKernelLocallyFiniteStatement)
    (hTower : ReturnEmbeddingStatement) : PrintedDynamicRankBudget := by
  intro X _ _ _ _ T
  have hper2 : (∀ x : X, T x ≠ x) → ∀ (x : X) (j : ℕ), 0 < j → j < 2 → (⇑T)^[j] x ≠ x := by
    intro h x j hj0 hj2
    obtain rfl : j = 1 := by omega
    rw [Function.iterate_one]
    exact h x
  have hper3 : (∀ x : X, T x ≠ x ∧ T (T x) ≠ x) →
      ∀ (x : X) (j : ℕ), 0 < j → j < 3 → (⇑T)^[j] x ≠ x := by
    intro h x j hj0 hj3
    rcases (by omega : j = 1 ∨ j = 1 + 1) with rfl | rfl
    · rw [Function.iterate_one]
      exact (h x).1
    · rw [Function.iterate_succ_apply', Function.iterate_one]
      exact (h x).2
  have hD : (∀ x : X, T x ≠ x ∧ T (T x) ≠ x) →
      mfHomKernel (ClopenCrossedProduct T (ZMod 2))ˣ = unitCoreKernel T (ZMod 2) ∧
        IsLocallyFiniteGroup (unitCoreKernel T (ZMod 2)) := by
    intro hper
    obtain ⟨C, _, _, _, _, S, -, -, -, ⟨E, hconj⟩⟩ := hTower X T 3 (by norm_num) (hper3 hper)
    haveI := hCount X T (ZMod 2)
    haveI : Countable (ClopenCrossedProduct T (ZMod 2))ˣ := countable_units _
    exact ⟨mfHomKernel_eq_ker_of_conj (unitCoreRestrict T (ZMod 2))
      (hModels X T (ZMod 2)).isLEF_units E ((hRad C S).1 3 (by norm_num)) hconj, hUnitLF X T⟩
  refine ⟨?_, ?_, ?_, hD, ?_⟩
  · intro hfix k _ _ n hn
    obtain ⟨C, _, _, _, _, S, hk, -, -, -⟩ := hTower X T 2 (by norm_num) (hper2 hfix)
    obtain ⟨⟨m, ψ, hψ⟩, hglE, helE, -⟩ := hk k
    haveI := hCount X T k
    haveI := hCount C S k
    haveI := countable_units_matrix (ClopenCrossedProduct T k) n
    have hDFLEF : IsDedekindFiniteMonoid (ClopenCrossedProduct T k) →
        Pestov91.IsLEFRing (ClopenCrossedProduct T k) :=
      fun h ↦ (hRefl X T k).2.1.mp ((hRefl X T k).1.mp h)
    have hnotDF : ¬ IsDedekindFiniteMonoid (ClopenCrossedProduct T k) →
        ¬ IsDedekindFiniteMonoid (ClopenCrossedProduct S k) :=
      not_isDedekindFiniteMonoid_of_tower_corner (fun h ↦ (hRefl C S k).1.mp h) m ψ hψ
    refine ⟨isOperatorMF_iff_isLEF_and_isLEF_iff_of_pieces
