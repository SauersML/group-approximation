/-!
# The Sturmian subshift is infinite and minimal

`simple_kazhdan_sofic_group.tex` (origin/main bf961c128), tex l.250–251: "the infinite minimal
Sturmian subshift of an irrational slope $\alpha\in(0,1)$ [MorseHedlund]".

The cited input is proved here, not assumed:
* `exists_bounded_return`: an irrational rotation returns to `[0, ε)` modulo `1` within a bounded
  number of steps from every starting point, by Dirichlet's approximation theorem;
* `exists_right_window`: the first `n` letters of a coding do not change when the starting point
  moves slightly to the right;
* `exists_occurrence_of_mem_carrier`: every word of a coding occurs in every point of the subshift;
* `sturmianSubshift_isMinimal`, `sturmianSubshift_infinite` (a coding is not periodic, since
  `α` is irrational), and the closed endpoint `printedSturmianSubshiftInfiniteMinimal`.
-/

namespace GroupApproximation
namespace SimpleKazhdanSofic

open SymbolicDynamics SymbolicDynamics.FullShift

/-- Bounded returns of an irrational rotation into `[0, ε)` modulo `1`. -/
theorem exists_bounded_return (α : ℝ) (hα : Irrational α) {ε : ℝ} (hε : 0 < ε) :
    ∃ L : ℕ, ∀ φ : ℝ, ∃ p : ℕ, p < L ∧ Int.fract (φ + (p : ℝ) * α) < ε := by
  obtain ⟨m, hm⟩ := exists_nat_one_div_lt hε
  obtain ⟨j, k, hk0, -, hjk⟩ := Real.exists_int_int_abs_mul_sub_le α (Nat.succ_pos m)
  set δ : ℝ := (k : ℝ) * α - (j : ℝ) with hδdef
  have hm1 : (1 : ℝ) / (((m + 1 : ℕ) : ℝ) + 1) ≤ 1 / ((m : ℝ) + 1) := by
    apply one_div_le_one_div_of_le (by positivity)
    push_cast
    linarith
  have hδε : |δ| < ε := lt_of_le_of_lt (le_trans hjk hm1) hm
  have hδ1 : |δ| < 1 := by
    have h2 : (1 : ℝ) / (((m + 1 : ℕ) : ℝ) + 1) < 1 := by
      rw [div_lt_one (by positivity)]
      push_cast
      linarith [(Nat.cast_nonneg m : (0 : ℝ) ≤ m)]
    exact lt_of_le_of_lt hjk h2
  have hδ0 : δ ≠ 0 := by
    have hirr : Irrational ((k : ℝ) * α) := hα.intCast_mul hk0.ne'
    exact sub_ne_zero.mpr (hirr.ne_int j)
  obtain ⟨kn, hkn⟩ : ∃ kn : ℕ, (kn : ℤ) = k := ⟨k.toNat, Int.toNat_of_nonneg hk0.le⟩
  have hkpos : 0 < kn := by omega
  have hkR : (k : ℝ) = (kn : ℝ) := by rw [← hkn, Int.cast_natCast]
  refine ⟨kn * (⌈1 / |δ|⌉₊ + 2), fun φ => ?_⟩
  set s := Int.fract φ with hs
  have hs0 : 0 ≤ s := Int.fract_nonneg φ
  have hs1 : s < 1 := Int.fract_lt_one φ
  have hsplit : ∀ m' : ℕ, Int.fract (φ + ((kn * m' : ℕ) : ℝ) * α) = Int.fract (s + (m' : ℝ) * δ) := by
    intro m'
    have e : φ + ((kn * m' : ℕ) : ℝ) * α =
        (s + (m' : ℝ) * δ) + ((⌊φ⌋ + (m' : ℤ) * j : ℤ) : ℝ) := by
      have hφ := Int.floor_add_fract φ
      rw [hδdef, hkR]
      push_cast
      linarith
    rw [e, Int.fract_add_intCast]
  have hceil : (1 : ℝ) / |δ| ≤ (⌈1 / |δ|⌉₊ : ℝ) := Nat.le_ceil _
  have hδpos : 0 < |δ| := abs_pos.mpr hδ0
  rcases lt_or_gt_of_ne hδ0 with hneg | hpos
  · -- negative step: go down from `s` in steps `|δ|`
    have habs : |δ| = -δ := abs_of_neg hneg
    refine ⟨kn * ⌊s / |δ|⌋₊, ?_, ?_⟩
    · have h1 : (⌊s / |δ|⌋₊ : ℝ) ≤ s / |δ| := Nat.floor_le (div_nonneg hs0 hδpos.le)
      have h2 : s / |δ| < 1 / |δ| := by
        rw [div_lt_div_iff_of_pos_right hδpos]
        exact hs1
      have h3 : ⌊s / |δ|⌋₊ < ⌈1 / |δ|⌉₊ + 2 := by
        have : (⌊s / |δ|⌋₊ : ℝ) < (⌈1 / |δ|⌉₊ : ℝ) + 2 := by linarith
        exact_mod_cast this
      gcongr
    · rw [hsplit]
      have h1 : (⌊s / |δ|⌋₊ : ℝ) ≤ s / |δ| := Nat.floor_le (div_nonneg hs0 hδpos.le)
      have h2 : s / |δ| < (⌊s / |δ|⌋₊ : ℝ) + 1 := Nat.lt_floor_add_one _
      have h1' : (⌊s / |δ|⌋₊ : ℝ) * |δ| ≤ s := (le_div_iff₀ hδpos).mp h1
      have h2' : s < ((⌊s / |δ|⌋₊ : ℝ) + 1) * |δ| := (div_lt_iff₀ hδpos).mp h2
      have hval : Int.fract (s + (⌊s / |δ|⌋₊ : ℝ) * δ) = s + (⌊s / |δ|⌋₊ : ℝ) * δ := by
        rw [Int.fract_eq_iff]
        refine ⟨by rw [habs] at h1'; linarith, by rw [habs] at h2'; linarith, 0, by simp⟩
      rw [hval]
      rw [habs] at h2' hδε
      linarith
  · -- positive step: go up from `s` past `1` in steps `δ`
    have habs : |δ| = δ := abs_of_pos hpos
    refine ⟨kn * (⌊(1 - s) / δ⌋₊ + 1), ?_, ?_⟩
    · have h1 : (⌊(1 - s) / δ⌋₊ : ℝ) ≤ (1 - s) / δ :=
        Nat.floor_le (div_nonneg (by linarith) hpos.le)
      have h2 : (1 - s) / δ ≤ 1 / |δ| := by
        rw [habs]
        exact div_le_div_of_nonneg_right (by linarith) hpos.le
      have h3 : ⌊(1 - s) / δ⌋₊ + 1 < ⌈1 / |δ|⌉₊ + 2 := by
        have : (⌊(1 - s) / δ⌋₊ : ℝ) + 1 < (⌈1 / |δ|⌉₊ : ℝ) + 2 := by linarith
        exact_mod_cast this
      gcongr
    · rw [hsplit]
      have h1 : (⌊(1 - s) / δ⌋₊ : ℝ) ≤ (1 - s) / δ :=
        Nat.floor_le (div_nonneg (by linarith) hpos.le)
      have h2 : (1 - s) / δ < (⌊(1 - s) / δ⌋₊ : ℝ) + 1 := Nat.lt_floor_add_one _
      have h1' : (⌊(1 - s) / δ⌋₊ : ℝ) * δ ≤ 1 - s := (le_div_iff₀ hpos).mp h1
      have h2' : 1 - s < ((⌊(1 - s) / δ⌋₊ : ℝ) + 1) * δ := (div_lt_iff₀ hpos).mp h2
      have hval : Int.fract (s + (((⌊(1 - s) / δ⌋₊ + 1 : ℕ)) : ℝ) * δ) =
          s + (((⌊(1 - s) / δ⌋₊ + 1 : ℕ)) : ℝ) * δ - 1 := by
        rw [Int.fract_eq_iff]
        push_cast
        refine ⟨by linarith, by rw [habs] at hδ1; linarith, 1, by push_cast; ring⟩
      rw [hval]
      rw [habs] at hδε
      push_cast
      linarith

/-- One letter of a coding does not change when the starting point moves slightly right. -/
theorem exists_right_window_one (α θ₀ : ℝ) (hα0 : 0 < α) (i : ℤ) :
    ∃ η : ℝ, 0 < η ∧ ∀ δ : ℝ, 0 ≤ δ → δ < η →
      sturmianCoding α (θ₀ + δ) i = sturmianCoding α θ₀ i := by
  have ht0 : 0 ≤ Int.fract (θ₀ + (i : ℝ) * α) := Int.fract_nonneg _
  have ht1 : Int.fract (θ₀ + (i : ℝ) * α) < 1 := Int.fract_lt_one _
  have hshift : ∀ δ : ℝ, 0 ≤ δ → Int.fract (θ₀ + (i : ℝ) * α) + δ < 1 →
      Int.fract (θ₀ + δ + (i : ℝ) * α) = Int.fract (θ₀ + (i : ℝ) * α) + δ := by
    intro δ hδ0 hδ1
    rw [Int.fract_eq_iff]
    refine ⟨by linarith, hδ1, ⌊θ₀ + (i : ℝ) * α⌋, ?_⟩
    have := Int.floor_add_fract (θ₀ + (i : ℝ) * α)
    linarith
  by_cases h : 1 - α ≤ Int.fract (θ₀ + (i : ℝ) * α)
  · refine ⟨1 - Int.fract (θ₀ + (i : ℝ) * α), by linarith, fun δ hδ0 hδ => ?_⟩
    rw [sturmianCoding_apply, sturmianCoding_apply, hshift δ hδ0 (by linarith)]
    exact decide_eq_decide.mpr ⟨fun _ => h, fun _ => by linarith⟩
  · have h' : Int.fract (θ₀ + (i : ℝ) * α) < 1 - α := lt_of_not_ge h
    refine ⟨1 - α - Int.fract (θ₀ + (i : ℝ) * α), by linarith, fun δ hδ0 hδ => ?_⟩
    rw [sturmianCoding_apply, sturmianCoding_apply, hshift δ hδ0 (by linarith)]
    exact decide_eq_decide.mpr ⟨fun hc => absurd hc (not_le.mpr (by linarith)), fun hc => absurd hc h⟩

/-- The first `n` letters of a coding do not change when the starting point moves slightly
right. -/
theorem exists_right_window (α θ₀ : ℝ) (hα0 : 0 < α) (n : ℕ) :
    ∃ η : ℝ, 0 < η ∧ ∀ δ : ℝ, 0 ≤ δ → δ < η → ∀ i : ℕ, i < n →
      sturmianCoding α (θ₀ + δ) (i : ℤ) = sturmianCoding α θ₀ (i : ℤ) := by
  induction n with
  | zero => exact ⟨1, one_pos, fun _ _ _ i hi => absurd hi (Nat.not_lt_zero i)⟩
  | succ n ih =>
    obtain ⟨η, hη, hηw⟩ := ih
    obtain ⟨η', hη', hη'w⟩ := exists_right_window_one α θ₀ hα0 (n : ℤ)
    refine ⟨min η η', lt_min hη hη', fun δ hδ0 hδ i hi => ?_⟩
    by_cases hin : i < n
    · exact hηw δ hδ0 (lt_of_lt_of_le hδ (min_le_left _ _)) i hin
    · obtain rfl : i = n := by omega
      exact hη'w δ hδ0 (lt_of_lt_of_le hδ (min_le_right _ _))

/-- Every word of a coding reappears in every coding within a bounded distance. -/
theorem exists_bounded_occurrence (α : ℝ) (hα : Irrational α) (hα0 : 0 < α) (θ₀ : ℝ) (n : ℕ) :
    ∃ L : ℕ, ∀ φ : ℝ, ∃ p : ℕ, p < L ∧
      WordGraph.word (sturmianCoding α φ) (p : ℤ) n = WordGraph.word (sturmianCoding α θ₀) 0 n := by
  obtain ⟨η, hη, hw⟩ := exists_right_window α θ₀ hα0 n
  obtain ⟨L, hL⟩ := exists_bounded_return α hα hη
  refine ⟨L, fun φ => ?_⟩
  obtain ⟨p, hpL, hp⟩ := hL (φ - θ₀)
  refine ⟨p, hpL, ?_⟩
  funext j
  simp only [WordGraph.word_apply, zero_add]
  have e1 := congrFun (shift_sturmianCoding α φ (p : ℤ)) ((j : ℕ) : ℤ)
  rw [shift_apply, Int.cast_natCast] at e1
  have e2 : φ + (p : ℝ) * α =
      θ₀ + Int.fract (φ - θ₀ + (p : ℝ) * α) + (⌊φ - θ₀ + (p : ℝ) * α⌋ : ℝ) := by
    have := Int.floor_add_fract (φ - θ₀ + (p : ℝ) * α)
    linarith
  rw [e1, e2, sturmianCoding_add_intCast]
  exact hw _ (Int.fract_nonneg _) hp j j.2

/-- Every word of a coding occurs in every point of the Sturmian subshift. -/
theorem exists_occurrence_of_mem_carrier (α : ℝ) (hα : Irrational α) (hα0 : 0 < α) (θ₀ : ℝ)
    (n : ℕ) {x : ℤ → Bool} (hx : x ∈ sturmianCarrier α) :
    ∃ q : ℤ, WordGraph.word x q n = WordGraph.word (sturmianCoding α θ₀) 0 n := by
  obtain ⟨L, hL⟩ := exists_bounded_occurrence α hα hα0 θ₀ n
  have hx' : ∀ N : ℕ, ∃ θ : ℝ, ∀ i : ℤ, -(N : ℤ) ≤ i → i ≤ N → x i = sturmianCoding α θ i := hx
  obtain ⟨φ, hφ⟩ := hx' (L + n)
  obtain ⟨p, hpL, hp⟩ := hL φ
  refine ⟨p, ?_⟩
  rw [← hp]
  funext j
  simp only [WordGraph.word_apply]
  have hc : ((L + n : ℕ) : ℤ) = (L : ℤ) + (n : ℤ) := Nat.cast_add L n
  have hj : (j : ℕ) < n := j.2
  exact hφ _ (by omega) (by omega)

/-- The Sturmian subshift of an irrational slope is minimal. -/
theorem sturmianSubshift_isMinimal (α : ℝ) (hα : Irrational α) (hα0 : 0 < α) :
    IsMinimalSubshift (sturmianSubshift α) := by
  intro Y hYX hYc hYs hYne
  refine Set.Subset.antisymm hYX ?_
  intro x hx
  obtain ⟨y, hy⟩ := hYne
  by_contra hxY
  obtain ⟨I, u, hu, hIU⟩ := isOpen_pi_iff.mp hYc.isOpen_compl x hxY
  obtain ⟨N, hN⟩ := exists_bound_of_finset I
  have hx' : ∀ N : ℕ, ∃ θ : ℝ, ∀ i : ℤ, -(N : ℤ) ≤ i → i ≤ N → x i = sturmianCoding α θ i := hx
  obtain ⟨φ, hφ⟩ := hx' N
  obtain ⟨q, hq⟩ := exists_occurrence_of_mem_carrier α hα hα0
    (φ + ((-(N : ℤ) : ℤ) : ℝ) * α) (2 * N + 1) (hYX hy)
  have hagree : ∀ i : ℤ, -(N : ℤ) ≤ i → i ≤ N → shift (q + N) y i = x i := by
    intro i hi1 hi2
    have hjn : (((i + N).toNat : ℕ) : ℤ) = i + N := Int.toNat_of_nonneg (by omega)
    have hlt : (i + N).toNat < 2 * N + 1 := by omega
    have h1 := congrFun hq ⟨(i + N).toNat, hlt⟩
    simp only [WordGraph.word_apply, Fin.val_mk, zero_add] at h1
    rw [← shift_sturmianCoding, shift_apply, hjn] at h1
    rw [shift_apply, hφ i hi1 hi2]
    have e1 : q + (N : ℤ) + i = q + (i + N) := by ring
    have e2 : -(N : ℤ) + (i + N) = i := by ring
    rw [e1, h1, e2]
  have hmem : shift (q + N) y ∈ (↑I : Set ℤ).pi u := by
    rw [Set.mem_pi]
    intro i hi
    have hi' : i ∈ I := Finset.mem_coe.mp hi
    rw [hagree i (hN i hi').1 (hN i hi').2]
    exact (hu i hi').2
  exact absurd (hYs (q + N) hy) (hIU hmem)

/-- The coding from `0` of an irrational rotation is not periodic. -/
theorem sturmianCoding_zero_not_periodic (α : ℝ) (hα : Irrational α) (hα0 : 0 < α) (hα1 : α < 1)
    (p : ℕ) (hp : 0 < p) : ¬ ∀ i : ℤ, sturmianCoding α 0 (i + p) = sturmianCoding α 0 i := by
  intro hper
  have hiter : ∀ K : ℕ, ∀ i : ℤ,
      sturmianCoding α 0 (i + ((K * p : ℕ) : ℤ)) = sturmianCoding α 0 i := by
    intro K
    induction K with
    | zero => intro i; simp
    | succ K ih =>
      intro i
      have e : i + (((K + 1) * p : ℕ) : ℤ) = (i + ((K * p : ℕ) : ℤ)) + (p : ℤ) := by
        push_cast
        ring
      rw [e, hper, ih]
  have hsum : ∀ K : ℕ,
      ∑ i ∈ Finset.range (K * p), (if sturmianCoding α 0 (i : ℤ) then (1 : ℤ) else 0) =
        (K : ℤ) * ∑ i ∈ Finset.range p, (if sturmianCoding α 0 (i : ℤ) then (1 : ℤ) else 0) := by
    intro K
    induction K with
    | zero => simp
    | succ K ih =>
      rw [add_mul, one_mul, Finset.sum_range_add, ih]
      have hc : ∀ i ∈ Finset.range p,
          (if sturmianCoding α 0 ((K * p + i : ℕ) : ℤ) then (1 : ℤ) else 0) =
            (if sturmianCoding α 0 (i : ℤ) then (1 : ℤ) else 0) := by
        intro i _
        have e : ((K * p + i : ℕ) : ℤ) = (i : ℤ) + ((K * p : ℕ) : ℤ) := by
          push_cast
          ring
        rw [e, hiter]
      rw [Finset.sum_congr rfl hc]
      push_cast
      ring
  have hfloor : ∀ K : ℕ, ⌊((K * p : ℕ) : ℝ) * α⌋ = (K : ℤ) * ⌊(p : ℝ) * α⌋ := by
    intro K
    have h1 := sum_sturmianCoding α 0 hα0 hα1 (K * p)
    have h2 := sum_sturmianCoding α 0 hα0 hα1 p
    rw [hsum K, h2] at h1
    simp only [zero_add, Int.floor_zero, sub_zero] at h1
    exact h1.symm
  have hsplit := Int.floor_add_fract ((p : ℝ) * α)
  have hfr : Int.fract ((p : ℝ) * α) = 0 := by
    by_contra hne
    have hpos : 0 < Int.fract ((p : ℝ) * α) := lt_of_le_of_ne (Int.fract_nonneg _) (Ne.symm hne)
    obtain ⟨K, hK⟩ := exists_nat_one_div_lt hpos
    have hlt := Int.sub_one_lt_floor ((((K + 1) * p : ℕ) : ℝ) * α)
    rw [hfloor (K + 1)] at hlt
    push_cast at hlt
    have e : ((K : ℝ) + 1) * (⌊(p : ℝ) * α⌋ : ℝ) =
        ((K : ℝ) + 1) * ((p : ℝ) * α) - ((K : ℝ) + 1) * Int.fract ((p : ℝ) * α) := by
      linear_combination ((K : ℝ) + 1) * hsplit
    have hK' : 1 < Int.fract ((p : ℝ) * α) * ((K : ℝ) + 1) := by
      rwa [div_lt_iff₀ (by positivity)] at hK
    nlinarith
  have hirr : Irrational ((p : ℝ) * α) := by
    have := hα.intCast_mul (m := (p : ℤ)) (by exact_mod_cast hp.ne')
    simpa using this
  exact hirr.ne_int ⌊(p : ℝ) * α⌋ (by linarith)

/-- The Sturmian subshift of an irrational slope is infinite. -/
theorem sturmianSubshift_infinite (α : ℝ) (hα : Irrational α) (hα0 : 0 < α) (hα1 : α < 1) :
    (sturmianSubshift α).carrier.Infinite := by
  refine Set.infinite_of_injective_forall_mem
    (f := fun k : ℤ => shift k (sturmianCoding α 0)) ?_ ?_
  · intro k l hkl
    by_contra hne
    have hkl' : shift k (sturmianCoding α 0) = shift l (sturmianCoding α 0) := hkl
    have hper : ∀ i : ℤ, sturmianCoding α 0 (i + (k - l)) = sturmianCoding α 0 i := by
      intro i
      have h := congrFun hkl' (i - l)
      simp only [shift_apply] at h
      have e1 : k + (i - l) = i + (k - l) := by ring
      have e2 : l + (i - l) = i := by ring
      rw [e1, e2] at h
      exact h
    rcases lt_or_gt_of_ne hne with hlt | hgt
    · have hd : (((l - k).toNat : ℕ) : ℤ) = l - k := Int.toNat_of_nonneg (by omega)
      refine sturmianCoding_zero_not_periodic α hα hα0 hα1 (l - k).toNat (by omega) ?_
      intro i
      rw [hd]
      have h := hper (i + (l - k))
      have e : i + (l - k) + (k - l) = i := by ring
      rw [e] at h
      exact h.symm
    · have hd : (((k - l).toNat : ℕ) : ℤ) = k - l := Int.toNat_of_nonneg (by omega)
      refine sturmianCoding_zero_not_periodic α hα hα0 hα1 (k - l).toNat (by omega) ?_
      intro i
      rw [hd]
      exact hper i
  · intro k
    exact mapsTo_shift_sturmianCarrier α k (sturmianCoding_mem_carrier α 0)

/-- Tex l.250–251, closed: for irrational `α ∈ (0,1)` the Sturmian subshift is infinite and
minimal. -/
theorem printedSturmianSubshiftInfiniteMinimal : PrintedSturmianSubshiftInfiniteMinimal :=
  fun α hα hα0 hα1 =>
    ⟨sturmianSubshift_infinite α hα hα0 hα1, sturmianSubshift_isMinimal α hα hα0⟩
