import GroupApproximation.KOne.RegularizedDeterminant

/-!
# The Toeplitz space of words

`non_mf_groups_exist.tex`, `cor:leavitt-mf-quotient`: `K₁(L_k(1,d)) ≅ k^×/(k^×)^{d-1}`.
This module builds the representation on which the regularized determinant of
`KOne/RegularizedDeterminant.lean` is evaluated.

`Space k d` is the vector space with basis the finite words over `Fin d`.  On it

* `opS j` appends the letter `j` at the end of a word,
* `opT j` deletes a final letter `j` and kills words that do not end in `j`,
* `opQ` is the projection onto the empty word.

These satisfy the two Toeplitz relations

    `opT i * opS j = δᵢⱼ`   (`opT_mul_opS`),     `∑ⱼ opS j * opT j = 1 - opQ`   (`sum_opS_mul_opT`),

so the Leavitt relation `∑ⱼ sⱼtⱼ = 1` holds modulo the rank-one `opQ`
(`finiteRank_opQ`).

`split k d : Space k d ≃ k × (Fin d → Space k d)` sends the empty word to
`(1, 0)` and `i :: w` to `(0, e_w in coordinate i)`: it reads the **first**
letter, while `opS`/`opT` act on the **last** one.  The two ends are independent
except on words of length at most one, which is why both generators are *tame*
for this splitting: their defects have rank at most one (`defect_opS`,
`defect_opT`, `tame_opS`, `tame_opT`).
-/

namespace GroupApproximation
namespace ToeplitzWords

open FiniteRankDet RegularizedDet

variable (k : Type*) [Field k] (d : ℕ)

/-- Finitely supported `k`-valued functions on the finite words over `Fin d`. -/
abbrev Space := List (Fin d) →₀ k

/-- `S_j`: append the letter `j`. -/
noncomputable def opS (j : Fin d) : Module.End k (Space k d) :=
  Finsupp.lmapDomain k k fun w : List (Fin d) => w ++ [j]

/-- `T_j`: delete a final letter `j`; kill a word that does not end in `j`. -/
noncomputable def opT (j : Fin d) : Module.End k (Space k d) :=
  Finsupp.linearCombination k fun w : List (Fin d) =>
    if w.getLast? = some j then Finsupp.single w.dropLast (1 : k) else 0

/-- `Q`: the projection onto the empty word. -/
noncomputable def opQ : Module.End k (Space k d) :=
  Finsupp.linearCombination k fun w : List (Fin d) =>
    if w = [] then Finsupp.single [] (1 : k) else 0

/-- Split a word by its first letter. -/
noncomputable def splitMap : Space k d →ₗ[k] k × (Fin d → Space k d) :=
  Finsupp.linearCombination k fun w : List (Fin d) =>
    match w with
    | [] => ((1 : k), (0 : Fin d → Space k d))
    | i :: w' => ((0 : k), Pi.single i (Finsupp.single w' (1 : k)))

/-- Glue a coefficient on the empty word and `d` spaces of words back together. -/
noncomputable def glueMap : k × (Fin d → Space k d) →ₗ[k] Space k d :=
  LinearMap.coprod (Finsupp.lsingle [])
    (∑ i : Fin d, Finsupp.lmapDomain k k (List.cons i) ∘ₗ LinearMap.proj i)

variable {k d}

/-! ### The operators on basis vectors -/

theorem opS_single (j : Fin d) (w : List (Fin d)) (c : k) :
    opS k d j (Finsupp.single w c) = Finsupp.single (w ++ [j]) c := by
  rw [opS, Finsupp.lmapDomain_apply, Finsupp.mapDomain_single]

theorem opT_single (j : Fin d) (w : List (Fin d)) (c : k) :
    opT k d j (Finsupp.single w c)
      = if w.getLast? = some j then Finsupp.single w.dropLast c else 0 := by
  rw [opT, Finsupp.linearCombination_single]
  split_ifs
  · rw [Finsupp.smul_single, smul_eq_mul, mul_one]
  · rw [smul_zero]

theorem opQ_single (w : List (Fin d)) (c : k) :
    opQ k d (Finsupp.single w c) = if w = [] then Finsupp.single [] c else 0 := by
  rw [opQ, Finsupp.linearCombination_single]
  split_ifs
  · rw [Finsupp.smul_single, smul_eq_mul, mul_one]
  · rw [smul_zero]

theorem opT_single_nil (j : Fin d) (c : k) : opT k d j (Finsupp.single [] c) = 0 := by
  rw [opT_single, if_neg (by simp)]

theorem opT_single_singleton (j i : Fin d) (c : k) :
    opT k d j (Finsupp.single [i] c) = if i = j then Finsupp.single [] c else 0 := by
  rw [opT_single, List.getLast?_singleton, List.dropLast_singleton]
  by_cases h : i = j
  · rw [if_pos (by rw [h]), if_pos h]
  · rw [if_neg (fun h' => h (Option.some.inj h')), if_neg h]

theorem opT_single_cons_cons (j i b : Fin d) (w : List (Fin d)) (c : k) :
    opT k d j (Finsupp.single (i :: b :: w) c)
      = Finsupp.lmapDomain k k (List.cons i) (opT k d j (Finsupp.single (b :: w) c)) := by
  rw [opT_single, opT_single, List.getLast?_cons_cons, List.dropLast_cons_cons]
  split_ifs
  · rw [Finsupp.lmapDomain_apply, Finsupp.mapDomain_single]
  · rw [map_zero]

/-! ### The Toeplitz relations -/

theorem opT_mul_opS (i j : Fin d) :
    opT k d i * opS k d j = if i = j then 1 else 0 := by
  refine Finsupp.lhom_ext fun w c => ?_
  rw [Module.End.mul_apply, opS_single, opT_single, List.getLast?_concat, List.dropLast_concat]
  by_cases h : i = j
  · rw [if_pos (by rw [h]), if_pos h, Module.End.one_apply]
  · rw [if_neg (fun h' => h (Option.some.inj h').symm), if_neg h, LinearMap.zero_apply]

theorem sum_opS_mul_opT :
    ∑ j : Fin d, opS k d j * opT k d j = 1 - opQ k d := by
  refine Finsupp.lhom_ext fun w c => ?_
  rw [LinearMap.sub_apply, Module.End.one_apply, opQ_single, LinearMap.sum_apply]
  rcases List.eq_nil_or_concat w with rfl | ⟨L, b, rfl⟩
  · have h0 : ∀ j : Fin d, (opS k d j * opT k d j) (Finsupp.single [] c) = 0 := by
      intro j
      rw [Module.End.mul_apply, opT_single_nil, map_zero]
    rw [Finset.sum_eq_zero fun j _ => h0 j, if_pos rfl, sub_self]
  · rw [List.concat_eq_append]
    have hne : L ++ [b] ≠ [] := by simp
    rw [if_neg hne, sub_zero, Finset.sum_eq_single b]
    · rw [Module.End.mul_apply, opT_single, if_pos List.getLast?_concat, List.dropLast_concat,
        opS_single]
    · intro j _ hj
      rw [Module.End.mul_apply, opT_single, List.getLast?_concat,
        if_neg (fun h => hj (Option.some.inj h).symm), map_zero]
    · intro hb
      exact absurd (Finset.mem_univ b) hb

theorem opQ_eq :
    opQ k d = (Finsupp.lsingle [] : k →ₗ[k] Space k d) ∘ₗ (Finsupp.lapply [] : Space k d →ₗ[k] k) := by
  refine Finsupp.lhom_ext fun w c => ?_
  rw [opQ_single]
  show _ = Finsupp.single [] ((Finsupp.single w c : Space k d) [])
  rw [Finsupp.single_apply]
  by_cases h : w = []
  · rw [if_pos h, if_pos h]
  · rw [if_neg h, if_neg h, Finsupp.single_zero]

/-- `Q` has rank one. -/
theorem finiteRank_opQ : FiniteRank (opQ k d) := by
  rw [opQ_eq]
  exact FiniteRank.of_range_le (S := LinearMap.range (Finsupp.lsingle [] : k →ₗ[k] Space k d))
    (LinearMap.range_comp_le_range _ _)

/-! ### The first-letter splitting -/

theorem splitMap_single_nil (c : k) :
    splitMap k d (Finsupp.single [] c) = (c, 0) := by
  rw [splitMap, Finsupp.linearCombination_single]
  refine Prod.ext ?_ ?_
  · simp
  · simp

theorem splitMap_single_cons (i : Fin d) (w : List (Fin d)) (c : k) :
    splitMap k d (Finsupp.single (i :: w) c) = (0, Pi.single i (Finsupp.single w c)) := by
  rw [splitMap, Finsupp.linearCombination_single]
  refine Prod.ext ?_ (funext fun j => ?_)
  · simp
  · by_cases h : j = i
    · subst h
      simp
    · simp [Pi.single_eq_of_ne h]

theorem splitMap_lmapDomain_cons (i : Fin d) (g : Space k d) :
    splitMap k d (Finsupp.lmapDomain k k (List.cons i) g) = (0, Pi.single i g) := by
  have h : splitMap k d ∘ₗ Finsupp.lmapDomain k k (List.cons i)
      = LinearMap.inr k k (Fin d → Space k d) ∘ₗ LinearMap.single k (fun _ : Fin d => Space k d) i := by
    refine Finsupp.lhom_ext fun w c => ?_
    rw [LinearMap.comp_apply, Finsupp.lmapDomain_apply, Finsupp.mapDomain_single,
      splitMap_single_cons]
    rfl
  exact LinearMap.congr_fun h g

theorem glueMap_apply (c : k) (f : Fin d → Space k d) :
    glueMap k d (c, f) = Finsupp.single [] c + ∑ i : Fin d, Finsupp.lmapDomain k k (List.cons i) (f i) := by
  simp [glueMap, LinearMap.coprod_apply, LinearMap.sum_apply]

theorem glueMap_single (i : Fin d) (g : Space k d) :
    glueMap k d (0, Pi.single i g) = Finsupp.lmapDomain k k (List.cons i) g := by
  rw [glueMap_apply, Finsupp.single_zero, zero_add, Finset.sum_eq_single i]
  · rw [Pi.single_eq_same]
  · intro j _ hj
    rw [Pi.single_eq_of_ne hj, map_zero]
  · intro hi
    exact absurd (Finset.mem_univ i) hi

theorem glueMap_inl (c : k) : glueMap k d (c, 0) = Finsupp.single [] c := by
  rw [glueMap_apply]
  simp

theorem glueMap_splitMap : glueMap k d ∘ₗ splitMap k d = LinearMap.id := by
  refine Finsupp.lhom_ext fun w c => ?_
  rw [LinearMap.comp_apply, LinearMap.id_apply]
  cases w with
  | nil => rw [splitMap_single_nil, glueMap_inl]
  | cons i w =>
    rw [splitMap_single_cons, glueMap_single, Finsupp.lmapDomain_apply, Finsupp.mapDomain_single]

theorem splitMap_glueMap : splitMap k d ∘ₗ glueMap k d = LinearMap.id := by
  refine LinearMap.prod_ext ?_ ?_
  · refine LinearMap.ext_ring ?_
    show splitMap k d (glueMap k d ((1 : k), 0)) = ((1 : k), 0)
    rw [glueMap_inl, splitMap_single_nil]
  · refine LinearMap.pi_ext' fun i => Finsupp.lhom_ext fun w c => ?_
    show splitMap k d (glueMap k d (0, Pi.single i (Finsupp.single w c)))
      = (0, Pi.single i (Finsupp.single w c))
    rw [glueMap_single, splitMap_lmapDomain_cons]

variable (k d) in
/-- **The first-letter splitting** `Space k d ≃ k × (Fin d → Space k d)`. -/
noncomputable def split : Space k d ≃ₗ[k] k × (Fin d → Space k d) :=
  LinearEquiv.ofLinear (splitMap k d) (glueMap k d) splitMap_glueMap glueMap_splitMap

theorem conj_split_apply (X : Module.End k (Space k d)) (p : k × (Fin d → Space k d)) :
    (split k d).conj X p = splitMap k d (X (glueMap k d p)) :=
  rfl

theorem piMap_single (X : Module.End k (Space k d)) (i : Fin d) (g : Space k d) :
    piMap (Fin d) X (Pi.single i g) = Pi.single i (X g) := by
  funext i'
  by_cases h : i' = i
  · subst h
    simp
  · simp [Pi.single_eq_of_ne h]

/-! ### Both generators are tame -/

/-- **The defect of `S_j` has rank one**: it sends the empty-word coefficient to
the one-letter word `j`. -/
theorem defect_opS (j : Fin d) :
    defect (split k d) (opS k d j)
      = (LinearMap.inr k k (Fin d → Space k d) ∘ₗ LinearMap.single k (fun _ : Fin d => Space k d) j ∘ₗ
          (Finsupp.lsingle [] : k →ₗ[k] Space k d)) ∘ₗ LinearMap.fst k k (Fin d → Space k d) := by
  refine LinearMap.prod_ext ?_ ?_
  · refine LinearMap.ext_ring ?_
    show splitMap k d (opS k d j (glueMap k d ((1 : k), 0))) - sep k (Fin d) (opS k d j) ((1 : k), 0)
      = (0, Pi.single j (Finsupp.single [] (1 : k)))
    rw [glueMap_inl, opS_single, List.nil_append, splitMap_single_cons]
    refine Prod.ext ?_ ?_
    · simp [sep]
    · simp [sep, piMap]
  · refine LinearMap.pi_ext' fun i => Finsupp.lhom_ext fun w c => ?_
    show splitMap k d (opS k d j (glueMap k d (0, Pi.single i (Finsupp.single w c))))
        - sep k (Fin d) (opS k d j) (0, Pi.single i (Finsupp.single w c))
      = (0, Pi.single j (Finsupp.single [] (0 : k)))
    rw [glueMap_single, Finsupp.lmapDomain_apply, Finsupp.mapDomain_single, opS_single,
      List.cons_append, splitMap_single_cons]
    refine Prod.ext ?_ ?_
    · simp [sep]
    · show Pi.single i (Finsupp.single (w ++ [j]) c) - piMap (Fin d) (opS k d j)
          (Pi.single i (Finsupp.single w c)) = Pi.single j (Finsupp.single [] (0 : k))
      rw [piMap_single, opS_single, sub_self, Finsupp.single_zero, Pi.single_zero]

/-- **The defect of `T_j` has rank one**: it reads the empty-word coefficient of
coordinate `j`. -/
theorem defect_opT (j : Fin d) :
    defect (split k d) (opT k d j)
      = LinearMap.inl k k (Fin d → Space k d) ∘ₗ (Finsupp.lapply [] : Space k d →ₗ[k] k) ∘ₗ
          LinearMap.proj j ∘ₗ LinearMap.snd k k (Fin d → Space k d) := by
  refine LinearMap.prod_ext ?_ ?_
  · refine LinearMap.ext_ring ?_
    show splitMap k d (opT k d j (glueMap k d ((1 : k), 0))) - sep k (Fin d) (opT k d j) ((1 : k), 0)
      = (((0 : Fin d → Space k d) j : Space k d) [], 0)
    rw [glueMap_inl, opT_single_nil, map_zero]
    refine Prod.ext ?_ ?_
    · simp [sep]
    · simp [sep, piMap]
  · refine LinearMap.pi_ext' fun i => Finsupp.lhom_ext fun w c => ?_
    show splitMap k d (opT k d j (glueMap k d (0, Pi.single i (Finsupp.single w c))))
        - sep k (Fin d) (opT k d j) (0, Pi.single i (Finsupp.single w c))
      = (((Pi.single i (Finsupp.single w c) : Fin d → Space k d) j : Space k d) [], 0)
    rw [glueMap_single, Finsupp.lmapDomain_apply, Finsupp.mapDomain_single]
    cases w with
    | nil =>
      rw [opT_single_singleton]
      refine Prod.ext ?_ ?_
      · by_cases h : i = j
        · subst h
          simp [sep, splitMap_single_nil]
        · simp [sep, h, Pi.single_eq_of_ne (Ne.symm h)]
      · show (splitMap k d (if i = j then Finsupp.single [] c else 0)).2
            - piMap (Fin d) (opT k d j) (Pi.single i (Finsupp.single [] c)) = 0
        rw [piMap_single, opT_single_nil, Pi.single_zero, sub_zero]
        split_ifs
        · rw [splitMap_single_nil]
        · rw [map_zero]; rfl
    | cons b w =>
      rw [opT_single_cons_cons, splitMap_lmapDomain_cons]
      refine Prod.ext ?_ ?_
      · simp [sep, Finsupp.single_apply]
      · show Pi.single i (opT k d j (Finsupp.single (b :: w) c))
            - piMap (Fin d) (opT k d j) (Pi.single i (Finsupp.single (b :: w) c)) = 0
        rw [piMap_single, sub_self]

theorem tame_opS (j : Fin d) : Tame (split k d) (opS k d j) := by
  show FiniteRank (defect (split k d) (opS k d j))
  rw [defect_opS]
  exact FiniteRank.of_range_le
    (S := LinearMap.range (LinearMap.inr k k (Fin d → Space k d) ∘ₗ
      LinearMap.single k (fun _ : Fin d => Space k d) j ∘ₗ (Finsupp.lsingle [] : k →ₗ[k] Space k d)))
    (LinearMap.range_comp_le_range (LinearMap.fst k k (Fin d → Space k d)) _)

theorem tame_opT (j : Fin d) : Tame (split k d) (opT k d j) := by
  show FiniteRank (defect (split k d) (opT k d j))
  rw [defect_opT]
  exact FiniteRank.of_range_le (S := LinearMap.range (LinearMap.inl k k (Fin d → Space k d)))
    (LinearMap.range_comp_le_range _ _)

end ToeplitzWords
