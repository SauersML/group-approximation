/-!
# Osin's Lemma 9.4 on class words: single-side pairs and the class-word cases

Osin (math/0411039v3, §9), proof of Lemma 9.4, Cases 1 and 2.  A connector pair on the class words
whose source and target segments lie inside single side words is a connector pair on the side words,
so the side-level Case 1 and Case 2 refute it.  The other pairs are the two residual shapes of
`OsinLemma94ClassPairShapes`.

* `UnboundEstimate.WordConnectorPair.exists_shift`: a connector pair moves to other words when its
  four endpoint vertices are the vertices of the new words at shifted positions, up to one common
  left factor.  The word metric is left invariant.
* `OsinLemma94RealizedPolygons.corner_mul_listVal_eq_mod`, `listVal_flatMap_take_rotate`: the
  corners of the polygon read cyclically from any side.
* `OsinLemma94ClassPolygons.corner_mul_prefix`, `vertex_eq_of_sideAt`: a vertex of a class word
  inside the word of side `s` is the vertex of side `s`, up to the left factor
  `(P.corner k (classBase k % k_i))⁻¹`.  The gaps have value one.
* `OsinLemma94ClassPolygons.exists_sidePair_of_singleSide`: a single-side class-word pair gives a
  side-level pair, backwards when the class pair is.
* `osinLemma94ClassCases_of_residuals`: `OsinLemma94ClassCasesInput` from the side-level Case 1 and
  Case 2 and the two residual shapes.

variable {G : Type u} [Group G] {Lambda : Type w}

/-- **A connector pair moves to shifted words.**  If the vertices of the source (target) word of
`C` at the positions between `off` and `off + |word' s|` (between `off'` and `off' + |word' t|`)
are `h` times the vertices of `word' s` (of `word' t`) at the shifted positions, the same
connectors give a pair on the new words, backwards when `C` is. -/
theorem exists_shift {D : RelGenSet G Lambda} {v v' : ℕ → G}
    {word word' : ℕ → List (RelLetter G Lambda)} {n n' eps : ℕ} {A B A' B' : Set ℕ}
    (C : WordConnectorPair D v word n A B eps) (h : G) {s t off off' : ℕ}
    (hs : s < n') (ht : t < n') (hsA : s ∈ A') (htB : t ∈ B') (hst : s ≠ t)
    (hsadm : HullSC.RelWord.IsAdmissible D (word' s))
    (htadm : HullSC.RelWord.IsAdmissible D (word' t))
    (ha : off ≤ C.a) (ha' : C.a' ≤ off + (word' s).length)
    (hb : off' ≤ C.b) (hb' : off' ≤ C.b') (hbl : C.b ≤ off' + (word' t).length)
    (hbl' : C.b' ≤ off' + (word' t).length)
    (hvs : ∀ x, off ≤ x → x ≤ off + (word' s).length →
      OsinComponents.vertex (v C.source) (word C.source) x =
        h * OsinComponents.vertex (v' s) (word' s) (x - off))
    (hvt : ∀ x, off' ≤ x → x ≤ off' + (word' t).length →
      OsinComponents.vertex (v C.target) (word C.target) x =
        h * OsinComponents.vertex (v' t) (word' t) (x - off')) :
    ∃ C' : WordConnectorPair D v' word' n' A' B' eps, C.b' < C.b → C'.b' < C'.b := by
  have hfwd := C.source_forward
  have eA := hvs C.a ha (by omega)
  have eA' := hvs C.a' (by omega) ha'
  have eB := hvt C.b hb hbl
  have eB' := hvt C.b' hb' hbl'
  obtain ⟨hletS, hmulS, hlenS⟩ := C.start_geodesic
  obtain ⟨hletE, hmulE, hlenE⟩ := C.end_geodesic
  have hlong := C.source_long
  have hlongT := C.target_long
  rw [eA, eA', wordDist_left_invariant] at hlong
  rw [eB, eB', wordDist_left_invariant] at hlongT
  rw [eB, eA, mul_assoc] at hmulS
  rw [eB, eA, wordDist_left_invariant] at hlenS
  rw [eA', eB', mul_assoc] at hmulE
  rw [eA', eB', wordDist_left_invariant] at hlenE
  refine ⟨{ source := s, target := t, source_lt := hs, target_lt := ht, source_mem := hsA,
    target_mem := htB, distinct := hst, source_admissible := hsadm,
    target_admissible := htadm, a := C.a - off, a' := C.a' - off, b := C.b - off',
    b' := C.b' - off', a_le := by omega, a'_le := by omega, b_le := by omega,
    b'_le := by omega, source_forward := by omega, source_long := hlong,
    target_long := hlongT, startConnector := C.startConnector,
    endConnector := C.endConnector, start_geodesic := ⟨hletS, mul_left_cancel hmulS, hlenS⟩,
    end_geodesic := ⟨hletE, mul_left_cancel hmulE, hlenE⟩, start_short := C.start_short,
    end_short := C.end_short }, fun hback => ?_⟩
  show C.b' - off' < C.b - off'
  omega

end UnboundEstimate.WordConnectorPair

namespace OsinLemma94RealizedPolygons

variable {G : Type u} [Group G] {Lambda : Type w}
  {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {lambda c : ℝ} {eps : ℕ}
  {Delta : DiscDiagram.{u, w, v} W}
  {cuts : SectionCuts D lambda c Delta.boundaryWord}
  {S : GloballyDistinguishedSectionFamily D lambda c eps Delta cuts}

/-- One step around the polygon: `corner_t · value(side t) = corner_{(t + 1) mod k_i}`. -/
theorem corner_mul_listVal_eq_mod (P : OsinLemma94RealizedPolygons S) (k : Fin P.count) {t : ℕ}
    (ht : t < P.sideCount k) :
    P.corner k t * RelLetter.listVal (P.word k t) = P.corner k ((t + 1) % P.sideCount k) := by
  rw [← P.corner_step]
  by_cases h : t + 1 < P.sideCount k
  · rw [Nat.mod_eq_of_lt h]
  · have e : t + 1 = P.sideCount k := by omega
    rw [e, Nat.mod_self, P.corner_closed]

/-- **The sides read cyclically from side `r`**: the first `m` of them multiply to
`corner_r⁻¹ · corner_{(m + r) mod k_i}`. -/
theorem listVal_flatMap_take_rotate (P : OsinLemma94RealizedPolygons S) (k : Fin P.count)
    (r : ℕ) :
    ∀ m, m ≤ P.sideCount k →
      RelLetter.listVal ((((List.range (P.sideCount k)).rotate r).take m).flatMap (P.word k)) =
        (P.corner k (r % P.sideCount k))⁻¹ * P.corner k ((m + r) % P.sideCount k) := by
  intro m
  induction m with
  | zero =>
      intro _
      rw [List.take_zero, List.flatMap_nil, RelLetter.listVal_nil, Nat.zero_add, inv_mul_cancel]
  | succ m ih =>
      intro hm
      have hn : 0 < P.sideCount k := by omega
      have hget : ((List.range (P.sideCount k)).rotate r)[m]? =
          some ((m + r) % P.sideCount k) := by
        rw [List.getElem?_rotate (by rw [List.length_range]; omega), List.length_range,
          List.getElem?_range (Nat.mod_lt _ hn)]
      rw [List.take_succ, hget, Option.toList_some, List.flatMap_append, List.flatMap_singleton,
        HullSC.RelWord.listVal_append, ih (by omega), mul_assoc,
        P.corner_mul_listVal_eq_mod k (Nat.mod_lt _ hn), Nat.mod_add_mod, Nat.add_right_comm]

end OsinLemma94RealizedPolygons

namespace OsinLemma94ClassPolygons

variable {G : Type u} [Group G] {Lambda : Type w}
  {W : Set (List (RelLetter G Lambda))}
  {D : RelGenSet G Lambda} {lambda c : ℝ} {eps : ℕ}
  {Delta : DiscDiagram.{u, w, v} W}
  {cuts : SectionCuts D lambda c Delta.boundaryWord}
  {S : GloballyDistinguishedSectionFamily D lambda c eps Delta cuts}
  {P : OsinLemma94RealizedPolygons S}

theorem mem_of_sideAt (Q : OsinLemma94ClassPolygons P) {k : Fin P.count} {i s off : ℕ}
    (h : Q.SideAt k i s off) : s ∈ Q.classSides k i := by
  obtain ⟨L1, L2, hL, -⟩ := h
  rw [hL]
  exact List.mem_append_right _ List.mem_cons_self

/-- **The corner of a class before one of its sides.**  Reading the class words from `classBase k`
up to side `s` of class `i` multiplies the side words from side `classBase k % k_i` up to side `s`,
since the gaps have value one. -/
theorem corner_mul_prefix (Q : OsinLemma94ClassPolygons P) (k : Fin P.count) {i s : ℕ}
    (hi : i < Q.classCount k) {L1 L2 : List ℕ} (hL : Q.classSides k i = L1 ++ s :: L2) :
    Q.corner k i *
        RelLetter.listVal
          (L1.flatMap fun t => P.word k t ++ Embedded.dartWord S.diagram (Q.gap k t)) =
      (P.corner k (Q.classBase k % P.sideCount k))⁻¹ * P.corner k s := by
  have hrange : List.range (Q.classCount k) =
      List.range i ++ i :: (List.range (Q.classCount k - (i + 1))).map ((i + 1) + ·) := by
    have e : Q.classCount k = (i + 1) + (Q.classCount k - (i + 1)) := by omega
    conv_lhs => rw [e]
    rw [List.range_add, List.range_succ, List.append_assoc, List.singleton_append]
  have hsplit : (List.range (P.sideCount k)).rotate (Q.classBase k) =
      ((List.range i).flatMap (Q.classSides k) ++ L1) ++ s :: (L2 ++
        ((List.range (Q.classCount k - (i + 1))).map ((i + 1) + ·)).flatMap (Q.classSides k)) := by
    rw [← Q.sides_eq k, hrange, List.flatMap_append, List.flatMap_cons, hL]
    simp only [List.append_assoc, List.cons_append]
  set A := (List.range i).flatMap (Q.classSides k) ++ L1 with hA
  have hmemR : ∀ t ∈ A, ∃ i' < Q.classCount k, t ∈ Q.classSides k i' := by
    intro t ht
    have htR : t ∈ (List.range (P.sideCount k)).rotate (Q.classBase k) := by
      rw [hsplit]
      exact List.mem_append_left _ ht
    rw [← Q.sides_eq k, List.mem_flatMap] at htR
    obtain ⟨i', hi', hti'⟩ := htR
    exact ⟨i', List.mem_range.mp hi', hti'⟩
  have hcorner : Q.corner k i *
      RelLetter.listVal
        (L1.flatMap fun t => P.word k t ++ Embedded.dartWord S.diagram (Q.gap k t)) =
        RelLetter.listVal
          (A.flatMap fun t => P.word k t ++ Embedded.dartWord S.diagram (Q.gap k t)) := by
    have e : (List.range i).flatMap (Q.word k) =
        ((List.range i).flatMap (Q.classSides k)).flatMap
          fun t => P.word k t ++ Embedded.dartWord S.diagram (Q.gap k t) := by
      rw [List.flatMap_assoc]
      exact congrArg (fun g => (List.range i).flatMap g) (funext fun i' => Q.word_eq k i')
    rw [hA, List.flatMap_append, HullSC.RelWord.listVal_append, ← e]
    rfl
  have hgaps : RelLetter.listVal
      (A.flatMap fun t => P.word k t ++ Embedded.dartWord S.diagram (Q.gap k t)) =
        RelLetter.listVal (A.flatMap (P.word k)) :=
    listVal_flatMap_append A (P.word k) (fun t => Embedded.dartWord S.diagram (Q.gap k t))
      (fun t ht => by
        obtain ⟨i', hi', hti'⟩ := hmemR t ht
        exact Q.gap_value k i' hi' t hti')
  have hlens := congrArg List.length hsplit
  rw [List.length_rotate, List.length_range, List.length_append, List.length_cons] at hlens
  have hlen : A.length ≤ P.sideCount k := by omega
  have hn : 0 < P.sideCount k := by omega
  have htake : ((List.range (P.sideCount k)).rotate (Q.classBase k)).take A.length = A := by
    rw [hsplit]
    exact List.take_left
  have hs : s = (A.length + Q.classBase k) % P.sideCount k := by
    have h1 : ((List.range (P.sideCount k)).rotate (Q.classBase k))[A.length]? = some s := by
      rw [hsplit, List.getElem?_append_right (le_refl _), Nat.sub_self, List.getElem?_cons_zero]
    rw [List.getElem?_rotate (by rw [List.length_range]; omega), List.length_range,
      List.getElem?_range (Nat.mod_lt _ hn)] at h1
    exact (Option.some_inj.mp h1).symm
  rw [hcorner, hgaps, ← htake, P.listVal_flatMap_take_rotate k (Q.classBase k) A.length hlen, hs]

/-- **A vertex of a class word inside the word of side `s`** is the vertex of side `s` at the
shifted position, up to the left factor `(P.corner k (classBase k % k_i))⁻¹`. -/
theorem vertex_eq_of_sideAt (Q : OsinLemma94ClassPolygons P) {k : Fin P.count} {i s off x : ℕ}
    (hi : i < Q.classCount k) (hat : Q.SideAt k i s off) (h1 : off ≤ x)
    (h2 : x ≤ off + (P.word k s).length) :
    OsinComponents.vertex (Q.corner k i) (Q.word k i) x =
      (P.corner k (Q.classBase k % P.sideCount k))⁻¹ *
        OsinComponents.vertex (P.corner k s) (P.word k s) (x - off) := by
  obtain ⟨L1, L2, hL, rfl⟩ := hat
  have hword : Q.word k i =
      (L1.flatMap fun t => P.word k t ++ Embedded.dartWord S.diagram (Q.gap k t)) ++
        (P.word k s ++ (Embedded.dartWord S.diagram (Q.gap k s) ++
          L2.flatMap fun t => P.word k t ++ Embedded.dartWord S.diagram (Q.gap k t))) := by
    rw [Q.word_eq, hL, List.flatMap_append, List.flatMap_cons]
    simp only [List.append_assoc]
  rw [HullSC.vertex_eq_mul_listVal_take, HullSC.vertex_eq_mul_listVal_take, hword,
    List.take_append, List.take_of_length_le h1, List.take_append_of_le_length (by omega),
    HullSC.RelWord.listVal_append]
  simp only [← mul_assoc]
  rw [Q.corner_mul_prefix k hi hL]

/-- **A single-side class-word pair is a side-level pair.**  The source side lies in an (A1)
class, so it is an (A1) side; the target side lies in a class that is not short, so it is not
short; different classes have different sides. -/
theorem exists_sidePair_of_singleSide (Q : OsinLemma94ClassPolygons P) (k : Fin P.count)
    (C : WordConnectorPair (symmetricLabelAlphabet D) (Q.corner k) (Q.word k) (Q.classCount k)
      (Q.relatorClasses k) (Q.longClasses k) eps)
    (hsingle : Q.SingleSidePair k C) :
    ∃ C' : WordConnectorPair (symmetricLabelAlphabet D) (P.corner k) (P.word k) (P.sideCount k)
      (P.relatorSides k) (P.longSides k) eps, C.b' < C.b → C'.b' < C'.b := by
  obtain ⟨⟨s, off, hsAt, ha, ha'⟩, t, off', htAt, hb, hb', hbl, hbl'⟩ := hsingle
  have hsmem := Q.mem_of_sideAt hsAt
  have htmem := Q.mem_of_sideAt htAt
  have hadm : ∀ {i x : ℕ}, x ∈ Q.classSides k i →
      HullSC.RelWord.IsAdmissible (symmetricLabelAlphabet D) (Q.word k i) →
        HullSC.RelWord.IsAdmissible (symmetricLabelAlphabet D) (P.word k x) := by
    intro i x hx hw letter hletter
    refine hw letter ?_
    rw [Q.word_eq]
    exact List.mem_flatMap.mpr ⟨x, hx, List.mem_append_left _ hletter⟩
  have hsrel : s ∈ P.relatorSides k := by
    obtain ⟨-, j, hj⟩ : 2 ≤ Q.classCount k ∧ ∃ j, Q.classKind k C.source = .cell j :=
      C.source_mem
    exact ⟨j, (Q.kind_eq k C.source C.source_lt s hsmem).trans hj⟩
  have htlong : t ∈ P.longSides k := by
    show P.kind k t ≠ .short
    rw [Q.kind_eq k C.target C.target_lt t htmem]
    exact C.target_mem
  have hst : s ≠ t := fun h =>
    Q.classSides_disjoint k C.source_lt C.target_lt C.distinct hsmem (h ▸ htmem)
  exact C.exists_shift ((P.corner k (Q.classBase k % P.sideCount k))⁻¹)
    (Q.lt_sideCount_of_mem k _ C.source_lt s hsmem)
    (Q.lt_sideCount_of_mem k _ C.target_lt t htmem) hsrel htlong hst
    (hadm hsmem C.source_admissible) (hadm htmem C.target_admissible) ha ha' hb hb' hbl hbl'
    (fun _ h1 h2 => Q.vertex_eq_of_sideAt C.source_lt hsAt h1 h2)
    (fun _ h1 h2 => Q.vertex_eq_of_sideAt C.target_lt htAt h1 h2)

end OsinLemma94ClassPolygons

/-- **Cases 1 and 2 of Lemma 9.4 on class words**, from the side-level cases on single-side pairs
and the two residual shapes. -/
theorem osinLemma94ClassCases_of_residuals
    (hone : OsinLemma94CaseOneInput.{u, w, v}) (htwo : OsinLemma94CaseTwoInput.{u, w, v})
    (hendpoint : OsinLemma94ClassCaseGapEndpointStatement.{u, w, v})
    (hspan : OsinLemma94ClassCaseGapSpanStatement.{u, w, v}) :
    OsinLemma94ClassCasesInput.{u, w, v} := by
  intro G _ Lambda D hhyper lambda c mu hlambda hlambda1 hc hmu hmu16
  obtain ⟨e1, hone1⟩ := hone D hhyper lambda c mu hlambda hlambda1 hc hmu hmu16
  obtain ⟨e2, htwo1⟩ := htwo D hhyper lambda c mu hlambda hlambda1 hc hmu hmu16
  obtain ⟨e3, hend1⟩ := hendpoint D hhyper lambda c mu hlambda hlambda1 hc hmu hmu16
  obtain ⟨e4, hspan1⟩ := hspan D hhyper lambda c mu hlambda hlambda1 hc hmu hmu16
  refine ⟨max e1 (max e2 (max e3 e4)), fun eps heps => ?_⟩
  simp only [max_le_iff] at heps
  obtain ⟨he1, he2, he3, he4⟩ := heps
  obtain ⟨r1, hr1, hone2⟩ := hone1 eps he1
  obtain ⟨r2, _, htwo2⟩ := htwo1 eps he2
  obtain ⟨r3, _, hend2⟩ := hend1 eps he3
  obtain ⟨r4, _, hspan2⟩ := hspan1 eps he4
  refine ⟨max r1 (max r2 (max r3 r4)), lt_of_lt_of_le hr1 (le_max_left _ _), fun rho hrho => ?_⟩
  simp only [max_le_iff] at hrho
  obtain ⟨hq1, hq2, hq3, hq4⟩ := hrho
  intro W hW Delta cuts hleast hcells hbelow S hcard hmin P hmax Q k C hback
  by_cases hsingle : Q.SingleSidePair k C
  · obtain ⟨C', hback'⟩ := Q.exists_sidePair_of_singleSide k C hsingle
    by_cases hcut : P.kind k C'.target = .cutting
    · exact htwo2 rho hq2 W hW Delta cuts hleast hcells S hcard hmin P hmax k C' (hback' hback)
        hcut
    · exact hone2 rho hq1 W hW Delta cuts hleast hcells hbelow S hcard hmin P hmax k C'
        (hback' hback) hcut
  · by_cases hgap : Q.GapEndpoint k C
    · exact hend2 rho hq3 W hW Delta cuts hleast hcells hbelow S hcard hmin P hmax Q k C hback hgap
    · exact hspan2 rho hq4 W hW Delta cuts hleast hcells hbelow S hcard hmin P hmax Q k C hback
        hgap hsingle
