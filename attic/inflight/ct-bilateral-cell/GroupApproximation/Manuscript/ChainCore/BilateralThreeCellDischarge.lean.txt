/-!
# `prop:bilateral-three`: the three-coordinate cell, closed

`non_mf_groups_exist.tex`, proof of Proposition `prop:bilateral-three`
(`\label{prop:bilateral-three}`, tex 1550–1593 at origin/main 68481e4d7):

> Take a clopen $P$ with $T(P)\subseteq P$ and nonempty $D=P\setminus T(P)$.  Put $p=1_P$,
> $d=1_D$, and $f_{ab}=u^a d u^{-b}$ for $a,b\in\mathbb Z$. [...]
> The unital subalgebra $S\subseteq pR_Xp$ generated by $s=up$, $t=pu^{-1}$ and $kp$
> satisfies $ts=p$, $st=p-d$.  It is finitely generated, so $L=\EL_3(S)$ has
> property~\textup{(T)}.  Embed it in $\GL_3(R_X)$ by adjoining identity on $1-p$. [...]
> Then $c\in C_B(L)$ and $WcW^{-1}=e_{13}(f_{0,-2})$, $\delta=[WcW^{-1},\ell]=e_{23}(-f_{0,-2})$,
> $[y,\delta]=z^{-1}$. [...] So $\langle z\rangle$ is a nontrivial finite central subgroup of
> $B$, of order $\operatorname{char}(k)$. [...] The identities
> $[z,e_{31}(f_{-2,0})]=e_{31}(f_{-1,0})$, $[e_{23}(f_{0,-1}),e_{31}(f_{-1,0})]=e_{21}(d)$
> then kill the defect root.

`bilateralThreeCellStatement : BilateralThreeCellStatement` is the cell piece of
`printedBilateralThree_of_pieces`:

* `S = cellRing T k hP hTP`, the subring of the corner ring `pR_Xp` generated by `s`, `t` and
  `kp` (the elements `p (a·1) p = a p`, `a ∈ k`), finitely generated
  (`isFinitelyGeneratedRing_cellRing`);
* `L = coreGroup` of its coefficients, the image of `EL₃(S)` under the injective corner
  embedding, with property (T) by `hasKazhdanPropertyT_range_of_injective` and the
  repository's Ershov--Jaikin-Zapirain theorem;
* `W`, `c`, `y`, `ℓ`, `z` from `BilateralThreeCellMatrices`, `g₁ = e₃₁(f_{-2,0})`,
  `g₂ = e₂₃(f_{0,-1})`, and `z^{char k} = 1` (`natCast_ringChar_eq_zero`).
-/

section Ring

variable {X : Type} [TopologicalSpace X] (T : X ≃ₜ X) (k : Type) [Field k] {P : Set X}

/-- The generators of `S` in the corner `pR_Xp` (tex 1554–1555): `s = up`, `t = pu⁻¹`, and `kp`
as the elements `p (a·1) p`, `a ∈ k`. -/
def cellGeneratorSet (hP : IsClopen P) (hTP : T '' P ⊆ P) :
    Set (bilateralCell T k hP hTP).Corner :=
  {(bilateralCell T k hP hTP).sCorner, (bilateralCell T k hP hTP).tCorner} ∪
    Set.range fun a : k =>
      (⟨(bilateralCell T k hP hTP).p * coeff T k (LocallyConstant.const X a) *
          (bilateralCell T k hP hTP).p, coeff T k (LocallyConstant.const X a), rfl⟩ :
        (bilateralCell T k hP hTP).Corner)

theorem cellGeneratorSet_finite [Finite k] (hP : IsClopen P) (hTP : T '' P ⊆ P) :
    (cellGeneratorSet T k hP hTP).Finite :=
  (Set.toFinite _).union (Set.finite_range _)

/-- **"The unital subalgebra $S\subseteq pR_Xp$ generated by $s=up$, $t=pu^{-1}$ and $kp$"**
(tex 1554–1555). -/
noncomputable def cellRing [Finite k] (hP : IsClopen P) (hTP : T '' P ⊆ P) :
    Subring (bilateralCell T k hP hTP).Corner :=
  Subring.closure ((cellGeneratorSet_finite T k hP hTP).toFinset : Set _)

/-- **"It is finitely generated"** (tex 1556). -/
theorem isFinitelyGeneratedRing_cellRing [Finite k] (hP : IsClopen P) (hTP : T '' P ⊆ P) :
    IsFinitelyGeneratedRing ↥(cellRing T k hP hTP) :=
  OneSidedMFRadical.FullDefectRing.isFinitelyGeneratedRing_subringClosure _

theorem sCorner_mem_cellRing [Finite k] (hP : IsClopen P) (hTP : T '' P ⊆ P) :
    (bilateralCell T k hP hTP).sCorner ∈ cellRing T k hP hTP :=
  Subring.subset_closure (by simp [cellGeneratorSet])

theorem tCorner_mem_cellRing [Finite k] (hP : IsClopen P) (hTP : T '' P ⊆ P) :
    (bilateralCell T k hP hTP).tCorner ∈ cellRing T k hP hTP :=
  Subring.subset_closure (by simp [cellGeneratorSet])

/-- `char k = 0` in `R_X`, so `⟨z⟩` has order dividing `char k` (tex 1586–1587). -/
theorem natCast_ringChar_eq_zero : ((ringChar k : ℕ) : ClopenCrossedProduct T k) = 0 := by
  first
    | (rw [← map_natCast ((coeff T k).comp (LocallyConstant.constRingHom (X := X))) (ringChar k),
          ringChar.Nat.cast_ringChar (R := k), map_zero]; done)
    | (rw [← map_natCast (coeff T k) (ringChar k)]
       have h : ((ringChar k : ℕ) : LocallyConstant X k) = 0 := by
         ext x
         show ((ringChar k : ℕ) : k) = 0
         exact ringChar.Nat.cast_ringChar (R := k)
       rw [h, map_zero])

end Ring

/-- **The three-coordinate cell of `prop:bilateral-three`** (tex 1550–1593), closed. -/
theorem bilateralThreeCellStatement : BilateralThreeCellStatement := by
  intro X _ T k _ _ P hP hTP
  have hcorner := (bilateralCell T k hP hTP).corner_of_mem_range (S := cellRing T k hP hTP)
  have hstable := (bilateralCell T k hP hTP).unit_conj_mem_range
    (sCorner_mem_cellRing T k hP hTP) (tCorner_mem_cellRing T k hP hTP)
  have hn : ((ringChar k : ℕ) : ClopenCrossedProduct T k) * (bilateralCell T k hP hTP).d = 0 := by
    rw [natCast_ringChar_eq_zero, zero_mul]
  refine ⟨coreGroup (Set.range fun a : ↥(cellRing T k hP hTP) =>
      (a : (bilateralCell T k hP hTP).Corner).1),
    (bilateralCell T k hP hTP).W, (bilateralCell T k hP hTP).c, (bilateralCell T k hP hTP).y,
    (bilateralCell T k hP hTP).ell, (bilateralCell T k hP hTP).z,
    elementaryUnit (2 : Fin 3) 0 (by decide) ((bilateralCell T k hP hTP).f (-2) 0),
    elementaryUnit (1 : Fin 3) 2 (by decide) ((bilateralCell T k hP hTP).f 0 (-1)),
    ?_, (bilateralCell T k hP hTP).W_conj_mem_coreGroup hcorner hstable,
    (bilateralCell T k hP hTP).c_commute_coreGroup hcorner,
    (bilateralCell T k hP hTP).ell_mem_coreGroup ⟨1, rfl⟩, ?_,
    (bilateralCell T k hP hTP).z_commute_coreGroup hcorner,
    (bilateralCell T k hP hTP).z_commute_W, (bilateralCell T k hP hTP).z_commute_c,
    (bilateralCell T k hP hTP).z_commute_y,
    (bilateralCell T k hP hTP).isOfFinOrder_z
      (Nat.pos_of_ne_zero (CharP.char_ne_zero_of_finite (R := k) (ringChar k))) hn, ?_⟩
  · rw [← (bilateralCell T k hP hTP).range_cornerEmbed (cellRing T k hP hTP)]
    exact hasKazhdanPropertyT_range_of_injective (isFinitelyGeneratedRing_cellRing T k hP hTP) _
      ((bilateralCell T k hP hTP).cornerEmbed_injective _)
  · rw [(bilateralCell T k hP hTP).delta_eq]
    exact (bilateralCell T k hP hTP).commutator_y_delta
  · rw [(bilateralCell T k hP hTP).commutator_z_root,
      (bilateralCell T k hP hTP).commutator_roots_defect, bilateralCell_d]
