/-!
# The corner copy of `EL₃(S)` inside `GL₃(R)`

`non_mf_groups_exist.tex`, proof of Proposition `prop:bilateral-three`
(`\label{prop:bilateral-three}`, tex 1554–1558 and 1586 at origin/main 68481e4d7):

> The unital subalgebra $S\subseteq pR_Xp$ generated by $s=up$, $t=pu^{-1}$ and $kp$
> satisfies $ts=p$, $st=p-d$.  It is finitely generated, so $L=\EL_3(S)$ has
> property~\textup{(T)}.  Embed it in $\GL_3(R_X)$ by adjoining identity on $1-p$.
> [...] So $\langle z\rangle$ is a nontrivial finite central subgroup of $B$, of order
> $\operatorname{char}(k)$.

For a `Cell` of a ring `R`:

* `Cell.Corner`: the corner ring `pRp` with identity `p` (Mathlib's `IsIdempotentElem.Corner`);
  `sCorner`, `tCorner`: `s` and `t` as corner elements;
* `Cell.cornerMatrixHom : M₃(pRp) →* M₃(R)`, `g ↦ diag(1 - p) + g`, "adjoining identity on
  $1-p$", and for a subring `S` of the corner `Cell.cornerEmbed S : EL₃(S) →* GL₃(R)` with
  `cornerEmbed_injective` and `range_cornerEmbed`: the image is `coreGroup` of the coefficient
  set of `S`, on which `u a u⁻¹ = s a t` stays in `S` (`unit_conj_mem_range`);
* `Cell.z_pow_eq_one_of_natCast_mul_d`, `Cell.isOfFinOrder_z`: `z^n = 1` when `n d = 0`.
-/

namespace GroupApproximation
namespace BilateralThreeCell

/-- `(1 + N)^m = 1 + m N` when `N² = 0`. -/
theorem one_add_pow_of_mul_self_eq_zero {M : Type*} [Ring M] (N : M) (hN : N * N = 0) (m : ℕ) :
    (1 + N) ^ m = 1 + m • N := by
  induction m with
  | zero => simp
  | succ m ih =>
      rw [pow_succ, ih, succ_nsmul]
      have e : (1 + m • N) * (1 + N) = 1 + (m • N + N) + m • (N * N) := by noncomm_ring
      rw [e, hN, smul_zero, add_zero]

namespace Cell

variable {R : Type*} [Ring R] (C : Cell R)

theorem isIdempotentElem_p : IsIdempotentElem C.p := C.p_mul_p

/-- The corner ring `pRp`, with identity `p` (tex 1554). -/
abbrev Corner : Type _ := C.isIdempotentElem_p.Corner

@[simp] theorem val_one : (1 : C.Corner).1 = C.p := rfl

@[simp] theorem val_zero : (0 : C.Corner).1 = 0 := rfl

@[simp] theorem val_add (x y : C.Corner) : (x + y).1 = x.1 + y.1 := rfl

@[simp] theorem val_mul (x y : C.Corner) : (x * y).1 = x.1 * y.1 := rfl

theorem p_mul_val (x : C.Corner) : C.p * x.1 = x.1 :=
  ((Subsemigroup.mem_corner_iff C.isIdempotentElem_p).mp x.2).1

theorem val_mul_p (x : C.Corner) : x.1 * C.p = x.1 :=
  ((Subsemigroup.mem_corner_iff C.isIdempotentElem_p).mp x.2).2

theorem p_mul_val_mul_p (x : C.Corner) : C.p * x.1 * C.p = x.1 := by
  rw [C.p_mul_val, C.val_mul_p]

/-- `s = up` in the corner. -/
def sCorner : C.Corner := ⟨C.s, C.s, C.p_mul_s_mul_p⟩

/-- `t = pu⁻¹` in the corner. -/
def tCorner : C.Corner := ⟨C.t, C.t, C.p_mul_t_mul_p⟩

/-- The inclusion of the corner as an additive map. -/
def cornerValAddHom : C.Corner →+ R where
  toFun x := x.1
  map_zero' := rfl
  map_add' _ _ := rfl

/-- "Adjoining identity on $1-p$" (tex 1558): `g ↦ diag(1 - p) + g` on `3 × 3` matrices. -/
def cornerMatrixHom : Matrix (Fin 3) (Fin 3) C.Corner →* Matrix (Fin 3) (Fin 3) R where
  toFun g := Matrix.diagonal (fun _ => 1 - C.p) + g.map Subtype.val
  map_one' := by
    ext i j
    by_cases h : i = j
    · subst h
      simp [Matrix.one_apply]
    · simp [Matrix.one_apply, h]
  map_mul' g h := by
    have hl : ∀ x : C.Corner, (1 - C.p) * x.1 = 0 := fun x => by
      rw [sub_mul, one_mul, C.p_mul_val, sub_self]
    have hr : ∀ x : C.Corner, x.1 * (1 - C.p) = 0 := fun x => by
      rw [mul_sub, mul_one, C.val_mul_p, sub_self]
    have hDD : (1 - C.p) * (1 - C.p) = 1 - C.p := by
      rw [sub_mul, one_mul, mul_sub, mul_one, C.p_mul_p, sub_self, sub_zero]
    have hmap : (g * h).map Subtype.val = g.map Subtype.val * h.map Subtype.val := by
      ext i j
      simp only [Matrix.map_apply, Matrix.mul_apply]
      exact map_sum C.cornerValAddHom (fun k => g i k * h k j) Finset.univ
    have e1 : Matrix.diagonal (fun _ => 1 - C.p) * h.map Subtype.val = 0 := by
      ext i j
      simp [Matrix.diagonal_mul, hl]
    have e2 : g.map Subtype.val * Matrix.diagonal (fun _ => 1 - C.p) = 0 := by
      ext i j
      simp [Matrix.mul_diagonal, hr]
    show Matrix.diagonal (fun _ => 1 - C.p) + (g * h).map Subtype.val =
      (Matrix.diagonal (fun _ => 1 - C.p) + g.map Subtype.val) *
        (Matrix.diagonal (fun _ => 1 - C.p) + h.map Subtype.val)
    rw [hmap, add_mul, mul_add, mul_add, Matrix.diagonal_mul_diagonal, e1, e2, add_zero, zero_add,
      hDD]

/-- `GL₃(S) → GL₃(R)` for a subring `S` of the corner. -/
def cornerUnitMap (S : Subring C.Corner) :
    (Matrix (Fin 3) (Fin 3) ↥S)ˣ →* (Matrix (Fin 3) (Fin 3) R)ˣ :=
  Units.map (C.cornerMatrixHom.comp
    (S.subtype.mapMatrix : Matrix (Fin 3) (Fin 3) ↥S →+* Matrix (Fin 3) (Fin 3) C.Corner))

/-- **"Embed it in $\GL_3(R_X)$ by adjoining identity on $1-p$"** (tex 1558):
`EL₃(S) → GL₃(R)`. -/
def cornerEmbed (S : Subring C.Corner) :
    elementaryGroup (Fin 3) ↥S →* (Matrix (Fin 3) (Fin 3) R)ˣ :=
  (C.cornerUnitMap S).comp (elementaryGroup (Fin 3) ↥S).subtype

@[simp] theorem cornerUnitMap_val (S : Subring C.Corner) (g : (Matrix (Fin 3) (Fin 3) ↥S)ˣ) :
    ((C.cornerUnitMap S g : (Matrix (Fin 3) (Fin 3) R)ˣ) : Matrix (Fin 3) (Fin 3) R) =
      Matrix.diagonal (fun _ => 1 - C.p) +
        (g : Matrix (Fin 3) (Fin 3) ↥S).map (fun x => (x : C.Corner).1) := rfl

/-- The image of a root: "adjoining identity on $1-p$" sends `1_S + a E_{ij}` to `1 + a E_{ij}`. -/
theorem cornerUnitMap_elementaryUnit (S : Subring C.Corner) (i j : Fin 3) (h : i ≠ j) (a : ↥S) :
    C.cornerUnitMap S (elementaryUnit i j h a) = elementaryUnit i j h (a : C.Corner).1 := by
  apply Units.ext
  change Matrix.diagonal (fun _ => 1 - C.p) +
      (1 + Matrix.single i j a : Matrix (Fin 3) (Fin 3) ↥S).map (fun x => (x : C.Corner).1) =
    1 + Matrix.single i j (a : C.Corner).1
  ext r s
  simp only [Matrix.add_apply, Matrix.map_apply, Matrix.diagonal_apply, Matrix.one_apply,
    Matrix.single_apply]
  split_ifs <;> simp_all

theorem cornerUnitMap_injective (S : Subring C.Corner) : Function.Injective (C.cornerUnitMap S) := by
  intro g g' hgg'
  apply Units.ext
  have hv : Matrix.diagonal (fun _ => 1 - C.p) +
        (g : Matrix (Fin 3) (Fin 3) ↥S).map (fun x => (x : C.Corner).1) =
      Matrix.diagonal (fun _ => 1 - C.p) +
        (g' : Matrix (Fin 3) (Fin 3) ↥S).map (fun x => (x : C.Corner).1) :=
    congrArg (fun x : (Matrix (Fin 3) (Fin 3) R)ˣ => (x : Matrix (Fin 3) (Fin 3) R)) hgg'
  have h2 := add_left_cancel hv
  ext r s
  exact Subtype.val_injective
    (Subtype.val_injective (congrArg (fun M : Matrix (Fin 3) (Fin 3) R => M r s) h2))

theorem cornerEmbed_injective (S : Subring C.Corner) : Function.Injective (C.cornerEmbed S) :=
  (C.cornerUnitMap_injective S).comp Subtype.val_injective

/-- The embedded `L` is `coreGroup` of the coefficient set of `S`. -/
theorem range_cornerEmbed (S : Subring C.Corner) :
    (C.cornerEmbed S).range = coreGroup (Set.range fun a : ↥S => (a : C.Corner).1) := by
  rw [cornerEmbed, MonoidHom.range_comp, Subgroup.range_subtype, elementaryGroup,
    MonoidHom.map_closure, coreGroup]
  congr 1
  ext g
  constructor
  · rintro ⟨_, ⟨i, j, h, a, rfl⟩, rfl⟩
    exact ⟨i, j, h, _, ⟨a, rfl⟩, (C.cornerUnitMap_elementaryUnit S i j h a).symm⟩
  · rintro ⟨i, j, h, _, ⟨a, rfl⟩, rfl⟩
    exact ⟨_, ⟨i, j, h, a, rfl⟩, C.cornerUnitMap_elementaryUnit S i j h a⟩

/-- The coefficients of `S` lie in the corner `pRp`. -/
theorem corner_of_mem_range {S : Subring C.Corner} :
    ∀ x ∈ Set.range (fun a : ↥S => (a : C.Corner).1), C.p * x * C.p = x := by
  rintro _ ⟨a, rfl⟩
  exact C.p_mul_val_mul_p a

/-- "sending a root coefficient $a\in S$ to $uau^{-1}=sat$" (tex 1567–1568): conjugation by `u`
preserves the coefficients of `S` when `s, t ∈ S`. -/
theorem unit_conj_mem_range {S : Subring C.Corner} (hs : C.sCorner ∈ S) (ht : C.tCorner ∈ S) :
    ∀ x ∈ Set.range (fun a : ↥S => (a : C.Corner).1),
      ↑C.u * x * ↑C.u⁻¹ ∈ Set.range (fun a : ↥S => (a : C.Corner).1) := by
  rintro _ ⟨a, rfl⟩
  refine ⟨⟨C.sCorner * a * C.tCorner, S.mul_mem (S.mul_mem hs a.2) ht⟩, ?_⟩
  rw [C.unit_conj_eq_s_mul_t (C.p_mul_val_mul_p a)]
  rfl

/-- `z^n = 1` when `n d = 0`. -/
theorem z_pow_eq_one_of_natCast_mul_d {n : ℕ} (h : (n : R) * C.d = 0) : C.z ^ n = 1 := by
  have hx : (n : R) * C.f (-1) (-2) = 0 := by
    rw [Cell.f, ← mul_assoc, ← mul_assoc, Nat.cast_comm, mul_assoc (C.pw (-1)), h, mul_zero,
      zero_mul]
  apply Units.ext
  rw [Units.val_pow_eq_pow_val, Units.val_one]
  change (1 + Matrix.single (2 : Fin 3) (2 : Fin 3) (C.f (-1) (-2))) ^ n = 1
  rw [one_add_pow_of_mul_self_eq_zero _
      (single_mul_single_eq_zero (Or.inr C.f_mul_self_neg_one_neg_two)),
    Matrix.smul_single, nsmul_eq_mul, hx, Matrix.single_zero, add_zero]

/-- "$\langle z\rangle$ is a [...] finite [...] subgroup" (tex 1586). -/
theorem isOfFinOrder_z {n : ℕ} (hn : 0 < n) (h : (n : R) * C.d = 0) : IsOfFinOrder C.z :=
  isOfFinOrder_iff_pow_eq_one.mpr ⟨n, hn, C.z_pow_eq_one_of_natCast_mul_d h⟩
