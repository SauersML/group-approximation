/-!
# The relation sequence: the Fox complex is exact at `ℤG^X`

For a presentation by signed words `rel : ρ → SignedWord X` of `G = ⟨X | rel⟩`
(`Presented rel`), every `1`-cycle of the Fox complex is a sum of translated relator rows:
`ker ∂₁ = range ∂₂` (`ker_foxD₁_eq_range_foxD₂`).  This is Gruenberg's relation sequence.

The proof lifts a cycle `z` along a set-theoretic section `G → F` (`liftG`).  The boundary
`y = ∂₁^F (lift z)` dies in `ℤG`, so it is a combination of differences `[g] − [h]` with
`h⁻¹g` in the normal closure `N` of the relators, and `[g] − [h] = ∂₁^F (h · ∂(h⁻¹g))`
(`exists_foxD₁_eq_sub_lift`).  For `n ∈ N` the pushforward of the Fox derivative `∂n` is a
relator combination (`mapG_foxFree_mem_range`, by induction on `N` as the subgroup generated
by conjugates of relators, where `∂(c r c⁻¹) ↦ π(c)·∂r`).  Since `∂₁^F` is injective
(`foxD₁_injective_freeGroup`, the augmentation ideal of a free group is free), the lift of
`z` is that combination, and pushing forward gives `z ∈ range ∂₂`.
-/

namespace GroupApproximation.FoxComplex

open FreeResolutionPrimeOrder FoxBoundary PresentedGroupRelatorReplay

universe v w

/-- Naturality of integral Fox boundaries under pushforward. -/
theorem mapG_foxFrom {F : Type*} [Group F] {G : Type*} [Group G] {Y : Type*} (φ : F →* G)
    (s : Y → F) (p : F) (w : SignedWord Y) :
    mapG φ (foxFrom s p w) = foxFrom (fun x ↦ φ (s x)) (φ p) w := by
  induction w generalizing p with
  | nil => rw [foxFrom_nil, foxFrom_nil, map_zero]
  | cons letter rest ih =>
      rcases letter with ⟨x, _ | _⟩
      · rw [foxFrom_cons_false, foxFrom_cons_false, map_add, map_neg, mapG_single, ih, map_mul,
          map_inv]
      · rw [foxFrom_cons_true, foxFrom_cons_true, map_add, mapG_single, ih, map_mul]

variable {X : Type v} {ρ : Type w} (rel : ρ → SignedWord X)

/-- The relators as elements of the free group. -/
def relSet : Set (FreeGroup X) := Set.range fun r ↦ word (rel r)

/-- The group presented by signed-word relators. -/
abbrev Presented : Type v := PresentedGroup (relSet rel)

theorem mk_rel (r : ρ) : PresentedGroup.mk (relSet rel) (word (rel r)) = 1 :=
  (QuotientGroup.eq_one_iff (word (rel r))).mpr (Subgroup.subset_normalClosure ⟨r, rfl⟩)

theorem mem_normalClosure_of_mk_eq_one {n : FreeGroup X}
    (h : PresentedGroup.mk (relSet rel) n = 1) : n ∈ Subgroup.normalClosure (relSet rel) :=
  (QuotientGroup.eq_one_iff n).mp h

theorem wordValue_ofP (w : SignedWord X) :
    wordValue (PresentedGroup.of : X → Presented rel) w =
      PresentedGroup.mk (relSet rel) (word w) :=
  wordValue_freeGroup_map (PresentedGroup.mk (relSet rel)) w

theorem wordValue_ofP_rel (r : ρ) :
    wordValue (PresentedGroup.of : X → Presented rel) (rel r) = 1 := by
  rw [wordValue_ofP, mk_rel]

/-- The presented group is generated by the generators through signed words. -/
theorem presented_generates (g : Presented rel) :
    ∃ w : SignedWord X, wordValue (PresentedGroup.of : X → Presented rel) w = g := by
  obtain ⟨f, rfl⟩ := PresentedGroup.mk_surjective (relSet rel) g
  induction f using Quot.ind with
  | mk w => exact ⟨w, wordValue_ofP rel w⟩

theorem mapG_foxFree_word (w : SignedWord X) :
    mapG (PresentedGroup.mk (relSet rel)) (foxFree (1 : FreeGroup X) (word w)) =
      foxZ (PresentedGroup.of : X → Presented rel) w := by
  show mapG (PresentedGroup.mk (relSet rel)) (foxFrom FreeGroup.of 1 w) =
    foxFrom (PresentedGroup.of : X → Presented rel) 1 w
  rw [mapG_foxFrom, map_one]
  rfl

/-- The pushforward of the Fox derivative of a conjugated relator is a translated relator row. -/
theorem mapG_foxFree_conj (c : FreeGroup X) (r : ρ) :
    mapG (PresentedGroup.mk (relSet rel)) (foxFree 1 (c * word (rel r) * c⁻¹)) =
      act (PresentedGroup.mk (relSet rel) c) (foxZ (PresentedGroup.of : X → Presented rel) (rel r)) := by
  have ha : PresentedGroup.mk (relSet rel) (word (rel r)) = 1 := mk_rel rel r
  have h3 : mapG (PresentedGroup.mk (relSet rel)) (act (c * word (rel r)) (foxFree 1 c⁻¹)) =
      -mapG (PresentedGroup.mk (relSet rel)) (foxFree 1 c) := by
    rw [foxFree_inv, map_neg, map_neg, act_act, mapG_act, map_mul, map_mul, ha, mul_one, map_inv,
      mul_inv_cancel, act_one]
  rw [foxFree_mul, foxFree_mul, map_add, map_add, h3, mapG_act, mapG_foxFree_word]
  abel

/-- **Normal-closure elements push forward to relator combinations.** -/
theorem mapG_foxFree_mem_range {n : FreeGroup X}
    (hn : n ∈ Subgroup.normalClosure (relSet rel)) :
    mapG (PresentedGroup.mk (relSet rel)) (foxFree 1 n) ∈
      (foxD₂ (PresentedGroup.of : X → Presented rel) rel).range := by
  have hn' : n ∈ Subgroup.closure (Group.conjugatesOfSet (relSet rel)) := hn
  clear hn
  induction hn' using Subgroup.closure_induction with
  | mem x hx =>
      obtain ⟨a, ⟨r, rfl⟩, hconj⟩ := Group.mem_conjugatesOfSet_iff.mp hx
      obtain ⟨c, rfl⟩ := isConj_iff.mp hconj
      refine ⟨Finsupp.single (r, PresentedGroup.mk (relSet rel) c) 1, ?_⟩
      rw [foxD₂_single, one_zsmul, mapG_foxFree_conj]
  | one =>
      rw [foxFree_one, map_zero]
      exact zero_mem _
  | mul x y hx hy ihx ihy =>
      have hx' : x ∈ Subgroup.normalClosure (relSet rel) := hx
      have hxN : PresentedGroup.mk (relSet rel) x = 1 :=
        (QuotientGroup.eq_one_iff (N := Subgroup.normalClosure (relSet rel)) x).mpr hx'
      rw [foxFree_mul, map_add, mapG_act, hxN, act_one]
      exact add_mem ihx ihy
  | inv x hx ihx =>
      have hx' : x ∈ Subgroup.normalClosure (relSet rel) := hx
      have hxN : PresentedGroup.mk (relSet rel) x = 1 :=
        (QuotientGroup.eq_one_iff (N := Subgroup.normalClosure (relSet rel)) x).mpr hx'
      rw [foxFree_inv, map_neg, mapG_act, map_inv, hxN, inv_one, act_one]
      exact neg_mem ihx

/-- A set-theoretic section of the quotient map. -/
noncomputable def sectionP : Presented rel → FreeGroup X :=
  Function.surjInv (PresentedGroup.mk_surjective (relSet rel))

theorem mk_sectionP (g : Presented rel) : PresentedGroup.mk (relSet rel) (sectionP rel g) = g :=
  Function.surjInv_eq (PresentedGroup.mk_surjective (relSet rel)) g

/-- Lifting along the section. -/
noncomputable def liftG {ι : Type*} : FreeZG (Presented rel) ι →+ FreeZG (FreeGroup X) ι :=
  Finsupp.mapDomain.addMonoidHom (Prod.map id (sectionP rel))

theorem liftG_single {ι : Type*} (i : ι) (g : Presented rel) (c : ℤ) :
    liftG rel (Finsupp.single (i, g) c) = Finsupp.single (i, sectionP rel g) c := by
  show Finsupp.mapDomain (Prod.map id (sectionP rel)) (Finsupp.single (i, g) c) = _
  rw [Finsupp.mapDomain_single]
  rfl

theorem mapG_liftG {ι : Type*} (f : FreeZG (Presented rel) ι) :
    mapG (PresentedGroup.mk (relSet rel)) (liftG rel f) = f := by
  refine Finsupp.induction_linear f ?_ ?_ ?_
  · rw [map_zero, map_zero]
  · intro f f' hf hf'
    rw [map_add, map_add, hf, hf']
  · rintro ⟨i, g⟩ c
    rw [liftG_single, mapG_single, mk_sectionP]

/-- **The lifting step**: a combination of cosets differences is a free-group boundary whose
pushforward is a relator combination. -/
theorem exists_foxD₁_eq_sub_lift (y : FreeZG (FreeGroup X) Unit) :
    ∃ v : FreeZG (FreeGroup X) X,
      foxD₁ FreeGroup.of v = y - liftG rel (mapG (PresentedGroup.mk (relSet rel)) y) ∧
        mapG (PresentedGroup.mk (relSet rel)) v ∈
          (foxD₂ (PresentedGroup.of : X → Presented rel) rel).range := by
  refine Finsupp.induction_linear y ?_ ?_ ?_
  · refine ⟨0, ?_, ?_⟩
    · rw [map_zero, map_zero, map_zero, sub_zero]
    · rw [map_zero]
      exact zero_mem _
  · rintro y y' ⟨v, hv, hvr⟩ ⟨v', hv', hvr'⟩
    refine ⟨v + v', ?_, ?_⟩
    · rw [map_add, hv, hv', map_add, map_add]
      abel
    · rw [map_add]
      exact add_mem hvr hvr'
  · rintro ⟨⟨⟩, g⟩ c
    have hm : PresentedGroup.mk (relSet rel)
        ((sectionP rel (PresentedGroup.mk (relSet rel) g))⁻¹ * g) = 1 := by
      rw [map_mul, map_inv, mk_sectionP, inv_mul_cancel]
    obtain ⟨u, hu⟩ := mapG_foxFree_mem_range rel (mem_normalClosure_of_mk_eq_one rel hm)
    refine ⟨c • act (sectionP rel (PresentedGroup.mk (relSet rel) g))
      (foxFree 1 ((sectionP rel (PresentedGroup.mk (relSet rel) g))⁻¹ * g)), ?_, ?_⟩
    · rw [map_zsmul, foxD₁_act, foxD₁_foxFree, mapG_single, liftG_single, map_sub, act_single,
        act_single, mul_inv_cancel_left, mul_one, smul_sub, Finsupp.smul_single_one,
        Finsupp.smul_single_one]
    · refine ⟨c • act (PresentedGroup.mk (relSet rel)
        (sectionP rel (PresentedGroup.mk (relSet rel) g))) u, ?_⟩
      rw [map_zsmul, foxD₂_act, hu, map_zsmul, mapG_act]

/-- **Exactness at `ℤG^X`**: every `1`-cycle is a relator combination. -/
theorem ker_foxD₁_le_range_foxD₂ (z : FreeZG (Presented rel) X)
    (hz : foxD₁ (PresentedGroup.of : X → Presented rel) z = 0) :
    z ∈ (foxD₂ (PresentedGroup.of : X → Presented rel) rel).range := by
  have hy : mapG (PresentedGroup.mk (relSet rel)) (foxD₁ FreeGroup.of (liftG rel z)) = 0 := by
    rw [mapG_foxD₁, mapG_liftG]
    exact hz
  obtain ⟨v, hv, hvr⟩ := exists_foxD₁_eq_sub_lift rel (foxD₁ FreeGroup.of (liftG rel z))
  rw [hy, map_zero, sub_zero] at hv
  have hveq : v = liftG rel z := foxD₁_injective_freeGroup hv
  rw [hveq, mapG_liftG] at hvr
  exact hvr

/-- **The relation sequence**: `ker ∂₁ = range ∂₂` for a presented group. -/
theorem ker_foxD₁_eq_range_foxD₂ :
    (foxD₁ (PresentedGroup.of : X → Presented rel)).ker =
      (foxD₂ (PresentedGroup.of : X → Presented rel) rel).range := by
  ext z
  constructor
  · intro hz
    exact ker_foxD₁_le_range_foxD₂ rel z ((AddMonoidHom.mem_ker).mp hz)
  · rintro ⟨y, rfl⟩
    exact (AddMonoidHom.mem_ker).mpr (foxD₁_foxD₂ _ rel (wordValue_ofP_rel rel) y)
