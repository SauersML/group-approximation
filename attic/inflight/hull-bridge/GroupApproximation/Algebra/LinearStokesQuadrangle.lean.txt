import GroupApproximation.Algebra.DiscreteStokesPaths
import GroupApproximation.Algebra.WordMetricComparison

/-!
# The linear Stokes inequality and the quadrangle bound in a Cayley graph

Let `S` be a symmetric generating set of a group `Q`.  A path in the Cayley graph
is a list of letters of `S`; `listPoints b l` lists its vertices read from `b`.

`LinearStokes S C` says: along every closed path `l`, the discrete integral of
`φ dψ` (`DiscreteStokes.wordSum`) is at most `C |l|` in absolute value, for all
`φ ψ : Q → ℤ` moving by at most one along each letter of the path.  A linear
isoperimetric inequality over relators of bounded length gives it
(`Algebra/DiscreteStokesArea.lean`).

With `φ` and `ψ` the distances to two consecutive sides of a loop, truncated at
the separations of the opposite sides, the quadrangle identity
(`DiscreteStokes.wordSum_quadrangle`) evaluates the integral exactly, and
`LinearStokes` then bounds the product of the two separations by the length of the
loop (`two_mul_mul_le_of_linearStokes`).  This is the only way area enters the
proof of Gromov's theorem that follows.
-/

namespace GroupApproximation
namespace DiscreteStokes

open WordMetric

universe v

variable {Q : Type v} [Group Q]

/-! ## Paths of group elements as free words -/

/-- A word of group elements, as a free word in positive letters. -/
def trueLetters (l : List Q) : List (Q × Bool) :=
  l.map fun x => (x, true)

@[simp] theorem trueLetters_nil : trueLetters ([] : List Q) = [] := rfl

@[simp] theorem trueLetters_cons (x : Q) (l : List Q) :
    trueLetters (x :: l) = (x, true) :: trueLetters l := rfl

theorem trueLetters_append (l m : List Q) :
    trueLetters (l ++ m) = trueLetters l ++ trueLetters m :=
  List.map_append

@[simp] theorem length_trueLetters (l : List Q) : (trueLetters l).length = l.length :=
  List.length_map _

@[simp] theorem wordVal_trueLetters (l : List Q) :
    wordVal (id : Q → Q) (trueLetters l) = l.prod := by
  induction l with
  | nil => rfl
  | cons x l ih => simp [wordVal_cons, ih]

/-- **The vertices of a path** of group elements read from `b`. -/
def listPoints : Q → List Q → List Q
  | b, [] => [b]
  | b, x :: l => b :: listPoints (b * x) l

@[simp] theorem listPoints_nil (b : Q) : listPoints b [] = [b] := rfl

@[simp] theorem listPoints_cons (b x : Q) (l : List Q) :
    listPoints b (x :: l) = b :: listPoints (b * x) l := rfl

theorem listPoints_ne_nil (b : Q) (l : List Q) : listPoints b l ≠ [] := by
  cases l <;> simp

@[simp] theorem pathPoints_trueLetters (b : Q) (l : List Q) :
    pathPoints (id : Q → Q) b (trueLetters l) = listPoints b l := by
  induction l generalizing b with
  | nil => rfl
  | cons x l ih => simp [ih]

/-! ## Distance to the vertices of a path -/

/-- The distance from `g` to the nearest point of a list (`0` for the empty list). -/
noncomputable def distToList (S : Set Q) (g : Q) : List Q → ℕ
  | [] => 0
  | [p] => wordDist S g p
  | p :: p' :: P => min (wordDist S g p) (distToList S g (p' :: P))

theorem distToList_le_of_mem (S : Set Q) (g : Q) {p : Q} :
    ∀ {P : List Q}, p ∈ P → distToList S g P ≤ wordDist S g p
  | [], h => absurd h (by simp)
  | [q], h => by
      rw [List.mem_singleton] at h
      subst h
      exact le_rfl
  | q :: q' :: P, h => by
      rw [List.mem_cons] at h
      rcases h with rfl | h
      · exact min_le_left _ _
      · exact le_trans (min_le_right _ _) (distToList_le_of_mem S g h)

theorem exists_mem_distToList_eq (S : Set Q) (g : Q) :
    ∀ {P : List Q}, P ≠ [] → ∃ p ∈ P, wordDist S g p = distToList S g P
  | [], h => absurd rfl h
  | [q], _ => ⟨q, by simp, rfl⟩
  | q :: q' :: P, _ => by
      obtain ⟨p, hp, hpd⟩ := exists_mem_distToList_eq S g (List.cons_ne_nil q' P)
      by_cases hle : wordDist S g q ≤ distToList S g (q' :: P)
      · exact ⟨q, by simp, (min_eq_left hle).symm⟩
      · refine ⟨p, List.mem_cons_of_mem _ hp, ?_⟩
        rw [hpd]
        exact (min_eq_right (not_le.mp hle).le).symm

theorem distToList_eq_zero_of_mem (S : Set Q) {g : Q} {P : List Q} (hg : g ∈ P) :
    distToList S g P = 0 := by
  have h := distToList_le_of_mem S g hg
  rw [wordDist_self] at h
  omega

theorem le_distToList (S : Set Q) (g : Q) {P : List Q} (hP : P ≠ []) {n : ℕ}
    (h : ∀ p ∈ P, n ≤ wordDist S g p) : n ≤ distToList S g P := by
  obtain ⟨p, hp, hpd⟩ := exists_mem_distToList_eq S g hP
  rw [← hpd]
  exact h p hp

/-- One letter moves the distance to a list by at most one, upwards. -/
theorem distToList_mul_le {S : Set Q} (hS : IsSymmetricGeneratingSet S) {P : List Q}
    (hP : P ≠ []) (g s : Q) (hs : s ∈ S) :
    distToList S (g * s) P ≤ distToList S g P + 1 := by
  obtain ⟨p, hp, hpd⟩ := exists_mem_distToList_eq S g hP
  have h1 := distToList_le_of_mem S (g * s) hp
  have h2 := wordDist_triangle hS (g * s) g p
  have h3 : wordDist S (g * s) g ≤ 1 := by
    have heq : (g * s)⁻¹ * g = s⁻¹ := by group
    unfold wordDist
    rw [heq]
    exact wordNorm_le_one_of_mem (hS.inv_mem s hs)
  omega

/-- One letter moves the distance to a list by at most one, downwards. -/
theorem distToList_le_mul {S : Set Q} (hS : IsSymmetricGeneratingSet S) {P : List Q}
    (hP : P ≠ []) (g s : Q) (hs : s ∈ S) :
    distToList S g P ≤ distToList S (g * s) P + 1 := by
  obtain ⟨p, hp, hpd⟩ := exists_mem_distToList_eq S (g * s) hP
  have h1 := distToList_le_of_mem S g hp
  have h2 := wordDist_triangle hS g (g * s) p
  have h3 : wordDist S g (g * s) ≤ 1 := by
    unfold wordDist
    rw [inv_mul_cancel_left]
    exact wordNorm_le_one_of_mem hs
  omega

/-- **The distance to the vertices of a path, truncated at `k`**, as an integer. -/
noncomputable def truncDist (S : Set Q) (P : List Q) (k : ℕ) (g : Q) : ℤ :=
  ((min (distToList S g P) k : ℕ) : ℤ)

theorem truncDist_eq_zero_of_mem (S : Set Q) {P : List Q} (k : ℕ) {q : Q} (hq : q ∈ P) :
    truncDist S P k q = 0 := by
  unfold truncDist
  rw [distToList_eq_zero_of_mem S hq]
  simp

theorem truncDist_eq_of_le (S : Set Q) {P : List Q} {k : ℕ} {q : Q}
    (h : k ≤ distToList S q P) : truncDist S P k q = k := by
  unfold truncDist
  rw [min_eq_right h]

/-- A truncated distance moves by at most one along every letter of `S`. -/
theorem truncDist_lipschitzAlong {S : Set Q} (hS : IsSymmetricGeneratingSet S)
    {P : List Q} (hP : P ≠ []) (k : ℕ) (l : List Q) (hl : ∀ x ∈ l, x ∈ S) :
    LipschitzAlong (id : Q → Q) (truncDist S P k) (trueLetters l) := by
  intro x hx g
  obtain ⟨s, hs, rfl⟩ := List.mem_map.mp hx
  have h1 := distToList_mul_le hS hP g s (hl s hs)
  have h2 := distToList_le_mul hS hP g s (hl s hs)
  simp only [letterVal_true, id_eq, truncDist]
  rw [abs_le]
  push_cast
  constructor <;> omega

/-- A truncated distance moves by at most one along every letter of `S`, everywhere. -/
theorem truncDist_lipschitz {S : Set Q} (hS : IsSymmetricGeneratingSet S)
    {P : List Q} (hP : P ≠ []) (k : ℕ) (g s : Q) (hs : s ∈ S) :
    |truncDist S P k (g * s) - truncDist S P k g| ≤ 1 := by
  have h1 := distToList_mul_le hS hP g s hs
  have h2 := distToList_le_mul hS hP g s hs
  simp only [truncDist]
  rw [abs_le]
  push_cast
  constructor <;> omega

/-! ## The linear Stokes inequality -/

/-- **The linear Stokes inequality** at constant `C`: along every closed path of
letters of `S`, the discrete integral of `φ dψ` is at most `C` times the length,
for all functions moving by at most one along every letter of `S`.  The Lipschitz
condition is asked everywhere, not only along the path, because the area bound
integrates along the relators of a filling, which leave the path. -/
def LinearStokes (S : Set Q) (C : ℕ) : Prop :=
  ∀ (φ ψ : Q → ℤ), (∀ (g s : Q), s ∈ S → |φ (g * s) - φ g| ≤ 1) →
    (∀ (g s : Q), s ∈ S → |ψ (g * s) - ψ g| ≤ 1) →
    ∀ (l : List Q), (∀ x ∈ l, x ∈ S) → ∀ b : Q, l.prod = 1 →
      |wordSum (id : Q → Q) φ ψ b (trueLetters l)| ≤ (C : ℤ) * l.length

/-- **The quadrangle bound.**  A closed path `l₁ l₂ l₃ l₄` whose third side is at
distance at least `a` from the first and whose fourth side is at distance at least
`c` from the second has length at least `2 a c / C`. -/
theorem two_mul_mul_le_of_linearStokes {S : Set Q} (hS : IsSymmetricGeneratingSet S)
    {C : ℕ} (hC : LinearStokes S C) (b : Q) (l₁ l₂ l₃ l₄ : List Q)
    (hl : ∀ x ∈ l₁ ++ l₂ ++ l₃ ++ l₄, x ∈ S) (hclosed : (l₁ ++ l₂ ++ l₃ ++ l₄).prod = 1)
    (a c : ℕ)
    (hsep₁ : ∀ q ∈ listPoints (b * (l₁ ++ l₂).prod) l₃, a ≤ distToList S q (listPoints b l₁))
    (hsep₂ : ∀ q ∈ listPoints (b * (l₁ ++ l₂ ++ l₃).prod) l₄,
      c ≤ distToList S q (listPoints (b * l₁.prod) l₂)) :
    2 * (a : ℤ) * c ≤ (C : ℤ) * (l₁ ++ l₂ ++ l₃ ++ l₄).length := by
  have hbound := hC (truncDist S (listPoints b l₁) a)
    (truncDist S (listPoints (b * l₁.prod) l₂) c) _ hl
    (truncDist_lipschitzAlong hS (listPoints_ne_nil b l₁) a _ hl)
    (truncDist_lipschitzAlong hS (listPoints_ne_nil _ l₂) c _ hl) b hclosed
  have hquad : wordSum (id : Q → Q) (truncDist S (listPoints b l₁) a)
      (truncDist S (listPoints (b * l₁.prod) l₂) c) b (trueLetters (l₁ ++ l₂ ++ l₃ ++ l₄)) =
        2 * (a : ℤ) * c := by
    rw [trueLetters_append, trueLetters_append, trueLetters_append]
    apply wordSum_quadrangle
    · intro q hq
      rw [pathPoints_trueLetters] at hq
      exact truncDist_eq_zero_of_mem S a hq
    · intro q hq
      rw [wordVal_trueLetters, pathPoints_trueLetters] at hq
      exact truncDist_eq_zero_of_mem S c hq
    · intro q hq
      rw [← trueLetters_append, wordVal_trueLetters, pathPoints_trueLetters] at hq
      exact truncDist_eq_of_le S (hsep₁ q hq)
    · intro q hq
      rw [← trueLetters_append, ← trueLetters_append, wordVal_trueLetters,
        pathPoints_trueLetters] at hq
      exact truncDist_eq_of_le S (hsep₂ q hq)
  rw [hquad] at hbound
  exact le_trans (le_abs_self _) hbound

end DiscreteStokes
end GroupApproximation
