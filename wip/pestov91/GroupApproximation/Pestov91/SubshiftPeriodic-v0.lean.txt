import GroupApproximation.Pestov91.SubshiftToeplitz

/-!
# Periodic return words for the Toeplitz subshift

`periodicToeplitz F` is the `2^(F+1)`-periodic configuration that agrees with `toeplitz` on
`[1, 2^(F+2) - 1]`. When `2k < 2^F`, its windows of radius `k` are exactly the windows of radius
`k` of the points of `toeplitzSubshift`:

* `periodicToeplitz_window_legal`: every window of `periodicToeplitz F` is a window of `toeplitz`;
* `toeplitz_window_occurs`: every window of `toeplitz` is a window of `periodicToeplitz F`.

`exists_periodic_model` packages both, with an arbitrary lower bound on the period. These are
the periodic models behind the local embeddings of the crossed product into matrix rings.
-/

namespace GroupApproximation.Pestov91

open SymbolicDynamics.FullShift

/-- The `2^(F+1)`-periodic configuration agreeing with `toeplitz` on `[1, 2^(F+1)]`. -/
def periodicToeplitz (F : ℕ) (n : ℤ) : Bool :=
  toeplitz ((n - 1) % 2 ^ (F + 1) + 1)

theorem periodicToeplitz_add_mul (F : ℕ) (n q : ℤ) :
    periodicToeplitz F (n + 2 ^ (F + 1) * q) = periodicToeplitz F n := by
  simp only [periodicToeplitz]
  rw [show n + 2 ^ (F + 1) * q - 1 = (n - 1) + 2 ^ (F + 1) * q by ring,
    Int.add_mul_emod_self_left]

/-- `periodicToeplitz F` agrees with `toeplitz` on `[1, 2^(F+2) - 1]`. -/
theorem periodicToeplitz_eq (F : ℕ) {n : ℤ} (hn1 : 1 ≤ n) (hn2 : n < 2 * 2 ^ (F + 1)) :
    periodicToeplitz F n = toeplitz n := by
  simp only [periodicToeplitz]
  have hN : (0 : ℤ) < 2 ^ (F + 1) := by positivity
  rcases lt_or_ge n (2 ^ (F + 1) + 1) with h | h
  · have h' : (n - 1) % 2 ^ (F + 1) = n - 1 := Int.emod_eq_of_lt (by linarith) (by linarith)
    rw [h', sub_add_cancel]
  · have h' : (n - 1) % 2 ^ (F + 1) = n - 1 - 2 ^ (F + 1) := by
      rw [← Int.sub_emod_right (n - 1) (2 ^ (F + 1))]
      exact Int.emod_eq_of_lt (by linarith) (by linarith)
    have hnd : ¬ (2 : ℤ) ^ (F + 1) ∣ n - 2 ^ (F + 1) := fun hd => by
      have h0 := Int.eq_zero_of_abs_lt_dvd hd (abs_lt.2 ⟨by linarith, by linarith⟩)
      linarith
    have h2 := toeplitz_add_of_dvd hnd (dvd_refl ((2 : ℤ) ^ (F + 1)))
    rw [sub_add_cancel] at h2
    rw [h', show n - 1 - 2 ^ (F + 1) + 1 = n - 2 ^ (F + 1) by ring, h2]

/-- Every window of radius `k` of `periodicToeplitz F` is a window of `toeplitz`. -/
theorem periodicToeplitz_window_legal {F k : ℕ} (hk : 2 * k < 2 ^ (F + 1)) (n : ℤ) :
    ∃ c : ℤ, Agree k (shift n (periodicToeplitz F)) (shift c toeplitz) := by
  refine ⟨(n - k - 1) % 2 ^ (F + 1) + k + 1, fun m hm => ?_⟩
  simp only [shift_apply]
  have hN : (0 : ℤ) < 2 ^ (F + 1) := by positivity
  have h0 := Int.emod_nonneg (n - k - 1) hN.ne'
  have h1 := Int.emod_lt_of_pos (n - k - 1) hN
  have hdec := Int.emod_add_mul_ediv (n - k - 1) (2 ^ (F + 1))
  have hm' := abs_le.1 hm
  have hkN : (2 * k : ℤ) < 2 ^ (F + 1) := by exact_mod_cast hk
  have hsplit : n + m = ((n - k - 1) % 2 ^ (F + 1) + k + 1 + m) +
      2 ^ (F + 1) * ((n - k - 1) / 2 ^ (F + 1)) := by
    linear_combination (-1 : ℤ) * hdec
  rw [hsplit, periodicToeplitz_add_mul,
    periodicToeplitz_eq F (n := (n - k - 1) % 2 ^ (F + 1) + k + 1 + m) (by linarith)
      (by linarith)]

/-- Every window of radius `k` of `toeplitz` is a window of `periodicToeplitz F`. -/
theorem toeplitz_window_occurs {F k : ℕ} (hk : 2 * k < 2 ^ F) (c : ℤ) :
    ∃ n : ℤ, Agree k (shift c toeplitz) (shift n (periodicToeplitz F)) := by
  have hpos : (0 : ℤ) < 2 ^ F := by positivity
  have hkF : (2 * k : ℤ) < 2 ^ F := by exact_mod_cast hk
  have hk0 : (0 : ℤ) ≤ k := Nat.cast_nonneg k
  have hpow : (2 : ℤ) ^ (F + 1) = 2 * 2 ^ F := by ring
  by_cases hex : ∃ m₀ : ℤ, |m₀| ≤ k ∧ (2 : ℤ) ^ F ∣ c + m₀
  · obtain ⟨m₀, hm₀, hdiv⟩ := hex
    have h2 := abs_le.1 hm₀
    -- the window meets at most one multiple of `2^F`
    have huniq : ∀ m : ℤ, |m| ≤ k → (2 : ℤ) ^ F ∣ c + m → m = m₀ := by
      intro m hm hdm
      have h1 := abs_le.1 hm
      have hsub : (2 : ℤ) ^ F ∣ m - m₀ := by
        have h3 := dvd_sub hdm hdiv
        rwa [show c + m - (c + m₀) = m - m₀ by ring] at h3
      have h4 := Int.eq_zero_of_abs_lt_dvd hsub (abs_lt.2 ⟨by linarith, by linarith⟩)
      linarith
    -- place that multiple at `2^F` or at `2^(F+1)`, whichever carries the right parity
    obtain ⟨e, he, hFe, hlow, hhigh⟩ : ∃ e : ℕ, toeplitz (c + m₀) = Nat.bodd e ∧ F ≤ e ∧
        (2 : ℤ) ^ F ≤ 2 ^ e ∧ (2 : ℤ) ^ e ≤ 2 * 2 ^ F := by
      have hbit : toeplitz (c + m₀) = Nat.bodd F ∨ toeplitz (c + m₀) = Nat.bodd (F + 1) := by
        rw [show F + 1 = Nat.succ F from rfl, Nat.bodd_succ]
        cases toeplitz (c + m₀) <;> cases Nat.bodd F <;> decide
      rcases hbit with h | h
      · exact ⟨F, h, le_rfl, le_rfl, by linarith⟩
      · exact ⟨F + 1, h, Nat.le_succ F, by rw [hpow]; linarith, hpow.le⟩
    refine ⟨2 ^ e - m₀, fun m hm => ?_⟩
    simp only [shift_apply]
    have h1 := abs_le.1 hm
    rw [periodicToeplitz_eq F (n := 2 ^ e - m₀ + m) (by linarith) (by linarith)]
    rcases eq_or_ne m m₀ with rfl | hne
    · rw [he, sub_add_cancel, ← mul_one ((2 : ℤ) ^ e),
        toeplitz_two_pow_mul (e := e) (s := 1) (by omega)]
    · have hndvd : ¬ (2 : ℤ) ^ F ∣ c + m := fun hd => hne (huniq m hm hd)
      have hd : (2 : ℤ) ^ F ∣ 2 ^ e - (c + m₀) := dvd_sub (pow_dvd_pow 2 hFe) hdiv
      rw [show (2 : ℤ) ^ e - m₀ + m = (c + m) + (2 ^ e - (c + m₀)) by ring,
        toeplitz_add_of_dvd hndvd hd]
  · push Not at hex
    refine ⟨c % 2 ^ F + 2 ^ F, fun m hm => ?_⟩
    simp only [shift_apply]
    have h0 := Int.emod_nonneg c hpos.ne'
    have h1 := Int.emod_lt_of_pos c hpos
    have hm' := abs_le.1 hm
    have hdec := Int.emod_add_mul_ediv c (2 ^ F)
    rw [periodicToeplitz_eq F (n := c % 2 ^ F + 2 ^ F + m) (by linarith) (by linarith),
      show c % 2 ^ F + 2 ^ F + m = (c + m) + 2 ^ F * (1 - c / 2 ^ F) by
        linear_combination hdec,
      toeplitz_add_of_dvd (hex m hm) (dvd_mul_right _ _)]

/-- Periodic return words. For every radius `k` and bound `B` there are a period `N ≥ B` and
an `N`-periodic configuration `z` such that every window of radius `k` of `z` is a window of a
point of the Toeplitz subshift, and every window of radius `k` of such a point is a window of
`z`. -/
theorem exists_periodic_model (k B : ℕ) :
    ∃ (N : ℕ) (z : ℤ → Bool), 0 < N ∧ B ≤ N ∧ (∀ n : ℤ, z (n + N) = z n) ∧
      (∀ n : ℤ, ∃ y ∈ toeplitzSubshift.carrier, Agree k (shift n z) y) ∧
      (∀ y ∈ toeplitzSubshift.carrier, ∃ n : ℤ, Agree k y (shift n z)) := by
  obtain ⟨F, hF⟩ : ∃ F : ℕ, F = 2 * k + B := ⟨_, rfl⟩
  have hkF : 2 * k < 2 ^ F := lt_of_le_of_lt (by omega) (Nat.lt_pow_self Nat.one_lt_two)
  have hFF : 2 ^ F ≤ 2 ^ (F + 1) := Nat.pow_le_pow_right Nat.zero_lt_two (Nat.le_succ F)
  have hBF : B < 2 ^ (F + 1) :=
    lt_of_le_of_lt (by omega) (lt_of_lt_of_le (Nat.lt_pow_self Nat.one_lt_two) hFF)
  refine ⟨2 ^ (F + 1), periodicToeplitz F, by positivity, hBF.le, fun n => ?_, fun n => ?_,
    fun y hy => ?_⟩
  · simpa using periodicToeplitz_add_mul F n 1
  · obtain ⟨c, hc⟩ := periodicToeplitz_window_legal (lt_of_lt_of_le hkF hFF) n
    exact ⟨shift c toeplitz, toeplitzSubshift.mapsTo c (self_mem_orbitClosure toeplitz), hc⟩
  · obtain ⟨c, hc⟩ := mem_orbitClosure_iff.1 hy k
    obtain ⟨n, hn⟩ := toeplitz_window_occurs hkF c
    exact ⟨n, hc.trans hn⟩

end GroupApproximation.Pestov91
