import GroupApproximation.Pestov91.NormalTower
import Mathlib.Data.Matrix.Composition

/-!
# Pestov 9.1: the corner monoid `Q_U` of a tower

Part 2 §6.2(b) of `research/artifacts/pestov-9-1-writeup-2026-09-13-part2.md`.
With `e = χ_U` and `t = towerUnit e m = χ_V`, `V = ⊔_{|a| ≤ m} T^a U`, the map

  `cornerHom F = (1 - t) I + (towerMap e m)_* F`,
  `F ∈ M_ι(M_(2m+1)(A)) = M_(ι × [-m,m])(A)`,

is a monoid map into `M_ι(R)`: its range is `Q_U = (1-t) I + M_ι(B_m(U))`, and its
multiplication is the one of `M_(ι × [-m,m])(A)`, pointwise on `U`.

* `cornerHom_eq_iff`: two matrices have the same image iff their entries agree
  after cutting by `e`, i.e. on `U` (this uses that `coeff` is injective).
* `exists_cornerHom_eq`: every `k` with `k - 1 ∈ M_ι(B_m(U))` is in the range.
-/

namespace GroupApproximation.Pestov91.TowerSetting

variable {A R : Type*} [CommRing A] [Ring R] (S : TowerSetting A R)

/-- The unit `χ_V` of the tower subring `B_m(U)`. -/
def towerUnit (e : A) (m : ℕ) : R := S.towerMap e m 1

theorem towerUnit_mem (e : A) (m : ℕ) : S.towerUnit e m ∈ S.tower e m :=
  S.towerMap_mem e m 1

theorem towerUnit_mul {e : A} (he : IsIdempotentElem e) {m : ℕ} (hret : S.LongReturn e (2 * m))
    {r : R} (hr : r ∈ S.tower e m) : S.towerUnit e m * r = r := by
  obtain ⟨F, rfl⟩ := hr
  rw [towerUnit, ← S.towerMap_mul he hret, one_mul]

theorem mul_towerUnit {e : A} (he : IsIdempotentElem e) {m : ℕ} (hret : S.LongReturn e (2 * m))
    {r : R} (hr : r ∈ S.tower e m) : r * S.towerUnit e m = r := by
  obtain ⟨F, rfl⟩ := hr
  rw [towerUnit, ← S.towerMap_mul he hret, mul_one]

/-- A tower matrix vanishes iff all its entries vanish on `U`. -/
theorem towerMap_eq_zero_iff {e : A} (he : IsIdempotentElem e) {m : ℕ}
    (hret : S.LongReturn e (2 * m)) (hinj : Function.Injective S.coeff)
    (F : Matrix (Level m) (Level m) A) : S.towerMap e m F = 0 ↔ ∀ a b, F a b * e = 0 := by
  constructor
  · intro h a b
    have hab := congrArg
      (fun r ↦ S.towerMap e m (Matrix.single a a 1) * r * S.towerMap e m (Matrix.single b b 1)) h
    simp only [mul_zero, zero_mul] at hab
    rw [← S.towerMap_mul he hret, ← S.towerMap_mul he hret, Matrix.single_mul_mul_single,
      one_mul, mul_one, S.towerMap_single, towerElem, Units.mul_left_eq_zero,
      map_eq_zero_iff _ hinj, map_eq_zero_iff _ (S.α ^ (a : ℤ)).injective] at hab
    exact hab
  · intro h
    rw [towerMap_apply]
    refine Finset.sum_eq_zero fun a _ ↦ Finset.sum_eq_zero fun b _ ↦ ?_
    rw [towerElem, h a b, map_zero, map_zero, zero_mul]

variable {ι : Type*} [Fintype ι] [DecidableEq ι]

/-- The tower map applied entrywise to `M_ι(M_(2m+1)(A)) = M_(ι × [-m,m])(A)`. -/
def cornerLinear (e : A) (m : ℕ) : Matrix (ι × Level m) (ι × Level m) A →+ Matrix ι ι R :=
  (S.towerMap e m).mapMatrix.comp
    (Matrix.compRingEquiv ι (Level m) A).symm.toRingHom.toAddMonoidHom

theorem cornerLinear_apply (e : A) (m : ℕ) (F : Matrix (ι × Level m) (ι × Level m) A)
    (p q : ι) :
    S.cornerLinear e m F p q = S.towerMap e m ((Matrix.compRingEquiv ι (Level m) A).symm F p q) :=
  rfl

theorem cornerLinear_mul {e : A} (he : IsIdempotentElem e) {m : ℕ}
    (hret : S.LongReturn e (2 * m)) (F G : Matrix (ι × Level m) (ι × Level m) A) :
    S.cornerLinear e m (F * G) = S.cornerLinear e m F * S.cornerLinear e m G := by
  ext p q
  rw [cornerLinear_apply, map_mul, Matrix.mul_apply, Matrix.mul_apply, map_sum]
  exact Finset.sum_congr rfl fun r _ ↦ S.towerMap_mul he hret _ _

theorem cornerLinear_one (e : A) (m : ℕ) :
    S.cornerLinear e m (1 : Matrix (ι × Level m) (ι × Level m) A) =
      Matrix.diagonal fun _ ↦ S.towerUnit e m := by
  ext p q
  rw [cornerLinear_apply, map_one, Matrix.one_apply, Matrix.diagonal_apply]
  split_ifs
  · rfl
  · exact map_zero _

/-- **The corner monoid `Q_U`**: `F ↦ (1 - χ_V) I + (towerMap e m)_* F`. -/
def cornerHom {e : A} (he : IsIdempotentElem e) {m : ℕ} (hret : S.LongReturn e (2 * m)) :
    Matrix (ι × Level m) (ι × Level m) A →* Matrix ι ι R where
  toFun F := Matrix.diagonal (fun _ ↦ 1 - S.towerUnit e m) + S.cornerLinear e m F
  map_one' := by
    dsimp only
    rw [cornerLinear_one, Matrix.diagonal_add, sub_add_cancel, Matrix.diagonal_one]
  map_mul' F G := by
    dsimp only
    have ht : IsIdempotentElem (S.towerUnit e m) := S.towerUnit_mul he hret (S.towerUnit_mem e m)
    have hDL : Matrix.diagonal (fun _ ↦ 1 - S.towerUnit e m) * S.cornerLinear e m G = 0 := by
      ext p q
      rw [Matrix.diagonal_mul, sub_mul, one_mul, cornerLinear_apply,
        S.towerUnit_mul he hret (S.towerMap_mem e m _), sub_self, Matrix.zero_apply]
    have hLD : S.cornerLinear e m F * Matrix.diagonal (fun _ ↦ 1 - S.towerUnit e m) = 0 := by
      ext p q
      rw [Matrix.mul_diagonal, mul_sub, mul_one, cornerLinear_apply,
        S.mul_towerUnit he hret (S.towerMap_mem e m _), sub_self, Matrix.zero_apply]
    have hDD : Matrix.diagonal (fun _ ↦ 1 - S.towerUnit e m) *
        Matrix.diagonal (fun _ ↦ 1 - S.towerUnit e m) =
          Matrix.diagonal (fun _ ↦ 1 - S.towerUnit e m) := by
      rw [Matrix.diagonal_mul_diagonal, ht.one_sub.eq]
    simp only [add_mul, mul_add, hDL, hLD, hDD, S.cornerLinear_mul he hret, add_zero, zero_add]

theorem cornerHom_apply {e : A} (he : IsIdempotentElem e) {m : ℕ}
    (hret : S.LongReturn e (2 * m)) (F : Matrix (ι × Level m) (ι × Level m) A) :
    S.cornerHom he hret F =
      Matrix.diagonal (fun _ ↦ 1 - S.towerUnit e m) + S.cornerLinear e m F :=
  rfl

/-- **Pointwise multiplication on `U`**: the corner monoid identifies exactly the
matrices that agree on `U`. -/
theorem cornerHom_eq_iff {e : A} (he : IsIdempotentElem e) {m : ℕ}
    (hret : S.LongReturn e (2 * m)) (hinj : Function.Injective S.coeff)
    (F G : Matrix (ι × Level m) (ι × Level m) A) :
    S.cornerHom he hret F = S.cornerHom he hret G ↔ ∀ p q, F p q * e = G p q * e := by
  rw [cornerHom_apply, cornerHom_apply, add_right_inj, ← sub_eq_zero, ← map_sub]
  constructor
  · intro h p q
    have hpq := congrFun (congrFun h p.1) q.1
    rw [Matrix.zero_apply, cornerLinear_apply, S.towerMap_eq_zero_iff he hret hinj] at hpq
    simpa [sub_mul, sub_eq_zero, Matrix.compRingEquiv_symm_apply] using hpq p.2 q.2
  · intro h
    ext p q
    rw [Matrix.zero_apply, cornerLinear_apply, S.towerMap_eq_zero_iff he hret hinj]
    intro a b
    simpa [sub_mul, sub_eq_zero, Matrix.compRingEquiv_symm_apply] using h (p, a) (q, b)

/-- Every `k` with `k - 1 ∈ M_ι(B_m(U))` lies in the corner monoid. -/
theorem exists_cornerHom_eq {e : A} (he : IsIdempotentElem e) {m : ℕ}
    (hret : S.LongReturn e (2 * m)) {k : Matrix ι ι R}
    (hk : ∀ p q, (k - 1) p q ∈ S.tower e m) : ∃ F, S.cornerHom he hret F = k := by
  have hk' : ∀ p q, ∃ F, S.towerMap e m F = (k - 1) p q := fun p q ↦
    AddMonoidHom.mem_range.mp (hk p q)
  choose K hK using hk'
  refine ⟨1 + Matrix.compRingEquiv ι (Level m) A (Matrix.of K), ?_⟩
  have hL : S.cornerLinear e m (Matrix.compRingEquiv ι (Level m) A (Matrix.of K)) = k - 1 := by
    ext p q
    rw [cornerLinear_apply, RingEquiv.symm_apply_apply]
    exact hK p q
  rw [cornerHom_apply, map_add, cornerLinear_one, ← add_assoc, Matrix.diagonal_add,
    sub_add_cancel, Matrix.diagonal_one, hL]
  abel

end GroupApproximation.Pestov91.TowerSetting
