/-!
# The centre of the Pestov 9.1 witness ring, and the trivial centre of `EL₃`

The witness ring is a skew monoid algebra `R = A ⋊ G`, with `A = LC(X, 𝔽₂)` and `G = ℤ`
written multiplicatively (blueprint D3).  §1.4 of
`research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md` computes
`Z(R) = 𝔽_q`, and §4.4 turns that into `Z(EL₃(R)) ⊆ 𝔽_q^× I₃`.  Over `𝔽₂` the centre of
`EL₃(R)` is therefore trivial, and the witness is `EL₃(R)` itself (blueprint D5).

This module proves the ring half without committing to a realisation of `A` or of the action:

* `skewMonoidAlgebra_central_zero_or_one`.  Let a commutative group `G` act on a ring `A`, and
  suppose every `G`-invariant element of `A` is `0` or `1` and every `g ≠ 1` moves some
  element.  Then every central element of `A ⋊ G` is `0` or `1`.  Commuting with `single g 1`
  makes each coefficient invariant, hence `0` or `1`.  Commuting with `single 1 a` gives
  `c_g * (g • a) = a * c_g`, so `c_g = 1` would make `g` act trivially.
* `skewMonoidAlgebra_centralUnits_trivial`: the only central unit is `1`.
* `center_elementaryGroup_skewMonoidAlgebra_eq_bot`: `Z(EL_ι(A ⋊ G)) = ⊥`.  It goes through
  `center_elementaryGroup_eq_bot_of_central_zero_or_one` from
  `GroupApproximation/Pestov91/Centre.lean`.
* For `A = LC(X, 𝔽₂)` with `(g • f) x = f (τ g x)`, the two hypotheses come from the dynamics.
  `locallyConstant_invariant_zero_or_one` needs dense orbits: an invariant function is
  constant on each orbit and has open fibres.  `locallyConstant_exists_smul_ne` needs
  freeness and total separation: a clopen set containing `x` but not `τ g x` has an indicator
  that `g` moves.

Both hypotheses are needed.  For the trivial action `A ⋊ G` is the group ring `A[G]`, which is
commutative when `A` is.
-/

namespace GroupApproximation
namespace Pestov91

section Coefficients

variable {R : Type*} [Ring R]

/-- If every central element of `R` is `0` or `1`, the only central unit of `R` is `1`.
The zero case forces `0 = 1`, where every unit is `1` anyway. -/
theorem centralUnits_trivial_of_central_zero_or_one
    (hR : ∀ c : R, (∀ r : R, r * c = c * r) → c = 0 ∨ c = 1) :
    ∀ v : Rˣ, (∀ y : R, (v : R) * y = y * (v : R)) → v = 1 := by
  intro v hv
  apply Units.ext
  rw [Units.val_one]
  rcases hR v fun y ↦ (hv y).symm with h0 | h1
  · calc (v : R) = 0 := h0
      _ = (v : R) * ↑v⁻¹ := by rw [h0, zero_mul]
      _ = 1 := v.mul_inv
  · exact h1

end Coefficients

section SkewMonoidAlgebra

variable {A G : Type*} [Ring A] [CommGroup G] [MulSemiringAction G A]

/-- **The centre of a crossed product by a free action with trivial invariants** is `{0, 1}`. -/
theorem skewMonoidAlgebra_central_zero_or_one
    (hinv : ∀ f : A, (∀ g : G, g • f = f) → f = 0 ∨ f = 1)
    (hfree : ∀ g : G, g ≠ 1 → ∃ a : A, g • a ≠ a) :
    ∀ c : SkewMonoidAlgebra A G, (∀ r : SkewMonoidAlgebra A G, r * c = c * r) →
      c = 0 ∨ c = 1 := by
  classical
  intro c hc
  have hinvariant : ∀ x g : G, g • c.coeff x = c.coeff x := by
    intro x g
    have h := congrArg (fun r : SkewMonoidAlgebra A G ↦ r.coeff (g * x))
      (hc (SkewMonoidAlgebra.single g 1))
    simp only [SkewMonoidAlgebra.coeff_single_mul, SkewMonoidAlgebra.coeff_mul_single,
      MulSemiringAction.smul_one, one_mul, mul_one, inv_mul_cancel_left,
      mul_inv_cancel_comm] at h
    exact h
  have hzero : ∀ x : G, x ≠ 1 → c.coeff x = 0 := by
    intro x hx
    rcases hinv _ (hinvariant x) with h0 | h1
    · exact h0
    · obtain ⟨a, ha⟩ := hfree x hx
      refine absurd ?_ ha
      have h := congrArg (fun r : SkewMonoidAlgebra A G ↦ r.coeff x)
        (hc (SkewMonoidAlgebra.single (1 : G) a))
      simp only [SkewMonoidAlgebra.coeff_single_mul, SkewMonoidAlgebra.coeff_mul_single,
        inv_one, one_mul, mul_one, one_smul, h1] at h
      exact h.symm
  have hc_eq : c = SkewMonoidAlgebra.single 1 (c.coeff 1) := by
    refine SkewMonoidAlgebra.ext fun x ↦ ?_
    rw [SkewMonoidAlgebra.coeff_single_apply]
    split_ifs with hx
    · subst hx
      rfl
    · exact hzero x (Ne.symm hx)
  rcases hinv _ (hinvariant 1) with h0 | h1
  · left
    rw [hc_eq, h0, SkewMonoidAlgebra.single_zero_right]
  · right
    rw [hc_eq, h1, SkewMonoidAlgebra.single_one_one]

/-- The only central unit of `A ⋊ G` is `1`, under the same two hypotheses. -/
theorem skewMonoidAlgebra_centralUnits_trivial
    (hinv : ∀ f : A, (∀ g : G, g • f = f) → f = 0 ∨ f = 1)
    (hfree : ∀ g : G, g ≠ 1 → ∃ a : A, g • a ≠ a) :
    ∀ v : (SkewMonoidAlgebra A G)ˣ,
      (∀ y : SkewMonoidAlgebra A G,
        (v : SkewMonoidAlgebra A G) * y = y * (v : SkewMonoidAlgebra A G)) → v = 1 :=
  centralUnits_trivial_of_central_zero_or_one (skewMonoidAlgebra_central_zero_or_one hinv hfree)

/-- **`Z(EL_ι(A ⋊ G)) = 1`** for a free action with trivial invariants and two indices. -/
theorem center_elementaryGroup_skewMonoidAlgebra_eq_bot {ι : Type*} [Fintype ι]
    [DecidableEq ι] [Nontrivial ι]
    (hinv : ∀ f : A, (∀ g : G, g • f = f) → f = 0 ∨ f = 1)
    (hfree : ∀ g : G, g ≠ 1 → ∃ a : A, g • a ≠ a) :
    Subgroup.center ↥(elementaryGroup ι (SkewMonoidAlgebra A G)) = ⊥ :=
  center_elementaryGroup_eq_bot_of_central_zero_or_one
    (skewMonoidAlgebra_central_zero_or_one hinv hfree)

end SkewMonoidAlgebra

section LocallyConstant

variable {X G : Type*} [TopologicalSpace X] [Group G]
  [MulSemiringAction G (LocallyConstant X (ZMod 2))]

/-- **Dense orbits give trivial invariants.**  If `G` acts on `LC(X, 𝔽₂)` by
`(g • f) x = f (τ g x)` and every orbit of `τ` is dense, a `G`-invariant function is constant,
so it is `0` or `1`. -/
theorem locallyConstant_invariant_zero_or_one (τ : G → X → X)
    (hτ : ∀ (g : G) (f : LocallyConstant X (ZMod 2)) (x : X), (g • f) x = f (τ g x))
    (hdense : ∀ x : X, Dense (Set.range fun g : G ↦ τ g x)) :
    ∀ f : LocallyConstant X (ZMod 2), (∀ g : G, g • f = f) → f = 0 ∨ f = 1 := by
  intro f hf
  by_cases hX : Nonempty X
  · obtain ⟨x₀⟩ := hX
    have hconst : ∀ x : X, f x = f x₀ := by
      intro x
      obtain ⟨_, ⟨g, rfl⟩, hg⟩ := (hdense x).exists_mem_open
        (f.isLocallyConstant.isOpen_fiber (f x₀)) ⟨x₀, rfl⟩
      have hg' : f (τ g x) = f x₀ := hg
      rw [← hg', ← hτ g f x, hf g]
    rcases (by decide : ∀ c : ZMod 2, c = 0 ∨ c = 1) (f x₀) with h0 | h1
    · exact Or.inl (LocallyConstant.ext fun x ↦ by simp [hconst x, h0])
    · exact Or.inr (LocallyConstant.ext fun x ↦ by simp [hconst x, h1])
  · exact Or.inl (LocallyConstant.ext fun x ↦ (hX ⟨x⟩).elim)

/-- **Freeness moves a clopen indicator.**  If `(g • f) x = f (τ g x)`, `X` is totally
separated and every `g ≠ 1` moves some point, then every `g ≠ 1` moves some locally constant
function. -/
theorem locallyConstant_exists_smul_ne [TotallySeparatedSpace X] (τ : G → X → X)
    (hτ : ∀ (g : G) (f : LocallyConstant X (ZMod 2)) (x : X), (g • f) x = f (τ g x))
    (hfree : ∀ g : G, g ≠ 1 → ∃ x : X, τ g x ≠ x) :
    ∀ g : G, g ≠ 1 → ∃ a : LocallyConstant X (ZMod 2), g • a ≠ a := by
  intro g hg
  obtain ⟨x, hx⟩ := hfree g hg
  obtain ⟨U, hU, hxU, hgxU⟩ := exists_isClopen_of_totally_separated hx.symm
  have hgxU' : τ g x ∉ U := hgxU
  refine ⟨LocallyConstant.charFn (ZMod 2) hU, fun h ↦ ?_⟩
  have h1 := congrArg (fun a : LocallyConstant X (ZMod 2) ↦ a x) h
  simp only [hτ, LocallyConstant.coe_charFn, Set.indicator_of_mem hxU,
    Set.indicator_of_notMem hgxU', Pi.one_apply] at h1
  exact (by decide : (0 : ZMod 2) ≠ 1) h1

end LocallyConstant
