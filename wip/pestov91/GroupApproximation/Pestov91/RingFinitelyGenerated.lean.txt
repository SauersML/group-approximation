/-!
# The crossed-product ring of a Bool subshift is finitely generated

The Pestov 9.1 witness is `S = EL₃(R)/Z` with `R = LC(X, 𝔽₂) ⋊ ℤ` for an
infinite minimal subshift `X ⊆ Bool^ℤ`.  Property `(T)` for `EL₃(R)` comes from
`IntegralColumnPlaneClosure.finitelyGeneratedRingGeneralRankElementaryPropertyT`,
whose coefficient hypothesis is `IsFinitelyGeneratedRing R`: some finite set
generates `R` as a ring with `1`.  This module proves that hypothesis.

The generators are the indicators of the two one-coordinate cylinders
`[x₀ = false]` and `[x₀ = true]`, together with the implementing unit `u` and
its inverse.  The proof has two halves.

1. **Coefficients** (`closure_range_coordinateIndicator`).  On a compact set
   `X` of Bool sequences, every locally constant `ZMod 2`-valued function lies
   in the subring generated by the coordinate indicators `[xᵢ = b]`, `i ∈ ℤ`.
   Compactness gives a finite window `I` such that the function depends only
   on `x|_I` (`exists_finset_dependsOn`); induction on `I` then splits the
   function along one coordinate at a time.

2. **The crossed product** (`subring_closure_eq_top_of_shift`).  Conjugation
   by `u` moves `[xᵢ = b]` to `[xᵢ₊₁ = b]`, so the subring generated by
   `[x₀ = b]`, `u` and `u⁻¹` contains every coordinate indicator, so every
   coefficient, so everything.  If the covariance moves `[xᵢ = b]` the other
   way, exchange `u` and its inverse.

The crossed-product half is stated for an arbitrary ring `R` with a
coefficient ring hom, a two-sided inverse pair and the covariance relation, so
it does not depend on how `R` is constructed.
-/

namespace GroupApproximation

namespace Pestov91

/-! ### Coordinate indicators on a set of Bool sequences -/

section Coordinates

variable (X : Set (ℤ → Bool))

/-- The indicator of the one-coordinate cylinder `[xᵢ = b]`, as a locally
constant `ZMod 2`-valued function on `X`. -/
def coordinateIndicator (i : ℤ) (b : Bool) : LocallyConstant X (ZMod 2) where
  toFun x := if x.1 i = b then 1 else 0
  isLocallyConstant :=
    ((IsLocallyConstant.iff_continuous fun x : X => x.1 i).mpr
      ((continuous_apply i).comp continuous_subtype_val)).comp
      fun c : Bool => if c = b then (1 : ZMod 2) else 0

theorem coordinateIndicator_apply (i : ℤ) (b : Bool) (x : X) :
    coordinateIndicator X i b x = if x.1 i = b then 1 else 0 :=
  rfl

/-- The two cylinders `[xᵢ = false]` and `[xᵢ = true]` partition `X`. -/
theorem coordinateIndicator_false_add_true (i : ℤ) :
    coordinateIndicator X i false + coordinateIndicator X i true = 1 := by
  refine LocallyConstant.ext fun x => ?_
  rw [LocallyConstant.add_apply, coordinateIndicator_apply, coordinateIndicator_apply,
    LocallyConstant.one_apply]
  cases x.1 i <;> simp

/-- A shift of the sequences moves the coordinate indicators by the same
offset. -/
theorem comap_coordinateIndicator (T : C(X, X)) (d : ℤ)
    (hT : ∀ (x : X) (n : ℤ), (T x).1 n = x.1 (n + d)) (i : ℤ) (b : Bool) :
    LocallyConstant.comap T (coordinateIndicator X i b) =
      coordinateIndicator X (i + d) b :=
  LocallyConstant.ext fun x => by
    rw [LocallyConstant.coe_comap_apply, coordinateIndicator_apply,
      coordinateIndicator_apply, hT]

/-- **A locally constant function on a compact set of sequences depends on
finitely many coordinates.** -/
theorem exists_finset_dependsOn [CompactSpace X] {Y : Type*}
    (f : LocallyConstant X Y) :
    ∃ I : Finset ℤ, ∀ x y : X, (∀ i ∈ I, x.1 i = y.1 i) → f x = f y := by
  have hloc : ∀ x : X, ∃ I : Finset ℤ,
      ∀ y : X, (∀ i ∈ I, y.1 i = x.1 i) → f y = f x := by
    intro x
    obtain ⟨t, ht, hts⟩ :=
      isOpen_induced_iff.mp (f.isLocallyConstant.isOpen_fiber (f x))
    have hmem : ∀ y : X, y.1 ∈ t ↔ f y = f x := fun y => Set.ext_iff.mp hts y
    obtain ⟨I, U, hU, hIU⟩ := isOpen_pi_iff.mp ht x.1 ((hmem x).mpr rfl)
    refine ⟨I, fun y hy => (hmem y).mp (hIU (Set.mem_pi.mpr fun i hi => ?_))⟩
    rw [hy i hi]
    exact (hU i hi).2
  choose J hJ using hloc
  obtain ⟨t, ht⟩ := isCompact_univ.elim_finite_subcover
    (fun z : X => Subtype.val ⁻¹' ((J z : Set ℤ).pi fun i => {z.1 i}))
    (fun z => (isOpen_set_pi (J z).finite_toSet fun i _ => isOpen_discrete _).preimage
      continuous_subtype_val)
    (fun z _ => Set.mem_iUnion.mpr ⟨z, Set.mem_pi.mpr fun _ _ => Set.mem_singleton _⟩)
  refine ⟨t.biUnion J, fun x y hxy => ?_⟩
  obtain ⟨z, hz, hxz⟩ := Set.mem_iUnion₂.mp (ht (Set.mem_univ x))
  have hxz' : ∀ i ∈ J z, x.1 i = z.1 i := hxz
  have hyz : ∀ i ∈ J z, y.1 i = z.1 i := fun i hi =>
    (hxy i (Finset.mem_biUnion.mpr ⟨z, hz, hi⟩)).symm.trans (hxz' i hi)
  exact (hJ z x hxz').trans (hJ z y hyz).symm

/-- **Functions of finitely many coordinates restrict into the coordinate
subring.**  If `g` on Bool sequences depends only on the coordinates in `I`,
its restriction to `X` lies in the subring generated by the coordinate
indicators.  The induction splits `g` along one coordinate at a time. -/
theorem exists_mem_closure_coordinateIndicator (I : Finset ℤ) :
    ∀ g : (ℤ → Bool) → ZMod 2,
      (∀ y z : ℤ → Bool, (∀ i ∈ I, y i = z i) → g y = g z) →
      ∃ c ∈ Subring.closure
          (Set.range fun p : ℤ × Bool => coordinateIndicator X p.1 p.2),
        ∀ x : X, c x = g x.1 := by
  classical
  induction I using Finset.induction_on with
  | empty =>
    intro g hg
    refine ⟨((g fun _ => false).val : LocallyConstant X (ZMod 2)),
      natCast_mem _ _, fun x => ?_⟩
    rw [hg x.1 (fun _ => false) fun i hi => absurd hi (Finset.notMem_empty i)]
    exact ZMod.natCast_zmod_val _
  | insert i J hi ih =>
    intro g hg
    have hsplit : ∀ (b : Bool) (y z : ℤ → Bool), (∀ j ∈ J, y j = z j) →
        g (Function.update y i b) = g (Function.update z i b) := by
      intro b y z hyz
      refine hg _ _ fun j hj => ?_
      rcases Finset.mem_insert.mp hj with rfl | hj
      · simp only [Function.update_self]
      · have hji : j ≠ i := fun h => hi (h ▸ hj)
        rw [Function.update_of_ne hji, Function.update_of_ne hji]
        exact hyz j hj
    obtain ⟨c₀, hc₀, hc₀x⟩ :=
      ih (fun y => g (Function.update y i false)) (hsplit false)
    obtain ⟨c₁, hc₁, hc₁x⟩ :=
      ih (fun y => g (Function.update y i true)) (hsplit true)
    refine ⟨coordinateIndicator X i false * c₀ + coordinateIndicator X i true * c₁,
      Subring.add_mem _
        (Subring.mul_mem _ (Subring.subset_closure ⟨(i, false), rfl⟩) hc₀)
        (Subring.mul_mem _ (Subring.subset_closure ⟨(i, true), rfl⟩) hc₁),
      fun x => ?_⟩
    have hx : Function.update x.1 i (x.1 i) = x.1 := Function.update_eq_self i x.1
    rw [LocallyConstant.add_apply, LocallyConstant.mul_apply, LocallyConstant.mul_apply,
      hc₀x x, hc₁x x, coordinateIndicator_apply, coordinateIndicator_apply]
    cases h : x.1 i
    · rw [h] at hx
      simp [hx]
    · rw [h] at hx
      simp [hx]

/-- **The coordinate indicators generate the coefficient ring.**  On a compact
set `X` of Bool sequences, every locally constant `ZMod 2`-valued function
lies in the subring generated by the indicators `[xᵢ = b]`. -/
theorem closure_range_coordinateIndicator [CompactSpace X] :
    Subring.closure
        (Set.range fun p : ℤ × Bool => coordinateIndicator X p.1 p.2) = ⊤ := by
  classical
  refine (Subring.eq_top_iff' _).mpr fun f => ?_
  obtain ⟨I, hI⟩ := exists_finset_dependsOn X f
  let g : (ℤ → Bool) → ZMod 2 := fun y =>
    if h : ∃ x : X, ∀ i ∈ I, x.1 i = y i then f h.choose else 0
  have hg : ∀ y z : ℤ → Bool, (∀ i ∈ I, y i = z i) → g y = g z := by
    intro y z hyz
    by_cases hy : ∃ x : X, ∀ i ∈ I, x.1 i = y i
    · have hz : ∃ x : X, ∀ i ∈ I, x.1 i = z i :=
        ⟨hy.choose, fun i hi => (hy.choose_spec i hi).trans (hyz i hi)⟩
      dsimp only [g]
      rw [dif_pos hy, dif_pos hz]
      exact hI _ _ fun i hi =>
        (hy.choose_spec i hi).trans ((hyz i hi).trans (hz.choose_spec i hi).symm)
    · have hz : ¬∃ x : X, ∀ i ∈ I, x.1 i = z i := fun ⟨x, hx⟩ =>
        hy ⟨x, fun i hi => (hx i hi).trans (hyz i hi).symm⟩
      dsimp only [g]
      rw [dif_neg hy, dif_neg hz]
  obtain ⟨c, hc, hcx⟩ := exists_mem_closure_coordinateIndicator X I g hg
  have hx : ∀ x : X, ∃ x' : X, ∀ i ∈ I, x'.1 i = x.1 i := fun x => ⟨x, fun _ _ => rfl⟩
  have hcf : c = f := LocallyConstant.ext fun x => by
    rw [hcx x]
    dsimp only [g]
    rw [dif_pos (hx x)]
    exact hI _ _ (hx x).choose_spec
  exact hcf ▸ hc

/-- The same for a closed set of sequences, which is compact. -/
theorem closure_range_coordinateIndicator_of_isClosed (hX : IsClosed X) :
    Subring.closure
        (Set.range fun p : ℤ × Bool => coordinateIndicator X p.1 p.2) = ⊤ :=
  haveI : CompactSpace X := isCompact_iff_compactSpace.mp hX.isCompact
  closure_range_coordinateIndicator X

/-! #### Calibration

`coordinateIndicator` is not degenerate: on the full shift the indicator of
`[x₀ = true]` is neither `0` nor `1`, and the generation theorem applies there,
the full shift being compact. -/

theorem coordinateIndicator_univ_ne_zero :
    coordinateIndicator (Set.univ : Set (ℤ → Bool)) 0 true ≠ 0 := by
  intro h
  have h1 := LocallyConstant.congr_fun h ⟨fun _ => true, Set.mem_univ _⟩
  simp [coordinateIndicator_apply] at h1

theorem coordinateIndicator_univ_ne_one :
    coordinateIndicator (Set.univ : Set (ℤ → Bool)) 0 true ≠ 1 := by
  intro h
  have h1 := LocallyConstant.congr_fun h ⟨fun _ => false, Set.mem_univ _⟩
  simp [coordinateIndicator_apply] at h1

theorem closure_range_coordinateIndicator_univ :
    Subring.closure (Set.range fun p : ℤ × Bool =>
      coordinateIndicator (Set.univ : Set (ℤ → Bool)) p.1 p.2) = ⊤ :=
  closure_range_coordinateIndicator_of_isClosed _ isClosed_univ

end Coordinates

/-! ### Adjoining a shifting unit -/

section Shift

variable {A R : Type*} [Ring A] [Ring R]

/-- **The shift generation lemma.**  Let `ι : A →+* R`, let `u, v` be mutually
inverse in `R`, and let `e i b` be elements of `A` generating `A` as a ring,
such that conjugation by `u` sends `ι (e i b)` to `ι (e (i + 1) b)`.  If `R` is
generated by `ι A` together with `u` and `v`, it is generated by the four
elements `ι (e 0 false)`, `ι (e 0 true)`, `u`, `v`. -/
theorem subring_closure_eq_top_of_shift (ι : A →+* R) (u v : R)
    (huv : u * v = 1) (hvu : v * u = 1) (e : ℤ → Bool → A)
    (hA : Subring.closure (Set.range fun p : ℤ × Bool => e p.1 p.2) = ⊤)
    (hshift : ∀ i b, ι (e (i + 1) b) = u * ι (e i b) * v)
    (hR : Subring.closure (Set.range ι ∪ {u, v}) = ⊤) :
    Subring.closure ({ι (e 0 false), ι (e 0 true), u, v} : Set R) = ⊤ := by
  set S := Subring.closure ({ι (e 0 false), ι (e 0 true), u, v} : Set R)
  have hu : u ∈ S := Subring.subset_closure (by simp)
  have hv : v ∈ S := Subring.subset_closure (by simp)
  have he : ∀ i b, ι (e i b) ∈ S := by
    intro i b
    induction i using Int.induction_on with
    | zero => cases b <;> exact Subring.subset_closure (by simp)
    | succ n ih =>
      rw [hshift]
      exact S.mul_mem (S.mul_mem hu ih) hv
    | pred n ih =>
      have h := hshift (-(n : ℤ) - 1) b
      rw [sub_add_cancel] at h
      have hback : ι (e (-(n : ℤ) - 1) b) = v * ι (e (-(n : ℤ)) b) * u := by
        rw [h]
        calc ι (e (-(n : ℤ) - 1) b)
            = (v * u) * ι (e (-(n : ℤ) - 1) b) * (v * u) := by
              rw [hvu, one_mul, mul_one]
          _ = v * (u * ι (e (-(n : ℤ) - 1) b) * v) * u := by
              simp only [mul_assoc]
      rw [hback]
      exact S.mul_mem (S.mul_mem hv ih) hu
  have hι : ∀ a, ι a ∈ S := by
    have hle : Subring.closure (Set.range fun p : ℤ × Bool => e p.1 p.2) ≤ S.comap ι :=
      Subring.closure_le.mpr (by
        rintro _ ⟨⟨i, b⟩, rfl⟩
        exact he i b)
    rw [hA] at hle
    exact fun a => hle (Subring.mem_top a)
  have hle : Subring.closure (Set.range ι ∪ {u, v}) ≤ S :=
    Subring.closure_le.mpr (by
      rintro _ (⟨a, rfl⟩ | h)
      · exact hι a
      · rcases h with rfl | rfl
        · exact hu
        · exact hv)
  rw [hR] at hle
  exact top_le_iff.mp hle

/-- Every integer power of a unit lies in any subring containing the unit and
its inverse. -/
theorem val_zpow_mem_closure (u : Rˣ) (s : Set R) (k : ℤ) :
    ((u ^ k : Rˣ) : R) ∈ Subring.closure (s ∪ {(u : R), ↑u⁻¹}) := by
  have hu : (u : R) ∈ Subring.closure (s ∪ {(u : R), ↑u⁻¹}) :=
    Subring.subset_closure (Set.mem_union_right _ (by simp))
  have hv : ((u⁻¹ : Rˣ) : R) ∈ Subring.closure (s ∪ {(u : R), ↑u⁻¹}) :=
    Subring.subset_closure (Set.mem_union_right _ (by simp))
  induction k using Int.induction_on with
  | zero => rw [zpow_zero, Units.val_one]; exact Subring.one_mem _
  | succ n ih => rw [zpow_add_one, Units.val_mul]; exact Subring.mul_mem _ ih hu
  | pred n ih => rw [zpow_sub_one, Units.val_mul]; exact Subring.mul_mem _ ih hv

/-- **Normal form gives the generation hypothesis.**  If every element of `R`
is a finite sum of terms `ι a * u ^ k`, then `R` is generated by `ι A`, `u`
and `u⁻¹`. -/
theorem closure_range_union_eq_top_of_forall_mem_span (ι : A →+* R) (u : Rˣ)
    (hspan : ∀ r : R, r ∈ AddSubmonoid.closure
      (Set.range fun p : A × ℤ => ι p.1 * ((u ^ p.2 : Rˣ) : R))) :
    Subring.closure (Set.range ι ∪ {(u : R), ↑u⁻¹}) = ⊤ := by
  refine (Subring.eq_top_iff' _).mpr fun r => ?_
  have hle : AddSubmonoid.closure
      (Set.range fun p : A × ℤ => ι p.1 * ((u ^ p.2 : Rˣ) : R)) ≤
      (Subring.closure (Set.range ι ∪ {(u : R), ↑u⁻¹})).toSubsemiring.toAddSubmonoid := by
    refine AddSubmonoid.closure_le.mpr ?_
    rintro _ ⟨⟨a, k⟩, rfl⟩
    exact Subring.mul_mem _ (Subring.subset_closure (Set.mem_union_left _ ⟨a, rfl⟩))
      (val_zpow_mem_closure u _ k)
  exact hle (hspan r)

end Shift

/-! ### The crossed-product ring is finitely generated -/

section CrossedProduct

variable (X : Set (ℤ → Bool)) [CompactSpace X] {R : Type*} [Ring R]

/-- **The crossed-product ring is finitely generated.**  Let `R` receive the
coefficients `LC(X, ZMod 2)` of a compact set `X` of Bool sequences through a
ring hom `ι`, and contain mutually inverse `u, v` whose conjugation moves the
coordinate indicators up by one.  If `ι` together with `u` and `v` generates
`R`, then `R` is a finitely generated ring, generated by `[x₀ = false]`,
`[x₀ = true]`, `u` and `v`. -/
theorem isFinitelyGeneratedRing_of_shift (ι : LocallyConstant X (ZMod 2) →+* R)
    (u v : R) (huv : u * v = 1) (hvu : v * u = 1)
    (hshift : ∀ i b, ι (coordinateIndicator X (i + 1) b) =
      u * ι (coordinateIndicator X i b) * v)
    (hR : Subring.closure (Set.range ι ∪ {u, v}) = ⊤) :
    IsFinitelyGeneratedRing R := by
  classical
  refine ⟨{ι (coordinateIndicator X 0 false), ι (coordinateIndicator X 0 true), u, v}, ?_⟩
  simp only [Finset.coe_insert, Finset.coe_singleton]
  exact subring_closure_eq_top_of_shift ι u v huv hvu (coordinateIndicator X)
    (closure_range_coordinateIndicator X) hshift hR

/-- The same with the implementing unit given as a unit of `R` and the
normal form given as finite sums `ι f * u ^ k`. -/
theorem isFinitelyGeneratedRing_of_normalForm (ι : LocallyConstant X (ZMod 2) →+* R)
    (u : Rˣ)
    (hshift : ∀ i b, ι (coordinateIndicator X (i + 1) b) =
      ↑u * ι (coordinateIndicator X i b) * ↑u⁻¹)
    (hspan : ∀ r : R, r ∈ AddSubmonoid.closure
      (Set.range fun p : LocallyConstant X (ZMod 2) × ℤ => ι p.1 * ((u ^ p.2 : Rˣ) : R))) :
    IsFinitelyGeneratedRing R :=
  isFinitelyGeneratedRing_of_shift X ι u ↑u⁻¹ u.mul_inv u.inv_mul hshift
    (closure_range_union_eq_top_of_forall_mem_span ι u hspan)

end CrossedProduct
