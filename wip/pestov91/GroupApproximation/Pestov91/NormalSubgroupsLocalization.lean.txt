/-!
# Normal subgroups of `EL_ι(R)`: localizing a noncentral element

Part 1 §4.2 of `research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`
turns a noncentral `g` in a normal subgroup `N ⊴ EL₃(R)` into a nontrivial `k ∈ N` that is `1`
plus a matrix over a small tower subring.  This module proves the ring-independent part of that
step, over any unital ring `R` and any index type with at least three indices, and names the
two hypotheses the witness ring has to supply.

* **§4.2(a).**  The coefficients whose roots commute with a fixed matrix form a non-unital subring
  (`rootCentralizer`; products use a third index).  So if `S` generates `R` as a non-unital ring,
  a noncentral `g` fails to commute with some `e_ij(s)`, `s ∈ S`
  (`exists_noncommuting_root_of_not_mem_center`).
* **§4.2(b).**  If `s` and every `g_pi s (g⁻¹)_jq` lie in a non-unital subring `B`, then
  `k = ⁅g, e_ij(s)⁆` and `k⁻¹` are `1` plus matrices over `B` (`commutator_elGen_sub_one_mem`),
  and `k ≠ 1` (`commutator_elGen_ne_one`).
* **The two named hypotheses.**  `RootLocalization ι R Good` is the data of (a),(b) for a
  concrete ring: a generating set whose elements sit in good subrings absorbing `g` and `g⁻¹`.
  `LocalizedRootDetection ι R Good` is cases (c),(d): a nontrivial `k ∈ N` over a good subring
  yields a nonzero root in `N`.  `Pestov91.NormalSubgroups` combines them.
-/

namespace GroupApproximation
namespace Pestov91

open scoped commutatorElement

/-! ### Matrices over a non-unital subring -/

section EntriesIn

variable {ι R : Type*} [Ring R]

theorem single_apply_mem [DecidableEq ι] {B : NonUnitalSubring R} {s : R} (hs : s ∈ B)
    (i j p q : ι) : Matrix.single i j s p q ∈ B := by
  rw [Matrix.single_apply]
  split_ifs
  · exact hs
  · exact zero_mem B

theorem mul_apply_mem [Fintype ι] {B : NonUnitalSubring R} {X Y : Matrix ι ι R}
    (hX : ∀ p q, X p q ∈ B) (hY : ∀ p q, Y p q ∈ B) (p q : ι) : (X * Y) p q ∈ B := by
  rw [Matrix.mul_apply]
  exact sum_mem fun r _ ↦ mul_mem (hX p r) (hY r q)

end EntriesIn

variable {ι R : Type*} [Fintype ι] [DecidableEq ι] [Ring R]

/-! ### §4.2(a): a noncommuting root from a generating set -/

/-- The coefficients `s` for which `M` commutes with every off-diagonal `E_{ij}(s)`.  With three
indices it is a non-unital subring, since `E_{ij}(st) = E_{ik}(s) E_{kj}(t)`. -/
def rootCentralizer (hcard : 3 ≤ Fintype.card ι) (M : Matrix ι ι R) : NonUnitalSubring R where
  carrier := {s | ∀ i j : ι, i ≠ j → M * Matrix.single i j s = Matrix.single i j s * M}
  zero_mem' := by
    intro i j _
    rw [Matrix.single_zero, mul_zero, zero_mul]
  add_mem' := by
    intro a b ha hb i j hij
    rw [Matrix.single_add, mul_add, add_mul, ha i j hij, hb i j hij]
  neg_mem' := by
    intro a ha i j hij
    rw [← Matrix.single_neg, mul_neg, neg_mul, ha i j hij]
  mul_mem' := by
    intro a b ha hb i j hij
    obtain ⟨k, hki, hkj⟩ := exists_third_index hcard i j
    rw [← Matrix.single_mul_single_same (c := a) i k j b, ← mul_assoc, ha i k hki.symm,
      mul_assoc, hb k j hkj, ← mul_assoc]

theorem mem_rootCentralizer {hcard : 3 ≤ Fintype.card ι} {M : Matrix ι ι R} {s : R} :
    s ∈ rootCentralizer hcard M ↔
      ∀ i j : ι, i ≠ j → M * Matrix.single i j s = Matrix.single i j s * M :=
  Iff.rfl

/-- An element of `EL_ι(R)` whose matrix commutes with every off-diagonal `E_{ij}(a)` is
central. -/
theorem mem_center_of_forall_commute_single (hcard : 3 ≤ Fintype.card ι)
    {g : ↥(elementaryGroup ι R)}
    (h : ∀ i j : ι, i ≠ j → ∀ a : R,
      ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) * Matrix.single i j a =
        Matrix.single i j a * ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R)) :
    g ∈ Subgroup.center ↥(elementaryGroup ι R) := by
  rw [Subgroup.mem_center_iff]
  intro z
  apply Subtype.ext
  apply Units.ext
  exact (commute_of_forall_single hcard _ h ((z : (Matrix ι ι R)ˣ) : Matrix ι ι R)).symm

/-- **§4.2(a).**  If `S` generates `R` as a non-unital ring, a noncentral element of `EL_ι(R)`
fails to commute with some root `e_{ij}(s)` with `s ∈ S`. -/
theorem exists_noncommuting_root_of_not_mem_center (hcard : 3 ≤ Fintype.card ι) {S : Set R}
    (hS : NonUnitalSubring.closure S = ⊤) {g : ↥(elementaryGroup ι R)}
    (hg : g ∉ Subgroup.center ↥(elementaryGroup ι R)) :
    ∃ (i j : ι) (_ : i ≠ j) (s : R), s ∈ S ∧
      ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) * Matrix.single i j s ≠
        Matrix.single i j s * ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) := by
  by_contra hcon
  push Not at hcon
  refine hg (mem_center_of_forall_commute_single hcard fun i j hij a ↦ ?_)
  have hle : NonUnitalSubring.closure S ≤
      rootCentralizer hcard ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) :=
    NonUnitalSubring.closure_le.mpr fun s hs ↦
      mem_rootCentralizer.mpr fun i' j' hij' ↦ hcon i' j' hij' s hs
  have ha : a ∈ NonUnitalSubring.closure S := by
    rw [hS]
    exact NonUnitalSubring.mem_top a
  exact mem_rootCentralizer.mp (hle ha) i j hij

/-- The commutator of `g` with a root that `g` does not commute with is nontrivial. -/
theorem commutator_elGen_ne_one {g : ↥(elementaryGroup ι R)} {i j : ι} (hij : i ≠ j) {s : R}
    (hs : ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) * Matrix.single i j s ≠
      Matrix.single i j s * ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R)) :
    ⁅g, elGen i j hij s⁆ ≠ 1 := by
  intro h1
  apply hs
  have hm : ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) * (1 + Matrix.single i j s) =
      (1 + Matrix.single i j s) * ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) :=
    congrArg (fun z : ↥(elementaryGroup ι R) ↦ ((z : (Matrix ι ι R)ˣ) : Matrix ι ι R))
      (commutatorElement_eq_one_iff_mul_comm.mp h1)
  rw [mul_add, add_mul, mul_one, one_mul] at hm
  exact add_left_cancel hm

/-! ### §4.2(b): localization of the commutator -/

/-- **§4.2(b), matrix form.**  If `G Gi = 1`, `s ∈ B` and every `G_pi s Gi_jq ∈ B`, then
`G e_ij(s) Gi e_ij(-s) - 1` and `e_ij(s) G e_ij(-s) Gi - 1` are matrices over `B`. -/
theorem commutator_sub_one_entries_mem (B : NonUnitalSubring R) {G Gi : Matrix ι ι R}
    (hGGi : G * Gi = 1) {i j : ι} {s : R} (hs : s ∈ B)
    (habs : ∀ p q : ι, G p i * s * Gi j q ∈ B) :
    (∀ p q : ι,
      (G * (1 + Matrix.single i j s) * Gi * (1 - Matrix.single i j s) - 1) p q ∈ B) ∧
    (∀ p q : ι,
      ((1 + Matrix.single i j s) * G * (1 - Matrix.single i j s) * Gi - 1) p q ∈ B) := by
  have hX : ∀ p q : ι, (G * Matrix.single i j s * Gi) p q ∈ B := by
    intro p q
    rw [Matrix.mul_apply]
    refine sum_mem fun r _ ↦ ?_
    rw [matrix_mul_single_apply]
    by_cases hr : r = j
    · rw [if_pos hr, hr]
      exact habs p q
    · rw [if_neg hr, zero_mul]
      exact zero_mem B
  have hS := single_apply_mem hs i j
  have e1 : G * (1 + Matrix.single i j s) * Gi * (1 - Matrix.single i j s) - 1 =
      G * Matrix.single i j s * Gi - Matrix.single i j s -
        G * Matrix.single i j s * Gi * Matrix.single i j s := by
    have e : G * (1 + Matrix.single i j s) * Gi * (1 - Matrix.single i j s) =
        G * Gi - G * Gi * Matrix.single i j s + G * Matrix.single i j s * Gi -
          G * Matrix.single i j s * Gi * Matrix.single i j s := by noncomm_ring
    rw [e, hGGi]
    noncomm_ring
  have e2 : (1 + Matrix.single i j s) * G * (1 - Matrix.single i j s) * Gi - 1 =
      Matrix.single i j s - G * Matrix.single i j s * Gi -
        Matrix.single i j s * (G * Matrix.single i j s * Gi) := by
    have e : (1 + Matrix.single i j s) * G * (1 - Matrix.single i j s) * Gi =
        G * Gi + Matrix.single i j s * (G * Gi) - G * Matrix.single i j s * Gi -
          Matrix.single i j s * (G * Matrix.single i j s * Gi) := by noncomm_ring
    rw [e, hGGi]
    noncomm_ring
  refine ⟨fun p q ↦ ?_, fun p q ↦ ?_⟩
  · rw [e1, Matrix.sub_apply, Matrix.sub_apply]
    exact sub_mem (sub_mem (hX p q) (hS p q)) (mul_apply_mem hX hS p q)
  · rw [e2, Matrix.sub_apply, Matrix.sub_apply]
    exact sub_mem (sub_mem (hS p q) (hX p q)) (mul_apply_mem hS hX p q)

theorem val_commutator_elGen (g : ↥(elementaryGroup ι R)) {i j : ι} (hij : i ≠ j) (s : R) :
    (((⁅g, elGen i j hij s⁆ : ↥(elementaryGroup ι R)) : (Matrix ι ι R)ˣ) : Matrix ι ι R) =
      ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) * (1 + Matrix.single i j s) *
        (((g⁻¹ : ↥(elementaryGroup ι R)) : (Matrix ι ι R)ˣ) : Matrix ι ι R) *
          (1 - Matrix.single i j s) :=
  rfl

theorem val_commutator_elGen_swap (g : ↥(elementaryGroup ι R)) {i j : ι} (hij : i ≠ j)
    (s : R) :
    (((⁅elGen i j hij s, g⁆ : ↥(elementaryGroup ι R)) : (Matrix ι ι R)ˣ) : Matrix ι ι R) =
      (1 + Matrix.single i j s) * ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) *
        (1 - Matrix.single i j s) *
          (((g⁻¹ : ↥(elementaryGroup ι R)) : (Matrix ι ι R)ˣ) : Matrix ι ι R) :=
  rfl

/-- **§4.2(b).**  If `s` and every `g_pi s (g⁻¹)_jq` lie in a non-unital subring `B`, then
`k = ⁅g, e_ij(s)⁆` and `k⁻¹` are `1` plus matrices over `B`. -/
theorem commutator_elGen_sub_one_mem (B : NonUnitalSubring R) (g : ↥(elementaryGroup ι R))
    {i j : ι} (hij : i ≠ j) {s : R} (hs : s ∈ B)
    (habs : ∀ p q : ι, ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) p i * s *
      (((g⁻¹ : ↥(elementaryGroup ι R)) : (Matrix ι ι R)ˣ) : Matrix ι ι R) j q ∈ B) :
    (∀ p q : ι, ((((⁅g, elGen i j hij s⁆ : ↥(elementaryGroup ι R)) : (Matrix ι ι R)ˣ) :
      Matrix ι ι R) - 1) p q ∈ B) ∧
    (∀ p q : ι, (((((⁅g, elGen i j hij s⁆)⁻¹ : ↥(elementaryGroup ι R)) : (Matrix ι ι R)ˣ) :
      Matrix ι ι R) - 1) p q ∈ B) := by
  obtain ⟨h1, h2⟩ :=
    commutator_sub_one_entries_mem B (ElementarySimplicity.elMat_mul_inv g) hs habs
  refine ⟨?_, ?_⟩
  · rw [val_commutator_elGen]
    exact h1
  · rw [commutatorElement_inv, val_commutator_elGen_swap]
    exact h2

/-! ### The two named hypotheses -/

variable (ι R) in
/-- **§4.2(a),(b) data.**  For every `g`, a generating set `S` of `R` (as a non-unital ring)
whose elements lie, for all `i j`, in a good non-unital subring `B` containing every
`g_pi s (g⁻¹)_jq`.  For the witness ring, `B` is a tower subring `B_m(U)`. -/
def RootLocalization (Good : NonUnitalSubring R → Prop) : Prop :=
  ∀ g : ↥(elementaryGroup ι R), ∃ S : Set R, NonUnitalSubring.closure S = ⊤ ∧
    ∀ s ∈ S, ∀ i j : ι, ∃ B : NonUnitalSubring R, Good B ∧ s ∈ B ∧
      ∀ p q : ι, ((g : (Matrix ι ι R)ˣ) : Matrix ι ι R) p i * s *
        (((g⁻¹ : ↥(elementaryGroup ι R)) : (Matrix ι ι R)ˣ) : Matrix ι ι R) j q ∈ B

variable (ι R) in
/-- **§4.2(c),(d) hypothesis.**  A nontrivial `k` in a normal subgroup `N` such that `k - 1` and
`k⁻¹ - 1` are matrices over a good subring gives a nonzero root in `N`. -/
def LocalizedRootDetection (Good : NonUnitalSubring R → Prop) : Prop :=
  ∀ N : Subgroup ↥(elementaryGroup ι R), N.Normal → ∀ B : NonUnitalSubring R, Good B →
    ∀ k ∈ N, k ≠ 1 →
      (∀ p q : ι, (((k : (Matrix ι ι R)ˣ) : Matrix ι ι R) - 1) p q ∈ B) →
      (∀ p q : ι,
        ((((k⁻¹ : ↥(elementaryGroup ι R)) : (Matrix ι ι R)ˣ) : Matrix ι ι R) - 1) p q ∈ B) →
      ∃ (i j : ι) (hij : i ≠ j) (a : R), a ≠ 0 ∧ elGen i j hij a ∈ N

/-! ### Model tests

The trivial predicate `(· = ⊤)` satisfies `RootLocalization` over every ring, and then
`LocalizedRootDetection` is plain root detection for noncentral elements.  The empty predicate
makes `LocalizedRootDetection` vacuous but `RootLocalization` false, so the pair cannot be
satisfied by degenerating `Good`. -/

theorem rootLocalization_top : RootLocalization ι R (· = ⊤) := fun _ ↦
  ⟨Set.univ, top_unique fun x _ ↦ NonUnitalSubring.subset_closure (Set.mem_univ x),
    fun s _ _ _ ↦ ⟨⊤, rfl, NonUnitalSubring.mem_top s, fun _ _ ↦ NonUnitalSubring.mem_top _⟩⟩

theorem localizedRootDetection_bot : LocalizedRootDetection ι R (fun _ ↦ False) :=
  fun _ _ _ hB ↦ hB.elim

theorem not_rootLocalization_bot [Nontrivial R] [Nonempty ι] :
    ¬ RootLocalization ι R (fun _ ↦ False) := by
  intro h
  obtain ⟨S, hS, hgood⟩ := h 1
  obtain ⟨i⟩ := ‹Nonempty ι›
  obtain ⟨s, hs⟩ : S.Nonempty := by
    by_contra hempty
    rw [Set.not_nonempty_iff_eq_empty] at hempty
    rw [hempty, NonUnitalSubring.closure_empty] at hS
    have h1 : (1 : R) ∈ (⊥ : NonUnitalSubring R) := by
      rw [hS]
      exact NonUnitalSubring.mem_top 1
    exact one_ne_zero (NonUnitalSubring.mem_bot.mp h1)
  obtain ⟨B, hB, -⟩ := hgood s hs i i
  exact hB

end Pestov91
