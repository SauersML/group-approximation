import GroupApproximation.Pestov91.Subshift
import Mathlib.NumberTheory.Padics.PadicVal.Basic
import Mathlib.Data.Nat.Bits

/-!
# An explicit infinite minimal subshift: the Toeplitz sequence

`toeplitz n` is the parity of the 2-adic valuation of `n`, with `toeplitz 0 = false`: the
two-sided period-doubling Toeplitz sequence. The subshift is `toeplitzSubshift`, the orbit
closure of `toeplitz`; its points form `ToeplitzSpace`.

* `toeplitz_add_of_dvd`: adding a multiple of `2^e` does not change `toeplitz a` when `2^e ∤ a`.
* `exists_ne_of_mem`: for a point `y` and `j ≠ 0` there is `m` with `|m| ≤ 2|j|` and
  `y m ≠ y (m + j)`. So the shift acts freely (`vadd_ne_self`), and long cylinders have long
  return times (`eq_zero_of_agree_shift`).
* `toeplitz_mem_orbitClosure`: `toeplitz` lies in the orbit closure of every point, so the
  subshift is minimal. It is infinite.
-/

namespace GroupApproximation.Pestov91

open SymbolicDynamics.FullShift Set

/-- The two-sided Toeplitz sequence: the parity of the 2-adic valuation, `false` at `0`. -/
def toeplitz (n : ℤ) : Bool := Nat.bodd (padicValInt 2 n)

theorem padicValInt_two_eq_of_forall_dvd_iff {a b : ℤ} (ha : a ≠ 0) (hb : b ≠ 0)
    (h : ∀ e : ℕ, (2 : ℤ) ^ e ∣ a ↔ (2 : ℤ) ^ e ∣ b) : padicValInt 2 a = padicValInt 2 b := by
  have key : ∀ e : ℕ, e ≤ padicValInt 2 a ↔ e ≤ padicValInt 2 b := fun e => by
    have h2a := padicValInt_dvd_iff (p := 2) e a
    have h2b := padicValInt_dvd_iff (p := 2) e b
    simp only [Nat.cast_ofNat, ha, hb, false_or] at h2a h2b
    rw [← h2a, ← h2b]
    exact h e
  exact le_antisymm ((key _).1 le_rfl) ((key _).2 le_rfl)

theorem padicValInt_add_of_dvd {e : ℕ} {a d : ℤ} (ha : ¬ (2 : ℤ) ^ e ∣ a)
    (hd : (2 : ℤ) ^ e ∣ d) : padicValInt 2 (a + d) = padicValInt 2 a := by
  have had : ¬ (2 : ℤ) ^ e ∣ a + d := fun h => ha ((dvd_add_left hd).1 h)
  refine padicValInt_two_eq_of_forall_dvd_iff (fun h => had (by rw [h]; exact dvd_zero _))
    (fun h => ha (by rw [h]; exact dvd_zero _)) fun f => ?_
  rcases le_or_gt f e with hfe | hef
  · exact dvd_add_left ((pow_dvd_pow 2 hfe).trans hd)
  · exact ⟨fun h => absurd ((pow_dvd_pow 2 hef.le).trans h) had,
      fun h => absurd ((pow_dvd_pow 2 hef.le).trans h) ha⟩

theorem padicValInt_two_pow_mul {e : ℕ} {s : ℤ} (hs : ¬ (2 : ℤ) ∣ s) :
    padicValInt 2 (2 ^ e * s) = e := by
  have hs0 : s ≠ 0 := fun h => hs (by rw [h]; exact dvd_zero _)
  have hpow : ((2 : ℤ) ^ e) = ((2 ^ e : ℕ) : ℤ) := by norm_cast
  rw [padicValInt.mul (p := 2) (pow_ne_zero _ two_ne_zero) hs0,
    padicValInt.eq_zero_of_not_dvd (p := 2) (z := s) (by exact_mod_cast hs), add_zero, hpow,
    padicValInt.of_nat, padicValNat.prime_pow (p := 2)]

theorem exists_two_pow_mul_odd {j : ℤ} (hj : j ≠ 0) :
    ∃ t : ℤ, ¬ (2 : ℤ) ∣ t ∧ j = 2 ^ padicValInt 2 j * t := by
  obtain ⟨t, ht⟩ := padicValInt_dvd (p := 2) j
  have ht' : j = 2 ^ padicValInt 2 j * t := by exact_mod_cast ht
  refine ⟨t, fun h2 => ?_, ht'⟩
  obtain ⟨u, rfl⟩ := h2
  have hdvd : (2 : ℤ) ^ (padicValInt 2 j + 1) ∣ j := ⟨u, ht'.trans (by ring)⟩
  have h := (padicValInt_dvd_iff (p := 2) (padicValInt 2 j + 1) j).1 (by exact_mod_cast hdvd)
  exact absurd (h.resolve_left hj) (by omega)

theorem toeplitz_add_of_dvd {e : ℕ} {a d : ℤ} (ha : ¬ (2 : ℤ) ^ e ∣ a) (hd : (2 : ℤ) ^ e ∣ d) :
    toeplitz (a + d) = toeplitz a := by
  simp only [toeplitz, padicValInt_add_of_dvd ha hd]

theorem toeplitz_two_pow_mul {e : ℕ} {s : ℤ} (hs : ¬ (2 : ℤ) ∣ s) :
    toeplitz (2 ^ e * s) = Nat.bodd e := by
  simp only [toeplitz, padicValInt_two_pow_mul hs]

theorem toeplitz_zero : toeplitz 0 = false := by
  simp only [toeplitz, padicValInt.zero, Nat.bodd_zero]

/-- Every integer is within `2^a` of an odd multiple of `2^a`. -/
theorem exists_near_two_pow_mul_odd (a : ℕ) (c : ℤ) :
    ∃ m s : ℤ, |m| ≤ 2 ^ a ∧ ¬ (2 : ℤ) ∣ s ∧ c + m = 2 ^ a * s := by
  have hpos : (0 : ℤ) < 2 ^ (a + 1) := by positivity
  have h0 := Int.emod_nonneg c hpos.ne'
  have h1 := Int.emod_lt_of_pos c hpos
  have h2 : (2 : ℤ) ^ (a + 1) = 2 * 2 ^ a := by ring
  refine ⟨2 ^ a - c % 2 ^ (a + 1), 2 * (c / 2 ^ (a + 1)) + 1,
    abs_le.2 ⟨by linarith, by linarith⟩, ?_, ?_⟩
  · generalize c / 2 ^ (a + 1) = q
    omega
  · linear_combination (-1 : ℤ) * Int.emod_add_mul_ediv c (2 ^ (a + 1))

/-- The Toeplitz sequence is nowhere locally periodic at scale `2|j|`. -/
theorem toeplitz_exists_ne (c : ℤ) {j : ℤ} (hj : j ≠ 0) :
    ∃ m : ℤ, |m| ≤ 2 * |j| ∧ toeplitz (c + m) ≠ toeplitz (c + m + j) := by
  obtain ⟨t, ht, hjt⟩ := exists_two_pow_mul_odd hj
  generalize padicValInt 2 j = a at hjt
  obtain ⟨m, s, hm, hs, hcm⟩ := exists_near_two_pow_mul_odd (a + 1) c
  have hle : (2 : ℤ) ^ a ≤ |j| := Int.le_of_dvd (abs_pos.2 hj) ((dvd_abs _ _).2 ⟨t, hjt⟩)
  have hpow : (2 : ℤ) ^ (a + 1) = 2 * 2 ^ a := by ring
  refine ⟨m, by linarith, ?_⟩
  have hodd : ¬ (2 : ℤ) ∣ 2 * s + t := fun h => ht ((dvd_add_right (dvd_mul_right 2 s)).1 h)
  have h1 : c + m + j = 2 ^ a * (2 * s + t) := by rw [hcm, hjt]; ring
  rw [h1, hcm, toeplitz_two_pow_mul hs, toeplitz_two_pow_mul hodd,
    show a + 1 = Nat.succ a from rfl, Nat.bodd_succ]
  cases Nat.bodd a <;> decide

/-- The Toeplitz subshift: the orbit closure of `toeplitz`. -/
abbrev toeplitzSubshift : Subshift Bool ℤ := orbitClosure toeplitz

/-- The points of the Toeplitz subshift, with the shift action of `ℤ`. -/
abbrev ToeplitzSpace : Type := ↥toeplitzSubshift.carrier

theorem exists_ne_of_mem {y : ℤ → Bool} (hy : y ∈ toeplitzSubshift.carrier) {j : ℤ}
    (hj : j ≠ 0) : ∃ m : ℤ, |m| ≤ 2 * |j| ∧ y m ≠ y (m + j) := by
  obtain ⟨c, hc⟩ := mem_orbitClosure_iff.1 hy (3 * j.natAbs)
  obtain ⟨m, hm, hne⟩ := toeplitz_exists_ne c hj
  have hj' : |j| = (j.natAbs : ℤ) := Int.abs_eq_natAbs j
  have hm0 := abs_nonneg j
  have hmj := abs_add_le m j
  refine ⟨m, hm, fun h => hne ?_⟩
  have h1 := hc m (by push_cast; linarith)
  have h2 := hc (m + j) (by push_cast; linarith)
  simp only [shift_apply] at h1 h2
  rw [← h1, h, h2, add_assoc]

/-- The shift acts freely on the Toeplitz subshift. -/
theorem vadd_ne_self (y : ToeplitzSpace) {j : ℤ} (hj : j ≠ 0) : j +ᵥ y ≠ y := by
  intro h
  obtain ⟨m, -, hne⟩ := exists_ne_of_mem y.2 hj
  have hm := congrArg (fun z : ToeplitzSpace => (z : ℤ → Bool) m) h
  simp only [coe_vadd, shift_apply] at hm
  exact hne (by rw [← hm, add_comm])

/-- Long cylinders have long return times: if `y` and `shift j y` agree with the same
configuration on `[-k, k]` and `3|j| ≤ k`, then `j = 0`. -/
theorem eq_zero_of_agree_shift {y z : ℤ → Bool} (hy : y ∈ toeplitzSubshift.carrier) {j : ℤ}
    {k : ℕ} (hk : 3 * |j| ≤ k) (h1 : Agree k z y) (h2 : Agree k z (shift j y)) : j = 0 := by
  by_contra hj
  obtain ⟨m, hm, hne⟩ := exists_ne_of_mem hy hj
  have hj0 := abs_nonneg j
  have e1 := h1 m (by linarith)
  have e2 := h2 m (by linarith)
  simp only [shift_apply] at e2
  exact hne (by rw [← e1, e2, add_comm])

/-- `toeplitz` lies in the orbit closure of each point of its subshift. -/
theorem toeplitz_mem_orbitClosure {y : ℤ → Bool} (hy : y ∈ toeplitzSubshift.carrier) :
    toeplitz ∈ (orbitClosure y).carrier := by
  refine mem_orbitClosure_iff.2 fun k => ?_
  obtain ⟨c, hc⟩ := mem_orbitClosure_iff.1 hy (k + 2 ^ (2 * k))
  obtain ⟨m₀, s, hm₀, hs, hcm⟩ := exists_near_two_pow_mul_odd (2 * k) c
  have hk2 : (k : ℤ) < 2 ^ (2 * k) := by
    exact_mod_cast lt_of_le_of_lt (by omega : k ≤ 2 * k) (Nat.lt_pow_self Nat.one_lt_two)
  refine ⟨m₀, fun m hm => ?_⟩
  have hmm := abs_add_le m₀ m
  have e := hc (m₀ + m) (by push_cast; linarith)
  simp only [shift_apply] at e ⊢
  rw [e, ← add_assoc, hcm, add_comm]
  rcases eq_or_ne m 0 with rfl | hm0
  · rw [zero_add, toeplitz_two_pow_mul hs, toeplitz_zero, Nat.bodd_mul]
    rfl
  · have hndvd : ¬ (2 : ℤ) ^ (2 * k) ∣ m :=
      fun hd => hm0 (Int.eq_zero_of_abs_lt_dvd hd (lt_of_le_of_lt hm hk2))
    exact (toeplitz_add_of_dvd hndvd (dvd_mul_right _ s)).symm

instance : AddAction.IsMinimal ℤ ToeplitzSpace :=
  isMinimal_orbitClosure fun _ hy => toeplitz_mem_orbitClosure hy

instance : Nonempty ToeplitzSpace := ⟨⟨toeplitz, self_mem_orbitClosure toeplitz⟩⟩

instance : Infinite ToeplitzSpace :=
  infinite_of_free fun y _ hn => vadd_ne_self y hn

end GroupApproximation.Pestov91
