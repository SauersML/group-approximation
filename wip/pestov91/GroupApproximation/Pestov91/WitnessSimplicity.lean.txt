import GroupApproximation.Pestov91.RingSimple
import GroupApproximation.Pestov91.SplitSimplicity

/-!
# Simplicity of `EL_ι(LC(X, K) ⋊ ℤ)` for a free minimal system

The Pestov 9.1 witness is `EL₃(R)` for `R = LC(X, 𝔽₂) ⋊ ℤ` over the Toeplitz subshift.  Its
simplicity is `SplitSimplicity.isSimpleGroup_elementaryGroup_crossedProduct`, whose ring inputs are
`IsSimpleRing R`, `Z(R) = {0, 1}`, the covariance `hconj`, the normal form `hspan` and the
localisation `hkey`.  This module discharges `IsSimpleRing R` and `hkey` from the dynamics, in the
hypothesis shapes of `RingSimple.isSimpleRing_of_minimal_free`:

* `IsSimpleRing R` is `isSimpleRing_of_minimal_free`;
* `hkey` is `locallyConstant_cut_key`, with its covering property from
  `exists_finset_cover_of_dense_orbits`.

So `isSimpleGroup_elementaryGroup_of_minimal_free` asks only for what the crossed-product, subshift
and centre lanes supply.  It is conditional on those hypotheses, all stated as arguments.
-/

namespace GroupApproximation
namespace Pestov91

/-- **`EL_ι(LC(X, K) ⋊ ℤ)` is simple for a free minimal system.**  With at least three indices, a
nontrivial ring `R` whose central elements are `0` or `1`, spanned by the `ι (c j) * u ^ j` with
`u ^ j` implementing `σ j f = f ∘ τ j`, over a compact totally separated `X` on which the `τ j`
(`j ≠ 0`) have no fixed point and every orbit is dense, has a simple elementary group. -/
theorem isSimpleGroup_elementaryGroup_of_minimal_free {ι R X K : Type*} [Fintype ι] [DecidableEq ι]
    [Ring R] [Nontrivial R] [TopologicalSpace X] [CompactSpace X] [TotallySeparatedSpace X]
    [Field K] (hcard : 3 ≤ Fintype.card ι)
    (hR : ∀ c : R, (∀ r : R, r * c = c * r) → c = 0 ∨ c = 1)
    (φ : LocallyConstant X K →+* R) (u : Rˣ) (τ : ℤ → X → X) (hτ : ∀ j, Continuous (τ j))
    (hfree : ∀ j : ℤ, j ≠ 0 → ∀ x, τ j x ≠ x) (hmin : ∀ x, Dense (Set.range fun j => τ j x))
    (σ : ℤ → LocallyConstant X K → LocallyConstant X K) (hσ : ∀ j f x, σ j f x = f (τ j x))
    (hconj : ∀ (j : ℤ) (a : LocallyConstant X K),
      ((u ^ j : Rˣ) : R) * φ a = φ (σ j a) * ((u ^ j : Rˣ) : R))
    (hspan : ∀ x : R, ∃ (S : Finset ℤ) (c : ℤ → LocallyConstant X K),
      x = ∑ j ∈ S, φ (c j) * ((u ^ j : Rˣ) : R)) :
    IsSimpleGroup (elementaryGroup ι R) :=
  haveI : IsSimpleRing R := isSimpleRing_of_minimal_free φ u τ hτ hfree hmin σ hσ hconj hspan
  isSimpleGroup_elementaryGroup_crossedProduct hcard hR φ u σ hconj hspan fun a ha F hF =>
    locallyConstant_cut_key τ hτ hfree
      (fun _ hW hne => exists_finset_cover_of_dense_orbits τ hτ hmin hW hne) σ hσ a ha F hF

end Pestov91
end GroupApproximation
