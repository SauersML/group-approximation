/-!
# Pestov Open Question 9.1 from its leaves

Pestov's Open Question 9.1 (*Hyperlinear and sofic groups: a brief guide*,
arXiv:0804.3968v8, p. 21) asks whether an infinite simple group with Kazhdan's
property `(T)` can be hyperlinear (sofic).  The witness of this development is
`EL₃(R)` for the crossed product `R = LC(X, 𝔽₂) ⋊ ℤ` of an infinite minimal
subshift `X`.  Its centre is trivial, so it is its own quotient by the centre.

This module states the answer with every ingredient not yet proved for the
witness as an explicit hypothesis.  All four theorems are conditional.  The
hypothesis-free endpoint is `GroupApproximation/Pestov91/Endpoint.lean`, and
these are the forms it applies (blueprint
`research/artifacts/pestov91-lean-blueprint-2026-09-13.md`, G8).

1. `exists_infinite_simple_kazhdan_sofic_hyperlinear_of_lef`: an infinite simple
   LEF group with property `(T)` is sofic (`isSofic_of_isLEF`) and hyperlinear
   (`isHyperlinear_of_isLEF`).

2. `exists_infinite_simple_kazhdan_lef_of_ringLeaves`, with its companion
   `exists_infinite_simple_kazhdan_sofic_hyperlinear_of_ringLeaves`: ring-level
   leaves for one ring `R` spanned by a covariant unit over a commutative
   coefficient ring `A`.  The leaves are:
   * `IsSimpleRing R`;
   * central elements of `R` are `0` or `1`;
   * `hconj`, `hspan` and the localisation `hkey`;
   * `Infinite R`;
   * `R` is finitely generated;
   * `EL₃(R)` is LEF.

   Simplicity of `EL₃(R)` is `isSimpleGroup_elementaryGroup_crossedProduct`,
   infiniteness is `elementaryModCentre_infinite`, and property `(T)` comes from
   `exists_infinite_simple_kazhdan_lef_of_elementary`.

3. `exists_infinite_simple_kazhdan_sofic_hyperlinear_of_dynamics` discharges
   `IsSimpleRing R` and `hkey` for `A = LC(X, K)`: `isSimpleRing_of_minimal_free`
   and `locallyConstant_cut_key`, from a compact totally separated `X` with a
   free minimal action `τ`.  What is left:
   * the dynamical system;
   * the presentation `φ`, `u`, `σ`, `hconj`, `hspan`;
   * `Infinite R` and the centre;
   * finite generation;
   * LEF.

The prose of this module was written by Claude (Anthropic).
-/

namespace GroupApproximation
namespace Pestov91

/-- **From an infinite simple Kazhdan LEF group to the answer.**  An LEF group
is sofic and hyperlinear. -/
theorem exists_infinite_simple_kazhdan_sofic_hyperlinear_of_lef
    (h : ∃ (E : Type) (_ : Group E), Infinite E ∧ IsSimpleGroup E ∧
      HasKazhdanPropertyT.{0, 0} E ∧ IsLEF E) :
    ∃ (E : Type) (_ : Group E), Infinite E ∧ IsSimpleGroup E ∧
      HasKazhdanPropertyT.{0, 0} E ∧ IsSofic E ∧ IsHyperlinear E := by
  obtain ⟨E, _, hinf, hsimple, hT, hlef⟩ := h
  exact ⟨E, inferInstance, hinf, hsimple, hT, isSofic_of_isLEF hlef, isHyperlinear_of_isLEF hlef⟩

/-- **The endpoint from ring-level leaves.**  Every binder is a ring fact that a
named lane proves for the witness ring (blueprint v2, G2–G7). -/
theorem exists_infinite_simple_kazhdan_lef_of_ringLeaves {R : Type} [Ring R] [IsSimpleRing R]
    [Infinite R] (hR : ∀ c : R, (∀ r : R, r * c = c * r) → c = 0 ∨ c = 1)
    {A : Type} [CommRing A] (φ : A →+* R) (u : Rˣ) (σ : ℤ → A → A)
    (hconj : ∀ (j : ℤ) (a : A), ((u ^ j : Rˣ) : R) * φ a = φ (σ j a) * ((u ^ j : Rˣ) : R))
    (hspan : ∀ x : R, ∃ (S : Finset ℤ) (c : ℤ → A), x = ∑ j ∈ S, φ (c j) * ((u ^ j : Rˣ) : R))
    (hkey : ∀ a : A, a ≠ 0 → ∀ F : Finset ℤ, (0 : ℤ) ∉ F → ∃ e b : A, b * e * a * e = e ∧
      (∀ j ∈ F, e * σ j e = 0) ∧ ∃ s : Finset ℤ, ∏ i ∈ s, (1 - σ i e) = 0)
    (hfg : IsFinitelyGeneratedRing R)
    (hlef : IsLEF ↥(elementaryGroup (Fin 3) R)) :
    ∃ (E : Type) (_ : Group E), Infinite E ∧ IsSimpleGroup E ∧
      HasKazhdanPropertyT.{0, 0} E ∧ IsLEF E :=
  exists_infinite_simple_kazhdan_lef_of_elementary hfg
    (isSimpleGroup_elementaryGroup_crossedProduct (by simp) hR φ u σ hconj hspan hkey)
    (elementaryModCentre_infinite (R := R) (0 : Fin 3) 1 (by decide)) hlef

/-- **The answer from ring-level leaves**, in the sofic and hyperlinear form. -/
theorem exists_infinite_simple_kazhdan_sofic_hyperlinear_of_ringLeaves {R : Type} [Ring R]
    [IsSimpleRing R] [Infinite R] (hR : ∀ c : R, (∀ r : R, r * c = c * r) → c = 0 ∨ c = 1)
    {A : Type} [CommRing A] (φ : A →+* R) (u : Rˣ) (σ : ℤ → A → A)
    (hconj : ∀ (j : ℤ) (a : A), ((u ^ j : Rˣ) : R) * φ a = φ (σ j a) * ((u ^ j : Rˣ) : R))
    (hspan : ∀ x : R, ∃ (S : Finset ℤ) (c : ℤ → A), x = ∑ j ∈ S, φ (c j) * ((u ^ j : Rˣ) : R))
    (hkey : ∀ a : A, a ≠ 0 → ∀ F : Finset ℤ, (0 : ℤ) ∉ F → ∃ e b : A, b * e * a * e = e ∧
      (∀ j ∈ F, e * σ j e = 0) ∧ ∃ s : Finset ℤ, ∏ i ∈ s, (1 - σ i e) = 0)
    (hfg : IsFinitelyGeneratedRing R)
    (hlef : IsLEF ↥(elementaryGroup (Fin 3) R)) :
    ∃ (E : Type) (_ : Group E), Infinite E ∧ IsSimpleGroup E ∧
      HasKazhdanPropertyT.{0, 0} E ∧ IsSofic E ∧ IsHyperlinear E :=
  exists_infinite_simple_kazhdan_sofic_hyperlinear_of_lef
    (exists_infinite_simple_kazhdan_lef_of_ringLeaves hR φ u σ hconj hspan hkey hfg hlef)

/-- **The answer from a free minimal system.**  Let `X` be compact and totally
separated with a free action `τ` of `ℤ` whose orbits are dense, and let the
infinite ring `R`, with centre `{0, 1}`, be spanned by `φ (LC(X, K))` and a unit
`u` implementing `σ j f = f ∘ τ j`.  If `R` is finitely generated and `EL₃(R)` is
LEF, then `EL₃(R)` is infinite, simple, Kazhdan, sofic and hyperlinear. -/
theorem exists_infinite_simple_kazhdan_sofic_hyperlinear_of_dynamics
    {R : Type} [Ring R] [Infinite R] {X K : Type} [TopologicalSpace X] [CompactSpace X]
    [TotallySeparatedSpace X] [Field K]
    (hR : ∀ c : R, (∀ r : R, r * c = c * r) → c = 0 ∨ c = 1)
    (φ : LocallyConstant X K →+* R) (u : Rˣ) (τ : ℤ → X → X) (hτ : ∀ j, Continuous (τ j))
    (hfree : ∀ j : ℤ, j ≠ 0 → ∀ x, τ j x ≠ x) (hmin : ∀ x, Dense (Set.range fun j => τ j x))
    (σ : ℤ → LocallyConstant X K → LocallyConstant X K) (hσ : ∀ j f x, σ j f x = f (τ j x))
    (hconj : ∀ (j : ℤ) (a : LocallyConstant X K),
      ((u ^ j : Rˣ) : R) * φ a = φ (σ j a) * ((u ^ j : Rˣ) : R))
    (hspan : ∀ x : R, ∃ (S : Finset ℤ) (c : ℤ → LocallyConstant X K),
      x = ∑ j ∈ S, φ (c j) * ((u ^ j : Rˣ) : R))
    (hfg : IsFinitelyGeneratedRing R)
    (hlef : IsLEF ↥(elementaryGroup (Fin 3) R)) :
    ∃ (E : Type) (_ : Group E), Infinite E ∧ IsSimpleGroup E ∧
      HasKazhdanPropertyT.{0, 0} E ∧ IsSofic E ∧ IsHyperlinear E := by
  haveI : IsSimpleRing R := isSimpleRing_of_minimal_free φ u τ hτ hfree hmin σ hσ hconj hspan
  exact exists_infinite_simple_kazhdan_sofic_hyperlinear_of_ringLeaves hR φ u σ hconj hspan
    (fun a ha F hF => locallyConstant_cut_key τ hτ hfree
      (fun _ hW hne => exists_finset_cover_of_dense_orbits τ hτ hmin hW hne) σ hσ a ha F hF)
    hfg hlef

end Pestov91
end GroupApproximation
