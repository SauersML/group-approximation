import GroupApproximation.Algebra.FinitaryLinearBinarySimple

/-!
# `ι_W` sends `GL_{ι×n}(𝔽₂)` into `EL_ι(R)`

Case (c) of §4.2 needs the copy `ι_W(SL_d(F_q))` to lie in `G = EL_3(R)`.  Over `𝔽₂`
every invertible matrix is elementary (`FinitaryLinear.elementaryGroup_zmodTwo_eq_top`),
so it suffices to place the image of each transvection `t = 1 + c e_((p,a),(q,b))`:

* if `p ≠ q`, `ι_W(t) = e_pq(c E a b)` (`towerEmbed_elementaryUnit`);
* if `p = q`, then `a ≠ b`.  Put `x = c E a b` and `y = E b b`, so `x y = x` and `y x = 0`.
  For `r ≠ p`, `[e_pr(x), e_rp(y)] = 1 + x E_pp = ι_W(t)`.  This is the commutator
  identity of §4.2(c), which the first review recomputed.

`towerEL` is the resulting homomorphism `GL_{ι×n}(𝔽₂) →* EL_ι(R)`.
-/

namespace GroupApproximation
namespace Pestov91

open Matrix
open scoped commutatorElement

variable {K R : Type*} [CommRing K] [Ring R] [Algebra K R]
variable {ι n : Type*} [Fintype ι] [DecidableEq ι] [Fintype n] [DecidableEq n]
variable {E : n → n → R}

theorem towerEmbed_elementaryUnit (p q : ι) (a b : n) (h : (p, a) ≠ (q, b)) (c : K) :
    towerEmbed E ((elementaryUnit (p, a) (q, b) h c : (Matrix (ι × n) (ι × n) K)ˣ) :
      Matrix (ι × n) (ι × n) K) = 1 + Matrix.single p q (algebraMap K R c * E a b) := by
  change towerEmbed E (1 + Matrix.single (p, a) (q, b) c) = _
  rw [towerEmbed_add_right, towerEmbed_one, blockEmbed_single]

/-- **The image of a transvection is elementary.**  The diagonal-block case is the
commutator `[e_pr(c E a b), e_rp(E b b)] = 1 + c E a b E_pp` of §4.2(c). -/
theorem towerUnits_elementaryUnit_mem (hE : IsMatrixUnits E) [Nontrivial ι] (x y : ι × n)
    (hxy : x ≠ y) (c : K) :
    towerUnits K ι hE (elementaryUnit x y hxy c) ∈ elementaryGroup ι R := by
  obtain ⟨p, a⟩ := x
  obtain ⟨q, b⟩ := y
  by_cases hpq : p = q
  · subst hpq
    have hab : a ≠ b := fun h ↦ hxy (by rw [h])
    obtain ⟨r, hr⟩ := exists_ne p
    have hXY : algebraMap K R c * E a b * E b b = algebraMap K R c * E a b := by
      rw [mul_assoc, hE a b b b, if_pos rfl]
    have hYX : E b b * (algebraMap K R c * E a b) = 0 := by
      rw [← mul_assoc, ← Algebra.commutes c (E b b), mul_assoc, hE b b a b,
        if_neg (Ne.symm hab), mul_zero]
    have hAA : Matrix.single p r (algebraMap K R c * E a b) *
        Matrix.single p r (algebraMap K R c * E a b) = 0 :=
      Matrix.single_mul_single_of_ne (c := algebraMap K R c * E a b) p r p hr _
    have hBB : Matrix.single r p (E b b) * Matrix.single r p (E b b) = 0 :=
      Matrix.single_mul_single_of_ne (c := E b b) r p r (Ne.symm hr) _
    have hBA : Matrix.single r p (E b b) * Matrix.single p r (algebraMap K R c * E a b) = 0 := by
      rw [Matrix.single_mul_single_same, hYX, Matrix.single_zero]
    have heq : towerUnits K ι hE (elementaryUnit (p, a) (p, b) hxy c) =
        ⁅elementaryUnit p r (Ne.symm hr) (algebraMap K R c * E a b),
          elementaryUnit r p hr (E b b)⁆ := by
      apply Units.ext
      rw [towerUnits_coe, towerEmbed_elementaryUnit]
      change 1 + Matrix.single p p (algebraMap K R c * E a b) =
        (1 + Matrix.single p r (algebraMap K R c * E a b)) * (1 + Matrix.single r p (E b b)) *
          (1 - Matrix.single p r (algebraMap K R c * E a b)) * (1 - Matrix.single r p (E b b))
      rw [unipotent_commutator_of_right_annihilates _ _ hAA hBB hBA,
        Matrix.single_mul_single_same, hXY]
    rw [heq, commutatorElement_def]
    exact Subgroup.mul_mem _ (Subgroup.mul_mem _ (Subgroup.mul_mem _ (elementaryUnit_mem _ _ _ _)
      (elementaryUnit_mem _ _ _ _)) (Subgroup.inv_mem _ (elementaryUnit_mem _ _ _ _)))
      (Subgroup.inv_mem _ (elementaryUnit_mem _ _ _ _))
  · have heq : towerUnits K ι hE (elementaryUnit (p, a) (q, b) hxy c) =
        elementaryUnit p q hpq (algebraMap K R c * E a b) := by
      apply Units.ext
      rw [towerUnits_coe, towerEmbed_elementaryUnit]
      rfl
    rw [heq]
    exact elementaryUnit_mem _ _ _ _

/-- **`ι_W(GL_{ι×n}(𝔽₂)) ≤ EL_ι(R)`**, since `GL = EL` over `𝔽₂`. -/
theorem towerUnits_mem [Algebra (ZMod 2) R] (hE : IsMatrixUnits E) [Nontrivial ι]
    (ε : (Matrix (ι × n) (ι × n) (ZMod 2))ˣ) :
    towerUnits (ZMod 2) ι hE ε ∈ elementaryGroup ι R := by
  have hε : ε ∈ elementaryGroup (ι × n) (ZMod 2) := by
    rw [FinitaryLinear.elementaryGroup_zmodTwo_eq_top]
    exact Subgroup.mem_top ε
  have hle : elementaryGroup (ι × n) (ZMod 2) ≤
      (elementaryGroup ι R).comap (towerUnits (ZMod 2) ι hE) := by
    refine (Subgroup.closure_le (K := (elementaryGroup ι R).comap (towerUnits (ZMod 2) ι hE))).mpr ?_
    rintro _ ⟨x, y, hxy, c, rfl⟩
    exact Subgroup.mem_comap.mpr (towerUnits_elementaryUnit_mem hE x y hxy c)
  exact hle hε

variable (ι) in
/-- `ι_W` as a homomorphism `GL_{ι×n}(𝔽₂) →* EL_ι(R)`. -/
def towerEL [Algebra (ZMod 2) R] [Nontrivial ι] (hE : IsMatrixUnits E) :
    (Matrix (ι × n) (ι × n) (ZMod 2))ˣ →* elementaryGroup ι R :=
  (towerUnits (ZMod 2) ι hE).codRestrict (elementaryGroup ι R) (towerUnits_mem hE)

theorem towerEL_coe [Algebra (ZMod 2) R] [Nontrivial ι] (hE : IsMatrixUnits E)
    (ε : (Matrix (ι × n) (ι × n) (ZMod 2))ˣ) :
    ((towerEL ι hE ε : elementaryGroup ι R) : (Matrix ι ι R)ˣ) = towerUnits (ZMod 2) ι hE ε :=
  rfl

end Pestov91
end GroupApproximation
