/-!
# Normal subgroups of `EL_ι(R)` are central or everything

Part 1 §4 of `research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`:
every normal subgroup of `G = EL₃(R)` is contained in `Z(G)` or equals `G`, and at `q = 2`,
where `Z(R) = 𝔽₂`, `G` is simple.

* **§4.1 over a simple ring.**  `normalLevel` (on main) is a two-sided ideal, and level `⊤`
  gives `N = ⊤` through `relativeElementary_normalLevel_le` and `relativeElementary_top`.  Over a
  simple ring one nonzero root in `N` therefore gives `N = ⊤`.
* **§4.3.**  `normal_le_center_or_eq_top`: a noncentral `g ∈ N` gives, by §4.2(a),(b)
  (`Pestov91.NormalSubgroupsLocalization`), a nontrivial `k ∈ N` over a good subring, and
  cases (c),(d) give a nonzero root in `N`.
* **Simplicity.**  With `Z(R) = {0, 1}` the centre of `G` is trivial (`Pestov91.Centre`), so
  `G` itself is simple (`isSimpleGroup_of_rootLocalization`).

Both theorems are conditional: `IsSimpleRing R`, `RootLocalization ι R Good`,
`LocalizedRootDetection ι R Good` and (for simplicity) `Z(R) = {0, 1}` are arguments.
-/

namespace GroupApproximation
namespace Pestov91

open scoped commutatorElement

variable {ι R : Type*} [Fintype ι] [DecidableEq ι] [Ring R]

/-! ### §4.1 over a simple ring -/

/-- A nonzero root in a normal subgroup makes its level ideal nonzero. -/
theorem normalLevel_ne_bot_of_elGen_mem (hcard : 3 ≤ Fintype.card ι)
    (N : Subgroup ↥(elementaryGroup ι R)) [N.Normal] {i j : ι} (hij : i ≠ j) {a : R}
    (ha : a ≠ 0) (hmem : elGen i j hij a ∈ N) : normalLevel hcard N ≠ ⊥ := by
  intro hbot
  have hlevel : a ∈ normalLevel hcard N := (mem_normalLevel_iff hcard N).mpr
    (Manuscript.OneSidedMFRadical.TorsionComplementaryIdempotents.elGen_mem_of_one_position
      (exists_third_index hcard) hij hmem)
  rw [hbot] at hlevel
  exact ha (TwoSidedIdeal.mem_bot.mp hlevel)

/-- **§4.1 over a simple ring.**  A normal subgroup of `EL_ι(R)` containing one nonzero root is
everything. -/
theorem normal_eq_top_of_elGen_mem_of_isSimpleRing [IsSimpleRing R]
    (hcard : 3 ≤ Fintype.card ι) (N : Subgroup ↥(elementaryGroup ι R)) [N.Normal]
    {i j : ι} (hij : i ≠ j) {a : R} (ha : a ≠ 0) (hmem : elGen i j hij a ∈ N) : N = ⊤ := by
  have htop : normalLevel hcard N = ⊤ :=
    (IsSimpleOrder.eq_bot_or_eq_top _).resolve_left
      (normalLevel_ne_bot_of_elGen_mem hcard N hij ha hmem)
  have hle := relativeElementary_normalLevel_le hcard N
  rw [htop, relativeElementary_top] at hle
  exact top_le_iff.mp hle

/-! ### §4.3: the dichotomy and simplicity -/

/-- **Every normal subgroup of `EL_ι(R)` is central or everything**, over a simple ring, given
the localization data of §4.2(a),(b) and the case analysis (c),(d) for the same `Good`. -/
theorem normal_le_center_or_eq_top [IsSimpleRing R] (hcard : 3 ≤ Fintype.card ι)
    {Good : NonUnitalSubring R → Prop} (hloc : RootLocalization ι R Good)
    (hcases : LocalizedRootDetection ι R Good)
    (N : Subgroup ↥(elementaryGroup ι R)) [hN : N.Normal] :
    N ≤ Subgroup.center ↥(elementaryGroup ι R) ∨ N = ⊤ := by
  by_cases hle : N ≤ Subgroup.center ↥(elementaryGroup ι R)
  · exact Or.inl hle
  refine Or.inr ?_
  obtain ⟨g, hgN, hgc⟩ := SetLike.not_le_iff_exists.mp hle
  obtain ⟨S, hS, hSloc⟩ := hloc g
  obtain ⟨i, j, hij, s, hsS, hcomm⟩ :=
    exists_noncommuting_root_of_not_mem_center hcard hS hgc
  obtain ⟨B, hB, hsB, habs⟩ := hSloc s hsS i j
  obtain ⟨hk, hkinv⟩ := commutator_elGen_sub_one_mem B g hij hsB habs
  obtain ⟨p, q, hpq, a, ha, hmem⟩ :=
    hcases N hN B hB ⁅g, elGen i j hij s⁆ (commutator_mem_left N hgN _)
      (commutator_elGen_ne_one hij hcomm) hk hkinv
  exact normal_eq_top_of_elGen_mem_of_isSimpleRing hcard N hpq ha hmem

/-- **`EL_ι(R)` is simple** over a simple ring with `Z(R) = {0, 1}`, given the localization data
of §4.2(a),(b) and the case analysis (c),(d) for the same `Good`. -/
theorem isSimpleGroup_of_rootLocalization [IsSimpleRing R] (hcard : 3 ≤ Fintype.card ι)
    (hcentre : ∀ c : R, (∀ r : R, r * c = c * r) → c = 0 ∨ c = 1)
    {Good : NonUnitalSubring R → Prop} (hloc : RootLocalization ι R Good)
    (hcases : LocalizedRootDetection ι R Good) :
    IsSimpleGroup ↥(elementaryGroup ι R) := by
  haveI : Nontrivial ι := Fintype.one_lt_card_iff_nontrivial.mp (by omega)
  obtain ⟨i, j, hij⟩ := exists_pair_ne ι
  have hcenter : Subgroup.center ↥(elementaryGroup ι R) = ⊥ :=
    center_elementaryGroup_eq_bot_of_central_zero_or_one hcentre
  refine
    { exists_pair_ne :=
        ⟨elGen i j hij 1, 1, fun h ↦ one_ne_zero ((elGen_eq_one_iff i j hij 1).mp h)⟩
      eq_bot_or_eq_top_of_normal := fun N hN ↦ ?_ }
  haveI := hN
  rcases normal_le_center_or_eq_top hcard hloc hcases N with hle | htop
  · left
    rw [hcenter] at hle
    exact le_bot_iff.mp hle
  · exact Or.inr htop
