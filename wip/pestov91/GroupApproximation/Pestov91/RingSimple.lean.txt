/-!
# Simplicity of the crossed product of a free minimal action on a totally separated space

Let `X` be a compact totally separated space and `τ n : X → X` (`n : ℤ`) continuous maps
with no fixed points for `n ≠ 0` that move every point into every nonempty open set. Let
`R` be a ring with a ring hom `ι : LC(X, K) →+* R` and a unit `u` satisfying
`u^n ι(f) = ι(f ∘ τ n) u^n`, such that every element of `R` is a finite sum `∑ ι(f_j) u^j`.
Then every two-sided ideal of `R` with a nonzero element contains `1`
(`one_mem_of_ne_zero_mem`), so a nontrivial such `R` is a simple ring
(`isSimpleRing_of_covariant`).

The proof is the direct one of the construction artifact, §1.3, over an arbitrary field `K`:
* a nonzero `r = ∑ ι(f_j) u^j` has a coefficient with `f_{j₀}(x) = c ≠ 0`;
* a clopen `W ∋ x` on which `f_{j₀} = c` and which is disjoint from its images under
  `τ (j - j₀)`, `j ≠ j₀`, gives `ι(χ_W) r u^{-j₀} ι(χ_W) = c ι(χ_W)`
  (`exists_charFn_mem`);
* the conjugates `u^n ι(χ_W) u^{-n} = ι(χ_W ∘ τ n)` lie in the ideal, finitely many of the
  sets `τ n ⁻¹' W` cover `X`, and `g ↦ h + g - h g` assembles `1`
  (`one_mem_of_charFn_mem`).

The dynamics enter only through the hypotheses `hfree` and `hmin`, stated as theorem
arguments. `isSimpleRing_of_free_minimal` restates the result for a free minimal continuous
`ℤ`-action with `τ n = ((-n) +ᵥ ·)`.
-/

namespace GroupApproximation.Pestov91

open LocallyConstant

variable {X : Type*} [TopologicalSpace X] {K : Type*} [Field K]

/-- A clopen neighbourhood `W` of `x` inside a clopen set `C` that is disjoint from its images
under `τ n` for the finitely many `n ∈ D`, provided no `τ n` with `n ∈ D` fixes `x`. -/
theorem exists_isClopen_disjoint_images [TotallySeparatedSpace X] (τ : ℤ → C(X, X)) {x : X}
    {C : Set X} (hC : IsClopen C) (hxC : x ∈ C) (D : Finset ℤ) (hD : ∀ n ∈ D, τ n x ≠ x) :
    ∃ W : Set X, IsClopen W ∧ x ∈ W ∧ W ⊆ C ∧ ∀ n ∈ D, ∀ y ∈ W, τ n y ∉ W := by
  induction D using Finset.induction_on with
  | empty => exact ⟨C, hC, hxC, subset_rfl, fun _ hn => by simp at hn⟩
  | insert a D _ ih =>
    obtain ⟨W, hW, hxW, hWC, hWsep⟩ := ih fun n hn => hD n (Finset.mem_insert_of_mem hn)
    obtain ⟨V, hV, hxV, haxV⟩ :=
      exists_isClopen_of_totally_separated (hD a (Finset.mem_insert_self a D)).symm
    refine ⟨W ∩ V ∩ τ a ⁻¹' Vᶜ, (hW.inter hV).inter (hV.compl.preimage (τ a).continuous),
      ⟨⟨hxW, hxV⟩, haxV⟩, fun y hy => hWC hy.1.1, fun n hn y hy hny => ?_⟩
    rcases Finset.mem_insert.mp hn with rfl | hn
    · exact hy.2 hny.1.2
    · exact hWsep n hn y hy.1.1 hny.1.1

section Covariant

variable {R : Type*} [Ring R] (τ : ℤ → C(X, X)) (ι : LocallyConstant X K →+* R) (u : Rˣ)

/-- Conjugating `ι f` by `u^n` composes `f` with `τ n`. -/
theorem conj_eq_comap
    (hcov : ∀ (n : ℤ) (f : LocallyConstant X K),
      ((u ^ n : Rˣ) : R) * ι f = ι (comap (τ n) f) * ((u ^ n : Rˣ) : R))
    (n : ℤ) (f : LocallyConstant X K) :
    ((u ^ n : Rˣ) : R) * ι f * ((u ^ (-n) : Rˣ) : R) = ι (comap (τ n) f) := by
  rw [hcov, mul_assoc, ← Units.val_mul, ← zpow_add, add_neg_cancel, zpow_zero, Units.val_one,
    mul_one]

/-- **Cutting down.** A nonzero element of a two-sided ideal yields the indicator of a nonempty
clopen set in the ideal. -/
theorem exists_charFn_mem [TotallySeparatedSpace X]
    (hfree : ∀ (n : ℤ) (x : X), τ n x = x → n = 0)
    (hcov : ∀ (n : ℤ) (f : LocallyConstant X K),
      ((u ^ n : Rˣ) : R) * ι f = ι (comap (τ n) f) * ((u ^ n : Rˣ) : R))
    (hspan : ∀ r : R, ∃ (s : Finset ℤ) (f : ℤ → LocallyConstant X K),
      r = ∑ j ∈ s, ι (f j) * ((u ^ j : Rˣ) : R))
    (I : TwoSidedIdeal R) {r : R} (hrI : r ∈ I) (hr0 : r ≠ 0) :
    ∃ (W : Set X) (hW : IsClopen W), W.Nonempty ∧ ι (charFn K hW) ∈ I := by
  obtain ⟨s, f, hr⟩ := hspan r
  obtain ⟨j₀, hj₀s, hj₀⟩ : ∃ j₀ ∈ s, f j₀ ≠ 0 := by
    by_contra hcon
    refine hr0 (hr.trans (Finset.sum_eq_zero fun j hj => ?_))
    have hfj : f j = 0 := by
      by_contra hne
      exact hcon ⟨j, hj, hne⟩
    rw [hfj, map_zero, zero_mul]
  obtain ⟨x, hx⟩ : ∃ x, f j₀ x ≠ 0 := by
    by_contra hcon
    refine hj₀ (LocallyConstant.ext fun y => ?_)
    by_contra hne
    exact hcon ⟨y, by simpa using hne⟩
  obtain ⟨W, hW, hxW, hWC, hWsep⟩ := exists_isClopen_disjoint_images τ
    ((f j₀).isLocallyConstant.isClopen_fiber (f j₀ x)) rfl ((s.erase j₀).image fun j => j - j₀)
    (fun n hn hfix => by
      obtain ⟨j, hj, rfl⟩ := Finset.mem_image.mp hn
      exact Finset.ne_of_mem_erase hj (sub_eq_zero.mp (hfree _ x hfix)))
  refine ⟨W, hW, ⟨x, hxW⟩, ?_⟩
  have hterm : ∀ j ∈ s, j ≠ j₀ → ι (charFn K hW) * (ι (f j) * ((u ^ j : Rˣ) : R)) *
      ((u ^ (-j₀) : Rˣ) : R) * ι (charFn K hW) = 0 := by
    intro j hj hne
    have hsplit : ((u ^ j : Rˣ) : R) * ((u ^ (-j₀) : Rˣ) : R) = ((u ^ (j - j₀) : Rˣ) : R) := by
      rw [← Units.val_mul, ← zpow_add, sub_eq_add_neg]
    have hprod : charFn K hW * f j * comap (τ (j - j₀)) (charFn K hW) = 0 := by
      refine LocallyConstant.ext fun y => ?_
      by_cases hy : y ∈ W
      · have hτy : τ (j - j₀) y ∉ W :=
          hWsep _ (Finset.mem_image.mpr ⟨j, Finset.mem_erase.mpr ⟨hne, hj⟩, rfl⟩) y hy
        simp [(charFn_eq_zero K _ hW).mpr hτy]
      · simp [(charFn_eq_zero K y hW).mpr hy]
    calc ι (charFn K hW) * (ι (f j) * ((u ^ j : Rˣ) : R)) * ((u ^ (-j₀) : Rˣ) : R) *
          ι (charFn K hW)
        = ι (charFn K hW) * ι (f j) * (((u ^ (j - j₀) : Rˣ) : R) * ι (charFn K hW)) := by
          rw [← hsplit]
          simp only [mul_assoc]
      _ = ι (charFn K hW * f j * comap (τ (j - j₀)) (charFn K hW)) *
          ((u ^ (j - j₀) : Rˣ) : R) := by
          rw [hcov]
          simp only [map_mul, mul_assoc]
      _ = 0 := by rw [hprod, map_zero, zero_mul]
  have hcut : charFn K hW * f j₀ * charFn K hW = const X (f j₀ x) * charFn K hW := by
    refine LocallyConstant.ext fun y => ?_
    by_cases hy : y ∈ W
    · simp [(charFn_eq_one K y hW).mpr hy, (hWC hy : f j₀ y = f j₀ x)]
    · simp [(charFn_eq_zero K y hW).mpr hy]
  have hunit : ((u ^ j₀ : Rˣ) : R) * ((u ^ (-j₀) : Rˣ) : R) = 1 := by
    rw [← Units.val_mul, ← zpow_add, add_neg_cancel, zpow_zero, Units.val_one]
  have hlead : ι (charFn K hW) * (ι (f j₀) * ((u ^ j₀ : Rˣ) : R)) * ((u ^ (-j₀) : Rˣ) : R) *
      ι (charFn K hW) = ι (const X (f j₀ x) * charFn K hW) := by
    calc ι (charFn K hW) * (ι (f j₀) * ((u ^ j₀ : Rˣ) : R)) * ((u ^ (-j₀) : Rˣ) : R) *
          ι (charFn K hW)
        = ι (charFn K hW) * ι (f j₀) *
          (((u ^ j₀ : Rˣ) : R) * ((u ^ (-j₀) : Rˣ) : R)) * ι (charFn K hW) := by
          simp only [mul_assoc]
      _ = ι (const X (f j₀ x) * charFn K hW) := by
          rw [hunit, mul_one, ← hcut, map_mul, map_mul]
  have hz : ι (const X (f j₀ x) * charFn K hW) ∈ I := by
    have hmem : ι (charFn K hW) * r * ((u ^ (-j₀) : Rˣ) : R) * ι (charFn K hW) ∈ I :=
      I.mul_mem_right _ _ (I.mul_mem_right _ _ (I.mul_mem_left _ _ hrI))
    rwa [hr, Finset.mul_sum, Finset.sum_mul, Finset.sum_mul,
      Finset.sum_eq_single_of_mem j₀ hj₀s hterm, hlead] at hmem
  have hc : const X (f j₀ x)⁻¹ * (const X (f j₀ x) * charFn K hW) = charFn K hW := by
    refine LocallyConstant.ext fun y => ?_
    simp only [coe_mul, coe_const, Pi.mul_apply, Function.const_apply]
    exact inv_mul_cancel_left₀ hx _
  have hinv := I.mul_mem_left (ι (const X (f j₀ x)⁻¹)) _ hz
  rwa [← map_mul, hc] at hinv

/-- **Covering.** If the indicator of a nonempty clopen set lies in a two-sided ideal, then so
does `1`: its conjugates cover `X`, finitely many suffice, and `h + g - h g` joins them. -/
theorem one_mem_of_charFn_mem [CompactSpace X]
    (hmin : ∀ U : Set X, IsOpen U → U.Nonempty → ∀ y : X, ∃ n : ℤ, τ n y ∈ U)
    (hcov : ∀ (n : ℤ) (f : LocallyConstant X K),
      ((u ^ n : Rˣ) : R) * ι f = ι (comap (τ n) f) * ((u ^ n : Rˣ) : R))
    (I : TwoSidedIdeal R) {W : Set X} (hW : IsClopen W) (hne : W.Nonempty)
    (hWI : ι (charFn K hW) ∈ I) : (1 : R) ∈ I := by
  have hconj : ∀ n : ℤ, ι (comap (τ n) (charFn K hW)) ∈ I := fun n => by
    rw [← conj_eq_comap τ ι u hcov]
    exact I.mul_mem_right _ _ (I.mul_mem_left _ _ hWI)
  obtain ⟨S, hS⟩ := isCompact_univ.elim_finite_subcover (fun n : ℤ => τ n ⁻¹' W)
    (fun n => hW.isOpen.preimage (τ n).continuous)
    (fun y _ => Set.mem_iUnion.mpr (hmin W hW.isOpen hne y))
  have hunion : ∀ T : Finset ℤ, ∃ g : LocallyConstant X K,
      ι g ∈ I ∧ ∀ y : X, (∃ n ∈ T, τ n y ∈ W) → g y = 1 := by
    intro T
    induction T using Finset.induction_on with
    | empty => exact ⟨0, by rw [map_zero]; exact I.zero_mem, fun _ ⟨_, hn, _⟩ => by simp at hn⟩
    | insert a T _ ih =>
      obtain ⟨g, hgI, hg⟩ := ih
      refine ⟨comap (τ a) (charFn K hW) + g - comap (τ a) (charFn K hW) * g, ?_, fun y hy => ?_⟩
      · rw [map_sub, map_add, map_mul]
        exact I.sub_mem (I.add_mem (hconj a) hgI) (I.mul_mem_right _ _ (hconj a))
      · obtain ⟨n, hn, hyW⟩ := hy
        simp only [LocallyConstant.coe_sub, LocallyConstant.coe_add, LocallyConstant.coe_mul,
          Pi.sub_apply, Pi.add_apply, Pi.mul_apply]
        rcases Finset.mem_insert.mp hn with rfl | hn
        · rw [LocallyConstant.coe_comap_apply, (charFn_eq_one K _ hW).mpr hyW]
          ring
        · rw [hg y ⟨n, hn, hyW⟩]
          ring
  obtain ⟨g, hgI, hg⟩ := hunion S
  have hg1 : g = 1 := LocallyConstant.ext fun y => by
    obtain ⟨n, hn, hyW⟩ := Set.mem_iUnion₂.mp (hS (Set.mem_univ y))
    simpa using hg y ⟨n, hn, hyW⟩
  rwa [hg1, map_one] at hgI

/-- **Simplicity.** A two-sided ideal with a nonzero element contains `1`. -/
theorem one_mem_of_ne_zero_mem [CompactSpace X] [TotallySeparatedSpace X]
    (hfree : ∀ (n : ℤ) (x : X), τ n x = x → n = 0)
    (hmin : ∀ U : Set X, IsOpen U → U.Nonempty → ∀ y : X, ∃ n : ℤ, τ n y ∈ U)
    (hcov : ∀ (n : ℤ) (f : LocallyConstant X K),
      ((u ^ n : Rˣ) : R) * ι f = ι (comap (τ n) f) * ((u ^ n : Rˣ) : R))
    (hspan : ∀ r : R, ∃ (s : Finset ℤ) (f : ℤ → LocallyConstant X K),
      r = ∑ j ∈ s, ι (f j) * ((u ^ j : Rˣ) : R))
    (I : TwoSidedIdeal R) {r : R} (hrI : r ∈ I) (hr0 : r ≠ 0) : (1 : R) ∈ I := by
  obtain ⟨W, hW, hne, hWI⟩ := exists_charFn_mem τ ι u hfree hcov hspan I hrI hr0
  exact one_mem_of_charFn_mem τ ι u hmin hcov I hW hne hWI

/-- A nontrivial ring spanned by `ι(LC(X, K))` and the powers of a covariant unit, over a free
minimal family `τ` on a compact totally separated space, is a simple ring. -/
theorem isSimpleRing_of_covariant [Nontrivial R] [CompactSpace X] [TotallySeparatedSpace X]
    (hfree : ∀ (n : ℤ) (x : X), τ n x = x → n = 0)
    (hmin : ∀ U : Set X, IsOpen U → U.Nonempty → ∀ y : X, ∃ n : ℤ, τ n y ∈ U)
    (hcov : ∀ (n : ℤ) (f : LocallyConstant X K),
      ((u ^ n : Rˣ) : R) * ι f = ι (comap (τ n) f) * ((u ^ n : Rˣ) : R))
    (hspan : ∀ r : R, ∃ (s : Finset ℤ) (f : ℤ → LocallyConstant X K),
      r = ∑ j ∈ s, ι (f j) * ((u ^ j : Rˣ) : R)) :
    IsSimpleRing R := by
  refine IsSimpleRing.of_eq_bot_or_eq_top fun I => ?_
  by_cases hI : I = ⊥
  · exact Or.inl hI
  · refine Or.inr ((TwoSidedIdeal.one_mem_iff I).mp ?_)
    obtain ⟨r, hrI, hr0⟩ : ∃ r ∈ I, r ≠ 0 := by
      by_contra hcon
      refine hI (TwoSidedIdeal.ext fun r => ⟨fun hr => ?_, fun hr => ?_⟩)
      · rw [TwoSidedIdeal.mem_bot]
        by_contra hr0
        exact hcon ⟨r, hr, hr0⟩
      · rw [TwoSidedIdeal.mem_bot] at hr
        rw [hr]
        exact I.zero_mem
    exact one_mem_of_ne_zero_mem τ ι u hfree hmin hcov hspan I hrI hr0

/-- The same statement for a free minimal continuous `ℤ`-action, with the convention
`u^n ι(f) u^{-n} = ι(f ∘ ((-n) +ᵥ ·))`, so that `u ι(χ_W) u⁻¹ = ι(χ_{1 +ᵥ W})`. -/
theorem isSimpleRing_of_free_minimal [Nontrivial R] [CompactSpace X] [TotallySeparatedSpace X]
    [AddAction ℤ X] [ContinuousConstVAdd ℤ X] [AddAction.IsMinimal ℤ X]
    (hfree : ∀ (n : ℤ) (x : X), n +ᵥ x = x → n = 0)
    (hcov : ∀ (n : ℤ) (f : LocallyConstant X K), ((u ^ n : Rˣ) : R) * ι f =
      ι (comap ⟨fun x => (-n) +ᵥ x, continuous_const_vadd (-n)⟩ f) * ((u ^ n : Rˣ) : R))
    (hspan : ∀ r : R, ∃ (s : Finset ℤ) (f : ℤ → LocallyConstant X K),
      r = ∑ j ∈ s, ι (f j) * ((u ^ j : Rˣ) : R)) :
    IsSimpleRing R :=
  isSimpleRing_of_covariant (fun n => ⟨fun x => (-n) +ᵥ x, continuous_const_vadd (-n)⟩) ι u
    (fun n x hx => neg_eq_zero.mp (hfree (-n) x hx))
    (fun U hU hne y => by
      obtain ⟨c, hc⟩ := hU.exists_vadd_mem ℤ y hne
      exact ⟨-c, by simpa using hc⟩)
    hcov hspan

end Covariant

/-! ### Model tests: the identity family is neither free nor minimal. -/

example : ¬ ∀ (n : ℤ) (x : Unit), (fun _ : ℤ => ContinuousMap.id Unit) n x = x → n = 0 :=
  fun h => one_ne_zero (h 1 () rfl)

example : ¬ ∀ U : Set Bool, IsOpen U → U.Nonempty →
    ∀ y : Bool, ∃ n : ℤ, (fun _ : ℤ => ContinuousMap.id Bool) n y ∈ U := fun h => by
  obtain ⟨_, hn⟩ := h {true} (isOpen_discrete _) ⟨true, rfl⟩ false
  simp at hn
