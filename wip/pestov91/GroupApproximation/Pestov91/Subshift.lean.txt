/-!
# The period-doubling subshift

The Pestov 9.1 witness ring is built over an infinite minimal subshift `X ⊆ Bool^ℤ`. Here `X` is
the orbit closure of the period-doubling Toeplitz sequence `pd n = ν₂(n) mod 2`, with
`pd 0 = false`. In the base-point representation of the blueprint
(`research/artifacts/pestov91-lean-blueprint-2026-09-13.md`, D1 and D2), points of `X` become
positions `n : ℤ` of `pd` and cylinders become windows. Everything below is 2-adic arithmetic
with explicit constants:

* `WindowEq x n y m k`: the radius-`k` windows of `x` at `n` and of `y` at `m` coincide;
* `pd_uniformlyRecurrent`, property (U): every radius-`k` window of `pd` recurs in every stretch
  of length `2^(2k+3)`;
* `pd_aperiodic`, property (P): windows of radius `k ≥ 2L` at distance `0 < |j| ≤ L` differ;
* `pd_periodicModel`, the periodic models of A§3.1: for all `k` and `B` there is a periodic
  sequence of period `N > B` with exactly the radius-`k` windows of `pd`.
-/

namespace GroupApproximation.Pestov91

/-- The period-doubling sequence: the parity of the 2-adic valuation (`pd 0 = false`). -/
def pd (n : ℤ) : Bool := (padicValInt 2 n).bodd

/-- `x` around `n` and `y` around `m` agree on the centred window of radius `k`. -/
def WindowEq (x : ℤ → Bool) (n : ℤ) (y : ℤ → Bool) (m : ℤ) (k : ℕ) : Prop :=
  ∀ i ∈ Finset.Icc (-(k : ℤ)) k, x (n + i) = y (m + i)

instance decidableWindowEq (x : ℤ → Bool) (n : ℤ) (y : ℤ → Bool) (m : ℤ) (k : ℕ) :
    Decidable (WindowEq x n y m k) :=
  inferInstanceAs (Decidable (∀ i ∈ Finset.Icc (-(k : ℤ)) k, x (n + i) = y (m + i)))

namespace WindowEq

variable {x y z : ℤ → Bool} {n m p : ℤ} {k k' : ℕ}

theorem mono (h : WindowEq x n y m k') (hk : k ≤ k') : WindowEq x n y m k := by
  intro i hi
  obtain ⟨hi1, hi2⟩ := Finset.mem_Icc.1 hi
  have hkk : (k : ℤ) ≤ k' := by exact_mod_cast hk
  exact h i (Finset.mem_Icc.2 ⟨by linarith, by linarith⟩)

theorem symm (h : WindowEq x n y m k) : WindowEq y m x n k := fun i hi => (h i hi).symm

theorem trans (h : WindowEq x n y m k) (h' : WindowEq y m z p k) : WindowEq x n z p k :=
  fun i hi => (h i hi).trans (h' i hi)

/-- Recentring: a radius-`k+|i|` window contains the radius-`k` window at offset `i`. -/
theorem shift (i : ℤ) (h : WindowEq x n y m (k + i.natAbs)) :
    WindowEq x (n + i) y (m + i) k := by
  intro j hj
  obtain ⟨hj1, hj2⟩ := Finset.mem_Icc.1 hj
  have habs : |i| = (i.natAbs : ℤ) := Int.abs_eq_natAbs i
  obtain ⟨h1, h2⟩ := abs_le.1 (le_refl |i|)
  have e := h (i + j) (Finset.mem_Icc.2 ⟨by push_cast; linarith, by push_cast; linarith⟩)
  rwa [← add_assoc, ← add_assoc] at e

end WindowEq

theorem pd_add_of_dvd {e : ℕ} {a d : ℤ} (ha : ¬ (2 : ℤ) ^ e ∣ a) (hd : (2 : ℤ) ^ e ∣ d) :
    pd (a + d) = pd a := by
  simp only [pd, padicValInt_add_of_dvd ha hd]

theorem pd_two_pow_mul {e : ℕ} {s : ℤ} (hs : ¬ (2 : ℤ) ∣ s) : pd (2 ^ e * s) = e.bodd := by
  simp only [pd, padicValInt_two_pow_mul hs]

theorem pd_zero : pd 0 = false := by
  simp only [pd, padicValInt.zero, Nat.bodd_zero]

/-! Model tests for `pd`. -/

theorem pd_one : pd 1 = false := by
  have h := pd_two_pow_mul (e := 0) (s := 1) (by omega)
  norm_num at h
  exact h.trans rfl

theorem pd_two : pd 2 = true := by
  have h := pd_two_pow_mul (e := 1) (s := 1) (by omega)
  norm_num at h
  exact h.trans rfl

theorem pd_four : pd 4 = false := by
  have h := pd_two_pow_mul (e := 2) (s := 1) (by omega)
  norm_num at h
  exact h.trans rfl

theorem pd_eight : pd 8 = true := by
  have h := pd_two_pow_mul (e := 3) (s := 1) (by omega)
  norm_num at h
  exact h.trans rfl

/-- `pd` is nowhere locally periodic at scale `2|j|`. -/
theorem pd_exists_ne (c : ℤ) {j : ℤ} (hj : j ≠ 0) :
    ∃ m : ℤ, |m| ≤ 2 * |j| ∧ pd (c + m) ≠ pd (c + m + j) := by
  obtain ⟨t, ht, hjt⟩ := exists_two_pow_mul_odd hj
  generalize padicValInt 2 j = a at hjt
  obtain ⟨m, s, hm, hs, hcm⟩ := exists_near_two_pow_mul_odd (a + 1) c
  have hle : (2 : ℤ) ^ a ≤ |j| := Int.le_of_dvd (abs_pos.2 hj) ((dvd_abs _ _).2 ⟨t, hjt⟩)
  have hpow : (2 : ℤ) ^ (a + 1) = 2 * 2 ^ a := by ring
  refine ⟨m, by linarith, ?_⟩
  have hodd : ¬ (2 : ℤ) ∣ 2 * s + t := fun h => ht ((dvd_add_right (dvd_mul_right 2 s)).1 h)
  have h1 : c + m + j = 2 ^ a * (2 * s + t) := by rw [hcm, hjt]; ring
  rw [h1, hcm, pd_two_pow_mul hs, pd_two_pow_mul hodd, show a + 1 = Nat.succ a from rfl,
    Nat.bodd_succ]
  cases a.bodd <;> decide

/-- (P) Aperiodicity with explicit return times: long centred windows at distance `0 < |j| ≤ L`
differ. -/
theorem pd_aperiodic (L : ℕ) :
    ∃ k₀ : ℕ, ∀ k : ℕ, k₀ ≤ k → ∀ n j : ℤ, j ≠ 0 → |j| ≤ (L : ℤ) →
      ¬ WindowEq pd n pd (n + j) k := by
  refine ⟨2 * L, fun k hk n j hj hjL hw => ?_⟩
  obtain ⟨m, hm, hne⟩ := pd_exists_ne n hj
  have hk' : (2 * L : ℤ) ≤ k := by exact_mod_cast hk
  obtain ⟨hm1, hm2⟩ := abs_le.1 hm
  have hj0 := abs_nonneg j
  have e := hw m (Finset.mem_Icc.2 ⟨by linarith, by linarith⟩)
  rw [add_right_comm n j m] at e
  exact hne e

/-- Model test: the constant sequence fails the statement of `pd_aperiodic`. -/
theorem not_aperiodic_const (L : ℕ) (hL : 1 ≤ L) :
    ¬ ∃ k₀ : ℕ, ∀ k : ℕ, k₀ ≤ k → ∀ n j : ℤ, j ≠ 0 → |j| ≤ (L : ℤ) →
      ¬ WindowEq (fun _ => false) n (fun _ => false) (n + j) k := by
  rintro ⟨k₀, hk₀⟩
  exact hk₀ k₀ le_rfl 0 1 one_ne_zero (by simpa using hL) fun _ _ => rfl

/-- A radius-`k` window of `pd` is reproduced at every position of one residue class modulo
`2^(K+2)`, once `2k < 2^K`. -/
theorem pd_exists_center (k : ℕ) {K : ℕ} (hK : 2 * k < 2 ^ K) (n : ℤ) :
    ∃ c : ℤ, ∀ p : ℤ, (2 : ℤ) ^ (K + 2) ∣ p - c → WindowEq pd p pd n k := by
  have hkK : (2 * k : ℤ) < 2 ^ K := by exact_mod_cast hK
  have hdvdK : (2 : ℤ) ^ K ∣ 2 ^ (K + 2) := pow_dvd_pow 2 (by omega)
  by_cases hex : ∃ i₀ ∈ Finset.Icc (-(k : ℤ)) k, (2 : ℤ) ^ K ∣ n + i₀
  · obtain ⟨i₀, hi₀, hdiv⟩ := hex
    obtain ⟨hi₀1, hi₀2⟩ := Finset.mem_Icc.1 hi₀
    -- the window meets exactly one multiple of `2^K`, at the offset `i₀`
    have huniq : ∀ i ∈ Finset.Icc (-(k : ℤ)) k, (2 : ℤ) ^ K ∣ n + i → i = i₀ := by
      intro i hi hdi
      obtain ⟨hi1, hi2⟩ := Finset.mem_Icc.1 hi
      have hsub : (2 : ℤ) ^ K ∣ i - i₀ := by
        have h3 := dvd_sub hdi hdiv
        rwa [show n + i - (n + i₀) = i - i₀ by ring] at h3
      have h4 := Int.eq_zero_of_abs_lt_dvd hsub (abs_lt.2 ⟨by linarith, by linarith⟩)
      linarith
    -- every multiple `2^K (u + 4v)` carries the value of `pd` at that offset
    obtain ⟨u, hu⟩ : ∃ u : ℤ, ∀ v : ℤ, pd (2 ^ K * (u + 4 * v)) = pd (n + i₀) := by
      have hbit : pd (n + i₀) = K.bodd ∨ pd (n + i₀) = (K + 1).bodd := by
        rw [show K + 1 = Nat.succ K from rfl, Nat.bodd_succ]
        cases pd (n + i₀) <;> cases K.bodd <;> decide
      rcases hbit with h | h
      · exact ⟨1, fun v => by rw [h, pd_two_pow_mul (s := 1 + 4 * v) (by omega)]⟩
      · refine ⟨2, fun v => ?_⟩
        rw [h, show (2 : ℤ) ^ K * (2 + 4 * v) = 2 ^ (K + 1) * (1 + 2 * v) by ring,
          pd_two_pow_mul (s := 1 + 2 * v) (by omega)]
    refine ⟨2 ^ K * u - i₀, fun p hp i hi => ?_⟩
    obtain ⟨v, hv⟩ := hp
    rcases eq_or_ne i i₀ with heq | hne
    · rw [heq, show p + i₀ = 2 ^ K * (u + 4 * v) by linear_combination hv, hu v]
    · have hndvd : ¬ (2 : ℤ) ^ K ∣ n + i := fun hd => hne (huniq i hi hd)
      have hpn : p - n = 2 ^ (K + 2) * v + 2 ^ K * u - (n + i₀) := by linear_combination hv
      have hd : (2 : ℤ) ^ K ∣ p - n := by
        rw [hpn]
        exact dvd_sub (dvd_add (dvd_mul_of_dvd_left hdvdK v) (dvd_mul_right _ u)) hdiv
      rw [show p + i = (n + i) + (p - n) by ring, pd_add_of_dvd hndvd hd]
  · push Not at hex
    refine ⟨n, fun p hp i hi => ?_⟩
    rw [show p + i = (n + i) + (p - n) by ring, pd_add_of_dvd (hex i hi) (hdvdK.trans hp)]

/-- (U) Uniform recurrence: every radius-`k` window of `pd` recurs in every stretch `[a, a + l)`. -/
theorem pd_uniformlyRecurrent (k : ℕ) :
    ∃ l : ℕ, ∀ n a : ℤ, ∃ t : ℕ, t < l ∧ WindowEq pd (a + t) pd n k := by
  have hK : 2 * k < 2 ^ (2 * k + 1) :=
    lt_of_le_of_lt (by omega) (Nat.lt_pow_self Nat.one_lt_two)
  refine ⟨2 ^ (2 * k + 1 + 2), fun n a => ?_⟩
  obtain ⟨c, hc⟩ := pd_exists_center k hK n
  obtain ⟨t, ht, htc⟩ := exists_lt_dvd_add_sub a c (N := 2 ^ (2 * k + 1 + 2)) (by positivity)
  exact ⟨t, ht, hc (a + t) (by exact_mod_cast htc)⟩

/-- The `2^(F+1)`-periodic sequence agreeing with `pd` on `[1, 2^(F+1)]`. -/
def pdPeriodic (F : ℕ) (n : ℤ) : Bool := pd ((n - 1) % 2 ^ (F + 1) + 1)

theorem pdPeriodic_add_mul (F : ℕ) (n q : ℤ) :
    pdPeriodic F (n + 2 ^ (F + 1) * q) = pdPeriodic F n := by
  simp only [pdPeriodic]
  rw [show n + 2 ^ (F + 1) * q - 1 = (n - 1) + 2 ^ (F + 1) * q by ring,
    Int.add_mul_emod_self_left]

/-- `pdPeriodic F` agrees with `pd` on `[1, 2^(F+2) - 1]`. -/
theorem pdPeriodic_eq (F : ℕ) {n : ℤ} (hn1 : 1 ≤ n) (hn2 : n < 2 * 2 ^ (F + 1)) :
    pdPeriodic F n = pd n := by
  simp only [pdPeriodic]
  have hN : (0 : ℤ) < 2 ^ (F + 1) := by positivity
  rcases lt_or_ge n (2 ^ (F + 1) + 1) with h | h
  · have h' : (n - 1) % 2 ^ (F + 1) = n - 1 := Int.emod_eq_of_lt (by linarith) (by linarith)
    rw [h', sub_add_cancel]
  · have h' : (n - 1) % 2 ^ (F + 1) = n - 1 - 2 ^ (F + 1) := by
      rw [← Int.sub_emod_right (n - 1) (2 ^ (F + 1))]
      exact Int.emod_eq_of_lt (by linarith) (by linarith)
    have hnd : ¬ (2 : ℤ) ^ (F + 1) ∣ n - 2 ^ (F + 1) := fun hd => by
      have h0 := Int.eq_zero_of_abs_lt_dvd hd (abs_lt.2 ⟨by linarith, by linarith⟩)
      linarith
    have h2 := pd_add_of_dvd hnd (dvd_refl ((2 : ℤ) ^ (F + 1)))
    rw [sub_add_cancel] at h2
    rw [h', show n - 1 - 2 ^ (F + 1) + 1 = n - 2 ^ (F + 1) by ring, h2]

/-- Every radius-`k` window of `pdPeriodic F` is a window of `pd`. -/
theorem pdPeriodic_window_legal {F k : ℕ} (hk : 2 * k < 2 ^ (F + 1)) (m : ℤ) :
    ∃ n : ℤ, WindowEq (pdPeriodic F) m pd n k := by
  refine ⟨(m - k - 1) % 2 ^ (F + 1) + k + 1, fun i hi => ?_⟩
  obtain ⟨hi1, hi2⟩ := Finset.mem_Icc.1 hi
  have hN : (0 : ℤ) < 2 ^ (F + 1) := by positivity
  have h0 := Int.emod_nonneg (m - k - 1) hN.ne'
  have h1 := Int.emod_lt_of_pos (m - k - 1) hN
  have hdec := Int.emod_add_mul_ediv (m - k - 1) (2 ^ (F + 1))
  have hkN : (2 * k : ℤ) < 2 ^ (F + 1) := by exact_mod_cast hk
  have hsplit : m + i = ((m - k - 1) % 2 ^ (F + 1) + k + 1 + i) +
      2 ^ (F + 1) * ((m - k - 1) / 2 ^ (F + 1)) := by
    linear_combination (-1 : ℤ) * hdec
  rw [hsplit, pdPeriodic_add_mul,
    pdPeriodic_eq F (n := (m - k - 1) % 2 ^ (F + 1) + k + 1 + i) (by linarith) (by linarith)]

/-- Every radius-`k` window of `pd` is a window of `pdPeriodic (F + 1)`, once `2k < 2^F`. -/
theorem pd_window_occurs {F k : ℕ} (hk : 2 * k < 2 ^ F) (n : ℤ) :
    ∃ m : ℤ, WindowEq pd n (pdPeriodic (F + 1)) m k := by
  obtain ⟨c, hc⟩ := pd_exists_center k hk n
  obtain ⟨t, ht, htc⟩ :=
    exists_lt_dvd_add_sub ((k : ℤ) + 1) c (N := 2 ^ (F + 2)) (by positivity)
  refine ⟨(k : ℤ) + 1 + t,
    (hc ((k : ℤ) + 1 + t) (by exact_mod_cast htc)).symm.trans fun i hi => ?_⟩
  obtain ⟨hi1, hi2⟩ := Finset.mem_Icc.1 hi
  have ht' : (t : ℤ) < 2 ^ (F + 2) := by exact_mod_cast ht
  have hkF : (2 * k : ℤ) < 2 ^ F := by exact_mod_cast hk
  have hk0 : (0 : ℤ) ≤ k := Nat.cast_nonneg k
  have ht0 : (0 : ℤ) ≤ t := Nat.cast_nonneg t
  have hp1 : (2 : ℤ) ^ (F + 2) = 4 * 2 ^ F := by ring
  have hp2 : (2 : ℤ) ^ (F + 1 + 1) = 4 * 2 ^ F := by ring
  exact (pdPeriodic_eq (F + 1) (n := (k : ℤ) + 1 + t + i) (by linarith) (by linarith)).symm

/-- Periodic models (A§3.1): a periodic sequence with period `N > B` and the same radius-`k`
windows as `pd`. -/
theorem pd_periodicModel (k B : ℕ) :
    ∃ (N : ℕ) (y : ℤ → Bool), B < N ∧ (∀ n : ℤ, y (n + N) = y n) ∧
      (∀ m : ℤ, ∃ n : ℤ, WindowEq y m pd n k) ∧ (∀ n : ℤ, ∃ m : ℤ, WindowEq pd n y m k) := by
  obtain ⟨F, hF⟩ : ∃ F : ℕ, F = 2 * k + B := ⟨_, rfl⟩
  have hF2 : F < 2 ^ F := Nat.lt_pow_self Nat.one_lt_two
  have hFF : 2 ^ F ≤ 2 ^ (F + 1 + 1) := Nat.pow_le_pow_right Nat.zero_lt_two (by omega)
  refine ⟨2 ^ (F + 1 + 1), pdPeriodic (F + 1), by omega, fun n => ?_, fun m => ?_,
    fun n => ?_⟩
  · simpa using pdPeriodic_add_mul (F + 1) n 1
  · exact pdPeriodic_window_legal (lt_of_lt_of_le (by omega) hFF) m
  · exact pd_window_occurs (by omega) n
