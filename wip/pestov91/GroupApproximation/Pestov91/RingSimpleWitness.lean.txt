/-!
# The Pestov 9.1 witness ring is simple

`WitnessRing = CrossedProduct toeplitzShift = LC(ToeplitzSpace, 𝔽₂) ⋊ ℤ` is a simple ring
(blueprint v2, leaf G2). The proof is `isSimpleRing_of_minimal_free` from
`GroupApproximation/Pestov91/RingSimple.lean`, the direct cut-and-cover argument of
`research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`, §1.3, over the
field `𝔽₂`. Nothing is cited: BCFS covers only the complex Steinberg algebra (review note N6).

* `isSimpleRing_crossedProduct`: for a homeomorphism `σ` of a nonempty compact totally separated
  space whose powers have no fixed points and dense orbits, `CrossedProduct σ` is simple. The
  data are `ι = C σ`, `u = u σ`, `τ j = σ ^ (-j)` and `σ' j = shift σ j`, with covariance
  `u_zpow_mul_C` and normal form `exists_eq_sum_C_mul_u_zpow`.
* `isSimpleRing_witnessRing`: the Toeplitz shift is free (`toeplitzShift_free`) and minimal
  (`toeplitzShift_dense_orbit`), so the witness ring is simple. No hypotheses.
-/

namespace GroupApproximation.Pestov91

/-- **Simplicity of `LC(X, 𝔽₂) ⋊_σ ℤ`.** A homeomorphism `σ` of a nonempty compact totally
separated space whose nonzero powers have no fixed points and whose orbits are dense gives a
simple crossed product. -/
theorem isSimpleRing_crossedProduct {X : Type*} [TopologicalSpace X] [Nonempty X] [CompactSpace X]
    [TotallySeparatedSpace X] (σ : X ≃ₜ X) (hfree : ∀ j : ℤ, j ≠ 0 → ∀ x, (σ ^ (-j)) x ≠ x)
    (hdense : ∀ x, Dense (Set.range fun j : ℤ => (σ ^ (-j)) x)) :
    IsSimpleRing (CrossedProduct σ) :=
  isSimpleRing_of_minimal_free (CrossedProduct.C σ) (CrossedProduct.u σ)
    (fun j => ⇑(σ ^ (-j))) (fun j => (σ ^ (-j)).continuous) hfree hdense
    (fun j => ⇑(CrossedProduct.shift σ j)) (CrossedProduct.shift_apply σ)
    (CrossedProduct.u_zpow_mul_C σ) (CrossedProduct.exists_eq_sum_C_mul_u_zpow σ)

/-- **Leaf G2.** The Pestov 9.1 witness ring `LC(ToeplitzSpace, 𝔽₂) ⋊ ℤ` is a simple ring. -/
theorem isSimpleRing_witnessRing : IsSimpleRing WitnessRing :=
  isSimpleRing_crossedProduct toeplitzShift toeplitzShift_free toeplitzShift_dense_orbit

instance : IsSimpleRing WitnessRing := isSimpleRing_witnessRing

/-! ### Model test: the identity homeomorphism of a nonempty space is not free. -/

example {X : Type*} [TopologicalSpace X] [Nonempty X] :
    ¬ ∀ j : ℤ, j ≠ 0 → ∀ x : X, ((1 : X ≃ₜ X) ^ (-j)) x ≠ x := fun h => by
  obtain ⟨x⟩ := ‹Nonempty X›
  exact h 1 one_ne_zero x (by rw [one_zpow, Homeomorph.one_apply])

end GroupApproximation.Pestov91
