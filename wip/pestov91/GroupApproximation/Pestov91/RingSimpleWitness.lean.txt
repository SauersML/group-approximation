/-- **Simplicity of the witness ring.** `LC(ToeplitzSpace, 𝔽₂) ⋊ ℤ` is a simple ring. -/
theorem isSimpleRing_witnessRing : IsSimpleRing WitnessRing :=
  isSimpleRing_of_minimal_free (CrossedProduct.C : WitnessCoeff →+* WitnessRing) CrossedProduct.unit
    (fun j y => (-j) +ᵥ y) witness_continuous_vadd witness_free witness_dense_orbit
    (fun j f => Multiplicative.ofAdd j • f) ofAdd_smul_witnessCoeff_apply
    CrossedProduct.unit_zpow_mul_C CrossedProduct.exists_sum_C_mul_unit_zpow

instance : IsSimpleRing WitnessRing := isSimpleRing_witnessRing
