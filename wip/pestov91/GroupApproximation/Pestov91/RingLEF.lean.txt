/-!
# Pestov 9.1: local matrix models make elementary groups LEF

Section 3.3 of
`research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`.

A ring `R` has *local matrix models* when every finite set `F ⊆ R` carries an
additive map `ψ : R →+ Matrix (Fin N) (Fin N) (ZMod 2)` that is unital,
injective on `F` and multiplicative on pairs from `F`.  Applied entrywise to an
invertible matrix whose entries and inverse entries lie in `F`, `ψ` gives an
invertible matrix over `M_N(𝔽₂)`, multiplicatively on such matrices.  The
target `GL_ι(M_N(𝔽₂))` is a finite group, so `GL_ι(R)` is LEF, and so is every
subgroup of it, in particular `EL_ι(R)`.

The models for `R = LC(X, 𝔽₂) ⋊ ℤ`, by periodic return words, are built in
`GroupApproximation.Pestov91.RingLEFModel`.

## Main results

* `IsLocalMatrixModel`, `HasLocalMatrixModels`: the ring-level input.
* `isLEF_units_matrix_of_hasLocalMatrixModels`: `GL_ι(R)` is LEF.
* `isLEF_elementaryGroup_of_hasLocalMatrixModels`: `EL_ι(R)` is LEF.
-/

namespace GroupApproximation.Pestov91

/-- A local model of a ring on a finite set `F`: an additive map that is
unital, injective on `F` and multiplicative on pairs from `F`. -/
structure IsLocalMatrixModel {R S : Type*} [Ring R] [Ring S] (F : Finset R)
    (ψ : R →+ S) : Prop where
  map_one : ψ 1 = 1
  injOn : Set.InjOn ψ F
  map_mul : ∀ a ∈ F, ∀ b ∈ F, ψ (a * b) = ψ a * ψ b

/-- Every finite subset of `R` has a local model in a matrix ring over `𝔽₂`. -/
def HasLocalMatrixModels (R : Type*) [Ring R] : Prop :=
  ∀ F : Finset R, ∃ (N : ℕ) (ψ : R →+ Matrix (Fin N) (Fin N) (ZMod 2)),
    IsLocalMatrixModel F ψ

namespace IsLocalMatrixModel

variable {R S : Type*} [Ring R] [Ring S] {F : Finset R} {ψ : R →+ S}

theorem mono (h : IsLocalMatrixModel F ψ) {F' : Finset R} (hF : F' ⊆ F) :
    IsLocalMatrixModel F' ψ where
  map_one := h.map_one
  injOn := h.injOn.mono (Finset.coe_subset.2 hF)
  map_mul a ha b hb := h.map_mul a (hF ha) b (hF hb)

/-- The partial-operation form: unital, injective on `F`, and additive and
multiplicative whenever the sum or the product lies in `F`. -/
theorem textbook (h : IsLocalMatrixModel F ψ) :
    ψ 1 = 1 ∧ Set.InjOn ψ F ∧
      (∀ a ∈ F, ∀ b ∈ F, a + b ∈ F → ψ (a + b) = ψ a + ψ b) ∧
      (∀ a ∈ F, ∀ b ∈ F, a * b ∈ F → ψ (a * b) = ψ a * ψ b) :=
  ⟨h.map_one, h.injOn, fun a _ b _ _ ↦ map_add ψ a b,
    fun a ha b hb _ ↦ h.map_mul a ha b hb⟩

/-- A local model stays one after composing with an injective ring map. -/
theorem comp_injective {T : Type*} [Ring T] (h : IsLocalMatrixModel F ψ) (θ : S →+* T)
    (hθ : Function.Injective θ) : IsLocalMatrixModel F (θ.toAddMonoidHom.comp ψ) where
  map_one := by
    rw [AddMonoidHom.comp_apply, h.map_one]
    exact map_one θ
  injOn _ ha _ hb hab := h.injOn ha hb (hθ hab)
  map_mul a ha b hb := by
    rw [AddMonoidHom.comp_apply, h.map_mul a ha b hb]
    exact map_mul θ _ _

/-- Entrywise application is multiplicative on matrices with entries in `F`. -/
theorem mapMatrix_mul {ι : Type*} [Fintype ι] (h : IsLocalMatrixModel F ψ)
    {A B : Matrix ι ι R} (hA : ∀ i j, A i j ∈ F) (hB : ∀ i j, B i j ∈ F) :
    ψ.mapMatrix (A * B) = ψ.mapMatrix A * ψ.mapMatrix B := by
  ext i j
  simp only [AddMonoidHom.mapMatrix_apply, Matrix.map_apply, Matrix.mul_apply, map_sum]
  exact Finset.sum_congr rfl fun k _ ↦ h.map_mul _ (hA i k) _ (hB k j)

/-- Entrywise application is unital. -/
theorem mapMatrix_one {ι : Type*} [DecidableEq ι] (h : IsLocalMatrixModel F ψ) :
    ψ.mapMatrix (1 : Matrix ι ι R) = 1 := by
  ext i j
  rw [AddMonoidHom.mapMatrix_apply, Matrix.map_apply, Matrix.one_apply, Matrix.one_apply]
  split_ifs
  · exact h.map_one
  · exact map_zero ψ

end IsLocalMatrixModel

/-- Models indexed by any finite type give models indexed by `Fin N`. -/
theorem hasLocalMatrixModels_of_fintype {R : Type*} [Ring R]
    (h : ∀ F : Finset R, ∃ (n : Type) (_ : Fintype n) (_ : DecidableEq n)
      (ψ : R →+ Matrix n n (ZMod 2)), IsLocalMatrixModel F ψ) :
    HasLocalMatrixModels R := fun F ↦ by
  obtain ⟨n, _, _, ψ, hψ⟩ := h F
  let e : Matrix n n (ZMod 2) ≃+* Matrix (Fin (Fintype.card n)) (Fin (Fintype.card n)) (ZMod 2) :=
    Matrix.reindexRingEquiv (R := ZMod 2) (Fintype.equivFin n)
  exact ⟨Fintype.card n, e.toRingHom.toAddMonoidHom.comp ψ, hψ.comp_injective _ e.injective⟩

section Units

variable {ι R : Type*} [Fintype ι] [DecidableEq ι] [Ring R] [DecidableEq R]

/-- The entries of the matrices in `s` and of their inverses. -/
def unitEntries (s : Finset (Matrix ι ι R)ˣ) : Finset R :=
  s.biUnion fun g ↦
    Finset.univ.image (fun p : ι × ι ↦ (g : Matrix ι ι R) p.1 p.2) ∪
      Finset.univ.image (fun p : ι × ι ↦ (↑g⁻¹ : Matrix ι ι R) p.1 p.2)

theorem val_apply_mem_unitEntries {s : Finset (Matrix ι ι R)ˣ} {g : (Matrix ι ι R)ˣ}
    (hg : g ∈ s) (i j : ι) : (g : Matrix ι ι R) i j ∈ unitEntries s :=
  Finset.mem_biUnion.2 ⟨g, hg, Finset.mem_union_left _
    (Finset.mem_image.2 ⟨(i, j), Finset.mem_univ _, rfl⟩)⟩

theorem inv_apply_mem_unitEntries {s : Finset (Matrix ι ι R)ˣ} {g : (Matrix ι ι R)ˣ}
    (hg : g ∈ s) (i j : ι) : (↑g⁻¹ : Matrix ι ι R) i j ∈ unitEntries s :=
  Finset.mem_biUnion.2 ⟨g, hg, Finset.mem_union_right _
    (Finset.mem_image.2 ⟨(i, j), Finset.mem_univ _, rfl⟩)⟩

end Units

/-- LEF from injective multiplicative models into finite groups in `Type`. -/
theorem isLEF_of_forall_finset_finite {J : Type*} [Group J]
    (h : ∀ s : Finset J, ∃ (G : Type) (_ : Group G) (_ : Finite G) (f : J → G),
      Set.InjOn f s ∧ ∀ x ∈ s, ∀ y ∈ s, f (x * y) = f x * f y) :
    IsLEF J := by
  refine (isLEF_iff_textbook J).2 fun s ↦ ?_
  obtain ⟨G, _, _, f, hinj, hmul⟩ := h s
  exact ⟨⟨G, inferInstance, Fintype.ofFinite G, Classical.decEq G⟩, f, hinj,
    fun x hx y hy _ ↦ hmul x hx y hy⟩

/-- **`GL_ι(R)` is LEF** when `R` has local matrix models over `𝔽₂`: the models
land in the finite group `GL_ι(M_N(𝔽₂))`. -/
theorem isLEF_units_matrix_of_hasLocalMatrixModels {R : Type*} [Ring R]
    (hR : HasLocalMatrixModels R) (ι : Type) [Fintype ι] [DecidableEq ι] :
    IsLEF (Matrix ι ι R)ˣ := by
  classical
  refine isLEF_of_forall_finset_finite fun s ↦ ?_
  obtain ⟨N, ψ, hψ⟩ := hR (unitEntries s)
  have hunit : ∀ g ∈ s, IsUnit (ψ.mapMatrix (g : Matrix ι ι R)) := fun g hg ↦
    ⟨⟨ψ.mapMatrix (g : Matrix ι ι R), ψ.mapMatrix (↑g⁻¹ : Matrix ι ι R),
      by rw [← hψ.mapMatrix_mul (val_apply_mem_unitEntries hg)
          (inv_apply_mem_unitEntries hg), Units.mul_inv, hψ.mapMatrix_one],
      by rw [← hψ.mapMatrix_mul (inv_apply_mem_unitEntries hg)
          (val_apply_mem_unitEntries hg), Units.inv_mul, hψ.mapMatrix_one]⟩, rfl⟩
  refine ⟨(Matrix ι ι (Matrix (Fin N) (Fin N) (ZMod 2)))ˣ, inferInstance, inferInstance,
    fun g ↦ if hg : IsUnit (ψ.mapMatrix (g : Matrix ι ι R)) then hg.unit else 1, ?_, ?_⟩
  · intro x hx y hy hxy
    have hx' := hunit x (Finset.mem_coe.1 hx)
    have hy' := hunit y (Finset.mem_coe.1 hy)
    dsimp only at hxy
    rw [dif_pos hx', dif_pos hy'] at hxy
    have hval := congrArg Units.val hxy
    rw [IsUnit.unit_spec, IsUnit.unit_spec] at hval
    apply Units.ext
    ext i j
    exact hψ.injOn (Finset.mem_coe.2 (val_apply_mem_unitEntries (Finset.mem_coe.1 hx) i j))
      (Finset.mem_coe.2 (val_apply_mem_unitEntries (Finset.mem_coe.1 hy) i j))
      (congrFun (congrFun hval i) j)
  · intro x hx y hy
    have hprod : ψ.mapMatrix ((x * y : (Matrix ι ι R)ˣ) : Matrix ι ι R) =
        ψ.mapMatrix (x : Matrix ι ι R) * ψ.mapMatrix (y : Matrix ι ι R) := by
      rw [Units.val_mul]
      exact hψ.mapMatrix_mul (val_apply_mem_unitEntries hx) (val_apply_mem_unitEntries hy)
    have hxy : IsUnit (ψ.mapMatrix ((x * y : (Matrix ι ι R)ˣ) : Matrix ι ι R)) := by
      rw [hprod]
      exact (hunit x hx).mul (hunit y hy)
    dsimp only
    rw [dif_pos hxy, dif_pos (hunit x hx), dif_pos (hunit y hy)]
    apply Units.ext
    rw [Units.val_mul, IsUnit.unit_spec, IsUnit.unit_spec, IsUnit.unit_spec, hprod]

/-- **`EL_ι(R)` is LEF** when `R` has local matrix models over `𝔽₂`. -/
theorem isLEF_elementaryGroup_of_hasLocalMatrixModels {R : Type*} [Ring R]
    (hR : HasLocalMatrixModels R) (ι : Type) [Fintype ι] [DecidableEq ι] :
    IsLEF (elementaryGroup ι R) :=
  isLEF_of_injective (elementaryGroup ι R).subtype Subtype.val_injective
    (isLEF_units_matrix_of_hasLocalMatrixModels hR ι)

/-- LEF passes to the quotient by a trivial normal subgroup; with `Z(EL₃(R)) = ⊥` this is the
quotient step of §3.3. -/
theorem isLEF_quotient_of_eq_bot {G : Type*} [Group G] {K : Subgroup G} [K.Normal]
    (hK : K = ⊥) (h : IsLEF G) : IsLEF (G ⧸ K) :=
  isLEF_of_injective
    ((QuotientGroup.quotientMulEquivOfEq hK).trans QuotientGroup.quotientBot).toMonoidHom
    (MulEquiv.injective _) h

/-! ### Model tests -/

/-- `𝔽₂` has local matrix models: the scalar embedding into `M_1(𝔽₂)`. -/
theorem hasLocalMatrixModels_zmodTwo : HasLocalMatrixModels (ZMod 2) := fun _ ↦
  ⟨1, (Matrix.scalar (α := ZMod 2) (Fin 1)).toAddMonoidHom,
    { map_one := map_one (Matrix.scalar (α := ZMod 2) (Fin 1))
      injOn := fun a _ b _ hab ↦ by simpa using congrFun (congrFun hab 0) 0
      map_mul := fun a _ b _ ↦ map_mul (Matrix.scalar (α := ZMod 2) (Fin 1)) a b }⟩

/-- The zero map is not a local model: it is not unital. -/
example (F : Finset (ZMod 2)) :
    ¬ IsLocalMatrixModel F (0 : ZMod 2 →+ Matrix (Fin 1) (Fin 1) (ZMod 2)) := fun h ↦ by
  have h1 := congrFun (congrFun h.map_one 0) 0
  simp at h1

/-- A ring of characteristic `3` has no local matrix models over `𝔽₂`:
additivity forces `ψ 2 = ψ 0`. -/
example : ¬ HasLocalMatrixModels (ZMod 3) := fun h ↦ by
  obtain ⟨N, ψ, hψ⟩ := h {0, 2}
  have h2 : ψ 2 = ψ 0 := by
    rw [show (2 : ZMod 3) = 1 + 1 by decide, map_add, hψ.map_one, map_zero]
    ext i j
    simp only [Matrix.add_apply, Matrix.one_apply, Matrix.zero_apply]
    split_ifs <;> decide
  exact absurd (hψ.injOn (by simp) (by simp) h2) (by decide)

/-- The conclusion on a real instance. -/
example : IsLEF (elementaryGroup (Fin 3) (ZMod 2)) :=
  isLEF_elementaryGroup_of_hasLocalMatrixModels hasLocalMatrixModels_zmodTwo (Fin 3)

end GroupApproximation.Pestov91
