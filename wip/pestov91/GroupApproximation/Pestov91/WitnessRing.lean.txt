/-!
# The witness ring `LC(X, 𝔽₂) ⋊ ℤ` over the Toeplitz subshift

`WitnessRing` is the crossed product `CrossedProduct toeplitzShift`: the locally constant
`𝔽₂`-valued functions on the period-doubling Toeplitz subshift `X`, crossed by the shift
`(T x)_h = x_(h+1)`. It is the ring `R` of the Pestov 9.1 witness `EL₃(R)`.

The ring-level facts are generic in `σ` and live in `GroupApproximation.Pestov91.CrossedProduct`:
* `CrossedProduct.u_zpow_mul_C`, the covariance `hconj`;
* `CrossedProduct.exists_eq_sum_C_mul_u_zpow`, the normal form `hspan`;
* `CrossedProduct.shift_apply`, the action `shift σ n f x = f ((σ ^ (-n)) x)`.

This module only fixes `σ := toeplitzShift` and records that the ring is infinite.
-/

namespace GroupApproximation.Pestov91

/-- The ring of the Pestov 9.1 witness: `LC(X, 𝔽₂) ⋊ ℤ` over the Toeplitz subshift. -/
abbrev WitnessRing : Type := CrossedProduct toeplitzShift

/-- The witness ring is infinite, because the monomials `uⁿ` are pairwise distinct. -/
instance : Infinite WitnessRing := skewMonoidAlgebra_infinite

end GroupApproximation.Pestov91
