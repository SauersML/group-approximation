/-!
# Pestov 9.1: copies of `GL_d(𝔽₂)` in the corner monoid

Part 2 §6.2(b)–(c) of `research/artifacts/pestov-9-1-writeup-2026-09-13-part2.md`, with
coefficient ring `A = LC(X, 𝔽₂)`.

* `cutHom w`: `ε ↦ w ε + (1 - w) I`, the matrix function equal to `ε` on `{w = 1}` and
  to `I` elsewhere.  Over `𝔽₂` every `w` is idempotent, so this is a monoid map for every `w`.
* `cornerCopy he hret w`: the copy `ι_W : GL_(ι × [-m,m])(𝔽₂) → GL_ι(R)`, which is
  `cornerHom ∘ cutHom w` on units and equals `1` off the tower over `W = {w = 1}`.
* `exists_corner_commutator`: if `k ≠ 1` and `k - 1, k⁻¹ - 1 ∈ M_ι(B_m(U))`, then some clopen
  `W ⊆ U` with `χ_W ≠ 0` and some `κ₀ ≠ 1` satisfy `⁅k, ι_W ε⁆ = ι_W ⁅κ₀, ε⁆` for all `ε`.
  This is review note N1 (`χ_(V(W))` is a central idempotent for `k` and `ι_W(GL_d)`), which
  here is multiplication pointwise on `U` (`cornerHom_eq_iff`).
-/

namespace GroupApproximation.Pestov91

open Matrix

variable {X : Type*} [TopologicalSpace X]

theorem zmodTwo_eq_zero_or_eq_one (z : ZMod 2) : z = 0 ∨ z = 1 := by
  revert z
  decide

/-- Every locally constant `𝔽₂`-valued function is idempotent. -/
theorem isIdempotentElem_locallyConstant_zmodTwo (f : LocallyConstant X (ZMod 2)) :
    IsIdempotentElem f := by
  change f * f = f
  ext x
  rw [LocallyConstant.mul_apply]
  rcases zmodTwo_eq_zero_or_eq_one (f x) with h | h <;> rw [h] <;> simp

/-- Cutting by `e` detects agreement on `{e = 1}`. -/
theorem locallyConstant_mul_eq_mul_iff {f g e : LocallyConstant X (ZMod 2)} :
    f * e = g * e ↔ ∀ x, e x = 1 → f x = g x := by
  constructor
  · intro h x hx
    have hx' := DFunLike.congr_fun h x
    simpa [LocallyConstant.mul_apply, hx] using hx'
  · intro h
    ext x
    rw [LocallyConstant.mul_apply, LocallyConstant.mul_apply]
    rcases zmodTwo_eq_zero_or_eq_one (e x) with h0 | h1
    · rw [h0, mul_zero, mul_zero]
    · rw [h1, h x h1]

section Eval

variable {n : Type*} [Fintype n] [DecidableEq n]

/-- Evaluation at `x` of a matrix of locally constant functions. -/
def evalMatrix (x : X) : Matrix n n (LocallyConstant X (ZMod 2)) →+* Matrix n n (ZMod 2) :=
  (LocallyConstant.evalRingHom x).mapMatrix

theorem evalMatrix_apply (x : X) (F : Matrix n n (LocallyConstant X (ZMod 2))) (p q : n) :
    evalMatrix x F p q = F p q x :=
  rfl

theorem evalMatrix_ext {F G : Matrix n n (LocallyConstant X (ZMod 2))}
    (h : ∀ x, evalMatrix x F = evalMatrix x G) : F = G := by
  ext p q x
  exact congrFun (congrFun (h x) p) q

theorem forall_mul_eq_mul_iff {e : LocallyConstant X (ZMod 2)}
    (F G : Matrix n n (LocallyConstant X (ZMod 2))) :
    (∀ p q, F p q * e = G p q * e) ↔ ∀ x, e x = 1 → evalMatrix x F = evalMatrix x G := by
  simp only [locallyConstant_mul_eq_mul_iff]
  constructor
  · intro h x hx
    ext p q
    exact h p q x hx
  · intro h p q x hx
    exact congrFun (congrFun (h x hx) p) q

theorem isClopen_evalMatrix_eq (F : Matrix n n (LocallyConstant X (ZMod 2)))
    (M : Matrix n n (ZMod 2)) : IsClopen {x | evalMatrix x F = M} := by
  have hset : {x | evalMatrix x F = M} = ⋂ p, ⋂ q, {x | F p q x = M p q} := by
    ext x
    simp only [Set.mem_setOf_eq, Set.mem_iInter]
    constructor
    · intro h p q
      exact congrFun (congrFun h p) q
    · intro h
      ext p q
      exact h p q
  rw [hset]
  exact isClopen_iInter_of_finite fun p ↦ isClopen_iInter_of_finite fun q ↦
    (F p q).isLocallyConstant.isClopen_fiber (M p q)

theorem evalMatrix_diagonal (x : X) (d : n → LocallyConstant X (ZMod 2)) :
    evalMatrix x (diagonal d) = diagonal fun i ↦ d i x := by
  ext p q
  rw [evalMatrix_apply, diagonal_apply, diagonal_apply]
  split_ifs <;> rfl

theorem evalMatrix_single (x : X) (p q : n) (c : LocallyConstant X (ZMod 2)) :
    evalMatrix x (single p q c) = single p q (c x) := by
  ext p' q'
  rw [evalMatrix_apply, single_apply, single_apply]
  split_ifs <;> rfl

theorem evalMatrix_const (x : X) (ε : Matrix n n (ZMod 2)) :
    evalMatrix x
      ((LocallyConstant.constRingHom : ZMod 2 →+* LocallyConstant X (ZMod 2)).mapMatrix ε) = ε := by
  ext p q
  rfl

/-- `ε ↦ w ε + (1 - w) I`. -/
def cutFun (w : LocallyConstant X (ZMod 2)) (ε : Matrix n n (ZMod 2)) :
    Matrix n n (LocallyConstant X (ZMod 2)) :=
  (LocallyConstant.constRingHom : ZMod 2 →+* LocallyConstant X (ZMod 2)).mapMatrix ε *
      diagonal (fun _ ↦ w) + diagonal (fun _ ↦ 1 - w)

theorem evalMatrix_cutFun (w : LocallyConstant X (ZMod 2)) (ε : Matrix n n (ZMod 2)) (x : X) :
    evalMatrix x (cutFun w ε) = if w x = 1 then ε else 1 := by
  rw [cutFun, map_add, map_mul, evalMatrix_const, evalMatrix_diagonal, evalMatrix_diagonal]
  have hsub : (1 - w) x = 1 - w x := rfl
  dsimp only
  rw [hsub]
  rcases zmodTwo_eq_zero_or_eq_one (w x) with h | h
  · rw [h, if_neg (show ¬(0 : ZMod 2) = 1 by decide), sub_zero, diagonal_zero, mul_zero,
      zero_add, diagonal_one]
  · rw [h, if_pos rfl, sub_self, diagonal_zero, add_zero, diagonal_one, mul_one]

/-- The cut map `ε ↦ w ε + (1 - w) I` as a monoid map. -/
def cutHom (w : LocallyConstant X (ZMod 2)) :
    Matrix n n (ZMod 2) →* Matrix n n (LocallyConstant X (ZMod 2)) where
  toFun := cutFun w
  map_one' := evalMatrix_ext fun x ↦ by
    rw [evalMatrix_cutFun, map_one]
    split_ifs <;> rfl
  map_mul' ε δ := evalMatrix_ext fun x ↦ by
    rw [map_mul, evalMatrix_cutFun, evalMatrix_cutFun, evalMatrix_cutFun]
    split_ifs <;> simp

theorem evalMatrix_cutHom (w : LocallyConstant X (ZMod 2)) (ε : Matrix n n (ZMod 2)) (x : X) :
    evalMatrix x (cutHom w ε) = if w x = 1 then ε else 1 :=
  evalMatrix_cutFun w ε x

theorem cutHom_one_add_single (w : LocallyConstant X (ZMod 2)) (P Q : n) (c : ZMod 2) :
    cutHom w (1 + single P Q c) = 1 + single P Q (LocallyConstant.const X c * w) := by
  apply evalMatrix_ext
  intro x
  rw [evalMatrix_cutHom, map_add, map_one, evalMatrix_single, LocallyConstant.mul_apply]
  rcases zmodTwo_eq_zero_or_eq_one (w x) with h | h
  · rw [h, if_neg (show ¬(0 : ZMod 2) = 1 by decide), mul_zero, single_zero, add_zero]
  · rw [h, if_pos rfl, mul_one]
    rfl

end Eval

namespace TowerSetting

variable {R : Type*} [Ring R] (S : TowerSetting (LocallyConstant X (ZMod 2)) R)
  {ι : Type*} [Fintype ι] [DecidableEq ι] {e : LocallyConstant X (ZMod 2)}
  (he : IsIdempotentElem e) {m : ℕ} (hret : S.LongReturn e (2 * m))

/-- **The copy `ι_W` of `GL_(ι × [-m,m])(𝔽₂)` inside `GL_ι(R)`**: equal to `ε` on the
tower over `W = {w = 1}` and to `1` elsewhere. -/
def cornerCopy (w : LocallyConstant X (ZMod 2)) :
    (Matrix (ι × Level m) (ι × Level m) (ZMod 2))ˣ →* (Matrix ι ι R)ˣ :=
  Units.map ((S.cornerHom he hret).comp (cutHom w))

theorem val_cornerCopy (w : LocallyConstant X (ZMod 2))
    (ε : (Matrix (ι × Level m) (ι × Level m) (ZMod 2))ˣ) :
    (S.cornerCopy he hret w ε : Matrix ι ι R) = S.cornerHom he hret (cutHom w ε) :=
  rfl

/-- **Case (c), localisation.** A nontrivial `k` whose entries and inverse entries lie in
`1 + B_m(U)` has a non-identity value `κ₀` on some clopen `W ⊆ U`, and commutators with the
copy `ι_W` are computed in `GL_d(𝔽₂)`. -/
theorem exists_corner_commutator (hinj : Function.Injective S.coeff) (k : (Matrix ι ι R)ˣ)
    (hk : k ≠ 1) (h1 : ∀ p q, ((k : Matrix ι ι R) - 1) p q ∈ S.tower e m)
    (h2 : ∀ p q, ((↑k⁻¹ : Matrix ι ι R) - 1) p q ∈ S.tower e m) :
    ∃ (w : LocallyConstant X (ZMod 2)) (κ₀ : (Matrix (ι × Level m) (ι × Level m) (ZMod 2))ˣ),
      w * e = w ∧ w ≠ 0 ∧ κ₀ ≠ 1 ∧
        ∀ ε, ⁅k, S.cornerCopy he hret w ε⁆ = S.cornerCopy he hret w ⁅κ₀, ε⁆ := by
  classical
  obtain ⟨κ, hκ⟩ := S.exists_cornerHom_eq he hret h1
  obtain ⟨κ', hκ'⟩ := S.exists_cornerHom_eq he hret h2
  have hU : ∀ x, e x = 1 →
      evalMatrix x κ * evalMatrix x κ' = 1 ∧ evalMatrix x κ' * evalMatrix x κ = 1 := by
    have hl : S.cornerHom he hret (κ * κ') = S.cornerHom he hret 1 := by
      rw [map_mul, map_one, hκ, hκ', Units.mul_inv]
    have hr : S.cornerHom he hret (κ' * κ) = S.cornerHom he hret 1 := by
      rw [map_mul, map_one, hκ, hκ', Units.inv_mul]
    rw [S.cornerHom_eq_iff he hret hinj, forall_mul_eq_mul_iff] at hl hr
    intro x hx
    exact ⟨by rw [← map_mul, hl x hx, map_one], by rw [← map_mul, hr x hx, map_one]⟩
  have hx0 : ∃ x0, e x0 = 1 ∧ evalMatrix x0 κ ≠ 1 := by
    by_contra hcon
    apply hk
    have hall : ∀ x, e x = 1 → evalMatrix x κ = evalMatrix x 1 := fun x hx ↦ by
      rw [map_one]
      by_contra hne
      exact hcon ⟨x, hx, hne⟩
    have hcorner := (S.cornerHom_eq_iff he hret hinj κ 1).mpr ((forall_mul_eq_mul_iff κ 1).mpr hall)
    rw [hκ, map_one] at hcorner
    exact Units.ext hcorner
  obtain ⟨x0, hex0, hκx0⟩ := hx0
  obtain ⟨hl0, hr0⟩ := hU x0 hex0
  let κ₀ : (Matrix (ι × Level m) (ι × Level m) (ZMod 2))ˣ :=
    ⟨evalMatrix x0 κ, evalMatrix x0 κ', hl0, hr0⟩
  let W : Set X := {x | e x = 1} ∩
    ({x | evalMatrix x κ = evalMatrix x0 κ} ∩ {x | evalMatrix x κ' = evalMatrix x0 κ'})
  have hW : IsClopen W := (e.isLocallyConstant.isClopen_fiber 1).inter
    ((isClopen_evalMatrix_eq κ _).inter (isClopen_evalMatrix_eq κ' _))
  let w : LocallyConstant X (ZMod 2) := LocallyConstant.charFn (ZMod 2) hW
  have hw1 : ∀ x, w x = 1 ↔ x ∈ W := fun x ↦ LocallyConstant.charFn_eq_one (ZMod 2) x hW
  refine ⟨w, κ₀, ?_, ?_, ?_, ?_⟩
  · ext x
    rw [LocallyConstant.mul_apply]
    rcases zmodTwo_eq_zero_or_eq_one (w x) with h | h
    · rw [h, zero_mul]
    · rw [h, one_mul, ((hw1 x).mp h).1]
  · intro hw0
    have hx0W := (hw1 x0).mpr ⟨hex0, rfl, rfl⟩
    rw [hw0] at hx0W
    simp at hx0W
  · intro h
    exact hκx0 (congrArg Units.val h)
  · intro ε
    apply Units.ext
    rw [commutatorElement_def, commutatorElement_def, ← map_inv]
    simp only [Units.val_mul, val_cornerCopy]
    rw [← hκ, ← hκ', ← map_mul, ← map_mul, ← map_mul, S.cornerHom_eq_iff he hret hinj,
      forall_mul_eq_mul_iff]
    intro x hx
    simp only [map_mul, evalMatrix_cutHom]
    by_cases hxW : w x = 1
    · obtain ⟨_, hxκ, hxκ'⟩ := (hw1 x).mp hxW
      simp only [if_pos hxW]
      rw [hxκ, hxκ']
      rfl
    · simp only [if_neg hxW, mul_one]
      exact (hU x hx).1

end TowerSetting
