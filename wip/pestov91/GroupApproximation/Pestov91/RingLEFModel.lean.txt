/-!
# Pestov 9.1: periodic models of a skew Laurent ring

Sections 3.1–3.2 of
`research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`, with the radius
condition of note N2 in `research/artifacts/ex-pestov91-second-review-2026-09-13.md`.

Let `R` be a ring in normal form `r = Σ_j C(a_j) v_j` over a coefficient ring `A`, with
`v_i C(b) = C(σ_i b) v_i`.  Given evaluations `E_n : A →+* 𝔽₂` indexed by `n ∈ ℤ/N`, put
`D(a) = diag(E_n(a))_n` and let `P^j` be the cyclic shift `δ_b ↦ δ_(b+j)`.  The model

  `ψ(Σ_j C(a_j) v_j) = Σ_j D(a_j) P^j`

is additive and unital.  It is multiplicative on `r, s` once `E_n(σ_i b) = E_(n-i)(b)` for the
exponents `i` of `r` and the coefficients `b` of `s`: this is `P^i D(b) = D(σ_i b) P^i`, exact at
every position because the index set is `ℤ/N` itself.  It is injective on a finite set whose
exponents lie in `[-w, w]` with `2w < N`, once the evaluations separate the coefficients: the
terms `D(a_j) P^j` sit on the disjoint entries `(n, n - j)`.

Every hypothesis is an explicit argument; the subshift supplies them.
-/

namespace GroupApproximation.Pestov91

section SkewModel

variable {A R M : Type*} [Ring A] [Ring R] [Ring M]

/-- The model `ψ(Σ_j C(a_j) v_j) = Σ_j D(a_j) p_j`, read off the normal form `coeff`. -/
def skewModel (coeff : R →+ (ℤ →₀ A)) (D : A →+* M) (p : ℤ → M) : R →+ M :=
  (Finsupp.liftAddHom fun j ↦ (AddMonoidHom.mulRight (p j)).comp D.toAddMonoidHom).comp coeff

theorem skewModel_apply (coeff : R →+ (ℤ →₀ A)) (D : A →+* M) (p : ℤ → M) (r : R) :
    skewModel coeff D p r = (coeff r).sum fun j a ↦ D a * p j := by
  rw [skewModel, AddMonoidHom.comp_apply, Finsupp.liftAddHom_apply]
  rfl

theorem skewModel_eq_of_coeff_eq_single (coeff : R →+ (ℤ →₀ A)) (D : A →+* M) (p : ℤ → M)
    {r : R} {j : ℤ} {a : A} (hr : coeff r = Finsupp.single j a) :
    skewModel coeff D p r = D a * p j := by
  rw [skewModel_apply, hr, Finsupp.sum_single_index]
  simp

theorem skewModel_one (coeff : R →+ (ℤ →₀ A)) (C : A →+* R) (v : ℤ → R) (D : A →+* M)
    (p : ℤ → M) (hv0 : v 0 = 1) (hp0 : p 0 = 1)
    (hmono : ∀ a j, coeff (C a * v j) = Finsupp.single j a) :
    skewModel coeff D p 1 = 1 := by
  have h1 : coeff 1 = Finsupp.single 0 1 := by
    simpa only [map_one, hv0, one_mul] using hmono 1 0
  rw [skewModel_eq_of_coeff_eq_single coeff D p h1, map_one, hp0, one_mul]

/-- **Multiplicativity** (Lemma 5.2(b) of the write-up): `ψ(rs) = ψ(r)ψ(s)` as soon as
`p_i D(b) = D(σ_i b) p_i` for the exponents `i` of `r` and the coefficients `b` of `s`. -/
theorem skewModel_mul (coeff : R →+ (ℤ →₀ A)) (C : A →+* R) (v : ℤ → R) (σ : ℤ → A →+* A)
    (D : A →+* M) (p : ℤ → M)
    (hv : ∀ i j, v (i + j) = v i * v j) (hp : ∀ i j, p (i + j) = p i * p j)
    (hmono : ∀ a j, coeff (C a * v j) = Finsupp.single j a)
    (hrepr : ∀ r, (coeff r).sum (fun j a ↦ C a * v j) = r)
    (hcomm : ∀ i b, v i * C b = C (σ i b) * v i) {r s : R}
    (hconj : ∀ i ∈ (coeff r).support, ∀ j ∈ (coeff s).support,
      p i * D (coeff s j) = D (σ i (coeff s j)) * p i) :
    skewModel coeff D p (r * s) = skewModel coeff D p r * skewModel coeff D p s := by
  have hprod : r * s = (coeff r).sum fun i a ↦ (coeff s).sum fun j b ↦
      C (a * σ i b) * v (i + j) := by
    conv_lhs => rw [← hrepr r, ← hrepr s]
    rw [Finsupp.sum_mul]
    refine Finsupp.sum_congr fun i _ ↦ ?_
    rw [Finsupp.mul_sum]
    refine Finsupp.sum_congr fun j _ ↦ ?_
    calc C (coeff r i) * v i * (C (coeff s j) * v j)
        = C (coeff r i) * (v i * C (coeff s j)) * v j := by simp only [mul_assoc]
      _ = C (coeff r i) * (C (σ i (coeff s j)) * v i) * v j := by rw [hcomm]
      _ = C (coeff r i * σ i (coeff s j)) * v (i + j) := by
        rw [map_mul, hv]
        simp only [mul_assoc]
  rw [hprod, map_finsuppSum, skewModel_apply, skewModel_apply, Finsupp.sum_mul]
  refine Finsupp.sum_congr fun i hi ↦ ?_
  rw [map_finsuppSum, Finsupp.mul_sum]
  refine Finsupp.sum_congr fun j hj ↦ ?_
  rw [skewModel_eq_of_coeff_eq_single coeff D p (hmono _ _), map_mul, hp]
  calc D (coeff r i) * D (σ i (coeff s j)) * (p i * p j)
      = D (coeff r i) * (D (σ i (coeff s j)) * p i) * p j := by simp only [mul_assoc]
    _ = D (coeff r i) * (p i * D (coeff s j)) * p j := by rw [hconj i hi j hj]
    _ = D (coeff r i) * p i * (D (coeff s j) * p j) := by simp only [mul_assoc]

end SkewModel

section Periodic

variable {N : ℕ}

/-- Integers in `[-w, w]` stay distinct modulo `N > 2w`. -/
theorem intCast_zmod_eq_iff_of_abs_le {w : ℕ} (hN : 2 * w < N) {i j : ℤ}
    (hi : |i| ≤ w) (hj : |j| ≤ w) : (i : ZMod N) = (j : ZMod N) ↔ i = j := by
  refine ⟨fun h ↦ ?_, fun h ↦ by rw [h]⟩
  obtain ⟨hi₁, hi₂⟩ := abs_le.mp hi
  obtain ⟨hj₁, hj₂⟩ := abs_le.mp hj
  have hlt : |j - i| < (N : ℤ) := abs_lt.mpr ⟨by omega, by omega⟩
  have h0 := Int.eq_zero_of_abs_lt_dvd ((ZMod.intCast_eq_intCast_iff_dvd_sub i j N).mp h) hlt
  omega

variable (N) in
/-- The cyclic shift `P^j : δ_b ↦ δ_(b+j)` on `𝔽₂^(ℤ/N)`. -/
def shiftMatrix (j : ℤ) : Matrix (ZMod N) (ZMod N) (ZMod 2) :=
  Matrix.of fun a b ↦ if a = b + (j : ZMod N) then 1 else 0

theorem shiftMatrix_apply (j : ℤ) (a b : ZMod N) :
    shiftMatrix N j a b = if a = b + (j : ZMod N) then 1 else 0 := rfl

theorem shiftMatrix_zero : shiftMatrix N 0 = 1 := by
  ext a b
  simp [shiftMatrix_apply, Matrix.one_apply]

theorem shiftMatrix_add [NeZero N] (i j : ℤ) :
    shiftMatrix N (i + j) = shiftMatrix N i * shiftMatrix N j := by
  ext a b
  rw [Matrix.mul_apply, Finset.sum_eq_single (b + (j : ZMod N))]
  · simp only [shiftMatrix_apply, if_pos rfl, mul_one, Int.cast_add]
    rw [← add_assoc, add_right_comm]
  · intro x _ hx
    rw [shiftMatrix_apply j x b, if_neg hx, mul_zero]
  · intro h
    exact absurd (Finset.mem_univ _) h

/-- Conjugating a diagonal matrix by the shift: `P^j diag(d) = diag(d(· - j)) P^j`. -/
theorem shiftMatrix_mul_diagonal [NeZero N] (j : ℤ) (d : ZMod N → ZMod 2) :
    shiftMatrix N j * Matrix.diagonal d =
      Matrix.diagonal (fun a ↦ d (a - j)) * shiftMatrix N j := by
  ext a b
  rw [Matrix.mul_diagonal, Matrix.diagonal_mul, shiftMatrix_apply]
  split_ifs with h
  · rw [h, add_sub_cancel_right, one_mul, mul_one]
  · rw [zero_mul, mul_zero]

variable {A R : Type*} [Ring A] [Ring R]

/-- The diagonal model `D(a) = diag(E_n(a))_n` of a family of evaluations. -/
def diagModel [NeZero N] (E : ZMod N → A →+* ZMod 2) :
    A →+* Matrix (ZMod N) (ZMod N) (ZMod 2) where
  toFun a := Matrix.diagonal fun n ↦ E n a
  map_one' := by simp
  map_mul' a b := by simp
  map_zero' := by simp
  map_add' a b := by simp

theorem diagModel_apply [NeZero N] (E : ZMod N → A →+* ZMod 2) (a : A) :
    diagModel E a = Matrix.diagonal fun n ↦ E n a := rfl

/-- The entry `(n, n - j)` of the model reads the coefficient of `v_j` at position `n`. -/
theorem skewModel_diag_shift_apply [NeZero N] (coeff : R →+ (ℤ →₀ A))
    (E : ZMod N → A →+* ZMod 2) {w : ℕ} (hN : 2 * w < N) {r : R}
    (hsupp : ∀ j ∈ (coeff r).support, |j| ≤ w) {j : ℤ} (hj : |j| ≤ w) (n : ZMod N) :
    skewModel coeff (diagModel E) (shiftMatrix N) r n (n - j) = E n (coeff r j) := by
  have hterm : ∀ x ∈ (coeff r).support, x ≠ j →
      (diagModel E (coeff r x) * shiftMatrix N x) n (n - j) = 0 := by
    intro x hx hxj
    rw [diagModel_apply, Matrix.diagonal_mul, shiftMatrix_apply, if_neg, mul_zero]
    intro h
    exact hxj ((intCast_zmod_eq_iff_of_abs_le hN (hsupp x hx) hj).1 (by linear_combination -h))
  rw [skewModel_apply, Finsupp.sum, Matrix.sum_apply]
  by_cases hmem : j ∈ (coeff r).support
  · rw [Finset.sum_eq_single_of_mem j hmem hterm, diagModel_apply, Matrix.diagonal_mul,
      shiftMatrix_apply, if_pos (sub_add_cancel n _).symm, mul_one]
  · rw [Finsupp.notMem_support_iff.1 hmem, map_zero]
    exact Finset.sum_eq_zero fun x hx ↦ hterm x hx fun hxj ↦ hmem (hxj ▸ hx)

/-- **Injectivity** (Lemma 5.2(c)): distinct exponents give disjoint entries. -/
theorem skewModel_diag_shift_injOn [NeZero N] (coeff : R →+ (ℤ →₀ A)) (C : A →+* R)
    (v : ℤ → R) (hrepr : ∀ r, (coeff r).sum (fun j a ↦ C a * v j) = r)
    (E : ZMod N → A →+* ZMod 2) (F : Finset R) {w : ℕ} (hN : 2 * w < N)
    (hsupp : ∀ r ∈ F, ∀ j ∈ (coeff r).support, |j| ≤ w)
    (hsep : ∀ r ∈ F, ∀ s ∈ F, ∀ j : ℤ, |j| ≤ w →
      (∀ n, E n (coeff r j) = E n (coeff s j)) → coeff r j = coeff s j) :
    Set.InjOn (skewModel coeff (diagModel E) (shiftMatrix N)) F := by
  intro r hr s hs hrs
  have hr' := Finset.mem_coe.1 hr
  have hs' := Finset.mem_coe.1 hs
  have hc : coeff r = coeff s := by
    ext j
    by_cases hj : |j| ≤ w
    · refine hsep r hr' s hs' j hj fun n ↦ ?_
      rw [← skewModel_diag_shift_apply coeff E hN (hsupp r hr') hj n,
        ← skewModel_diag_shift_apply coeff E hN (hsupp s hs') hj n, hrs]
    · rw [Finsupp.notMem_support_iff.1 fun h ↦ hj (hsupp r hr' j h),
        Finsupp.notMem_support_iff.1 fun h ↦ hj (hsupp s hs' j h)]
  rw [← hrepr r, ← hrepr s, hc]

/-- **The periodic model is a local matrix model** (§3.2 with note N2): exponents in `[-w, w]`,
`2w < N`, exact shift identities on the coefficients, and separating evaluations. -/
theorem isLocalMatrixModel_skewModel_diag_shift [NeZero N]
    (coeff : R →+ (ℤ →₀ A)) (C : A →+* R) (v : ℤ → R) (σ : ℤ → A →+* A)
    (hv0 : v 0 = 1) (hv : ∀ i j, v (i + j) = v i * v j)
    (hmono : ∀ a j, coeff (C a * v j) = Finsupp.single j a)
    (hrepr : ∀ r, (coeff r).sum (fun j a ↦ C a * v j) = r)
    (hcomm : ∀ i b, v i * C b = C (σ i b) * v i)
    (E : ZMod N → A →+* ZMod 2) (F : Finset R) {w : ℕ} (hN : 2 * w < N)
    (hsupp : ∀ r ∈ F, ∀ j ∈ (coeff r).support, |j| ≤ w)
    (hshift : ∀ s ∈ F, ∀ j ∈ (coeff s).support, ∀ i : ℤ, |i| ≤ w →
      ∀ n, E n (σ i (coeff s j)) = E (n - i) (coeff s j))
    (hsep : ∀ r ∈ F, ∀ s ∈ F, ∀ j : ℤ, |j| ≤ w →
      (∀ n, E n (coeff r j) = E n (coeff s j)) → coeff r j = coeff s j) :
    IsLocalMatrixModel F (skewModel coeff (diagModel E) (shiftMatrix N)) where
  map_one := skewModel_one coeff C v _ _ hv0 shiftMatrix_zero hmono
  injOn := skewModel_diag_shift_injOn coeff C v hrepr E F hN hsupp hsep
  map_mul r hr s hs := skewModel_mul coeff C v σ _ _ hv shiftMatrix_add hmono hrepr hcomm
    fun i hi j hj ↦ by
      rw [diagModel_apply, diagModel_apply, shiftMatrix_mul_diagonal]
      congr 2
      funext n
      exact (hshift s hs j hj i (hsupp r hr i hi) n).symm

end Periodic
