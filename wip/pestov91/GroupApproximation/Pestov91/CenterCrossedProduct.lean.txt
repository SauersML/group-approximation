import GroupApproximation.Pestov91.Center
import GroupApproximation.Pestov91.CrossedProduct

/-!
# The centre of `LC(X, 𝔽₂) ⋊_σ ℤ` for a free homeomorphism with dense orbits

This is the σ-level instance of `Pestov91.Center` (A§1.4).  `CrossedProduct σ` is the skew
monoid algebra of `LC(X, 𝔽₂)` over `ShiftIndex σ`, and `ShiftIndex.ofInt σ n` acts by
`f ↦ f ∘ σ⁻ⁿ` (`CrossedProduct.shift_apply`, blueprint M8).  So the generic theorems apply with
`τ g := σ ^ (-n)` for `g = ofInt σ n`:

* dense orbits make every invariant coefficient `0` or `1`;
* freeness and total separation make every `g ≠ 1` move some clopen indicator.

The result `crossedProduct_central_zero_or_one` takes exactly the two dynamical inputs in the
form blueprint G1 pins them for `toeplitzShift`.
-/

namespace GroupApproximation
namespace Pestov91

variable {X : Type*} [TopologicalSpace X]

/-- The coefficient action of the crossed product, read at a point:
`(g • f) x = f (σ⁻ⁿ x)` for `g = ofInt σ n`. -/
theorem shiftIndex_smul_apply (σ : X ≃ₜ X) (g : ShiftIndex σ)
    (f : LocallyConstant X (ZMod 2)) (x : X) :
    (g • f) x = f ((σ ^ (-((ShiftIndex.ofInt σ).symm g))) x) := by
  obtain ⟨n, rfl⟩ := (ShiftIndex.ofInt σ).surjective g
  rw [CrossedProduct.ofInt_smul, CrossedProduct.shift_apply, Equiv.symm_apply_apply]

/-- **`Z(LC(X, 𝔽₂) ⋊_σ ℤ) = 𝔽₂`.**  Let `X` be totally separated, let every nonzero power of
`σ` move some point, and let every orbit of `σ` be dense.  Then every central element of the
crossed product is `0` or `1`. -/
theorem crossedProduct_central_zero_or_one [TotallySeparatedSpace X] (σ : X ≃ₜ X)
    (hfree : ∀ j : ℤ, j ≠ 0 → ∃ y : X, (σ ^ (-j)) y ≠ y)
    (hdense : ∀ y : X, Dense (Set.range fun j : ℤ ↦ (σ ^ (-j)) y)) :
    ∀ c : CrossedProduct σ, (∀ r : CrossedProduct σ, r * c = c * r) → c = 0 ∨ c = 1 := by
  refine skewMonoidAlgebra_central_zero_or_one
    (locallyConstant_invariant_zero_or_one
      (fun g ↦ ⇑(σ ^ (-((ShiftIndex.ofInt σ).symm g)))) (shiftIndex_smul_apply σ) ?_)
    (locallyConstant_exists_smul_ne
      (fun g ↦ ⇑(σ ^ (-((ShiftIndex.ofInt σ).symm g)))) (shiftIndex_smul_apply σ) ?_)
  · intro y
    have h : (Set.range fun g : ShiftIndex σ ↦ (σ ^ (-((ShiftIndex.ofInt σ).symm g))) y) =
        Set.range fun j : ℤ ↦ (σ ^ (-j)) y :=
      (ShiftIndex.ofInt σ).symm.surjective.range_comp (fun j : ℤ ↦ (σ ^ (-j)) y)
    exact (congrArg Dense h).mpr (hdense y)
  · intro g hg
    refine hfree _ fun h0 ↦ hg ?_
    rw [← (ShiftIndex.ofInt σ).apply_symm_apply g, h0, ShiftIndex.ofInt_zero]

end Pestov91
end GroupApproximation
