/-!
# Root detection in `EL_3(R)`: cases (c) and (d)

§4.2 of `research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`, with
`q = 2`.  Cases (a) and (b) produce `k ∈ N`, `k ≠ 1`, which on the tower over a clopen set `U`
is a locally constant `κ : U → GL_d(𝔽₂)`.  Pick a clopen `W ⊆ U` on which `κ = κ₀`.

* **Case (c)** (`κ₀` not scalar).  The copy `ι_W` of `GL_d(𝔽₂)` lies in `G`
  (`towerUnits_mem`), and `N_W = ι_W⁻¹(N)` is normal in `GL_d(𝔽₂)`.  It contains every
  `[κ₀, δ]` by the N1 identity (`towerUnits_commutator`).  `GL_d(𝔽₂)` is simple
  (`FinitaryLinear.isSimpleGroup_units_matrix_zmodTwo`) and has trivial centre
  (`center_units_zmodTwo_eq_bot`), so `κ₀ ≠ 1` forces `N_W = GL_d(𝔽₂)`.  In particular
  `e_ij(χ_(T^a W)) = ι_W(1 + e_((i,a),(j,a))) ∈ N` (`elGen_mem_of_corner_ne_one`).  Over `𝔽₂`
  the perfectness and `PSL`-simplicity steps of the printed proof are replaced by
  simplicity of `GL_d(𝔽₂) = EL_d(𝔽₂)`, which is already on main.
* **Case (d)** (`κ₀ = λ I`, `λ ≠ 1`).  Over `𝔽₂` there is no such `λ`, because `𝔽₂ˣ = 1`
  (`units_zmodTwo_eq_one`).  Every `κ₀ ≠ 1` is non-scalar and falls under case (c).  The
  ring-level computation of the printed case (d) is still recorded, over any ring, as
  `scalarUnit_commutator`: `[c I, e_ij(r)] = e_ij(c r c⁻¹ - r)`.
-/

namespace GroupApproximation
namespace Pestov91

open Matrix
open scoped commutatorElement

/-- A normal subgroup contains the commutator of any of its elements with anything. -/
theorem commutatorElement_mem_of_mem_left {G : Type*} [Group G] (N : Subgroup G) [N.Normal]
    {x : G} (hx : x ∈ N) (y : G) : ⁅x, y⁆ ∈ N := by
  rw [commutatorElement_def]
  have h1 : y * x⁻¹ * y⁻¹ ∈ N := (inferInstance : N.Normal).conj_mem _ (N.inv_mem hx) y
  simpa only [mul_assoc] using N.mul_mem hx h1

/-- **`GL_κ(𝔽₂)` has trivial centre** for `#κ ≥ 3`: it is simple and not abelian. -/
theorem center_units_zmodTwo_eq_bot {κ : Type*} [Fintype κ] [DecidableEq κ]
    (hcard : 3 ≤ Fintype.card κ) : Subgroup.center (Matrix κ κ (ZMod 2))ˣ = ⊥ := by
  haveI := FinitaryLinear.isSimpleGroup_units_matrix_zmodTwo κ hcard
  rcases IsSimpleGroup.eq_bot_or_eq_top_of_normal (Subgroup.center (Matrix κ κ (ZMod 2))ˣ)
      inferInstance with h | h
  · exact h
  · exfalso
    obtain ⟨x, y, z, hxy, hxz, hyz⟩ := Fintype.two_lt_card_iff.mp (by omega)
    have hmem : elementaryUnit x y hxy (1 : ZMod 2) ∈ Subgroup.center (Matrix κ κ (ZMod 2))ˣ := by
      rw [h]
      exact Subgroup.mem_top _
    have hc := Subgroup.mem_center_iff.mp hmem (elementaryUnit y z hyz 1)
    have hcomm : ⁅elementaryUnit x y hxy (1 : ZMod 2), elementaryUnit y z hyz 1⁆ = 1 :=
      commutatorElement_eq_one_iff_mul_comm.mpr hc.symm
    rw [elementaryUnit_commutator x y z hxy hyz hxz, mul_one,
      ← elementaryUnit_zero x z hxz] at hcomm
    exact absurd (elementaryUnit_injective x z hxz hcomm) (by decide)

section CaseC

variable {R : Type*} [Ring R] [Algebra (ZMod 2) R]
variable {ι n : Type*} [Fintype ι] [DecidableEq ι] [Fintype n] [DecidableEq n]
variable {E : n → n → R}

/-- **Case (c), group form.**  If the `P`-corner of `k ∈ N` is `κ₀ ≠ 1` and `P` commutes with
`k` (N1), then `N` contains the whole copy `ι_W(GL_{ι×n}(𝔽₂))`. -/
theorem towerEL_mem_of_corner_ne_one (hcard : 3 ≤ Fintype.card ι) [Nontrivial ι] [Nonempty n]
    (N : Subgroup (elementaryGroup ι R)) [N.Normal] (hE : IsMatrixUnits E)
    {k : elementaryGroup ι R} (hk : k ∈ N)
    (κ₀ : (Matrix (ι × n) (ι × n) (ZMod 2))ˣ) (hκ₀ : κ₀ ≠ 1)
    (hleft : (diagonal fun _ ↦ matrixUnitsSum E) * ((k : (Matrix ι ι R)ˣ) : Matrix ι ι R) =
      blockEmbed E (κ₀ : Matrix (ι × n) (ι × n) (ZMod 2)))
    (hright : ((k : (Matrix ι ι R)ˣ) : Matrix ι ι R) * (diagonal fun _ ↦ matrixUnitsSum E) =
      blockEmbed E (κ₀ : Matrix (ι × n) (ι × n) (ZMod 2)))
    (ε : (Matrix (ι × n) (ι × n) (ZMod 2))ˣ) : towerEL ι hE ε ∈ N := by
  have hcardP : 3 ≤ Fintype.card (ι × n) :=
    hcard.trans (Fintype.card_le_of_injective (fun i ↦ (i, Classical.arbitrary n))
      fun _ _ h ↦ congrArg Prod.fst h)
  haveI := FinitaryLinear.isSimpleGroup_units_matrix_zmodTwo (ι × n) hcardP
  have hcomm : ∀ δ, towerEL ι hE ⁅κ₀, δ⁆ = ⁅k, towerEL ι hE δ⁆ := by
    intro δ
    apply Subtype.ext
    rw [towerEL_coe, ← towerUnits_commutator hE (k : (Matrix ι ι R)ˣ) κ₀ δ hleft hright]
    rfl
  have hmemNW : ∀ δ, ⁅κ₀, δ⁆ ∈ N.comap (towerEL ι hE) := by
    intro δ
    rw [Subgroup.mem_comap, hcomm]
    exact commutatorElement_mem_of_mem_left N hk _
  rcases IsSimpleGroup.eq_bot_or_eq_top_of_normal (N.comap (towerEL ι hE))
      (Subgroup.Normal.comap inferInstance (towerEL ι hE)) with hbot | htop
  · exfalso
    have hcent : κ₀ ∈ Subgroup.center (Matrix (ι × n) (ι × n) (ZMod 2))ˣ := by
      rw [Subgroup.mem_center_iff]
      intro δ
      have h := hmemNW δ
      rw [hbot, Subgroup.mem_bot, commutatorElement_eq_one_iff_mul_comm] at h
      exact h.symm
    rw [center_units_zmodTwo_eq_bot hcardP, Subgroup.mem_bot] at hcent
    exact hκ₀ hcent
  · have hε : ε ∈ N.comap (towerEL ι hE) := by
      rw [htop]
      exact Subgroup.mem_top ε
    exact hε

/-- **Case (c), the consumed root.**  Under the hypotheses of `towerEL_mem_of_corner_ne_one`,
`N` contains `e_ij(E a a)` for all `i ≠ j` and every level `a`; for the subshift tower
`E a a = χ_(T^a W) ≠ 0`. -/
theorem elGen_mem_of_corner_ne_one (hcard : 3 ≤ Fintype.card ι) [Nonempty n]
    (N : Subgroup (elementaryGroup ι R)) [N.Normal] (hE : IsMatrixUnits E)
    {k : elementaryGroup ι R} (hk : k ∈ N)
    (κ₀ : (Matrix (ι × n) (ι × n) (ZMod 2))ˣ) (hκ₀ : κ₀ ≠ 1)
    (hleft : (diagonal fun _ ↦ matrixUnitsSum E) * ((k : (Matrix ι ι R)ˣ) : Matrix ι ι R) =
      blockEmbed E (κ₀ : Matrix (ι × n) (ι × n) (ZMod 2)))
    (hright : ((k : (Matrix ι ι R)ˣ) : Matrix ι ι R) * (diagonal fun _ ↦ matrixUnitsSum E) =
      blockEmbed E (κ₀ : Matrix (ι × n) (ι × n) (ZMod 2))) :
    ∀ (i j : ι) (hij : i ≠ j) (a : n), elGen i j hij (E a a) ∈ N := by
  haveI : Nontrivial ι := Fintype.one_lt_card_iff_nontrivial.mp (by omega)
  intro i j hij a
  have hne : ((i, a) : ι × n) ≠ (j, a) := fun h ↦ hij (congrArg Prod.fst h)
  have heq : towerEL ι hE (elementaryUnit (i, a) (j, a) hne (1 : ZMod 2)) =
      elGen i j hij (E a a) := by
    apply Subtype.ext
    apply Units.ext
    change towerEmbed E ((elementaryUnit (i, a) (j, a) hne (1 : ZMod 2) :
        (Matrix (ι × n) (ι × n) (ZMod 2))ˣ) : Matrix (ι × n) (ι × n) (ZMod 2)) =
      1 + Matrix.single i j (E a a)
    rw [towerEmbed_elementaryUnit, map_one, one_mul]
  rw [← heq]
  exact towerEL_mem_of_corner_ne_one hcard N hE hk κ₀ hκ₀ hleft hright _

end CaseC

/-- **Case (d) is vacuous over `𝔽₂`**: the only scalar unit is `1`. -/
theorem units_zmodTwo_eq_one (u : (ZMod 2)ˣ) : u = 1 := Subsingleton.elim u 1

section CaseD

variable {S ι : Type*} [Ring S] [Fintype ι] [DecidableEq ι]

/-- The scalar unit `c I`. -/
def scalarUnit (c : Sˣ) : (Matrix ι ι S)ˣ where
  val := diagonal fun _ ↦ (c : S)
  inv := diagonal fun _ ↦ ((c⁻¹ : Sˣ) : S)
  val_inv := by
    rw [Matrix.diagonal_mul_diagonal]
    simp
  inv_val := by
    rw [Matrix.diagonal_mul_diagonal]
    simp

/-- **The ring-level computation of case (d)**: `[c I, e_ij(r)] = e_ij(c r c⁻¹ - r)`.  In §4.2(d)
`c r c⁻¹ = λ₀⁻¹ r`, and the commutator is the root `e_ij((λ₀⁻¹ - 1) r)`. -/
theorem scalarUnit_commutator (c : Sˣ) (i j : ι) (h : i ≠ j) (r : S) :
    ⁅scalarUnit c, elementaryUnit i j h r⁆ =
      elementaryUnit i j h ((c : S) * r * ((c⁻¹ : Sˣ) : S) - r) := by
  apply Units.ext
  change (diagonal fun _ ↦ (c : S)) * (1 + Matrix.single i j r) *
      (diagonal fun _ ↦ ((c⁻¹ : Sˣ) : S)) * (1 - Matrix.single i j r) =
    1 + Matrix.single i j ((c : S) * r * ((c⁻¹ : Sˣ) : S) - r)
  have hDD : (diagonal fun _ : ι ↦ (c : S)) * (diagonal fun _ ↦ ((c⁻¹ : Sˣ) : S)) = 1 :=
    (scalarUnit c).val_inv
  have hsd : (diagonal fun _ : ι ↦ (c : S)) * Matrix.single i j r *
      (diagonal fun _ ↦ ((c⁻¹ : Sˣ) : S)) = Matrix.single i j ((c : S) * r * ((c⁻¹ : Sˣ) : S)) := by
    ext p q
    simp only [Matrix.mul_diagonal, Matrix.diagonal_mul, Matrix.single_apply]
    split_ifs <;> simp
  have hSS : Matrix.single i j ((c : S) * r * ((c⁻¹ : Sˣ) : S)) * Matrix.single i j r = 0 :=
    Matrix.single_mul_single_of_ne (c := (c : S) * r * ((c⁻¹ : Sˣ) : S)) i j i h.symm r
  have e1 : (diagonal fun _ : ι ↦ (c : S)) * (1 + Matrix.single i j r) *
      (diagonal fun _ ↦ ((c⁻¹ : Sˣ) : S)) =
      (diagonal fun _ : ι ↦ (c : S)) * (diagonal fun _ ↦ ((c⁻¹ : Sˣ) : S)) +
        (diagonal fun _ : ι ↦ (c : S)) * Matrix.single i j r *
          (diagonal fun _ ↦ ((c⁻¹ : Sˣ) : S)) := by
    noncomm_ring
  have e2 : (1 + Matrix.single i j ((c : S) * r * ((c⁻¹ : Sˣ) : S))) * (1 - Matrix.single i j r) =
      1 + (Matrix.single i j ((c : S) * r * ((c⁻¹ : Sˣ) : S)) - Matrix.single i j r) -
        Matrix.single i j ((c : S) * r * ((c⁻¹ : Sˣ) : S)) * Matrix.single i j r := by
    noncomm_ring
  rw [e1, hDD, hsd, matrix_single_sub, e2, hSS, sub_zero]

end CaseD
