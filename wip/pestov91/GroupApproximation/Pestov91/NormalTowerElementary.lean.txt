/-!
# Pestov 9.1: the copies `ι_W(GL_d(𝔽₂))` lie in `EL_ι(R)`

Part 2 §6.2(c) of `research/artifacts/pestov-9-1-writeup-2026-09-13-part2.md`.  Over `𝔽₂`
every invertible matrix is elementary (`FinitaryLinear.elementaryGroup_zmodTwo_eq_top`), so it
suffices to place the image of each transvection `I + c e_((p,a),(p',b))`:

* `p ≠ p'`: it is the root `e_pp'(c E_ab(W))`;
* `p = p'`, `a ≠ b`: with `x = c E_ab(W)`, `y = E_bb(W)` one has `xy = x`, `yx = 0`, `x² = 0`, and
  `[e_pp''(x), e_p''p(y)] = I + x E_pp`.

`exists_corner_copy` bundles case (c) of root detection: the copy `φ = ι_W`, a non-identity
`κ₀`, the commutator identity, `φ(GL_d(𝔽₂)) ≤ EL_ι(R)` and `φ(e_((i,0),(j,0))(1)) = e_ij(χ_W)`.
With `FinitaryLinear.isSimpleGroup_units_matrix_zmodTwo` this puts `e_ij(χ_W) ≠ 1` in every
normal subgroup of `EL_ι(R)` that contains `k`.
-/

namespace GroupApproximation.Pestov91

open Matrix

/-- The middle level `0 ∈ [-m, m]`. -/
def Level.zero (m : ℕ) : Level m :=
  ⟨0, Finset.mem_Icc.mpr ⟨by omega, by omega⟩⟩

namespace TowerSetting

section General

variable {A R : Type*} [CommRing A] [Ring R] (S : TowerSetting A R)
  {ι : Type*} [Fintype ι] [DecidableEq ι]

theorem cornerLinear_single (e : A) (m : ℕ) (i j : ι) (a b : Level m) (y : A) :
    S.cornerLinear e m (single (i, a) (j, b) y) = single i j (S.towerElem e a b y) := by
  have hsymm : (compRingEquiv ι (Level m) A).symm (single (i, a) (j, b) y) =
      single i j (single a b y) := by
    rw [RingEquiv.symm_apply_eq, compRingEquiv_apply, comp_single_single]
  ext p q
  rw [cornerLinear_apply, hsymm, single_apply, single_apply]
  split_ifs
  · exact S.towerMap_single e m a b y
  · exact map_zero _

theorem cornerHom_one_add_single {e : A} (he : IsIdempotentElem e) {m : ℕ}
    (hret : S.LongReturn e (2 * m)) (i j : ι) (a b : Level m) (y : A) :
    S.cornerHom he hret (1 + single (i, a) (j, b) y) = 1 + single i j (S.towerElem e a b y) := by
  rw [cornerHom_apply, map_add, cornerLinear_one, cornerLinear_single, ← add_assoc,
    diagonal_add, sub_add_cancel, diagonal_one]

end General

variable {X : Type*} [TopologicalSpace X] {R : Type*} [Ring R]
  (S : TowerSetting (LocallyConstant X (ZMod 2)) R) {ι : Type*} [Fintype ι] [DecidableEq ι]
  {e : LocallyConstant X (ZMod 2)} (he : IsIdempotentElem e) {m : ℕ}
  (hret : S.LongReturn e (2 * m))

theorem val_cornerCopy_elementaryUnit (w : LocallyConstant X (ZMod 2)) (i j : ι) (a b : Level m)
    (hne : (i, a) ≠ (j, b)) (c : ZMod 2) :
    (S.cornerCopy he hret w (elementaryUnit (i, a) (j, b) hne c) : Matrix ι ι R) =
      1 + single i j (S.towerElem e a b (LocallyConstant.const X c * w)) := by
  rw [val_cornerCopy]
  change S.cornerHom he hret (cutHom w (1 + single (i, a) (j, b) c)) = _
  rw [cutHom_one_add_single, S.cornerHom_one_add_single he hret]

theorem cornerCopy_elementaryUnit_mem [Nontrivial ι] {w : LocallyConstant X (ZMod 2)}
    (hw : w * e = w) (i j : ι) (a b : Level m) (hne : (i, a) ≠ (j, b)) (c : ZMod 2) :
    S.cornerCopy he hret w (elementaryUnit (i, a) (j, b) hne c) ∈ elementaryGroup ι R := by
  by_cases hij : i = j
  · subst hij
    have hab : (a : ℤ) ≠ b := fun h ↦ hne (by rw [Subtype.ext h])
    obtain ⟨i'', hi''⟩ := exists_ne i
    set x := S.towerElem e a b (LocallyConstant.const X c * w) with hx
    set y := S.towerElem e b b w with hy
    have hxx : x * x = 0 :=
      S.towerElem_mul_towerElem_of_ne hret hab.symm (Level.abs_sub_le b a) _ _
    have hyx : y * x = 0 :=
      S.towerElem_mul_towerElem_of_ne hret hab.symm (Level.abs_sub_le b a) _ _
    have hxy : x * y = x := by
      rw [hx, hy, S.towerElem_mul_towerElem_same he, mul_assoc,
        (isIdempotentElem_locallyConstant_zmodTwo w).eq]
    have hcomm : S.cornerCopy he hret w (elementaryUnit (i, a) (i, b) hne c) =
        ⁅elementaryUnit i i'' hi''.symm x, elementaryUnit i'' i hi'' y⁆ := by
      apply Units.ext
      rw [S.val_cornerCopy_elementaryUnit he hret, commutatorElement_def]
      change 1 + single i i x =
        (1 + single i i'' x) * (1 + single i'' i y) * (1 - single i i'' x) * (1 - single i'' i y)
      have hBB : single i'' i y * single i'' i y = 0 :=
        single_mul_single_of_ne (h := hi''.symm) _ _ _ _
      have hAA' : single i i'' x * single i i'' x = 0 :=
        single_mul_single_of_ne (h := hi'') _ _ _ _
      have hBA : single i'' i y * single i i'' x = 0 := by
        rw [single_mul_single_same, hyx, single_zero]
      have hAB : single i i'' x * single i'' i y = single i i x := by
        rw [single_mul_single_same, hxy]
      have hCA : single i i x * single i i'' x = 0 := by
        rw [single_mul_single_same, hxx, single_zero]
      have hCB : single i i x * single i'' i y = 0 :=
        single_mul_single_of_ne (h := hi''.symm) _ _ _ _
      have hAC : single i i'' x * single i i x = 0 :=
        single_mul_single_of_ne (h := hi'') _ _ _ _
      have hBC : single i'' i y * single i i x = 0 := by
        rw [single_mul_single_same, hyx, single_zero]
      noncomm_ring [hAA', hBB, hBA, hAB, hCA, hCB, hAC, hBC]
    rw [hcomm, commutatorElement_def]
    refine (elementaryGroup ι R).mul_mem ((elementaryGroup ι R).mul_mem
      ((elementaryGroup ι R).mul_mem (elementaryUnit_mem _ _ _ _) (elementaryUnit_mem _ _ _ _))
      ((elementaryGroup ι R).inv_mem (elementaryUnit_mem _ _ _ _)))
      ((elementaryGroup ι R).inv_mem (elementaryUnit_mem _ _ _ _))
  · have hunit : S.cornerCopy he hret w (elementaryUnit (i, a) (j, b) hne c) =
        elementaryUnit i j hij (S.towerElem e a b (LocallyConstant.const X c * w)) :=
      Units.ext (S.val_cornerCopy_elementaryUnit he hret w i j a b hne c)
    rw [hunit]
    exact elementaryUnit_mem _ _ _ _

/-- **`ι_W(GL_d(𝔽₂)) ≤ EL_ι(R)`.** -/
theorem cornerCopy_mem_elementaryGroup [Nontrivial ι] {w : LocallyConstant X (ZMod 2)}
    (hw : w * e = w) (ε : (Matrix (ι × Level m) (ι × Level m) (ZMod 2))ˣ) :
    S.cornerCopy he hret w ε ∈ elementaryGroup ι R := by
  have hε : ε ∈ elementaryGroup (ι × Level m) (ZMod 2) := by
    rw [FinitaryLinear.elementaryGroup_zmodTwo_eq_top]
    exact Subgroup.mem_top ε
  have hmap : (elementaryGroup (ι × Level m) (ZMod 2)).map (S.cornerCopy he hret w) ≤
      elementaryGroup ι R := by
    rw [elementaryGroup, Subgroup.map_le_iff_le_comap, Subgroup.closure_le]
    rintro _ ⟨⟨i, a⟩, ⟨j, b⟩, hne, c, rfl⟩
    exact S.cornerCopy_elementaryUnit_mem he hret hw i j a b hne c
  exact hmap ⟨ε, hε, rfl⟩

/-- **Case (c) of root detection, bundled.** -/
theorem exists_corner_copy [Nontrivial ι] (hinj : Function.Injective S.coeff)
    (k : (Matrix ι ι R)ˣ) (hk : k ≠ 1)
    (h1 : ∀ p q, ((k : Matrix ι ι R) - 1) p q ∈ S.tower e m)
    (h2 : ∀ p q, ((↑k⁻¹ : Matrix ι ι R) - 1) p q ∈ S.tower e m) :
    ∃ (w : LocallyConstant X (ZMod 2))
      (φ : (Matrix (ι × Level m) (ι × Level m) (ZMod 2))ˣ →* (Matrix ι ι R)ˣ)
      (κ₀ : (Matrix (ι × Level m) (ι × Level m) (ZMod 2))ˣ),
      S.coeff w ≠ 0 ∧ κ₀ ≠ 1 ∧ (∀ ε, ⁅k, φ ε⁆ = φ ⁅κ₀, ε⁆) ∧
        φ.range ≤ elementaryGroup ι R ∧
        ∀ (i j : ι) (h : i ≠ j),
          φ (elementaryUnit (i, Level.zero m) (j, Level.zero m)
              (fun hij ↦ h (congrArg Prod.fst hij)) 1) = elementaryUnit i j h (S.coeff w) := by
  obtain ⟨w, κ₀, hwe, hw0, hκ₀, hcomm⟩ := S.exists_corner_commutator he hret hinj k hk h1 h2
  refine ⟨w, S.cornerCopy he hret w, κ₀, (map_ne_zero_iff _ hinj).mpr hw0, hκ₀, hcomm, ?_, ?_⟩
  · rintro _ ⟨ε, rfl⟩
    exact S.cornerCopy_mem_elementaryGroup he hret hwe ε
  · intro i j h
    apply Units.ext
    rw [S.val_cornerCopy_elementaryUnit he hret]
    have hc : LocallyConstant.const X (1 : ZMod 2) * w * e = w := by
      rw [mul_assoc, hwe]
      exact one_mul w
    change 1 + single i j (S.coeff ((S.α ^ (0 : ℤ)) (LocallyConstant.const X (1 : ZMod 2) * w * e)) *
        ((S.u ^ ((0 : ℤ) - 0) : Rˣ) : R)) = 1 + single i j (S.coeff w)
    rw [hc, sub_self, zpow_zero, zpow_zero, RingAut.one_apply, Units.val_one, mul_one]

end TowerSetting
