/-!
# Pestov 9.1: `LC(X, 𝔽₂) ⋊ ℤ` has local matrix models, so `EL₃` of it is LEF

Section 3 of `research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`.

Let `X` be a subshift with periodic models, and let `ℤ` act on `A = LC(X, 𝔽₂)` by
`ofAdd i • f = f ∘ T^(-i)`, that is `(ofAdd i • f) y = f ((-i) +ᵥ y)` (the artifact's
`α(f) = f ∘ T⁻¹`).  Every finite `F ⊆ R = A ⋊ ℤ` has a local matrix model:

* `w` bounds the exponents of `F` and `ρ` the coordinate radii of its coefficients;
* a periodic chart `p : ℤ/N → X` at radius `k = ρ + w` gives the evaluations `E_n(f) = f(p_n)`;
* gluing gives `E_n(ofAdd i • f) = E_(n-i)(f)` for `|i| ≤ w` (note N2: `ρ + |i| ≤ k`);
* covering at radius `k ≥ ρ` makes the evaluations separate the coefficients;
* `N ≥ 2w + 1` separates the exponents.

So `isLocalMatrixModel_skewModel_diag_shift` applies, and `EL₃(R)` is LEF.  The action and its
coordinate formula `hact` are hypotheses.
-/

namespace GroupApproximation.Pestov91

open SkewMonoidAlgebra (single)
open Multiplicative (ofAdd toAdd)

section Coeff

variable {A : Type*} [Ring A] [MulSemiringAction (Multiplicative ℤ) A]

/-- The normal form of `A ⋊ ℤ` as a finitely supported function on `ℤ`. -/
def coeffHom : SkewMonoidAlgebra A (Multiplicative ℤ) →+ (ℤ →₀ A) where
  toFun r := Finsupp.equivMapDomain toAdd r.coeff
  map_zero' := Finsupp.ext fun _ ↦ rfl
  map_add' _ _ := Finsupp.ext fun _ ↦ rfl

theorem coeffHom_apply (r : SkewMonoidAlgebra A (Multiplicative ℤ)) (n : ℤ) :
    coeffHom r n = r.coeff (ofAdd n) := rfl

theorem coeffHom_C_mul_unit_zpow (a : A) (j : ℤ) :
    coeffHom (CrossedProduct.C a * ↑(CrossedProduct.unit ^ j)) = Finsupp.single j a := by
  ext n
  rw [CrossedProduct.val_unit_zpow, CrossedProduct.C_mul_single, mul_one, coeffHom_apply,
    SkewMonoidAlgebra.coeff_single]
  by_cases h : j = n
  · subst h
    simp
  · rw [Finsupp.single_eq_of_ne fun h' ↦ h (ofAdd.injective h').symm,
      Finsupp.single_eq_of_ne (Ne.symm h)]

theorem coeffHom_sum (r : SkewMonoidAlgebra A (Multiplicative ℤ)) :
    (coeffHom r).sum (fun j a ↦ CrossedProduct.C a * ↑(CrossedProduct.unit ^ j)) = r := by
  simp only [CrossedProduct.val_unit_zpow, CrossedProduct.C_mul_single, mul_one]
  rw [Finsupp.sum, show (coeffHom r).support = r.support.map toAdd.toEmbedding from rfl,
    Finset.sum_map]
  exact (SkewMonoidAlgebra.sum_def' r single).symm.trans (SkewMonoidAlgebra.sum_single r)

end Coeff

section Subshift

variable {X : Subshift Bool ℤ}

/-- **Exact shift identity on a chart** (§3.2(a), note N2): for `f` of radius `ρ` and
`ρ + |i| ≤ k`, `(ofAdd i • f)(p_n) = f(p_(n-i))`, seams included. -/
theorem chart_shift_apply [MulSemiringAction (Multiplicative ℤ) (LocallyConstant X.carrier (ZMod 2))]
    (hact : ∀ (i : ℤ) (f : LocallyConstant X.carrier (ZMod 2)) (y : X.carrier),
      (ofAdd i • f) y = f ((-i) +ᵥ y))
    {k N : ℕ} {p : ZMod N → X.carrier}
    (hglue : ∀ (n : ZMod N) (i m : ℤ), |m| ≤ k → |i + m| ≤ k →
      (p (n + i) : ℤ → Bool) m = (p n : ℤ → Bool) (i + m))
    {f : LocallyConstant X.carrier (ZMod 2)} {ρ : ℕ}
    (hf : ∀ y y' : X.carrier, Agree ρ y y' → f y = f y')
    {i : ℤ} (hi : (ρ : ℤ) + |i| ≤ k) (n : ZMod N) :
    (ofAdd i • f) (p n) = f (p (n - i)) := by
  rw [hact]
  refine hf _ _ fun m hm ↦ ?_
  obtain ⟨hm₁, hm₂⟩ := abs_le.mp hm
  obtain ⟨hi₁, hi₂⟩ := abs_le.mp (show |i| ≤ (k : ℤ) - ρ by linarith)
  rw [coe_vadd, shift_apply, sub_eq_add_neg, ← Int.cast_neg]
  exact (hglue n (-i) m (abs_le.mpr ⟨by omega, by omega⟩)
    (abs_le.mpr ⟨by omega, by omega⟩)).symm

/-- **Covering separates coefficients** (§3.2(c)). -/
theorem chart_ext {k N : ℕ} {p : ZMod N → X.carrier}
    (hcover : ∀ y : X.carrier, ∃ n : ZMod N, Agree k (p n) y)
    {f g : LocallyConstant X.carrier (ZMod 2)} {ρ : ℕ} (hρk : ρ ≤ k)
    (hf : ∀ y y' : X.carrier, Agree ρ y y' → f y = f y')
    (hg : ∀ y y' : X.carrier, Agree ρ y y' → g y = g y')
    (h : ∀ n, f (p n) = g (p n)) : f = g := by
  refine LocallyConstant.ext fun y ↦ ?_
  obtain ⟨n, hn⟩ := hcover y
  rw [← hf _ _ (hn.mono hρk), ← hg _ _ (hn.mono hρk), h n]

/-- **Local matrix models of `LC(X, 𝔽₂) ⋊ ℤ`** (§3.1–3.2 with note N2). -/
theorem hasLocalMatrixModels_crossedProduct
    [MulSemiringAction (Multiplicative ℤ) (LocallyConstant X.carrier (ZMod 2))]
    (hact : ∀ (i : ℤ) (f : LocallyConstant X.carrier (ZMod 2)) (y : X.carrier),
      (ofAdd i • f) y = f ((-i) +ᵥ y))
    (hX : HasPeriodicModels X) :
    HasLocalMatrixModels
      (SkewMonoidAlgebra (LocallyConstant X.carrier (ZMod 2)) (Multiplicative ℤ)) := by
  classical
  refine hasLocalMatrixModels_of_fintype fun F ↦ ?_
  choose ρf hρf using fun f : LocallyConstant X.carrier (ZMod 2) ↦
    exists_radius_locallyConstant f
  let ρ : ℕ := F.sup fun r ↦ (coeffHom r).support.sup fun j ↦ ρf (coeffHom r j)
  let w : ℕ := F.sup fun r ↦ (coeffHom r).support.sup Int.natAbs
  obtain ⟨N, z, -, hNB, hz, hwin, hcov⟩ := hX (ρ + w) (2 * w + 1)
  haveI : NeZero N := ⟨by omega⟩
  obtain ⟨p, hglue, hcover⟩ := exists_periodicChart (X := X) hz hwin hcov
  have hsupp : ∀ r ∈ F, ∀ j ∈ (coeffHom r).support, |j| ≤ w := fun r hr j hj ↦ by
    rw [Int.abs_eq_natAbs]
    exact_mod_cast (Finset.le_sup (f := Int.natAbs) hj).trans
      (Finset.le_sup (f := fun r ↦ (coeffHom r).support.sup Int.natAbs) hr)
  have hrad : ∀ r ∈ F, ∀ j : ℤ, ∀ y y' : X.carrier, Agree ρ y y' →
      coeffHom r j y = coeffHom r j y' := by
    intro r hr j y y' hyy'
    by_cases hj : j ∈ (coeffHom r).support
    · exact hρf _ y y' (hyy'.mono ((Finset.le_sup (f := fun j ↦ ρf (coeffHom r j)) hj).trans
        (Finset.le_sup (f := fun r ↦ (coeffHom r).support.sup fun j ↦ ρf (coeffHom r j)) hr)))
    · rw [Finsupp.notMem_support_iff.1 hj]
      rfl
  refine ⟨ZMod N, inferInstance, inferInstance, _, isLocalMatrixModel_skewModel_diag_shift
    coeffHom CrossedProduct.C (fun j ↦ ↑(CrossedProduct.unit ^ j))
    (fun i ↦ MulSemiringAction.toRingHom (Multiplicative ℤ) _ (ofAdd i)) (by simp)
    (fun i j ↦ by simp [zpow_add]) coeffHom_C_mul_unit_zpow coeffHom_sum
    (fun i b ↦ CrossedProduct.unit_zpow_mul_C i b)
    (fun n ↦ LocallyConstant.evalRingHom (p n)) F (w := w) (by omega) hsupp ?_ ?_⟩
  · intro s hs j _ i hi n
    exact chart_shift_apply hact hglue (hrad s hs j) (by push_cast; linarith) n
  · intro r hr s hs j _ h
    exact chart_ext hcover (by omega) (hrad r hr j) (hrad s hs j) h

/-- **`EL₃(LC(X, 𝔽₂) ⋊ ℤ)` is LEF** (§3.3). -/
theorem isLEF_elementaryGroup_crossedProduct
    [MulSemiringAction (Multiplicative ℤ) (LocallyConstant X.carrier (ZMod 2))]
    (hact : ∀ (i : ℤ) (f : LocallyConstant X.carrier (ZMod 2)) (y : X.carrier),
      (ofAdd i • f) y = f ((-i) +ᵥ y))
    (hX : HasPeriodicModels X) :
    IsLEF ↥(elementaryGroup (Fin 3)
      (SkewMonoidAlgebra (LocallyConstant X.carrier (ZMod 2)) (Multiplicative ℤ))) :=
  isLEF_elementaryGroup_of_hasLocalMatrixModels (hasLocalMatrixModels_crossedProduct hact hX)
    (Fin 3)

end Subshift
