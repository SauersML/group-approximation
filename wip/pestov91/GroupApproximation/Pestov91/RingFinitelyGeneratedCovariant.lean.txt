/-! ### An abstract ring spanned by a covariant unit -/

section Covariant

/-- **A ring spanned by a unit covariant with the shift is finitely generated.**
Let `R` receive `LC(X, ZMod 2)` through `φ`, for a compact set `X` of Bool
sequences.  Suppose a unit `u` implements `σ j f = f ∘ τ j`, where `τ j` reads
coordinate `j + n` at position `n`, and every element of `R` is a finite sum
`∑ φ (c j) * u ^ j`.  Then `R` is generated as a ring by `u`, `u⁻¹`,
`φ [x₀ = false]` and `φ [x₀ = true]`. -/
theorem isFinitelyGeneratedRing_of_covariantShift {R : Type*} [Ring R]
    (X : Set (ℤ → Bool)) [CompactSpace X] (φ : LocallyConstant X (ZMod 2) →+* R) (u : Rˣ)
    (τ : ℤ → X → X) (hτ : ∀ (j : ℤ) (x : X) (n : ℤ), (τ j x).1 n = x.1 (j + n))
    (σ : ℤ → LocallyConstant X (ZMod 2) → LocallyConstant X (ZMod 2))
    (hσ : ∀ (j : ℤ) (f : LocallyConstant X (ZMod 2)) (x : X), σ j f x = f (τ j x))
    (hconj : ∀ (j : ℤ) (a : LocallyConstant X (ZMod 2)),
      ((u ^ j : Rˣ) : R) * φ a = φ (σ j a) * ((u ^ j : Rˣ) : R))
    (hspan : ∀ x : R, ∃ (S : Finset ℤ) (c : ℤ → LocallyConstant X (ZMod 2)),
      x = ∑ j ∈ S, φ (c j) * ((u ^ j : Rˣ) : R)) :
    IsFinitelyGeneratedRing R := by
  classical
  refine ⟨{φ (coordinateIndicator X 0 false), φ (coordinateIndicator X 0 true), (u : R),
    ((u⁻¹ : Rˣ) : R)}, ?_⟩
  set S := Subring.closure (({φ (coordinateIndicator X 0 false),
    φ (coordinateIndicator X 0 true), (u : R), ((u⁻¹ : Rˣ) : R)} : Finset R) : Set R)
  have hu : (u : R) ∈ S := Subring.subset_closure (by simp)
  have hv : ((u⁻¹ : Rˣ) : R) ∈ S := Subring.subset_closure (by simp)
  have hpow : ∀ j : ℤ, ((u ^ j : Rˣ) : R) ∈ S := by
    intro j
    induction j using Int.induction_on with
    | zero => rw [zpow_zero, Units.val_one]; exact S.one_mem
    | succ i ih => rw [zpow_add_one, Units.val_mul]; exact S.mul_mem ih hu
    | pred i ih => rw [zpow_sub_one, Units.val_mul]; exact S.mul_mem ih hv
  have he : ∀ (k : ℤ) (b : Bool), φ (coordinateIndicator X k b) ∈ S := by
    intro k b
    have hσk : σ k (coordinateIndicator X 0 b) = coordinateIndicator X k b :=
      LocallyConstant.ext fun x => by
        rw [hσ, coordinateIndicator_apply, coordinateIndicator_apply, hτ, add_zero]
    have h := hconj k (coordinateIndicator X 0 b)
    rw [hσk] at h
    have hk : φ (coordinateIndicator X k b) =
        ((u ^ k : Rˣ) : R) * φ (coordinateIndicator X 0 b) * ((u ^ (-k) : Rˣ) : R) := by
      rw [h, zpow_neg, mul_assoc, Units.mul_inv, mul_one]
    rw [hk]
    refine S.mul_mem (S.mul_mem (hpow k) ?_) (hpow (-k))
    cases b
    · exact Subring.subset_closure (by simp)
    · exact Subring.subset_closure (by simp)
  have hφ : ∀ a, φ a ∈ S := by
    have hle : Subring.closure
        (Set.range fun p : ℤ × Bool => coordinateIndicator X p.1 p.2) ≤ S.comap φ :=
      Subring.closure_le.mpr (by
        rintro _ ⟨⟨k, b⟩, rfl⟩
        exact he k b)
    rw [closure_range_coordinateIndicator X] at hle
    exact fun a => hle (Subring.mem_top a)
  refine (Subring.eq_top_iff' S).mpr fun x => ?_
  obtain ⟨T, c, rfl⟩ := hspan x
  exact Subring.sum_mem _ fun j _ => S.mul_mem (hφ (c j)) (hpow j)

end Covariant

/-! ### G5: the witness ring (after S1 `WitnessRing.lean` lands) -/

/-- **The Pestov 9.1 witness ring is finitely generated** (blueprint G5).  Under
the S1 action `(ofAdd j • f) y = f ((-j) +ᵥ y)`, the cylinder `[xₖ = b]` is the
translate of `[x₀ = b]` by `-k`, so `isFinitelyGeneratedRing_crossedProduct`
applies. -/
theorem isFinitelyGeneratedRing_witnessRing : IsFinitelyGeneratedRing WitnessRing :=
  isFinitelyGeneratedRing_crossedProduct toeplitzSubshift.carrier fun k b =>
    ⟨-k, LocallyConstant.ext fun y => by
      simp [ofAdd_smul_witnessCoeff_apply, coordinateIndicator_apply]⟩
