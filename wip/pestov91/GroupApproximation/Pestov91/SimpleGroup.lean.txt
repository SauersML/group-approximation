import GroupApproximation.Pestov91.SplitSimplicity

/-!
# Simplicity of `EL_ι(R)` from root detection over a simple ring

The simplicity glue of the Pestov 9.1 witness, generic over the coefficient ring.  Let `N` be a
nontrivial normal subgroup of `EL_ι(R)` with `|ι| ≥ 3`.

1. **Root detection**, a hypothesis here, supplies a nonzero `x` with `x_{ij}(x) ∈ N` for one pair
   `i ≠ j`.
2. **Spreading.**  `Pestov91.elGen_mem_of_single_root` puts `x_{kl}(x)` in `N` for every pair
   `k ≠ l`, so `x` lies in the level ideal `normalLevel hcard N`.
3. **Simple ring.**  The level ideal is a nonzero two-sided ideal, hence `⊤`.  The relative
   elementary group at level `⊤` is all of `EL_ι(R)` (`relativeElementary_top`), and it lies in `N`
   (`relativeElementary_normalLevel_le`).  So `N = ⊤`.

No division hypothesis is used.  `ElementarySimplicity.isSimpleGroup_of_rootDetection` needs a unit
sandwich `a x b = 1` for every nonzero `x`, and that fails over a stably finite ring such as the
crossed product of the witness.  Simplicity of the ring replaces it.
`Pestov91.isSimpleGroup_elementaryGroup_of_split` runs the same argument with root detection
derived from split annihilators.  The theorem here takes root detection itself as the input, so it
serves any route that detects roots.
-/

namespace GroupApproximation
namespace Pestov91

variable {ι R : Type*} [Fintype ι] [DecidableEq ι] [Ring R]

/-- A nonzero root in a normal subgroup makes its level ideal nonzero. -/
theorem normalLevel_ne_bot_of_elGen_mem (hcard : 3 ≤ Fintype.card ι)
    (N : Subgroup ↥(elementaryGroup ι R)) [N.Normal] {i j : ι} (hij : i ≠ j) {a : R}
    (ha : a ≠ 0) (hmem : elGen i j hij a ∈ N) : normalLevel hcard N ≠ ⊥ := by
  intro hbot
  have hlevel : a ∈ normalLevel hcard N :=
    (mem_normalLevel_iff hcard N).mpr fun k l hkl => by
      simpa only [one_mul, mul_one] using elGen_mem_of_single_root hcard N hij hmem hkl 1 1
  rw [hbot, TwoSidedIdeal.mem_bot] at hlevel
  exact ha hlevel

/-- **A normal subgroup containing one nonzero root is everything**, over a simple ring with at
least three indices. -/
theorem normal_eq_top_of_elGen_mem_of_isSimpleRing [IsSimpleRing R]
    (hcard : 3 ≤ Fintype.card ι) (N : Subgroup ↥(elementaryGroup ι R)) [N.Normal]
    {i j : ι} (hij : i ≠ j) {a : R} (ha : a ≠ 0) (hmem : elGen i j hij a ∈ N) : N = ⊤ := by
  have htop : normalLevel hcard N = ⊤ :=
    (IsSimpleOrder.eq_bot_or_eq_top _).resolve_left
      (normalLevel_ne_bot_of_elGen_mem hcard N hij ha hmem)
  have hle := relativeElementary_normalLevel_le hcard N
  rw [htop, relativeElementary_top] at hle
  exact top_le_iff.mp hle

/-- **`EL_ι(R)` is simple** over a simple ring `R` with `|ι| ≥ 3`, once every nontrivial normal
subgroup contains a nonzero elementary root. -/
theorem isSimpleGroup_of_rootDetection_of_isSimpleRing [IsSimpleRing R]
    (hcard : 3 ≤ Fintype.card ι) (hnt : Nontrivial ↥(elementaryGroup ι R))
    (hdetect : ElementarySimplicity.RootDetection ι R) :
    IsSimpleGroup ↥(elementaryGroup ι R) where
  exists_pair_ne := hnt.exists_pair_ne
  eq_bot_or_eq_top_of_normal := by
    intro N hN
    by_cases hbot : N = ⊥
    · exact Or.inl hbot
    · haveI : N.Normal := hN
      obtain ⟨i, j, hij, x, hx0, hmem⟩ := hdetect N hN hbot
      exact Or.inr (normal_eq_top_of_elGen_mem_of_isSimpleRing hcard N hij hx0 hmem)

end Pestov91
end GroupApproximation
