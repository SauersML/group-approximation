import GroupApproximation.Pestov91.NormalSubgroupsCasesEmbed
import Mathlib.GroupTheory.Commutator.Basic

/-!
# The central-idempotent line (N1) and the commutator identity of §4.2(c)

Let `f = ∑ₐ E a a` and `P = diag(f)`.  In §4.2(c) the localized commutator `k ∈ N` restricts,
on the tower over `W`, to the constant matrix `κ₀`.  The second review
(`research/artifacts/ex-pestov91-second-review-2026-09-13.md`, note N1) asks for the line that
justifies `[k, ι_W(ε)] = ι_W([κ₀, ε])`: `P` commutes with `k` and with every `ι_W(ε)`.
Here that line is the pair of hypotheses

  `P * k = blockEmbed E κ₀`,   `k * P = blockEmbed E κ₀`,

saying that `P` commutes with `k` and that the `P`-corner of `k` is `κ₀` read through the
matrix units.  `ι_W(ε)` commutes with `P` by construction (`towerEmbed_mul_diagonal_sum`,
`diagonal_sum_mul_towerEmbed`).  Every matrix splits as `X = X P + X (1 - P)`, and the two
sides of the identity agree on both pieces:
* times `P`, both equal `blockEmbed κ₀ · blockEmbed ε · blockEmbed κ₀⁻¹ · blockEmbed ε⁻¹`;
* times `1 - P`, both equal `1 - P`.

As N1 requires, `ι_W` is read on all of `GL_{ι×n}(K)`, since `κ₀` need not have determinant `1`.
-/

namespace GroupApproximation
namespace Pestov91

open Matrix
open scoped commutatorElement

variable {K R : Type*} [CommRing K] [Ring R] [Algebra K R]
variable {ι n : Type*} [Fintype ι] [DecidableEq ι] [Fintype n] [DecidableEq n]
variable {E : n → n → R}

/-- **N1, the commutator identity of §4.2(c)**: if `P = diag(∑ₐ E a a)` commutes with `k` and
the `P`-corner of `k` is `κ₀`, then `[k, ι_W(ε)] = ι_W([κ₀, ε])` for every `ε ∈ GL_{ι×n}(K)`. -/
theorem towerUnits_commutator (hE : IsMatrixUnits E) (k : (Matrix ι ι R)ˣ)
    (κ₀ ε : (Matrix (ι × n) (ι × n) K)ˣ)
    (hleft : (diagonal fun _ ↦ matrixUnitsSum E) * (k : Matrix ι ι R) =
      blockEmbed E (κ₀ : Matrix (ι × n) (ι × n) K))
    (hright : (k : Matrix ι ι R) * (diagonal fun _ ↦ matrixUnitsSum E) =
      blockEmbed E (κ₀ : Matrix (ι × n) (ι × n) K)) :
    ⁅k, towerUnits K ι hE ε⁆ = towerUnits K ι hE ⁅κ₀, ε⁆ := by
  have hPk : (diagonal fun _ ↦ matrixUnitsSum E) * (k : Matrix ι ι R) =
      (k : Matrix ι ι R) * diagonal fun _ ↦ matrixUnitsSum E := hleft.trans hright.symm
  have hPkinv : (diagonal fun _ ↦ matrixUnitsSum E) * ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) =
      ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) * diagonal fun _ ↦ matrixUnitsSum E := by
    calc (diagonal fun _ ↦ matrixUnitsSum E) * ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R)
        = ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) * (k : Matrix ι ι R) *
            (diagonal fun _ ↦ matrixUnitsSum E) * ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) := by
          rw [Units.inv_mul, one_mul]
      _ = ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) *
            ((diagonal fun _ ↦ matrixUnitsSum E) * (k : Matrix ι ι R)) *
            ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) := by
          rw [hPk, mul_assoc ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R)]
      _ = ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) * diagonal fun _ ↦ matrixUnitsSum E := by
          rw [mul_assoc, mul_assoc, Units.mul_inv, mul_one]
  have hkinvP : ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) * (diagonal fun _ ↦ matrixUnitsSum E) =
      blockEmbed E ((κ₀⁻¹ : (Matrix (ι × n) (ι × n) K)ˣ) : Matrix (ι × n) (ι × n) K) := by
    calc ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) * (diagonal fun _ ↦ matrixUnitsSum E)
        = (diagonal fun _ ↦ matrixUnitsSum E) * ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) :=
          hPkinv.symm
      _ = blockEmbed E ((κ₀⁻¹ : (Matrix (ι × n) (ι × n) K)ˣ) : Matrix (ι × n) (ι × n) K) *
            blockEmbed E (κ₀ : Matrix (ι × n) (ι × n) K) *
            ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) := by
          rw [← blockEmbed_mul hE, Units.inv_mul, blockEmbed_one]
      _ = blockEmbed E ((κ₀⁻¹ : (Matrix (ι × n) (ι × n) K)ˣ) : Matrix (ι × n) (ι × n) K) *
            ((diagonal fun _ ↦ matrixUnitsSum E) * (k : Matrix ι ι R)) *
            ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) := by
          rw [hleft]
      _ = blockEmbed E ((κ₀⁻¹ : (Matrix (ι × n) (ι × n) K)ˣ) : Matrix (ι × n) (ι × n) K) *
            (diagonal fun _ ↦ matrixUnitsSum E) := by
          rw [mul_assoc, mul_assoc, Units.mul_inv, mul_one]
      _ = blockEmbed E ((κ₀⁻¹ : (Matrix (ι × n) (ι × n) K)ˣ) : Matrix (ι × n) (ι × n) K) :=
          blockEmbed_mul_diagonal_sum hE _
  have hkB : ∀ M : Matrix (ι × n) (ι × n) K, (k : Matrix ι ι R) * blockEmbed E M =
      blockEmbed E (κ₀ : Matrix (ι × n) (ι × n) K) * blockEmbed E M := by
    intro M
    conv_lhs => rw [← diagonal_sum_mul_blockEmbed hE M, ← mul_assoc, hright]
  have hk_t : ∀ (M : Matrix (ι × n) (ι × n) K) (X : Matrix ι ι R),
      (k : Matrix ι ι R) * (blockEmbed E M * X) =
        blockEmbed E (κ₀ : Matrix (ι × n) (ι × n) K) * (blockEmbed E M * X) := by
    intro M X
    rw [← mul_assoc, hkB, mul_assoc]
  have hT_t : ∀ (M : Matrix (ι × n) (ι × n) K) (X : Matrix ι ι R),
      towerEmbed E (ε : Matrix (ι × n) (ι × n) K) * (blockEmbed E M * X) =
        blockEmbed E (ε : Matrix (ι × n) (ι × n) K) * (blockEmbed E M * X) := by
    intro M X
    rw [← mul_assoc, towerEmbed_mul_blockEmbed hE, mul_assoc]
  have hkinv_t : ∀ X : Matrix ι ι R, ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) *
      ((diagonal fun _ ↦ matrixUnitsSum E) * X) =
        blockEmbed E ((κ₀⁻¹ : (Matrix (ι × n) (ι × n) K)ˣ) : Matrix (ι × n) (ι × n) K) * X := by
    intro X
    rw [← mul_assoc, hkinvP]
  have hTinvP : towerEmbed E ((ε⁻¹ : (Matrix (ι × n) (ι × n) K)ˣ) : Matrix (ι × n) (ι × n) K) *
      (diagonal fun _ ↦ matrixUnitsSum E) = (diagonal fun _ ↦ matrixUnitsSum E) *
        blockEmbed E ((ε⁻¹ : (Matrix (ι × n) (ι × n) K)ˣ) : Matrix (ι × n) (ι × n) K) := by
    rw [towerEmbed_mul_diagonal_sum hE, diagonal_sum_mul_blockEmbed hE]
  have hkQ : (k : Matrix ι ι R) * (1 - diagonal fun _ ↦ matrixUnitsSum E) =
      (1 - diagonal fun _ ↦ matrixUnitsSum E) * (k : Matrix ι ι R) := by
    rw [mul_sub, sub_mul, mul_one, one_mul, hPk]
  have hkinvQ : ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) * (1 - diagonal fun _ ↦ matrixUnitsSum E) =
      (1 - diagonal fun _ ↦ matrixUnitsSum E) * ((k⁻¹ : (Matrix ι ι R)ˣ) : Matrix ι ι R) := by
    rw [mul_sub, sub_mul, mul_one, one_mul, hPkinv]
  have hk_Qt : ∀ X : Matrix ι ι R,
      (k : Matrix ι ι R) * ((1 - diagonal fun _ ↦ matrixUnitsSum E) * X) =
        (1 - diagonal fun _ ↦ matrixUnitsSum E) * ((k : Matrix ι ι R) * X) := by
    intro X
    rw [← mul_assoc, hkQ, mul_assoc]
  have hT_Qt : ∀ X : Matrix ι ι R, towerEmbed E (ε : Matrix (ι × n) (ι × n) K) *
      ((1 - diagonal fun _ ↦ matrixUnitsSum E) * X) =
        (1 - diagonal fun _ ↦ matrixUnitsSum E) * X := by
    intro X
    rw [← mul_assoc, towerEmbed_mul_compl hE]
  have hsplit : ∀ X : Matrix ι ι R, X = X * (diagonal fun _ ↦ matrixUnitsSum E) +
      X * (1 - diagonal fun _ ↦ matrixUnitsSum E) := by
    intro X
    rw [mul_sub, mul_one]
    abel
  apply Units.ext
  rw [commutatorElement_def, ← map_inv (towerUnits K ι hE) ε, Units.val_mul, Units.val_mul,
    Units.val_mul, towerUnits_coe, towerUnits_coe, towerUnits_coe, commutatorElement_def,
    Units.val_mul, Units.val_mul, Units.val_mul]
  refine (hsplit _).trans ?_
  refine Eq.trans ?_ (hsplit _).symm
  congr 1
  · conv_rhs => rw [towerEmbed_mul_diagonal_sum hE, blockEmbed_mul hE, blockEmbed_mul hE,
      blockEmbed_mul hE]
    simp only [mul_assoc]
    rw [hTinvP, hkinv_t, hT_t, hk_t]
  · conv_rhs => rw [towerEmbed_mul_compl hE]
    simp only [mul_assoc]
    rw [towerEmbed_mul_compl hE, hkinvQ, hT_Qt, hk_Qt, Units.mul_inv, mul_one]

end Pestov91
end GroupApproximation
