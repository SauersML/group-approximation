import Mathlib.RingTheory.NonUnitalSubring.Basic
import Mathlib.Topology.Compactness.Compact
import Mathlib.Topology.LocallyConstant.Algebra

/-!
# Small indicators generate the crossed product as a non-unital ring

§4.2(a) of `research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md` needs a
generating set of `R = LC(X, 𝔽₂) ⋊ ℤ` all of whose elements sit in towers over clopen sets with
long return times.  This module supplies it over any compact space `X` and any ring `R` spanned
by `ι(LC(X, 𝔽₂))` and the powers of a unit `u` (the hypothesis `hspan` of `Pestov91.RingSimple`):

  `{ι(χ_W) u^ε : W clopen and small, ε ∈ {-1, 0, 1}}` generates `R` as a non-unital ring,

whenever "small" is hereditary and every point has a small clopen neighbourhood
(`closure_smallCharFn_mul_zpow_eq_top`).  For the witness, "small" is "disjoint from its images
under `τ n`, `0 < |n| ≤ L`", and small neighbourhoods come from freeness.

The proof peels one set of a finite small clopen cover at a time, `g = g χ_W + (g - g χ_W)`; over
`𝔽₂` the product `g χ_W` is the indicator of `{g = 1} ∩ W`, which is small.  Then `1`, `u` and
`u⁻¹` lie in the closure, hence every `u^j`, hence every `ι(f) u^j`.
-/

namespace GroupApproximation
namespace Pestov91

open LocallyConstant

variable {X : Type*} [TopologicalSpace X] {R : Type*} [Ring R]

/-- Over `𝔽₂`, cutting a locally constant function down to a clopen set gives the indicator of
`{g = 1} ∩ W`. -/
theorem mul_charFn_eq_charFn_inter (g : LocallyConstant X (ZMod 2)) {W : Set X}
    (hW : IsClopen W) :
    g * charFn (ZMod 2) hW = charFn (ZMod 2) ((g.isLocallyConstant.isClopen_fiber 1).inter hW) := by
  have h01 : ∀ z : ZMod 2, z = 0 ∨ z = 1 := by decide
  ext y
  rw [LocallyConstant.coe_mul, Pi.mul_apply, coe_charFn, coe_charFn]
  by_cases hy : y ∈ W
  · rcases h01 (g y) with h | h
    · simp [Set.indicator_apply, hy, h]
    · simp [Set.indicator_apply, hy, h]
  · simp [Set.indicator_apply, hy]

/-- **Small indicators generate.**  If `Small` is hereditary and every point has a small clopen
neighbourhood, the elements `ι(χ_W) u^ε` with `W` small clopen and `ε ∈ {-1, 0, 1}` generate `R`
as a non-unital ring. -/
theorem closure_smallCharFn_mul_zpow_eq_top [CompactSpace X]
    (Small : Set X → Prop) (hmono : ∀ W W' : Set X, W' ⊆ W → Small W → Small W')
    (hnbhd : ∀ x : X, ∃ W : Set X, IsClopen W ∧ x ∈ W ∧ Small W)
    (ι : LocallyConstant X (ZMod 2) →+* R) (u : Rˣ)
    (hspan : ∀ r : R, ∃ (s : Finset ℤ) (f : ℤ → LocallyConstant X (ZMod 2)),
      r = ∑ j ∈ s, ι (f j) * ((u ^ j : Rˣ) : R)) :
    NonUnitalSubring.closure
      {r : R | ∃ (W : Set X) (hW : IsClopen W) (ε : ℤ), Small W ∧ (ε = -1 ∨ ε = 0 ∨ ε = 1) ∧
        r = ι (charFn (ZMod 2) hW) * ((u ^ ε : Rˣ) : R)} = ⊤ := by
  classical
  set C := NonUnitalSubring.closure
    {r : R | ∃ (W : Set X) (hW : IsClopen W) (ε : ℤ), Small W ∧ (ε = -1 ∨ ε = 0 ∨ ε = 1) ∧
      r = ι (charFn (ZMod 2) hW) * ((u ^ ε : Rˣ) : R)}
  choose W hW hxW hsmall using hnbhd
  obtain ⟨t, ht⟩ := CompactSpace.elim_nhds_subcover W fun x ↦ (hW x).isOpen.mem_nhds (hxW x)
  -- Functions supported in the union of the sets `W x`, `x ∈ s`, give elements of `C`.
  have hpeel : ∀ s : Finset X, ∀ g : LocallyConstant X (ZMod 2),
      (∀ y : X, g y ≠ 0 → ∃ x ∈ s, y ∈ W x) → ∀ ε : ℤ, (ε = -1 ∨ ε = 0 ∨ ε = 1) →
        ι g * ((u ^ ε : Rˣ) : R) ∈ C := by
    intro s
    induction s using Finset.induction_on with
    | empty =>
      intro g hg ε _
      have hg0 : g = 0 := LocallyConstant.ext fun y ↦ by
        by_contra hy
        obtain ⟨x, hx, -⟩ := hg y hy
        simp at hx
      rw [hg0, map_zero, zero_mul]
      exact zero_mem C
    | insert a s _ ih =>
      intro g hg ε hε
      have hsupp : ∀ y : X, (g - g * charFn (ZMod 2) (hW a)) y ≠ 0 → ∃ x ∈ s, y ∈ W x := by
        intro y hy
        by_cases hya : y ∈ W a
        · exact absurd (show (g - g * charFn (ZMod 2) (hW a)) y = 0 by
            simp [coe_charFn, Set.indicator_apply, hya]) hy
        · have hgy : g y ≠ 0 := by simpa [coe_charFn, Set.indicator_apply, hya] using hy
          obtain ⟨x, hx, hyx⟩ := hg y hgy
          rcases Finset.mem_insert.mp hx with rfl | hxs
          · exact absurd hyx hya
          · exact ⟨x, hxs, hyx⟩
      have hsplit : ι g * ((u ^ ε : Rˣ) : R) =
          ι (g * charFn (ZMod 2) (hW a)) * ((u ^ ε : Rˣ) : R) +
            ι (g - g * charFn (ZMod 2) (hW a)) * ((u ^ ε : Rˣ) : R) := by
        rw [← add_mul, ← map_add, add_sub_cancel]
      rw [hsplit]
      refine add_mem (NonUnitalSubring.subset_closure ?_) (ih _ hsupp ε hε)
      refine ⟨_, (g.isLocallyConstant.isClopen_fiber 1).inter (hW a), ε,
        hmono _ _ Set.inter_subset_right (hsmall a), hε, ?_⟩
      rw [mul_charFn_eq_charFn_inter]
  have hall : ∀ g : LocallyConstant X (ZMod 2), ∀ ε : ℤ, (ε = -1 ∨ ε = 0 ∨ ε = 1) →
      ι g * ((u ^ ε : Rˣ) : R) ∈ C := fun g ↦
    hpeel t g fun y _ ↦ by
      have hy : y ∈ ⋃ x ∈ t, W x := by
        rw [ht]
        exact Set.mem_univ y
      obtain ⟨x, hx, hyx⟩ := Set.mem_iUnion₂.mp hy
      exact ⟨x, hx, hyx⟩
  have hι : ∀ g : LocallyConstant X (ZMod 2), ι g ∈ C := fun g ↦ by
    have h := hall g 0 (Or.inr (Or.inl rfl))
    rwa [zpow_zero, Units.val_one, mul_one] at h
  have hone : (1 : R) ∈ C := by
    have h := hι 1
    rwa [map_one] at h
  have hu : ((u : Rˣ) : R) ∈ C := by
    have h := hall 1 1 (Or.inr (Or.inr rfl))
    rwa [map_one, one_mul, zpow_one] at h
  have huinv : ((u⁻¹ : Rˣ) : R) ∈ C := by
    have h := hall 1 (-1) (Or.inl rfl)
    rwa [map_one, one_mul, zpow_neg_one] at h
  have hpow : ∀ j : ℤ, ((u ^ j : Rˣ) : R) ∈ C := fun j ↦ by
    induction j using Int.induction_on with
    | zero =>
      rw [zpow_zero, Units.val_one]
      exact hone
    | succ n ih =>
      rw [zpow_add_one, Units.val_mul]
      exact mul_mem ih hu
    | pred n ih =>
      rw [zpow_sub_one, Units.val_mul]
      exact mul_mem ih huinv
  refine eq_top_iff.mpr fun r _ ↦ ?_
  obtain ⟨s, f, rfl⟩ := hspan r
  exact sum_mem fun j _ ↦ mul_mem (hι (f j)) (hpow j)

end Pestov91
end GroupApproximation
