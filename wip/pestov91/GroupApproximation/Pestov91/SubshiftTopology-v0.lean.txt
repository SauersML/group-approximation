/-!
# Two-sided subshifts over `Bool`

Configurations are sequences `ℤ → Bool` with the product topology. The shift is Mathlib's
`SymbolicDynamics.FullShift.shift`, `(shift n x) m = x (n + m)`, so `shift 1` is the shift
`(T x)_m = x_(m+1)` of the construction. This file supplies what the Pestov 9.1 modules use
about a subshift `X`:

* `Agree k x y`: `x` and `y` agree on the window `[-k, k]`. These sets are cylinders, hence
  open, and every open set around a point contains one of them (`exists_agree_subset`).
* `orbitClosure x`, the closure of the shift orbit of `x`, as a Mathlib `Subshift`, with its
  window description `mem_orbitClosure_iff`.
* The shift action of `ℤ` on the points of a subshift: continuous, on a compact space.
* `isMinimal_orbitClosure`: the orbit closure of `x` is minimal as soon as `x` lies in the
  orbit closure of each of its points. Then finitely many translates of a nonempty open set
  cover the subshift (`exists_finite_cover_vadd`).
* `exists_radius`: a locally constant function on a subshift depends only on a finite window.
-/

namespace GroupApproximation.Pestov91

open SymbolicDynamics.FullShift Set Topology
open scoped Pointwise

/-- `x` and `y` agree on the centred window `[-k, k]`. -/
def Agree (k : ℕ) (x y : ℤ → Bool) : Prop :=
  ∀ m : ℤ, |m| ≤ k → x m = y m

namespace Agree

theorem refl (k : ℕ) (x : ℤ → Bool) : Agree k x x := fun _ _ => rfl

theorem symm {k : ℕ} {x y : ℤ → Bool} (h : Agree k x y) : Agree k y x :=
  fun m hm => (h m hm).symm

theorem trans {k : ℕ} {x y z : ℤ → Bool} (h : Agree k x y) (h' : Agree k y z) :
    Agree k x z :=
  fun m hm => (h m hm).trans (h' m hm)

theorem mono {k k' : ℕ} {x y : ℤ → Bool} (hk : k ≤ k') (h : Agree k' x y) : Agree k x y :=
  fun m hm => h m (hm.trans (by exact_mod_cast hk))

end Agree

theorem setOf_agree_eq_cylinder (k : ℕ) (x : ℤ → Bool) :
    {y | Agree k x y} = cylinder (Finset.Icc (-(k : ℤ)) k) x := by
  ext y
  refine ⟨fun h => mem_cylinder.2 fun i hi => ?_, fun h => ?_⟩
  · exact (show Agree k x y from h i (abs_le.2 (Finset.mem_Icc.1 hi))).symm
  · show Agree k x y
    exact fun m hm => (mem_cylinder.1 h m (Finset.mem_Icc.2 (abs_le.1 hm))).symm

theorem isOpen_setOf_agree (k : ℕ) (x : ℤ → Bool) : IsOpen {y | Agree k x y} := by
  rw [setOf_agree_eq_cylinder]
  exact isOpen_cylinder _ _

/-- Every open set around `y` contains all configurations agreeing with `y` on a window. -/
theorem exists_agree_subset {t : Set (ℤ → Bool)} (ht : IsOpen t) {y : ℤ → Bool} (hy : y ∈ t) :
    ∃ k : ℕ, ∀ z, Agree k y z → z ∈ t := by
  obtain ⟨I, u, hu, hsub⟩ := isOpen_pi_iff.1 ht y hy
  refine ⟨I.sup Int.natAbs, fun z hz => hsub (Set.mem_pi.2 fun a ha => ?_)⟩
  have ha' : |a| ≤ ((I.sup Int.natAbs : ℕ) : ℤ) := by
    rw [Int.abs_eq_natAbs]
    exact_mod_cast Finset.le_sup (f := Int.natAbs) (Finset.mem_coe.1 ha)
  rw [← hz a ha']
  exact (hu a (Finset.mem_coe.1 ha)).2

theorem mem_closure_iff_agree {s : Set (ℤ → Bool)} {y : ℤ → Bool} :
    y ∈ closure s ↔ ∀ k : ℕ, ∃ z ∈ s, Agree k y z := by
  constructor
  · intro h k
    obtain ⟨z, hz, hzs⟩ := mem_closure_iff.1 h _ (isOpen_setOf_agree k y) (Agree.refl k y)
    exact ⟨z, hzs, hz⟩
  · intro h
    refine mem_closure_iff.2 fun t ht hy => ?_
    obtain ⟨k, hk⟩ := exists_agree_subset ht hy
    obtain ⟨z, hzs, hz⟩ := h k
    exact ⟨z, hk z hz, hzs⟩

/-- The closure of the shift orbit of `x`, as a subshift. -/
def orbitClosure (x : ℤ → Bool) : Subshift Bool ℤ where
  carrier := closure (range fun n : ℤ => shift n x)
  isClosed := isClosed_closure
  mapsTo g := by
    refine MapsTo.closure ?_ (continuous_shift g)
    rintro _ ⟨n, rfl⟩
    refine ⟨g + n, ?_⟩
    funext h
    show x (g + n + h) = x (n + (g + h))
    congr 1
    ring

theorem shift_mem_orbitClosure (x : ℤ → Bool) (n : ℤ) : shift n x ∈ (orbitClosure x).carrier :=
  subset_closure ⟨n, rfl⟩

theorem self_mem_orbitClosure (x : ℤ → Bool) : x ∈ (orbitClosure x).carrier := by
  simpa using shift_mem_orbitClosure x 0

/-- A configuration lies in the orbit closure of `x` iff each of its windows is a window of `x`. -/
theorem mem_orbitClosure_iff {x y : ℤ → Bool} :
    y ∈ (orbitClosure x).carrier ↔ ∀ k : ℕ, ∃ n : ℤ, Agree k y (shift n x) := by
  change y ∈ closure _ ↔ _
  simp only [mem_closure_iff_agree, exists_range_iff]

section Action

variable {X : Subshift Bool ℤ}

/-- The shift action of `ℤ` on the points of a subshift. -/
instance : AddAction ℤ X.carrier where
  vadd n y := ⟨shift n y, X.mapsTo n y.2⟩
  zero_vadd y := Subtype.ext (shift_zero _)
  add_vadd a b y := Subtype.ext <| by
    funext h
    show (y : ℤ → Bool) (a + b + h) = (y : ℤ → Bool) (b + (a + h))
    congr 1
    ring

@[simp] theorem coe_vadd (n : ℤ) (y : X.carrier) :
    ((n +ᵥ y : X.carrier) : ℤ → Bool) = shift n y := rfl

instance : ContinuousConstVAdd ℤ X.carrier :=
  ⟨fun n => ((continuous_shift n).comp continuous_subtype_val).subtype_mk
    fun y => X.mapsTo n y.2⟩

instance : CompactSpace X.carrier :=
  isCompact_iff_compactSpace.1 X.isClosed.isCompact

theorem image_val_orbit (y : X.carrier) :
    ((↑) : X.carrier → ℤ → Bool) '' AddAction.orbit ℤ y =
      range fun n : ℤ => shift n (y : ℤ → Bool) := by
  ext z
  constructor
  · rintro ⟨w, hw, rfl⟩
    obtain ⟨n, rfl⟩ := AddAction.mem_orbit_iff.1 hw
    exact ⟨n, rfl⟩
  · rintro ⟨n, rfl⟩
    exact ⟨n +ᵥ y, AddAction.mem_orbit y n, rfl⟩

/-- Finitely many translates of a nonempty open subset of a minimal subshift cover it. -/
theorem exists_finite_cover_vadd [AddAction.IsMinimal ℤ X.carrier] {U : Set X.carrier}
    (hU : IsOpen U) (hne : U.Nonempty) : ∃ I : Finset ℤ, ⋃ n ∈ I, n +ᵥ U = univ := by
  obtain ⟨I, hI⟩ := isCompact_univ.exists_finite_cover_vadd ℤ hU hne
  exact ⟨I, eq_univ_of_univ_subset hI⟩

/-- A subshift with a free shift action and a point is infinite. -/
theorem infinite_of_free [Nonempty X.carrier]
    (hfree : ∀ y : X.carrier, ∀ n : ℤ, n ≠ 0 → n +ᵥ y ≠ y) : Infinite X.carrier := by
  obtain ⟨y⟩ := ‹Nonempty X.carrier›
  refine Infinite.of_injective (fun n : ℤ => n +ᵥ y) fun a b hab => ?_
  by_contra hne
  apply hfree y (a - b) (sub_ne_zero.2 hne)
  have h : (-b) +ᵥ (a +ᵥ y) = (-b) +ᵥ (b +ᵥ y) := congrArg _ hab
  rwa [← add_vadd, ← add_vadd, neg_add_cancel, zero_vadd, neg_add_eq_sub] at h

/-- A locally constant function on a subshift depends only on a finite window. -/
theorem exists_radius {β : Type*} {f : X.carrier → β} (hf : IsLocallyConstant f) :
    ∃ ρ : ℕ, ∀ y y' : X.carrier, Agree ρ y y' → f y = f y' := by
  have hloc : ∀ y : X.carrier, ∃ k : ℕ, ∀ y' : X.carrier, Agree k y y' → f y' = f y := by
    intro y
    obtain ⟨t, ht, hts⟩ := isOpen_induced_iff.1 (hf.isOpen_fiber (f y))
    have hy : y ∈ Subtype.val ⁻¹' t := by rw [hts]; exact rfl
    obtain ⟨k, hk⟩ := exists_agree_subset ht hy
    refine ⟨k, fun y' hy' => ?_⟩
    have hmem : y' ∈ Subtype.val ⁻¹' t := hk y' hy'
    rw [hts] at hmem
    exact hmem
  choose k hk using hloc
  obtain ⟨s, hs⟩ := isCompact_univ.elim_finite_subcover
    (fun y : X.carrier => {y' : X.carrier | Agree (k y) y y'})
    (fun y => (isOpen_setOf_agree (k y) y).preimage continuous_subtype_val)
    (fun y _ => mem_iUnion.2 ⟨y, Agree.refl _ _⟩)
  refine ⟨s.sup k, fun y y' hyy' => ?_⟩
  obtain ⟨c, hc, hyc⟩ := mem_iUnion₂.1 (hs (mem_univ y))
  have hyc' : Agree (k c) c y' := Agree.trans hyc (Agree.mono (Finset.le_sup hc) hyy')
  rw [hk c y hyc, hk c y' hyc']

theorem exists_radius_locallyConstant {β : Type*} (f : LocallyConstant X.carrier β) :
    ∃ ρ : ℕ, ∀ y y' : X.carrier, Agree ρ y y' → f y = f y' :=
  exists_radius f.isLocallyConstant

end Action

/-- The orbit closure of `x` is minimal once `x` lies in the orbit closure of each of its
points. -/
theorem isMinimal_orbitClosure {x : ℤ → Bool}
    (hx : ∀ y ∈ (orbitClosure x).carrier, x ∈ (orbitClosure y).carrier) :
    AddAction.IsMinimal ℤ (orbitClosure x).carrier := by
  refine ⟨fun y z => closure_subtype.2 ?_⟩
  rw [image_val_orbit]
  refine closure_minimal ?_ isClosed_closure z.2
  rintro _ ⟨n, rfl⟩
  exact (orbitClosure (y : ℤ → Bool)).mapsTo n (hx y y.2)

end GroupApproximation.Pestov91
