/-!
# Simplicity of `EL₃(LC(X, 𝔽₂) ⋊_σ ℤ)` for a free minimal homeomorphism

The σ-level step of blueprint v2 leaf G4.  It instantiates
`isSimpleGroup_elementaryGroup_of_minimal_free` with the data of the crossed product
`CrossedProduct σ` (lane pc-crossed-product), exactly as `isSimpleRing_crossedProduct` (lane
pc-ring-simple) instantiates `isSimpleRing_of_minimal_free`:

  `ι = C σ`, `u = u σ`, `τ j = σ ^ (-j)`, `σ' j = shift σ j`,

with covariance `u_zpow_mul_C` and normal form `exists_eq_sum_C_mul_u_zpow`.  The remaining
hypotheses are freeness and density of the orbits of `σ`, and `Z(CrossedProduct σ) = {0, 1}`;
for the Toeplitz shift they are G1 and G3.
-/

namespace GroupApproximation
namespace Pestov91

/-- **`EL₃(LC(X, 𝔽₂) ⋊_σ ℤ)` is simple** for a homeomorphism `σ` of a nonempty compact totally
separated space whose nonzero powers have no fixed points and whose orbits are dense, once the
central elements of the crossed product are `0` and `1`. -/
theorem isSimpleGroup_elementaryGroup_crossedProduct_of_minimal_free {X : Type*}
    [TopologicalSpace X] [Nonempty X] [CompactSpace X] [TotallySeparatedSpace X] (σ : X ≃ₜ X)
    (hfree : ∀ j : ℤ, j ≠ 0 → ∀ x, (σ ^ (-j)) x ≠ x)
    (hdense : ∀ x, Dense (Set.range fun j : ℤ => (σ ^ (-j)) x))
    (hR : ∀ c : CrossedProduct σ, (∀ r : CrossedProduct σ, r * c = c * r) → c = 0 ∨ c = 1) :
    IsSimpleGroup ↥(elementaryGroup (Fin 3) (CrossedProduct σ)) :=
  isSimpleGroup_elementaryGroup_of_minimal_free (by simp) hR (CrossedProduct.C σ)
    (CrossedProduct.u σ) (fun j => ⇑(σ ^ (-j))) (fun j => (σ ^ (-j)).continuous) hfree hdense
    (fun j => ⇑(CrossedProduct.shift σ j)) (CrossedProduct.shift_apply σ)
    (CrossedProduct.u_zpow_mul_C σ) (CrossedProduct.exists_eq_sum_C_mul_u_zpow σ)

end Pestov91
end GroupApproximation
