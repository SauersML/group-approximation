/-!
# Pestov 9.1: tower subrings and bounded-propagation absorption

§1.5–1.6 and §4.2 of
`research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`
(Part 1 §3.5–3.6 and Part 2 §6.2 of `pestov-9-1-writeup-2026-09-13-part*.md`).

The ring `R = LC(X, 𝔽₂) ⋊ ℤ` enters only through its crossed-product relations,
collected in `TowerSetting`: a commutative coefficient ring `A`, an automorphism
`α` of `A`, a ring map `coeff : A →+* R` and a unit `u` with
`u · coeff f = coeff (α f) · u`.  For the subshift ring `A = LC(X, 𝔽₂)` and
`α f = f ∘ T⁻¹`, so `α χ_Y = χ_{TY}`.

For an idempotent `e = χ_U` the tower generators are
`towerElem e a b x = coeff (α^a (x e)) u^(a-b)`, which is `E_ab(W)` for
`x = χ_W`, `W ⊆ U`.  `towerMap e m` sums them over a matrix indexed by the
levels `[-m, m]`, and `tower e m = B_m(U)` is its range.

* `towerMap_mul`: when `e · α^j e = 0` for `0 < |j| ≤ 2m` (`LongReturn`), the
  tower map is multiplicative, so `B_m(U)` is a ring image of `M_(2m+1)(A)`.
* `mul_mul_mem_tower`: `r b r' ∈ B_(m+w)(U)` for `b ∈ B_m(U)` and `r, r'` of
  propagation `≤ w`.  With coefficients in `A` rather than in `LC(U)`, this is an
  exact identity on generators (`coeff_mul_towerElem_mul`); membership needs no
  return-time hypothesis.
-/

namespace GroupApproximation.Pestov91

/-- The crossed-product relations used by the tower calculus. -/
structure TowerSetting (A R : Type*) [CommRing A] [Ring R] where
  /-- The automorphism of the coefficient ring, `f ↦ f ∘ T⁻¹` for a subshift. -/
  α : RingAut A
  /-- The coefficient embedding. -/
  coeff : A →+* R
  /-- The implementing unit. -/
  u : Rˣ
  u_mul_coeff : ∀ f : A, (u : R) * coeff f = coeff (α f) * u

/-- The levels `[-m, m]` of a tower. -/
abbrev Level (m : ℕ) : Type := ↥(Finset.Icc (-(m : ℤ)) m)

theorem Level.abs_le {m : ℕ} (a : Level m) : |(a : ℤ)| ≤ m :=
  _root_.abs_le.mpr (Finset.mem_Icc.mp a.2)

namespace TowerSetting

variable {A R : Type*} [CommRing A] [Ring R] (S : TowerSetting A R)

theorem zpow_apply_zpow_apply (p q : ℤ) (z : A) :
    (S.α ^ p) ((S.α ^ q) z) = (S.α ^ (p + q)) z := by
  rw [zpow_add, RingAut.mul_apply]

theorem u_inv_mul_coeff (f : A) :
    ((S.u⁻¹ : Rˣ) : R) * S.coeff f = S.coeff (S.α⁻¹ f) * ((S.u⁻¹ : Rˣ) : R) := by
  have hα : S.α (S.α⁻¹ f) = f := by
    rw [← RingAut.mul_apply, mul_inv_cancel, RingAut.one_apply]
  have h := S.u_mul_coeff (S.α⁻¹ f)
  rw [hα] at h
  calc ((S.u⁻¹ : Rˣ) : R) * S.coeff f
      = ((S.u⁻¹ : Rˣ) : R) * (S.coeff f * S.u) * ((S.u⁻¹ : Rˣ) : R) := by
        simp [mul_assoc]
    _ = ((S.u⁻¹ : Rˣ) : R) * (S.u * S.coeff (S.α⁻¹ f)) * ((S.u⁻¹ : Rˣ) : R) := by
        rw [h]
    _ = S.coeff (S.α⁻¹ f) * ((S.u⁻¹ : Rˣ) : R) := by
        simp [← mul_assoc]

/-- `u^j · coeff f = coeff (α^j f) · u^j` for every `j ∈ ℤ`. -/
theorem zpow_mul_coeff (j : ℤ) (f : A) :
    ((S.u ^ j : Rˣ) : R) * S.coeff f = S.coeff ((S.α ^ j) f) * ((S.u ^ j : Rˣ) : R) := by
  induction j using Int.induction_on generalizing f with
  | zero => simp
  | succ n ih =>
    rw [zpow_add_one, zpow_add_one, Units.val_mul, mul_assoc, S.u_mul_coeff, ← mul_assoc,
      ih, mul_assoc, RingAut.mul_apply]
  | pred n ih =>
    rw [zpow_sub_one, zpow_sub_one, Units.val_mul, mul_assoc, S.u_inv_mul_coeff,
      ← mul_assoc, ih, mul_assoc, RingAut.mul_apply]

/-- The product of two monomials `coeff p · u^i`. -/
theorem coeff_mul_zpow_mul (p q : A) (i j : ℤ) :
    S.coeff p * ((S.u ^ i : Rˣ) : R) * (S.coeff q * ((S.u ^ j : Rˣ) : R)) =
      S.coeff (p * (S.α ^ i) q) * ((S.u ^ (i + j) : Rˣ) : R) := by
  rw [mul_assoc, ← mul_assoc ((S.u ^ i : Rˣ) : R), S.zpow_mul_coeff, map_mul, zpow_add,
    Units.val_mul]
  simp only [mul_assoc]

/-- `e · α^j e = 0` for `0 < |j| ≤ n`: for `e = χ_U`, the translates `T^j U`
with `0 < |j| ≤ n` miss `U`. -/
def LongReturn (e : A) (n : ℕ) : Prop :=
  ∀ j : ℤ, j ≠ 0 → |j| ≤ n → e * (S.α ^ j) e = 0

theorem LongReturn.mono {S : TowerSetting A R} {e : A} {n n' : ℕ} (h : S.LongReturn e n)
    (hn : n' ≤ n) : S.LongReturn e n' :=
  fun j hj hjn ↦ h j hj (hjn.trans (by exact_mod_cast hn))

/-- Return times are invariant under translation: `T^i U` returns as late as `U`. -/
theorem longReturn_zpow {e : A} {n : ℕ} (h : S.LongReturn e n) (i : ℤ) :
    S.LongReturn ((S.α ^ i) e) n := by
  intro j hj hjn
  rw [S.zpow_apply_zpow_apply, add_comm, ← S.zpow_apply_zpow_apply, ← map_mul,
    h j hj hjn, map_zero]

/-- The tower generator `E_ab(x) = coeff (α^a (x e)) u^(a-b)`; for `x = χ_W`,
`W ⊆ U`, `e = χ_U`, it is `E_ab(W) = χ_(T^a W) u^(a-b)`. -/
def towerElem (e : A) (a b : ℤ) (x : A) : R :=
  S.coeff ((S.α ^ a) (x * e)) * ((S.u ^ (a - b) : Rˣ) : R)

/-- Matrix units compose: `E_ab(x) E_bd(y) = E_ad(xy)`. -/
theorem towerElem_mul_towerElem_same {e : A} (he : IsIdempotentElem e) (a b d : ℤ) (x y : A) :
    S.towerElem e a b x * S.towerElem e b d y = S.towerElem e a d (x * y) := by
  have hxy : x * e * (y * e) = x * y * e := by
    calc x * e * (y * e) = x * y * (e * e) := by ring
      _ = x * y * e := by rw [he.eq]
  rw [towerElem, towerElem, towerElem, S.coeff_mul_zpow_mul, S.zpow_apply_zpow_apply,
    sub_add_cancel, ← map_mul, hxy, sub_add_sub_cancel]

/-- Matrix units at different levels are orthogonal: `E_ab(x) E_cd(y) = 0` for
`b ≠ c`, once the tower levels are disjoint. -/
theorem towerElem_mul_towerElem_of_ne {e : A} {n : ℕ} (hret : S.LongReturn e n) {a b c d : ℤ}
    (hbc : b ≠ c) (hcb : |c - b| ≤ n) (x y : A) :
    S.towerElem e a b x * S.towerElem e c d y = 0 := by
  rw [towerElem, towerElem, S.coeff_mul_zpow_mul, S.zpow_apply_zpow_apply,
    show a - b + c = a + (c - b) by ring, ← S.zpow_apply_zpow_apply, ← map_mul]
  have h0 : x * e * (S.α ^ (c - b)) (y * e) = 0 := by
    rw [map_mul, show x * e * ((S.α ^ (c - b)) y * (S.α ^ (c - b)) e) =
        x * (S.α ^ (c - b)) y * (e * (S.α ^ (c - b)) e) by ring,
      hret (c - b) (sub_ne_zero.mpr hbc.symm) hcb, mul_zero]
  rw [h0, map_zero, map_zero, zero_mul]

/-- The tower map `M_(2m+1)(A) → R`, `F ↦ Σ_{a,b ∈ [-m,m]} E_ab(F_ab)`. -/
def towerMap (e : A) (m : ℕ) : Matrix (Level m) (Level m) A →+ R where
  toFun F := ∑ a : Level m, ∑ b : Level m, S.towerElem e (a : ℤ) (b : ℤ) (F a b)
  map_zero' := by simp [towerElem]
  map_add' F G := by simp [towerElem, add_mul, Finset.sum_add_distrib]

theorem towerMap_apply (e : A) (m : ℕ) (F : Matrix (Level m) (Level m) A) :
    S.towerMap e m F = ∑ a : Level m, ∑ b : Level m, S.towerElem e (a : ℤ) (b : ℤ) (F a b) :=
  rfl

theorem towerMap_single (e : A) (m : ℕ) (a b : Level m) (x : A) :
    S.towerMap e m (Matrix.single a b x) = S.towerElem e a b x := by
  rw [towerMap_apply, Finset.sum_eq_single a, Finset.sum_eq_single b]
  · simp
  · intro b' _ hb'
    simp [Ne.symm hb', towerElem]
  · simp
  · intro a' _ ha'
    simp [Ne.symm ha', towerElem]
  · simp

theorem Level.abs_sub_le {m : ℕ} (b c : Level m) : |(c : ℤ) - b| ≤ ((2 * m : ℕ) : ℤ) := by
  have hb := Finset.mem_Icc.mp b.2
  have hc := Finset.mem_Icc.mp c.2
  rw [_root_.abs_le]
  push_cast
  constructor <;> omega

/-- **Part 1 §1.5.** On a tower with disjoint levels the tower map is
multiplicative. -/
theorem towerMap_mul {e : A} (he : IsIdempotentElem e) {m : ℕ} (hret : S.LongReturn e (2 * m))
    (F G : Matrix (Level m) (Level m) A) :
    S.towerMap e m (F * G) = S.towerMap e m F * S.towerMap e m G := by
  induction F using Matrix.induction_on' with
  | h_zero => simp
  | h_add F₁ F₂ h₁ h₂ => rw [add_mul, map_add, map_add, add_mul, h₁, h₂]
  | h_std_basis a b x =>
    induction G using Matrix.induction_on' with
    | h_zero => simp
    | h_add G₁ G₂ h₁ h₂ => rw [mul_add, map_add, map_add, mul_add, h₁, h₂]
    | h_std_basis c d y =>
      rw [S.towerMap_single, S.towerMap_single]
      by_cases hbc : b = c
      · subst hbc
        rw [Matrix.single_mul_single_same, S.towerMap_single,
          S.towerElem_mul_towerElem_same he]
      · rw [Matrix.single_mul_single_of_ne (h := hbc), map_zero,
          S.towerElem_mul_towerElem_of_ne hret (fun h ↦ hbc (Subtype.ext h))
            (Level.abs_sub_le b c)]

/-- The tower subring `B_m(U)`: the range of the tower map. -/
def tower (e : A) (m : ℕ) : AddSubgroup R := (S.towerMap e m).range

theorem towerMap_mem (e : A) (m : ℕ) (F : Matrix (Level m) (Level m) A) :
    S.towerMap e m F ∈ S.tower e m :=
  ⟨F, rfl⟩

theorem towerElem_mem (e : A) {m : ℕ} {a b : ℤ} (ha : |a| ≤ m) (hb : |b| ≤ m) (x : A) :
    S.towerElem e a b x ∈ S.tower e m :=
  ⟨Matrix.single ⟨a, Finset.mem_Icc.mpr (_root_.abs_le.mp ha)⟩
      ⟨b, Finset.mem_Icc.mpr (_root_.abs_le.mp hb)⟩ x,
    S.towerMap_single e m _ _ x⟩

theorem tower_mono (e : A) {m m' : ℕ} (h : m ≤ m') : S.tower e m ≤ S.tower e m' := by
  rintro _ ⟨F, rfl⟩
  rw [towerMap_apply]
  refine AddSubgroup.sum_mem _ fun a _ ↦ AddSubgroup.sum_mem _ fun b _ ↦ ?_
  exact S.towerElem_mem e ((Level.abs_le a).trans (by exact_mod_cast h))
    ((Level.abs_le b).trans (by exact_mod_cast h)) _

/-- `B_m(U)` is closed under products once the levels are disjoint. -/
theorem mul_mem_tower {e : A} (he : IsIdempotentElem e) {m : ℕ} (hret : S.LongReturn e (2 * m))
    {r s : R} (hr : r ∈ S.tower e m) (hs : s ∈ S.tower e m) : r * s ∈ S.tower e m := by
  obtain ⟨F, rfl⟩ := hr
  obtain ⟨G, rfl⟩ := hs
  exact ⟨F * G, S.towerMap_mul he hret F G⟩

/-- A coefficient cut by `e`, times `u^i`, is the tower generator `E_(i,0)` over
`α^(-i) e`; so `χ_C = E_00(C)`, `χ_C u = E_10(T⁻¹ C)`, `χ_C u⁻¹ = E_(-1,0)(T C)`. -/
theorem coeff_mul_zpow_eq_towerElem (e f : A) (i : ℤ) :
    S.coeff (f * e) * ((S.u ^ i : Rˣ) : R) =
      S.towerElem ((S.α ^ (-i)) e) i 0 ((S.α ^ (-i)) f) := by
  rw [towerElem, sub_zero, ← map_mul, S.zpow_apply_zpow_apply, add_neg_cancel, zpow_zero,
    RingAut.one_apply]

/-- Elements of propagation at most `w`: the additive span of the monomials
`coeff f · u^i` with `|i| ≤ w`. -/
def propagation (w : ℕ) : AddSubgroup R :=
  AddSubgroup.closure {r | ∃ (f : A) (i : ℤ), |i| ≤ w ∧ S.coeff f * ((S.u ^ i : Rˣ) : R) = r}

/-- The absorption identity on generators. -/
theorem coeff_mul_towerElem_mul (e f f' : A) (i j a b : ℤ) (x : A) :
    S.coeff f * ((S.u ^ i : Rˣ) : R) * S.towerElem e a b x *
        (S.coeff f' * ((S.u ^ j : Rˣ) : R)) =
      S.towerElem e (i + a) (b - j) ((S.α ^ (-(i + a))) f * x * (S.α ^ (-b)) f') := by
  rw [towerElem, towerElem, S.coeff_mul_zpow_mul, S.coeff_mul_zpow_mul,
    show i + (a - b) + j = i + a - (b - j) by ring]
  congr 2
  simp only [map_mul, S.zpow_apply_zpow_apply, add_neg_cancel, zpow_zero, RingAut.one_apply,
    show i + a + -b = i + (a - b) by ring]
  ring

/-- **Absorption (Part 1 §1.6).** If `r, r'` have propagation `≤ w` and
`b ∈ B_m(U)`, then `r b r' ∈ B_(m+w)(U)`. -/
theorem mul_mul_mem_tower (e : A) {m w : ℕ} {r r' b : R} (hr : r ∈ S.propagation w)
    (hr' : r' ∈ S.propagation w) (hb : b ∈ S.tower e m) : r * b * r' ∈ S.tower e (m + w) := by
  obtain ⟨F, rfl⟩ := hb
  rw [towerMap_apply, Finset.mul_sum, Finset.sum_mul]
  refine AddSubgroup.sum_mem _ fun a _ ↦ ?_
  rw [Finset.mul_sum, Finset.sum_mul]
  refine AddSubgroup.sum_mem _ fun c _ ↦ ?_
  induction hr using AddSubgroup.closure_induction with
  | mem r hr =>
    obtain ⟨f, i, hi, rfl⟩ := hr
    induction hr' using AddSubgroup.closure_induction with
    | mem r' hr' =>
      obtain ⟨f', j, hj, rfl⟩ := hr'
      rw [S.coeff_mul_towerElem_mul]
      have ha := _root_.abs_le.mp (Level.abs_le a)
      have hc := _root_.abs_le.mp (Level.abs_le c)
      have hi' := _root_.abs_le.mp hi
      have hj' := _root_.abs_le.mp hj
      refine S.towerElem_mem e (_root_.abs_le.mpr ⟨?_, ?_⟩) (_root_.abs_le.mpr ⟨?_, ?_⟩) _ <;>
        push_cast <;> omega
    | zero => rw [mul_zero]; exact zero_mem _
    | add s t _ _ hs ht => rw [mul_add]; exact add_mem hs ht
    | neg s _ hs => rw [mul_neg]; exact neg_mem hs
  | zero => rw [zero_mul, zero_mul]; exact zero_mem _
  | add s t _ _ hs ht => rw [add_mul, add_mul]; exact add_mem hs ht
  | neg s _ hs => rw [neg_mul, neg_mul]; exact neg_mem hs

/-- Absorption for matrices: `g M g'` has entries in `B_(m+w)(U)`. -/
theorem matrix_mul_mul_mem_tower {ι : Type*} [Fintype ι] (e : A) {m w : ℕ}
    {g g' M : Matrix ι ι R} (hg : ∀ p q, g p q ∈ S.propagation w)
    (hg' : ∀ p q, g' p q ∈ S.propagation w) (hM : ∀ p q, M p q ∈ S.tower e m) (p q : ι) :
    (g * M * g') p q ∈ S.tower e (m + w) := by
  simp only [Matrix.mul_apply, Finset.sum_mul]
  exact AddSubgroup.sum_mem _ fun k _ ↦ AddSubgroup.sum_mem _ fun l _ ↦
    S.mul_mul_mem_tower e (hg p l) (hg' k q) (hM l k)

/-- `M_ι(B_m(U))` is closed under products. -/
theorem matrix_mul_mem_tower {ι : Type*} [Fintype ι] {e : A} (he : IsIdempotentElem e) {m : ℕ}
    (hret : S.LongReturn e (2 * m)) {M N : Matrix ι ι R} (hM : ∀ p q, M p q ∈ S.tower e m)
    (hN : ∀ p q, N p q ∈ S.tower e m) (p q : ι) : (M * N) p q ∈ S.tower e m := by
  rw [Matrix.mul_apply]
  exact AddSubgroup.sum_mem _ fun k _ ↦ S.mul_mem_tower he hret (hM p k) (hN k q)

end TowerSetting
