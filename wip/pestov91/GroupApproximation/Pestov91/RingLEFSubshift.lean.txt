import GroupApproximation.Pestov91.SubshiftPeriodic
import Mathlib.Algebra.Ring.Periodic
import Mathlib.Data.ZMod.Basic
import Mathlib.Tactic.LinearCombination

/-!
# Pestov 9.1: periodic charts of a subshift

Section 3.1 of `research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`.

`HasPeriodicModels X` says that for every radius `k` and bound `B` there is an `N`-periodic
configuration `z`, `N ≥ B`, whose radius-`k` windows are exactly the radius-`k` windows of the
points of `X`, seams included.  Choosing for each position `n ∈ ℤ/N` a point `p_n ∈ X` that
agrees with `shift n z` on `[-k, k]` gives a chart:

* gluing, `(p_(n+i))_m = (p_n)_(i+m)` whenever `|m|, |i + m| ≤ k`, exact across the seams
  because `z` is periodic;
* covering, every point of `X` agrees on `[-k, k]` with some `p_n`.

The Toeplitz subshift has periodic models (`hasPeriodicModels_toeplitz`, from
`exists_periodic_model`).
-/

namespace GroupApproximation.Pestov91

open SymbolicDynamics.FullShift

/-- Periodic models of `X` at every radius `k` and above every period bound `B` (§3.1). -/
def HasPeriodicModels (X : Subshift Bool ℤ) : Prop :=
  ∀ k B : ℕ, ∃ (N : ℕ) (z : ℤ → Bool), 0 < N ∧ B ≤ N ∧ (∀ n : ℤ, z (n + N) = z n) ∧
    (∀ n : ℤ, ∃ y ∈ X.carrier, Agree k (shift n z) y) ∧
    (∀ y ∈ X.carrier, ∃ n : ℤ, Agree k y (shift n z))

/-- The Toeplitz subshift has periodic models. -/
theorem hasPeriodicModels_toeplitz : HasPeriodicModels toeplitzSubshift :=
  exists_periodic_model

/-- A subshift with periodic models is nonempty: the empty subshift fails the definition. -/
theorem HasPeriodicModels.nonempty {X : Subshift Bool ℤ} (h : HasPeriodicModels X) :
    X.carrier.Nonempty := by
  obtain ⟨N, z, -, -, -, hwin, -⟩ := h 0 0
  obtain ⟨y, hy, -⟩ := hwin 0
  exact ⟨y, hy⟩

/-- An `N`-periodic configuration takes equal values at indices congruent modulo `N`. -/
theorem periodic_apply_eq_of_intCast_eq {α : Type*} {N : ℕ} {y : ℤ → α}
    (hy : Function.Periodic y (N : ℤ)) {a b : ℤ} (h : (a : ZMod N) = (b : ZMod N)) :
    y a = y b := by
  obtain ⟨t, ht⟩ := (ZMod.intCast_eq_intCast_iff_dvd_sub a b N).mp h
  have hb : b = a + t * N := by linear_combination ht
  have h2 : y (a + t * N) = y a := by simpa using hy.int_mul t a
  rw [hb, h2]

/-- **Periodic charts** (§3.1): points `p_n ∈ X`, `n ∈ ℤ/N`, glued exactly across seams and
covering `X` at radius `k`. -/
theorem exists_periodicChart {X : Subshift Bool ℤ} {k N : ℕ} [NeZero N] {z : ℤ → Bool}
    (hz : Function.Periodic z (N : ℤ))
    (hwin : ∀ n : ℤ, ∃ y ∈ X.carrier, Agree k (shift n z) y)
    (hcov : ∀ y ∈ X.carrier, ∃ n : ℤ, Agree k y (shift n z)) :
    ∃ p : ZMod N → X.carrier,
      (∀ (n : ZMod N) (i m : ℤ), |m| ≤ k → |i + m| ≤ k →
        (p (n + i) : ℤ → Bool) m = (p n : ℤ → Bool) (i + m)) ∧
      (∀ y : X.carrier, ∃ n : ZMod N, Agree k (p n) y) := by
  choose q hqX hq using hwin
  refine ⟨fun n ↦ ⟨q (n.val : ℤ), hqX _⟩, fun n i m hm him ↦ ?_, fun y ↦ ?_⟩
  · show q ((n + (i : ZMod N)).val : ℤ) m = q (n.val : ℤ) (i + m)
    rw [← hq _ m hm, ← hq _ (i + m) him, shift_apply, shift_apply]
    refine periodic_apply_eq_of_intCast_eq hz ?_
    simp only [Int.cast_add, Int.cast_natCast, ZMod.natCast_zmod_val, add_assoc]
  · obtain ⟨n, hn⟩ := hcov y y.2
    refine ⟨(n : ZMod N), fun m hm ↦ ?_⟩
    show q ((n : ZMod N).val : ℤ) m = (y : ℤ → Bool) m
    rw [← hq _ m hm, hn m hm, shift_apply, shift_apply]
    refine periodic_apply_eq_of_intCast_eq hz ?_
    simp only [Int.cast_add, Int.cast_natCast, ZMod.natCast_zmod_val]

end GroupApproximation.Pestov91
