/-!
# The crossed product `LC(X, 𝔽₂) ⋊ ℤ`

The ring of the Pestov 9.1 witness is the algebraic crossed product of the locally constant
`𝔽₂`-valued functions on a space `X` by a homeomorphism `σ`:

  `R = LC(X, 𝔽₂) ⋊_σ ℤ`,  elements `∑ⱼ fⱼ uʲ` with finitely many `fⱼ ≠ 0`,
  `(f uⁱ)(g uʲ) = f · (g ∘ σ⁻ⁱ) · uⁱ⁺ʲ`,  so `u f u⁻¹ = f ∘ σ⁻¹` and `u χ_Y u⁻¹ = χ_(σ Y)`.

This is the convention of
`research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`, §1.2, where
`α(f) = f ∘ T⁻¹`.  Everything here holds for an arbitrary homeomorphism of an arbitrary
topological space; minimality and freeness enter only where they are used.

The ring is Mathlib's `SkewMonoidAlgebra` over `ℤ`, written multiplicatively, acting on
coefficients through the powers of `σ`.  The index type is the synonym `ShiftIndex σ` of
`Multiplicative ℤ`, so that this action is an instance and `Ring`, `Nontrivial` and the rest of
Mathlib's API come by instance search.  The API below is indexed by `ℤ`.

## Main declarations

* `CrossedProduct σ`: the ring.
* `CrossedProduct.single σ n f = f uⁿ`, `CrossedProduct.coeff r n`, `CrossedProduct.support r`,
  the normal form `CrossedProduct.sum_single`, and `CrossedProduct.ext`.
* `CrossedProduct.shift σ n f = f ∘ σ⁻ⁿ` and `CrossedProduct.single_mul_single`.
* `CrossedProduct.C σ : LC(X, 𝔽₂) →+* CrossedProduct σ`, injective.
* `CrossedProduct.u σ`, a unit with `uⁿ = single σ n 1` and `uⁿ (C f) u⁻ⁿ = C (shift σ n f)`.
-/

namespace GroupApproximation.Pestov91

/-- `𝔽₂` has two elements, available here without importing `Mathlib.Data.ZMod.Basic`. -/
instance zmodTwo_nontrivial : Nontrivial (ZMod 2) := ⟨⟨0, 1, by decide⟩⟩

variable {X : Type*} [TopologicalSpace X]

/-- Locally constant functions on a nonempty space, with values in a nontrivial type, form a
nontrivial type. -/
instance instNontrivialLocallyConstant {Y : Type*} [Nontrivial Y] [Nonempty X] :
    Nontrivial (LocallyConstant X Y) := by
  obtain ⟨y₁, y₂, h⟩ := exists_pair_ne Y
  exact ⟨⟨LocallyConstant.const X y₁, LocallyConstant.const X y₂,
    fun e => h (congrArg (fun g : LocallyConstant X Y => g (Classical.arbitrary X)) e)⟩⟩

/-- The index group `ℤ` of the crossed product over `σ`, written multiplicatively and tagged
with `σ`, so that the action of `σ` on coefficients is an instance. -/
def ShiftIndex (_σ : X ≃ₜ X) : Type := Multiplicative ℤ

namespace ShiftIndex

variable (σ : X ≃ₜ X)

instance : CommGroup (ShiftIndex σ) := inferInstanceAs (CommGroup (Multiplicative ℤ))

/-- The exponent `n : ℤ` as an index. -/
def ofInt : ℤ ≃ ShiftIndex σ := Multiplicative.ofAdd

theorem ofInt_add (m n : ℤ) : ofInt σ (m + n) = ofInt σ m * ofInt σ n := rfl

theorem ofInt_zero : ofInt σ 0 = 1 := rfl

instance : Infinite (ShiftIndex σ) := Infinite.of_injective (ofInt σ) (ofInt σ).injective

end ShiftIndex

variable (X) in
/-- A homeomorphism `h` acts on locally constant functions by the ring automorphism
`f ↦ f ∘ h⁻¹`. -/
def homeomorphRingAut : (X ≃ₜ X) →* RingAut (LocallyConstant X (ZMod 2)) where
  toFun h := LocallyConstant.congrLeftRingEquiv h
  map_one' := RingEquiv.ext fun _ => LocallyConstant.ext fun _ => rfl
  map_mul' _ _ := RingEquiv.ext fun _ => LocallyConstant.ext fun _ => rfl

/-- The action of the index group on coefficients, `n ↦ (f ↦ f ∘ σ⁻ⁿ)`. -/
def shiftAut (σ : X ≃ₜ X) : ShiftIndex σ →* RingAut (LocallyConstant X (ZMod 2)) :=
  (homeomorphRingAut X).comp (zpowersHom (X ≃ₜ X) σ)

instance (σ : X ≃ₜ X) : MulSemiringAction (ShiftIndex σ) (LocallyConstant X (ZMod 2)) :=
  MulSemiringAction.compHom _ (shiftAut σ)

/-- The algebraic crossed product `LC(X, 𝔽₂) ⋊_σ ℤ`: finitely supported sums `∑ₙ fₙ uⁿ` with
`(f uⁱ)(g uʲ) = f · (g ∘ σ⁻ⁱ) · uⁱ⁺ʲ`. -/
abbrev CrossedProduct (σ : X ≃ₜ X) : Type _ :=
  SkewMonoidAlgebra (LocallyConstant X (ZMod 2)) (ShiftIndex σ)

namespace CrossedProduct

section Shift

variable (σ : X ≃ₜ X)

/-- `shift σ n f = f ∘ σ⁻ⁿ`, the automorphism by which `uⁿ` conjugates coefficients. -/
def shift (n : ℤ) : LocallyConstant X (ZMod 2) ≃+* LocallyConstant X (ZMod 2) :=
  shiftAut σ (ShiftIndex.ofInt σ n)

theorem ofInt_smul (n : ℤ) (f : LocallyConstant X (ZMod 2)) :
    ShiftIndex.ofInt σ n • f = shift σ n f := rfl

theorem shift_apply (n : ℤ) (f : LocallyConstant X (ZMod 2)) (x : X) :
    shift σ n f x = f ((σ ^ (-n)) x) := by
  change f ((σ ^ n).symm x) = _
  rw [zpow_neg, Homeomorph.inv_apply]

theorem shift_zero (f : LocallyConstant X (ZMod 2)) : shift σ 0 f = f :=
  LocallyConstant.ext fun x => by rw [shift_apply, neg_zero, zpow_zero, Homeomorph.one_apply]

theorem shift_add (m n : ℤ) (f : LocallyConstant X (ZMod 2)) :
    shift σ (m + n) f = shift σ m (shift σ n f) :=
  DFunLike.congr_fun (map_mul (shiftAut σ) (ShiftIndex.ofInt σ m) (ShiftIndex.ofInt σ n)) f

/-- The sign convention, as in the artifact's `u χ_Y u⁻¹ = χ_(T Y)`: `shift σ n` sends the
indicator of `Y` to the indicator of `(σ⁻ⁿ)⁻¹ Y = σⁿ Y`. -/
theorem shift_charFn (n : ℤ) {Y : Set X} (hY : IsClopen Y) :
    shift σ n (LocallyConstant.charFn (ZMod 2) hY) =
      LocallyConstant.charFn (ZMod 2) (hY.preimage (σ ^ (-n)).continuous) :=
  LocallyConstant.ext fun x => by rw [shift_apply]; rfl

end Shift

variable {σ : X ≃ₜ X}

section Basic

variable (σ) in
/-- `single σ n f` is the monomial `f uⁿ`. -/
noncomputable def single (n : ℤ) (f : LocallyConstant X (ZMod 2)) : CrossedProduct σ :=
  SkewMonoidAlgebra.single (ShiftIndex.ofInt σ n) f

/-- The coefficient of `uⁿ` in `r`. -/
def coeff (r : CrossedProduct σ) (n : ℤ) : LocallyConstant X (ZMod 2) :=
  SkewMonoidAlgebra.coeff r (ShiftIndex.ofInt σ n)

theorem coeff_single_same (n : ℤ) (f : LocallyConstant X (ZMod 2)) :
    coeff (single σ n f) n = f :=
  Finsupp.single_eq_same

theorem coeff_single_of_ne {m n : ℤ} (h : m ≠ n) (f : LocallyConstant X (ZMod 2)) :
    coeff (single σ m f) n = 0 :=
  Finsupp.single_eq_of_ne ((ShiftIndex.ofInt σ).injective.ne h.symm)

theorem coeff_single (m n : ℤ) (f : LocallyConstant X (ZMod 2)) :
    coeff (single σ m f) n = if m = n then f else 0 := by
  split_ifs with h
  · subst h
    exact coeff_single_same m f
  · exact coeff_single_of_ne h f

theorem coeff_zero (n : ℤ) : coeff (0 : CrossedProduct σ) n = 0 := rfl

theorem coeff_add (r s : CrossedProduct σ) (n : ℤ) :
    coeff (r + s) n = coeff r n + coeff s n :=
  DFunLike.congr_fun (SkewMonoidAlgebra.coeff_add r s) (ShiftIndex.ofInt σ n)

theorem coeff_neg (r : CrossedProduct σ) (n : ℤ) : coeff (-r) n = -coeff r n :=
  DFunLike.congr_fun (SkewMonoidAlgebra.coeff_neg r) (ShiftIndex.ofInt σ n)

theorem coeff_sub (r s : CrossedProduct σ) (n : ℤ) :
    coeff (r - s) n = coeff r n - coeff s n :=
  DFunLike.congr_fun (SkewMonoidAlgebra.coeff_sub r s) (ShiftIndex.ofInt σ n)

/-- Two elements with the same coefficients are equal. -/
theorem ext {r s : CrossedProduct σ} (h : ∀ n : ℤ, coeff r n = coeff s n) : r = s :=
  SkewMonoidAlgebra.ext fun a => h ((ShiftIndex.ofInt σ).symm a)

theorem single_zero (n : ℤ) : single σ n 0 = 0 :=
  SkewMonoidAlgebra.single_zero _

theorem single_add (n : ℤ) (f g : LocallyConstant X (ZMod 2)) :
    single σ n (f + g) = single σ n f + single σ n g :=
  SkewMonoidAlgebra.single_add _ _ _

theorem single_injective (n : ℤ) : Function.Injective (single σ n) :=
  SkewMonoidAlgebra.single_injective (ShiftIndex.ofInt σ n)

/-- The exponents `n` with `coeff r n ≠ 0`. -/
def support (r : CrossedProduct σ) : Finset ℤ :=
  (SkewMonoidAlgebra.support r).map (ShiftIndex.ofInt σ).symm.toEmbedding

theorem mem_support {r : CrossedProduct σ} {n : ℤ} : n ∈ support r ↔ coeff r n ≠ 0 :=
  Finset.mem_map_equiv.trans SkewMonoidAlgebra.mem_support_iff

/-- **Normal form.** Every element is the finite sum of its monomials. -/
theorem sum_single (r : CrossedProduct σ) : ∑ n ∈ support r, single σ n (coeff r n) = r := by
  rw [support, Finset.sum_map]
  exact SkewMonoidAlgebra.sum_single r

/-- Induction over sums of monomials. -/
@[elab_as_elim]
theorem induction_on {p : CrossedProduct σ → Prop} (r : CrossedProduct σ) (zero : p 0)
    (single : ∀ (n : ℤ) (f : LocallyConstant X (ZMod 2)), p (single σ n f))
    (add : ∀ r s, p r → p s → p (r + s)) : p r :=
  SkewMonoidAlgebra.induction_on r zero (fun a f => single ((ShiftIndex.ofInt σ).symm a) f) add

end Basic

section Lift

variable {S : Type*} [NonUnitalNonAssocSemiring S]

variable (σ) in
/-- The additive map `∑ₙ fₙ uⁿ ↦ ∑ₙ D fₙ · P n` for an additive `D` and any `P : ℤ → S`: the
shape of the local matrix models `ψ_k(∑ fⱼ uʲ) = ∑ D_k(fⱼ) P_kʲ`. -/
noncomputable def liftAdd (D : LocallyConstant X (ZMod 2) →+ S) (P : ℤ → S) :
    CrossedProduct σ →+ S :=
  SkewMonoidAlgebra.liftNC D fun a => P ((ShiftIndex.ofInt σ).symm a)

variable (σ) in
theorem liftAdd_single (D : LocallyConstant X (ZMod 2) →+ S) (P : ℤ → S) (n : ℤ)
    (f : LocallyConstant X (ZMod 2)) : liftAdd σ D P (single σ n f) = D f * P n :=
  SkewMonoidAlgebra.liftNC_single _ _ _ _

theorem liftAdd_apply (D : LocallyConstant X (ZMod 2) →+ S) (P : ℤ → S) (r : CrossedProduct σ) :
    liftAdd σ D P r = ∑ n ∈ support r, D (coeff r n) * P n :=
  calc liftAdd σ D P r = liftAdd σ D P (∑ n ∈ support r, single σ n (coeff r n)) := by
        rw [sum_single]
    _ = ∑ n ∈ support r, D (coeff r n) * P n := by
        rw [map_sum]
        exact Finset.sum_congr rfl fun n _ => liftAdd_single σ D P n (coeff r n)

end Lift

section Mul

theorem one_def : (1 : CrossedProduct σ) = single σ 0 1 := rfl

/-- `(f uⁱ)(g uʲ) = f · (g ∘ σ⁻ⁱ) · uⁱ⁺ʲ`. -/
theorem single_mul_single (i j : ℤ) (f g : LocallyConstant X (ZMod 2)) :
    single σ i f * single σ j g = single σ (i + j) (f * shift σ i g) := by
  refine SkewMonoidAlgebra.single_mul_single.trans ?_
  rfl

theorem coeff_single_mul (i : ℤ) (f : LocallyConstant X (ZMod 2)) (r : CrossedProduct σ)
    (n : ℤ) : coeff (single σ i f * r) n = f * shift σ i (coeff r (n - i)) := by
  rw [sub_eq_neg_add]
  exact SkewMonoidAlgebra.coeff_single_mul f (ShiftIndex.ofInt σ i) r (ShiftIndex.ofInt σ n)

theorem coeff_mul_single (r : CrossedProduct σ) (j : ℤ) (g : LocallyConstant X (ZMod 2))
    (n : ℤ) : coeff (r * single σ j g) n = coeff r (n - j) * shift σ (n - j) g := by
  rw [sub_eq_add_neg]
  exact SkewMonoidAlgebra.coeff_mul_single r g (ShiftIndex.ofInt σ j) (ShiftIndex.ofInt σ n)

variable (σ) in
/-- The coefficient embedding `f ↦ f u⁰`. -/
noncomputable def C : LocallyConstant X (ZMod 2) →+* CrossedProduct σ :=
  SkewMonoidAlgebra.singleOneRingHom

theorem C_apply (f : LocallyConstant X (ZMod 2)) : C σ f = single σ 0 f := rfl

theorem C_injective : Function.Injective (C σ) := fun _ _ h =>
  SkewMonoidAlgebra.single_injective (ShiftIndex.ofInt σ 0) h

variable (σ) in
/-- The unit `u = 1 u¹`, with `u f u⁻¹ = f ∘ σ⁻¹`. -/
noncomputable def u : (CrossedProduct σ)ˣ where
  val := single σ 1 1
  inv := single σ (-1) 1
  val_inv := by rw [single_mul_single, map_one, mul_one, add_neg_cancel, ← one_def]
  inv_val := by rw [single_mul_single, map_one, mul_one, neg_add_cancel, ← one_def]

theorem val_u : (u σ : CrossedProduct σ) = single σ 1 1 := rfl

theorem val_u_inv : (↑(u σ)⁻¹ : CrossedProduct σ) = single σ (-1) 1 := rfl

theorem val_u_zpow (n : ℤ) : (↑(u σ ^ n) : CrossedProduct σ) = single σ n 1 := by
  induction n using Int.induction_on with
  | zero => rw [zpow_zero, Units.val_one, one_def]
  | succ i ih => rw [zpow_add_one, Units.val_mul, ih, val_u, single_mul_single, map_one, mul_one]
  | pred i ih =>
    rw [zpow_sub_one, Units.val_mul, ih, val_u_inv, single_mul_single, map_one, mul_one,
      sub_eq_add_neg]

/-- The monomial `f uⁿ` is `C f · uⁿ`. -/
theorem single_eq_C_mul_u_zpow (n : ℤ) (f : LocallyConstant X (ZMod 2)) :
    single σ n f = C σ f * ↑(u σ ^ n) := by
  rw [val_u_zpow, C_apply, single_mul_single, map_one, mul_one, zero_add]

variable (σ) in
/-- `uʲ · C a = C (shift σ j a) · uʲ`, the shape `hconj` of `RingSimple` and `SplitSimplicity`. -/
theorem u_zpow_mul_C (j : ℤ) (a : LocallyConstant X (ZMod 2)) :
    ((u σ ^ j : (CrossedProduct σ)ˣ) : CrossedProduct σ) * C σ a =
      C σ (shift σ j a) * ((u σ ^ j : (CrossedProduct σ)ˣ) : CrossedProduct σ) := by
  rw [val_u_zpow, C_apply, C_apply, single_mul_single, single_mul_single, map_one, one_mul,
    mul_one, add_zero, zero_add]

/-- Conjugation by `uⁿ` on coefficients is `shift σ n`. -/
theorem u_zpow_mul_C_mul_inv (n : ℤ) (f : LocallyConstant X (ZMod 2)) :
    ↑(u σ ^ n) * C σ f * ↑(u σ ^ n)⁻¹ = C σ (shift σ n f) := by
  rw [u_zpow_mul_C, Units.mul_inv_cancel_right]

/-- **Normal form** in terms of the coefficient embedding and the powers of `u`. -/
theorem sum_C_mul_u_zpow (r : CrossedProduct σ) :
    ∑ n ∈ support r, C σ (coeff r n) * ↑(u σ ^ n) = r :=
  calc ∑ n ∈ support r, C σ (coeff r n) * ↑(u σ ^ n)
      = ∑ n ∈ support r, single σ n (coeff r n) :=
        Finset.sum_congr rfl fun n _ => (single_eq_C_mul_u_zpow n (coeff r n)).symm
    _ = r := sum_single r

variable (σ) in
/-- The normal form in the shape `hspan` of `RingSimple` and `SplitSimplicity`. -/
theorem exists_eq_sum_C_mul_u_zpow (x : CrossedProduct σ) :
    ∃ (S : Finset ℤ) (c : ℤ → LocallyConstant X (ZMod 2)),
      x = ∑ j ∈ S, C σ (c j) * ((u σ ^ j : (CrossedProduct σ)ˣ) : CrossedProduct σ) :=
  ⟨support x, coeff x, (sum_C_mul_u_zpow x).symm⟩

end Mul

section ModelTests

/-- The crossed product over a nonempty space is a nontrivial ring. -/
theorem nontrivial [Nonempty X] : Nontrivial (CrossedProduct σ) := inferInstance

/-- Model test: `u ≠ 1`, since the coefficient of `u⁰` in `u` is `0`. -/
theorem val_u_ne_one [Nonempty X] : (u σ : CrossedProduct σ) ≠ 1 := by
  intro h
  have h0 : coeff (u σ : CrossedProduct σ) 0 = coeff (1 : CrossedProduct σ) 0 := by rw [h]
  rw [val_u, one_def, coeff_single_same, coeff_single_of_ne one_ne_zero] at h0
  exact zero_ne_one h0

/-- Over the identity homeomorphism every `shift` is the identity. -/
theorem shift_one (n : ℤ) (f : LocallyConstant X (ZMod 2)) : shift (1 : X ≃ₜ X) n f = f :=
  LocallyConstant.ext fun x => by rw [shift_apply, one_zpow, Homeomorph.one_apply]

/-- Model test, degenerate case: over the identity homeomorphism `u` commutes with the
coefficients. -/
theorem val_u_mul_C_one (f : LocallyConstant X (ZMod 2)) :
    ↑(u (1 : X ≃ₜ X)) * C (1 : X ≃ₜ X) f = C (1 : X ≃ₜ X) f * ↑(u (1 : X ≃ₜ X)) := by
  have h := u_zpow_mul_C (σ := (1 : X ≃ₜ X)) 1 f
  rwa [zpow_one, shift_one] at h

end ModelTests

end CrossedProduct
