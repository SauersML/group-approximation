import GroupApproximation.Leavitt.ElementaryGroup
import Mathlib.Data.Matrix.Basis

/-!
# A copy of `M_{ι×n}(K)` inside `M_ι(R)` from a system of matrix units

Case (c) of the root detection for the Pestov 9.1 witness (§4.2(c) of
`research/artifacts/simple-kazhdan-lef-subshift-elementary-group-2026-09-12.md`)
works inside a tower `W, TW, …` of pairwise disjoint clopen levels.  The elements
`E a b = χ_(T^a W) u^(a-b)` multiply like matrix units,
`E a b * E c d = δ_bc E a d`, and they give a unital multiplicative map

  `ι_W : M_{ι×n}(K) → M_ι(R)`,  `ι_W(M)_{pq} = δ_pq (1 - f) + ∑_{a,b} M_{(p,a),(q,b)} E a b`,

where `f = ∑ₐ E a a` is the unit of the corner.  This module builds `ι_W` for any
system of matrix units in any `K`-algebra `R`.  Nothing about subshifts enters.

* `cornerHom E : M_n(K) →+ R` is multiplicative (`cornerHom_mul`) and sends `1` to `f`.
* `blockEmbed E` applies `cornerHom` blockwise; it is additive and multiplicative, and
  `blockEmbed E 1 = diag(f)`.
* `towerEmbed E M = diag(1 - f) + blockEmbed E M` is unital and multiplicative
  (`towerHom`), so it restricts to a group map on units (`towerUnits`).
-/

namespace GroupApproximation
namespace Pestov91

open Matrix

variable {K R : Type*} [CommRing K] [Ring R] [Algebra K R]
variable {ι n : Type*} [Fintype ι] [DecidableEq ι] [Fintype n] [DecidableEq n]

/-- A system of matrix units in `R` indexed by `n`: `E a b * E c d = δ_bc E a d`. -/
def IsMatrixUnits (E : n → n → R) : Prop :=
  ∀ a b c d, E a b * E c d = if b = c then E a d else 0

/-- The unit `f = ∑ₐ E a a` of the corner spanned by a system of matrix units. -/
def matrixUnitsSum (E : n → n → R) : R := ∑ a, E a a

namespace IsMatrixUnits

variable {E : n → n → R}

theorem sum_mul (hE : IsMatrixUnits E) (a b : n) : matrixUnitsSum E * E a b = E a b := by
  rw [matrixUnitsSum, Finset.sum_mul, Finset.sum_eq_single a]
  · rw [hE a a a b, if_pos rfl]
  · intro c _ hca
    rw [hE c c a b, if_neg hca]
  · intro ha
    exact absurd (Finset.mem_univ a) ha

theorem mul_sum (hE : IsMatrixUnits E) (a b : n) : E a b * matrixUnitsSum E = E a b := by
  rw [matrixUnitsSum, Finset.mul_sum, Finset.sum_eq_single b]
  · rw [hE a b b b, if_pos rfl]
  · intro c _ hcb
    rw [hE a b c c, if_neg (Ne.symm hcb)]
  · intro hb
    exact absurd (Finset.mem_univ b) hb

theorem sum_mul_sum (hE : IsMatrixUnits E) : matrixUnitsSum E * matrixUnitsSum E = matrixUnitsSum E := by
  conv_lhs => rw [matrixUnitsSum, Finset.mul_sum]
  exact Finset.sum_congr rfl fun a _ ↦ hE.sum_mul a a

end IsMatrixUnits

variable (K) in
/-- The corner map `A ↦ ∑_{a,b} A a b • E a b`. -/
def cornerHom (E : n → n → R) : Matrix n n K →+ R where
  toFun A := ∑ a, ∑ b, algebraMap K R (A a b) * E a b
  map_zero' := by simp
  map_add' A B := by
    simp only [Matrix.add_apply, map_add, add_mul, Finset.sum_add_distrib]

theorem cornerHom_apply (E : n → n → R) (A : Matrix n n K) :
    cornerHom K E A = ∑ a, ∑ b, algebraMap K R (A a b) * E a b := rfl

variable {E : n → n → R}

theorem cornerHom_one : cornerHom K E (1 : Matrix n n K) = matrixUnitsSum E := by
  rw [cornerHom_apply, matrixUnitsSum]
  refine Finset.sum_congr rfl fun a _ ↦ ?_
  rw [Finset.sum_eq_single a]
  · simp
  · intro b _ hba
    simp [Ne.symm hba]
  · intro ha
    exact absurd (Finset.mem_univ a) ha

theorem cornerHom_mul (hE : IsMatrixUnits E) (A B : Matrix n n K) :
    cornerHom K E (A * B) = cornerHom K E A * cornerHom K E B := by
  have hterm : ∀ a b c d, algebraMap K R (A a b) * E a b * (algebraMap K R (B c d) * E c d) =
      if b = c then algebraMap K R (A a b) * algebraMap K R (B c d) * E a d else 0 := by
    intro a b c d
    rw [mul_assoc, ← mul_assoc (E a b), ← Algebra.commutes (B c d) (E a b), mul_assoc,
      hE a b c d, ← mul_assoc]
    split_ifs
    · rfl
    · rw [mul_zero]
  have hinner : ∀ a b, ∑ c, ∑ d, algebraMap K R (A a b) * E a b *
      (algebraMap K R (B c d) * E c d) =
      ∑ d, algebraMap K R (A a b) * algebraMap K R (B b d) * E a d := by
    intro a b
    rw [Finset.sum_eq_single b]
    · refine Finset.sum_congr rfl fun d _ ↦ ?_
      rw [hterm, if_pos rfl]
    · intro c _ hcb
      refine Finset.sum_eq_zero fun d _ ↦ ?_
      rw [hterm, if_neg (Ne.symm hcb)]
    · intro hb
      exact absurd (Finset.mem_univ b) hb
  rw [cornerHom_apply, cornerHom_apply, cornerHom_apply]
  simp only [Finset.sum_mul]
  simp only [Finset.mul_sum]
  simp only [hinner]
  refine Finset.sum_congr rfl fun a _ ↦ ?_
  conv_rhs => rw [Finset.sum_comm]
  refine Finset.sum_congr rfl fun d _ ↦ ?_
  rw [Matrix.mul_apply, map_sum, Finset.sum_mul]
  simp only [map_mul]

theorem sum_mul_cornerHom (hE : IsMatrixUnits E) (A : Matrix n n K) :
    matrixUnitsSum E * cornerHom K E A = cornerHom K E A := by
  rw [cornerHom_apply, Finset.mul_sum]
  refine Finset.sum_congr rfl fun a _ ↦ ?_
  rw [Finset.mul_sum]
  refine Finset.sum_congr rfl fun b _ ↦ ?_
  rw [← mul_assoc, ← Algebra.commutes (A a b) (matrixUnitsSum E), mul_assoc, hE.sum_mul]

theorem cornerHom_mul_sum (hE : IsMatrixUnits E) (A : Matrix n n K) :
    cornerHom K E A * matrixUnitsSum E = cornerHom K E A := by
  rw [cornerHom_apply, Finset.sum_mul]
  refine Finset.sum_congr rfl fun a _ ↦ ?_
  rw [Finset.sum_mul]
  refine Finset.sum_congr rfl fun b _ ↦ ?_
  rw [mul_assoc, hE.mul_sum]

/-- Blockwise corner map: `blockEmbed E M p q = cornerHom E (M_{(p,·),(q,·)})`. -/
def blockEmbed (E : n → n → R) (M : Matrix (ι × n) (ι × n) K) : Matrix ι ι R :=
  Matrix.of fun p q ↦ cornerHom K E (Matrix.of fun a b ↦ M (p, a) (q, b))

theorem blockEmbed_apply (E : n → n → R) (M : Matrix (ι × n) (ι × n) K) (p q : ι) :
    blockEmbed E M p q = cornerHom K E (Matrix.of fun a b ↦ M (p, a) (q, b)) := rfl

theorem blockEmbed_add (M M' : Matrix (ι × n) (ι × n) K) :
    blockEmbed E (M + M') = blockEmbed E M + blockEmbed E M' := by
  ext p q
  rw [Matrix.add_apply, blockEmbed_apply, blockEmbed_apply, blockEmbed_apply, ← map_add]
  congr 1
  ext a b
  simp only [Matrix.of_apply, Matrix.add_apply]

theorem blockEmbed_single (p q : ι) (a b : n) (c : K) :
    blockEmbed E (Matrix.single (p, a) (q, b) c) = Matrix.single p q (algebraMap K R c * E a b) := by
  ext p' q'
  rw [blockEmbed_apply, cornerHom_apply]
  by_cases hpq : p = p' ∧ q = q'
  · obtain ⟨rfl, rfl⟩ := hpq
    rw [Matrix.single_apply_same, Finset.sum_eq_single a]
    · rw [Finset.sum_eq_single b]
      · rw [Matrix.of_apply, Matrix.single_apply_same]
      · intro b' _ hb'
        rw [Matrix.of_apply, Matrix.single_apply_of_ne, map_zero, zero_mul]
        exact fun h ↦ hb' (congrArg Prod.snd h.2).symm
      · intro hb
        exact absurd (Finset.mem_univ b) hb
    · intro a' _ ha'
      refine Finset.sum_eq_zero fun b' _ ↦ ?_
      rw [Matrix.of_apply, Matrix.single_apply_of_ne, map_zero, zero_mul]
      exact fun h ↦ ha' (congrArg Prod.snd h.1).symm
    · intro ha
      exact absurd (Finset.mem_univ a) ha
  · rw [Matrix.single_apply_of_ne (h := hpq)]
    refine Finset.sum_eq_zero fun a' _ ↦ Finset.sum_eq_zero fun b' _ ↦ ?_
    rw [Matrix.of_apply, Matrix.single_apply_of_ne, map_zero, zero_mul]
    exact fun h ↦ hpq ⟨congrArg Prod.fst h.1, congrArg Prod.fst h.2⟩

theorem blockEmbed_one :
    blockEmbed E (1 : Matrix (ι × n) (ι × n) K) = diagonal fun _ ↦ matrixUnitsSum E := by
  ext p q
  rw [blockEmbed_apply, Matrix.diagonal_apply]
  split_ifs with hpq
  · subst hpq
    rw [← cornerHom_one (K := K)]
    congr 1
    ext a b
    simp [Matrix.one_apply]
  · have h0 : (Matrix.of fun a b ↦ (1 : Matrix (ι × n) (ι × n) K) (p, a) (q, b)) = 0 := by
      ext a b
      simp [Matrix.one_apply, hpq]
    rw [h0, map_zero]

theorem block_of_mul (M M' : Matrix (ι × n) (ι × n) K) (p q : ι) :
    (Matrix.of fun a b ↦ (M * M') (p, a) (q, b) : Matrix n n K) =
      ∑ r, (Matrix.of fun a b ↦ M (p, a) (r, b)) * (Matrix.of fun a b ↦ M' (r, a) (q, b)) := by
  ext a d
  simp only [Matrix.of_apply, Matrix.mul_apply, Matrix.sum_apply, Fintype.sum_prod_type]

theorem blockEmbed_mul (hE : IsMatrixUnits E) (M M' : Matrix (ι × n) (ι × n) K) :
    blockEmbed E (M * M') = blockEmbed E M * blockEmbed E M' := by
  ext p q
  rw [Matrix.mul_apply, blockEmbed_apply, block_of_mul, map_sum]
  refine Finset.sum_congr rfl fun r _ ↦ ?_
  rw [cornerHom_mul hE, blockEmbed_apply, blockEmbed_apply]

theorem diagonal_sum_mul_blockEmbed (hE : IsMatrixUnits E) (M : Matrix (ι × n) (ι × n) K) :
    (diagonal fun _ ↦ matrixUnitsSum E) * blockEmbed E M = blockEmbed E M := by
  ext p q
  simp only [Matrix.diagonal_mul, blockEmbed_apply]
  exact sum_mul_cornerHom hE _

theorem blockEmbed_mul_diagonal_sum (hE : IsMatrixUnits E) (M : Matrix (ι × n) (ι × n) K) :
    blockEmbed E M * (diagonal fun _ ↦ matrixUnitsSum E) = blockEmbed E M := by
  ext p q
  simp only [Matrix.mul_diagonal, blockEmbed_apply]
  exact cornerHom_mul_sum hE _

theorem diagonal_compl_mul_blockEmbed (hE : IsMatrixUnits E) (M : Matrix (ι × n) (ι × n) K) :
    (diagonal fun _ ↦ 1 - matrixUnitsSum E) * blockEmbed E M = 0 := by
  ext p q
  simp only [Matrix.diagonal_mul, blockEmbed_apply, Matrix.zero_apply, sub_mul, one_mul]
  rw [sum_mul_cornerHom hE, sub_self]

theorem blockEmbed_mul_diagonal_compl (hE : IsMatrixUnits E) (M : Matrix (ι × n) (ι × n) K) :
    blockEmbed E M * (diagonal fun _ ↦ 1 - matrixUnitsSum E) = 0 := by
  ext p q
  simp only [Matrix.mul_diagonal, blockEmbed_apply, Matrix.zero_apply, mul_sub, mul_one]
  rw [cornerHom_mul_sum hE, sub_self]

/-- `ι_W(M) = diag(1 - f) + blockEmbed E M`, the unital copy of `M_{ι×n}(K)`. -/
def towerEmbed (E : n → n → R) (M : Matrix (ι × n) (ι × n) K) : Matrix ι ι R :=
  (diagonal fun _ ↦ 1 - matrixUnitsSum E) + blockEmbed E M

theorem diagonal_compl_eq (E : n → n → R) :
    (diagonal fun _ : ι ↦ 1 - matrixUnitsSum E) = 1 - diagonal fun _ ↦ matrixUnitsSum E := by
  rw [← Matrix.diagonal_one, Matrix.diagonal_sub]

theorem towerEmbed_one : towerEmbed E (1 : Matrix (ι × n) (ι × n) K) = 1 := by
  rw [towerEmbed, blockEmbed_one, Matrix.diagonal_add]
  simp only [sub_add_cancel]
  exact Matrix.diagonal_one

theorem towerEmbed_add_right (M M' : Matrix (ι × n) (ι × n) K) :
    towerEmbed E (M + M') = towerEmbed E M + blockEmbed E M' := by
  rw [towerEmbed, towerEmbed, blockEmbed_add, add_assoc]

theorem towerEmbed_mul (hE : IsMatrixUnits E) (M M' : Matrix (ι × n) (ι × n) K) :
    towerEmbed E (M * M') = towerEmbed E M * towerEmbed E M' := by
  have hD : ((diagonal fun _ : ι ↦ 1 - matrixUnitsSum E) * diagonal fun _ ↦ 1 - matrixUnitsSum E) =
      diagonal fun _ ↦ 1 - matrixUnitsSum E := by
    rw [Matrix.diagonal_mul_diagonal]
    congr 1
    funext _
    rw [sub_mul, one_mul, mul_sub, mul_one, hE.sum_mul_sum, sub_self, sub_zero]
  simp only [towerEmbed, add_mul, mul_add, hD, diagonal_compl_mul_blockEmbed hE,
    blockEmbed_mul_diagonal_compl hE, blockEmbed_mul hE, zero_add, add_zero]

theorem towerEmbed_mul_diagonal_sum (hE : IsMatrixUnits E) (M : Matrix (ι × n) (ι × n) K) :
    towerEmbed E M * (diagonal fun _ ↦ matrixUnitsSum E) = blockEmbed E M := by
  have hQP : ((diagonal fun _ : ι ↦ 1 - matrixUnitsSum E) *
      diagonal fun _ ↦ matrixUnitsSum E) = 0 := by
    rw [Matrix.diagonal_mul_diagonal, ← Matrix.diagonal_zero]
    congr 1
    funext _
    rw [sub_mul, one_mul, hE.sum_mul_sum, sub_self]
  rw [towerEmbed, add_mul, hQP, zero_add, blockEmbed_mul_diagonal_sum hE]

theorem diagonal_sum_mul_towerEmbed (hE : IsMatrixUnits E) (M : Matrix (ι × n) (ι × n) K) :
    (diagonal fun _ ↦ matrixUnitsSum E) * towerEmbed E M = blockEmbed E M := by
  have hPQ : ((diagonal fun _ : ι ↦ matrixUnitsSum E) *
      diagonal fun _ ↦ 1 - matrixUnitsSum E) = 0 := by
    rw [Matrix.diagonal_mul_diagonal, ← Matrix.diagonal_zero]
    congr 1
    funext _
    rw [mul_sub, mul_one, hE.sum_mul_sum, sub_self]
  rw [towerEmbed, mul_add, hPQ, zero_add, diagonal_sum_mul_blockEmbed hE]

theorem towerEmbed_mul_compl (hE : IsMatrixUnits E) (M : Matrix (ι × n) (ι × n) K) :
    towerEmbed E M * (1 - diagonal fun _ ↦ matrixUnitsSum E) =
      1 - diagonal fun _ ↦ matrixUnitsSum E := by
  rw [mul_sub, mul_one, towerEmbed_mul_diagonal_sum hE, towerEmbed, diagonal_compl_eq]
  abel

theorem compl_mul_towerEmbed (hE : IsMatrixUnits E) (M : Matrix (ι × n) (ι × n) K) :
    (1 - diagonal fun _ ↦ matrixUnitsSum E) * towerEmbed E M =
      1 - diagonal fun _ ↦ matrixUnitsSum E := by
  rw [sub_mul, one_mul, diagonal_sum_mul_towerEmbed hE, towerEmbed, diagonal_compl_eq]
  abel

theorem towerEmbed_mul_blockEmbed (hE : IsMatrixUnits E) (M M' : Matrix (ι × n) (ι × n) K) :
    towerEmbed E M * blockEmbed E M' = blockEmbed E M * blockEmbed E M' := by
  rw [towerEmbed, add_mul, diagonal_compl_mul_blockEmbed hE, zero_add]

variable (K ι) in
/-- `ι_W` as a monoid homomorphism `M_{ι×n}(K) →* M_ι(R)`. -/
def towerHom (hE : IsMatrixUnits E) : Matrix (ι × n) (ι × n) K →* Matrix ι ι R where
  toFun := towerEmbed E
  map_one' := towerEmbed_one
  map_mul' := towerEmbed_mul hE

variable (K ι) in
/-- `ι_W` on invertible matrices, `GL_{ι×n}(K) →* GL_ι(R)`. -/
def towerUnits (hE : IsMatrixUnits E) : (Matrix (ι × n) (ι × n) K)ˣ →* (Matrix ι ι R)ˣ :=
  Units.map (towerHom K ι hE)

theorem towerUnits_coe (hE : IsMatrixUnits E) (ε : (Matrix (ι × n) (ι × n) K)ˣ) :
    ((towerUnits K ι hE ε : (Matrix ι ι R)ˣ) : Matrix ι ι R) =
      towerEmbed E (ε : Matrix (ι × n) (ι × n) K) := rfl

end Pestov91
end GroupApproximation
