/-- **The endpoint from ring-level leaves.** Every binder is a ring fact that a named lane proves for
`WitnessRing` (blueprint v2, G2–G7). -/
theorem exists_infinite_simple_kazhdan_lef_of_ringLeaves {R : Type} [Ring R] [IsSimpleRing R] [Infinite R]
    (hR : ∀ c : R, (∀ r : R, r * c = c * r) → c = 0 ∨ c = 1)
    {A : Type} [CommRing A] (φ : A →+* R) (u : Rˣ) (σ : ℤ → A → A)
    (hconj : ∀ (j : ℤ) (a : A), ((u ^ j : Rˣ) : R) * φ a = φ (σ j a) * ((u ^ j : Rˣ) : R))
    (hspan : ∀ x : R, ∃ (S : Finset ℤ) (c : ℤ → A), x = ∑ j ∈ S, φ (c j) * ((u ^ j : Rˣ) : R))
    (hkey : ∀ a : A, a ≠ 0 → ∀ F : Finset ℤ, (0 : ℤ) ∉ F → ∃ e b : A, b * e * a * e = e ∧
      (∀ j ∈ F, e * σ j e = 0) ∧ ∃ s : Finset ℤ, ∏ i ∈ s, (1 - σ i e) = 0)
    (hfg : IsFinitelyGeneratedRing R)
    (hlef : IsLEF ↥(elementaryGroup (Fin 3) R)) :
    ∃ (E : Type) (_ : Group E), Infinite E ∧ IsSimpleGroup E ∧
      HasKazhdanPropertyT.{0, 0} E ∧ IsLEF E :=
  exists_infinite_simple_kazhdan_lef_of_elementary hfg
    (isSimpleGroup_elementaryGroup_crossedProduct (by simp) hR φ u σ hconj hspan hkey)
    (elementaryModCentre_infinite (R := R) (0 : Fin 3) 1 (by decide)) hlef
