/-! ### The Pestov 9.1 crossed product -/

section Witness

variable (X : Set (ℤ → Bool)) [CompactSpace X]

/-- A homeomorphism reading the next coordinate moves the coordinate indicators
down by one under `CrossedProduct.shift σ 1 = (· ∘ σ⁻¹)`. -/
theorem shift_coordinateIndicator (σ : X ≃ₜ X)
    (hσ : ∀ (x : X) (i : ℤ), (σ x).1 i = x.1 (1 + i)) (i : ℤ) (b : Bool) :
    CrossedProduct.shift σ 1 (coordinateIndicator X (i + 1) b) =
      coordinateIndicator X i b := by
  refine LocallyConstant.ext fun y => ?_
  have hy : ((σ ^ (-1 : ℤ)) y).1 (i + 1) = y.1 i := by
    rw [zpow_neg_one, Homeomorph.inv_apply]
    have h := hσ (σ.symm y) i
    rw [Homeomorph.apply_symm_apply, add_comm] at h
    exact h.symm
  rw [CrossedProduct.shift_apply, coordinateIndicator_apply, coordinateIndicator_apply, hy]

/-- **The Pestov 9.1 coefficient ring is finitely generated.**  For a compact
set `X` of Bool sequences and a homeomorphism `σ` of `X` reading the next
coordinate (the shift `(T x)ₘ = xₘ₊₁`), the crossed product
`LC(X, ZMod 2) ⋊_σ ℤ` is a finitely generated ring: `[x₀ = false]`,
`[x₀ = true]`, `u` and `u⁻¹` generate it.  This is the hypothesis of the
Ershov--Jaikin-Zapirain theorem. -/
theorem isFinitelyGeneratedRing_crossedProduct (σ : X ≃ₜ X)
    (hσ : ∀ (x : X) (i : ℤ), (σ x).1 i = x.1 (1 + i)) :
    IsFinitelyGeneratedRing (CrossedProduct σ) := by
  refine isFinitelyGeneratedRing_of_shift X (CrossedProduct.C σ)
    ↑(CrossedProduct.u σ)⁻¹ ↑(CrossedProduct.u σ)
    (CrossedProduct.u σ).inv_mul (CrossedProduct.u σ).mul_inv (fun i b => ?_) ?_
  · have h := CrossedProduct.u_zpow_mul_C_mul_inv (σ := σ) 1
      (coordinateIndicator X (i + 1) b)
    rw [zpow_one, shift_coordinateIndicator X σ hσ] at h
    calc CrossedProduct.C σ (coordinateIndicator X (i + 1) b)
        = (↑(CrossedProduct.u σ)⁻¹ * ↑(CrossedProduct.u σ)) *
            CrossedProduct.C σ (coordinateIndicator X (i + 1) b) *
            (↑(CrossedProduct.u σ)⁻¹ * ↑(CrossedProduct.u σ)) := by
          rw [Units.inv_mul, one_mul, mul_one]
      _ = ↑(CrossedProduct.u σ)⁻¹ *
            (↑(CrossedProduct.u σ) * CrossedProduct.C σ (coordinateIndicator X (i + 1) b) *
              ↑(CrossedProduct.u σ)⁻¹) * ↑(CrossedProduct.u σ) := by
          simp only [mul_assoc]
      _ = ↑(CrossedProduct.u σ)⁻¹ * CrossedProduct.C σ (coordinateIndicator X i b) *
            ↑(CrossedProduct.u σ) := by
          rw [h]
  · have h := closure_range_union_eq_top_of_forall_eq_sum (CrossedProduct.C σ)
      (CrossedProduct.u σ) fun r => ⟨CrossedProduct.support r, CrossedProduct.coeff r,
        (CrossedProduct.sum_single r).symm.trans
          (Finset.sum_congr rfl fun n _ => CrossedProduct.single_eq_C_mul_u_zpow n _)⟩
    rwa [Set.pair_comm] at h

end Witness
